{-
  Trees: reify, and reduction under an operation.

    reify     a tree t : Term_Σ(Val A) as a computation (ret at the leaves)
    op-cong*  if every argument reduces, so does the operation
              (Under operations)
-}
open import CBPV.Theory

module CBPV.Trees (Sg : Signature) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Nat using (ℕ; zero; suc)
open import Cubical.Data.Sigma
open import Cubical.Data.FinData using (Fin; zero; suc)

open Signature Sg
open import CBPV.Fin
open import CBPV.Syntax Sg
open import CBPV.Reduction Sg

private variable
  A : VTy
  B B' : CTy

------------------------------------------------------------------------
-- Many steps under an operation, one argument at a time.

opC : (o : Op) → (Fin (ar o) → CComp B) → CComp B
opC o k = op o k

↦*-≡ : {M M' N N' : CComp B} → M ≡ M' → N ≡ N' → M ↦* N → M' ↦* N'
↦*-≡ p q = subst2 _↦*_ p q

-- Reduce argument j of op o f along M ↦* N.
steps-at : (o : Op) (f : Fin (ar o) → CComp B) (j : Fin (ar o)) {M N : CComp B}
  → f j ≡ M → M ↦* N → op o f ↦* op o (f [ j ≔ N ])
steps-at o f j e done =
  ↦*-≡ refl (cong (opC o) (sym (upd-same f j) ∙ cong (f [ j ≔_]) e)) done
steps-at o f j {N = N} e (_∷_ {M = M₁} s p) =
  opArg o f j (subst (_↦ M₁) (sym e) s)
  ∷ ↦*-≡ refl (cong (opC o) (upd-upd f j M₁ N))
      (steps-at o (f [ j ≔ M₁ ]) j (upd-yes f j M₁ refl) p)

-- A context with n argument slots, along which argument reductions lift.
record ArgCtx (n : ℕ) (B B' : CTy) : Type where
  field
    C     : (Fin n → CComp B) → CComp B'
    lift* : (f : Fin n → CComp B) (j : Fin n) {N : CComp B}
          → f j ↦* N → C f ↦* C (f [ j ≔ N ])
open ArgCtx

-- Reduce all arguments, one position at a time.
parallel : (n : ℕ) (K : ArgCtx n B B') (k k' : Fin n → CComp B)
  → (∀ j → k j ↦* k' j) → K .C k ↦* K .C k'
parallel zero K k k' ps =
  ↦*-≡ refl (cong (K .C) (funExt λ ())) done
parallel (suc n) K k k' ps =
  K .lift* k zero (ps zero)
  ++* ↦*-≡ (cong (K .C) (sym (upd-zero k (k' zero)))) (cong (K .C) (cons-η k'))
        (parallel n K' (k ∘ suc) (k' ∘ suc) (ps ∘ suc))
  where
  K' : ArgCtx n _ _
  K' .C g = K .C (cons (k' zero) g)
  K' .lift* g j p =
    ↦*-≡ refl (cong (K .C) (upd-cons-suc (k' zero) g j _)) (K .lift* (cons (k' zero) g) (suc j) p)

-- If every argument reduces, so does the operation.
op-cong* : (o : Op) (k k' : Fin (ar o) → CComp B) → (∀ j → k j ↦* k' j) → op o k ↦* op o k'
op-cong* {B = B} o = parallel (ar o) K
  where
  K : ArgCtx (ar o) B B
  K .C           = op o
  K .lift* f j p = steps-at o f j refl p

------------------------------------------------------------------------
-- Trees as computations

reify : Term Sg (CVal A) → CComp (F A)
reify (var V)   = ret V
reify (op o ts) = op o (λ j → reify (ts j))
