{-
  Tree reduction is locally confluent, hence (Newman's lemma with strong
  normalization) confluent, so normal forms are unique.

    ⇝-det        redexes are deterministic
    local-conf   M ↦ M₁ and M ↦ M₂ join, by induction on M ↦ M₁:
                   the only overlap is an operation moving out through a
                   frame vs. a step inside one of its arguments, and two
                   steps inside different arguments commute
    confluence   M ↦* M₁ and M ↦* M₂ join
    nf-unique    M reaches at most one normal form
    tree-unique  M reaches at most one tree reify(t)
-}

------------------------------------------------------------------------
-- Joining

Join : CComp B → CComp B → Type
Join M N = Σ[ X ∈ CComp _ ] (M ↦* X) × (N ↦* X)

swap : {M N : CComp B} → Join M N → Join N M
swap (X , p , q) = X , q , p

one : {M N : CComp B} → M ↦ N → M ↦* N
one s = s ∷ done

------------------------------------------------------------------------
-- Updating one argument

upd-map : {n : _} {X Y : Type} (f : X → Y) (k : Fin n → X) (j : Fin n) (x : X)
  → (λ i → f (k i)) [ j ≔ f x ] ≡ (λ i → f ((k [ j ≔ x ]) i))
upd-map f k j x = funExt go
  where
  go : ∀ i → ((λ i → f (k i)) [ j ≔ f x ]) i ≡ f ((k [ j ≔ x ]) i)
  go i with discreteFin i j
  ... | yes _ = refl
  ... | no _  = refl

upd-comm : {n : _} {X : Type} (k : Fin n → X) (j j' : Fin n) (x y : X) → ¬ j ≡ j'
  → (k [ j ≔ x ]) [ j' ≔ y ] ≡ (k [ j' ≔ y ]) [ j ≔ x ]
upd-comm k j j' x y ne = funExt go
  where
  go : ∀ i → ((k [ j ≔ x ]) [ j' ≔ y ]) i ≡ ((k [ j' ≔ y ]) [ j ≔ x ]) i
  go i = h (discreteFin i j) (discreteFin i j')
    where
    h : Dec (i ≡ j) → Dec (i ≡ j') → ((k [ j ≔ x ]) [ j' ≔ y ]) i ≡ ((k [ j' ≔ y ]) [ j ≔ x ]) i
    h (yes p) (yes q) = ⊥.rec (ne (sym p ∙ q))
    h (yes p) (no q)  = upd-no _ j' y q ∙ upd-yes k j x p ∙ sym (upd-yes _ j x p)
    h (no p)  (yes q) = upd-yes _ j' y q ∙ sym (upd-no _ j x p ∙ upd-yes k j' y q)
    h (no p)  (no q)  = upd-no _ j' y q ∙ upd-no k j x p ∙ sym (upd-no _ j x p ∙ upd-no k j' y q)

------------------------------------------------------------------------
-- Redexes are deterministic

red-det : {R R₁ R₂ : CComp B} → R ↦red R₁ → R ↦red R₂ → R₁ ≡ R₂
red-det (bind-ret V N)        (bind-ret .V .N)        = refl
red-det (force-thunk M)       (force-thunk .M)        = refl
red-det (case+-inl V M₁ M₂)   (case+-inl .V .M₁ .M₂)  = refl
red-det (case+-inr V M₁ M₂)   (case+-inr .V .M₁ .M₂)  = refl
red-det (case×-pair V W M)    (case×-pair .V .W .M)   = refl
red-det (app-lam M V)         (app-lam .M .V)         = refl
red-det (π₁-pair M N)         (π₁-pair .M .N)         = refl
red-det (π₂-pair M N)         (π₂-pair .M .N)         = refl

⇝-det : {R R₁ R₂ : CComp B} → R ⇝ R₁ → R ⇝ R₂ → R₁ ≡ R₂
⇝-det (β r₁)             (β r₂)               = red-det r₁ r₂
⇝-det (alg-app o k V)    (alg-app .o .k .V)   = refl
⇝-det (alg-π₁ o k)       (alg-π₁ .o .k)       = refl
⇝-det (alg-π₂ o k)       (alg-π₂ .o .k)       = refl
⇝-det (alg-bind o k N)   (alg-bind .o .k .N)  = refl

------------------------------------------------------------------------
-- A redex against a step inside its frame

red-appF : {M : CComp (A ⇒ B)} {V : CVal A} {R₁ : CComp B} {M' : CComp (A ⇒ B)}
  → app M V ⇝ R₁ → M ↦ M' → Join R₁ (app M' V)
red-appF (β (app-lam M V)) s = ⊥.rec (lam-no-step s)
red-appF (alg-app o k V) (opArg .o .k j {K'} s) =
  _ , one (subst (op o (λ i → app (k i) V) ↦_) (cong (opC o) (upd-map (λ X → app X V) k j K'))
             (opArg o (λ i → app (k i) V) j (appF V s)))
    , one (red (alg-app o (k [ j ≔ K' ]) V))
red-appF (alg-app o k V) (red (β ()))

red-π₁F : {M : CComp (B₁ & B₂)} {R₁ : CComp B₁} {M' : CComp (B₁ & B₂)}
  → π₁ M ⇝ R₁ → M ↦ M' → Join R₁ (π₁ M')
red-π₁F (β (π₁-pair M N)) s = ⊥.rec (cpair-no-step s)
red-π₁F (alg-π₁ o k) (opArg .o .k j {K'} s) =
  _ , one (subst (op o (λ i → π₁ (k i)) ↦_) (cong (opC o) (upd-map π₁ k j K'))
             (opArg o (λ i → π₁ (k i)) j (π₁F s)))
    , one (red (alg-π₁ o (k [ j ≔ K' ])))
red-π₁F (alg-π₁ o k) (red (β ()))

red-π₂F : {M : CComp (B₁ & B₂)} {R₁ : CComp B₂} {M' : CComp (B₁ & B₂)}
  → π₂ M ⇝ R₁ → M ↦ M' → Join R₁ (π₂ M')
red-π₂F (β (π₂-pair M N)) s = ⊥.rec (cpair-no-step s)
red-π₂F (alg-π₂ o k) (opArg .o .k j {K'} s) =
  _ , one (subst (op o (λ i → π₂ (k i)) ↦_) (cong (opC o) (upd-map π₂ k j K'))
             (opArg o (λ i → π₂ (k i)) j (π₂F s)))
    , one (red (alg-π₂ o (k [ j ≔ K' ])))
red-π₂F (alg-π₂ o k) (red (β ()))

red-bindF : {M : CComp (F A)} {N : Comp (ε ▸ A) B} {R₁ : CComp B} {M' : CComp (F A)}
  → bind M N ⇝ R₁ → M ↦ M' → Join R₁ (bind M' N)
red-bindF (β (bind-ret V N)) s = ⊥.rec (ret-no-step s)
red-bindF (alg-bind o k N) (opArg .o .k j {K'} s) =
  _ , one (subst (op o (λ i → bind (k i) N) ↦_) (cong (opC o) (upd-map (λ X → bind X N) k j K'))
             (opArg o (λ i → bind (k i) N) j (bindF N s)))
    , one (red (alg-bind o (k [ j ≔ K' ]) N))
red-bindF (alg-bind o k N) (red (β ()))

------------------------------------------------------------------------
-- Local confluence, by induction on the first step

local-conf : {M M₁ M₂ : CComp B} → M ↦ M₁ → M ↦ M₂ → Join M₁ M₂
local-conf {M₁ = M₁} (red r₁) (red r₂) = M₁ , done , subst (_↦* M₁) (⇝-det r₁ r₂) done
local-conf (red r) (appF V s)  = red-appF r s
local-conf (red r) (π₁F s)     = red-π₁F r s
local-conf (red r) (π₂F s)     = red-π₂F r s
local-conf (red r) (bindF N s) = red-bindF r s
local-conf (red (β ())) (opArg _ _ _ _)
local-conf (appF V s) (red r)  = swap (red-appF r s)
local-conf (π₁F s) (red r)     = swap (red-π₁F r s)
local-conf (π₂F s) (red r)     = swap (red-π₂F r s)
local-conf (bindF N s) (red r) = swap (red-bindF r s)
local-conf (opArg _ _ _ _) (red (β ()))
local-conf (appF V s₁) (appF .V s₂) =
  let (X , p₁ , p₂) = local-conf s₁ s₂ in
  _ , steps-plug (appˢ • V) p₁ , steps-plug (appˢ • V) p₂
local-conf (π₁F s₁) (π₁F s₂) =
  let (X , p₁ , p₂) = local-conf s₁ s₂ in
  _ , steps-plug (π₁ˢ •) p₁ , steps-plug (π₁ˢ •) p₂
local-conf (π₂F s₁) (π₂F s₂) =
  let (X , p₁ , p₂) = local-conf s₁ s₂ in
  _ , steps-plug (π₂ˢ •) p₁ , steps-plug (π₂ˢ •) p₂
local-conf (bindF N s₁) (bindF .N s₂) =
  let (X , p₁ , p₂) = local-conf s₁ s₂ in
  _ , steps-plug (bindˢ • N) p₁ , steps-plug (bindˢ • N) p₂
local-conf (opArg o k j {K₁} s₁) (opArg .o .k j' {K₂} s₂) with discreteFin j j'
... | yes e =
  let (X , p₁ , p₂) = local-conf s₁ (subst (λ i → k i ↦ K₂) (sym e) s₂) in
  X' X
    , ↦*-≡ refl (cong (opC o) (upd-upd k j K₁ X)) (steps-at o (k [ j ≔ K₁ ]) j (upd-yes k j K₁ refl) p₁)
    , ↦*-≡ (cong (λ i → op o (k [ i ≔ K₂ ])) e) (cong (opC o) (upd-upd k j K₂ X))
        (steps-at o (k [ j ≔ K₂ ]) j (upd-yes k j K₂ refl) p₂)
  where
  X' : CComp _ → CComp _
  X' X = op o (k [ j ≔ X ])
... | no ne =
  op o ((k [ j ≔ K₁ ]) [ j' ≔ K₂ ])
    , one (opArg o (k [ j ≔ K₁ ]) j' (subst (_↦ K₂) (sym (upd-no k j K₁ (λ q → ne (sym q)))) s₂))
    , one (subst (op o (k [ j' ≔ K₂ ]) ↦_) (cong (opC o) (sym (upd-comm k j j' K₁ K₂ ne)))
             (opArg o (k [ j' ≔ K₂ ]) j (subst (_↦ K₁) (sym (upd-no k j' K₂ ne)) s₁)))

------------------------------------------------------------------------
-- Confluence (Newman's lemma) and unique normal forms

to : {M M' : CComp B} → M ↦* M' → N.Star _↦_ M M'
to done    = N.done
to (s ∷ p) = s N.∷ to p

from : {M M' : CComp B} → N.Star _↦_ M M' → M ↦* M'
from N.done    = done
from (s N.∷ p) = s ∷ from p

local-conf' : N.LocallyConfluent (_↦_ {B = B})
local-conf' s₁ s₂ = let (X , p , q) = local-conf s₁ s₂ in X , to p , to q

confluence : {M M₁ M₂ : CComp B} → M ↦* M₁ → M ↦* M₂ → Join M₁ M₂
confluence p q =
  let (X , r , t) = N.newman _↦_ strong-normalization local-conf' (to p) (to q) in
  X , from r , from t

Normal : CComp B → Type
Normal M = ∀ {M'} → ¬ (M ↦ M')

normal-done : {M X : CComp B} → Normal M → M ↦* X → M ≡ X
normal-done n done    = refl
normal-done n (s ∷ p) = ⊥.rec (n s)

nf-unique : {M N₁ N₂ : CComp B} → Normal N₁ → Normal N₂ → M ↦* N₁ → M ↦* N₂ → N₁ ≡ N₂
nf-unique n₁ n₂ p q =
  let (X , r , t) = confluence p q in normal-done n₁ r ∙ sym (normal-done n₂ t)

reify-normal : (t : Term Sg (CVal A)) → Normal (reify t)
reify-normal (var V)   s = ret-no-step s
reify-normal (op o ts) s = let (j , _ , sj , _) = op-step s in reify-normal (ts j) sj

-- Tree adequacy's tree is unique.
tree-unique : {M : CComp (F A)} (t t' : Term Sg (CVal A))
  → M ↦* reify t → M ↦* reify t' → reify t ≡ reify t'
tree-unique t t' = nf-unique (reify-normal t) (reify-normal t')
