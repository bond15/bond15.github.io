{-
  Finite families: updating one entry, consing, and sums of naturals
  over Fin n (for the termination metrics of §4 and §5).
-}
module OptionC.Fin where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Nat hiding (znots; snotz)
open import Cubical.Data.Nat.Order
open import Cubical.Data.FinData using (Fin; zero; suc; discreteFin; znots; snotz; injSucFin)
import Cubical.Data.Empty as ⊥
open import Cubical.Relation.Nullary

private variable
  n : ℕ
  X : Type

-- f [ j ≔ x ]: replace the j-th entry
_[_≔_] : (Fin n → X) → Fin n → X → Fin n → X
(f [ j ≔ x ]) i with discreteFin i j
... | yes _ = x
... | no _  = f i

upd-yes : (f : Fin n → X) (j : Fin n) (x : X) {i : Fin n} → i ≡ j → (f [ j ≔ x ]) i ≡ x
upd-yes f j x {i} p with discreteFin i j
... | yes _  = refl
... | no ¬p  = ⊥.rec (¬p p)

upd-no : (f : Fin n → X) (j : Fin n) (x : X) {i : Fin n} → ¬ i ≡ j → (f [ j ≔ x ]) i ≡ f i
upd-no f j x {i} ¬p with discreteFin i j
... | yes p = ⊥.rec (¬p p)
... | no _  = refl

upd-same : (f : Fin n → X) (j : Fin n) → f [ j ≔ f j ] ≡ f
upd-same f j = funExt λ i → go i (discreteFin i j)
  where
  go : (i : Fin _) → Dec (i ≡ j) → (f [ j ≔ f j ]) i ≡ f i
  go i (yes p) = upd-yes f j (f j) p ∙ cong f (sym p)
  go i (no ¬p) = upd-no f j (f j) ¬p

upd-upd : (f : Fin n → X) (j : Fin n) (x y : X) → (f [ j ≔ x ]) [ j ≔ y ] ≡ f [ j ≔ y ]
upd-upd f j x y = funExt λ i → go i (discreteFin i j)
  where
  go : (i : Fin _) → Dec (i ≡ j) → ((f [ j ≔ x ]) [ j ≔ y ]) i ≡ (f [ j ≔ y ]) i
  go i (yes p) = upd-yes _ j y p ∙ sym (upd-yes f j y p)
  go i (no ¬p) = upd-no _ j y ¬p ∙ upd-no f j x ¬p ∙ sym (upd-no f j y ¬p)

cons : X → (Fin n → X) → Fin (suc n) → X
cons x g zero    = x
cons x g (suc i) = g i

cons-η : (f : Fin (suc n) → X) → cons (f zero) (f ∘ suc) ≡ f
cons-η f = funExt λ { zero → refl ; (suc i) → refl }

upd-zero : (f : Fin (suc n) → X) (y : X) → f [ zero ≔ y ] ≡ cons y (f ∘ suc)
upd-zero f y = funExt λ
  { zero    → upd-yes f zero y refl
  ; (suc i) → upd-no f zero y snotz }

upd-cons-suc : (x : X) (g : Fin n → X) (i : Fin n) (y : X)
  → (cons x g) [ suc i ≔ y ] ≡ cons x (g [ i ≔ y ])
upd-cons-suc x g i y = funExt go
  where
  go : (l : Fin _) → ((cons x g) [ suc i ≔ y ]) l ≡ cons x (g [ i ≔ y ]) l
  go' : (l : Fin _) → Dec (l ≡ i) → ((cons x g) [ suc i ≔ y ]) (suc l) ≡ cons x (g [ i ≔ y ]) (suc l)
  go' l (yes p) = upd-yes (cons x g) (suc i) y (cong suc p) ∙ sym (upd-yes g i y p)
  go' l (no ¬p) = upd-no (cons x g) (suc i) y (λ q → ¬p (injSucFin q)) ∙ sym (upd-no g i y ¬p)
  go zero    = upd-no (cons x g) (suc i) y znots
  go (suc l) = go' l (discreteFin l i)

------------------------------------------------------------------------
-- Sums

sumFin : (n : ℕ) → (Fin n → ℕ) → ℕ
sumFin zero    f = 0
sumFin (suc n) f = f zero + sumFin n (f ∘ suc)

sum-cong : (n : ℕ) {f g : Fin n → ℕ} → (∀ i → f i ≡ g i) → sumFin n f ≡ sumFin n g
sum-cong n p = cong (sumFin n) (funExt p)

-- Pointwise ≤ gives ≤ of sums.
sum-≤ : (n : ℕ) {f g : Fin n → ℕ} → (∀ i → f i ≤ g i) → sumFin n f ≤ sumFin n g
sum-≤ zero    p = ≤-refl
sum-≤ (suc n) p = ≤-+-≤ (p zero) (sum-≤ n (p ∘ suc))

-- Equal except at j, strictly smaller at j: strictly smaller sum.
sum-< : (n : ℕ) (f g : Fin n → ℕ) (j : Fin n)
  → (∀ i → ¬ i ≡ j → f i ≡ g i) → f j < g j → sumFin n f < sumFin n g
sum-< (suc n) f g zero same less =
  <-+-≤ less (≤-reflexive (sum-cong n λ i → same (suc i) snotz))
sum-< (suc n) f g (suc j) same less =
  subst (λ a → a + sumFin n (f ∘ suc) < g zero + sumFin n (g ∘ suc)) (sym (same zero znots))
    (<-k+ (sum-< n (f ∘ suc) (g ∘ suc) j (λ i ne → same (suc i) (λ p → ne (injSucFin p))) less))

-- Splitting off one summand.
sum-split : (n : ℕ) (f : Fin n → ℕ) (j : Fin n)
  → f j ≤ sumFin n f
sum-split (suc n) f zero    = sumFin n (f ∘ suc) , +-comm (sumFin n (f ∘ suc)) (f zero)
sum-split (suc n) f (suc j) = ≤-trans (sum-split n (f ∘ suc) j) (f zero , refl)

------------------------------------------------------------------------
-- Punching a position out of Fin (suc m)

punchIn : {m : ℕ} → Fin (suc m) → Fin m → Fin (suc m)
punchIn zero    j       = suc j
punchIn (suc t) zero    = zero
punchIn (suc t) (suc j) = suc (punchIn t j)

punchIn-avoid : {m : ℕ} (t : Fin (suc m)) (j : Fin m) → ¬ punchIn t j ≡ t
punchIn-avoid zero    j       p = snotz p
punchIn-avoid (suc t) zero    p = znots p
punchIn-avoid (suc t) (suc j) p = punchIn-avoid t j (injSucFin p)

punchOut : {m : ℕ} (t j : Fin (suc m)) → ¬ j ≡ t → Fin m
punchOut           zero    zero    ne = ⊥.rec (ne refl)
punchOut           zero    (suc j) ne = j
punchOut {m = zero}  (suc t) j     ne = ⊥.rec (¬Fin0 t)
  where open import Cubical.Data.FinData using (¬Fin0)
punchOut {m = suc m} (suc t) zero    ne = zero
punchOut {m = suc m} (suc t) (suc j) ne = suc (punchOut t j (λ p → ne (cong suc p)))

punchIn-punchOut : {m : ℕ} (t j : Fin (suc m)) (ne : ¬ j ≡ t) → punchIn t (punchOut t j ne) ≡ j
punchIn-punchOut           zero    zero    ne = ⊥.rec (ne refl)
punchIn-punchOut           zero    (suc j) ne = refl
punchIn-punchOut {m = zero}  (suc t) j     ne = ⊥.rec (¬Fin0 t)
  where open import Cubical.Data.FinData using (¬Fin0)
punchIn-punchOut {m = suc m} (suc t) zero    ne = refl
punchIn-punchOut {m = suc m} (suc t) (suc j) ne = cong suc (punchIn-punchOut t j _)

-- Σ_{Fin (suc m)} h = h t + Σ_{Fin m} (h ∘ punchIn t)
sum-punch : (m : ℕ) (h : Fin (suc m) → ℕ) (t : Fin (suc m))
  → sumFin (suc m) h ≡ h t + sumFin m (h ∘ punchIn t)
sum-punch m       h zero    = refl
sum-punch zero    h (suc t) = ⊥.rec (¬Fin0 t)
  where open import Cubical.Data.FinData using (¬Fin0)
sum-punch (suc m) h (suc t) =
  cong (h zero +_) (sum-punch m (h ∘ suc) t)
  ∙ +-assoc (h zero) (h (suc t)) _
  ∙ cong (_+ sumFin m (h ∘ suc ∘ punchIn t)) (+-comm (h zero) (h (suc t)))
  ∙ sym (+-assoc (h (suc t)) (h zero) _)

-- Sums along an injection are bounded by the full sum.
sum-inj : (a m : ℕ) (f : Fin a → Fin m) → (∀ x y → f x ≡ f y → x ≡ y)
  → (h : Fin m → ℕ) → sumFin a (h ∘ f) ≤ sumFin m h
sum-inj zero    m       f inj h = zero-≤
sum-inj (suc a) zero    f inj h = ⊥.rec (¬Fin0 (f zero))
  where open import Cubical.Data.FinData using (¬Fin0)
sum-inj (suc a) (suc m) f inj h =
  subst (_≤ sumFin (suc m) h) (sym lhs)
    (subst (h t + sumFin a ((h ∘ punchIn t) ∘ f') ≤_) (sym (sum-punch m h t))
      (≤-k+ (sum-inj a m f' inj' (h ∘ punchIn t))))
  where
  t = f zero
  ne : (x : Fin a) → ¬ f (suc x) ≡ t
  ne x p = snotz (inj (suc x) zero p)
  f' : Fin a → Fin m
  f' x = punchOut t (f (suc x)) (ne x)
  inj' : ∀ x y → f' x ≡ f' y → x ≡ y
  inj' x y p = injSucFin (inj (suc x) (suc y)
    (sym (punchIn-punchOut t _ (ne x)) ∙ cong (punchIn t) p ∙ punchIn-punchOut t _ (ne y)))
  lhs : sumFin (suc a) (h ∘ f) ≡ h t + sumFin a ((h ∘ punchIn t) ∘ f')
  lhs = cong (h t +_) (sum-cong a λ x → cong h (sym (punchIn-punchOut t _ (ne x))))

------------------------------------------------------------------------
-- Fin (m + b) as a disjoint union

finL : {m b : ℕ} → Fin m → Fin (m + b)
finL zero    = zero
finL (suc x) = suc (finL x)

finR : {m b : ℕ} → Fin b → Fin (m + b)
finR {m = zero}  k = k
finR {m = suc m} k = suc (finR {m = m} k)

sum-+ : (m b : ℕ) (h : Fin (m + b) → ℕ)
  → sumFin (m + b) h ≡ sumFin m (h ∘ finL) + sumFin b (h ∘ finR {m = m})
sum-+ zero    b h = refl
sum-+ (suc m) b h = cong (h zero +_) (sum-+ m b (h ∘ suc)) ∙ +-assoc (h zero) _ _

open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)

splitFin : {m b : ℕ} → Fin (m + b) → Fin m ⊎ Fin b
splitFin {m = zero}  k       = inr k
splitFin {m = suc m} zero    = inl zero
splitFin {m = suc m} (suc k) = ⊎.map suc (λ y → y) (splitFin {m = m} k)

splitFin-L : {m b : ℕ} (x : Fin m) → splitFin {m = m} {b = b} (finL x) ≡ inl x
splitFin-L zero    = refl
splitFin-L {m = suc m} {b} (suc x) = cong (⊎.map suc (λ y → y)) (splitFin-L {b = b} x)

splitFin-R : {m b : ℕ} (y : Fin b) → splitFin {m = m} (finR {m = m} y) ≡ inr y
splitFin-R {m = zero}  y = refl
splitFin-R {m = suc m} y = cong (⊎.map suc (λ y → y)) (splitFin-R {m = m} y)

punchIn-inj : {m : ℕ} (t : Fin (suc m)) (j j' : Fin m) → punchIn t j ≡ punchIn t j' → j ≡ j'
punchIn-inj zero    j       j'       p = injSucFin p
punchIn-inj (suc t) zero    zero     p = refl
punchIn-inj (suc t) zero    (suc j') p = ⊥.rec (znots p)
punchIn-inj (suc t) (suc j) zero     p = ⊥.rec (snotz p)
punchIn-inj (suc t) (suc j) (suc j') p = cong suc (punchIn-inj t j j' (injSucFin p))
