{-
  Renaming of CBPV terms (both modes), and its laws.

  Lemmas are stated with a pointwise hypothesis on the renamings rather
  than an equation between renamings, so no function extensionality on
  renamings is needed.  The structural clauses of the traversal and of
  each lemma were generated mechanically, one per constructor.
-}
open import OptionC.Theory

module OptionC.Renaming (Sg : Signature) where

open import Cubical.Foundations.Prelude

open import OptionC.Syntax Sg

private variable
  m : Mode
  Γ Δ Θ : Ctx
  A A' : VTy
  B B' : CTy

Ren : Ctx → Ctx → Type
Ren Γ Δ = ∀ {A} → Var Δ A → Var Γ A

liftR : Ren Γ Δ → Ren (Γ ▸ A) (Δ ▸ A)
liftR r vz     = vz
liftR r (vs x) = vs (r x)

renV : Ren Γ Δ → Val m Δ A → Val m Γ A
renC : Ren Γ Δ → Comp m Δ B → Comp m Γ B
renS : Ren Γ Δ → Stack m Δ B B' → Stack m Γ B B'

renV r (var x) = var (r x)
renV r (absurd a0) = absurd (renV r a0)
renV r ⋆ = ⋆
renV r (inl a0) = inl (renV r a0)
renV r (inr a0) = inr (renV r a0)
renV r (vpair a0 a1) = vpair (renV r a0) (renV r a1)
renV r (thunk a0) = thunk (renC r a0)
renV r (vcase+ ⦃ p ⦄ a0 a1 a2) = vcase+ ⦃ p ⦄ (renV r a0) (renV (liftR r) a1) (renV (liftR r) a2)
renV r (vcase× ⦃ p ⦄ a0 a1) = vcase× ⦃ p ⦄ (renV r a0) (renV (liftR (liftR r)) a1)

renC r (case+ a0 a1 a2) = case+ (renV r a0) (renC (liftR r) a1) (renC (liftR r) a2)
renC r (case× a0 a1) = case× (renV r a0) (renC (liftR (liftR r)) a1)
renC r (force a0) = force (renV r a0)
renC r triv = triv
renC r (lam a0) = lam (renC (liftR r) a0)
renC r (app a0 a1) = app (renC r a0) (renV r a1)
renC r (cpair a0 a1) = cpair (renC r a0) (renC r a1)
renC r (π₁ a0) = π₁ (renC r a0)
renC r (π₂ a0) = π₂ (renC r a0)
renC r (ret a0) = ret (renV r a0)
renC r (bind a0 a1) = bind (renC r a0) (renC (liftR r) a1)
renC r (op o k) = op o (λ j → renC r (k j))

renS r • = •
renS r (appˢ a0 a1) = appˢ (renS r a0) (renV r a1)
renS r (π₁ˢ a0) = π₁ˢ (renS r a0)
renS r (π₂ˢ a0) = π₂ˢ (renS r a0)
renS r (bindˢ a0 a1) = bindˢ (renS r a0) (renC (liftR r) a1)
renS r (trivˢ ⦃ p ⦄) = trivˢ ⦃ p ⦄
renS r (lamˢ ⦃ p ⦄ a0) = lamˢ ⦃ p ⦄ (renS (liftR r) a0)
renS r (pairˢ ⦃ p ⦄ a0 a1) = pairˢ ⦃ p ⦄ (renS r a0) (renS r a1)

wkV : Val m Γ A → Val m (Γ ▸ A') A
wkV = renV vs

wkC : Comp m Γ B → Comp m (Γ ▸ A') B
wkC = renC vs

wkS : Stack m Γ B B' → Stack m (Γ ▸ A') B B'
wkS = renS vs

------------------------------------------------------------------------
-- Fusion: r ∘ r' = r''  ⇒  ren r ∘ ren r' = ren r''

RenFuse : Ren Γ Δ → Ren Δ Θ → Ren Γ Θ → Type
RenFuse {Θ = Θ} r r' r'' = ∀ {A} (x : Var Θ A) → r (r' x) ≡ r'' x

liftR-fuse : {r : Ren Γ Δ} {r' : Ren Δ Θ} {r'' : Ren Γ Θ}
  → RenFuse r r' r'' → RenFuse (liftR {A = A} r) (liftR r') (liftR r'')
liftR-fuse h vz     = refl
liftR-fuse h (vs x) = cong vs (h x)

ren-fuseV : {r : Ren Γ Δ} {r' : Ren Δ Θ} {r'' : Ren Γ Θ} → RenFuse r r' r''
  → (V : Val m Θ A) → renV r (renV r' V) ≡ renV r'' V
ren-fuseC : {r : Ren Γ Δ} {r' : Ren Δ Θ} {r'' : Ren Γ Θ} → RenFuse r r' r''
  → (M : Comp m Θ B) → renC r (renC r' M) ≡ renC r'' M
ren-fuseS : {r : Ren Γ Δ} {r' : Ren Δ Θ} {r'' : Ren Γ Θ} → RenFuse r r' r''
  → (S : Stack m Θ B B') → renS r (renS r' S) ≡ renS r'' S

ren-fuseV {r = r} {r' = r'} {r'' = r''} h (var x) = cong var (h x)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (absurd a0) i = absurd (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h ⋆ = refl
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (inl a0) i = inl (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (inr a0) i = inr (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (vpair a0 a1) i = vpair (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a1 i)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (thunk a0) i = thunk (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (vcase+ ⦃ p ⦄ a0 a1 a2) i = vcase+ ⦃ p ⦄ (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseV {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a1 i) (ren-fuseV {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a2 i)
ren-fuseV {r = r} {r' = r'} {r'' = r''} h (vcase× ⦃ p ⦄ a0 a1) i = vcase× ⦃ p ⦄ (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseV {r = (liftR (liftR r))} {r' = (liftR (liftR r'))} {r'' = (liftR (liftR r''))} (liftR-fuse (liftR-fuse h)) a1 i)

ren-fuseC {r = r} {r' = r'} {r'' = r''} h (case+ a0 a1 a2) i = case+ (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseC {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a1 i) (ren-fuseC {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a2 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (case× a0 a1) i = case× (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseC {r = (liftR (liftR r))} {r' = (liftR (liftR r'))} {r'' = (liftR (liftR r''))} (liftR-fuse (liftR-fuse h)) a1 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (force a0) i = force (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h triv = refl
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (lam a0) i = lam (ren-fuseC {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a0 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (app a0 a1) i = app (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a1 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (cpair a0 a1) i = cpair (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a1 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (π₁ a0) i = π₁ (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (π₂ a0) i = π₂ (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (ret a0) i = ret (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (bind a0 a1) i = bind (ren-fuseC {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseC {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a1 i)
ren-fuseC {r = r} {r' = r'} {r'' = r''} h (op o k) i = op o (λ j → ren-fuseC {r = r} {r' = r'} {r'' = r''} h (k j) i)

ren-fuseS {r = r} {r' = r'} {r'' = r''} h • = refl
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (appˢ a0 a1) i = appˢ (ren-fuseS {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseV {r = r} {r' = r'} {r'' = r''} h a1 i)
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (π₁ˢ a0) i = π₁ˢ (ren-fuseS {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (π₂ˢ a0) i = π₂ˢ (ren-fuseS {r = r} {r' = r'} {r'' = r''} h a0 i)
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (bindˢ a0 a1) i = bindˢ (ren-fuseS {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseC {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a1 i)
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (trivˢ ⦃ p ⦄) = refl
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (lamˢ ⦃ p ⦄ a0) i = lamˢ ⦃ p ⦄ (ren-fuseS {r = (liftR r)} {r' = (liftR r')} {r'' = (liftR r'')} (liftR-fuse h) a0 i)
ren-fuseS {r = r} {r' = r'} {r'' = r''} h (pairˢ ⦃ p ⦄ a0 a1) i = pairˢ ⦃ p ⦄ (ren-fuseS {r = r} {r' = r'} {r'' = r''} h a0 i) (ren-fuseS {r = r} {r' = r'} {r'' = r''} h a1 i)

-- Identity: r = id  ⇒  ren r = id

RenId : Ren Γ Γ → Type
RenId {Γ = Γ} r = ∀ {A} (x : Var Γ A) → r x ≡ x

liftR-id : {r : Ren Γ Γ} → RenId r → RenId (liftR {A = A} r)
liftR-id h vz     = refl
liftR-id h (vs x) = cong vs (h x)

ren-idV : {r : Ren Γ Γ} → RenId r → (V : Val m Γ A) → renV r V ≡ V
ren-idC : {r : Ren Γ Γ} → RenId r → (M : Comp m Γ B) → renC r M ≡ M
ren-idS : {r : Ren Γ Γ} → RenId r → (S : Stack m Γ B B') → renS r S ≡ S

ren-idV {r = r} h (var x) = cong var (h x)
ren-idV {r = r} h (absurd a0) i = absurd (ren-idV {r = r} h a0 i)
ren-idV {r = r} h ⋆ = refl
ren-idV {r = r} h (inl a0) i = inl (ren-idV {r = r} h a0 i)
ren-idV {r = r} h (inr a0) i = inr (ren-idV {r = r} h a0 i)
ren-idV {r = r} h (vpair a0 a1) i = vpair (ren-idV {r = r} h a0 i) (ren-idV {r = r} h a1 i)
ren-idV {r = r} h (thunk a0) i = thunk (ren-idC {r = r} h a0 i)
ren-idV {r = r} h (vcase+ ⦃ p ⦄ a0 a1 a2) i = vcase+ ⦃ p ⦄ (ren-idV {r = r} h a0 i) (ren-idV {r = (liftR r)} (liftR-id h) a1 i) (ren-idV {r = (liftR r)} (liftR-id h) a2 i)
ren-idV {r = r} h (vcase× ⦃ p ⦄ a0 a1) i = vcase× ⦃ p ⦄ (ren-idV {r = r} h a0 i) (ren-idV {r = (liftR (liftR r))} (liftR-id (liftR-id h)) a1 i)

ren-idC {r = r} h (case+ a0 a1 a2) i = case+ (ren-idV {r = r} h a0 i) (ren-idC {r = (liftR r)} (liftR-id h) a1 i) (ren-idC {r = (liftR r)} (liftR-id h) a2 i)
ren-idC {r = r} h (case× a0 a1) i = case× (ren-idV {r = r} h a0 i) (ren-idC {r = (liftR (liftR r))} (liftR-id (liftR-id h)) a1 i)
ren-idC {r = r} h (force a0) i = force (ren-idV {r = r} h a0 i)
ren-idC {r = r} h triv = refl
ren-idC {r = r} h (lam a0) i = lam (ren-idC {r = (liftR r)} (liftR-id h) a0 i)
ren-idC {r = r} h (app a0 a1) i = app (ren-idC {r = r} h a0 i) (ren-idV {r = r} h a1 i)
ren-idC {r = r} h (cpair a0 a1) i = cpair (ren-idC {r = r} h a0 i) (ren-idC {r = r} h a1 i)
ren-idC {r = r} h (π₁ a0) i = π₁ (ren-idC {r = r} h a0 i)
ren-idC {r = r} h (π₂ a0) i = π₂ (ren-idC {r = r} h a0 i)
ren-idC {r = r} h (ret a0) i = ret (ren-idV {r = r} h a0 i)
ren-idC {r = r} h (bind a0 a1) i = bind (ren-idC {r = r} h a0 i) (ren-idC {r = (liftR r)} (liftR-id h) a1 i)
ren-idC {r = r} h (op o k) i = op o (λ j → ren-idC {r = r} h (k j) i)

ren-idS {r = r} h • = refl
ren-idS {r = r} h (appˢ a0 a1) i = appˢ (ren-idS {r = r} h a0 i) (ren-idV {r = r} h a1 i)
ren-idS {r = r} h (π₁ˢ a0) i = π₁ˢ (ren-idS {r = r} h a0 i)
ren-idS {r = r} h (π₂ˢ a0) i = π₂ˢ (ren-idS {r = r} h a0 i)
ren-idS {r = r} h (bindˢ a0 a1) i = bindˢ (ren-idS {r = r} h a0 i) (ren-idC {r = (liftR r)} (liftR-id h) a1 i)
ren-idS {r = r} h (trivˢ ⦃ p ⦄) = refl
ren-idS {r = r} h (lamˢ ⦃ p ⦄ a0) i = lamˢ ⦃ p ⦄ (ren-idS {r = (liftR r)} (liftR-id h) a0 i)
ren-idS {r = r} h (pairˢ ⦃ p ⦄ a0 a1) i = pairˢ ⦃ p ⦄ (ren-idS {r = r} h a0 i) (ren-idS {r = r} h a1 i)

-- Weakening commutes with lifted renamings.
wk-renV : (r : Ren Γ Δ) (V : Val m Δ A) → renV (liftR {A = A'} r) (wkV V) ≡ wkV (renV r V)
wk-renV r V = ren-fuseV (λ _ → refl) V ∙ sym (ren-fuseV (λ _ → refl) V)

wk-renC : (r : Ren Γ Δ) (M : Comp m Δ B) → renC (liftR {A = A'} r) (wkC M) ≡ wkC (renC r M)
wk-renC r M = ren-fuseC (λ _ → refl) M ∙ sym (ren-fuseC (λ _ → refl) M)

wk-renS : (r : Ren Γ Δ) (S : Stack m Δ B B') → renS (liftR {A = A'} r) (wkS S) ≡ wkS (renS r S)
wk-renS r S = ren-fuseS (λ _ → refl) S ∙ sym (ren-fuseS (λ _ → refl) S)
