{-
  Strong normalization of configuration reduction, for an affine
  operational model.

  The configuration measure is the sum of the tree measure ‖_‖
  (CBPV.Measure) over the subterms:

    ‖⟨s, N⃗⟩‖T = Σ_p ‖N_p‖

  Every ↦T step decreases it (decrease-T):

    Pure    a step N ↦ N' at the hole decreases ‖N‖ (step-dec), and the
            other positions are unchanged;
    Effect  the reduct is T(γ ⊎ (S[M₁], …))(step(s,i,op)) (effect-via-step).
            Affinity (ρ injective) means each neighbor γ(q) and each
            continuation S[Mₖ] occurs at most once, so the measure of the
            reduct is at most Σ_q ‖γ(q)‖ + Σₖ ‖S[Mₖ]‖ (step-sum), and
            Σₖ ‖S[Mₖ]‖ < ‖S[op(M⃗)]‖ (stack-op-measure).

  Strong normalization is stated as accessibility (Acc) for the converse
  of ↦T.
-}

------------------------------------------------------------------------
-- Sums over holes

holeIn : {m : ℕ} (i : Fin (suc m)) → Fin m → Fin∖ (suc m) i
holeIn i k = punchIn i k , punchIn-avoid i k

-- Σ over the positions other than i.
holeSum : (n : ℕ) (i : Fin n) → (Fin∖ n i → ℕ) → ℕ
holeSum zero    i g = ⊥.rec (¬Fin0 i)
holeSum (suc m) i g = sumFin m (g ∘ holeIn i)

fill-other : {n : ℕ} (i : Fin n) (γ : Fin∖ n i → X) (x : X) (j : Fin n) (ne : ¬ j ≡ i)
  → fillAt i γ x j ≡ γ (j , ne)
fill-other i γ x j ne with discreteFin j i
... | yes p   = ⊥.rec (ne p)
... | no ne'  = cong (λ q → γ (j , q)) (isProp¬ _ ne' ne)

-- Σ over C⟪x⟫ = g x + Σ over the rest of C.
plug-sum : (C : ∂ poly X) (x : X) (g : X → ℕ)
  → sumFin _ (g ∘ (C ⟪ x ⟫) .snd) ≡ g x + holeSum _ (C .snd .fst) (g ∘ C .snd .snd)
plug-sum {X = X} (s , i , γ) x g = go (poly .P s) i γ
  where
  go : (n : ℕ) (i : Fin n) (γ : Fin∖ n i → X)
     → sumFin n (g ∘ fillAt i γ x) ≡ g x + holeSum n i (g ∘ γ)
  go zero    i γ = ⊥.rec (¬Fin0 i)
  go (suc m) i γ =
    sum-punch m (g ∘ fillAt i γ x) i
    ∙ cong₂ _+_ (cong g (fill-hole i γ x))
        (sum-cong m λ k → cong g (fill-other i γ x (punchIn i k) (punchIn-avoid i k)))

-- Σ along an injective renaming ρ into the holes and the arguments.
step-sum : (n : ℕ) (i : Fin n) {a b : ℕ}
  (ρ' : Fin a → Fin∖ n i ⊎ Fin b) → isInjective ρ'
  → (γ : Fin∖ n i → X) (δ : Fin b → X) (g : X → ℕ)
  → sumFin a (g ∘ ⊎.rec γ δ ∘ ρ') ≤ holeSum n i (g ∘ γ) + sumFin b (g ∘ δ)
step-sum zero    i ρ' inj γ δ g = ⊥.rec (¬Fin0 i)
step-sum (suc m) i {a} {b} ρ' inj γ δ g =
  subst (_≤ holeSum (suc m) i (g ∘ γ) + sumFin b (g ∘ δ)) (sym lhs-eq)
    (subst (sumFin a (G ∘ dec ∘ enc ∘ ρ') ≤_) rhs-eq
      (sum-inj a (m + b) (enc ∘ ρ') inj-encρ (G ∘ dec)))
  where
  G : Fin∖ (suc m) i ⊎ Fin b → ℕ
  G = g ∘ ⊎.rec γ δ
  enc : Fin∖ (suc m) i ⊎ Fin b → Fin (m + b)
  enc (inl (j , ne)) = finL (punchOut i j ne)
  enc (inr k)        = finR {m = m} k
  inlF : Fin∖ (suc m) i → Fin∖ (suc m) i ⊎ Fin b
  inlF = inl
  dec : Fin (m + b) → Fin∖ (suc m) i ⊎ Fin b
  dec = ⊎.map (holeIn i) (λ y → y) ∘ splitFin {m = m}
  dec-enc : ∀ x → dec (enc x) ≡ x
  dec-enc (inl (j , ne)) =
    cong (⊎.map (holeIn i) (λ y → y)) (splitFin-L {b = b} (punchOut i j ne))
    ∙ cong inlF (ΣPathP (punchIn-punchOut i j ne , isProp→PathP (λ _ → isProp¬ _) _ _))
  dec-enc (inr k) = cong (⊎.map (holeIn i) (λ y → y)) (splitFin-R {m = m} k)
  inj-encρ : ∀ x y → enc (ρ' x) ≡ enc (ρ' y) → x ≡ y
  inj-encρ x y p = inj x y (sym (dec-enc (ρ' x)) ∙ cong dec p ∙ dec-enc (ρ' y))
  lhs-eq : sumFin a (g ∘ ⊎.rec γ δ ∘ ρ') ≡ sumFin a (G ∘ dec ∘ enc ∘ ρ')
  lhs-eq = sum-cong a λ x → cong G (sym (dec-enc (ρ' x)))
  rhs-eq : sumFin (m + b) (G ∘ dec) ≡ holeSum (suc m) i (g ∘ γ) + sumFin b (g ∘ δ)
  rhs-eq = sum-+ m b (G ∘ dec)
      ∙ cong₂ _+_ (sum-cong m λ k → cong (G ∘ ⊎.map (holeIn i) (λ y → y)) (splitFin-L {b = b} k))
                  (sum-cong b λ k → cong (G ∘ ⊎.map (holeIn i) (λ y → y)) (splitFin-R {m = m} k))

------------------------------------------------------------------------
-- The configuration measure

‖_‖T : Config A → ℕ
‖ t ‖T = sumFin _ (λ p → ‖ t .snd p ‖)

-- Every ↦T step decreases the measure, for an affine operational model.
decrease-T : Affine → {t t' : Config A} → t ↦T t' → ‖ t' ‖T < ‖ t ‖T
decrease-T aff (pure C@(s , i , γ) {M} {M'} st) =
  subst2 _<_ (sym (plug-sum C M' ‖_‖)) (sym (plug-sum C M ‖_‖))
    (<-+k (step-dec st))
decrease-T aff (effect C@(s , i , γ) S o k) =
  subst (_< ‖ C ⟪ plug S (op o k) ⟫ ‖T) (sym rhs-eq-measure)
    (≤<-trans
      (step-sum (poly .P s) i (ρ s i o) (aff s i o) γ (λ j → plug S (k j)) ‖_‖)
      (subst (holeSum _ i (‖_‖ ∘ γ) + sumFin (ar o) (λ j → ‖ plug S (k j) ‖) <_)
        (+-comm (holeSum _ i (‖_‖ ∘ γ)) ‖ plug S (op o k) ‖
          ∙ sym (plug-sum C (plug S (op o k)) ‖_‖))
        (<-k+ (stack-op-measure S o k))))
  where
  rhs-eq-path : μ ((∂map η C) ⟪ ops o (λ j → η (plug S (k j))) ⟫)
           ≡ pmap (⊎.rec γ (λ j → plug S (k j))) (step s i o)
  rhs-eq-path = effect-via-step s i γ S o k ∙ map-poly _ (step s i o)
  rhs-eq-measure : ‖ μ ((∂map η C) ⟪ ops o (λ j → η (plug S (k j))) ⟫) ‖T
             ≡ sumFin _ (‖_‖ ∘ ⊎.rec γ (λ j → plug S (k j)) ∘ ρ s i o)
  rhs-eq-measure = cong ‖_‖T rhs-eq-path

------------------------------------------------------------------------
-- Theorem (Strong normalization for ↦T)

acc-measure : {Y : Type} {R : Y → Y → Type} (m : Y → ℕ)
  → (∀ {x y} → R y x → m y < m x) → ∀ x → Acc R x
acc-measure {R = R} m dec x = go x (<-wellfounded (m x))
  where
  go : ∀ x → Acc _<_ (m x) → Acc R x
  go x (acc rec) = acc λ y r → go y (rec (m y) (dec r))

-- The converse of ↦T.
_↤T_ : Config A → Config A → Type
t' ↤T t = t ↦T t'

-- Every ↦T sequence is finite.
strong-normalization-T : Affine → (t : Config A) → Acc _↤T_ t
strong-normalization-T aff = acc-measure ‖_‖T (decrease-T aff)
