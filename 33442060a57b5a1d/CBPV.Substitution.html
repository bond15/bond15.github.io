module CBPV.Substitution (Sg : Signature) where

open import Cubical.Foundations.Prelude

private variable
  Γ Δ Θ : Ctx
  A A' A'' : VTy
  B B' : CTy

-- Δ ⇒ Γ in the paper: a value in Γ for each variable of Δ.
Subst : Ctx → Ctx → Type
Subst Γ Δ = ∀ {A} → Var Δ A → Val Γ A

liftS : Subst Γ Δ → Subst (Γ ▸ A) (Δ ▸ A)
liftS σ vz     = var vz
liftS σ (vs x) = wkV (σ x)

subV : Subst Γ Δ → Val Δ A → Val Γ A
subC : Subst Γ Δ → Comp Δ B → Comp Γ B

subV r (var x) = r x
subV r (absurd a0) = absurd (subV r a0)
subV r ⋆ = ⋆
subV r (inl a0) = inl (subV r a0)
subV r (inr a0) = inr (subV r a0)
subV r (vpair a0 a1) = vpair (subV r a0) (subV r a1)
subV r (thunk a0) = thunk (subC r a0)

subC r (case+ a0 a1 a2) = case+ (subV r a0) (subC (liftS r) a1) (subC (liftS r) a2)
subC r (case× a0 a1) = case× (subV r a0) (subC (liftS (liftS r)) a1)
subC r (force a0) = force (subV r a0)
subC r triv = triv
subC r (lam a0) = lam (subC (liftS r) a0)
subC r (app a0 a1) = app (subC r a0) (subV r a1)
subC r (cpair a0 a1) = cpair (subC r a0) (subC r a1)
subC r (π₁ a0) = π₁ (subC r a0)
subC r (π₂ a0) = π₂ (subC r a0)
subC r (ret a0) = ret (subV r a0)
subC r (bind a0 a1) = bind (subC r a0) (subC (liftS r) a1)
subC r (op o k) = op o (λ j → subC r (k j))

idS : Subst Γ Γ
idS = var

-- γ , V/x
infixl 20 _,ₛ_
_,ₛ_ : Subst Γ Δ → Val Γ A → Subst Γ (Δ ▸ A)
(σ ,ₛ V) vz     = V
(σ ,ₛ V) (vs x) = σ x

-- M[V/x]
_[_]₀ : Comp (Γ ▸ A) B → Val Γ A → Comp Γ B
M [ V ]₀ = subC (idS ,ₛ V) M

-- M[V/x, W/y]
_[_,_]₁ : Comp (Γ ▸ A ▸ A') B → Val Γ A → Val Γ A' → Comp Γ B
M [ V , W ]₁ = subC (idS ,ₛ V ,ₛ W) M

-- sub σ ∘ ren r = sub (σ ∘ r)

SubRen : Subst Γ Δ → Ren Δ Θ → Subst Γ Θ → Type
SubRen {Θ = Θ} σ r σ'' = ∀ {A} (x : Var Θ A) → σ (r x) ≡ σ'' x

liftS-ren : {σ : Subst Γ Δ} {r : Ren Δ Θ} {σ'' : Subst Γ Θ}
  → SubRen σ r σ'' → SubRen (liftS {A = A} σ) (liftR r) (liftS σ'')
liftS-ren h vz     = refl
liftS-ren h (vs x) = cong wkV (h x)

sub-renV : {σ : Subst Γ Δ} {r : Ren Δ Θ} {σ'' : Subst Γ Θ} → SubRen σ r σ''
  → (V : Val Θ A) → subV σ (renV r V) ≡ subV σ'' V
sub-renC : {σ : Subst Γ Δ} {r : Ren Δ Θ} {σ'' : Subst Γ Θ} → SubRen σ r σ''
  → (M : Comp Θ B) → subC σ (renC r M) ≡ subC σ'' M

sub-renV {σ = σ} {r = r} {σ'' = σ''} h (var x) = h x
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (absurd a0) i = absurd (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h ⋆ = refl
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (inl a0) i = inl (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (inr a0) i = inr (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (vpair a0 a1) i = vpair (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (thunk a0) i = thunk (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i)

sub-renC {σ = σ} {r = r} {σ'' = σ''} h (case+ a0 a1 a2) i = case+ (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a1 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a2 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (case× a0 a1) i = case× (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS (liftS σ))} {r = (liftR (liftR r))} {σ'' = (liftS (liftS σ''))} (liftS-ren (liftS-ren h)) a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (force a0) i = force (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h triv = refl
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (lam a0) i = lam (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (app a0 a1) i = app (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (cpair a0 a1) i = cpair (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (π₁ a0) i = π₁ (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (π₂ a0) i = π₂ (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (ret a0) i = ret (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (bind a0 a1) i = bind (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (op o k) i = op o (λ j → sub-renC {σ = σ} {r = r} {σ'' = σ''} h (k j) i)

-- ren r ∘ sub σ = sub (ren r ∘ σ)

RenSub : Ren Γ Δ → Subst Δ Θ → Subst Γ Θ → Type
RenSub {Θ = Θ} r σ σ'' = ∀ {A} (x : Var Θ A) → renV r (σ x) ≡ σ'' x

liftR-sub : {r : Ren Γ Δ} {σ : Subst Δ Θ} {σ'' : Subst Γ Θ}
  → RenSub r σ σ'' → RenSub (liftR {A = A} r) (liftS σ) (liftS σ'')
liftR-sub h vz     = refl
liftR-sub {r = r} {σ = σ} h (vs x) = wk-renV r (σ x) ∙ cong wkV (h x)

ren-subV : {r : Ren Γ Δ} {σ : Subst Δ Θ} {σ'' : Subst Γ Θ} → RenSub r σ σ''
  → (V : Val Θ A) → renV r (subV σ V) ≡ subV σ'' V
ren-subC : {r : Ren Γ Δ} {σ : Subst Δ Θ} {σ'' : Subst Γ Θ} → RenSub r σ σ''
  → (M : Comp Θ B) → renC r (subC σ M) ≡ subC σ'' M

ren-subV {r = r} {σ = σ} {σ'' = σ''} h (var x) = h x
ren-subV {r = r} {σ = σ} {σ'' = σ''} h (absurd a0) i = absurd (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subV {r = r} {σ = σ} {σ'' = σ''} h ⋆ = refl
ren-subV {r = r} {σ = σ} {σ'' = σ''} h (inl a0) i = inl (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subV {r = r} {σ = σ} {σ'' = σ''} h (inr a0) i = inr (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subV {r = r} {σ = σ} {σ'' = σ''} h (vpair a0 a1) i = vpair (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i) (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a1 i)
ren-subV {r = r} {σ = σ} {σ'' = σ''} h (thunk a0) i = thunk (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a0 i)

ren-subC {r = r} {σ = σ} {σ'' = σ''} h (case+ a0 a1 a2) i = case+ (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i) (ren-subC {r = (liftR r)} {σ = (liftS σ)} {σ'' = (liftS σ'')} (liftR-sub h) a1 i) (ren-subC {r = (liftR r)} {σ = (liftS σ)} {σ'' = (liftS σ'')} (liftR-sub h) a2 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (case× a0 a1) i = case× (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i) (ren-subC {r = (liftR (liftR r))} {σ = (liftS (liftS σ))} {σ'' = (liftS (liftS σ''))} (liftR-sub (liftR-sub h)) a1 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (force a0) i = force (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h triv = refl
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (lam a0) i = lam (ren-subC {r = (liftR r)} {σ = (liftS σ)} {σ'' = (liftS σ'')} (liftR-sub h) a0 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (app a0 a1) i = app (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a0 i) (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a1 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (cpair a0 a1) i = cpair (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a0 i) (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a1 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (π₁ a0) i = π₁ (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (π₂ a0) i = π₂ (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (ret a0) i = ret (ren-subV {r = r} {σ = σ} {σ'' = σ''} h a0 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (bind a0 a1) i = bind (ren-subC {r = r} {σ = σ} {σ'' = σ''} h a0 i) (ren-subC {r = (liftR r)} {σ = (liftS σ)} {σ'' = (liftS σ'')} (liftR-sub h) a1 i)
ren-subC {r = r} {σ = σ} {σ'' = σ''} h (op o k) i = op o (λ j → ren-subC {r = r} {σ = σ} {σ'' = σ''} h (k j) i)

-- Weakening commutes with lifted substitutions.
wk-subV : (σ : Subst Γ Δ) (V : Val Δ A) → subV (liftS {A = A'} σ) (wkV V) ≡ wkV (subV σ V)
wk-subV σ V = sub-renV (λ _ → refl) V ∙ sym (ren-subV (λ _ → refl) V)

-- Composing substitutions.

SubSub : Subst Γ Δ → Subst Δ Θ → Subst Γ Θ → Type
SubSub {Θ = Θ} σ σ' σ'' = ∀ {A} (x : Var Θ A) → subV σ (σ' x) ≡ σ'' x

liftS-sub : {σ : Subst Γ Δ} {σ' : Subst Δ Θ} {σ'' : Subst Γ Θ}
  → SubSub σ σ' σ'' → SubSub (liftS {A = A} σ) (liftS σ') (liftS σ'')
liftS-sub h vz     = refl
liftS-sub {σ = σ} {σ' = σ'} h (vs x) = wk-subV σ (σ' x) ∙ cong wkV (h x)

sub-subV : {σ : Subst Γ Δ} {σ' : Subst Δ Θ} {σ'' : Subst Γ Θ} → SubSub σ σ' σ''
  → (V : Val Θ A) → subV σ (subV σ' V) ≡ subV σ'' V
sub-subC : {σ : Subst Γ Δ} {σ' : Subst Δ Θ} {σ'' : Subst Γ Θ} → SubSub σ σ' σ''
  → (M : Comp Θ B) → subC σ (subC σ' M) ≡ subC σ'' M

sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h (var x) = h x
sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h (absurd a0) i = absurd (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)
sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h ⋆ = refl
sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h (inl a0) i = inl (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)
sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h (inr a0) i = inr (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)
sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h (vpair a0 a1) i = vpair (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i) (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a1 i)
sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h (thunk a0) i = thunk (sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)

sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (case+ a0 a1 a2) i = case+ (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i) (sub-subC {σ = (liftS σ)} {σ' = (liftS σ')} {σ'' = (liftS σ'')} (liftS-sub h) a1 i) (sub-subC {σ = (liftS σ)} {σ' = (liftS σ')} {σ'' = (liftS σ'')} (liftS-sub h) a2 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (case× a0 a1) i = case× (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i) (sub-subC {σ = (liftS (liftS σ))} {σ' = (liftS (liftS σ'))} {σ'' = (liftS (liftS σ''))} (liftS-sub (liftS-sub h)) a1 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (force a0) i = force (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h triv = refl
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (lam a0) i = lam (sub-subC {σ = (liftS σ)} {σ' = (liftS σ')} {σ'' = (liftS σ'')} (liftS-sub h) a0 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (app a0 a1) i = app (sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i) (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a1 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (cpair a0 a1) i = cpair (sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i) (sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h a1 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (π₁ a0) i = π₁ (sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (π₂ a0) i = π₂ (sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)
sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (ret a0) i = ret (sub-subV {σ = σ} {σ' = σ'} {σ'' = σ''} h a0 i)

sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (op o k) i = op o (λ j → sub-subC {σ = σ} {σ' = σ'} {σ'' = σ''} h (k j) i)

-- σ = id  ⇒  sub σ = id

SubId : Subst Γ Γ → Type
SubId {Γ = Γ} σ = ∀ {A} (x : Var Γ A) → σ x ≡ var x

liftS-id : {σ : Subst Γ Γ} → SubId σ → SubId (liftS {A = A} σ)
liftS-id h vz     = refl
liftS-id h (vs x) = cong wkV (h x)

sub-idV : {σ : Subst Γ Γ} → SubId σ → (V : Val Γ A) → subV σ V ≡ V
sub-idC : {σ : Subst Γ Γ} → SubId σ → (M : Comp Γ B) → subC σ M ≡ M

sub-idV {σ = σ} h (var x) = h x
sub-idV {σ = σ} h (absurd a0) i = absurd (sub-idV {σ = σ} h a0 i)
sub-idV {σ = σ} h ⋆ = refl
sub-idV {σ = σ} h (inl a0) i = inl (sub-idV {σ = σ} h a0 i)
sub-idV {σ = σ} h (inr a0) i = inr (sub-idV {σ = σ} h a0 i)
sub-idV {σ = σ} h (vpair a0 a1) i = vpair (sub-idV {σ = σ} h a0 i) (sub-idV {σ = σ} h a1 i)
sub-idV {σ = σ} h (thunk a0) i = thunk (sub-idC {σ = σ} h a0 i)

sub-idC {σ = σ} h (case+ a0 a1 a2) i = case+ (sub-idV {σ = σ} h a0 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a1 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a2 i)
sub-idC {σ = σ} h (case× a0 a1) i = case× (sub-idV {σ = σ} h a0 i) (sub-idC {σ = (liftS (liftS σ))} (liftS-id (liftS-id h)) a1 i)
sub-idC {σ = σ} h (force a0) i = force (sub-idV {σ = σ} h a0 i)
sub-idC {σ = σ} h triv = refl
sub-idC {σ = σ} h (lam a0) i = lam (sub-idC {σ = (liftS σ)} (liftS-id h) a0 i)
sub-idC {σ = σ} h (app a0 a1) i = app (sub-idC {σ = σ} h a0 i) (sub-idV {σ = σ} h a1 i)
sub-idC {σ = σ} h (cpair a0 a1) i = cpair (sub-idC {σ = σ} h a0 i) (sub-idC {σ = σ} h a1 i)
sub-idC {σ = σ} h (π₁ a0) i = π₁ (sub-idC {σ = σ} h a0 i)
sub-idC {σ = σ} h (π₂ a0) i = π₂ (sub-idC {σ = σ} h a0 i)
sub-idC {σ = σ} h (ret a0) i = ret (sub-idV {σ = σ} h a0 i)
sub-idC {σ = σ} h (bind a0 a1) i = bind (sub-idC {σ = σ} h a0 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a1 i)
sub-idC {σ = σ} h (op o k) i = op o (λ j → sub-idC {σ = σ} h (k j) i)

-- Single substitution after a lifted substitution:
--   M[γ↑][V/x] = M[γ, V/x]

sub₀-lift : (σ : Subst Γ Δ) (M : Comp (Δ ▸ A) B) (V : Val Γ A)
  → (subC (liftS σ) M) [ V ]₀ ≡ subC (σ ,ₛ V) M
sub₀-lift σ M V = sub-subC h M
  where
  h : SubSub (idS ,ₛ V) (liftS σ) (σ ,ₛ V)
  h vz     = refl
  h (vs x) = sub-renV (λ _ → refl) (σ x) ∙ sub-idV (λ _ → refl) (σ x)

sub₁-lift : (σ : Subst Γ Δ) (M : Comp (Δ ▸ A ▸ A') B) (V : Val Γ A) (W : Val Γ A')
  → (subC (liftS (liftS σ)) M) [ V , W ]₁ ≡ subC (σ ,ₛ V ,ₛ W) M
sub₁-lift σ M V W = sub-subC h M
  where
  h : SubSub (idS ,ₛ V ,ₛ W) (liftS (liftS σ)) (σ ,ₛ V ,ₛ W)
  h vz          = refl
  h (vs vz)     = refl
  h (vs (vs x)) = sub-renV (λ _ → refl) (wkV (σ x))
                ∙ sub-renV (λ _ → refl) (σ x) ∙ sub-idV (λ _ → refl) (σ x)
