{-
  Option C, the termination measure on closed computations, and its
  decrease lemmas.

    reducts      finite branching: an enumeration of all one-step
                 reducts of a closed computation (head redex, steps
                 under a frame, steps inside one argument of an
                 operation)
    ‖_‖          the measure, by well-founded recursion on strong
                 normalization (OptionC.SNFundamental):

                   ‖ret V‖    = 0
                   ‖op(M⃗)‖   = 1 + Σᵢ ‖Mᵢ‖
                   ‖M‖        = 1 + max over all M ↦ M' of ‖M'‖   (otherwise)

    step-dec     (a)  M ↦ M'  ⇒  ‖M'‖ < ‖M‖
    steps-dec         M ↦* M' in n steps  ⇒  n + ‖M'‖ ≤ ‖M‖
    stack-op-bound, stack-op-measure
                 (b)  depth(S) + 1 + Σₖ ‖S[Mₖ]‖ ≤ ‖S[op(M⃗)]‖, hence
                      Σₖ ‖S[Mₖ]‖ < ‖S[op(M⃗)]‖, from whole-stack and (a)

  The recursion is on the order ≺: M' ≺ M if M ↦ M' or M = op(M⃗) and
  M' is one of the Mᵢ.  It is well founded on every strongly normalizing
  term (sn→≺): a step of an argument lifts to a step of the operation
  (opArg), so an argument is accessible as soon as every reduct of the
  operation is.

  ‖_‖ is opaque: its value is meas M (wf≺ M), and unfolding wf≺ would
  unfold the proof of strong normalization.  The defining equations are
  exported as ‖‖-ret, ‖‖-op and ‖‖-other.  The value does not depend on
  the accessibility proof (isPropAcc).
-}

------------------------------------------------------------------------
-- Finite enumerations

-- Enum P: finitely many elements, each satisfying P, and every element
-- satisfying P is among them.
record Enum {X : Type} (P : X → Type) : Type where
  constructor enum
  field
    size     : ℕ
    elt      : Fin size → X
    sound    : (i : Fin size) → P (elt i)
    complete : {x : X} → P x → Σ[ i ∈ Fin size ] elt i ≡ x
open Enum public

module _ {X : Type} {P : X → Type} where

  emptyE : ({x : X} → ¬ P x) → Enum P
  emptyE none = enum 0 (λ ()) (λ ()) (λ p → ⊥.rec (none p))

  singleE : (x : X) → P x → ({y : X} → P y → x ≡ y) → Enum P
  singleE x p u = enum 1 (λ _ → x) (λ _ → p) (λ q → zero , u q)

module _ {X : Type} {P Q : X → Type} where

  convE : ({x : X} → P x → Q x) → ({x : X} → Q x → P x) → Enum P → Enum Q
  convE f g e = enum (e .size) (e .elt) (λ i → f (e .sound i)) (λ q → e .complete (g q))

module _ {X : Type} {P Q R : X → Type} where

  unionE : Enum P → Enum Q
    → ({x : X} → P x → R x) → ({x : X} → Q x → R x) → ({x : X} → R x → P x ⊎ Q x)
    → Enum R
  unionE e e' f g h = enum (e .size + e' .size) el so co
    where
    pick : Fin (e .size) ⊎ Fin (e' .size) → X
    pick = ⊎.rec (e .elt) (e' .elt)

    el : Fin (e .size + e' .size) → X
    el i = pick (splitFin {m = e .size} i)

    so' : (c : Fin (e .size) ⊎ Fin (e' .size)) → R (pick c)
    so' (inl a) = f (e .sound a)
    so' (inr b) = g (e' .sound b)

    so : (i : Fin (e .size + e' .size)) → R (el i)
    so i = so' (splitFin {m = e .size} i)

    co' : {x : X} → P x ⊎ Q x → Σ[ i ∈ Fin (e .size + e' .size) ] el i ≡ x
    co' (inl p) =
      finL (e .complete p .fst)
      , cong pick (splitFin-L {b = e' .size} (e .complete p .fst)) ∙ e .complete p .snd
    co' (inr q) =
      finR {m = e .size} (e' .complete q .fst)
      , cong pick (splitFin-R {m = e .size} (e' .complete q .fst)) ∙ e' .complete q .snd

    co : {x : X} → R x → Σ[ i ∈ Fin (e .size + e' .size) ] el i ≡ x
    co r = co' (h r)

mapE : {X Y : Type} {P : X → Type} {R : Y → Type} (φ : X → Y) → Enum P
  → ({x : X} → P x → R (φ x)) → ({y : Y} → R y → Σ[ x ∈ X ] P x × (φ x ≡ y))
  → Enum R
mapE φ e f h =
  enum (e .size) (φ ∘ e .elt) (λ i → f (e .sound i))
    (λ r → e .complete (h r .snd .fst) .fst
         , cong φ (e .complete (h r .snd .fst) .snd) ∙ h r .snd .snd)

-- A union of n enumerations.
bigE : {X : Type} (n : ℕ) {P : Fin n → X → Type} → ((j : Fin n) → Enum (P j))
  → Enum (λ x → Σ[ j ∈ Fin n ] P j x)
bigE zero    es = emptyE λ { (() , _) }
bigE (suc n) {P} es =
  unionE (es zero) (bigE n {P = λ j → P (suc j)} (λ j → es (suc j)))
    (λ p → zero , p) (λ { (j , p) → suc j , p })
    λ { (zero , p) → inl p ; (suc j , p) → inr (j , p) }

-- Finite branching

-- The head of a closed computation: an elimination form (Neutral), or
-- an introduction form, or an operation.
data HeadView : {B : CTy} → CComp B → Type where
  hneu  : {M : CComp B} → Neutral M → HeadView M
  hlam  : {M : Comp cbpv (ε ▸ A) B} → HeadView (lam M)
  hpair : {B₁ B₂ : CTy} {M : CComp B₁} {N : CComp B₂} → HeadView (cpair M N)
  hret  : {V : CVal A} → HeadView (ret V)
  htriv : HeadView triv
  hop   : (o : Op) (k : Fin (ar o) → CComp B) → HeadView (op o k)

headView : (M : CComp B) → HeadView M
headView (case+ _ _ _) = hneu tt
headView (case× _ _)   = hneu tt
headView (force _)     = hneu tt
headView triv          = htriv
headView (lam _)       = hlam
headView (app _ _)     = hneu tt
headView (cpair _ _)   = hpair
headView (π₁ _)        = hneu tt
headView (π₂ _)        = hneu tt
headView (ret _)       = hret
headView (bind _ _)    = hneu tt
headView (op o k)      = hop o k

-- x ≡_ has exactly one element.
pointE : {X : Type} (x : X) → Enum (x ≡_)
pointE x = singleE x refl (λ p → p)

-- Reducts of M V, πᵢ M, x ← M; N from the head of M and the reducts of M.
appE : {M : CComp (A ⇒ B)} {V : CVal A} → HeadView M → Enum (M ↦_) → Enum (app M V ↦_)
appE {V = V} (hneu n) e =
  mapE (λ M' → app M' V) e (appF V)
    (λ s → app-neutral-step n s .fst , app-neutral-step n s .snd .fst
         , sym (app-neutral-step n s .snd .snd))
appE {V = V} (hlam {M = M}) e =
  singleE (M [ V ]₀) (red (β (app-lam M V))) (λ s → sym (app-lam-step s))
appE {V = V} (hop o k) e =
  unionE (pointE (op o (λ j → app (k j) V)))
    (mapE {R = λ y → Σ[ M' ∈ _ ] (op o k ↦ M') × (app M' V ≡ y)}
       (λ M' → app M' V) e (λ s → _ , s , refl) (λ r → r))
    (λ p → subst (app (op o k) V ↦_) p (red (alg-app o k V)))
    (λ { (M' , s , p) → subst (app (op o k) V ↦_) p (appF V s) })
    inv
  where
  inv : {X : CComp _} → app (op o k) V ↦ X
    → (op o (λ j → app (k j) V) ≡ X) ⊎ (Σ[ M' ∈ _ ] (op o k ↦ M') × (app M' V ≡ X))
  inv (red (β ()))
  inv (red (alg-app _ _ _)) = inl refl
  inv (appF _ s)            = inr (_ , s , refl)

π₁E : {B₁ B₂ : CTy} {M : CComp (B₁ & B₂)} → HeadView M → Enum (M ↦_) → Enum (π₁ M ↦_)
π₁E (hneu n) e =
  mapE π₁ e π₁F
    (λ s → π₁-neutral-step n s .fst , π₁-neutral-step n s .snd .fst
         , sym (π₁-neutral-step n s .snd .snd))
π₁E (hpair {M = M}) e = singleE M (red (β (π₁-pair M _))) (λ s → sym (π₁-pair-step s))
π₁E (hop o k) e =
  unionE (pointE (op o (λ j → π₁ (k j))))
    (mapE {R = λ y → Σ[ M' ∈ _ ] (op o k ↦ M') × (π₁ M' ≡ y)} π₁ e (λ s → _ , s , refl) (λ r → r))
    (λ p → subst (π₁ (op o k) ↦_) p (red (alg-π₁ o k)))
    (λ { (M' , s , p) → subst (π₁ (op o k) ↦_) p (π₁F s) })
    inv
  where
  inv : {X : CComp _} → π₁ (op o k) ↦ X
    → (op o (λ j → π₁ (k j)) ≡ X) ⊎ (Σ[ M' ∈ _ ] (op o k ↦ M') × (π₁ M' ≡ X))
  inv (red (β ()))
  inv (red (alg-π₁ _ _)) = inl refl
  inv (π₁F s)            = inr (_ , s , refl)

π₂E : {B₁ B₂ : CTy} {M : CComp (B₁ & B₂)} → HeadView M → Enum (M ↦_) → Enum (π₂ M ↦_)
π₂E (hneu n) e =
  mapE π₂ e π₂F
    (λ s → π₂-neutral-step n s .fst , π₂-neutral-step n s .snd .fst
         , sym (π₂-neutral-step n s .snd .snd))
π₂E (hpair {N = N}) e = singleE N (red (β (π₂-pair _ N))) (λ s → sym (π₂-pair-step s))
π₂E (hop o k) e =
  unionE (pointE (op o (λ j → π₂ (k j))))
    (mapE {R = λ y → Σ[ M' ∈ _ ] (op o k ↦ M') × (π₂ M' ≡ y)} π₂ e (λ s → _ , s , refl) (λ r → r))
    (λ p → subst (π₂ (op o k) ↦_) p (red (alg-π₂ o k)))
    (λ { (M' , s , p) → subst (π₂ (op o k) ↦_) p (π₂F s) })
    inv
  where
  inv : {X : CComp _} → π₂ (op o k) ↦ X
    → (op o (λ j → π₂ (k j)) ≡ X) ⊎ (Σ[ M' ∈ _ ] (op o k ↦ M') × (π₂ M' ≡ X))
  inv (red (β ()))
  inv (red (alg-π₂ _ _)) = inl refl
  inv (π₂F s)            = inr (_ , s , refl)

bindE : {M : CComp (F A)} {N : Comp cbpv (ε ▸ A) B} → HeadView M → Enum (M ↦_)
  → Enum (bind M N ↦_)
bindE {N = N} (hneu n) e =
  mapE (λ M' → bind M' N) e (bindF N)
    (λ s → bind-neutral-step n s .fst , bind-neutral-step n s .snd .fst
         , sym (bind-neutral-step n s .snd .snd))
bindE {N = N} (hret {V = V}) e =
  singleE (N [ V ]₀) (red (β (bind-ret V N))) (λ s → sym (bind-ret-step s))
bindE {N = N} (hop o k) e =
  unionE (pointE (op o (λ j → bind (k j) N)))
    (mapE {R = λ y → Σ[ M' ∈ _ ] (op o k ↦ M') × (bind M' N ≡ y)}
       (λ M' → bind M' N) e (λ s → _ , s , refl) (λ r → r))
    (λ p → subst (bind (op o k) N ↦_) p (red (alg-bind o k N)))
    (λ { (M' , s , p) → subst (bind (op o k) N ↦_) p (bindF N s) })
    inv
  where
  inv : {X : CComp _} → bind (op o k) N ↦ X
    → (op o (λ j → bind (k j) N) ≡ X) ⊎ (Σ[ M' ∈ _ ] (op o k ↦ M') × (bind M' N ≡ X))
  inv s with bind-op-step s
  ... | inl p              = inl (sym p)
  ... | inr (M' , s' , p)  = inr (M' , s' , sym p)

-- Reducts of op o k: one argument steps.
opE : (o : Op) (k : Fin (ar o) → CComp B) → ((j : Fin (ar o)) → Enum (k j ↦_))
  → Enum (op o k ↦_)
opE {B = B} o k es =
  convE
    (λ { (j , N , s , p) → subst (op o k ↦_) p (opArg o k j s) })
    (λ s → let (j , N , s' , p) = op-step s in j , N , s' , sym p)
    (bigE (ar o) {P = P} λ j →
      mapE (λ N → op o (k [ j ≔ N ])) (es j) (λ s → _ , s , refl) (λ r → r))
  where
  P : Fin (ar o) → CComp B → Type
  P j y = Σ[ N ∈ CComp B ] (k j ↦ N) × (op o (k [ j ≔ N ]) ≡ y)

-- Finite branching: every closed computation has finitely many
-- one-step reducts, and reducts enumerates all of them.
reducts : (M : CComp B) → Enum (M ↦_)
reducts (force (thunk M))      =
  singleE M (red (β (force-thunk M))) (λ s → sym (force-thunk-step s))
reducts (force (absurd V))     = ⊥.rec (no-closed-𝟘 V)
reducts (case+ (inl V) M₁ M₂)  =
  singleE (M₁ [ V ]₀) (red (β (case+-inl V M₁ M₂))) (λ s → sym (case+-inl-step s))
reducts (case+ (inr V) M₁ M₂)  =
  singleE (M₂ [ V ]₀) (red (β (case+-inr V M₁ M₂))) (λ s → sym (case+-inr-step s))
reducts (case+ (absurd V) _ _) = ⊥.rec (no-closed-𝟘 V)
reducts (case× (vpair V W) M)  =
  singleE (M [ V , W ]₁) (red (β (case×-pair V W M))) (λ s → sym (case×-step s))
reducts (case× (absurd V) _)   = ⊥.rec (no-closed-𝟘 V)
reducts triv                   = emptyE triv-no-step
reducts (lam M)                = emptyE lam-no-step
reducts (cpair M N)            = emptyE cpair-no-step
reducts (ret V)                = emptyE ret-no-step
reducts (app M V)              = appE (headView M) (reducts M)
reducts (π₁ M)                 = π₁E (headView M) (reducts M)
reducts (π₂ M)                 = π₂E (headView M) (reducts M)
reducts (bind M N)             = bindE (headView M) (reducts M)
reducts (op o k)               = opE o k (λ j → reducts (k j))

------------------------------------------------------------------------
-- The recursion order: reducts and arguments of operations

infix 4 _≺_

data _≺_ {B : CTy} : CComp B → CComp B → Type where
  red≺ : {M M' : CComp B} → M ↦ M' → M' ≺ M
  arg≺ : (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) → k j ≺ op o k

private
  Lifts : CComp B → Type
  Lifts {B} N = (N' : CComp B) → N ↦ N' → Acc _≺_ N'

  -- If every reduct of op o k is accessible, so is every reduct of k j.
  arg-lf : (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) → Lifts (op o k) → Lifts (k j)
  arg-lf o k j lf N' s with lf _ (opArg o k j s)
  ... | acc r = subst (Acc _≺_) (upd-yes k j N' refl) (r _ (arg≺ o (k [ j ≔ N' ]) j))

  -- If every reduct of N is accessible, so is N (structural recursion
  -- on N for the arguments of an operation).
  ≺-acc : (N : CComp B) → Lifts N → Acc _≺_ N
  ≺-below : (N : CComp B) → Lifts N → (y : CComp B) → y ≺ N → Acc _≺_ y

  ≺-acc N lf = acc (≺-below N lf)

  ≺-below N         lf y      (red≺ s)     = lf y s
  ≺-below .(op o k) lf .(k j) (arg≺ o k j) = ≺-acc (k j) (arg-lf o k j lf)

-- Every strongly normalizing computation is accessible for ≺.
sn→≺ : {M : CComp B} → SN M → Acc _≺_ M
sn→≺ {M = M} (acc r) = ≺-acc M (λ N' s → sn→≺ (r N' s))

opaque
  wf≺ : (M : CComp B) → Acc _≺_ M
  wf≺ M = sn→≺ (strong-normalization M)

------------------------------------------------------------------------
-- The measure

maxFin : (n : ℕ) → (Fin n → ℕ) → ℕ
maxFin zero    f = 0
maxFin (suc n) f = max (f zero) (maxFin n (f ∘ suc))

≤-maxFin : (n : ℕ) (f : Fin n → ℕ) (i : Fin n) → f i ≤ maxFin n f
≤-maxFin (suc n) f zero    = left-≤-max
≤-maxFin (suc n) f (suc i) = ≤-trans (≤-maxFin n (f ∘ suc) i) (right-≤-max {m = f zero})

-- 1 + the maximum of f over the enumerated reducts of M.
maxStep : (M : CComp B) → ((y : CComp B) → M ↦ y → ℕ) → ℕ
maxStep M f = suc (maxFin (reducts M .size) (λ i → f (reducts M .elt i) (reducts M .sound i)))

meas : (M : CComp B) → Acc _≺_ M → ℕ
meas (ret V)  _       = 0
meas (op o k) (acc r) = suc (sumFin (ar o) λ j → meas (k j) (r (k j) (arg≺ o k j)))
meas M        (acc r) = maxStep M (λ y s → meas y (r y (red≺ s)))

meas-irr : (M : CComp B) (a a' : Acc _≺_ M) → meas M a ≡ meas M a'
meas-irr M a a' = cong (meas M) (isPropAcc M a a')

-- Neither a return nor an operation.
IsRetOp : CComp B → Type
IsRetOp (ret _)  = Unit
IsRetOp (op _ _) = Unit
IsRetOp _        = ⊥

private
  other-eq : (M : CComp B) (r : (y : CComp B) → y ≺ M → Acc _≺_ y)
    → maxStep M (λ y s → meas y (r y (red≺ s))) ≡ maxStep M (λ y _ → meas y (wf≺ y))
  other-eq M r = cong (maxStep M) (funExt λ y → funExt λ s → meas-irr y (r y (red≺ s)) (wf≺ y))

  meas-op : (o : Op) (k : Fin (ar o) → CComp B) (a : Acc _≺_ (op o k))
    → meas (op o k) a ≡ suc (sumFin (ar o) λ j → meas (k j) (wf≺ (k j)))
  meas-op o k (acc r) = cong suc (sum-cong (ar o) λ j → meas-irr (k j) _ _)

  meas-other : (M : CComp B) → ¬ IsRetOp M → (a : Acc _≺_ M)
    → meas M a ≡ maxStep M (λ y _ → meas y (wf≺ y))
  meas-other (ret _)         n _       = ⊥.rec (n tt)
  meas-other (op _ _)        n _       = ⊥.rec (n tt)
  meas-other M@(case+ _ _ _) _ (acc r) = other-eq M r
  meas-other M@(case× _ _)   _ (acc r) = other-eq M r
  meas-other M@(force _)     _ (acc r) = other-eq M r
  meas-other M@triv          _ (acc r) = other-eq M r
  meas-other M@(lam _)       _ (acc r) = other-eq M r
  meas-other M@(app _ _)     _ (acc r) = other-eq M r
  meas-other M@(cpair _ _)   _ (acc r) = other-eq M r
  meas-other M@(π₁ _)        _ (acc r) = other-eq M r
  meas-other M@(π₂ _)        _ (acc r) = other-eq M r
  meas-other M@(bind _ _)    _ (acc r) = other-eq M r

opaque
  -- The termination measure on closed computations.
  ‖_‖ : CComp B → ℕ
  ‖ M ‖ = meas M (wf≺ M)

  ‖‖-ret : {V : CVal A} → ‖ ret V ‖ ≡ 0
  ‖‖-ret = refl

  ‖‖-op : (o : Op) (k : Fin (ar o) → CComp B) → ‖ op o k ‖ ≡ suc (sumFin (ar o) λ j → ‖ k j ‖)
  ‖‖-op o k = meas-op o k (wf≺ (op o k))

  ‖‖-other : (M : CComp B) → ¬ IsRetOp M
    → ‖ M ‖ ≡ suc (maxFin (reducts M .size) (λ i → ‖ reducts M .elt i ‖))
  ‖‖-other M n = meas-other M n (wf≺ M)

------------------------------------------------------------------------
-- (a) Every step decreases the measure.

private
  redex-not-retop : {R R' : CComp B} → R ⇝ R' → ¬ IsRetOp R
  redex-not-retop (β (bind-ret _ _))     ()
  redex-not-retop (β (force-thunk _))    ()
  redex-not-retop (β (case+-inl _ _ _))  ()
  redex-not-retop (β (case+-inr _ _ _))  ()
  redex-not-retop (β (case×-pair _ _ _)) ()
  redex-not-retop (β (app-lam _ _))      ()
  redex-not-retop (β (π₁-pair _ _))      ()
  redex-not-retop (β (π₂-pair _ _))      ()
  redex-not-retop (alg-app _ _ _)        ()
  redex-not-retop (alg-π₁ _ _)           ()
  redex-not-retop (alg-π₂ _ _)           ()
  redex-not-retop (alg-bind _ _ _)       ()

  -- Not an operation: by the maximum over the enumerated reducts.
  other-dec : {M M' : CComp B} → M ↦ M' → ¬ IsRetOp M → ‖ M' ‖ < ‖ M ‖
  other-dec {M = M} {M'} s n =
    subst (‖ M' ‖ <_) (sym (‖‖-other M n)) (suc-≤-suc le)
    where
    e = reducts M
    i = e .complete s .fst
    le : ‖ M' ‖ ≤ maxFin (e .size) (λ i → ‖ e .elt i ‖)
    le = subst (_≤ maxFin (e .size) (λ i → ‖ e .elt i ‖)) (cong ‖_‖ (e .complete s .snd))
           (≤-maxFin (e .size) (λ i → ‖ e .elt i ‖) i)

  -- An operation: one summand decreases.
  op-dec : (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) (N : CComp B)
    → ‖ N ‖ < ‖ k j ‖ → ‖ op o (k [ j ≔ N ]) ‖ < ‖ op o k ‖
  op-dec o k j N d =
    subst2 _<_ (sym (‖‖-op o (k [ j ≔ N ]))) (sym (‖‖-op o k))
      (suc-≤-suc
        (sum-< (ar o) (λ i → ‖ (k [ j ≔ N ]) i ‖) (λ i → ‖ k i ‖) j
          (λ i ne → cong ‖_‖ (upd-no k j N ne))
          (subst (_< ‖ k j ‖) (sym (cong ‖_‖ (upd-yes k j N refl))) d)))

step-dec : {M M' : CComp B} → M ↦ M' → ‖ M' ‖ < ‖ M ‖
step-dec (opArg o k j {N} s) = op-dec o k j N (step-dec s)
step-dec s@(red r)           = other-dec s (redex-not-retop r)
step-dec s@(appF _ _)        = other-dec s (λ ())
step-dec s@(π₁F _)           = other-dec s (λ ())
step-dec s@(π₂F _)           = other-dec s (λ ())
step-dec s@(bindF _ _)       = other-dec s (λ ())

-- Along M ↦* M', the measure drops by at least the number of steps.
steps-dec : {M M' : CComp B} (p : M ↦* M') → len p + ‖ M' ‖ ≤ ‖ M ‖
steps-dec done    = ≤-refl
steps-dec (s ∷ p) = ≤-trans (suc-≤-suc (steps-dec p)) (step-dec s)

-- (b) The Effect bound

-- ‖S[op(M⃗)]‖ ≥ depth(S) + ‖op(S[M⃗])‖ = depth(S) + 1 + Σₖ ‖S[Mₖ]‖
stack-op-bound : (S : CStack B B') (o : Op) (k : Fin (ar o) → CComp B)
  → depth S + suc (sumFin (ar o) (λ j → ‖ plug S (k j) ‖)) ≤ ‖ plug S (op o k) ‖
stack-op-bound S o k =
  subst (_≤ ‖ plug S (op o k) ‖)
    (cong₂ _+_ (whole-stack-len S o k) (‖‖-op o (λ j → plug S (k j))))
    (steps-dec (whole-stack S o k))

-- Σₖ ‖S[Mₖ]‖ < ‖S[op(M⃗)]‖, for every stack S (including •).
stack-op-measure : (S : CStack B B') (o : Op) (k : Fin (ar o) → CComp B)
  → sumFin (ar o) (λ j → ‖ plug S (k j) ‖) < ‖ plug S (op o k) ‖
stack-op-measure S o k = ≤-trans (≤SumRight {k = depth S}) (stack-op-bound S o k)
