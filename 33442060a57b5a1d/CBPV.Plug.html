{-
  Paper Fig. 2, "Operations": stack plugging S[M] and stack composition
  S'[S], with their laws:

    •[M] = M         (S' ∘ S)[M] = S'[S[M]]         (S[M])[γ] = S[γ][M[γ]]
-}
open import CBPV.Theory

module CBPV.Plug (Sg : Signature) where

open import Cubical.Foundations.Prelude

open import CBPV.Syntax Sg
open import CBPV.Renaming Sg
open import CBPV.Substitution Sg

private variable
  m : Mode
  Γ Δ : Ctx
  A : VTy
  B B' B'' : CTy

-- S[M]
plug : Stack m Γ B B' → Comp m Γ B → Comp m Γ B'
plug •           M = M
plug (appˢ S V)  M = app (plug S M) V
plug (π₁ˢ S)     M = π₁ (plug S M)
plug (π₂ˢ S)     M = π₂ (plug S M)
plug (bindˢ S N) M = bind (plug S M) N
plug trivˢ       M = triv
plug (lamˢ S)    M = lam (plug S (wkC M))
plug (pairˢ S T) M = cpair (plug S M) (plug T M)

-- S'[S]
infixr 9 _∘ˢ_
_∘ˢ_ : Stack m Γ B' B'' → Stack m Γ B B' → Stack m Γ B B''
•           ∘ˢ S = S
appˢ S' V   ∘ˢ S = appˢ (S' ∘ˢ S) V
π₁ˢ S'      ∘ˢ S = π₁ˢ (S' ∘ˢ S)
π₂ˢ S'      ∘ˢ S = π₂ˢ (S' ∘ˢ S)
bindˢ S' N  ∘ˢ S = bindˢ (S' ∘ˢ S) N
trivˢ       ∘ˢ S = trivˢ
lamˢ S'     ∘ˢ S = lamˢ (S' ∘ˢ wkS S)
pairˢ S' T' ∘ˢ S = pairˢ (S' ∘ˢ S) (T' ∘ˢ S)

plug-ren : (r : Ren Γ Δ) (S : Stack m Δ B B') (M : Comp m Δ B)
  → renC r (plug S M) ≡ plug (renS r S) (renC r M)
plug-ren r •           M = refl
plug-ren r (appˢ S V)  M = cong (λ X → app X (renV r V)) (plug-ren r S M)
plug-ren r (π₁ˢ S)     M = cong π₁ (plug-ren r S M)
plug-ren r (π₂ˢ S)     M = cong π₂ (plug-ren r S M)
plug-ren r (bindˢ S N) M = cong (λ X → bind X (renC (liftR r) N)) (plug-ren r S M)
plug-ren r trivˢ       M = refl
plug-ren r (lamˢ S)    M =
  cong lam (plug-ren (liftR r) S (wkC M) ∙ cong (plug (renS (liftR r) S)) (wk-renC r M))
plug-ren r (pairˢ S T) M = cong₂ cpair (plug-ren r S M) (plug-ren r T M)

plug-wk : (S : Stack m Γ B B') (M : Comp m Γ B)
  → wkC {A' = A} (plug S M) ≡ plug (wkS S) (wkC M)
plug-wk = plug-ren vs

plug-sub : (σ : Subst m Γ Δ) (S : Stack m Δ B B') (M : Comp m Δ B)
  → subC σ (plug S M) ≡ plug (subS σ S) (subC σ M)
plug-sub σ •           M = refl
plug-sub σ (appˢ S V)  M = cong (λ X → app X (subV σ V)) (plug-sub σ S M)
plug-sub σ (π₁ˢ S)     M = cong π₁ (plug-sub σ S M)
plug-sub σ (π₂ˢ S)     M = cong π₂ (plug-sub σ S M)
plug-sub σ (bindˢ S N) M = cong (λ X → bind X (subC (liftS σ) N)) (plug-sub σ S M)
plug-sub σ trivˢ       M = refl
plug-sub σ (lamˢ S)    M =
  cong lam (plug-sub (liftS σ) S (wkC M) ∙ cong (plug (subS (liftS σ) S)) (wk-subC σ M))
plug-sub σ (pairˢ S T) M = cong₂ cpair (plug-sub σ S M) (plug-sub σ T M)

plug-∘ : (S' : Stack m Γ B' B'') (S : Stack m Γ B B') (M : Comp m Γ B)
  → plug (S' ∘ˢ S) M ≡ plug S' (plug S M)
plug-∘ •           S M = refl
plug-∘ (appˢ S' V) S M = cong (λ X → app X V) (plug-∘ S' S M)
plug-∘ (π₁ˢ S')    S M = cong π₁ (plug-∘ S' S M)
plug-∘ (π₂ˢ S')    S M = cong π₂ (plug-∘ S' S M)
plug-∘ (bindˢ S' N) S M = cong (λ X → bind X N) (plug-∘ S' S M)
plug-∘ trivˢ       S M = refl
plug-∘ (lamˢ S')   S M =
  cong lam (plug-∘ S' (wkS S) (wkC M) ∙ cong (plug S') (sym (plug-wk S M)))
plug-∘ (pairˢ S' T') S M = cong₂ cpair (plug-∘ S' S M) (plug-∘ T' S M)
