{-
  Paper §2.3, Fig. 1: terms of CBPV(𝒯) and CBPV⁺(𝒯), intrinsically
  typed.

    Val   m Γ A      Γ ⊢ V : A
    Comp  m Γ B      Γ | · ⊢ M : B
    Stack m Γ B B'   Γ | • : B ⊢ S : B'

  The paper presents computations and stacks as one judgement with a
  stoup Δ.  We split it into Comp (empty stoup) and Stack (full stoup).
  A stack rule of Fig. 1 with stoup Δ becomes a Stack constructor whose
  hole is that of its premise:  M V with M a stack is `appˢ S V`,
  x ← M ; N with M a stack is `bindˢ S N`, and so on.  The grey rules
  with stoup Δ⁺ (complex stacks: triv, λ, ⟨_,_⟩) exist only in cbpv⁺.

  Names: σ₁/σ₂ are inl/inr, (V , V') is vpair, ⟨M , N⟩ is cpair,
  x ← M ; N is bind M N, the value-level grey case rules are vcase+ and
  vcase×.
-}
open import OptionC.Theory

module OptionC.Syntax (Sg : Signature) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.FinData using (Fin)

open import OptionC.Types public

open Signature Sg

private variable
  m : Mode
  Γ Δ : Ctx
  A A' A'' : VTy
  B B' B'' B₁ B₂ : CTy

data Val   (m : Mode) (Γ : Ctx) : VTy → Type
data Comp  (m : Mode) (Γ : Ctx) : CTy → Type
data Stack (m : Mode) (Γ : Ctx) : CTy → CTy → Type

data Val m Γ where
  var    : Var Γ A → Val m Γ A
  absurd : Val m Γ 𝟘 → Val m Γ A
  ⋆      : Val m Γ 𝟙
  inl    : Val m Γ A → Val m Γ (A ⊕ A')
  inr    : Val m Γ A' → Val m Γ (A ⊕ A')
  vpair  : Val m Γ A → Val m Γ A' → Val m Γ (A ⊗ A')
  thunk  : Comp m Γ B → Val m Γ (U B)
  -- complex values (CBPV⁺ only)
  vcase+ : ⦃ _ : Plus m ⦄ → Val m Γ (A ⊕ A')
         → Val m (Γ ▸ A) A'' → Val m (Γ ▸ A') A'' → Val m Γ A''
  vcase× : ⦃ _ : Plus m ⦄ → Val m Γ (A ⊗ A')
         → Val m (Γ ▸ A ▸ A') A'' → Val m Γ A''

data Comp m Γ where
  case+ : Val m Γ (A ⊕ A') → Comp m (Γ ▸ A) B → Comp m (Γ ▸ A') B → Comp m Γ B
  case× : Val m Γ (A ⊗ A') → Comp m (Γ ▸ A ▸ A') B → Comp m Γ B
  force : Val m Γ (U B) → Comp m Γ B
  triv  : Comp m Γ ⊤
  lam   : Comp m (Γ ▸ A) B → Comp m Γ (A ⇒ B)
  app   : Comp m Γ (A ⇒ B) → Val m Γ A → Comp m Γ B
  cpair : Comp m Γ B₁ → Comp m Γ B₂ → Comp m Γ (B₁ & B₂)
  π₁    : Comp m Γ (B₁ & B₂) → Comp m Γ B₁
  π₂    : Comp m Γ (B₁ & B₂) → Comp m Γ B₂
  ret   : Val m Γ A → Comp m Γ (F A)
  bind  : Comp m Γ (F A) → Comp m (Γ ▸ A) B → Comp m Γ B
  op    : (o : Op) → (Fin (ar o) → Comp m Γ B) → Comp m Γ B

-- Stack m Γ B B' : a stack with hole of type B and result type B'.
data Stack m Γ where
  •      : Stack m Γ B B
  appˢ   : Stack m Γ B (A ⇒ B') → Val m Γ A → Stack m Γ B B'
  π₁ˢ    : Stack m Γ B (B₁ & B₂) → Stack m Γ B B₁
  π₂ˢ    : Stack m Γ B (B₁ & B₂) → Stack m Γ B B₂
  bindˢ  : Stack m Γ B (F A) → Comp m (Γ ▸ A) B' → Stack m Γ B B'
  -- complex stacks (CBPV⁺ only)
  trivˢ  : ⦃ _ : Plus m ⦄ → Stack m Γ B ⊤
  lamˢ   : ⦃ _ : Plus m ⦄ → Stack m (Γ ▸ A) B B' → Stack m Γ B (A ⇒ B')
  pairˢ  : ⦃ _ : Plus m ⦄ → Stack m Γ B B₁ → Stack m Γ B B₂ → Stack m Γ B (B₁ & B₂)
