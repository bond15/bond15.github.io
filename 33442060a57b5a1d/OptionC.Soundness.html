{-
  Soundness of Option C's reduction for the denotation:
      M ↦ M'  ⇒  ⟦M⟧ = ⟦M'⟧          for closed M, M' : Comp(B).

  Adapted from B's TreeSoundness.  The β-steps are proved as in B.  Each
  algebraicity redex is B's whole-stack lemma den-plug-op at the
  one-frame stack (appˢ • V, π₁ˢ •, π₂ˢ • or bindˢ • N).  The frame rules
  are congruence of ⟦-⟧, and opArg is proved as in B.
-}
open import Cubical.Foundations.Prelude using (Type)
open import OptionC.Theory
open import OptionC.FreeModel

module OptionC.Soundness (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.FinData using (Fin; discreteFin)
open import Cubical.Relation.Nullary

open Theory 𝒯
open Model
open FreeModel fm

open import OptionC.Fin
open import OptionC.Syntax sig
open import OptionC.Plug sig
open import OptionC.Denotation 𝒯 fm
open import OptionC.DenotationLemmas 𝒯 fm
open import OptionC.Reduction sig

private variable
  B : CTy

sound-red : {N N' : CComp B} → N ↦red N' → ⟦ N ⟧C ρ∅ ≡ ⟦ N' ⟧C ρ∅
sound-red (bind-ret {B = B} V N) =
  ext-η ⟦ B ⟧c (λ a → ⟦ N ⟧C (ρ∅ ▹ a)) (⟦ V ⟧V ρ∅) ∙ sym (den-sub₀ ρ∅ N V)
sound-red (force-thunk M)     = refl
sound-red (case+-inl V M₁ M₂) = sym (den-sub₀ ρ∅ M₁ V)
sound-red (case+-inr V M₁ M₂) = sym (den-sub₀ ρ∅ M₂ V)
sound-red (case×-pair V W M)  = sym (den-sub₁ ρ∅ M V W)
sound-red (app-lam M V)       = sym (den-sub₀ ρ∅ M V)
sound-red (π₁-pair M N)       = refl
sound-red (π₂-pair M N)       = refl

-- ⟦R⟧ = ⟦R'⟧ for every redex.
sound-⇝ : {R R' : CComp B} → R ⇝ R' → ⟦ R ⟧C ρ∅ ≡ ⟦ R' ⟧C ρ∅
sound-⇝ (β r)            = sound-red r
sound-⇝ (alg-app o k V)  = den-plug-op (appˢ • V) o k ρ∅
sound-⇝ (alg-π₁ o k)     = den-plug-op (π₁ˢ •) o k ρ∅
sound-⇝ (alg-π₂ o k)     = den-plug-op (π₂ˢ •) o k ρ∅
sound-⇝ (alg-bind o k N) = den-plug-op (bindˢ • N) o k ρ∅

-- Soundness of ↦.
sound↦ : {M M' : CComp B} → M ↦ M' → ⟦ M ⟧C ρ∅ ≡ ⟦ M' ⟧C ρ∅
sound↦ (red r)      = sound-⇝ r
sound↦ (appF V s)   = cong (λ f → f (⟦ V ⟧V ρ∅)) (sound↦ s)
sound↦ (π₁F s)      = cong fst (sound↦ s)
sound↦ (π₂F s)      = cong snd (sound↦ s)
sound↦ (bindF {B = B} N s) = cong (ext ⟦ B ⟧c (λ a → ⟦ N ⟧C (ρ∅ ▹ a))) (sound↦ s)
sound↦ {B = B} (opArg o k j {N} s) =
  cong (⟦ B ⟧c .alg o) (funExt λ i → args i (discreteFin i j))
  where
  args : (i : Fin (ar o)) → Dec (i ≡ j) → ⟦ k i ⟧C ρ∅ ≡ ⟦ (k [ j ≔ N ]) i ⟧C ρ∅
  args i (yes p) = cong (λ l → ⟦ k l ⟧C ρ∅) p ∙ sound↦ s
                 ∙ cong (λ X → ⟦ X ⟧C ρ∅) (sym (upd-yes k j N p))
  args i (no ¬p) = cong (λ X → ⟦ X ⟧C ρ∅) (sym (upd-no k j N ¬p))

sound↦* : {M M' : CComp B} → M ↦* M' → ⟦ M ⟧C ρ∅ ≡ ⟦ M' ⟧C ρ∅
sound↦* done    = refl
sound↦* (s ∷ p) = sound↦ s ∙ sound↦* p
