{-
  Normal forms and progress.

    progress       : a closed computation steps, or is in normal form
                     (canonical, or an operation whose arguments are normal);
    neutral-not-NF : a neutral term is not a normal form.
-}
open import CBPV.Theory

module CBPV.NormalForms (Sg : Signature) where

private variable
  A : VTy
  B : CTy

------------------------------------------------------------------------
-- Normal forms

Normal : CComp B → Type
Normal M = ∀ {M'} → ¬ (M ↦ M')

-- Normal forms: N ::= ret V | λx.M | ⟨M, M'⟩ | triv | op(N₁, …, Nₙ).
data NF : CComp B → Type where
  nret  : {V : CVal A} → NF (ret V)
  nlam  : {A' : VTy} {M : Comp cbpv (ε ▸ A') B} → NF (lam M)
  npair : {B₁ B₂ : CTy} {M : CComp B₁} {N : CComp B₂} → NF (cpair M N)
  ntriv : NF triv
  nop   : (o : Op) {k : Fin (ar o) → CComp B} → (∀ j → NF (k j)) → NF (op o k)

nf-normal : {M : CComp B} → NF M → Normal M
nf-normal nret       s = canonical-no-step tt s
nf-normal nlam       s = canonical-no-step tt s
nf-normal npair      s = canonical-no-step tt s
nf-normal ntriv      s = canonical-no-step tt s
nf-normal (nop o ns) s =
  let (j , N , sj , _) = op-step s in nf-normal (ns j) sj

------------------------------------------------------------------------
-- Progress

no-closed-𝟘 : CVal 𝟘 → ⊥
no-closed-𝟘 (absurd V) = no-closed-𝟘 V

-- Search a finite family of alternatives.
search : {n : ℕ} {P Q : Fin n → Type}
  → (∀ j → P j ⊎ Q j) → (Σ[ j ∈ Fin n ] P j) ⊎ (∀ j → Q j)
search {zero}  d = inr λ ()
search {suc n} d with d zero | search (λ j → d (suc j))
... | inl p | _            = inl (zero , p)
... | inr q | inl (j , p)  = inl (suc j , p)
... | inr q | inr qs       = inr λ { zero → q ; (suc j) → qs j }

progress : (M : CComp B) → (Σ[ M' ∈ CComp B ] M ↦ M') ⊎ NF M
progress (ret V)    = inr nret
progress (lam M)    = inr nlam
progress (cpair M N) = inr npair
progress triv       = inr ntriv
progress (op o k) with search (λ j → progress (k j))
... | inl (j , (N , s)) = inl (_ , opArg o k j s)
... | inr hs            = inr (nop o hs)
progress (force (thunk M))       = inl (_ , red (β (force-thunk M)))
progress (force (absurd V))      = ⊥.rec (no-closed-𝟘 V)
progress (case+ (inl V) M₁ M₂)   = inl (_ , red (β (case+-inl V M₁ M₂)))
progress (case+ (inr V) M₁ M₂)   = inl (_ , red (β (case+-inr V M₁ M₂)))
progress (case+ (absurd V) _ _)  = ⊥.rec (no-closed-𝟘 V)
progress (case× (vpair V W) M)   = inl (_ , red (β (case×-pair V W M)))
progress (case× (absurd V) _)    = ⊥.rec (no-closed-𝟘 V)
progress (app M V) with progress M
... | inl (M' , s) = inl (_ , appF V s)
... | inr nlam     = inl (_ , red (β (app-lam _ V)))
... | inr (nop o _) = inl (_ , red (alg-app o _ V))
progress (π₁ M) with progress M
... | inl (M' , s)  = inl (_ , π₁F s)
... | inr npair     = inl (_ , red (β (π₁-pair _ _)))
... | inr (nop o _) = inl (_ , red (alg-π₁ o _))
progress (π₂ M) with progress M
... | inl (M' , s)  = inl (_ , π₂F s)
... | inr npair     = inl (_ , red (β (π₂-pair _ _)))
... | inr (nop o _) = inl (_ , red (alg-π₂ o _))
progress (bind M N) with progress M
... | inl (M' , s)  = inl (_ , bindF N s)
... | inr nret      = inl (_ , red (β (bind-ret _ N)))
... | inr (nop o _) = inl (_ , red (alg-bind o _ N))

------------------------------------------------------------------------
-- A neutral term is not a normal form.
neutral-not-NF : {M : CComp B} → Neutral M → NF M → ⊥
neutral-not-NF n nret      = n
neutral-not-NF n nlam      = n
neutral-not-NF n npair     = n
neutral-not-NF n ntriv     = n
neutral-not-NF n (nop o _) = n
