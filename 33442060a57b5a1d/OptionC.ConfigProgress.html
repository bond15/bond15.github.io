{-
  Option C, progress for configurations (paper §5.4), proved
  syntactically from tree progress (OptionC.NormalForms.progress), not
  from the logical relation (Option B's D11).

  A configuration ⟨s, N⃗⟩ is final, T(ret)(ṽ), or takes a step.  At each
  position p, tree progress says N_p takes a tree step (a Pure step of
  the configuration), or is normal; at type F A a normal form
  is ret V or an operation (an Effect step with the empty stack).
-}
open import OptionC.Theory
open import OptionC.OperationalModel

module OptionC.ConfigProgress (𝒯 : Theory) (om : OperationalModel 𝒯) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.FinData using (Fin)

open Theory 𝒯
open OperationalModel om
open import OptionC.Polynomial renaming (plug to _⟪_⟫)
open import OptionC.Syntax sig
open import OptionC.Reduction sig
open import OptionC.NormalForms sig using (progress; NF; nret; nop; search)
open import OptionC.ConfigReduction 𝒯 om

private variable
  A : VTy

-- M steps in every one-hole context.
StepsIn : CComp (F A) → Type
StepsIn {A} M = (C : ∂ poly (CComp (F A))) → Σ[ t' ∈ Config A ] C ⟪ M ⟫ ↦T t'

-- A closed computation of type F A is a return, or steps in every
-- context: a tree step is a Pure step, an operation is an Effect step.
classify : (M : CComp (F A)) → (Σ[ M' ∈ CComp (F A) ] StepsIn M) ⊎ (Σ[ V ∈ CVal A ] ret V ≡ M)
classify M with progress M
... | inl (M' , s)  = inl (M' , λ C → _ , pure C s)
... | inr nret      = inr (_ , refl)
... | inr (nop o _) = inl (M , λ C → _ , effect C • o _)

-- Progress for configurations.
progress-T : (t : Config A)
  → (Σ[ t' ∈ Config A ] t ↦T t') ⊎ (Σ[ ṽ ∈ T (CVal A) ] t ≡ pmap ret ṽ)
progress-T (s , γ) with search (λ p → classify (γ p))
... | inl (p , (_ , steps)) =
  inl (steps (unplug (s , γ) p) .fst
      , subst (_↦T steps (unplug (s , γ) p) .fst) (plug-unplug (s , γ) p)
          (steps (unplug (s , γ) p) .snd))
... | inr rets =
  inr ((s , λ p → rets p .fst) , ΣPathP (refl , funExt λ p → sym (rets p .snd)))
