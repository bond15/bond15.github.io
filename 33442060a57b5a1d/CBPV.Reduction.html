{-
  Reduction on closed CBPV(𝒯) computations in congruence
  form, with algebraicity through one frame at a time.

  Redexes R ⇝ R':

    β-steps (_↦red_)      x ← ret V; N ⇝ N[V/x]      force(thunk M) ⇝ M
                          case+, case×, (λx.M) V, πᵢ⟨M₁,M₂⟩
    algebraicity          op(M⃗) V      ⇝ op(M₁ V, …, Mₙ V)
                          πᵢ op(M⃗)     ⇝ op(πᵢ M₁, …, πᵢ Mₙ)
                          x ← op(M⃗); N ⇝ op(x ← M₁; N, …, x ← Mₙ; N)

  Reduction:

    red    R ⇝ R'   ⇒  R ↦ R'
    appF   M ↦ M'   ⇒  M V ↦ M' V
    π₁F    M ↦ M'   ⇒  π₁ M ↦ π₁ M'           (and π₂F)
    bindF  M ↦ M'   ⇒  (x ← M; N) ↦ (x ← M'; N)
    opArg  Mⱼ ↦ N   ⇒  op(…, Mⱼ, …) ↦ op(…, N, …)

  Closure under stacks is given per frame (appF, π₁F, π₂F, bindF); the
  general rule M ↦ M' ⇒ S[M] ↦ S[M'] is step-plug.  Per-frame rules keep
  the index of every constructor a plain term former, so inversion is
  by pattern matching.  There is no reflexivity, and no step under λ,
  thunk, the branches of a case, or the continuation N of a bind.

  Facts proved here:
    1. ↦ is closed under stacks                       step-plug
    2. ↦ is closed under operations                   step-opArg (= opArg)
    3. the whole-stack rule is derivable, in depth S steps
                                                      whole-stack, whole-stack-len
    4. ret V, λ, ⟨M,N⟩, triv are normal; an operation steps only inside
       an argument                                    canonical-no-step, op-step
    5. ↦ is not deterministic                         nondet
  Soundness for the denotation is in CBPV.Soundness.
-}
open import CBPV.Theory

private variable
  A A' : VTy
  B B' B'' B₁ B₂ : CTy

-- Val(A), Comp(B), closed stacks: closed CBPV(𝒯) terms.
CVal : VTy → Type
CVal A = Val cbpv ε A

CComp : CTy → Type
CComp B = Comp cbpv ε B

CStack : CTy → CTy → Type
CStack B B' = Stack cbpv ε B B'

------------------------------------------------------------------------
-- Redexes

infix 4 _↦red_ _⇝_ _↦_ _↦*_

-- β-steps.
data _↦red_ : CComp B → CComp B → Type where
  bind-ret    : (V : CVal A) (N : Comp cbpv (ε ▸ A) B) → bind (ret V) N ↦red N [ V ]₀
  force-thunk : (M : CComp B) → force (thunk M) ↦red M
  case+-inl   : (V : CVal A) (M₁ : Comp cbpv (ε ▸ A) B) (M₂ : Comp cbpv (ε ▸ A') B)
              → case+ (inl V) M₁ M₂ ↦red M₁ [ V ]₀
  case+-inr   : (V : CVal A') (M₁ : Comp cbpv (ε ▸ A) B) (M₂ : Comp cbpv (ε ▸ A') B)
              → case+ (inr V) M₁ M₂ ↦red M₂ [ V ]₀
  case×-pair  : (V : CVal A) (W : CVal A') (M : Comp cbpv (ε ▸ A ▸ A') B)
              → case× (vpair V W) M ↦red M [ V , W ]₁
  app-lam     : (M : Comp cbpv (ε ▸ A) B) (V : CVal A) → app (lam M) V ↦red M [ V ]₀
  π₁-pair     : (M : CComp B₁) (N : CComp B₂) → π₁ (cpair M N) ↦red M
  π₂-pair     : (M : CComp B₁) (N : CComp B₂) → π₂ (cpair M N) ↦red N

-- R ⇝ R': a β-step, or algebraicity through one frame.
data _⇝_ : CComp B → CComp B → Type where
  β        : {R R' : CComp B} → R ↦red R' → R ⇝ R'
  alg-app  : (o : Op) (k : Fin (ar o) → CComp (A ⇒ B)) (V : CVal A)
           → app (op o k) V ⇝ op o (λ j → app (k j) V)
  alg-π₁   : (o : Op) (k : Fin (ar o) → CComp (B₁ & B₂))
           → π₁ (op o k) ⇝ op o (λ j → π₁ (k j))
  alg-π₂   : (o : Op) (k : Fin (ar o) → CComp (B₁ & B₂))
           → π₂ (op o k) ⇝ op o (λ j → π₂ (k j))
  alg-bind : (o : Op) (k : Fin (ar o) → CComp (F A)) (N : Comp cbpv (ε ▸ A) B)
           → bind (op o k) N ⇝ op o (λ j → bind (k j) N)

------------------------------------------------------------------------
-- Reduction, in congruence form

-- R ↦ R' for a redex; one elimination frame at a time around a step
-- (closure under stacks, per frame); one operation argument at a time.
-- There is no reflexivity and no step under λ, thunk, the branches of
-- a case, or the continuation of a bind.
data _↦_ : CComp B → CComp B → Type where
  red    : {R R' : CComp B} → R ⇝ R' → R ↦ R'
  appF   : {M M' : CComp (A ⇒ B)} (V : CVal A) → M ↦ M' → app M V ↦ app M' V
  π₁F    : {M M' : CComp (B₁ & B₂)} → M ↦ M' → π₁ M ↦ π₁ M'
  π₂F    : {M M' : CComp (B₁ & B₂)} → M ↦ M' → π₂ M ↦ π₂ M'
  bindF  : {M M' : CComp (F A)} (N : Comp cbpv (ε ▸ A) B) → M ↦ M' → bind M N ↦ bind M' N
  opArg  : (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) {N : CComp B}
         → k j ↦ N → op o k ↦ op o (k [ j ≔ N ])

data _↦*_ : CComp B → CComp B → Type where
  done : {M : CComp B} → M ↦* M
  _∷_  : {L M N : CComp B} → L ↦ M → M ↦* N → L ↦* N

infixr 5 _∷_ _++*_

_++*_ : {L M N : CComp B} → L ↦* M → M ↦* N → L ↦* N
done    ++* q = q
(s ∷ p) ++* q = s ∷ (p ++* q)

-- The number of steps.
len : {M N : CComp B} → M ↦* N → ℕ
len done    = 0
len (_ ∷ p) = suc (len p)

len-++* : {L M N : CComp B} (p : L ↦* M) (q : M ↦* N) → len (p ++* q) ≡ len p + len q
len-++* done    q = refl
len-++* (s ∷ p) q = cong suc (len-++* p q)

------------------------------------------------------------------------
-- Fact 1: closed under stacks.  (In mode cbpv a stack has only the
-- frames •, appˢ, π₁ˢ, π₂ˢ, bindˢ: the complex stacks need an instance
-- of Plus cbpv, which is empty.)

step-plug : (S : CStack B B') {M M' : CComp B} → M ↦ M' → plug S M ↦ plug S M'
step-plug •           s = s
step-plug (appˢ S V)  s = appF V (step-plug S s)
step-plug (π₁ˢ S)     s = π₁F (step-plug S s)
step-plug (π₂ˢ S)     s = π₂F (step-plug S s)
step-plug (bindˢ S N) s = bindF N (step-plug S s)

-- The same for many steps, preserving the number of steps.
steps-plug : (S : CStack B B') {M M' : CComp B} → M ↦* M' → plug S M ↦* plug S M'
steps-plug S done    = done
steps-plug S (s ∷ p) = step-plug S s ∷ steps-plug S p

steps-plug-len : (S : CStack B B') {M M' : CComp B} (p : M ↦* M')
  → len (steps-plug S p) ≡ len p
steps-plug-len S done    = refl
steps-plug-len S (s ∷ p) = cong suc (steps-plug-len S p)

------------------------------------------------------------------------
-- Fact 2: closed under operations.  This is the constructor opArg.

step-opArg : (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) {N : CComp B}
  → k j ↦ N → op o k ↦ op o (k [ j ≔ N ])
step-opArg = opArg

------------------------------------------------------------------------
-- Fact 3: the whole-stack rule S[op(M⃗)] ↦ op(S[M⃗]) is derivable, in
-- depth S steps (one algebraicity step per frame, innermost first).

depth : CStack B B' → ℕ
depth •           = 0
depth (appˢ S V)  = suc (depth S)
depth (π₁ˢ S)     = suc (depth S)
depth (π₂ˢ S)     = suc (depth S)
depth (bindˢ S N) = suc (depth S)

-- Push through the inner stack S, then one algebraicity step for the
-- outer frame Fr.
private
  push : (Fr : CStack B' B'') (S : CStack B B') (o : Op) (k : Fin (ar o) → CComp B)
    → plug S (op o k) ↦* op o (λ j → plug S (k j))
    → plug Fr (op o (λ j → plug S (k j))) ↦ op o (λ j → plug Fr (plug S (k j)))
    → plug Fr (plug S (op o k)) ↦* op o (λ j → plug Fr (plug S (k j)))
  push Fr S o k p s = steps-plug Fr p ++* (s ∷ done)

whole-stack : (S : CStack B B') (o : Op) (k : Fin (ar o) → CComp B)
  → plug S (op o k) ↦* op o (λ j → plug S (k j))
whole-stack •           o k = done
whole-stack (appˢ S V)  o k =
  push (appˢ • V) S o k (whole-stack S o k) (red (alg-app o _ V))
whole-stack (π₁ˢ S)     o k =
  push (π₁ˢ •) S o k (whole-stack S o k) (red (alg-π₁ o _))
whole-stack (π₂ˢ S)     o k =
  push (π₂ˢ •) S o k (whole-stack S o k) (red (alg-π₂ o _))
whole-stack (bindˢ S N) o k =
  push (bindˢ • N) S o k (whole-stack S o k) (red (alg-bind o _ N))

private
  push-len : (Fr : CStack B' B'') (S : CStack B B') (o : Op) (k : Fin (ar o) → CComp B)
    (p : plug S (op o k) ↦* op o (λ j → plug S (k j)))
    (s : plug Fr (op o (λ j → plug S (k j))) ↦ op o (λ j → plug Fr (plug S (k j))))
    → len (push Fr S o k p s) ≡ suc (len p)
  push-len Fr S o k p s =
    len-++* (steps-plug Fr p) (s ∷ done)
    ∙ cong (_+ 1) (steps-plug-len Fr p)
    ∙ +-comm (len p) 1

whole-stack-len : (S : CStack B B') (o : Op) (k : Fin (ar o) → CComp B)
  → len (whole-stack S o k) ≡ depth S
whole-stack-len •           o k = refl
whole-stack-len (appˢ S V)  o k =
  push-len (appˢ • V) S o k _ _ ∙ cong suc (whole-stack-len S o k)
whole-stack-len (π₁ˢ S)     o k =
  push-len (π₁ˢ •) S o k _ _ ∙ cong suc (whole-stack-len S o k)
whole-stack-len (π₂ˢ S)     o k =
  push-len (π₂ˢ •) S o k _ _ ∙ cong suc (whole-stack-len S o k)
whole-stack-len (bindˢ S N) o k =
  push-len (bindˢ • N) S o k _ _ ∙ cong suc (whole-stack-len S o k)

------------------------------------------------------------------------
-- Fact 4: normal forms

-- ret V, λ, ⟨M,N⟩ and triv.
Canonical : CComp B → Type
Canonical (ret _)     = Unit
Canonical (lam _)     = Unit
Canonical (cpair _ _) = Unit
Canonical triv        = Unit
Canonical _           = ⊥

private
  redex-not-canonical : {R R' : CComp B} → R ⇝ R' → ¬ Canonical R
  redex-not-canonical (β (bind-ret _ _))     ()
  redex-not-canonical (β (force-thunk _))    ()
  redex-not-canonical (β (case+-inl _ _ _))  ()
  redex-not-canonical (β (case+-inr _ _ _))  ()
  redex-not-canonical (β (case×-pair _ _ _)) ()
  redex-not-canonical (β (app-lam _ _))      ()
  redex-not-canonical (β (π₁-pair _ _))      ()
  redex-not-canonical (β (π₂-pair _ _))      ()
  redex-not-canonical (alg-app _ _ _)        ()
  redex-not-canonical (alg-π₁ _ _)           ()
  redex-not-canonical (alg-π₂ _ _)           ()
  redex-not-canonical (alg-bind _ _ _)       ()

-- No step out of a canonical form.  (The frame and operation cases are
-- absurd: Canonical computes to ⊥ on app, π₁, π₂, bind and op.)
canonical-no-step : {M M' : CComp B} → Canonical M → ¬ (M ↦ M')
canonical-no-step c  (red r)         = redex-not-canonical r c
canonical-no-step () (appF _ _)
canonical-no-step () (π₁F _)
canonical-no-step () (π₂F _)
canonical-no-step () (bindF _ _)
canonical-no-step () (opArg _ _ _ _)

ret-no-step : {V : CVal A} {M' : CComp (F A)} → ¬ (ret V ↦ M')
ret-no-step = canonical-no-step tt

lam-no-step : {M : Comp cbpv (ε ▸ A) B} {M' : CComp (A ⇒ B)} → ¬ (lam M ↦ M')
lam-no-step = canonical-no-step tt

cpair-no-step : {M : CComp B₁} {N : CComp B₂} {M' : CComp (B₁ & B₂)} → ¬ (cpair M N ↦ M')
cpair-no-step = canonical-no-step tt

triv-no-step : {M' : CComp ⊤} → ¬ (triv ↦ M')
triv-no-step = canonical-no-step tt

-- What a step from M says when M is an operation.
private
  OpStep : CComp B → CComp B → Type
  OpStep (op o k) M' =
    Σ[ j ∈ Fin (ar o) ] Σ[ N ∈ CComp _ ] (k j ↦ N) × (M' ≡ op o (k [ j ≔ N ]))
  OpStep _ M' = Unit

  redex-opStep : {R R' : CComp B} → R ⇝ R' → OpStep R R'
  redex-opStep (β (bind-ret _ _))     = tt
  redex-opStep (β (force-thunk _))    = tt
  redex-opStep (β (case+-inl _ _ _))  = tt
  redex-opStep (β (case+-inr _ _ _))  = tt
  redex-opStep (β (case×-pair _ _ _)) = tt
  redex-opStep (β (app-lam _ _))      = tt
  redex-opStep (β (π₁-pair _ _))      = tt
  redex-opStep (β (π₂-pair _ _))      = tt
  redex-opStep (alg-app _ _ _)        = tt
  redex-opStep (alg-π₁ _ _)           = tt
  redex-opStep (alg-π₂ _ _)           = tt
  redex-opStep (alg-bind _ _ _)       = tt

  opStep : {M M' : CComp B} → M ↦ M' → OpStep M M'
  opStep (red r)          = redex-opStep r
  opStep (appF V s)       = tt
  opStep (π₁F s)          = tt
  opStep (π₂F s)          = tt
  opStep (bindF N s)      = tt
  opStep (opArg o k j {N} s) = j , N , s , refl

-- An operation steps only inside one of its arguments.
op-step : {o : Op} {k : Fin (ar o) → CComp B} {M' : CComp B} → op o k ↦ M'
  → Σ[ j ∈ Fin (ar o) ] Σ[ N ∈ CComp B ] (k j ↦ N) × (M' ≡ op o (k [ j ≔ N ]))
op-step = opStep

------------------------------------------------------------------------
-- Fact 5: ↦ is not deterministic.
--
-- If an argument k j of op o k steps, then app (op o k) V steps both by
-- algebraicity and inside the argument, and the two reducts differ
-- (one is an operation, the other an application).

private
  IsOp : CComp B → Type
  IsOp (op _ _) = Unit
  IsOp _        = ⊥

op≢app : {o : Op} {k : Fin (ar o) → CComp B} {M : CComp (A ⇒ B)} {V : CVal A}
  → ¬ (op o k ≡ app M V)
op≢app p = subst IsOp p tt

nondet : (o : Op) (k : Fin (ar o) → CComp (A ⇒ B)) (V : CVal A) (j : Fin (ar o)) {N : CComp (A ⇒ B)}
  → k j ↦ N
  → (app (op o k) V ↦ op o (λ i → app (k i) V))
  × (app (op o k) V ↦ app (op o (k [ j ≔ N ])) V)
  × (¬ (op o (λ i → app (k i) V) ≡ app (op o (k [ j ≔ N ])) V))
nondet o k V j s = red (alg-app o k V) , appF V (opArg o k j s) , op≢app
