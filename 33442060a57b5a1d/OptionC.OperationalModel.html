{-
  Paper §5.2, Definition "Operational Model": a free model monad T for
  𝒯 whose carrier is a polynomial functor and whose induced functorial
  action is the polynomial one.
-}
module OptionC.OperationalModel where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function

open import OptionC.Theory
open import OptionC.FreeModel
open import OptionC.Polynomial

module _ (𝒯 : Theory) where

  record OperationalModel : Type₁ where
    field
      poly : Poly
      free : FreeModel 𝒯 ⟦ poly ⟧

    open FreeModel free public

    field
      -- "the induced functorial structure on T is polynomial"
      map-poly : {X Y : Type} (f : X → Y) (t : ⟦ poly ⟧ X) → map f t ≡ pmap f t

    map-plug : {X Y : Type} (f : X → Y) (C : ∂ poly X) (x : X)
      → map f (plug C x) ≡ plug (∂map f C) (f x)
    map-plug f C x = map-poly f (plug C x) ∙ plug-map f C x
