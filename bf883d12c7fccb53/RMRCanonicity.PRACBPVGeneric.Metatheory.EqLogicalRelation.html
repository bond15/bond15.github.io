-- The equational logical relation, its fundamental theorem, and
-- equational canonicity (POPL's EqLogicalRelation).
--
-- As in POPL (its D6), the predicates are written by hand, with 𝒞⟦ F A ⟧
-- inductive, and the fundamental theorem is proved by induction on terms.
-- As in POPL (its D7), terms are not quotiented, so the predicates are
-- closed under ≈ by construction: values of sums and products are
-- related up to ≈, and 𝒞⟦ F A ⟧ has a third constructor conv.
--
-- CLOSED TERMS ARE THOSE OF LOCK TELESCOPES.  POPL's relation is on
-- closed terms.  Here an operation's continuations live behind locks, so
-- the closed computations of interest live at lock telescopes ∅ w ⧀ ⋯
-- (IsLT, from the termination relation).  As there, the relation is
-- defined at every telescope, and its Kripke quantifiers (at U B and
-- A ⇒ B) range over chains into lock telescopes (Chain.agda); stage B's
-- relation is reused for IsLT, LT-ne and the chain laws, not changed.
--
--   FPred R Θ M      the 𝒞⟦ F A ⟧ predicate (POPL's FPred):
--                      ret   R Θ V                        → FPred R Θ (return V)
--                      node  ∀ a ps. FPred R (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)
--                                                         → FPred R Θ (op o vs k)
--                      conv  M ≈c M' , FPred R Θ M'       → FPred R Θ M
--   𝒱⟦ 𝟘 ⟧ Θ V = ⊥                         𝒱⟦ 𝟙 ⟧ Θ V = ⊤
--   𝒱⟦ A ⊕ A' ⟧ Θ V = Σ X. V ≈ inl X × 𝒱⟦ A ⟧ Θ X  ⊎  Σ X'. V ≈ inr X' × 𝒱⟦ A' ⟧ Θ X'
--   𝒱⟦ A × A' ⟧ Θ V = Σ X X'. V ≈ pair X X' × 𝒱⟦ A ⟧ Θ X × 𝒱⟦ A' ⟧ Θ X'
--   𝒱⟦ U B ⟧ Θ V = ∀ Θ' lock telescope, χ : Chain Θ Θ'.  𝒞⟦ B ⟧ Θ' (force (chV χ V))
--   𝒱⟦ [ κ ] A ⟧ Θ V = ∀ q.  𝒱⟦ A ⟧ (Θ ⧀⟨ κ , q ⟩) (openV V q)
--   𝒱⟦ base b ⟧ Θ V = ⊤
--   𝒞⟦ F A ⟧ = FPred 𝒱⟦ A ⟧
--   𝒞⟦ A ⇒ B ⟧ Θ M = ∀ Θ' lock telescope, χ, V.  𝒱⟦ A ⟧ Θ' V → 𝒞⟦ B ⟧ Θ' (app (chC χ M) V)
--
-- CLOSURE.  𝒱-conv, 𝒞-conv: the predicates are closed under ≈ (at U B and
-- A ⇒ B along every chain: ch≈, EqSubstitution; at [ κ ] A by open≈).
-- mono𝒱, mono𝒞: Kripke monotonicity along chains (a conv is moved by
-- ch≈c).  𝒞-op: the predicates are closed under the operations (POPL's
-- CPred condition; at A ⇒ B by op-stack at app• hole V).  seq𝒞 is POPL's
-- fund-bind (F-β at a leaf, op-stack at hole to• N at a node, to≈ at a
-- conv); case𝒞, case₊𝒞 use the β laws after the value's ≈.
--
-- FUNDAMENTAL THEOREM.  fundV / fundC, as stage B's: for IsLT Σ, a chain
-- χ : Chain Θ Σ and a related environment (𝒢 χ), the term under χ is
-- related.  The compatibility lemmas are stage B's with each anti-reduction
-- step replaced by conv along the corresponding β law.
--
-- EQUATIONAL CANONICITY (POPL's corollary).  On a lock telescope every
-- computation of type F A is ≈ to reify t for an operation tree t whose
-- leaves are values of lock telescopes, i.e. closed values (OpTree and
-- reify, Termination):
--
--   equational-canonicity : IsLT Θ → (M : Comp Θ (F A)) → Σ t. M ≈c reify t

----------------------------------------------------------------------
-- the relation

-- 𝒞⟦ F A ⟧, inductively generated, with ≈-closure
data FPred {A : ValTy} (R : (Θ : Tele) → Val Θ A → Type) : (Θ : Tele) → Comp Θ (F A) → Type where
  ret : ∀ {Θ} {V : Val Θ A} → R Θ V → FPred R Θ (return V)
  node : ∀ {Θ} (o : Op) {vs : Vals Θ (params o)}
    {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (F A)} →
    (∀ a ps → FPred R (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)) → FPred R Θ (op o vs k)
  conv : ∀ {Θ} {M M' : Comp Θ (F A)} → M ≈c M' → FPred R Θ M' → FPred R Θ M

𝒱⟦_⟧ : (A : ValTy) (Θ : Tele) → Val Θ A → Type
𝒞⟦_⟧ : (B : CompTy) (Θ : Tele) → Comp Θ B → Type

𝒱⟦ 𝟘 ⟧ Θ V = ⊥
𝒱⟦ 𝟙 ⟧ Θ V = Unit
𝒱⟦ A ⊕ A' ⟧ Θ V =
  (Σ[ X ∈ Val Θ A ] (V ≈v inl X) ×ˢ 𝒱⟦ A ⟧ Θ X) ⊎ (Σ[ X ∈ Val Θ A' ] (V ≈v inr X) ×ˢ 𝒱⟦ A' ⟧ Θ X)
𝒱⟦ A × A' ⟧ Θ V =
  Σ[ X ∈ Val Θ A ] Σ[ X' ∈ Val Θ A' ] (V ≈v pair X X') ×ˢ 𝒱⟦ A ⟧ Θ X ×ˢ 𝒱⟦ A' ⟧ Θ X'
𝒱⟦ U B ⟧ Θ V = ∀ {Θ'} → IsLT Θ' → (χ : Chain Θ Θ') → 𝒞⟦ B ⟧ Θ' (force (chV χ V))
𝒱⟦ [ κ ] A ⟧ Θ V = ∀ q → 𝒱⟦ A ⟧ (Θ ⧀⟨ κ , q ⟩) (openV V q)
𝒱⟦ base b ⟧ Θ V = Unit

𝒞⟦ F A ⟧ = FPred 𝒱⟦ A ⟧
𝒞⟦ A ⇒ B ⟧ Θ M = ∀ {Θ'} → IsLT Θ' → (χ : Chain Θ Θ') (V : Val Θ' A) → 𝒱⟦ A ⟧ Θ' V →
  𝒞⟦ B ⟧ Θ' (app (chC χ M) V)

-- related environments: every neutral goes to a related value
𝒢 : ∀ {Θ Θ'} → Chain Θ Θ' → Type
𝒢 {Θ} {Θ'} χ = ∀ {A} (n : Ne Θ A) → 𝒱⟦ A ⟧ Θ' (chV χ (ne n))

----------------------------------------------------------------------
-- closure under ≈

𝒱-conv : ∀ A {Θ} {V V' : Val Θ A} → V ≈v V' → 𝒱⟦ A ⟧ Θ V' → 𝒱⟦ A ⟧ Θ V
𝒞-conv : ∀ B {Θ} {M M' : Comp Θ B} → M ≈c M' → 𝒞⟦ B ⟧ Θ M' → 𝒞⟦ B ⟧ Θ M

𝒱-conv 𝟘 p ()
𝒱-conv 𝟙 p v = ⋆
𝒱-conv (A ⊕ A') p (inj₁ (X ,ˢ q ,ˢ x)) = inj₁ (X ,ˢ ≈trans p q ,ˢ x)
𝒱-conv (A ⊕ A') p (inj₂ (X ,ˢ q ,ˢ x)) = inj₂ (X ,ˢ ≈trans p q ,ˢ x)
𝒱-conv (A × A') p (X ,ˢ X' ,ˢ q ,ˢ x ,ˢ x') = X ,ˢ X' ,ˢ ≈trans p q ,ˢ x ,ˢ x'
𝒱-conv (U B) p h l χ = 𝒞-conv B (force≈ (ch≈v χ p)) (h l χ)
𝒱-conv ([ κ ] A) p h q = 𝒱-conv A (open≈ p q) (h q)
𝒱-conv (base b) p v = ⋆

𝒞-conv (F A) p c = conv p c
𝒞-conv (A ⇒ B) p h l χ V v = 𝒞-conv B (app≈ (ch≈c χ p) ≈refl) (h l χ V v)

----------------------------------------------------------------------
-- monotonicity along chains

mono𝒱 : ∀ A {Θ Θ'} (χ : Chain Θ Θ') (V : Val Θ A) → 𝒱⟦ A ⟧ Θ V → 𝒱⟦ A ⟧ Θ' (chV χ V)
mono𝒱 𝟘 χ V ()
mono𝒱 𝟙 χ V v = ⋆
mono𝒱 (A ⊕ A') χ V (inj₁ (X ,ˢ q ,ˢ x)) =
  inj₁ (chV χ X ,ˢ ≈trans (ch≈v χ q) (≡→≈v (ch-inl χ X)) ,ˢ mono𝒱 A χ X x)
mono𝒱 (A ⊕ A') χ V (inj₂ (X ,ˢ q ,ˢ x)) =
  inj₂ (chV χ X ,ˢ ≈trans (ch≈v χ q) (≡→≈v (ch-inr χ X)) ,ˢ mono𝒱 A' χ X x)
mono𝒱 (A × A') χ V (X ,ˢ X' ,ˢ q ,ˢ x ,ˢ x') =
  chV χ X ,ˢ chV χ X' ,ˢ ≈trans (ch≈v χ q) (≡→≈v (ch-pair χ X X')) ,ˢ mono𝒱 A χ X x ,ˢ mono𝒱 A' χ X' x'
mono𝒱 (U B) χ V h l χ' = subst (λ X → 𝒞⟦ B ⟧ _ (force X)) (ch-++V χ χ' V) (h l (χ ++ᶜ χ'))
mono𝒱 ([ κ ] A) χ V h q =
  subst (𝒱⟦ A ⟧ _) (ch-openV χ q V) (mono𝒱 A (lockᶜ κ χ q) (openV V (posᶜ κ χ q)) (h (posᶜ κ χ q)))
mono𝒱 (base b) χ V v = ⋆

-- along a renaming
mono𝒱R : ∀ A {Θ Θ'} (ρ : Ren Θ Θ') (V : Val Θ A) → 𝒱⟦ A ⟧ Θ V → 𝒱⟦ A ⟧ Θ' (renV ρ V)
mono𝒱R A ρ V v = subst (𝒱⟦ A ⟧ _) (sub-renV ρ V) (mono𝒱 A (⌜ ρ ⌝ ∷ᶜ []ᶜ) V v)

monoF : ∀ {A Θ Θ'} (χ : Chain Θ Θ') {M : Comp Θ (F A)} → FPred 𝒱⟦ A ⟧ Θ M →
  FPred 𝒱⟦ A ⟧ Θ' (chC χ M)
monoF {A} {Θ' = Θ'} χ (ret {V = V} v) =
  subst (FPred 𝒱⟦ A ⟧ Θ') (sym (ch-return χ V)) (ret (mono𝒱 A χ V v))
monoF {A} {Θ' = Θ'} χ (node o {vs} {k} ts) =
  subst (FPred 𝒱⟦ A ⟧ Θ') (sym (ch-op χ o vs k))
    (node o λ a ps → monoF (wordᶜ (slot o a) χ ps) (ts a (wposᶜ (slot o a) χ ps)))
monoF χ (conv p t) = conv (ch≈c χ p) (monoF χ t)

mono𝒞 : ∀ B {Θ Θ'} (χ : Chain Θ Θ') (M : Comp Θ B) → 𝒞⟦ B ⟧ Θ M → 𝒞⟦ B ⟧ Θ' (chC χ M)
mono𝒞 (F A) χ M t = monoF χ t
mono𝒞 (A ⇒ B) χ M h l χ' V v = subst (λ X → 𝒞⟦ B ⟧ _ (app X V)) (ch-++C χ χ' M) (h l (χ ++ᶜ χ') V v)

----------------------------------------------------------------------
-- operations, sequencing, case, case₊

-- the predicates are closed under the operations
𝒞-op : ∀ B {Θ} (o : Op) {vs : Vals Θ (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B} →
  (∀ a ps → 𝒞⟦ B ⟧ (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)) → 𝒞⟦ B ⟧ Θ (op o vs k)
𝒞-op (F A) o h = node o h
𝒞-op (A ⇒ B) o {vs} {k} h {Θ'} l χ V v =
  subst (λ X → 𝒞⟦ B ⟧ Θ' (app X V)) (sym (ch-op χ o vs k))
    (𝒞-conv B (op-stack (app• hole V) o (chVs χ vs) k')
      (𝒞-op B o λ a ps →
        h a (wposᶜ (slot o a) χ ps) (IsLT-⧀* l (slot o a) ps) (wordᶜ (slot o a) χ ps)
          (renV (dropW idR (slot o a) ps) V) (mono𝒱R A (dropW idR (slot o a) ps) V v)))
  where
  k' : (a : Ar o) (ps : WPos (slot o a) (world Θ')) → Comp (Θ' ⧀*⟨ slot o a , ps ⟩) (A ⇒ B)
  k' a ps = chC (wordᶜ (slot o a) χ ps) (k a (wposᶜ (slot o a) χ ps))

-- the Kripke hypothesis on the continuation of a sequencing
SeqHyp : ∀ {Θ A} B → Comp (Θ , A) B → Type
SeqHyp {Θ} {A} B N = ∀ {Θ'} → IsLT Θ' → (χ : Chain Θ Θ') (V : Val Θ' A) → 𝒱⟦ A ⟧ Θ' V →
  𝒞⟦ B ⟧ Θ' (sub₁ (chC (liftᶜ χ) N) V)

-- POPL's fund-bind: by induction on 𝒞⟦ F A ⟧
seq𝒞 : ∀ {Θ A B} {M : Comp Θ (F A)} {N : Comp (Θ , A) B} → IsLT Θ →
  FPred 𝒱⟦ A ⟧ Θ M → SeqHyp B N → 𝒞⟦ B ⟧ Θ (M to N)
seq𝒞 {B = B} {N = N} l (ret {V = V} v) hN = 𝒞-conv B (F-β V N) (hN l []ᶜ V v)
seq𝒞 {B = B} {N = N} l (node o {vs} {k} ts) hN =
  𝒞-conv B (op-stack (hole to• N) o vs k)
    (𝒞-op B o λ a ps → seq𝒞 (IsLT-⧀* l (slot o a) ps) (ts a ps) λ {Θ'} l' χ' V v →
      subst (λ X → 𝒞⟦ B ⟧ Θ' (sub₁ (chC (liftᶜ χ') X) V))
        (sub-liftS-ren (dropW idR (slot o a) ps) N)
        (hN l' (⌜ dropW idR (slot o a) ps ⌝ ∷ᶜ χ') V v))
seq𝒞 {B = B} l (conv p t) hN = 𝒞-conv B (to≈ p ≈refl) (seq𝒞 l t hN)

case𝒞 : ∀ {Θ A A' B} {M : Comp (Θ , A , A') B} (X : Val Θ (A × A')) → 𝒱⟦ A × A' ⟧ Θ X →
  (∀ V V' → 𝒱⟦ A ⟧ Θ V → 𝒱⟦ A' ⟧ Θ V' → 𝒞⟦ B ⟧ Θ (sub₂ M V V')) → 𝒞⟦ B ⟧ Θ (case X M)
case𝒞 {B = B} {M} X (V ,ˢ V' ,ˢ p ,ˢ v ,ˢ v') h =
  𝒞-conv B (≈trans (case≈ p ≈refl) (×-β V V' M)) (h V V' v v')

case₊𝒞 : ∀ {Θ A A' B} {M : Comp (Θ , A) B} {N : Comp (Θ , A') B} (X : Val Θ (A ⊕ A')) →
  𝒱⟦ A ⊕ A' ⟧ Θ X → (∀ V → 𝒱⟦ A ⟧ Θ V → 𝒞⟦ B ⟧ Θ (sub₁ M V)) →
  (∀ V' → 𝒱⟦ A' ⟧ Θ V' → 𝒞⟦ B ⟧ Θ (sub₁ N V')) → 𝒞⟦ B ⟧ Θ (case₊ X M N)
case₊𝒞 {B = B} {M} {N} X (inj₁ (V ,ˢ p ,ˢ v)) hM hN =
  𝒞-conv B (≈trans (case₊≈ p ≈refl ≈refl) (+-β₁ V M N)) (hM V v)
case₊𝒞 {B = B} {M} {N} X (inj₂ (V ,ˢ p ,ˢ v)) hM hN =
  𝒞-conv B (≈trans (case₊≈ p ≈refl ≈refl) (+-β₂ V M N)) (hN V v)

----------------------------------------------------------------------
-- related environments (as stage B)

-- on a lock telescope every neutral is a name, hence related
𝒢-LT : ∀ {Θ} → IsLT Θ → 𝒢 ([]ᶜ {Θ})
𝒢-LT {Θ} l n =
  subst (λ A → (m : Ne Θ A) → 𝒱⟦ A ⟧ Θ (ne m)) (sym (LT-ne l n .snd)) (λ _ → ⋆) n

mono𝒢 : ∀ {Θ Θ' Θ''} (χ : Chain Θ Θ') → 𝒢 χ → (χ' : Chain Θ' Θ'') → 𝒢 (χ ++ᶜ χ')
mono𝒢 χ G χ' {A} n = subst (𝒱⟦ A ⟧ _) (sym (ch-++V χ χ' (ne n))) (mono𝒱 A χ' (chV χ (ne n)) (G n))

𝒢-lock : ∀ {Θ Θ'} (κ : Lk) (χ : Chain Θ Θ') (q : Pos κ (world Θ')) → 𝒢 χ → 𝒢 (lockᶜ κ χ q)
𝒢-lock κ χ q G {A} (up n) =
  subst (𝒱⟦ A ⟧ _) (sym (ch-↑ χ q (ne n))) (mono𝒱R A (dropL idR q) (chV χ (ne n)) (G n))
𝒢-lock κ χ q G {A} (opn n) = subst (𝒱⟦ A ⟧ _) (sym (ch-openV χ q (ne n))) (G n q)
𝒢-lock κ χ q G (nm k) = ⋆

𝒢-word : ∀ {Θ Θ'} (μ : List Lk) (χ : Chain Θ Θ') (ps : WPos μ (world Θ')) → 𝒢 χ → 𝒢 (wordᶜ μ χ ps)
𝒢-word [] χ ps G = subst 𝒢 (sym (wordᶜ-[] χ ps)) G
𝒢-word (κ ∷ []) χ p G = 𝒢-lock κ χ p G
𝒢-word (κ ∷ κ' ∷ μ) χ (p ,ˢ ps) G =
  transport (λ i → 𝒢 (wordᶜ-cons κ κ' μ χ p ps (~ i)))
    (𝒢-word (κ' ∷ μ) (lockᶜ κ χ p) ps (𝒢-lock κ χ p G))

𝒢-β : ∀ {Θ Θ' A} (χ : Chain Θ Θ') {V : Val Θ' A} → 𝒢 χ → 𝒱⟦ A ⟧ Θ' V →
  𝒢 (liftᶜ χ ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ)
𝒢-β {A = A} χ {V} G v here = subst (𝒱⟦ A ⟧ _) (sym (ch-β-here χ V)) v
𝒢-β χ {V} G v {B} (wkv n) = subst (𝒱⟦ B ⟧ _) (sym (ch-β χ V n)) (G n)

𝒢-β₂ : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') {V : Val Θ' A} {V' : Val Θ' A'} → 𝒢 χ →
  𝒱⟦ A ⟧ Θ' V → 𝒱⟦ A' ⟧ Θ' V' → 𝒢 (liftᶜ (liftᶜ χ) ++ᶜ (idS ,ₛ V ,ₛ V') ∷ᶜ []ᶜ)
𝒢-β₂ {A' = A'} χ {V} {V'} G v v' here = subst (𝒱⟦ A' ⟧ _) (sym (ch-β₂-snd χ V V')) v'
𝒢-β₂ {A = A} χ {V} {V'} G v v' (wkv here) = subst (𝒱⟦ A ⟧ _) (sym (ch-β₂-fst χ V V')) v
𝒢-β₂ χ {V} {V'} G v v' {B} (wkv (wkv n)) = subst (𝒱⟦ B ⟧ _) (sym (ch-β₂ χ V V' n)) (G n)

----------------------------------------------------------------------
-- semantic typing and the compatibility lemmas

record ⊨V_ {Θ A} (V : Val Θ A) : Type where
  field
    runV : ∀ {Σ} → IsLT Σ → (χ : Chain Θ Σ) → 𝒢 χ → 𝒱⟦ A ⟧ Σ (chV χ V)

record ⊨C_ {Θ B} (M : Comp Θ B) : Type where
  field
    runC : ∀ {Σ} → IsLT Σ → (χ : Chain Θ Σ) → 𝒢 χ → 𝒞⟦ B ⟧ Σ (chC χ M)

open ⊨V_
open ⊨C_

compat-ne : ∀ {Θ A} (n : Ne Θ A) → ⊨V (ne n)
compat-ne n .runV l χ G = G n

compat-tt : ∀ {Θ} → ⊨V (tt {Θ})
compat-tt .runV l χ G = ⋆

compat-inl : ∀ {Θ A A'} {V : Val Θ A} → ⊨V V → ⊨V (inl {A' = A'} V)
compat-inl {V = V} hV .runV l χ G = inj₁ (chV χ V ,ˢ ≡→≈v (ch-inl χ V) ,ˢ hV .runV l χ G)

compat-inr : ∀ {Θ A A'} {V : Val Θ A'} → ⊨V V → ⊨V (inr {A = A} V)
compat-inr {V = V} hV .runV l χ G = inj₂ (chV χ V ,ˢ ≡→≈v (ch-inr χ V) ,ˢ hV .runV l χ G)

compat-pair : ∀ {Θ A A'} {V : Val Θ A} {V' : Val Θ A'} → ⊨V V → ⊨V V' → ⊨V (pair V V')
compat-pair {V = V} {V'} hV hV' .runV l χ G =
  chV χ V ,ˢ chV χ V' ,ˢ ≡→≈v (ch-pair χ V V') ,ˢ hV .runV l χ G ,ˢ hV' .runV l χ G

compat-thunk : ∀ {Θ B} {M : Comp Θ B} → ⊨C M → ⊨V (thunk M)
compat-thunk {B = B} {M} hM .runV l χ G {Σ'} l' χ' =
  subst (λ X → 𝒞⟦ B ⟧ Σ' (force X)) (sym (ch-thunk (χ ++ᶜ χ') M) ∙ ch-++V χ χ' (thunk M))
    (𝒞-conv B (U-β _) (hM .runC l' (χ ++ᶜ χ') (mono𝒢 χ G χ')))

compat-const : ∀ {Θ b} (c : Cst b (world Θ)) → ⊨V (const {Θ} c)
compat-const c .runV l χ G = ⋆

compat-shut : ∀ {Θ κ A} {V : (p : Pos κ (world Θ)) → Val (Θ ⧀⟨ κ , p ⟩) A} →
  (∀ p → ⊨V (V p)) → ⊨V (shut V)
compat-shut {κ = κ} {A} {V} hV .runV {Σ} l χ G q =
  subst (𝒱⟦ A ⟧ (Σ ⧀⟨ κ , q ⟩)) (cong (λ X → openV X q) (sym (ch-shut χ V)))
    (hV (posᶜ κ χ q) .runV (lt⧀ l) (lockᶜ κ χ q) (𝒢-lock κ χ q G))

compat-return : ∀ {Θ A} {V : Val Θ A} → ⊨V V → ⊨C (return V)
compat-return {A = A} {V} hV .runC {Σ} l χ G =
  subst (FPred 𝒱⟦ A ⟧ Σ) (sym (ch-return χ V)) (ret (hV .runV l χ G))

compat-to : ∀ {Θ A B} {M : Comp Θ (F A)} {N : Comp (Θ , A) B} → ⊨C M → ⊨C N → ⊨C (M to N)
compat-to {B = B} {M} {N} hM hN .runC {Σ} l χ G =
  subst (𝒞⟦ B ⟧ Σ) (sym (ch-to χ M N))
    (seq𝒞 l (hM .runC l χ G) λ {Σ'} l' χ' V v →
      subst (𝒞⟦ B ⟧ Σ') (ch-sub₁ χ χ' N V)
        (hN .runC l' (liftᶜ (χ ++ᶜ χ') ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ) (𝒢-β (χ ++ᶜ χ') (mono𝒢 χ G χ') v)))

compat-force : ∀ {Θ B} {V : Val Θ (U B)} → ⊨V V → ⊨C (force V)
compat-force {B = B} {V} hV .runC {Σ} l χ G =
  subst (𝒞⟦ B ⟧ Σ) (sym (ch-force χ V)) (hV .runV l χ G l []ᶜ)

compat-case : ∀ {Θ A A' B} {V : Val Θ (A × A')} {M : Comp (Θ , A , A') B} →
  ⊨V V → ⊨C M → ⊨C (case V M)
compat-case {B = B} {V} {M} hV hM .runC {Σ} l χ G =
  subst (𝒞⟦ B ⟧ Σ) (sym (ch-case χ V M))
    (case𝒞 (chV χ V) (hV .runV l χ G) λ V₁ V₂ v₁ v₂ →
      subst (𝒞⟦ B ⟧ Σ) (ch-++C (liftᶜ (liftᶜ χ)) _ M)
        (hM .runC l (liftᶜ (liftᶜ χ) ++ᶜ (idS ,ₛ V₁ ,ₛ V₂) ∷ᶜ []ᶜ) (𝒢-β₂ χ G v₁ v₂)))

compat-case₊ : ∀ {Θ A A' B} {V : Val Θ (A ⊕ A')} {M : Comp (Θ , A) B} {N : Comp (Θ , A') B} →
  ⊨V V → ⊨C M → ⊨C N → ⊨C (case₊ V M N)
compat-case₊ {B = B} {V} {M} {N} hV hM hN .runC {Σ} l χ G =
  subst (𝒞⟦ B ⟧ Σ) (sym (ch-case₊ χ V M N))
    (case₊𝒞 (chV χ V) (hV .runV l χ G)
      (λ V₁ v₁ → subst (𝒞⟦ B ⟧ Σ) (ch-++C (liftᶜ χ) _ M)
        (hM .runC l (liftᶜ χ ++ᶜ (idS ,ₛ V₁) ∷ᶜ []ᶜ) (𝒢-β χ G v₁)))
      (λ V₂ v₂ → subst (𝒞⟦ B ⟧ Σ) (ch-++C (liftᶜ χ) _ N)
        (hN .runC l (liftᶜ χ ++ᶜ (idS ,ₛ V₂) ∷ᶜ []ᶜ) (𝒢-β χ G v₂))))

compat-absurd : ∀ {Θ B} {V : Val Θ 𝟘} → ⊨V V → ⊨C (absurd {B = B} V)
compat-absurd hV .runC l χ G = ⊥.rec (hV .runV l χ G)

compat-lam : ∀ {Θ A B} {M : Comp (Θ , A) B} → ⊨C M → ⊨C (lam M)
compat-lam {A = A} {B} {M} hM .runC {Σ} l χ G =
  subst (𝒞⟦ A ⇒ B ⟧ Σ) (sym (ch-lam χ M)) λ {Σ'} l' χ' V v →
    subst (λ X → 𝒞⟦ B ⟧ Σ' (app X V)) (sym (ch-lam χ' (chC (liftᶜ χ) M)))
      (𝒞-conv B (⇒-β _ V)
        (subst (𝒞⟦ B ⟧ Σ') (ch-sub₁ χ χ' M V)
          (hM .runC l' (liftᶜ (χ ++ᶜ χ') ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ) (𝒢-β (χ ++ᶜ χ') (mono𝒢 χ G χ') v))))

compat-app : ∀ {Θ A B} {M : Comp Θ (A ⇒ B)} {V : Val Θ A} → ⊨C M → ⊨V V → ⊨C (app M V)
compat-app {B = B} {M} {V} hM hV .runC {Σ} l χ G =
  subst (𝒞⟦ B ⟧ Σ) (sym (ch-app χ M V)) (hM .runC l χ G l []ᶜ (chV χ V) (hV .runV l χ G))

compat-op : ∀ {Θ B} (o : Op) (vs : Vals Θ (params o))
  {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B} →
  (∀ a ps → ⊨C (k a ps)) → ⊨C (op o vs k)
compat-op {B = B} o vs {k} hk .runC {Σ} l χ G =
  subst (𝒞⟦ B ⟧ Σ) (sym (ch-op χ o vs k))
    (𝒞-op B o λ a ps →
      hk a (wposᶜ (slot o a) χ ps) .runC
        (IsLT-⧀* l (slot o a) ps) (wordᶜ (slot o a) χ ps) (𝒢-word (slot o a) χ ps G))

----------------------------------------------------------------------
-- the fundamental theorem

semV : ∀ {Θ A} (V : Val Θ A) → ⊨V V
semC : ∀ {Θ B} (M : Comp Θ B) → ⊨C M

semV (ne n) = compat-ne n
semV tt = compat-tt
semV (inl V) = compat-inl (semV V)
semV (inr V) = compat-inr (semV V)
semV (pair V V') = compat-pair (semV V) (semV V')
semV (thunk M) = compat-thunk (semC M)
semV (const c) = compat-const c
semV (shut V) = compat-shut λ p → semV (V p)

semC (return V) = compat-return (semV V)
semC (M to N) = compat-to (semC M) (semC N)
semC (force V) = compat-force (semV V)
semC (case V M) = compat-case (semV V) (semC M)
semC (case₊ V M N) = compat-case₊ (semV V) (semC M) (semC N)
semC (absurd V) = compat-absurd (semV V)
semC (lam M) = compat-lam (semC M)
semC (app M V) = compat-app (semC M) (semV V)
semC (op o vs k) = compat-op o vs λ a ps → semC (k a ps)

fundV : ∀ {Θ Σ A} (V : Val Θ A) → IsLT Σ → (χ : Chain Θ Σ) → 𝒢 χ → 𝒱⟦ A ⟧ Σ (chV χ V)
fundV V = semV V .runV

fundC : ∀ {Θ Σ B} (M : Comp Θ B) → IsLT Σ → (χ : Chain Θ Σ) → 𝒢 χ → 𝒞⟦ B ⟧ Σ (chC χ M)
fundC M = semC M .runC

-- equational canonicity

-- every computation of a lock telescope is related
related : ∀ {Θ B} → IsLT Θ → (M : Comp Θ B) → 𝒞⟦ B ⟧ Θ M
related l M = fundC M l []ᶜ (𝒢-LT l)

-- a derivation of 𝒞⟦ F A ⟧ is a canonical form up to ≈
canonical-form : ∀ {A Θ} {R : (Θ : Tele) → Val Θ A → Type} {M : Comp Θ (F A)} → FPred R Θ M →
  Σ[ t ∈ OpTree A Θ ] (M ≈c reify t)
canonical-form (ret {V = V} v) = leaf V ,ˢ ≈refl
canonical-form (node o {vs} ts) =
  node o vs (λ a ps → canonical-form (ts a ps) .fst) ,ˢ
  op≈ (≈vs-refl vs) (λ a ps → canonical-form (ts a ps) .snd)
canonical-form (conv p t) = canonical-form t .fst ,ˢ ≈trans p (canonical-form t .snd)

equational-canonicity : ∀ {Θ A} → IsLT Θ → (M : Comp Θ (F A)) → Σ[ t ∈ OpTree A Θ ] (M ≈c reify t)
equational-canonicity l M = canonical-form (related l M)
