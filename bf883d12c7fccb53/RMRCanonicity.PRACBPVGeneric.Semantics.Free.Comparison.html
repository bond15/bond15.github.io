-- Relating the two denotations (parity stage G, prerequisite).
--
--   Semantics.Denotation        ⟦ F A ⟧ = the term algebra T_P ⟦ A ⟧
--                               (tree soundness, closing, the machine)
--   Semantics.Free.Denotation   ⟦ F A ⟧ = the free model T ⟦ A ⟧ of the
--                               theory (≈ soundness, EqCanonicity)
--
-- The two interpret U B, and so A ⇒ B and every type above them,
-- differently, and A ⇒ B is contravariant in A: there is no map between
-- the interpretations at every type.  The standard tool is a binary
-- Kripke logical relation between them, one relation per type and per
-- telescope, closed under the presheaf actions (mono):
--
--   Rv 𝟘 = ⊥     Rv 𝟙 = ⊤     Rv (A ⊕ A') = inl ~ inl, inr ~ inr
--   Rv (A × A')  componentwise        Rv ([ κ ] A)  at every position
--   Rv (base b)  equality             Rv (U B) = Rc B
--   Rc (F A)     Lifted (Rv A): t ~ c when c is t evaluated in the free
--                model with related leaves (leaf~ : var a ~ η a',
--                node~ : ops o ts ~ the node of the cs, for ts ~ cs)
--   Rc (A ⇒ B)   related arguments, at any later world, to related results
--   RΘ (∅ w)     equality of world maps        RΘ (Θ , A)  componentwise
--   RΘ (Θ ⧀⟨ κ , p ⟩)  the same map k into the world, and points of the two
--                locks with representatives ⟪ v , q , g , e ⟫, ⟪ v , q , g , e' ⟫
--                for related environments e, e' (LRel)
--
-- Every Rc B is closed under the operations (opsC: the algebras of the two
-- interpretations agree on related continuations), and
--
--   FUNDAMENTAL LEMMA (fundNe, fundV, fundVs, fundC).  For related
--   environments, the two denotations of every term are related.
--
-- At a first-order result type (FirstOrder: 𝟘, 𝟙, ⊕, ×, [ κ ], base; no U)
-- the relation is a function, same : ⟦ A ⟧ → ⟦ A ⟧' (same-rel), and Lifted
-- is the graph of the CANONICAL ALGEBRA MAP canon : T_P ⟦ A ⟧ → T ⟦ A ⟧, the
-- unique homomorphism extending η ∘ same (evaluation of terms in the free
-- model; lift-canon).  So
--
--   comparison    ⟦ M ⟧ ; canon = ⟦ M ⟧'    for M : Comp (∅ w) (F A)
--   comparison₀   at the identity: canon (denClosed M) = denClosed' M
--   machine-comparison
--                 the machine's map into the free model on ⟦ A ⟧
--                 (Machine.Run.h: denClosed, then evaluation at the unit),
--                 followed by T (same), is the free-model denotation
--
-- and related-closed states the lemma for closed computations of every
-- type.  Tree soundness needs no transfer: Semantics.Free.Soundness proves
-- it for the free-model denotation directly.

------------------------------------------------------------------------
-- relations between two presheaves, at every world

Rel : 𝒱 → 𝒱 → Type₁
Rel X X' = (w : W.ob) → El X w → El X' w → Type

-- a relation lifted to the terms over X and the free model on X': a term
-- is related to its evaluation in the free model at related leaves
data Lifted {X X' : 𝒱} (R : Rel X X') :
  (w : W.ob) → Term (PM.polynomial L O) X w → El Fr.∣ Fr.T X' ∣ w → Type where
  leaf~ : ∀ {w a a' c} → R w a a' → c ≡ Fr.η X' .N-ob w a' → Lifted R w (var a) c
  node~ : ∀ {w o ts c} (cs : (p : SN.Position w o) → El Fr.∣ Fr.T X' ∣ (SN.scope w o p)) →
    (∀ p → Lifted R _ (ts p) (cs p)) →
    c ≡ Algebra.algebra (Fr.T X' .fst) .NatTrans.N-ob w (o , cs) → Lifted R w (ops o ts) c

------------------------------------------------------------------------
-- the relation at each type

Rv : (A : ValTy) → Rel Tm.⟦ A ⟧v Fr.⟦ A ⟧v
Rc : (B : CompTy) → Rel Tm.⟦ B ⟧u Fr.⟦ B ⟧u

Rv 𝟘 w a a' = ⊥
Rv 𝟙 w a a' = Unit
Rv (A ⊕ A') w (inl a) (inl a') = Rv A w a a'
Rv (A ⊕ A') w (inl a) (inr a') = ⊥
Rv (A ⊕ A') w (inr a) (inl a') = ⊥
Rv (A ⊕ A') w (inr a) (inr a') = Rv A' w a a'
Rv (A Syn.× A') w a a' = Rv A w (a .fst) (a' .fst) × Rv A' w (a .snd) (a' .snd)
Rv (U B) = Rc B
Rv ([ κ ] A) w r r' = ∀ q → Rv A _ (r q) (r' q)
Rv (base b) w c c' = c ≡ c'

Rc (F A) = Lifted (Rv A)
Rc (A ⇒ B) w e e' = ∀ v (h : W [ w , v ]) a a' → Rv A v a a' →
  Rc B v (e .N-ob v (h , a)) (e' .N-ob v (h , a'))

------------------------------------------------------------------------
-- monotonicity: the relations are closed under the presheaf actions

monoV : ∀ A {w v} (f : W [ w , v ]) {a a'} → Rv A w a a' →
  Rv A v (Tm.⟦ A ⟧v .F-hom f a) (Fr.⟦ A ⟧v .F-hom f a')
monoC : ∀ B {w v} (f : W [ w , v ]) {b b'} → Rc B w b b' →
  Rc B v (Tm.⟦ B ⟧u .F-hom f b) (Fr.⟦ B ⟧u .F-hom f b')
monoLift : ∀ A {w v} (f : W [ w , v ]) {t c} → Lifted (Rv A) w t c →
  Lifted (Rv A) v (Tm.⟦ F A ⟧u .F-hom f t) (Fr.⟦ F A ⟧u .F-hom f c)

monoV 𝟘 f r = ⊥.rec r
monoV 𝟙 f r = tt
monoV (A ⊕ A') f {inl a} {inl a'} r = monoV A f r
monoV (A ⊕ A') f {inl a} {inr a'} r = ⊥.rec r
monoV (A ⊕ A') f {inr a} {inl a'} r = ⊥.rec r
monoV (A ⊕ A') f {inr a} {inr a'} r = monoV A' f r
monoV (A Syn.× A') f r = monoV A f (r .fst) , monoV A' f (r .snd)
monoV (U B) f r = monoC B f r
monoV ([ κ ] A) f r q = monoV A (K.smap κ f q) (r (K.pos κ f q))
monoV (base b) f r = cong (Cst b .F-hom f) r

monoC (F A) f r = monoLift A f r
monoC (A ⇒ B) f r v h a a' ra = r v (f W.⋆ h) a a' ra

monoLift A f (leaf~ r p) =
  leaf~ (monoV A f r) (cong (Fr.⟦ F A ⟧u .F-hom f) p ∙ sym (Fr.η Fr.⟦ A ⟧v .N-hom _ _ f _))
monoLift A f (node~ {o = o} cs rs p) =
  node~ (λ q → Fr.⟦ F A ⟧u .F-hom (smapP f o q) (cs (tpos f o q)))
    (λ q → monoLift A (smapP f o q) (rs (tpos f o q)))
    (cong (Fr.⟦ F A ⟧u .F-hom f) p
     ∙ sym (funExt⁻ (Algebra.algebra Fr.⟦ F A ⟧c .NatTrans.N-hom f) (o , cs)))

------------------------------------------------------------------------
-- the relations are closed under the operations

opsC : ∀ B w (o : SN.Shape w)
  (ks : (p : SN.Position w o) → El Tm.⟦ B ⟧u (SN.scope w o p))
  (ks' : (p : SN.Position w o) → El Fr.⟦ B ⟧u (SN.scope w o p)) →
  (∀ p → Rc B _ (ks p) (ks' p)) →
  Rc B w (Algebra.algebra Tm.⟦ B ⟧c .NatTrans.N-ob w (o , ks))
         (Algebra.algebra Fr.⟦ B ⟧c .NatTrans.N-ob w (o , ks'))
opsC (F A) w o ks ks' rs = node~ ks' rs refl
-- the exponential's operation runs B's, the argument moved along ι
opsC (A ⇒ B) w o ks ks' rs v h a a' ra =
  subst2 (Rc B v)
    (sym (Tm.expAlgebra-ob Tm.⟦ A ⟧v Tm.⟦ B ⟧c w (o , ks) v h a))
    (sym (Fr.expAlgebra-ob Fr.⟦ A ⟧v Fr.⟦ B ⟧c w (o , ks') v h a'))
    (opsC B v _ _ _ λ p →
      rs (tpos h o p) _ (smapP h o p W.⋆ W.id) _ _ (monoV A (ι v (Poly.Shapes (PM.polynomial L O) .F-hom h o) p) ra))

------------------------------------------------------------------------
-- the relation at telescopes

-- a relation lifted to the lock of κ: points with representatives
-- ⟪ v , q , g , e ⟫ and ⟪ v , q , g , e' ⟫ for related e, e'
record LRel (κ : Lk) {Γ Γ' : 𝒱} (R : Rel Γ Γ') (t : W.ob)
  (x : El (K.L κ Γ) t) (x' : El (K.L κ Γ') t) : Type where
  constructor lrel
  field
    v : W.ob
    q : Pos κ v
    g : W [ scope κ v q , t ]
    e : El Γ v
    e' : El Γ' v
    rel : R v e e'
    at : x ≡ K.pt κ Γ (v , q , g , e)
    at' : x' ≡ K.pt κ Γ' (v , q , g , e')

RΘ : (Θ : Tele) → Rel Tm.⟦ Θ ⟧t Fr.⟦ Θ ⟧t
RΘ (∅ w) v h h' = h ≡ h'
RΘ (Θ ▸ A) v γ γ' = RΘ Θ v (γ .fst) (γ' .fst) × Rv A v (γ .snd) (γ' .snd)
RΘ (Θ ⧀⟨ κ , p ⟩) t s s' = (s .snd .fst ≡ s' .snd .fst) × LRel κ (RΘ Θ) t (s .fst) (s' .fst)

-- related environments lie over the same world map
over : ∀ Θ {v γ γ'} → RΘ Θ v γ γ' → Tm.πt Θ .N-ob v γ ≡ Fr.πt Θ .N-ob v γ'
over (∅ w) r = r
over (Θ ▸ A) r = over Θ (r .fst)
over (Θ ⧀⟨ κ , p ⟩) r = r .fst

monoΘ : ∀ Θ {v u} (f : W [ v , u ]) {γ γ'} → RΘ Θ v γ γ' →
  RΘ Θ u (Tm.⟦ Θ ⟧t .F-hom f γ) (Fr.⟦ Θ ⟧t .F-hom f γ')
monoΘ (∅ w) f r = cong (W._⋆ f) r
monoΘ (Θ ▸ A) f r = monoΘ Θ f (r .fst) , monoV A f (r .snd)
monoΘ (Θ ⧀⟨ κ , p ⟩) f (k , lrel v q g e e' r at at') =
  cong (W._⋆ f) k ,
  lrel v q (g W.⋆ f) e e' r (cong (K.L κ _ .F-hom f) at) (cong (K.L κ _ .F-hom f) at')

------------------------------------------------------------------------
-- the maps out of a lock and into R_κ preserve the relation

-- two unlocks of maps into R_κ sending related environments to
-- position-wise related elements, at related points of the lock
unlock-rel : ∀ κ {Γ Γ' X X' : 𝒱} (R : Rel Γ Γ') (S : Rel X X')
  (φ : PshHom Γ (K.R κ X)) (φ' : PshHom Γ' (K.R κ X')) →
  (∀ {v u} (f : W [ v , u ]) {x x'} → S v x x' → S u (X .F-hom f x) (X' .F-hom f x')) →
  (∀ {v e e'} → R v e e' → ∀ q → S _ (φ .N-ob v e q) (φ' .N-ob v e' q)) →
  ∀ {t x x'} → LRel κ R t x x' → S t (K.unlock κ φ .N-ob t x) (K.unlock κ φ' .N-ob t x')
unlock-rel κ R S φ φ' monoS hφ (lrel v q g e e' r at at') =
  subst2 (S _) (sym (cong (K.unlock κ φ .N-ob _) at)) (sym (cong (K.unlock κ φ' .N-ob _) at'))
    (monoS g (hφ r q))

-- R_κ's introduction rule on the two sides: related at every position.
-- The free side's point is moved into the term side's summand (copair-at),
-- as the two telescopes lie over the same world map (over)
shut-rel : ∀ {Θ} κ {X X' : 𝒱} (S : Rel X X')
  (φ : (p : Pos κ (world Θ)) → PshHom Tm.⟦ Θ ⧀⟨ κ , p ⟩ ⟧t X)
  (φ' : (p : Pos κ (world Θ)) → PshHom Fr.⟦ Θ ⧀⟨ κ , p ⟩ ⟧t X') →
  (∀ p {t s s'} → RΘ (Θ ⧀⟨ κ , p ⟩) t s s' → S t (φ p .N-ob t s) (φ' p .N-ob t s')) →
  ∀ {v γ γ'} → RΘ Θ v γ γ' → ∀ q →
  S _ (K.Over.shut κ (Tm.πt Θ) φ .N-ob v γ q) (K.Over.shut κ (Fr.πt Θ) φ' .N-ob v γ' q)
shut-rel {Θ} κ S φ φ' hφ {v} {γ} {γ'} r q =
  subst (S _ _) (sym moved) (hφ p₁ (refl , lrel v q W.id γ γ' r refl refl))
  where
  p₁ = K.pos κ (Tm.πt Θ .N-ob v γ) q
  k₁ = K.smap κ (Tm.πt Θ .N-ob v γ) q

  x' : El (K.L κ Fr.⟦ Θ ⟧t) (scope κ v q)
  x' = K.pt κ Fr.⟦ Θ ⟧t (v , q , W.id , γ')

  -- the free side's point lies in the term side's summand
  r₁ : K.Over.Lπ κ (Fr.πt Θ) .N-ob _ x' ≡ (p₁ , k₁)
  r₁ = cong (λ h → K.pos κ h q , K.smap κ h q W.⋆ W.id) (sym (over Θ r))
       ∙ cong (p₁ ,_) (W.⋆IdR k₁)

  moved : K.Over.shut κ (Fr.πt Θ) φ' .N-ob v γ' q ≡ φ' p₁ .N-ob _ (x' , k₁ , r₁)
  moved = sym (K.Over.copair-at κ (Fr.πt Θ) φ' _ _ _) ∙ K.Over.copair-at κ (Fr.πt Θ) φ' x' _ r₁

-- iterated along a lock word
shutW-rel : ∀ {Θ} (μ : List Lk) {X X' : 𝒱} (S : Rel X X')
  (k : (ps : WPos μ (world Θ)) → PshHom Tm.⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X)
  (k' : (ps : WPos μ (world Θ)) → PshHom Fr.⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X') →
  (∀ ps {t s s'} → RΘ (Θ ⧀*⟨ μ , ps ⟩) t s s' → S t (k ps .N-ob t s) (k' ps .N-ob t s')) →
  ∀ {v γ γ'} → RΘ Θ v γ γ' → ∀ ps →
  S _ (Tm.shutW μ k .N-ob v γ ps) (Fr.shutW μ k' .N-ob v γ' ps)
shutW-rel [] S k k' hk r ps = hk ps r
shutW-rel (κ ∷ []) S k k' hk r p = shut-rel κ S k k' hk r p
shutW-rel {Θ} (κ ∷ μ@(_ ∷ _)) S k k' hk r (p , ps) =
  shut-rel κ (λ t y y' → ∀ ps → S _ (y ps) (y' ps))
    (λ p → Tm.shutW μ (λ ps → k (p , ps))) (λ p → Fr.shutW μ (λ ps → k' (p , ps)))
    (λ p r' → shutW-rel μ S (λ ps → k (p , ps)) (λ ps → k' (p , ps)) (λ ps → hk (p , ps)) r') r p ps

-- sequencing and case analysis preserve the relation

-- the two binds, on related computations and environments
bind-rel : ∀ {Θ A B} (N : Comp (Θ ▸ A) B) →
  (∀ {v γ γ' a a'} → RΘ Θ v γ γ' → Rv A v a a' →
     Rc B v (Tm.⟦ N ⟧C .N-ob v (γ , a)) (Fr.⟦ N ⟧C .N-ob v (γ' , a'))) →
  ∀ {v t c} → Lifted (Rv A) v t c → ∀ {γ γ'} → RΘ Θ v γ γ' →
  Rc B v (Tm.bind Tm.⟦ Θ ⟧t Tm.⟦ A ⟧v Tm.⟦ B ⟧c Tm.⟦ N ⟧C .N-ob v (t , γ))
         (Fr.bind Fr.⟦ Θ ⟧t Fr.⟦ A ⟧v Fr.⟦ B ⟧m Fr.⟦ N ⟧C .N-ob v (c , γ'))
bind-rel {Θ} {A} {B} N hN {v} (leaf~ {a = a} {a'} ra p) {γ} {γ'} rγ =
  subst2 (Rc B v)
    (sym (Tm.bind-η Tm.⟦ Θ ⟧t Tm.⟦ A ⟧v Tm.⟦ B ⟧c Tm.⟦ N ⟧C v a γ))
    (sym (cong (λ c → Fr.bind Fr.⟦ Θ ⟧t Fr.⟦ A ⟧v Fr.⟦ B ⟧m Fr.⟦ N ⟧C .N-ob v (c , γ')) p
          ∙ Fr.bind-η Fr.⟦ Θ ⟧t Fr.⟦ A ⟧v Fr.⟦ B ⟧m Fr.⟦ N ⟧C v a' γ'))
    (hN rγ ra)
bind-rel {Θ} {A} {B} N hN {v} (node~ {o = o} {ts} cs rs p) {γ} {γ'} rγ =
  subst2 (Rc B v)
    (sym (Tm.bind-op Tm.⟦ Θ ⟧t Tm.⟦ A ⟧v Tm.⟦ B ⟧c Tm.⟦ N ⟧C v o ts γ))
    (sym (cong (λ c → Fr.bind Fr.⟦ Θ ⟧t Fr.⟦ A ⟧v Fr.⟦ B ⟧m Fr.⟦ N ⟧C .N-ob v (c , γ')) p
          ∙ Fr.bind-op Fr.⟦ Θ ⟧t Fr.⟦ A ⟧v Fr.⟦ B ⟧m Fr.⟦ N ⟧C v o cs γ'))
    (opsC B v o _ _ λ q → bind-rel N hN (rs q) (monoΘ Θ (ι v o q) rγ))

-- the two copairings, on related injections
case₊-rel : ∀ {Θ A A' B} (M : Comp (Θ ▸ A) B) (N : Comp (Θ ▸ A') B) →
  (∀ {v γ γ' a a'} → RΘ Θ v γ γ' → Rv A v a a' →
     Rc B v (Tm.⟦ M ⟧C .N-ob v (γ , a)) (Fr.⟦ M ⟧C .N-ob v (γ' , a'))) →
  (∀ {v γ γ' a a'} → RΘ Θ v γ γ' → Rv A' v a a' →
     Rc B v (Tm.⟦ N ⟧C .N-ob v (γ , a)) (Fr.⟦ N ⟧C .N-ob v (γ' , a'))) →
  ∀ {v γ γ'} → RΘ Θ v γ γ' → ∀ s s' → Rv (A ⊕ A') v s s' →
  Rc B v (Fr.casePsh Tm.⟦ M ⟧C Tm.⟦ N ⟧C .N-ob v (γ , s))
         (Fr.casePsh Fr.⟦ M ⟧C Fr.⟦ N ⟧C .N-ob v (γ' , s'))
case₊-rel M N hM hN r (inl a) (inl a') ra = hM r ra
case₊-rel M N hM hN r (inl a) (inr a') ()
case₊-rel M N hM hN r (inr a) (inl a') ()
case₊-rel M N hM hN r (inr a) (inr a') ra = hN r ra

------------------------------------------------------------------------
-- the fundamental lemma

fundNe : ∀ {Θ A} (n : Ne Θ A) {v γ γ'} → RΘ Θ v γ γ' →
  Rv A v (Tm.⟦ n ⟧ne .N-ob v γ) (Fr.⟦ n ⟧ne .N-ob v γ')
fundNe here r = r .snd
fundNe (wkv n) r = fundNe n (r .fst)
fundNe (up {Δ} {κ} {p} {A} n) r =
  unlock-rel κ (RΘ Δ) (Rv A)
    (Tm.⟦ n ⟧ne ⋆PshHom K.box κ Tm.⟦ A ⟧v) (Fr.⟦ n ⟧ne ⋆PshHom K.box κ Fr.⟦ A ⟧v)
    (monoV A) (λ r' q → monoV A (K.ι κ _ q) (fundNe n r')) (r .snd)
fundNe (opn {Δ} {κ} {p} {A} n) r =
  unlock-rel κ (RΘ Δ) (Rv A) Tm.⟦ n ⟧ne Fr.⟦ n ⟧ne (monoV A) (λ r' q → fundNe n r' q) (r .snd)
fundNe (nm {Δ} {κ} {p} {b} k) r =
  unlock-rel κ (RΘ Δ) (Rv (base b)) (Fr.namesR κ k) (Fr.namesR κ k) (monoV (base b))
    (λ r' q → refl) (r .snd)

fundV : ∀ {Θ A} (V : Val Θ A) {v γ γ'} → RΘ Θ v γ γ' →
  Rv A v (Tm.⟦ V ⟧V .N-ob v γ) (Fr.⟦ V ⟧V .N-ob v γ')
fundVs : ∀ {Θ bs} (vs : Vals Θ bs) {v γ γ'} → RΘ Θ v γ γ' →
  Tm.⟦ vs ⟧Vs .N-ob v γ ≡ Fr.⟦ vs ⟧Vs .N-ob v γ'
fundC : ∀ {Θ B} (M : Comp Θ B) {v γ γ'} → RΘ Θ v γ γ' →
  Rc B v (Tm.⟦ M ⟧C .N-ob v γ) (Fr.⟦ M ⟧C .N-ob v γ')

fundV (Syn.ne n) r = fundNe n r
fundV Syn.tt r = tt
fundV (Syn.inl V) r = fundV V r
fundV (Syn.inr V) r = fundV V r
fundV (Syn.pair V V') r = fundV V r , fundV V' r
fundV (Syn.thunk M) r = fundC M r
fundV {Θ} (Syn.const {b} c) r = cong (λ h → Cst b .F-hom h c) (over Θ r)
fundV {Θ} (Syn.shut {κ} {A} V) r = shut-rel κ (Rv A) (λ p → Tm.⟦ V p ⟧V) (λ p → Fr.⟦ V p ⟧V)
  (λ p s → fundV (V p) s) r

fundVs Syn.[]ᵥ r = refl
fundVs (V Syn.∷ᵥ vs) r = ΣPathP (fundV V r , fundVs vs r)

fundC (Syn.return V) r = leaf~ (fundV V r) refl
fundC (Syn._to_ M N) r = bind-rel N (λ rγ ra → fundC N (rγ , ra)) (fundC M r) r
fundC (Syn.force V) r = fundV V r
fundC (Syn.case V M) r = fundC M ((r , fundV V r .fst) , fundV V r .snd)
fundC {Θ} {B} (Syn.case₊ V M N) {v} {γ} {γ'} r =
  case₊-rel M N (λ rγ ra → fundC M (rγ , ra)) (λ rγ ra → fundC N (rγ , ra)) r
    (Tm.⟦ V ⟧V .N-ob v γ) (Fr.⟦ V ⟧V .N-ob v γ') (fundV V r)
fundC (Syn.absurd V) {v} {γ} r = ⊥.rec (Tm.⟦ V ⟧V .N-ob v γ)
fundC {Θ} (Syn.lam M) r u h a a' ra = fundC M (monoΘ Θ h r , ra)
fundC (Syn.app M V) r = fundC M r _ W.id _ _ (fundV V r)
-- the parameters agree, the continuations are related through their words
fundC (Syn.op {B} o vs k) {v} {γ} {γ'} r =
  subst (Rc B v _)
    (cong (λ cs → Algebra.algebra Fr.⟦ B ⟧c .NatTrans.N-ob v ((o , cs) , ks')) (fundVs vs r))
    (opsC B v (o , Tm.⟦ vs ⟧Vs .N-ob v γ) _ ks' λ (a , ps) →
      shutW-rel (slot o a) (Rc B) (λ ps → Tm.⟦ k a ps ⟧C) (λ ps → Fr.⟦ k a ps ⟧C)
        (λ ps s → fundC (k a ps) s) r ps)
  where
  ks' : (p : Σ (Syn.Ar o) λ a → WPos (slot o a) v) → El Fr.⟦ B ⟧u (Syn.wscope (slot o (p .fst)) v (p .snd))
  ks' (a , ps) = Fr.shutW (slot o a) (λ ps → Fr.⟦ k a ps ⟧C) .N-ob v γ' ps

-- closed computations, at every type
related-closed : ∀ {w B} (M : Comp (∅ w) B) →
  Rc B w (Tm.⟦ M ⟧C .N-ob w W.id) (Fr.⟦ M ⟧C .N-ob w W.id)
related-closed M = fundC M refl

------------------------------------------------------------------------
-- first-order result types: the relation is a function

data FirstOrder : ValTy → Type where
  fo𝟘 : FirstOrder 𝟘
  fo𝟙 : FirstOrder 𝟙
  _fo⊕_ : ∀ {A A'} → FirstOrder A → FirstOrder A' → FirstOrder (A ⊕ A')
  _fo×_ : ∀ {A A'} → FirstOrder A → FirstOrder A' → FirstOrder (A Syn.× A')
  fo[_] : ∀ κ {A} → FirstOrder A → FirstOrder ([ κ ] A)
  fobase : ∀ b → FirstOrder (base b)

-- the coproduct of two maps
_⊎PshHom_ : ∀ {X Y X' Y' : 𝒱} → PshHom X X' → PshHom Y Y' →
  PshHom (X Fr.⊎Psh Y) (X' Fr.⊎Psh Y')
(α ⊎PshHom β) .N-ob v = ⊎.map (α .N-ob v) (β .N-ob v)
(α ⊎PshHom β) .N-hom u v f (inl x) = cong inl (α .N-hom u v f x)
(α ⊎PshHom β) .N-hom u v f (inr y) = cong inr (β .N-hom u v f y)

-- at a first-order type the two interpretations agree (U does not occur)
same : ∀ {A} → FirstOrder A → PshHom Tm.⟦ A ⟧v Fr.⟦ A ⟧v
same fo𝟘 = EmptyPsh-rec _
same fo𝟙 = idPshHom
same (f fo⊕ f') = same f ⊎PshHom same f'
same (f fo× f') = same f ×PshHom same f'
same (fo[ κ ] f) = K.Rmap κ (same f)
same (fobase b) = idPshHom

same-rel : ∀ {A} (fo : FirstOrder A) {w a a'} → Rv A w a a' → same fo .N-ob w a ≡ a'
same-rel fo𝟘 r = ⊥.rec r
same-rel fo𝟙 r = refl
same-rel (f fo⊕ f') {a = inl a} {inl a'} r = cong inl (same-rel f r)
same-rel (f fo⊕ f') {a = inl a} {inr a'} ()
same-rel (f fo⊕ f') {a = inr a} {inl a'} ()
same-rel (f fo⊕ f') {a = inr a} {inr a'} r = cong inr (same-rel f' r)
same-rel (f fo× f') r = ΣPathP (same-rel f (r .fst) , same-rel f' (r .snd))
same-rel (fo[ κ ] f) r = funExt λ q → same-rel f (r q)
same-rel (fobase b) r = r

-- the canonical algebra map from the term algebra on ⟦ A ⟧ to the free
-- model on ⟦ A ⟧: the unique homomorphism extending η ∘ same (evaluation
-- of terms in the free model)
canon : ∀ {A} → FirstOrder A → Hom (PM.polynomial L O) Tm.⟦ F A ⟧c Fr.⟦ F A ⟧c
canon {A} fo = evaluateHom Fr.⟦ F A ⟧c (PshHom→NatTrans (same fo ⋆PshHom Fr.η Fr.⟦ A ⟧v))

-- Lifted is canon's graph
lift-canon : ∀ {A} (fo : FirstOrder A) {w t c} → Lifted (Rv A) w t c →
  canon fo .Hom.transformation .NatTrans.N-ob w t ≡ c
lift-canon {A} fo (leaf~ r p) = cong (Fr.η Fr.⟦ A ⟧v .N-ob _) (same-rel fo r) ∙ sym p
lift-canon {A} fo {w} (node~ {o = o} cs rs p) =
  cong (λ c → Algebra.algebra Fr.⟦ F A ⟧c .NatTrans.N-ob w (o , c)) (funExt λ q → lift-canon fo (rs q))
  ∙ sym p

------------------------------------------------------------------------
-- the comparison theorem

-- the term-algebra denotation, followed by the canonical map, is the
-- free-model denotation
comparison : ∀ {A w} (fo : FirstOrder A) (M : Comp (∅ w) (F A)) →
  Tm.⟦ M ⟧C ⋆PshHom NatTrans→PshHom (canon fo .Hom.transformation) ≡ Fr.⟦ M ⟧C
comparison fo M = makePshHomPath (funExt λ v → funExt λ h → lift-canon fo (fundC M refl))

-- at the identity: closed denotations
comparison₀ : ∀ {A w} (fo : FirstOrder A) (M : Comp (∅ w) (F A)) →
  canon fo .Hom.transformation .NatTrans.N-ob w (Tm.⟦ M ⟧C .N-ob w W.id) ≡ Fr.⟦ M ⟧C .N-ob w W.id
comparison₀ fo M = lift-canon fo (related-closed M)

-- the machine's map into the free model

module _ {A : ValTy} where
  private
    module Adj = UnitCounit._⊣_ (FreeModel.adjunction fm)

  -- the free functor on same: T ⟦ A ⟧ → T ⟦ A ⟧'
  Tsame : FirstOrder A → Hom (PM.polynomial L O) (Fr.T Tm.⟦ A ⟧v .fst) Fr.⟦ F A ⟧c
  Tsame fo = FreeModel.free fm ⟪ PshHom→NatTrans (same fo) ⟫

  -- Machine.Run.h on a closed computation: its term-algebra denotation at
  -- the identity, evaluated in the free model at the unit
  hClosed : ∀ {w} → Comp (∅ w) (F A) → El Fr.∣ Fr.T Tm.⟦ A ⟧v ∣ w
  hClosed {w} M = evaluate (Fr.T Tm.⟦ A ⟧v .fst) (Adj.η .NatTrans.N-ob Tm.⟦ A ⟧v) .NatTrans.N-ob w (Tm.⟦ M ⟧C .N-ob w W.id)

  -- T (same) after evaluation at the unit is the canonical map
  relabel : (fo : FirstOrder A) {w : W.ob} (t : Term (PM.polynomial L O) Tm.⟦ A ⟧v w) →
    Tsame fo .Hom.transformation .NatTrans.N-ob w
      (evaluate (Fr.T Tm.⟦ A ⟧v .fst) (Adj.η .NatTrans.N-ob Tm.⟦ A ⟧v) .NatTrans.N-ob w t)
    ≡ canon fo .Hom.transformation .NatTrans.N-ob w t
  relabel fo {w} (var a) i = Adj.η .NatTrans.N-hom (PshHom→NatTrans (same fo)) (~ i) .NatTrans.N-ob w a
  relabel fo {w} (ops o ts) =
    Fr.hom-op (Tsame fo) w o _
    ∙ cong (λ c → Algebra.algebra Fr.⟦ F A ⟧c .NatTrans.N-ob w (o , c)) (funExt λ q → relabel fo (ts q))

  -- the machine's map, followed by T (same), is the free-model denotation
  machine-comparison : (fo : FirstOrder A) {w : W.ob} (M : Comp (∅ w) (F A)) →
    Tsame fo .Hom.transformation .NatTrans.N-ob w (hClosed M) ≡ Fr.⟦ M ⟧C .N-ob w W.id
  machine-comparison fo {w} M = relabel fo (Tm.⟦ M ⟧C .N-ob w W.id) ∙ comparison₀ fo M
