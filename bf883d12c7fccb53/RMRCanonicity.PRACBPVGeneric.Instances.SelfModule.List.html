-- The machine at Q = T for list nondeterminism WITHOUT equations
-- (Semantics.SelfMachine): W = 1, no locks, operations or (two
-- continuations) and fail (none), and no monoid equations.
--
-- The free model is then the binary trees with or-nodes, fail-leaves and
-- variables, not lists: or is not associative, fail is not its unit.  A
-- configuration is such a tree whose leaves are threads (closed
-- computations); running or in a thread splits it into two threads, and
-- running fail ends it.  RMR's list configurations (Q = T = List, ρ =
-- concat) are the quotient by the monoid equations; their free model is a
-- higher inductive type and is out of scope here.
--
--   finCont        or has two continuations, fail none
--   list-sn/-wn    strong and weak normalization at Q = T
--   orFail-run     η (or (return tt) fail) ⟶* orFailDone: the tree
--                  or(thread returning tt, fail), in two effect steps
--                  (op-root at the root, op-in at the second leaf)
--   orFail-normal  the run ends in a terminal configuration
module RMRCanonicity.PRACBPVGeneric.Instances.SelfModule.List where

------------------------------------------------------------------------
-- the signature

data ListOp : Type where
  or fail : ListOp

discreteListOp : Discrete ListOp
discreteListOp or or = yes refl
discreteListOp or fail = no λ e → ⊥.rec (subst (λ { or → Unit ; fail → ⊥ }) e tt)
discreteListOp fail or = no λ e → ⊥.rec (subst (λ { or → ⊥ ; fail → Unit }) e tt)
discreteListOp fail fail = yes refl

ListAr : ListOp → Type
ListAr or = Bool
ListAr fail = ⊥

isSetListAr : ∀ o → isSet (ListAr o)
isSetListAr or = isSetBool
isSetListAr fail = isProp→isSet isProp⊥

listOps : OpSig (noLocks TerminalCategory)
listOps = constOps TerminalCategory ListOp (Discrete→isSet discreteListOp) ListAr isSetListAr

open import RMRCanonicity.PRACBPVGeneric.Syntax (noLocks TerminalCategory) listOps hiding (_×_)
open import RMRCanonicity.PRACBPVGeneric.Closing (noLocks TerminalCategory) listOps using (closeK)

-- or M N and fail
orK : ∀ {Θ B} → Comp Θ B → Comp Θ B → (a : Bool) (ps : WPos [] (world Θ)) → Comp (Θ ⧀*⟨ [] , ps ⟩) B
orK M N true _ = M
orK M N false _ = N

orC : ∀ {Θ B} → Comp Θ B → Comp Θ B → Comp Θ B
orC M N = op or []ᵥ (orK M N)

failK : ∀ {Θ B} (a : ⊥) (ps : WPos [] (world Θ)) → Comp (Θ ⧀*⟨ [] , ps ⟩) B
failK ()

failC : ∀ {Θ B} → Comp Θ B
failC = op fail []ᵥ failK

------------------------------------------------------------------------
-- the machine

isSetLk : isSet ⊥
isSetLk = isProp→isSet isProp⊥

isSetBase : isSet NoBase
isSetBase = isProp→isSet λ ()

isFinOrdBool : isFinOrd Bool
isFinOrdBool = 2 ,Σ invEquiv SumFin2≃Bool

-- or has two continuations, fail none
finCont : ∀ o w → isFinOrd (Σ[ a ∈ Ar o ] WPos (slot o a) w)
finCont or w = isFinOrdΣ Bool isFinOrdBool _ λ _ → isFinOrdUnit
finCont fail w = isFinOrdΣ ⊥ isFinOrd⊥ _ λ ()

module _ {A : ValTy} where
  open SelfMachine (noLocks TerminalCategory) listOps isSetLk isSetBase isSetUnit* A finCont

  list-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  list-sn = strongNormalization

  list-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  list-wn = weakNormalization

------------------------------------------------------------------------
-- a run: or (return tt) fail

open SelfMachine (noLocks TerminalCategory) listOps isSetLk isSetBase isSetUnit* 𝟙 finCont

private
  X = CompPsh (F 𝟙)

orFail : Comp (∅ tt*) (F 𝟙)
orFail = orC (return tt) failC

-- after the first step: the node or, threads return tt and fail
orFail₁ : Config
orFail₁ = nodeC or tt (orK (return tt) failC)

-- the second leaf
leaf₂ : Σ[ q ∈ Σ[ a ∈ Bool ] Unit ] Unit
leaf₂ = (false ,Σ tt) ,Σ tt

leaf₁≢leaf₂ : ¬ Path (Σ[ q ∈ Σ[ a ∈ Bool ] Unit ] Unit) ((true ,Σ tt) ,Σ tt) leaf₂
leaf₁≢leaf₂ e = true≢false (cong (λ z → z .fst .fst) e)

-- the end: the tree or(thread returning tt, fail)
orFailDone : Config
orFailDone =
  (tt* ,Σ ops (or ,Σ tt) (λ { (true ,Σ _) → var tt ; (false ,Σ _) → ops (fail ,Σ tt) λ _ → var tt })) ,Σ
  λ { ((true ,Σ _) ,Σ _) → return tt ; ((false ,Σ _) ,Σ ((() ,Σ _) ,Σ _)) }

orFailDone-terminal : Terminal orFailDone
orFailDone-terminal ((true ,Σ _) ,Σ _) = tt ,Σ refl
orFailDone-terminal ((false ,Σ _) ,Σ ((() ,Σ _) ,Σ _))

-- the second step's result is orFailDone: read off as terms (conf-ext);
-- the second leaf holds the node fail, the first its thread
step₂ : ER.step SM.selfModule (CompAlg (F 𝟙)) (ctxAt orFail₁ leaf₂)
          ((fail ,Σ tt) ,Σ λ (a ,Σ ps) → closeK [] (failK a) ps)
        ≡ orFailDone
step₂ = SM.conf-ext X (SM.to-from X _ ∙ cong (tt* ,Σ_) (cong (ops (or ,Σ tt)) (funExt λ
  { (true ,Σ tt) → plug-remaining SM.TermQ′ (ER.T .fst ⟅ X ⟆) C node ((true ,Σ tt) ,Σ tt) leaf₁≢leaf₂
  ; (false ,Σ tt) → plug-hole SM.TermQ′ (ER.T .fst ⟅ X ⟆) C node ∙ cong (ops (fail ,Σ tt)) (funExt λ { (() ,Σ _) }) })))
  where
  C = (ER.∂Q ⟪ IsMonad.η (ER.T .snd) .NatTrans.N-ob X ⟫) (ctxAt orFail₁ leaf₂)
  node = ops (fail ,Σ tt) λ (a ,Σ ps) → var (closeK [] (failK a) ps)

-- or splits the thread; then fail ends the second one
orFail-run : ηC orFail ⟶* orFailDone
orFail-run = op-root then op-in orFail₁ leaf₂ refl then (step₂ by-equation done)

orFail-normal : Σ[ y ∈ Config ] (ηC orFail ⟶* y) × Terminal y
orFail-normal = orFailDone ,Σ orFail-run ,Σ orFailDone-terminal

orFail-run-sound : denote (ηC orFail) ≡ denote orFailDone
orFail-run-sound = sound⟶* orFail-run
