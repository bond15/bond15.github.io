-- Tree reduction is deterministic (6.1).
--
-- A step function  next : Comp Θ B → Maybe (Comp Θ B)  finds the redex of
-- a computation: it walks down the head of to and app (the frames of a
-- stack) and fires the head step it finds there.  Every step agrees with
-- it:
--
--   head-next   M ⇝ M'  →  next M ≡ just M'
--   plug-next   M ⇝ M'  →  next (plug S M) ≡ just (plug S M')
--   step-next   M ↦ M'  →  next M ≡ just M'
--
-- So two steps from the same computation have the same target (↦-det),
-- and return and op do not step (ret-normal, op-normal): next is nothing
-- on them.  An op is normal because tree reduction has no rule under the
-- locks of its continuations (only ↦*'s op* goes there).
--
-- ↦-to and ↦-app add a frame to a step (the frames of the logical
-- relation's anti-reduction lemma).
open import Cubical.Foundations.Prelude hiding (Sub; _,_; comp)
open import Cubical.Data.Maybe using (Maybe; nothing; just; ¬nothing≡just; just-inj)
  renaming (map-Maybe to mapM)
open import Cubical.Data.Empty as ⊥ using ()
open import Cubical.Relation.Nullary using (¬_)
open import Cubical.Categories.Category using (Category)

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Metatheory.Determinism
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
open import RMRCanonicity.PRACBPVGeneric.Reduction L O

----------------------------------------------------------------------
-- frames

↦-to : ∀ {Θ A B} {M M' : Comp Θ (F A)} (N : Comp (Θ , A) B) → M ↦ M' → (M to N) ↦ (M' to N)
↦-to N (step S r) = step (S to• N) r

↦-app : ∀ {Θ A B} {M M' : Comp Θ (A ⇒ B)} (V : Val Θ A) → M ↦ M' → app M V ↦ app M' V
↦-app V (step S r) = step (app• S V) r

----------------------------------------------------------------------
-- the step function

next : ∀ {Θ B} → Comp Θ B → Maybe (Comp Θ B)
-- M to N, given the step of M
nextTo : ∀ {Θ A B} → Comp Θ (F A) → Comp (Θ , A) B → Maybe (Comp Θ (F A)) → Maybe (Comp Θ B)
-- app M V, given the step of M
nextApp : ∀ {Θ A B} → Comp Θ (A ⇒ B) → Val Θ A → Maybe (Comp Θ (A ⇒ B)) → Maybe (Comp Θ B)

next (return V) = nothing
next (M to N) = nextTo M N (next M)
next (force (ne n)) = nothing
next (force (thunk M)) = just M
next (case (ne n) M) = nothing
next (case (pair V V') M) = just (sub₂ M V V')
next (case₊ (ne n) M N) = nothing
next (case₊ (inl V) M N) = just (sub₁ M V)
next (case₊ (inr V) M N) = just (sub₁ N V)
next (absurd V) = nothing
next (lam M) = nothing
next (app M V) = nextApp M V (next M)
next (op o vs k) = nothing

nextTo (return V) N _ = just (sub₁ N V)
nextTo (op o vs k) N _ =
  just (op o vs λ a ps → k a ps to renC (keep (dropW idR (slot o a) ps)) N)
nextTo (M to M') N r = mapM (_to N) r
nextTo (force V) N r = mapM (_to N) r
nextTo (case V M) N r = mapM (_to N) r
nextTo (case₊ V M M') N r = mapM (_to N) r
nextTo (absurd V) N r = mapM (_to N) r
nextTo (app M V) N r = mapM (_to N) r

nextApp (lam M) V _ = just (sub₁ M V)
nextApp (op o vs k) V _ =
  just (op o vs λ a ps → app (k a ps) (renV (dropW idR (slot o a) ps) V))
nextApp (M to N) V r = mapM (λ M' → app M' V) r
nextApp (force V') V r = mapM (λ M' → app M' V) r
nextApp (case V' M) V r = mapM (λ M' → app M' V) r
nextApp (case₊ V' M N) V r = mapM (λ M' → app M' V) r
nextApp (absurd V') V r = mapM (λ M' → app M' V) r
nextApp (app M V') V r = mapM (λ M' → app M' V) r

----------------------------------------------------------------------
-- every step agrees with next

head-next : ∀ {Θ B} {M M' : Comp Θ B} → M ⇝ M' → next M ≡ just M'
head-next β-to = refl
head-next β-force = refl
head-next β-case = refl
head-next β-case₊-inl = refl
head-next β-case₊-inr = refl
head-next β-app = refl
head-next (op-to o vs) = refl
head-next (op-app o vs) = refl

-- a computation that steps is not return or op, so to and app pass its step on
nextTo-step : ∀ {Θ A B} (M : Comp Θ (F A)) (N : Comp (Θ , A) B) {M'} →
  next M ≡ just M' → nextTo M N (next M) ≡ just (M' to N)
nextTo-step (return V) N e = ⊥.rec (¬nothing≡just e)
nextTo-step (op o vs k) N e = ⊥.rec (¬nothing≡just e)
nextTo-step (M to M₁) N e = cong (mapM (_to N)) e
nextTo-step (force V) N e = cong (mapM (_to N)) e
nextTo-step (case V M) N e = cong (mapM (_to N)) e
nextTo-step (case₊ V M M') N e = cong (mapM (_to N)) e
nextTo-step (absurd V) N e = cong (mapM (_to N)) e
nextTo-step (app M V) N e = cong (mapM (_to N)) e

nextApp-step : ∀ {Θ A B} (M : Comp Θ (A ⇒ B)) (V : Val Θ A) {M'} →
  next M ≡ just M' → nextApp M V (next M) ≡ just (app M' V)
nextApp-step (lam M) V e = ⊥.rec (¬nothing≡just e)
nextApp-step (op o vs k) V e = ⊥.rec (¬nothing≡just e)
nextApp-step (M to N) V e = cong (mapM (λ M' → app M' V)) e
nextApp-step (force V') V e = cong (mapM (λ M' → app M' V)) e
nextApp-step (case V' M) V e = cong (mapM (λ M' → app M' V)) e
nextApp-step (case₊ V' M N) V e = cong (mapM (λ M' → app M' V)) e
nextApp-step (absurd V') V e = cong (mapM (λ M' → app M' V)) e
nextApp-step (app M V') V e = cong (mapM (λ M' → app M' V)) e

plug-next : ∀ {Θ B B'} (S : Stack Θ B B') {M M' : Comp Θ B} → M ⇝ M' →
  next (plug S M) ≡ just (plug S M')
plug-next hole r = head-next r
plug-next (S to• N) {M} r = nextTo-step (plug S M) N (plug-next S r)
plug-next (app• S V) {M} r = nextApp-step (plug S M) V (plug-next S r)

step-next : ∀ {Θ B} {M M' : Comp Θ B} → M ↦ M' → next M ≡ just M'
step-next (step S r) = plug-next S r

----------------------------------------------------------------------
-- determinism and normal forms

↦-det : ∀ {Θ B} {M M₁ M₂ : Comp Θ B} → M ↦ M₁ → M ↦ M₂ → M₁ ≡ M₂
↦-det r₁ r₂ = just-inj _ _ (sym (step-next r₁) ∙ step-next r₂)

op-normal : ∀ {Θ B} (o : Op) {vs : Vals Θ (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B} {M'} →
  ¬ (op o vs k ↦ M')
op-normal o r = ¬nothing≡just (step-next r)

ret-normal : ∀ {Θ A} {V : Val Θ A} {M'} → ¬ (return V ↦ M')
ret-normal r = ¬nothing≡just (step-next r)
