-- Algebras of an RMR signature with a strength, on presheaves of worlds
-- (copied from the old core's Denotation, generic in the signature):
--
--   the free algebra on A        the term algebra  T A  (no equations)
--   the exponential algebra      (G ⇒ U B) with the structure
--       Ext (G ⇒ B) × G  --strength-->  Ext ((G ⇒ B) × G)  --Ext ev-->  Ext B  --α-->  B
--     (the strength moves the argument into each continuation along ι)
--   bind                          the Kleisli extension  T A × G → U B  of  G × A → U B,
--                                 through the exponential algebra
open import Cubical.Foundations.Prelude
open import Cubical.Data.Sigma using (ΣPathP)
open import Cubical.Categories.Category using (Category; _^op; _[_,_])
open import Cubical.Categories.Functor using (Functor; _∘F_; _⟅_⟆)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Presheaf.Constructions.BinProduct
  using (_×Psh_; π₁; π₂; ×PshIntro; _×PshHom_)
open import Cubical.Categories.Presheaf.Morphism.Alt
  using (PshHom; NatTrans→PshHom; PshHom→NatTrans; _⋆PshHom_; idPshHom; makePshHomPath)
open import Polynomial.Presheaf.Discrete.Base using (Poly)

open import RightModuleReduction.Theory
  using (Signature; module SignatureNotation; Algebra; Term; TermPsh; evaluate; var; ops)
open import RightModuleReduction.TermAlgebra using (TermAlg)

module RMRCanonicity.PRACBPVGeneric.Semantics.Algebra
  {W : Category ℓ-zero ℓ-zero} (P : Signature W)
  (str : NatTrans (Poly.s P) (Poly.t P ∘F Poly.p P)) where

open import RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf W

open SignatureNotation P
open Functor
open PshHom

private
  module W = Category W

-- the strength at a position
ι : ∀ w (o : Shape w) (p : Position w o) → W [ w , scope w o p ]
ι w o p = str .NatTrans.N-ob ((w , o) , p)

-- the signature's extension functor on presheaves
Ext : 𝒱 → 𝒱
Ext X = Poly.Extension P ⟅ X ⟆

-- carry an argument into every continuation's world along ι
strength : (X A : 𝒱) → PshHom (Ext X ×Psh A) (Ext (X ×Psh A))
strength X A .N-ob w ((o , k) , a) = o , λ p → k p , A .F-hom (ι w o p) a
strength X A .N-hom c c' f ((o , k) , a) =
  ΣPathP (refl , funExt λ p → ΣPathP (refl , path p))
  where
  path : ∀ p → _
  path p =
    sym (funExt⁻ (A .F-seq f (ι c _ p)) a)
    ∙ cong (λ g → A .F-hom g a) (sym (str .NatTrans.N-hom ((f , refl) , refl)))
    ∙ funExt⁻ (A .F-seq _ _) a

-- the exponential algebra: run the operation after moving the argument along ι
expMap : (A : 𝒱) (B : Algebra P) →
  PshHom (Ext (Exp A (Algebra.Carrier B)) ×Psh A) (Algebra.Carrier B)
expMap A B =
  strength (Exp A (Algebra.Carrier B)) A
    ⋆PshHom NatTrans→PshHom (Poly.Extension P .F-hom
      (PshHom→NatTrans (appExp A (Algebra.Carrier B))))
    ⋆PshHom NatTrans→PshHom (Algebra.algebra B)

-- opaque: expAlgebra-ob is its only unfolding
opaque
  expAlgebra : (A : 𝒱) (B : Algebra P) →
    NatTrans (Ext (Exp A (Algebra.Carrier B))) (Exp A (Algebra.Carrier B))
  expAlgebra A B = PshHom→NatTrans (λExp A (Algebra.Carrier B) (expMap A B))

  expAlgebra-ob : (A : 𝒱) (B : Algebra P) (w : W.ob)
    (e : El (Ext (Exp A (Algebra.Carrier B))) w)
    (v : W.ob) (h : W [ w , v ]) (a : El A v) →
    expAlgebra A B .NatTrans.N-ob w e .N-ob v (h , a)
      ≡ expMap A B .N-ob v (Ext (Exp A (Algebra.Carrier B)) .F-hom h e , a)
  expAlgebra-ob A B w e v h a = refl

expAlg : 𝒱 → Algebra P → Algebra P
expAlg A B .Algebra.Carrier = Exp A (Algebra.Carrier B)
expAlg A B .Algebra.algebra = expAlgebra A B

-- returning a value: a leaf of the term algebra
η : (X : 𝒱) → PshHom X (TermPsh P X)
η X .N-ob w x = var x
η X .N-hom c c' f x = refl

-- sequencing: extend N along the free algebra, through the exponential algebra
bindEnv : (G : 𝒱) (A : 𝒱) (B : Algebra P) →
  PshHom (G ×Psh A) (Algebra.Carrier B) →
  NatTrans A (Algebra.Carrier (expAlg G B))
bindEnv G A B N =
  PshHom→NatTrans (λExp G (Algebra.Carrier B) (×PshIntro (π₂ _ _) (π₁ _ _) ⋆PshHom N))

opaque
  bind : (G : 𝒱) (A : 𝒱) (B : Algebra P) →
    PshHom (G ×Psh A) (Algebra.Carrier B) →
    PshHom (TermPsh P A ×Psh G) (Algebra.Carrier B)
  bind G A B N =
    (NatTrans→PshHom (evaluate (expAlg G B) (bindEnv G A B N)) ×PshHom idPshHom)
    ⋆PshHom appExp G (Algebra.Carrier B)

-- interface lemmas (copied from the old core's DenotationLemmas)

-- the position and map of the extension's action at a canonical lift
tpos : ∀ {w w'} (f : W [ w , w' ]) (o : Shape w) → Position w' (Poly.Shapes P .F-hom f o) →
  Position w o
tpos f o p = Poly.transportPosition P (f , refl) p

smapP : ∀ {w w'} (f : W [ w , w' ]) (o : Shape w) (p : Position w' (Poly.Shapes P .F-hom f o)) →
  W [ scope w o (tpos f o p) , scope w' (Poly.Shapes P .F-hom f o) p ]
smapP f o p = Poly.s P .F-hom ((f , refl) , refl)

-- the exponential's operation at the identity world map
expAlgebra-id : (A : 𝒱) (B : Algebra P) (v : W.ob) (o : Shape v)
  (ks : (p : Position v o) → El (Exp A (Algebra.Carrier B)) (scope v o p))
  (a : El A v) →
  expAlgebra A B .NatTrans.N-ob v (o , ks) .N-ob v (W.id , a)
    ≡ Algebra.algebra B .NatTrans.N-ob v
        (o , λ p → ks p .N-ob (scope v o p) (W.id , A .F-hom (ι v o p) a))
expAlgebra-id A B v o ks a =
  expAlgebra-ob A B v (o , ks) v W.id a
  ∙ cong (λ y → expMap A B .N-ob v (y , a))
      (funExt⁻ (Ext (Exp A (Algebra.Carrier B)) .F-id) (o , ks))

-- evaluating into the exponential commutes with changing the environment
module _ {G G' X : 𝒱} {B : Algebra P} (φ : PshHom G' G)
  (env : NatTrans X (Algebra.Carrier (expAlg G B)))
  (env' : NatTrans X (Algebra.Carrier (expAlg G' B)))
  (hyp : ∀ {w} (a : El X w) v (g : W [ w , v ]) (x' : El G' v) →
     env' .NatTrans.N-ob w a .N-ob v (g , x') ≡ env .NatTrans.N-ob w a .N-ob v (g , φ .N-ob v x'))
  where

  evalPre : ∀ {w} (t : Term P X w) v (g : W [ w , v ]) (x' : El G' v) →
    evaluate (expAlg G' B) env' .NatTrans.N-ob w t .N-ob v (g , x')
      ≡ evaluate (expAlg G B) env .NatTrans.N-ob w t .N-ob v (g , φ .N-ob v x')
  evalPre (var a) v g x' = hyp a v g x'
  evalPre {w} (ops o args) v g x' =
    expAlgebra-ob G' B w _ v g x'
    ∙ cong (λ c → Algebra.algebra B .NatTrans.N-ob v (Poly.Shapes P .F-hom g o , c))
      (funExt λ p →
        evalPre (args (tpos g o p)) (scope v _ p) _ (G' .F-hom (ι v _ p) x')
        ∙ cong (λ y → evaluate (expAlg G B) env .NatTrans.N-ob _ (args (tpos g o p))
                        .N-ob (scope v _ p) (_ , y))
            (φ .N-hom _ _ (ι v _ p) x'))
    ∙ sym (expAlgebra-ob G B w _ v g (φ .N-ob v x'))

-- bind's interface: it commutes with changing the environment along φ,
-- sends a variable to N, and passes operations through
opaque
  unfolding bind

  bind-pre : (G G' X : 𝒱) (B : Algebra P) (φ : PshHom G' G)
    (N : PshHom (G ×Psh X) (Algebra.Carrier B)) (N' : PshHom (G' ×Psh X) (Algebra.Carrier B))
    (hyp : ∀ v (x' : El G' v) (a : El X v) →
      N' .N-ob v (x' , a) ≡ N .N-ob v (φ .N-ob v x' , a))
    (v : W.ob) (t : Term P X v) (x' : El G' v) →
    bind G' X B N' .N-ob v (t , x') ≡ bind G X B N .N-ob v (t , φ .N-ob v x')
  bind-pre G G' X B φ N N' hyp v t x' =
    evalPre {G = G} {G' = G'} {X = X} {B = B} φ (bindEnv G X B N) (bindEnv G' X B N')
      (λ a u g y → hyp u y (X .F-hom g a)) t v W.id x'

  bind-η : (G X : 𝒱) (B : Algebra P) (N : PshHom (G ×Psh X) (Algebra.Carrier B))
    (v : W.ob) (a : El X v) (x : El G v) →
    bind G X B N .N-ob v (var a , x) ≡ N .N-ob v (x , a)
  bind-η G X B N v a x = cong (λ a' → N .N-ob v (x , a')) (funExt⁻ (X .F-id) a)

  bind-op : (G X : 𝒱) (B : Algebra P) (N : PshHom (G ×Psh X) (Algebra.Carrier B))
    (v : W.ob) (o : Shape v) (ts : (p : Position v o) → Term P X (scope v o p))
    (x : El G v) →
    bind G X B N .N-ob v (ops o ts , x)
      ≡ Algebra.algebra B .NatTrans.N-ob v
          (o , λ p → bind G X B N .N-ob (scope v o p) (ts p , G .F-hom (ι v o p) x))
  bind-op G X B N v o ts x = expAlgebra-id G B v o _ x
