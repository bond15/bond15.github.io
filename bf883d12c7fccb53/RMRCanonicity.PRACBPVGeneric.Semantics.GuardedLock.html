-- On an instance the canonical lock is the evident functor.  For the
-- guarded tick (W = ω^op) the coend collapses:
--
--   L_tick Γ t ≅ Γ (t + 1)          (◀, "earlier": restriction along +1)
--
-- because every point ⟪ v + 1 , tt , g , e ⟫ is identified with
-- ⟪ t + 1 , tt , id , Γ (v + 1 → t + 1) e ⟫.  (For local state the same
-- computation gives the Day convolution with y 1, a genuine quotient in
-- general.)
open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Isomorphism using (Iso)
open import Cubical.Data.Nat using (ℕ; zero; suc)
open import Cubical.Data.Unit using (tt)
open import Cubical.HITs.SetQuotients using (eq/; squash/; rec; elimProp) renaming ([_] to ⟪_⟫)
open import Cubical.Categories.Functor using (Functor)

open import RMRCanonicity.PRACBPVGeneric.Instances.Guarded
  using (Omega; _≤ω_; s≤s; ≤ω-refl; isProp≤ω; Etick; strTick)

module RMRCanonicity.PRACBPVGeneric.Semantics.GuardedLock where

open import RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf Omega
open import RMRCanonicity.PRACBPVGeneric.Semantics.Lock Etick strTick

open Functor

module _ (Γ : 𝒱) where
  -- read a point at the next world
  earlier : ∀ t → El (L Γ) t → El Γ (suc t)
  earlier t = rec (Γ .F-ob (suc t) .snd) pt' resp
    where
    pt' : Raw Γ t → El Γ (suc t)
    pt' (suc v , q , g , e) = Γ .F-hom (s≤s g) e

    resp : ∀ a b → Shift Γ t a b → pt' a ≡ pt' b
    resp _ _ (shift {suc v} {suc v'} (s≤s f) q g e) =
      sym (funExt⁻ (Γ .F-seq (s≤s f) (s≤s g)) e)
      ∙ cong (λ m → Γ .F-hom m e) (isProp≤ω _ _)

  later : ∀ t → El Γ (suc t) → El (L Γ) t
  later t γ = ⟪ suc t , tt , ≤ω-refl , γ ⟫

  earlierIso : ∀ t → Iso (El (L Γ) t) (El Γ (suc t))
  earlierIso t .Iso.fun = earlier t
  earlierIso t .Iso.inv = later t
  earlierIso t .Iso.sec γ = funExt⁻ (Γ .F-id) γ
  earlierIso t .Iso.ret = elimProp (λ _ → squash/ _ _) λ where
    (suc v , q , g , e) →
      eq/ _ _ (shift (s≤s g) tt ≤ω-refl e)
      ∙ cong (λ m → pt Γ (suc v , tt , m , e)) (isProp≤ω _ _)

  -- (stage 6, additive) earlier is natural in the world: with earlierIso
  -- this is an isomorphism of presheaves L_tick Γ ≅ Γ (− + 1)
  earlier-nat : ∀ {t t'} (f : t' ≤ω t) (x : El (L Γ) t) →
    earlier t' (L Γ .F-hom f x) ≡ Γ .F-hom (s≤s f) (earlier t x)
  earlier-nat {t' = t'} f = elimProp (λ _ → Γ .F-ob (suc t') .snd _ _) λ where
    (suc v , q , g , e) →
      cong (λ m → Γ .F-hom m e) (isProp≤ω _ _)
      ∙ funExt⁻ (Γ .F-seq (s≤s g) (s≤s f)) e
