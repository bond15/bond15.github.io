-- The maximum of a ℕ-valued function on a finite set (Def 2.3 of the
-- build spec).
--
-- For a finite set X and g : X → ℕ, a least upper bound of g is an n with
--
--     IsLub X g n  :=  (∀ x → g x ≤ n) × (∀ m → (∀ x → g x ≤ m) → n ≤ m).
--
-- Being a least upper bound is a proposition, and least upper bounds are
-- unique (≤ is antisymmetric), so Σ n (IsLub X g n) is a proposition.  A
-- finite set only comes with a TRUNCATED enumeration ∥ X ≃ Fin k ∥; since
-- the goal is a proposition we may eliminate the truncation, and on an
-- enumerated set the lub is computed by recursion on k (the maximum of
-- g 0 and the lub of the rest).  The empty set has lub 0.
--
--   IsLub, isPropIsLub, isPropLub, lub
--   max X g        the lub's value
--   max-ub         it bounds every value
--   max-least      it is below every upper bound
--   max-mono       pointwise ≤ gives ≤ of maxima
--   max-pull       restricting along any map π : Y → X does not increase it
--                  (no surjectivity or injectivity of π needed)
module RMRCanonicity.PRACBPVGeneric.FinMax where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Equiv
open import Cubical.Foundations.HLevels
open import Cubical.Data.Sigma
open import Cubical.Data.Nat using (ℕ ; zero ; suc) renaming (max to maxℕ)
open import Cubical.Data.Nat.Order
  using (_≤_ ; isProp≤ ; ≤-antisym ; ≤-trans ; zero-≤ ; left-≤-max ; right-≤-max ; maxLUB)
open import Cubical.Data.SumFin.Base using (Fin ; fzero ; fsuc)
open import Cubical.Data.FinSet.Base using (FinSet)
import Cubical.HITs.PropositionalTruncation as PT

private
  variable
    ℓ ℓ' : Level

-- n is a least upper bound of g
IsLub : (X : Type ℓ) → (X → ℕ) → ℕ → Type ℓ
IsLub X g n = (∀ x → g x ≤ n) × (∀ m → (∀ x → g x ≤ m) → n ≤ m)

isPropIsLub : (X : Type ℓ) (g : X → ℕ) (n : ℕ) → isProp (IsLub X g n)
isPropIsLub X g n =
  isProp× (isPropΠ λ _ → isProp≤) (isPropΠ2 λ _ _ → isProp≤)

-- least upper bounds are unique
isPropLub : (X : Type ℓ) (g : X → ℕ) → isProp (Σ[ n ∈ ℕ ] IsLub X g n)
isPropLub X g (n , ub , least) (n' , ub' , least') =
  Σ≡Prop (isPropIsLub X g) (≤-antisym (least n' ub') (least' n ub))

private
  -- the lub on an enumerated set, by recursion on the size
  -- (Fin is Cubical.Data.SumFin's, the one FinSet enumerates by:
  --  Fin 0 = ⊥, Fin (suc k) = ⊤ ⊎ Fin k)
  lubFin : ∀ k (g : Fin k → ℕ) → Σ[ n ∈ ℕ ] IsLub (Fin k) g n
  lubFin zero g = 0 , (λ ()) , λ m _ → zero-≤
  lubFin (suc k) g =
    let (n , ub , least) = lubFin k (λ i → g (fsuc i))
    in maxℕ (g fzero) n
     , (λ { fzero → left-≤-max ; (fsuc i) → ≤-trans (ub i) (right-≤-max {m = g fzero}) })
     , λ m bound → maxLUB (bound fzero) (least m (λ i → bound (fsuc i)))

  -- least upper bounds transfer along an equivalence
  lubEquiv : {X : Type ℓ} {Y : Type ℓ'} (e : X ≃ Y) (g : X → ℕ) →
    Σ[ n ∈ ℕ ] IsLub Y (λ y → g (invEq e y)) n → Σ[ n ∈ ℕ ] IsLub X g n
  lubEquiv e g (n , ub , least) =
      n
    , (λ x → subst (λ z → g z ≤ n) (retEq e x) (ub (equivFun e x)))
    , λ m bound → least m (λ y → bound (invEq e y))

-- the lub over a finite set: eliminate the truncated enumeration into the
-- proposition Σ n (IsLub X g n)
lub : (X : FinSet ℓ) (g : X .fst → ℕ) → Σ[ n ∈ ℕ ] IsLub (X .fst) g n
lub X g =
  PT.rec (isPropLub (X .fst) g)
    (λ e → lubEquiv e g (lubFin _ _))
    (X .snd .snd)

max : (X : FinSet ℓ) → (X .fst → ℕ) → ℕ
max X g = lub X g .fst

max-ub : (X : FinSet ℓ) (g : X .fst → ℕ) → ∀ x → g x ≤ max X g
max-ub X g = lub X g .snd .fst

max-least : (X : FinSet ℓ) (g : X .fst → ℕ) → ∀ m → (∀ x → g x ≤ m) → max X g ≤ m
max-least X g = lub X g .snd .snd

max-mono : (X : FinSet ℓ) (g h : X .fst → ℕ) → (∀ x → g x ≤ h x) → max X g ≤ max X h
max-mono X g h g≤h = max-least X g (max X h) (λ x → ≤-trans (g≤h x) (max-ub X h x))

max-pull : (X : FinSet ℓ) (Y : FinSet ℓ') (g : X .fst → ℕ) (π : Y .fst → X .fst) →
  max Y (λ y → g (π y)) ≤ max X g
max-pull X Y g π = max-least Y (λ y → g (π y)) (max X g) (λ y → max-ub X g (π y))
