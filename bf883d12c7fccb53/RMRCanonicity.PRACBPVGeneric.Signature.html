-- The parameter of the syntax, in two layers.
--
-- (i) A LOCK SIGNATURE (the mode theory): base types b with presheaves of
--     constants Const b : W → Set, and a set Lk of LOCK KINDS.  A lock kind
--     κ is a strong familial arity with names:
--
--       E κ   : W^op → Fam W     E κ w = Σ_{p ∈ Pos κ w} y (scope κ w p)
--       str κ : E κ ⇒ y          the strength (ι κ w p : w → scope κ w p)
--       names κ , gen κ          the names the lock binds: a global element
--                                of R_κ (Π_{b ∈ names κ} Const b),
--                                i.e. a map L_κ 1 → Π Const
--
--     Each lock kind is a modality: [ κ ] A denotes R_κ ⟦A⟧, the right
--     adjoint of the lock L_κ.
--
-- (ii) An OPERATION SIGNATURE over it: each operation o has typed value
--     parameters, and an arity in Fam (Lk*), a family of lock words:
--
--       params o : List Base            Par_o = Π_{b ∈ params o} Const b
--       Ar o     : a set                the slots (continuations) of o
--       slot o a : List Lk              the locks the slot a is behind
--
--     A word μ denotes the composite arity E μ (y for [], E κ ⊙ E μ' for
--     κ ∷ μ'), and the family denotes the sum  E o = Σ_{a ∈ Ar o} E (slot o a).
--     The signature functor is
--       Ext X w = Σ_o Par_o w × R_{E o} X w,
--     a parametric right adjoint with Ext 1 = Σ_o Par_o; its strength comes
--     from the lock kinds' (Polynomial.agda).
--
-- All laws live in the Functor / NatTrans / Global records of the lock
-- kinds.  Operations carry no laws; the only side conditions are that
-- operations and slots form sets.
module RMRCanonicity.PRACBPVGeneric.Signature where

open import Cubical.Foundations.Prelude
open import Cubical.Categories.Category using (Category; _[_,_]; _^op)
open import Cubical.Categories.Functor using (Functor)
open import Cubical.Categories.NaturalTransformation using (NatTrans; idTrans)
open import Cubical.Categories.Instances.Sets using (SET)
open import Cubical.Data.List using (List; []; _∷_)

open import RMRCanonicity.PRACBPVGeneric.Fam

record LockKind (W : Category ℓ-zero ℓ-zero) {Base : Type}
  (Const : Base → Functor W (SET ℓ-zero)) : Type₁ where
  field
    E : Functor (W ^op) (Fam W)
    str : NatTrans E (Yo W)
    names : List Base
    gen : Global W (Arity.R W E (CstsF W Const names))

record LockSig (W : Category ℓ-zero ℓ-zero) : Type₁ where
  field
    Base : Type
    Const : Base → Functor W (SET ℓ-zero)
    Lk : Type
    kind : Lk → LockKind W Const

record OpSig {W : Category ℓ-zero ℓ-zero} (L : LockSig W) : Type₁ where
  open LockSig L
  field
    Op : Type
    isSetOp : isSet Op
    params : Op → List Base
    Ar : Op → Type
    isSetAr : ∀ o → isSet (Ar o)
    slot : (o : Op) → Ar o → List Lk

-- The notation the syntax and the polynomial share (defined once, so that
-- both state their types with the same names).
module Notation {W : Category ℓ-zero ℓ-zero} (L : LockSig W) where
  open LockSig L public
  private
    module W = Category W
    module K (κ : Lk) = LockKind (kind κ)

  World : Type
  World = W.ob

  Cst : Base → World → Type
  Cst b w = Const b .Functor.F-ob w .fst

  Cs : World → List Base → Type
  Cs = Csts W Const

  -- a lock kind, read off its arity
  E : Lk → Functor (W ^op) (Fam W)
  E κ = K.E κ

  Pos : Lk → World → Type
  Pos κ = Arity.Pos W (E κ)

  scope : (κ : Lk) (w : World) → Pos κ w → World
  scope κ = Arity.scope W (E κ)

  pos : ∀ κ {w w'} → W [ w , w' ] → Pos κ w' → Pos κ w
  pos κ = Arity.pos W (E κ)

  smap : ∀ κ {w w'} (f : W [ w , w' ]) (p : Pos κ w') →
    W [ scope κ w (pos κ f p) , scope κ w' p ]
  smap κ = Arity.smap W (E κ)

  -- the strength at a position (Kripke access across the lock)
  ι : ∀ κ w (p : Pos κ w) → W [ w , scope κ w p ]
  ι κ = ιOf W (K.str κ)

  names : Lk → List Base
  names κ = K.names κ

  -- the names bound by the lock at position p (generic elements)
  gen : ∀ κ w (p : Pos κ w) → Cs (scope κ w p) (names κ)
  gen κ = Global.el (K.gen κ)

  -- a lock word denotes the composite of its lock kinds' arities,
  -- bracketed without a trailing unit (so [ κ ] denotes E κ itself)
  wordE : List Lk → Functor (W ^op) (Fam W)
  wordE [] = Yo W
  wordE (κ ∷ []) = E κ
  wordE (κ ∷ μ@(_ ∷ _)) = _⊙_ W (E κ) (wordE μ)

  wordStr : (μ : List Lk) → NatTrans (wordE μ) (Yo W)
  wordStr [] = idTrans (Yo W)
  wordStr (κ ∷ []) = K.str κ
  wordStr (κ ∷ μ@(_ ∷ _)) = _⊙ˢ_ W (K.str κ) (wordStr μ)

  -- positions and scope of a lock word
  WPos : List Lk → World → Type
  WPos μ = Arity.Pos W (wordE μ)

  wscope : (μ : List Lk) (w : World) → WPos μ w → World
  wscope μ = Arity.scope W (wordE μ)

  -- the strength of a lock word: the composite of the lock kinds' ι
  wι : ∀ μ w (ps : WPos μ w) → W [ w , wscope μ w ps ]
  wι μ = ιOf W (wordStr μ)
