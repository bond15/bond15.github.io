-- The free model of a theory without equations: the term algebra, with
-- its universal property (FreeAdjunction.FromUniversalProperty).
-- Extension is evaluateHom; uniqueness is by induction on terms.  So the
-- free-model monad is T X = terms over X, with η = var and μ = evaluate.
--
--   freeModel      the term model, a FreeModel of the theory
--   unit-is-var    the adjunction's unit is var (by definition)
--   evaluate-var   evaluating a term at the unit gives the term back
open import Cubical.Foundations.Prelude
open import Cubical.Data.Empty using (⊥)
import Cubical.Data.Empty as ⊥
open import Cubical.Categories.Category using (Category; _^op)
open import Cubical.Categories.Functor using (Functor; _∘F_; 𝟙⟨_⟩)
open import Cubical.Categories.NaturalTransformation using (NatTrans; makeNatTransPath)
open import Cubical.Categories.Presheaf.Base using (PresheafCategory)
open import Cubical.Categories.Adjoint using (module UnitCounit)

open import RightModuleReduction.Theory
  using (Theory; Algebra; Satisfies; Term; var; ops; TermPsh; evaluate)
open import RightModuleReduction.TermAlgebra using (TermAlg; evaluateHom)
open import RightModuleReduction.Model using (Hom; Hom-path; Mod; modelForgetful; FreeModel)
import RightModuleReduction.FreeAdjunction as FA

module RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel {W : Category ℓ-zero ℓ-zero}
  (𝒯 : Theory W) (noEquations : Theory.Equation 𝒯 → ⊥) where

open Functor
open NatTrans

private
  𝒞 = PresheafCategory (W ^op) ℓ-zero

-- with no equations, every algebra is a model
anyModel : (A : Algebra (Theory.signature 𝒯)) → Satisfies 𝒯 A
anyModel A e = ⊥.rec (noEquations e)

-- homomorphisms out of a term algebra are determined on the variables
term-unique : ∀ {X} {A : Algebra (Theory.signature 𝒯)}
  (h₁ h₂ : Hom (Theory.signature 𝒯) (TermAlg (Theory.signature 𝒯) X) A) →
  (∀ w x → h₁ .Hom.transformation .N-ob w (var x) ≡ h₂ .Hom.transformation .N-ob w (var x)) →
  h₁ ≡ h₂
term-unique {X} {A} h₁ h₂ hv = Hom-path (Theory.signature 𝒯) (makeNatTransPath (funExt λ w → funExt (go w)))
  where
  -- a homomorphism sends ops to the algebra
  pres : (h : Hom (Theory.signature 𝒯) (TermAlg (Theory.signature 𝒯) X) A) → ∀ w o args →
    h .Hom.transformation .N-ob w (ops o args)
      ≡ Algebra.algebra A .N-ob w (o , λ p → h .Hom.transformation .N-ob _ (args p))
  pres h w o args i = h .Hom.preserves i .N-ob w (o , args)

  go : ∀ w (t : Term (Theory.signature 𝒯) X w) →
    h₁ .Hom.transformation .N-ob w t ≡ h₂ .Hom.transformation .N-ob w t
  go w (var x) = hv w x
  go w (ops o args) =
    pres h₁ w o args
    ∙ cong (λ c → Algebra.algebra A .N-ob w (o , c)) (funExt λ p → go _ (args p))
    ∙ sym (pres h₂ w o args)

-- the term algebra on X, as a model
TermModel : 𝒞 .Category.ob → Mod 𝒯 .Category.ob
TermModel X = TermAlg (Theory.signature 𝒯) X , anyModel (TermAlg (Theory.signature 𝒯) X)

η : (X : 𝒞 .Category.ob) → NatTrans X (TermPsh (Theory.signature 𝒯) X)
η X .N-ob w x = var x
η X .N-hom f = refl

-- the free functor: substitution of variables
free : Functor 𝒞 (Mod 𝒯)
free .F-ob = TermModel
free .F-hom {X} {Y} f = evaluateHom (TermAlg (Theory.signature 𝒯) Y) (seq f)
  where
  seq : NatTrans X Y → NatTrans X (TermPsh (Theory.signature 𝒯) Y)
  seq f .N-ob w x = var (f .N-ob w x)
  seq f .N-hom g = funExt λ x → cong var (funExt⁻ (f .N-hom g) x)
free .F-id {X} = term-unique _ _ λ w x → refl
free .F-seq {X} {Y} {Z} f g = term-unique _ _ λ w x → refl

unit : NatTrans 𝟙⟨ 𝒞 ⟩ (modelForgetful 𝒯 ∘F free)
unit .N-ob X = η X
unit .N-hom f = makeNatTransPath refl

module UP = FA.FromUniversalProperty 𝒞 (Mod 𝒯) free (modelForgetful 𝒯) unit
  (λ {X} {B} f → evaluateHom (B .fst) f)
  (λ f → makeNatTransPath refl)
  (λ {X} {B} f h p → term-unique h (evaluateHom (B .fst) f) λ w x i → p i .N-ob w x)

-- the free model of the equation-free theory: terms
freeModel : FreeModel 𝒯
freeModel .FreeModel.free = free
freeModel .FreeModel.adjunction = UP.adjunction

private
  module Adj = UnitCounit._⊣_ (FreeModel.adjunction freeModel)

-- the unit of the adjunction is var
unit-is-var : ∀ X w (x : X .F-ob w .fst) → Adj.η .N-ob X .N-ob w x ≡ var x
unit-is-var X w x = refl

-- evaluating a term in the free algebra at the unit gives it back
evaluate-var : ∀ X w (t : Term (Theory.signature 𝒯) X w) →
  evaluate (TermAlg (Theory.signature 𝒯) X) (Adj.η .N-ob X) .N-ob w t ≡ t
evaluate-var X w (var x) = refl
evaluate-var X w (ops o args) = cong (ops o) (funExt λ p → evaluate-var X _ (args p))
