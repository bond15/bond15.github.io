-- The error machine (W = 1, no locks, one operation raise with no slots).
--
--   S _ = 1                     one state (nothing to remember)
--   H = 1                       one outcome: raised
--   δ _ raise = halt
--
-- so Q X ≅ X(tt) + 1, RMR's error configurations: running a value, or
-- raised.  The run: a raise under a to-frame first pushes the frame into
-- raise's (no) continuations (a tree step), is closed, and halts.
module RMRCanonicity.PRACBPVGeneric.Instances.Machines.Error where

open import Cubical.Foundations.Prelude hiding (_,_)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit; Unit*; tt*)
open import Cubical.Data.Unit.Properties using (isSetUnit*)
open import Cubical.Data.Empty using (⊥; isProp⊥)
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Foundations.HLevels using (isSetΣ)
open import Cubical.Data.Sigma using (Σ-syntax) renaming (_,_ to _,Σ_)
open import Cubical.Categories.Instances.Terminal using (TerminalCategory)

open import RightModuleReduction.Theory using (module SignatureNotation)

open import RMRCanonicity.PRACBPVGeneric.Instances.Constant using (NoBase; noLocks; errorOps; raise; module Error)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
open Error
open import RMRCanonicity.PRACBPVGeneric.Reduction (noLocks TerminalCategory) errorOps using (step; op-to)
import RMRCanonicity.PRACBPVGeneric.Semantics.RunMachine as RunMachine

-- shapes, positions and scopes of the derived polynomial
module P = SignatureNotation (PM.polynomial (noLocks TerminalCategory) errorOps)

------------------------------------------------------------------------
-- the runner

isSetLk : isSet ⊥
isSetLk = isProp→isSet isProp⊥

isSetBase : isSet NoBase
isSetBase = isProp→isSet λ ()

-- states and outcomes: one each
S : Unit* {ℓ-zero} → Type
S _ = Unit

isSetConf : isSet (Σ (Unit* {ℓ-zero}) S)
isSetConf = isSetΣ isSetUnit* λ _ → isSetUnit

-- raise halts
δ : ∀ {w} → S w → (o : P.Shape w) → Unit ⊎ (Σ[ p ∈ P.Position w o ] S (P.scope w o p))
δ _ (raise ,Σ tt) = inl tt

------------------------------------------------------------------------
-- runs

module _ {A A' : ValTy} where
  open RunMachine (noLocks TerminalCategory) errorOps isSetLk isSetBase S Unit isSetConf isSetUnit δ A

  -- raise to N: push the frame into raise's continuations (none), close,
  -- and halt
  raise-to : (N : Comp (∅ tt* , A') (F A)) → running tt (raiseC to N) ⟶* halted tt
  raise-to N = pure-at (step hole (op-to raise []ᵥ)) then op-at

  -- the runner halts on the program's denotation.  (Not by refl: ⟦ to ⟧
  -- is the opaque bind; machine soundness supplies the equation.)
  raise-to-halts : (N : Comp (∅ tt* , A') (F A)) →
    run tt (denClosed (raiseC to N)) ≡ Rn.halted ⟦ A ⟧v tt
  raise-to-halts N = run-halt (raise-to N)
