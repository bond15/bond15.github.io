{-# OPTIONS --no-allow-unsolved-metas --no-lossy-unification #-}

module RightModuleReduction.IndexedContainer where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels
open import Cubical.Data.Sigma using (ΣPathP)
import Cubical.Data.Empty as Empty
open import Cubical.Relation.Nullary using (Discrete; yes; no; ¬_; isProp¬)
open import Cubical.Categories.Category using (Category; _^op)
open import Cubical.Categories.Functor using (Functor; _⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Instances.Sets using (SET)
open import Cubical.Categories.Presheaf.Base using (PresheafCategory)

open Category
open Functor
open NatTrans

-- A Set-valued container whose positions specify input worlds.
-- Decidable equality suffices for filling a selected position; finiteness
-- and transport of shapes along output-world maps are not required.
record IndexedContainer (I : Type) : Type₁ where
  field
    Shape : hSet ℓ-zero
    Pos : Shape .fst → hSet ℓ-zero
    decPos : (s : Shape .fst) → Discrete (Pos s .fst)
    input : (s : Shape .fst) → (p : Pos s .fst) → I

module _ {W : Category ℓ-zero ℓ-zero} (Q′ : IndexedContainer (W .ob)) where
  open IndexedContainer Q′

  private
    𝒱 = PresheafCategory (W ^op) ℓ-zero

  Extension : Functor 𝒱 (SET ℓ-zero)
  Extension .F-ob A =
    (Σ[ s ∈ Shape .fst ] ((p : Pos s .fst) → (A ⟅ input s p ⟆) .fst)) ,
    isSetΣ (Shape .snd) (λ s → isSetΠ λ p → (A ⟅ input s p ⟆) .snd)
  Extension .F-hom f (s , a) = s , λ p → f .N-ob (input s p) (a p)
  Extension .F-id = refl
  Extension .F-seq f g = refl

  -- A shape, a selected position, and fillers for every other position.
  Contexts : Functor 𝒱 (SET ℓ-zero)
  Contexts .F-ob A =
    (Σ (Σ[ s ∈ Shape .fst ] Pos s .fst) λ { (s , p) →
      (r : Pos s .fst) → ¬ r ≡ p → (A ⟅ input s r ⟆) .fst }) ,
    isSetΣ (isSetΣ (Shape .snd) (λ s → Pos s .snd))
      (λ { (s , p) → isSetΠ λ r → isSetΠ λ _ → (A ⟅ input s r ⟆) .snd })
  Contexts .F-hom f ((s , p) , a) =
    (s , p) , λ r r≠p → f .N-ob (input s r) (a r r≠p)
  Contexts .F-id = refl
  Contexts .F-seq f g = refl

  private
    fillPosition : (A : 𝒱 .ob) (s : Shape .fst) (p : Pos s .fst)
      (a : (r : Pos s .fst) → ¬ r ≡ p → (A ⟅ input s r ⟆) .fst) →
      (A ⟅ input s p ⟆) .fst → (r : Pos s .fst) → (A ⟅ input s r ⟆) .fst
    fillPosition A s p a x r with decPos s r p
    ... | yes e = subst (λ r → (A ⟅ input s r ⟆) .fst) (sym e) x
    ... | no r≠p = a r r≠p

    map-subst : {B : Type} (X Y : B → Type) (f : (b : B) → X b → Y b)
      {b c : B} (e : b ≡ c) (x : X b) →
      f c (subst X e x) ≡ subst Y e (f b x)
    map-subst X Y f {b} e x = J
      (λ c e → f c (subst X e x) ≡ subst Y e (f b x))
      (cong (f b) (substRefl {B = X} x) ∙ sym (substRefl {B = Y} (f b x))) e

    fillPosition-map : (A B : 𝒱 .ob) (f : NatTrans A B)
      (s : Shape .fst) (p : Pos s .fst)
      (a : (r : Pos s .fst) → ¬ r ≡ p → (A ⟅ input s r ⟆) .fst)
      (x : (A ⟅ input s p ⟆) .fst) (r : Pos s .fst) →
      f .N-ob (input s r) (fillPosition A s p a x r) ≡
      fillPosition B s p (λ r r≠p → f .N-ob (input s r) (a r r≠p))
        (f .N-ob (input s p) x) r
    fillPosition-map A B f s p a x r with decPos s r p
    ... | yes e = map-subst
      (λ r → (A ⟅ input s r ⟆) .fst)
      (λ r → (B ⟅ input s r ⟆) .fst)
      (λ r → f .N-ob (input s r)) (sym e) x
    ... | no r≠p = refl

  plug : (A : 𝒱 .ob) (C@((s , p) , a) : (Contexts ⟅ A ⟆) .fst) →
    (A ⟅ input s p ⟆) .fst → (Extension ⟅ A ⟆) .fst
  plug A ((s , p) , a) x = s , fillPosition A s p a x

  plug-map : (A : 𝒱 .ob) {B : 𝒱 .ob} (f : NatTrans A B)
    (C@((s , p) , a) : (Contexts ⟅ A ⟆) .fst)
    (x : (A ⟅ input s p ⟆) .fst) →
    (Extension ⟪ f ⟫) (plug A C x) ≡
    plug B ((Contexts ⟪ f ⟫) C) (f .N-ob (input s p) x)
  plug-map A {B} f ((s , p) , a) x =
    ΣPathP (refl , funExt (fillPosition-map A B f s p a x))

  plug-hole : (A : 𝒱 .ob) (C@((s , p) , a) : (Contexts ⟅ A ⟆) .fst)
    (x : (A ⟅ input s p ⟆) .fst) → plug A C x .snd p ≡ x
  plug-hole A ((s , p) , a) x with decPos s p p
  ... | yes e = cong (λ e → subst (λ r → (A ⟅ input s r ⟆) .fst) (sym e) x)
      (Pos s .snd p p e refl) ∙ substRefl {B = λ r → (A ⟅ input s r ⟆) .fst} x
  ... | no ne = Empty.rec (ne refl)

  plug-remaining : (A : 𝒱 .ob) (C@((s , p) , a) : (Contexts ⟅ A ⟆) .fst)
    (x : (A ⟅ input s p ⟆) .fst) (r : Pos s .fst) (ne : ¬ r ≡ p) →
    plug A C x .snd r ≡ a r ne
  plug-remaining A ((s , p) , a) x r ne with decPos s r p
  ... | yes e = Empty.rec (ne e)
  ... | no ne′ = cong (a r) (isProp¬ _ ne′ ne)
