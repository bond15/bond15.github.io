-- Canonicity of tree reduction (POPL's TreeCanonicity, with its
-- correction C8), for the free-model denotation.
--
-- C8 (as POPL): no bijection ⟦ · ⟧_A is needed, and reify ⟦ M ⟧ does not
-- type-check when the theory has equations (⟦ M ⟧ lives in the free model
-- T ⟦ A ⟧, reify takes an operation tree).  Termination and tree soundness
-- give, for every A, a tree that M reduces to and whose image in the free
-- model is ⟦ M ⟧:
--
--   tree-canonicity    on a lock telescope,
--                      Σ t. (M ↦* reify t) × (⌊ t ⌋ ≡ ⟦ M ⟧)
--   tree-canonicity₀   at a root, with the closed denotation (⌊ t ⌋₀)
--
-- ⌊_⌋ and den-reify are EqCanonicity's; termination is stage B's
-- (Metatheory.Termination); tree soundness for the free-model denotation
-- is Semantics.Free.Soundness (sound↦*).  The free model is fm, for a
-- theory whose models are closed under exponentials (ec).
open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Data.Sigma using (Σ-syntax; _×_)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Presheaf.Morphism.Alt using (PshHom)

open import RightModuleReduction.Model using (FreeModel)

open import RMRCanonicity.PRACBPVGeneric.Signature
import RMRCanonicity.PRACBPVGeneric.Semantics.Machine as Machine
import RMRCanonicity.PRACBPVGeneric.Semantics.StrongTheory as StrongTheory

module RMRCanonicity.PRACBPVGeneric.Semantics.TreeCanonicity
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (eqs : Machine.Equations L O) (fm : FreeModel (Machine.theory L O eqs))
  (ec : StrongTheory.ExpClosed L O eqs) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O using (F; ∅; Comp)
open import RMRCanonicity.PRACBPVGeneric.Reduction L O using (_↦*_)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Free.Denotation L O eqs fm ec using (⟦_⟧C)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Free.Soundness L O eqs fm ec using (sound↦*)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Free.Closed L O eqs fm ec using (denClosed)
open import RMRCanonicity.PRACBPVGeneric.Semantics.EqCanonicity L O eqs fm ec using (⌊_⌋; ⌊_⌋₀; den-reify)
open import RMRCanonicity.PRACBPVGeneric.Metatheory.LogicalRelation L O using (IsLT; lt∅)
open import RMRCanonicity.PRACBPVGeneric.Metatheory.Termination L O using (OpTree; reify; termination)

open PshHom

private
  module W = Category W

-- tree reduction reaches an operation tree whose image is the denotation
tree-canonicity : ∀ {Θ A} → IsLT Θ → (M : Comp Θ (F A)) →
  Σ[ t ∈ OpTree A Θ ] (M ↦* reify t) × (⌊ t ⌋ ≡ ⟦ M ⟧C)
tree-canonicity l M =
  termination l M .fst ,
  termination l M .snd ,
  sym (den-reify (termination l M .fst)) ∙ sym (sound↦* (termination l M .snd))

-- at a root: the closed denotation
tree-canonicity₀ : ∀ {w A} (M : Comp (∅ w) (F A)) →
  Σ[ t ∈ OpTree A (∅ w) ] (M ↦* reify t) × (⌊ t ⌋₀ ≡ denClosed M)
tree-canonicity₀ {w} M =
  tree-canonicity lt∅ M .fst ,
  tree-canonicity lt∅ M .snd .fst ,
  cong (λ φ → φ .N-ob w W.id) (tree-canonicity lt∅ M .snd .snd)
