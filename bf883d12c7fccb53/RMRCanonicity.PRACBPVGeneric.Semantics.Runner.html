-- Deterministic runners: effect data for an equation-free theory
-- (math spec §10, Thm 10.1).
--
-- For an equation-free theory with signature P, the free-model monad is
-- the term monad T X = terms over X (Semantics.FreeModel): η = var and
-- μ = evaluate id.  A runner is
--
--   S w       the states at a world w
--   H         the halting outcomes
--   δ σ o     at a state σ ∈ S w and an operation node o ∈ Shape w,
--             either halt with h ∈ H, or pick a position p of o and the
--             next state, at the scope of p
--
-- From a runner:
--
--   RunQ′          the configuration container.  A running shape (w , σ)
--                  has one hole, at world w; a halted shape h has none:
--                    Q X = Σ_w S w × X w  +  H
--   run σ t        run a term t ∈ T X w from σ: a variable is a running
--                  configuration; at an operation node δ decides whether
--                  to halt or to go on in the chosen child, at the new
--                  state (continue)
--   next σ o k     one node with children k in X: halted, or running the
--                  chosen child at the new state
--   ρ              ρ_X ((w , σ) , t) = run σ t,  ρ_X h = h
--   runnerModule   (RunQ′ , ρ) is a right T-module (Thm 10.1): naturality,
--                  unit and associativity, each by induction on terms
--   effect-at      RMR's effect step at a running configuration whose
--                  hole holds an operation of any P-algebra: it steps to
--                  next σ o k, the configuration δ chooses
--
-- The module is natural in X, not in worlds, so the polynomial's
-- reindexing (pos-is / smap-is) never enters.  δ's next state is typed at
-- the chosen child's scope, so nothing is transported either.
open import Cubical.Foundations.Prelude
open import Cubical.Data.Empty using (⊥)
import Cubical.Data.Empty as ⊥
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.Sum using (_⊎_; inl; inr; isSet⊎)
open import Cubical.Data.Sigma using (ΣPathP; Σ-syntax)
open import Cubical.Relation.Nullary using (yes)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Functor using (Functor; _∘F_; _⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Monad.Base using (IsMonad)

open import RightModuleReduction.Theory
  using (Theory; Algebra; Term; var; ops; module SignatureNotation)
open import RightModuleReduction.IndexedContainer using (IndexedContainer; plug; plug-hole)
open import RightModuleReduction.RightModule using (RightModule)
import RightModuleReduction.EffectReduction as EffectReduction

import RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel as FM

module RMRCanonicity.PRACBPVGeneric.Semantics.Runner {W : Category ℓ-zero ℓ-zero}
  (𝒯 : Theory W) (noEq : Theory.Equation 𝒯 → ⊥)
  (S : Category.ob W → Type) (H : Type)
  (isSetConf : isSet (Σ (Category.ob W) S)) (isSetH : isSet H)
  (δ : ∀ {w} → S w → (o : SignatureNotation.Shape (Theory.signature 𝒯) w) →
       H ⊎ (Σ[ p ∈ SignatureNotation.Position (Theory.signature 𝒯) w o ]
              S (SignatureNotation.scope (Theory.signature 𝒯) w o p)))
  where

open import RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf W using (𝒱; El)
open SignatureNotation (Theory.signature 𝒯) using (Shape; Position; scope)

open NatTrans

private
  module W = Category W

------------------------------------------------------------------------
-- the configuration container

RunQ′ : IndexedContainer W.ob
RunQ′ .IndexedContainer.Shape = (Σ W.ob S ⊎ H) , isSet⊎ isSetConf isSetH
RunQ′ .IndexedContainer.Pos (inl _) = Unit , isSetUnit
RunQ′ .IndexedContainer.Pos (inr _) = ⊥ , isProp→isSet ⊥.isProp⊥
RunQ′ .IndexedContainer.decPos (inl _) tt tt = yes refl
RunQ′ .IndexedContainer.decPos (inr _) ()
RunQ′ .IndexedContainer.input (inl (w , _)) _ = w
RunQ′ .IndexedContainer.input (inr _) ()

-- RMR's effect reduction for the term monad and this container
module ER = EffectReduction 𝒯 (FM.freeModel 𝒯 noEq) RunQ′
open ER using (T; Q; ∂Q)

-- configurations over X
Conf : 𝒱 → Type
Conf X = (Q ⟅ X ⟆) .fst

-- a running configuration: a state at w and the content of the hole
running : (X : 𝒱) {w : W.ob} → S w → El X w → Conf X
running X {w} σ x = inl (w , σ) , λ _ → x

-- a halted configuration: an outcome, and no hole
halted : (X : 𝒱) → H → Conf X
halted X h = inr h , λ ()

------------------------------------------------------------------------
-- running a term

-- what δ decides at an operation node
Outcome : ∀ {w} → Shape w → Type
Outcome {w} o = H ⊎ (Σ[ p ∈ Position w o ] S (scope w o p))

-- halt, or go on at the chosen position with the new state
continue : (X : 𝒱) {w : W.ob} {o : Shape w} → Outcome o →
  ((p : Position w o) → S (scope w o p) → Conf X) → Conf X
continue X (inl h) k = halted X h
continue X (inr (p , σ')) k = k p σ'

-- run a term from a state (structural recursion on the term)
run : ∀ {X w} → S w → Term (Theory.signature 𝒯) X w → Conf X
run {X} σ (var x) = running X σ x
run {X} σ (ops o k) = continue X (δ σ o) λ p σ' → run σ' (k p)

-- the configuration after one operation node whose children are in X
next : (X : 𝒱) {w : W.ob} → S w → (o : Shape w) →
  ((p : Position w o) → El X (scope w o p)) → Conf X
next X σ o k = continue X (δ σ o) λ p σ' → running X σ' (k p)

-- maps of configurations that keep outcomes commute with continue
continue-map : ∀ {X Y : 𝒱} (g : Conf X → Conf Y) → (∀ h → halted Y h ≡ g (halted X h)) →
  ∀ {w} {o : Shape w} (r : Outcome o)
  {kX : (p : Position w o) → S (scope w o p) → Conf X}
  {kY : (p : Position w o) → S (scope w o p) → Conf Y} →
  (∀ p σ' → kY p σ' ≡ g (kX p σ')) → continue Y r kY ≡ g (continue X r kX)
continue-map g gh (inl h) e = gh h
continue-map g gh (inr (p , σ')) e = e p σ'

------------------------------------------------------------------------
-- the action ρ : Q ∘ T ⇒ Q

-- run the term in the hole; an outcome stays
act : (X : 𝒱) → Conf (T .fst ⟅ X ⟆) → Conf X
act X (inl (w , σ) , t) = run σ (t tt)
act X (inr h , _) = halted X h

-- naturality: T α renames the variables and keeps the operation nodes
run-natural : ∀ {X Y : 𝒱} (α : NatTrans X Y) {w} (σ : S w) (t : Term (Theory.signature 𝒯) X w) →
  run σ ((T .fst ⟪ α ⟫) .N-ob w t) ≡ (Q ⟪ α ⟫) (run σ t)
run-natural α σ (var x) = refl
run-natural {X} {Y} α σ (ops o k) =
  continue-map {X} {Y} (Q ⟪ α ⟫) (λ h → ΣPathP (refl , funExt λ ())) (δ σ o)
    λ p σ' → run-natural α σ' (k p)

-- associativity: running a flattened term is running the outer term,
-- then the inner one where it stopped
run-assoc : ∀ {X : 𝒱} {w} (σ : S w) (t : Term (Theory.signature 𝒯) (T .fst ⟅ X ⟆) w) →
  run σ (IsMonad.μ (T .snd) .N-ob X .N-ob w t) ≡ act X (run σ t)
run-assoc σ (var u) = refl
run-assoc {X} σ (ops o k) =
  continue-map {T .fst ⟅ X ⟆} {X} (act X) (λ h → refl) (δ σ o) λ p σ' → run-assoc σ' (k p)

ρ : NatTrans (Q ∘F T .fst) Q
ρ .N-ob = act
ρ .N-hom α = funExt λ
  { (inl (w , σ) , t) → run-natural α σ (t tt)
  ; (inr h , t) → ΣPathP (refl , funExt λ ()) }

-- Thm 10.1: a deterministic runner is a right module of the term monad
runnerModule : RightModule T Q
runnerModule .RightModule.ρ = ρ
runnerModule .RightModule.ρ-unit X = funExt λ
  { (inl _ , x) → refl
  ; (inr h , x) → ΣPathP (refl , funExt λ ()) }
runnerModule .RightModule.ρ-assoc X = funExt λ
  { (inl (w , σ) , t) → run-assoc σ (t tt)
  ; (inr h , t) → refl }

------------------------------------------------------------------------
-- the effect step at a running configuration

-- the one-hole context of a running configuration
context : (X : 𝒱) {w : W.ob} → S w → (∂Q ⟅ X ⟆) .fst
context X {w} σ = (inl (w , σ) , tt) , λ r ne → ⊥.rec (ne refl)

plug-context : (X : 𝒱) {w : W.ob} (σ : S w) (x : El X w) →
  plug RunQ′ X (context X σ) x ≡ running X σ x
plug-context X σ x = ΣPathP (refl , funExt λ _ → plug-hole RunQ′ X (context X σ) x)

plug-mapped-context : {X Y : 𝒱} (f : NatTrans X Y) {w : W.ob} (σ : S w) (y : El Y w) →
  plug RunQ′ Y ((∂Q ⟪ f ⟫) (context X σ)) y ≡ running Y σ y
plug-mapped-context {X} {Y} f σ y =
  ΣPathP (refl , funExt λ _ → plug-hole RunQ′ Y ((∂Q ⟪ f ⟫) (context X σ)) y)

-- RMR's effect step, for any P-algebra in the hole: the node's operation
-- steps to the configuration δ chooses.  (The step lifts the children
-- with η = var, so ρ runs one node: run σ (ops o (var ∘ k)) = next σ o k.)
effect-at : (alg : Algebra (Theory.signature 𝒯)) {w : W.ob} (σ : S w) (o : Shape w)
  (k : (p : Position w o) → El (Algebra.Carrier alg) (scope w o p)) →
  ER._↦_ runnerModule alg
    (running (Algebra.Carrier alg) σ (Algebra.algebra alg .N-ob w (o , k)))
    (next (Algebra.Carrier alg) σ o k)
effect-at alg {w} σ o k =
  subst2 (ER._↦_ runnerModule alg)
    (plug-context (Algebra.Carrier alg) σ (Algebra.algebra alg .N-ob w (o , k)))
    (cong (act (Algebra.Carrier alg))
      (plug-mapped-context (IsMonad.η (T .snd) .N-ob (Algebra.Carrier alg)) σ
        (ops o λ p → var (k p))))
    (ER.effect-step (context (Algebra.Carrier alg) σ) (o , k))
