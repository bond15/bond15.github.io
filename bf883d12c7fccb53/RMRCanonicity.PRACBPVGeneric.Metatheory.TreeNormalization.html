-- Strong normalization of tree reduction (POPL's TreeNormalization,
-- strong-normalization; its D9: stated constructively, as accessibility
-- for the converse of ↦).
--
-- On a lock telescope every computation of type F A has a derivation of
-- the logical relation (Termination.related): a chain of tree steps (back)
-- ending in return or in an op.  return and op do not step
-- (Determinism.ret-normal, op-normal), and a computation has at most one
-- step (Determinism.↦-det), so every step from M is the derivation's first
-- one, and accessibility follows by induction on the derivation (tree-acc).
--
-- POPL derives the same statement from a decreasing metric (C7) because
-- its tree reduction also reduces under op (opArg).  Here ↦ never goes
-- under an op (only ↦*'s op* does), so the chain of back steps is the
-- whole story and no metric is needed.
--
--   _↤_                    the converse of ↦
--   tree-acc               a derivation of the logical relation gives Acc
--   strong-normalization   every computation of type F A on a lock
--                          telescope is accessible: there is no infinite
--                          ↦-sequence from it
open import Cubical.Foundations.Prelude hiding (Sub; _,_; comp)
open import Cubical.Data.Empty as ⊥ using ()
open import Cubical.Induction.WellFounded using (Acc; acc)
open import Cubical.Categories.Category using (Category)

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Metatheory.TreeNormalization
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
open import RMRCanonicity.PRACBPVGeneric.Reduction L O
open import RMRCanonicity.PRACBPVGeneric.Metatheory.Determinism L O using (↦-det; ret-normal; op-normal)
open import RMRCanonicity.PRACBPVGeneric.Metatheory.LogicalRelation L O
  using (IsLT; Tree; leaf; node; back)
open import RMRCanonicity.PRACBPVGeneric.Metatheory.Termination L O using (related)

infix 4 _↤_

-- the converse of tree reduction: M' ↤ M iff M ↦ M'
_↤_ : ∀ {Θ B} → Comp Θ B → Comp Θ B → Type
M' ↤ M = M ↦ M'

-- a derivation of the logical relation (for any relation at the leaves)
-- makes its computation accessible
tree-acc : ∀ {A Θ} {R : (Θ : Tele) → Val Θ A → Type} {M : Comp Θ (F A)} →
  Tree R Θ M → Acc _↤_ M
tree-acc (leaf _) = acc λ _ s → ⊥.rec (ret-normal s)
tree-acc (node o _) = acc λ _ s → ⊥.rec (op-normal o s)
tree-acc (back s t) = acc λ _ s' → subst (Acc _↤_) (↦-det s s') (tree-acc t)

-- strong normalization of tree reduction
strong-normalization : ∀ {Θ A} → IsLT Θ → (M : Comp Θ (F A)) → Acc _↤_ M
strong-normalization l M = tree-acc (related l M)
