-- The denotational semantics of PRACBPV, CANONICAL version: a telescope
-- denotes a presheaf over the representable of its world, and a lock is
-- literally a summand of a left adjoint.
--
--   Types               ⟦ 0 ⟧ = 0 (initial)   ⟦ A + A' ⟧ = +   (pointwise)
--                       ⟦ 1 ⟧ = 1   ⟦ A × A' ⟧ = ×   ⟦ base b ⟧ = Const b
--                       ⟦ [ κ ] A ⟧ = R_κ ⟦ A ⟧                 (right adjoint)
--                       ⟦ U B ⟧ = carrier of ⟦ B ⟧
--                       ⟦ F A ⟧ = T ⟦ A ⟧, the free (term) algebra
--                       ⟦ A ⇒ B ⟧ = ⟦ A ⟧ ⇒ ⟦ B ⟧, the exponential algebra (strength)
--   Telescopes          ⟦ Θ ⟧ ∈ 𝒱 with  π_Θ : ⟦ Θ ⟧ → y (world Θ)
--                       ⟦ ∅ w ⟧ = y w
--                       ⟦ Θ , A ⟧ = ⟦ Θ ⟧ × ⟦ A ⟧
--                       ⟦ Θ ⧀⟨ κ , p ⟩ ⟧ = the p-summand of L_κ ⟦ Θ ⟧
--                          (pullback of L_κ ⟦ Θ ⟧ → L_κ y w ≅ E_κ w along the p-th injection)
--   Terms               maps out of ⟦ Θ ⟧
--   Neutrals out of a lock: each is (inclusion of the summand) ; unlock φ,
--                       the transpose of a map φ : ⟦ Δ ⟧ → R_κ X:
--                         up n   φ = ⟦ n ⟧ ; box     (the strength)
--                         opn n  φ = ⟦ n ⟧           (the counit)
--                         nm k   φ = the generic elements gen κ
--   inl, inr            the coproduct injections
--   case₊ V M N         the copairing of ⟦ M ⟧ and ⟦ N ⟧ in context (casePsh),
--                       at ⟨ id , ⟦ V ⟧ ⟩
--   absurd V            ⟦ V ⟧ followed by the map out of the initial presheaf
--   shut V              the transpose of the copairing of the ⟦ V p ⟧
--   op o vs k           the algebra of ⟦ B ⟧ at (o , ⟦ vs ⟧), the continuations
--                       transposed through their lock words (shutW)

module RMRCanonicity.PRACBPVGeneric.Semantics.Denotation
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O as Syn
  using ( Lk; Base; Const; Cs; Op; params; Ar; slot; names; gen; pick; _∋_; ∋z; ∋s
        ; wordE; wordStr; WPos; wscope
        ; ValTy; CompTy; 𝟘; 𝟙; _⊕_; U; [_]_; base; F; _⇒_
        ; Tele; ∅; _⧀⟨_,_⟩; _⧀*⟨_,_⟩; world
        ; Ne; here; wkv; up; opn; nm
        ; Val; Vals; Comp; Stack )
  renaming (_,_ to _▸_)
-- The polynomial is always written PM.polynomial L O.  A module copy
-- (open import … L O using (polynomial)) or an abbreviation (P = …) is not
-- syntactically the same term, and comparing the two unfolds the
-- Grothendieck construction: `Ext X ≡ Poly.Extension P ⟅ X ⟆` by refl did
-- not finish in 60 s, against 5 s for the whole module without it.
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
open PM L O using (pos-is; smap-is)

open import RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf W public
open import RMRCanonicity.PRACBPVGeneric.Semantics.Algebra (PM.polynomial L O) (PM.strength L O) public

open Functor
open PshHom

private
  module W = Category W

-- the lock of each lock kind, and of each lock word
module K (κ : Lk) = RMRCanonicity.PRACBPVGeneric.Semantics.Lock (Syn.E κ) (LockKind.str (LockSig.kind L κ))
module KW (μ : List Lk) = RMRCanonicity.PRACBPVGeneric.Semantics.Lock (wordE μ) (wordStr μ)

------------------------------------------------------------------------
-- parameters: the presheaves of constants

Cst : Base → 𝒱
Cst b = fromFun (Const b)

Par : List Base → 𝒱
Par bs = fromFun (CstsF W Const bs)

-- picking a constant commutes with moving along a world map
pick-cmaps : ∀ {bs b u v} (k : bs ∋ b) (f : W [ u , v ]) (cs : Cs u bs) →
  pick k (cmaps W Const bs f cs) ≡ Const b .F-hom f (pick k cs)
pick-cmaps ∋z f cs = refl
pick-cmaps (∋s k) f cs = pick-cmaps k f (cs .snd)

------------------------------------------------------------------------
-- types

⟦_⟧v : ValTy → 𝒱
⟦_⟧c : CompTy → Algebra (PM.polynomial L O)

⟦ 𝟘 ⟧v = EmptyPsh
⟦ 𝟙 ⟧v = UnitPsh
⟦ A ⊕ A' ⟧v = ⟦ A ⟧v ⊎Psh ⟦ A' ⟧v
⟦ A Syn.× A' ⟧v = ⟦ A ⟧v ×Psh ⟦ A' ⟧v
⟦ U B ⟧v = Algebra.Carrier ⟦ B ⟧c
⟦ [ κ ] A ⟧v = K.R κ ⟦ A ⟧v
⟦ base b ⟧v = Cst b

⟦ F A ⟧c = TermAlg (PM.polynomial L O) ⟦ A ⟧v
⟦ A ⇒ B ⟧c = expAlg ⟦ A ⟧v ⟦ B ⟧c

⟦_⟧u : CompTy → 𝒱
⟦ B ⟧u = Algebra.Carrier ⟦ B ⟧c

------------------------------------------------------------------------
-- telescopes: a presheaf over the representable of the world

⟦_⟧t : Tele → 𝒱
πt : (Θ : Tele) → PshHom ⟦ Θ ⟧t (Y (world Θ))

⟦ ∅ w ⟧t = Y w
⟦ Θ ▸ A ⟧t = ⟦ Θ ⟧t ×Psh ⟦ A ⟧v
⟦ Θ ⧀⟨ κ , p ⟩ ⟧t = K.Over.Summand κ (πt Θ) p

πt (∅ w) = idPshHom
πt (Θ ▸ A) = π₁ _ _ ⋆PshHom πt Θ
πt (Θ ⧀⟨ κ , p ⟩) = K.Over.πS κ (πt Θ) p

-- the summand inclusion of a lock
incl : ∀ {Δ κ p} → PshHom ⟦ Δ ⧀⟨ κ , p ⟩ ⟧t (K.L κ ⟦ Δ ⟧t)
incl {Δ} {κ} {p} = K.Over.inc κ (πt Δ) p

------------------------------------------------------------------------
-- neutrals

-- the generic elements: the names a κ-lock binds, as a map into R_κ
namesR : ∀ {Γ : 𝒱} (κ : Lk) {b} → names κ ∋ b → PshHom Γ (K.R κ (Cst b))
namesR κ k .N-ob v e q = pick k (gen κ v q)
namesR {Γ} κ {b} k .N-hom c c' f e = funExt λ q →
  cong (pick k) (sym (funExt⁻ (Global.nat (LockKind.gen (LockSig.kind L κ)) f) q))
  ∙ pick-cmaps k (K.smap κ f q) _

⟦_⟧ne : ∀ {Θ A} → Ne Θ A → PshHom ⟦ Θ ⟧t ⟦ A ⟧v
⟦ here ⟧ne = π₂ _ _
⟦ wkv n ⟧ne = π₁ _ _ ⋆PshHom ⟦ n ⟧ne
⟦ up {Δ} {κ} {p} {A} n ⟧ne = incl {Δ} {κ} {p} ⋆PshHom K.unlock κ (⟦ n ⟧ne ⋆PshHom K.box κ ⟦ A ⟧v)
⟦ opn {Δ} {κ} {p} n ⟧ne = incl {Δ} {κ} {p} ⋆PshHom K.unlock κ ⟦ n ⟧ne
⟦ nm {Δ} {κ} {p} k ⟧ne = incl {Δ} {κ} {p} ⋆PshHom K.unlock κ (namesR κ k)

------------------------------------------------------------------------
-- continuations behind a lock word: R_μ's introduction rule, iterated

-- R_{[]} X ≅ X  and  R_κ (R_μ X) ≅ R_{κ ⊙ μ} X, both identities on elements
unitR : ∀ {X : 𝒱} → PshHom X (KW.R [] X)
unitR .N-ob v x _ = x
unitR .N-hom c c' f x = refl

uncurryR : ∀ {X : 𝒱} (κ κ' : Lk) (μ : List Lk) →
  PshHom (K.R κ (KW.R (κ' ∷ μ) X)) (KW.R (κ ∷ κ' ∷ μ) X)
uncurryR κ κ' μ .N-ob v r (p , ps) = r p ps
uncurryR κ κ' μ .N-hom c c' f r = refl

shutW : ∀ {Θ} {X : 𝒱} (μ : List Lk) →
  ((ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X) → PshHom ⟦ Θ ⟧t (KW.R μ X)
shutW [] k = k tt ⋆PshHom unitR
shutW {Θ} (κ ∷ []) k = K.Over.shut κ (πt Θ) k
shutW {Θ} (κ ∷ μ@(κ' ∷ μ')) k =
  K.Over.shut κ (πt Θ) (λ p → shutW {Θ ⧀⟨ κ , p ⟩} μ (λ ps → k (p , ps))) ⋆PshHom uncurryR κ κ' μ'

------------------------------------------------------------------------
-- an operation node: shape (o , parameters) and continuations

opNode : ∀ {Γ X : 𝒱} (o : Op) → PshHom Γ (Par (params o)) →
  ((a : Ar o) → PshHom Γ (KW.R (slot o a) X)) → PshHom Γ (Ext X)
opNode o vs ks .N-ob v e = (o , vs .N-ob v e) , λ (a , ps) → ks a .N-ob v e ps
opNode {Γ} {X} o vs ks .N-hom c c' f e =
  ΣPathP (ΣPathP (refl , vs .N-hom c c' f e) , funExt λ (a , ps) →
    funExt⁻ (ks a .N-hom c c' f e) ps
    ∙ sym (λ i → X .F-hom (smap-is f o (vs .N-ob c' e) (a , ps) i)
                    (ks _ .N-ob c' e (pos-is f o (vs .N-ob c' e) (a , ps) i .snd))))

-- the denotation of op o vs k, from those of vs and of the continuations
opDen : ∀ {Θ} (B : CompTy) (o : Op) → PshHom ⟦ Θ ⟧t (Par (params o)) →
  ((a : Ar o) (ps : WPos (slot o a) (world Θ)) → PshHom ⟦ Θ ⧀*⟨ slot o a , ps ⟩ ⟧t ⟦ B ⟧u) →
  PshHom ⟦ Θ ⟧t ⟦ B ⟧u
opDen {Θ} B o vs ks =
  opNode o vs (λ a → shutW {Θ} (slot o a) (ks a)) ⋆PshHom NatTrans→PshHom (Algebra.algebra ⟦ B ⟧c)

------------------------------------------------------------------------
-- values and computations: maps out of the telescope

⟦_⟧V : ∀ {Θ A} → Val Θ A → PshHom ⟦ Θ ⟧t ⟦ A ⟧v
⟦_⟧Vs : ∀ {Θ bs} → Vals Θ bs → PshHom ⟦ Θ ⟧t (Par bs)
⟦_⟧C : ∀ {Θ B} → Comp Θ B → PshHom ⟦ Θ ⟧t ⟦ B ⟧u

⟦ Syn.ne n ⟧V = ⟦ n ⟧ne
⟦ Syn.tt ⟧V = UnitPsh-intro
⟦ Syn.inl {A} {A'} V ⟧V = ⟦ V ⟧V ⋆PshHom inlPsh ⟦ A ⟧v ⟦ A' ⟧v
⟦ Syn.inr {A} {A'} V ⟧V = ⟦ V ⟧V ⋆PshHom inrPsh ⟦ A ⟧v ⟦ A' ⟧v
⟦ Syn.pair V V' ⟧V = ×PshIntro ⟦ V ⟧V ⟦ V' ⟧V
⟦ Syn.thunk M ⟧V = ⟦ M ⟧C
-- a constant at the world: through the telescope's world, by Yoneda
⟦_⟧V {Θ} (Syn.const {b} c) = πt Θ ⋆PshHom yoneda (Cst b) c
-- R_κ's introduction: the transpose of the copairing of the summands
⟦_⟧V {Θ} (Syn.shut {κ} V) = K.Over.shut κ (πt Θ) (λ p → ⟦ V p ⟧V)

⟦ Syn.[]ᵥ ⟧Vs .N-ob v e = tt
⟦ Syn.[]ᵥ ⟧Vs .N-hom c c' f e = refl
⟦ V Syn.∷ᵥ vs ⟧Vs .N-ob v e = ⟦ V ⟧V .N-ob v e , ⟦ vs ⟧Vs .N-ob v e
⟦ V Syn.∷ᵥ vs ⟧Vs .N-hom c c' f e = ΣPathP (⟦ V ⟧V .N-hom c c' f e , ⟦ vs ⟧Vs .N-hom c c' f e)

⟦ Syn.return V ⟧C = ⟦ V ⟧V ⋆PshHom η _
⟦_⟧C {Θ} (Syn._to_ {A} {B} M N) =
  ×PshIntro ⟦ M ⟧C idPshHom ⋆PshHom bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧c ⟦ N ⟧C
⟦ Syn.force V ⟧C = ⟦ V ⟧V
⟦ Syn.case V M ⟧C =
  ×PshIntro (×PshIntro idPshHom (⟦ V ⟧V ⋆PshHom π₁ _ _)) (⟦ V ⟧V ⋆PshHom π₂ _ _) ⋆PshHom ⟦ M ⟧C
⟦ Syn.case₊ V M N ⟧C = ×PshIntro idPshHom ⟦ V ⟧V ⋆PshHom casePsh ⟦ M ⟧C ⟦ N ⟧C
⟦ Syn.absurd {B} V ⟧C = ⟦ V ⟧V ⋆PshHom EmptyPsh-rec ⟦ B ⟧u
⟦ Syn.lam {A = A} {B} M ⟧C = λExp ⟦ A ⟧v ⟦ B ⟧u ⟦ M ⟧C
⟦ Syn.app {A = A} {B} M V ⟧C = ×PshIntro ⟦ M ⟧C ⟦ V ⟧V ⋆PshHom appExp ⟦ A ⟧v ⟦ B ⟧u
-- the algebra of ⟦ B ⟧ at the shape (o , ⟦ vs ⟧), each continuation
-- transposed through its slot's lock word
⟦_⟧C {Θ} (Syn.op {B} o vs k) = opDen B o ⟦ vs ⟧Vs (λ a ps → ⟦ k a ps ⟧C)

------------------------------------------------------------------------
-- stacks: maps U ⟦ B ⟧ × ⟦ Θ ⟧ → U ⟦ B' ⟧ (as in the old core)

⟦_⟧S : ∀ {Θ B B'} → Stack Θ B B' → PshHom (⟦ B ⟧u ×Psh ⟦ Θ ⟧t) ⟦ B' ⟧u
⟦ Syn.hole ⟧S = π₁ _ _
⟦_⟧S {Θ} (Syn._to•_ {A} {B' = B'} S N) =
  ×PshIntro ⟦ S ⟧S (π₂ _ _) ⋆PshHom bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B' ⟧c ⟦ N ⟧C
⟦ Syn.app• {A = A} {B' = B'} S V ⟧S =
  ×PshIntro ⟦ S ⟧S (π₂ _ _ ⋆PshHom ⟦ V ⟧V) ⋆PshHom appExp ⟦ A ⟧v ⟦ B' ⟧u
