-- The term monad as a right module over itself: configurations are terms
-- (Q = T, the POPL setting), for an equation-free theory.
--
-- Data: an equation-free theory with signature P (so the free-model monad
-- is T X = terms over X, Semantics.FreeModel), worlds forming a set, and
-- operations with discrete positions.
--
-- THE CONTAINER.  RMR's configurations are an indexed container
-- Q X = Σ_s Π_{r ∈ Pos s} X (input s r).  Terms are one:
--
--   shapes      Σ_w Term P 1 w        a world and a term with no data at
--                                     its variables (1 the terminal
--                                     presheaf)
--   positions   Leaf t                the leaves (variable occurrences) of t
--   input       leafWorld t l         the world of the leaf l
--
-- so Q X ≅ Σ_w T X w: a term splits into its shape and the variables at
-- its leaves (fromTerm, with inverse toTerm: confIso).  Positions are discrete
-- (discreteLeaf), and finite when the operations' positions are
-- (finLeaf).
--
-- THE MODULE.  ρ = μ: a configuration of terms is a term whose leaves
-- hold terms, and ρ substitutes them (graft t f = μ (t with f at its
-- leaves); ρ-is-μ).  The module laws are the monad laws, read through the
-- isomorphism Q X ≅ Σ_w T X w (conf-ext: configurations are equal when
-- their terms are):
--
--   ρ natural    graft-map, assemble-map
--   ρ-unit       graft-unit
--   ρ-assoc      graft-assoc
--
-- AFFINITY (self-affine).  At the generic redex the hole p of a shape t
-- holds an operation node o whose children are the fresh generators
-- (inr q), and every other leaf r holds its own generator (inl r).
-- Grafting replaces each leaf by what it holds, so a leaf of the result
-- is a leaf l of t followed by a leaf of what l holds (split, a bijection
-- with inverse unsplit), and it holds that leaf's generator (split-at).
-- The generators are pairwise distinct, so the origins of the result's
-- holes are too.

module RMRCanonicity.PRACBPVGeneric.Semantics.SelfModule {W : Category ℓ-zero ℓ-zero}
  (𝒯 : Theory W) (noEq : Theory.Equation 𝒯 → ⊥)
  (isSetWorld : isSet (Category.ob W))
  (discretePos : ∀ w o → Discrete (SignatureNotation.Position (Theory.signature 𝒯) w o))
  where

open import RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf W using (𝒱; El)

open Functor
open NatTrans

private
  module W = Category W
  P = Theory.signature 𝒯
  open SignatureNotation P using (Shape; Position)

------------------------------------------------------------------------
-- shapes and leaves

-- the terminal presheaf
𝟏 : 𝒱
𝟏 .F-ob _ = Unit , isSetUnit
𝟏 .F-hom _ _ = tt
𝟏 .F-id = refl
𝟏 .F-seq _ _ = refl

-- the leaves of a term
Leaf : ∀ {w} → Term P 𝟏 w → Type
Leaf (var _) = Unit
Leaf {w} (ops o k) = Σ[ q ∈ Position w o ] Leaf (k q)

-- the world of a leaf
leafWorld : ∀ {w} (t : Term P 𝟏 w) → Leaf t → W.ob
leafWorld {w} (var _) _ = w
leafWorld (ops o k) (q , l) = leafWorld (k q) l

discreteLeaf : ∀ {w} (t : Term P 𝟏 w) → Discrete (Leaf t)
discreteLeaf (var _) _ _ = yes refl
discreteLeaf {w} (ops o k) = discreteΣ (discretePos w o) λ q → discreteLeaf (k q)

-- finitely many leaves, numbered, when operations have finitely many
-- positions
finLeaf : (∀ w o → isFinOrd (Position w o)) → ∀ {w} (t : Term P 𝟏 w) → isFinOrd (Leaf t)
finLeaf finPos (var _) = isFinOrdUnit
finLeaf finPos {w} (ops o k) = isFinOrdΣ _ (finPos w o) _ λ q → finLeaf finPos (k q)

------------------------------------------------------------------------
-- the configuration container

TermQ′ : IndexedContainer W.ob
TermQ′ .IndexedContainer.Shape = (Σ W.ob (Term P 𝟏)) , isSetΣ isSetWorld (λ w → TermPsh P 𝟏 .F-ob w .snd)
TermQ′ .IndexedContainer.Pos (w , t) = Leaf t , Discrete→isSet (discreteLeaf t)
TermQ′ .IndexedContainer.decPos (w , t) = discreteLeaf t
TermQ′ .IndexedContainer.input (w , t) = leafWorld t

-- RMR's effect reduction for the term monad and this container
module ER = EffectReduction 𝒯 (FM.freeModel 𝒯 noEq) TermQ′
open ER using (T; Q; ∂Q)

-- configurations over X
Conf : 𝒱 → Type
Conf X = (Q ⟅ X ⟆) .fst

------------------------------------------------------------------------
-- Q X ≅ Σ_w T X w

-- the shape of a term: its variables forgotten
shape : ∀ {X w} → Term P X w → Term P 𝟏 w
shape (var _) = var tt
shape (ops o k) = ops o λ q → shape (k q)

-- the variable at a leaf
leafVar : ∀ {X w} (u : Term P X w) (l : Leaf (shape u)) → El X (leafWorld (shape u) l)
leafVar (var x) _ = x
leafVar (ops o k) (q , l) = leafVar (k q) l

-- a shape with variables at its leaves
assemble : (X : 𝒱) {w : W.ob} (t : Term P 𝟏 w) → ((l : Leaf t) → El X (leafWorld t l)) → Term P X w
assemble X (var _) f = var (f tt)
assemble X (ops o k) f = ops o λ q → assemble X (k q) λ l → f (q , l)

fromTerm : (X : 𝒱) → Σ W.ob (Term P X) → Conf X
fromTerm X (w , u) = (w , shape u) , leafVar u

toTerm : (X : 𝒱) → Conf X → Σ W.ob (Term P X)
toTerm X ((w , t) , f) = w , assemble X t f

assemble-shape : ∀ {X w} (u : Term P X w) → assemble X (shape u) (leafVar u) ≡ u
assemble-shape (var x) = refl
assemble-shape (ops o k) = cong (ops o) (funExt λ q → assemble-shape (k q))

shape-assemble : (X : 𝒱) {w : W.ob} (t : Term P 𝟏 w) (f : (l : Leaf t) → El X (leafWorld t l)) →
  Path (Σ[ t′ ∈ Term P 𝟏 w ] ((l : Leaf t′) → El X (leafWorld t′ l)))
    (shape (assemble X t f) , leafVar (assemble X t f)) (t , f)
shape-assemble X (var _) f = refl
shape-assemble X (ops o k) f i =
  ops o (λ q → shape-assemble X (k q) (λ l → f (q , l)) i .fst) ,
  λ (q , l) → shape-assemble X (k q) (λ l → f (q , l)) i .snd l

to-from : (X : 𝒱) (x : Σ W.ob (Term P X)) → toTerm X (fromTerm X x) ≡ x
to-from X (w , u) = cong (w ,_) (assemble-shape u)

from-to : (X : 𝒱) (c : Conf X) → fromTerm X (toTerm X c) ≡ c
from-to X ((w , t) , f) i = (w , shape-assemble X t f i .fst) , shape-assemble X t f i .snd

-- Q X ≅ Σ_w T X w
confIso : (X : 𝒱) → Iso (Conf X) (Σ W.ob (Term P X))
confIso X = iso (toTerm X) (fromTerm X) (to-from X) (from-to X)

-- configurations are equal when their terms are
conf-ext : (X : 𝒱) {c₁ c₂ : Conf X} → toTerm X c₁ ≡ toTerm X c₂ → c₁ ≡ c₂
conf-ext X {c₁} {c₂} e = sym (from-to X c₁) ∙∙ cong (fromTerm X) e ∙∙ from-to X c₂

------------------------------------------------------------------------
-- the action ρ = μ

private
  T₀ = T .fst
  module Tm = IsMonad (T .snd)

-- substitute the term f l at each leaf l of t
graft : ∀ {X w} (t : Term P 𝟏 w) → ((l : Leaf t) → Term P X (leafWorld t l)) → Term P X w
graft {X} {w} t f = Tm.μ .N-ob X .N-ob w (assemble (T₀ ⟅ X ⟆) t f)

act : (X : 𝒱) → Conf (T₀ ⟅ X ⟆) → Conf X
act X ((w , t) , f) = fromTerm X (w , graft t f)

-- graft commutes with renaming the variables
graft-map : ∀ {X Y} (α : NatTrans X Y) {w} (t : Term P 𝟏 w)
  (f : (l : Leaf t) → Term P X (leafWorld t l)) →
  graft t (λ l → (T₀ ⟪ α ⟫) .N-ob _ (f l)) ≡ (T₀ ⟪ α ⟫) .N-ob w (graft t f)
graft-map α (var _) f = refl
graft-map α (ops o k) f = cong (ops o) (funExt λ q → graft-map α (k q) λ l → f (q , l))

-- renaming the variables at the leaves is renaming the term
assemble-map : ∀ {X Y} (α : NatTrans X Y) {w} (u : Term P X w) →
  assemble Y (shape u) (λ l → α .N-ob _ (leafVar u l)) ≡ (T₀ ⟪ α ⟫) .N-ob w u
assemble-map α (var x) = refl
assemble-map α (ops o k) = cong (ops o) (funExt λ q → assemble-map α (k q))

-- grafting variables changes nothing (μ ∘ T η = id)
graft-unit : (X : 𝒱) {w : W.ob} (t : Term P 𝟏 w) (f : (l : Leaf t) → El X (leafWorld t l)) →
  graft t (λ l → var (f l)) ≡ assemble X t f
graft-unit X (var _) f = refl
graft-unit X (ops o k) f = cong (ops o) (funExt λ q → graft-unit X (k q) λ l → f (q , l))

-- flattening the leaves first, or after (μ ∘ T μ = μ ∘ μ)
graft-assoc : (X : 𝒱) {w : W.ob} (t : Term P 𝟏 w) (f : (l : Leaf t) → Term P (T₀ ⟅ X ⟆) (leafWorld t l)) →
  graft t (λ l → Tm.μ .N-ob X .N-ob _ (f l)) ≡ Tm.μ .N-ob X .N-ob w (graft t f)
graft-assoc X (var _) f = refl
graft-assoc X (ops o k) f = cong (ops o) (funExt λ q → graft-assoc X (k q) λ l → f (q , l))

-- read through confIso, ρ is μ
ρ-is-μ : (X : 𝒱) {w : W.ob} (t : Term P 𝟏 w) (f : (l : Leaf t) → Term P X (leafWorld t l)) →
  toTerm X (act X ((w , t) , f)) ≡ (w , Tm.μ .N-ob X .N-ob w (assemble (T₀ ⟅ X ⟆) t f))
ρ-is-μ X t f = to-from X _

ρ : NatTrans (Q ∘F T₀) Q
ρ .N-ob = act
ρ .N-hom {X} {Y} α = funExt λ ((w , t) , f) → conf-ext Y
  (to-from Y _ ∙ cong (w ,_) (graft-map α t f ∙ sym (assemble-map α (graft t f))))

-- configurations of terms form a right T-module, with ρ = μ
selfModule : RightModule T Q
selfModule .RightModule.ρ = ρ
selfModule .RightModule.ρ-unit X = funExt λ ((w , t) , f) → conf-ext X
  (to-from X _ ∙ cong (w ,_) (graft-unit X t f))
selfModule .RightModule.ρ-assoc X = funExt λ ((w , t) , f) → conf-ext X
  (to-from X _
   ∙ cong (w ,_) (graft-assoc X t f ∙ cong (Tm.μ .N-ob X .N-ob w) (sym (assemble-shape (graft t f))))
   ∙ sym (to-from X _))

------------------------------------------------------------------------
-- the leaves of a graft

-- a leaf's world and variable
at : ∀ {X w} (u : Term P X w) → Leaf (shape u) → Σ W.ob (El X)
at u l = leafWorld (shape u) l , leafVar u l

module _ {X : 𝒱} where
  -- a leaf of graft t g is a leaf l of t, then a leaf of g l
  split : ∀ {w} (t : Term P 𝟏 w) (g : (l : Leaf t) → Term P X (leafWorld t l)) →
    Leaf (shape (graft t g)) → Σ[ l ∈ Leaf t ] Leaf (shape (g l))
  split (var _) g a = tt , a
  split (ops o k) g (q , a) =
    (q , split (k q) (λ l → g (q , l)) a .fst) , split (k q) (λ l → g (q , l)) a .snd

  unsplit : ∀ {w} (t : Term P 𝟏 w) (g : (l : Leaf t) → Term P X (leafWorld t l)) →
    Σ[ l ∈ Leaf t ] Leaf (shape (g l)) → Leaf (shape (graft t g))
  unsplit (var _) g (_ , b) = b
  unsplit (ops o k) g ((q , l) , b) = q , unsplit (k q) (λ l → g (q , l)) (l , b)

  unsplit-split : ∀ {w} (t : Term P 𝟏 w) (g : (l : Leaf t) → Term P X (leafWorld t l))
    (a : Leaf (shape (graft t g))) → unsplit t g (split t g a) ≡ a
  unsplit-split (var _) g a = refl
  unsplit-split (ops o k) g (q , a) = cong (q ,_) (unsplit-split (k q) (λ l → g (q , l)) a)

  -- and it holds what that leaf of g l holds
  split-at : ∀ {w} (t : Term P 𝟏 w) (g : (l : Leaf t) → Term P X (leafWorld t l))
    (a : Leaf (shape (graft t g))) → at (graft t g) a ≡ at (g (split t g a .fst)) (split t g a .snd)
  split-at (var _) g a = refl
  split-at (ops o k) g (q , a) = split-at (k q) (λ l → g (q , l)) a

-- a map out of a Σ type that is injective on each fibre and remembers the
-- base point is injective
Σ-inj : {A L : Type} {B : A → Type} (K : Σ A B → L) (τ : L → A) →
  (∀ a b → τ (K (a , b)) ≡ a) → (∀ a b₁ b₂ → K (a , b₁) ≡ K (a , b₂) → b₁ ≡ b₂) →
  ∀ x y → K x ≡ K y → x ≡ y
Σ-inj {B = B} K τ lab inj (a₁ , b₁) (a₂ , b₂) e =
  pth ∙ cong (a₂ ,_) (inj a₂ (subst B ea b₁) b₂ (cong K (sym pth) ∙ e))
  where
  ea : a₁ ≡ a₂
  ea = sym (lab a₁ b₁) ∙∙ cong τ e ∙∙ lab a₂ b₂

  pth : (a₁ , b₁) ≡ (a₂ , subst B ea b₁)
  pth = ΣPathP (ea , subst-filler B ea b₁)

------------------------------------------------------------------------
-- affinity at the generic redex

private
  module Aff = Affinity 𝒯 (FM.freeModel 𝒯 noEq) TermQ′

module _ (w : W.ob) (t : Term P 𝟏 w) (p : Leaf t) (o : Shape (leafWorld t p)) where
  private
    Gen : 𝒱
    Gen = Aff.Gen selfModule (w , t) p o

    Source : Type
    Source = Aff.Source selfModule (w , t) p o

    -- the hole holds the operation node over the fresh generators ...
    node : Term P Gen (leafWorld t p)
    node = ops o λ q → var (inr q , W.id)

    -- ... and every other leaf its own generator
    C : (∂Q ⟅ T₀ ⟅ Gen ⟆ ⟆) .fst
    C = (∂Q ⟪ Tm.η .N-ob Gen ⟫) (Aff.genCtx selfModule (w , t) p o)

    g : (l : Leaf t) → Term P Gen (leafWorld t l)
    g = plug TermQ′ (T₀ ⟅ Gen ⟆) C node .snd

    -- the leaf a generator belongs to
    τ : Source → Leaf t
    τ (inl (r , _)) = r
    τ (inr _) = p

    -- the generators held by a term
    Labelled : (l : Leaf t) {v : W.ob} → Term P Gen v → Type
    Labelled l u = ∀ b → τ (leafVar u b .fst) ≡ l

    FibInj : {v : W.ob} → Term P Gen v → Type
    FibInj u = ∀ b₁ b₂ → leafVar u b₁ .fst ≡ leafVar u b₂ .fst → b₁ ≡ b₂

    node-labelled : Labelled p node
    node-labelled _ = refl

    node-inj : FibInj node
    node-inj (q₁ , _) (q₂ , _) e = ΣPathP (isEmbedding→Inj isEmbedding-inr q₁ q₂ e , refl)

    -- each leaf l holds generators of l, pairwise distinct
    leaf-cases : (l : Leaf t) → Dec (l ≡ p) → Labelled l (g l) × FibInj (g l)
    leaf-cases l (yes e) =
      subst (λ l → Labelled l (g l) × FibInj (g l)) (sym e)
        (subst (λ u → Labelled p u × FibInj u) (sym (plug-hole TermQ′ (T₀ ⟅ Gen ⟆) C node))
          (node-labelled , node-inj))
    leaf-cases l (no ne) =
      subst (λ u → Labelled l u × FibInj u) (sym (plug-remaining TermQ′ (T₀ ⟅ Gen ⟆) C node l ne))
        ((λ _ → refl) , λ _ _ _ → refl)

    K : Σ[ l ∈ Leaf t ] Leaf (shape (g l)) → Source
    K (l , b) = leafVar (g l) b .fst

    K-inj : ∀ x y → K x ≡ K y → x ≡ y
    K-inj = Σ-inj K τ (λ l → leaf-cases l (discreteLeaf t l p) .fst)
                      (λ l → leaf-cases l (discreteLeaf t l p) .snd)

    origin : Leaf (shape (graft t g)) → Source
    origin a = at (graft t g) a .snd .fst

  self-affine-at : ∀ r₁ r₂ → origin r₁ ≡ origin r₂ → r₁ ≡ r₂
  self-affine-at r₁ r₂ e =
    sym (unsplit-split t g r₁)
    ∙∙ cong (unsplit t g) (K-inj (split t g r₁) (split t g r₂)
         (cong (λ z → z .snd .fst) (sym (split-at t g r₁)) ∙∙ e ∙∙ cong (λ z → z .snd .fst) (split-at t g r₂)))
    ∙∙ unsplit-split t g r₂

-- the self module is affine
self-affine : Aff.ModuleAffine selfModule
self-affine (w , t) p o = self-affine-at w t p o

-- finitely many holes, numbered
self-finQ : (∀ w o → isFinOrd (Position w o)) → ∀ s → isFinOrd (IndexedContainer.Pos TermQ′ s .fst)
self-finQ finPos (w , t) = finLeaf finPos t
