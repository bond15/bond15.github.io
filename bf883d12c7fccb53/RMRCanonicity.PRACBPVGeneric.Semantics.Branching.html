-- Branching monads (Def 1.1 of the build spec): the monads M on Set in
-- whose Kleisli category comodels (runners) take their transitions.
--
-- A branching monad is a FINITARY CONTAINER
--
--     M X = Σ[ s ∈ Sh ] (Fin (ar s) → X)        (shapes s, positions Fin (ar s))
--
-- (an M-tree of shape s has ar s leaves, numbered left to right) with a
-- unit shape e and a shape-level Kleisli extension
--
--     bindS : (s : Sh) → (Fin (ar s) → M Y) → M Y,     bind (s , ℓ) k = bindS s (k ∘ ℓ),
--
-- satisfying the three monad laws (unit-l, unit-r, assoc) and two laws on
-- the leaf list  leaves (s , ℓ) = [ ℓ 0 , … , ℓ (ar s − 1) ]:
--
--     leaves-ret  : leaves (ret x)      ≡ [ x ]
--     leaves-bind : leaves (bindS s g)  ≡ leaves (g 0) ++ … ++ leaves (g (ar s − 1)).
--
-- Since bind sees the leaves only through k ∘ ℓ, the Kleisli structure is
-- natural in X by construction.  The law leaves-bind says that the
-- multiplication μ is cartesian AND that the bijection between the positions
-- of μ(s , g) and the pairs (i , position of g i) is the LEXICOGRAPHIC one:
-- the leaves of the composite are the leaves of the children, concatenated
-- in order, each exactly once.  This is strictly stronger than mere
-- cartesianness of μ (which asks only for SOME natural bijection of
-- positions): it fixes the order, which is what the leaf bookkeeping of
-- the machine (leaves-plug, leaves-graft) needs, and it makes the
-- bystander leaves of a configuration non-duplicable, so strong
-- normalization needs no affinity hypothesis.  So: branching monads are
-- the finitary polynomial monads on Set whose multiplication is cartesian
-- in lexicographic position order.
--
-- Instances (with leaves-inj: a tree is determined by its leaf list):
--   IdM      Sh = 1,    ar = 1                        the identity monad
--   MaybeM   Sh = Bool, ar true = 1, ar false = 0     1 + (−): halting / exceptions
--   ListM    Sh = ℕ,    ar n = n                      lists: finite nondeterminism
--   WriterM  Sh = A,    ar = 1                        A × (−) for a monoid A
-- Reader (Sh = 1, positions R, diagonal μ) and State are polynomial but
-- their multiplication is not cartesian; they are not branching monads and
-- appear instead as theories over IdM.
--
--   BranchingMonad: Sh, isSetSh, ar, Br, e, bindS, ret, bind, leaves,
--     unit-l, unit-r, assoc, leaves-ret, leaves-bind;
--     derived: bind-ret-l, map, map-is, isSet⟦⟧, leaves-bindM, leaves-map
--   ⟦_⟧, LeavesInjective
--   list lemmas: concatMap, concatMap-++, concatMap-concatMap,
--     concatMap-tabulate, concatMap-cong, tabulate-map

------------------------------------------------------------------------
-- List lemmas

concatMap : (A → List B) → List A → List B
concatMap f [] = []
concatMap f (x ∷ xs) = f x ++ concatMap f xs

concatMap-++ : (f : A → List B) (xs ys : List A) →
  concatMap f (xs ++ ys) ≡ concatMap f xs ++ concatMap f ys
concatMap-++ f [] ys = refl
concatMap-++ f (x ∷ xs) ys =
  cong (f x ++_) (concatMap-++ f xs ys) ∙ sym (++-assoc (f x) _ _)

concatMap-concatMap : (f : B → List C) (h : A → List B) (xs : List A) →
  concatMap f (concatMap h xs) ≡ concatMap (λ x → concatMap f (h x)) xs
concatMap-concatMap f h [] = refl
concatMap-concatMap f h (x ∷ xs) =
  concatMap-++ f (h x) (concatMap h xs) ∙ cong (concatMap f (h x) ++_) (concatMap-concatMap f h xs)

concatMap-cong : {f g : A → List B} → (∀ x → f x ≡ g x) → ∀ xs → concatMap f xs ≡ concatMap g xs
concatMap-cong p xs i = concatMap (λ x → p x i) xs

concatMap-tabulate : (f : B → List C) (k : A → B) (n : ℕ) (ℓ : Fin n → A) →
  concatMap f (tabulate n (λ i → k (ℓ i))) ≡ concatMap (λ x → f (k x)) (tabulate n ℓ)
concatMap-tabulate f k zero ℓ = refl
concatMap-tabulate f k (suc n) ℓ = cong (f (k (ℓ zero)) ++_) (concatMap-tabulate f k n (λ i → ℓ (suc i)))

tabulate-map : (f : A → B) (n : ℕ) (ℓ : Fin n → A) →
  tabulate n (λ i → f (ℓ i)) ≡ mapL f (tabulate n ℓ)
tabulate-map f zero ℓ = refl
tabulate-map f (suc n) ℓ = cong (f (ℓ zero) ∷_) (tabulate-map f n (λ i → ℓ (suc i)))

------------------------------------------------------------------------
-- Branching monads

record BranchingMonad : Type₁ where
  field
    Sh      : Type
    isSetSh : isSet Sh
    ar      : Sh → ℕ

  -- M-trees with leaves in X: a shape and its leaves, numbered left to right
  Br : Type → Type
  Br X = Σ[ s ∈ Sh ] (Fin (ar s) → X)

  field
    e     : Sh
    bindS : ∀ {Y : Type} (s : Sh) → (Fin (ar s) → Br Y) → Br Y

  ret : ∀ {X : Type} → X → Br X
  ret x = e , λ _ → x

  bind : ∀ {X Y : Type} → Br X → (X → Br Y) → Br Y
  bind m k = bindS (m .fst) (λ i → k (m .snd i))

  leaves : ∀ {X : Type} → Br X → List X
  leaves m = tabulate (ar (m .fst)) (m .snd)

  field
    unit-l      : ∀ {Y : Type} (m : Br Y) → bindS e (λ _ → m) ≡ m
    unit-r      : ∀ {Y : Type} (s : Sh) (ℓ : Fin (ar s) → Y) → bindS s (λ i → ret (ℓ i)) ≡ (s , ℓ)
    assoc       : ∀ {Y Z : Type} (s : Sh) (g : Fin (ar s) → Br Y) (k : Y → Br Z) →
                  bind (bindS s g) k ≡ bindS s (λ i → bind (g i) k)
    leaves-ret  : ∀ {X : Type} (x : X) → leaves (ret x) ≡ [ x ]
    leaves-bind : ∀ {Y : Type} (s : Sh) (g : Fin (ar s) → Br Y) →
                  leaves (bindS s g) ≡ concatMap leaves (tabulate (ar s) g)

  -- derived structure

  bind-ret-l : ∀ {X Y : Type} (x : X) (k : X → Br Y) → bind (ret x) k ≡ k x
  bind-ret-l x k = unit-l (k x)

  map : ∀ {X Y : Type} → (X → Y) → Br X → Br Y
  map f m = bind m (λ x → ret (f x))

  -- map acts on the leaves only
  map-is : ∀ {X Y : Type} (f : X → Y) (s : Sh) (ℓ : Fin (ar s) → X) →
    map f (s , ℓ) ≡ (s , λ i → f (ℓ i))
  map-is f s ℓ = unit-r s (λ i → f (ℓ i))

  isSet⟦⟧ : ∀ {X : Type} → isSet X → isSet (Br X)
  isSet⟦⟧ isSetX = isSetΣ isSetSh (λ _ → isSetΠ λ _ → isSetX)

  -- leaves-bind for a general bind
  leaves-bindM : ∀ {X Y : Type} (m : Br X) (k : X → Br Y) →
    leaves (bind m k) ≡ concatMap (λ x → leaves (k x)) (leaves m)
  leaves-bindM m k =
    leaves-bind (m .fst) (λ i → k (m .snd i))
    ∙ concatMap-tabulate leaves k (ar (m .fst)) (m .snd)

  leaves-map : ∀ {X Y : Type} (f : X → Y) (m : Br X) → leaves (map f m) ≡ mapL f (leaves m)
  leaves-map f m =
    cong leaves (map-is f (m .fst) (m .snd)) ∙ tabulate-map f (ar (m .fst)) (m .snd)

open BranchingMonad

-- the underlying functor on types
⟦_⟧ : BranchingMonad → Type → Type
⟦ M ⟧ X = Br M X

-- a tree is determined by its list of leaves
LeavesInjective : BranchingMonad → Type₁
LeavesInjective M = ∀ {X : Type} (m m' : ⟦ M ⟧ X) → leaves M m ≡ leaves M m' → m ≡ m'

------------------------------------------------------------------------
-- The identity monad: one shape with one leaf

private
  fin1-ext : {X : Type} {ℓ ℓ' : Fin 1 → X} → ℓ zero ≡ ℓ' zero → ℓ ≡ ℓ'
  fin1-ext p = funExt λ { zero → p }

IdM : BranchingMonad
IdM .Sh = Unit
IdM .isSetSh = isSetUnit
IdM .ar _ = 1
IdM .e = tt
IdM .bindS _ g = g zero
IdM .unit-l m = refl
IdM .unit-r s ℓ = ΣPathP (refl , fin1-ext refl)
IdM .assoc s g k = refl
IdM .leaves-ret x = refl
IdM .leaves-bind s g = sym (++-unit-r _)

IdM-leaves-inj : LeavesInjective IdM
IdM-leaves-inj m m' p = ΣPathP (refl , fin1-ext (cons-inj₁ p))

------------------------------------------------------------------------
-- 1 + (−): a tree is a leaf (true) or a halt with no leaves (false)

private
  noLeaves : {X : Type} → Fin 0 → X
  noLeaves ()

  arMaybe : Bool → ℕ
  arMaybe true = 1
  arMaybe false = 0

  bindMaybe : {Y : Type} (s : Bool) → (Fin (arMaybe s) → Σ[ s ∈ Bool ] (Fin (arMaybe s) → Y)) →
    Σ[ s ∈ Bool ] (Fin (arMaybe s) → Y)
  bindMaybe true g = g zero
  bindMaybe false g = false , noLeaves

MaybeM : BranchingMonad
MaybeM .Sh = Bool
MaybeM .isSetSh = isSetBool
MaybeM .ar = arMaybe
MaybeM .e = true
MaybeM .bindS = bindMaybe
MaybeM .unit-l m = refl
MaybeM .unit-r true ℓ = ΣPathP (refl , fin1-ext refl)
MaybeM .unit-r false ℓ = ΣPathP (refl , funExt λ ())
MaybeM .assoc true g k = refl
MaybeM .assoc false g k = refl
MaybeM .leaves-ret x = refl
MaybeM .leaves-bind true g = sym (++-unit-r _)
MaybeM .leaves-bind false g = refl

MaybeM-leaves-inj : LeavesInjective MaybeM
MaybeM-leaves-inj (true , ℓ) (true , ℓ') p = ΣPathP (refl , fin1-ext (cons-inj₁ p))
MaybeM-leaves-inj (true , ℓ) (false , ℓ') p = ⊥.rec (¬cons≡nil p)
MaybeM-leaves-inj (false , ℓ) (true , ℓ') p = ⊥.rec (¬nil≡cons p)
MaybeM-leaves-inj (false , ℓ) (false , ℓ') p = ΣPathP (refl , funExt λ ())

------------------------------------------------------------------------
-- Lists: a tree of shape n is a list of n leaves.  Trees are lists via
-- lookup-tabulate-iso, and the laws are the list monad's.

private
  LBr : Type → Type
  LBr X = Σ[ n ∈ ℕ ] (Fin n → X)

  lleaves : {X : Type} → LBr X → List X
  lleaves m = tabulate (m .fst) (m .snd)

  fromList : {X : Type} → List X → LBr X
  fromList {X} = Iso.fun (lookup-tabulate-iso X)

  -- trees are determined by their leaves
  fromList-lleaves : {X : Type} (m : LBr X) → fromList (lleaves m) ≡ m
  fromList-lleaves {X} m = Iso.sec (lookup-tabulate-iso X) m

  lleaves-fromList : {X : Type} (xs : List X) → lleaves (fromList xs) ≡ xs
  lleaves-fromList = tabulate-lookup

  lleaves-inj : {X : Type} (m m' : LBr X) → lleaves m ≡ lleaves m' → m ≡ m'
  lleaves-inj m m' p = sym (fromList-lleaves m) ∙∙ cong fromList p ∙∙ fromList-lleaves m'

  bindSL : {Y : Type} (n : ℕ) → (Fin n → LBr Y) → LBr Y
  bindSL n g = fromList (concatMap lleaves (tabulate n g))

  bindL : {X Y : Type} → LBr X → (X → LBr Y) → LBr Y
  bindL m k = bindSL (m .fst) (λ i → k (m .snd i))

  retL : {X : Type} → X → LBr X
  retL x = 1 , λ _ → x

  lleaves-bindSL : {Y : Type} (n : ℕ) (g : Fin n → LBr Y) →
    lleaves (bindSL n g) ≡ concatMap lleaves (tabulate n g)
  lleaves-bindSL n g = lleaves-fromList _

  lleaves-bindL : {X Y : Type} (m : LBr X) (k : X → LBr Y) →
    lleaves (bindL m k) ≡ concatMap (λ x → lleaves (k x)) (lleaves m)
  lleaves-bindL m k =
    lleaves-bindSL (m .fst) (λ i → k (m .snd i)) ∙ concatMap-tabulate lleaves k (m .fst) (m .snd)

  concatMap-ret : {X : Type} (n : ℕ) (ℓ : Fin n → X) →
    concatMap lleaves (tabulate n (λ i → retL (ℓ i))) ≡ tabulate n ℓ
  concatMap-ret zero ℓ = refl
  concatMap-ret (suc n) ℓ = cong (ℓ zero ∷_) (concatMap-ret n (λ i → ℓ (suc i)))

  unit-lL : {Y : Type} (m : LBr Y) → bindSL 1 (λ _ → m) ≡ m
  unit-lL m = cong fromList (++-unit-r (lleaves m)) ∙ fromList-lleaves m

  unit-rL : {Y : Type} (n : ℕ) (ℓ : Fin n → Y) → bindSL n (λ i → retL (ℓ i)) ≡ (n , ℓ)
  unit-rL n ℓ = cong fromList (concatMap-ret n ℓ) ∙ fromList-lleaves (n , ℓ)

  -- associativity, compared on leaves
  assocL : {Y Z : Type} (n : ℕ) (g : Fin n → LBr Y) (k : Y → LBr Z) →
    bindL (bindSL n g) k ≡ bindSL n (λ i → bindL (g i) k)
  assocL n g k = lleaves-inj _ _
    (lleaves-bindL (bindSL n g) k
     ∙ cong (concatMap (λ y → lleaves (k y))) (lleaves-bindSL n g)
     ∙ concatMap-concatMap (λ y → lleaves (k y)) lleaves (tabulate n g)
     ∙ sym (concatMap-cong (λ m → lleaves-bindL m k) (tabulate n g))
     ∙ sym (concatMap-tabulate lleaves (λ m → bindL m k) n g)
     ∙ sym (lleaves-bindSL n (λ i → bindL (g i) k)))

ListM : BranchingMonad
ListM .Sh = ℕ
ListM .isSetSh = isSetℕ
ListM .ar n = n
ListM .e = 1
ListM .bindS = bindSL
ListM .unit-l = unit-lL
ListM .unit-r = unit-rL
ListM .assoc = assocL
ListM .leaves-ret x = refl
ListM .leaves-bind = lleaves-bindSL

ListM-leaves-inj : LeavesInjective ListM
ListM-leaves-inj = lleaves-inj

-- Writer: A × (−) for a monoid A (one leaf, the shape records the output)

module _ (𝔸 : Monoid ℓ-zero) where
  open MonoidStr (𝔸 .snd)

  WriterM : BranchingMonad
  WriterM .Sh = 𝔸 .fst
  WriterM .isSetSh = is-set
  WriterM .ar _ = 1
  WriterM .e = ε
  WriterM .bindS a g = a · g zero .fst , g zero .snd
  WriterM .unit-l m = ΣPathP (·IdL (m .fst) , refl)
  WriterM .unit-r a ℓ = ΣPathP (·IdR a , fin1-ext refl)
  WriterM .assoc a g k =
    ΣPathP (sym (·Assoc a (g zero .fst) (k (g zero .snd zero) .fst)) , refl)
  WriterM .leaves-ret x = refl
  WriterM .leaves-bind a g = refl

  -- (WriterM is not leaves-injective: the leaf list forgets the output.)
