-- Errors as Q = T (POPL's Instances/Errors; paper §5.3, Example
-- (errors)).
--
-- W = 1, no locks.  One operation raise with no continuations and no
-- equations (Machine.noEquations).  Free model T X = X + 1, as a container
-- with two shapes: run (true) with one position, the running thread, and
-- raised (false) with none.  So a configuration is a running thread or the
-- error, and ρ = μ (Instances.Equational.Common, Semantics.PolySelfModule).
--
-- With no equations the theory is the one of Instances/SelfModule/Error
-- (the term model, Q = T = terms); the free model here is X + 1 instead of
-- the terms over X, which the machine identifies on reachable
-- configurations (the configuration raise of the term model is raised
-- here).
--
--   errFree, freeModel, Q′, err-module
--   err-affine        at most one hole per configuration
--   err-finQ, err-finOp
--   err-pure          a tree step in the running thread
--   err-effect        POPL's derived rule  S[raise] ⟶ raised
--   err-sn, err-wn, err-ground-canonicity
--   err-run           ⟨run , raise to return tt⟩ ⟶* raised, the only final
--                     configuration (err-run-unique), and its comparison
--                     with the free-model denotation (err-run-comparison)
module RMRCanonicity.PRACBPVGeneric.Instances.Equational.Errors where

isSetErrOp : isSet ErrOp
isSetErrOp = isProp→isSet λ { raise raise → refl }

-- raise, with no continuations
module C = Common ErrOp isSetErrOp (λ _ → ⊥) (λ _ → isProp→isSet isProp⊥)
open C using (𝟙W; L₁; O₁; isSetLk; isSetBase; chaotic; finOpOf)

open import RMRCanonicity.PRACBPVGeneric.Syntax L₁ O₁
  using (ValTy; 𝟙; F; ∅; Comp; Val; op; return; _to_; []ᵥ; tt)
open import RMRCanonicity.PRACBPVGeneric.Reduction L₁ O₁ using (_↦_; step; op-to)
open import RMRCanonicity.PRACBPVGeneric.Syntax L₁ O₁ using (hole)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Ground L₁ O₁ using (Ground; g𝟙; reflect; reflect-den)

-- no equations
errEqs : Machine.Equations L₁ O₁
errEqs = Machine.noEquations L₁ O₁

------------------------------------------------------------------------
-- the free model X + 1, presented by the container (run: one position,
-- raised: none)

Sh : hSet ℓ-zero
Sh = Bool , isSetBool

Ps : Bool → hSet ℓ-zero
Ps true = Unit , isSetUnit
Ps false = ⊥ , isProp→isSet isProp⊥

decPs : ∀ s → Discrete (Ps s .fst)
decPs true _ _ = yes refl
decPs false ()

open C.Presented errEqs Sh Ps decPs

-- the empty family of threads
noThreads : ∀ {Z : Type} → ⊥ → Z
noThreads ()

-- raise: the error configuration
opE : ∀ {Z} → ErrOp → (⊥ → Ext Z) → Ext Z
opE raise _ = false , noThreads

opE-nat : ∀ {Z Z'} (g : Z → Z') (o : ErrOp) (k : ⊥ → Ext Z) → pmap g (opE o k) ≡ opE o (λ a → pmap g (k a))
opE-nat g raise k = ΣPathP (refl , funExt λ ())

open WithOps true opE opE-nat

-- POPL's errFree: run γ extends to f (γ tt), raised to the model's raise
extE : (N : Model) {Z : Type} → (Z → ∣ N ∣) → Ext Z → ∣ N ∣
extE N f (true , γ) = f (γ tt)
extE N f (false , γ) = Nalg N raise noThreads

errFree : PolyFree
errFree .PolyFree.sat X ()
errFree .PolyFree.ext = extE
errFree .PolyFree.ext-η N f z = refl
errFree .PolyFree.ext-hom N f raise k = cong (Nalg N raise) (funExt λ ())
errFree .PolyFree.ext-unique N f h hh hη (true , γ) = hη (γ tt)
errFree .PolyFree.ext-unique N f h hh hη (false , γ) =
  cong h (ΣPathP (refl , funExt λ ())) ∙ hh raise noThreads ∙ cong (Nalg N raise) (funExt λ ())

open WithFree errFree

------------------------------------------------------------------------
-- finiteness

err-finQ : ∀ s → isFinOrd (Ps s .fst)
err-finQ true = isFinOrdUnit
err-finQ false = isFinOrd⊥

err-finOp = finOpOf λ _ → isFinOrd⊥

err-ec : StrongTheory.ExpClosed L₁ O₁ errEqs
err-ec = StrongTheory.expClosed-chaotic L₁ O₁ chaotic errEqs

-- at most one hole
isPropPs : ∀ s → isProp (Ps s .fst)
isPropPs true = isPropUnit
isPropPs false = isProp⊥

------------------------------------------------------------------------
-- the machine

module _ {A : ValTy} where
  open PolyMachine L₁ O₁ isSetLk isSetBase errEqs freeModel Q′ tt* presentation A

  -- ρ = μ
  err-module : RightModule (PS.ER.T) (PS.ER.Q)
  err-module = PS.polyModule presentation

  err-affine : Affine
  err-affine = affine-subsingleton isPropPs

  open WithAffinity err-finOp err-finQ err-affine

  -- a running thread, and the error
  running : Comp (∅ tt*) (F A) → Config
  running M = true , λ _ → M

  raised : Config
  raised = false , noThreads

  -- a tree step in the running thread
  err-pure : {M M' : Comp (∅ tt*) (F A)} → M ↦ M' → running M ⟶ running M'
  err-pure {M} {M'} r = subst (running M ⟶_)
    (ΣPathP (refl , funExt λ _ → plug-hole Q′ (CompPsh (F A)) (ctxAt (running M) tt) M'))
    (pure-in (running M) tt r)

  -- POPL's derived rule:  S[raise] ⟶ raised
  err-effect : (k : ⊥ → Unit → Comp (∅ tt*) (F A)) → running (op raise []ᵥ k) ⟶ raised
  err-effect k = subst (running (op raise []ᵥ k) ⟶_)
    (spliced-hole (running (op raise []ᵥ k)) tt ((raise , tt) , λ q → k (q .fst) (q .snd)))
    (effect-in (running (op raise []ᵥ k)) tt refl)

  err-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  err-sn = strongNormalization

  err-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  err-wn = weakNormalization

  err-ground-canonicity : Ground A → (M : Comp (∅ tt*) (F A)) →
    Σ[ ṽ ∈ Final ] (ηC M ⟶* retF ṽ) × (∀ ṽ' → ηC M ⟶* retF ṽ' → ṽ ≡ ṽ')
  err-ground-canonicity = ground-canonicity-T

------------------------------------------------------------------------
-- a run

-- raise to return tt
raiseThen : Comp (∅ tt*) (F 𝟙)
raiseThen = raise𝟙 to return tt
  where
  raise𝟙 : Comp (∅ tt*) (F 𝟙)
  raise𝟙 = op raise []ᵥ noThreads

open PolyMachine L₁ O₁ isSetLk isSetBase errEqs freeModel Q′ tt* presentation 𝟙
open Comparing err-ec using (module Cmp; module Fr; final-comparison)

-- the final configuration: the error, no values
raisedF : Final
raisedF = false , noThreads

err-run : ηC raiseThen ⟶* retF raisedF
err-run =
  err-pure (step hole (op-to raise []ᵥ))
  then err-effect _
  then (ΣPathP (refl , funExt λ ()) by-equation done)

err-run-unique : ∀ ṽ → ηC raiseThen ⟶* retF ṽ → ṽ ≡ raisedF
err-run-unique ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p err-run

err-run-comparison :
  (ER.Q ⟪ PshHom→NatTrans (Cmp.same Cmp.fo𝟙) ⟫) (denF raisedF)
    ≡ from Fr.⟦ 𝟙 ⟧v (Fr.⟦ raiseThen ⟧C .PshHom.N-ob tt* refl)
err-run-comparison = final-comparison Cmp.fo𝟙 raiseThen _ err-run
