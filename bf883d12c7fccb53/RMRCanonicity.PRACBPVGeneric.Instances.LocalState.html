-- Local state: locations are VALUES, allocation is a LOCK KIND.
--
--   W = Inj; one base type Loc whose constants at world n are the cells Fin n.
--   ONE lock kind, fresh:
--     arity     E n = y (n + 1)         one position, scope n + 1
--     strength  the prefix inclusion n → n + 1
--     names     one Loc: the fresh cell (the last one)
--   [ fresh ] A is Max's 𝕃 ⊸ A, and Θ ⧀⟨ fresh , tt ⟩ is Max's Γ # ℓ.
--
--   Operations (lawless):
--     lookup    params [Loc]   slots Bool, unlocked
--     update b  params [Loc]   one slot, unlocked
--     new b     params []      one slot behind [fresh]
--   so read / write have NO locks in their continuations, there is no
--   modal type for lookup or update, and new true / new false share the one
--   modality [ fresh ].
--
--   Two test operations beyond Max's calculus are COMMENTED OUT below
--   (search for "newIf" / "new₂"): newIf b (slots at n + 1 and n) and
--   new₂ b (one slot behind the word [fresh , fresh]).

------------------------------------------------------------------------
-- the lock signature

data LSBase : Type where
  Loc : LSBase

CellPsh : Functor Inj (SET ℓ-zero)
CellPsh .F-ob n = Fin n ,Σ isSetFin
CellPsh .F-hom ρ i = ρ .fst i
CellPsh .F-id = refl
CellPsh .F-seq f g = refl

LSConst : LSBase → Functor Inj (SET ℓ-zero)
LSConst Loc = CellPsh

-- fresh: E n = y (n + 1)
Efresh : Functor (Inj ^op) (Fam Inj)
Efresh .F-ob n = (Unit ,Σ isSetUnit) ,Σ (λ _ → n + 1)
Efresh .F-hom f = (λ _ → tt) ,Σ (λ _ → extendRight f 1)
Efresh .F-id = ΣPathP (refl ,Σ funExt λ _ → extendRight-id)
Efresh .F-seq f g = ΣPathP (refl ,Σ funExt λ _ → sym (extendRight-seq g f))

-- the strength: the prefix inclusion
strFresh : NatTrans Efresh (Yo Inj)
strFresh .N-ob n = (λ _ → tt) ,Σ (λ _ → extendInjection {n} {1})
strFresh .N-hom f = ΣPathP (refl ,Σ funExt λ _ → extendRight-extendInjection f)

-- the name: the fresh (last) cell, a natural generic element
genFresh : Global Inj (Arity.R Inj Efresh (CstsF Inj LSConst (Loc ∷ [])))
genFresh .Global.el n _ = suffixFin {n} {1} zero ,Σ tt
genFresh .Global.nat f = funExt λ _ → cong (_,Σ tt) (extendRight-map f 1 (inr zero))

data LSLk : Type where
  fresh : LSLk

lsLocks : LockSig Inj
lsLocks .LockSig.Base = LSBase
lsLocks .LockSig.Const = LSConst
lsLocks .LockSig.Lk = LSLk
lsLocks .LockSig.kind fresh = record
  { E = Efresh ; str = strFresh ; names = Loc ∷ [] ; gen = genFresh }

------------------------------------------------------------------------
-- the operations

data LSOp : Type where
  lookup : LSOp
  update : Bool → LSOp
  new : Bool → LSOp
  -- newIf : Bool → LSOp
  -- new₂ : Bool → LSOp

private
  Code = Unit ⊎ (Bool ⊎ Bool)

  enc : LSOp → Code
  enc lookup = inl tt
  enc (update b) = inr (inl b)
  enc (new b) = inr (inr b)

  dec : Code → LSOp
  dec (inl _) = lookup
  dec (inr (inl b)) = update b
  dec (inr (inr b)) = new b

  dec-enc : ∀ o → dec (enc o) ≡ o
  dec-enc lookup = refl
  dec-enc (update b) = refl
  dec-enc (new b) = refl

isSetLSOp : isSet LSOp
isSetLSOp = isSetRetract enc dec dec-enc
  (isSet⊎ isSetUnit (isSet⊎ isSetBool isSetBool))

LSAr : LSOp → Type
LSAr lookup = Bool
LSAr (update b) = Unit
LSAr (new b) = Unit
-- LSAr (newIf b) = Bool
-- LSAr (new₂ b) = Unit

isSetLSAr : ∀ o → isSet (LSAr o)
isSetLSAr lookup = isSetBool
isSetLSAr (update b) = isSetUnit
isSetLSAr (new b) = isSetUnit
-- isSetLSAr (newIf b) = isSetBool
-- isSetLSAr (new₂ b) = isSetUnit

lsOps : OpSig lsLocks
lsOps .OpSig.Op = LSOp
lsOps .OpSig.isSetOp = isSetLSOp
lsOps .OpSig.params lookup = Loc ∷ []
lsOps .OpSig.params (update b) = Loc ∷ []
lsOps .OpSig.params (new b) = []
-- lsOps .OpSig.params (newIf b) = Loc ∷ []
-- lsOps .OpSig.params (new₂ b) = []
lsOps .OpSig.Ar = LSAr
lsOps .OpSig.isSetAr = isSetLSAr
lsOps .OpSig.slot lookup _ = []
lsOps .OpSig.slot (update b) _ = []
lsOps .OpSig.slot (new b) _ = fresh ∷ []
-- lsOps .OpSig.slot (newIf b) true = fresh ∷ []
-- lsOps .OpSig.slot (newIf b) false = []
-- lsOps .OpSig.slot (new₂ b) _ = fresh ∷ fresh ∷ []

------------------------------------------------------------------------
-- the syntax at local state

open import RMRCanonicity.PRACBPVGeneric.Syntax lsLocks lsOps public

𝕃 : ValTy
𝕃 = base Loc

cell : ∀ {Θ} → Fin (world Θ) → Val Θ 𝕃
cell i = const i

-- read V {M | N}: no lock, no renaming of the branches
read : ∀ {Θ B} → Val Θ 𝕃 → Comp Θ B → Comp Θ B → Comp Θ B
read V M N = op lookup (V ∷ᵥ []ᵥ) λ { true _ → M ; false _ → N }

-- V := b ; M
write : ∀ {Θ B} → Val Θ 𝕃 → Bool → Comp Θ B → Comp Θ B
write V b M = op (update b) (V ∷ᵥ []ᵥ) λ _ _ → M

-- Max's  Γ # ℓ  (independent of the initial value)
_#ℓ : Tele → Tele
Θ #ℓ = Θ ⧀⟨ fresh , tt ⟩

-- new ℓ := b in M
newℓ : ∀ {Θ B} (b : Bool) → Comp (Θ #ℓ) B → Comp Θ B
newℓ b M = op (new b) []ᵥ λ _ _ → M

-- ℓ: the name bound by the nearest fresh lock
ℓ : ∀ {Δ} → Val (Δ #ℓ) 𝕃
ℓ = ne (nm ∋z)

-- Max's Cartesian  new x := b in M : substitute ℓ for x
newx : ∀ {Θ B} (b : Bool) → Comp (Θ , 𝕃) B → Comp Θ B
newx b M = newℓ b (subC (⌜ dropL idR tt ⌝ ,ₛ ℓ) M)

-- Max's 𝕃 ⊸ A
𝕃⊸_ : ValTy → ValTy
𝕃⊸ A = [ fresh ] A

------------------------------------------------------------------------
-- examples

-- allocate, then pass the location to a function that reads it
passLoc : Comp (∅ 0) (F 𝟙)
passLoc = newℓ true (app (lam (read (ne here) (return tt) (return tt))) ℓ)

passLocx : Comp (∅ 0) (F 𝟙)
passLocx = newx true (app (lam (read (ne here) (return tt) (return tt))) (ne here))

-- x : Loc bound BEFORE the allocation is reached with up; ℓ is nm
before : Comp (∅ 0 , 𝕃) (F 𝟙)
before = newℓ false (write (ne (up here)) true (read ℓ (return tt) (return tt)))

-- y : Loc bound AFTER the allocation is a plain variable (no up); an
-- argument may instantiate it with ℓ itself, as passLoc does
after : Comp (∅ 0) (𝕃 ⇒ F 𝟙)
after = newℓ false (lam (write (ne here) true (read (wkV ℓ) (return tt) (return tt))))

-- ν ℓ. ℓ : 𝕃 ⊸ 𝕃
generic : ∀ {Θ} → Val Θ (𝕃⊸ 𝕃)
generic = shut λ _ → ℓ

-- an abstracted value is used under ANY new: the modality belongs to the
-- lock kind, not to the operation
useAbs : ∀ (b : Bool) → Comp (∅ 0 , 𝕃⊸ 𝕃) (F 𝕃)
useAbs b = newℓ b (return (ne (opn here)))

-- substituting a cell for a location variable gives the operation lookup
-- at that cell
resolve : ∀ {n} (i : Fin n) →
  sub₁ {Θ = ∅ n} (read (ne here) (return tt) (return tt)) (cell i)
    ≡ read (cell i) (return tt) (return tt)
resolve i = cong (op lookup (cell i ∷ᵥ []ᵥ)) (funExt λ { true → refl ; false → refl })

-- the machine's step after new: closing the lock turns ℓ into the fresh
-- (last) cell of world n + 1
afterNew-ℓ : ∀ {n} → closeC {u = n} {κ = fresh} tt (return ℓ) ≡ return (cell (suffixFin {n} {1} zero))
afterNew-ℓ = refl

closeRead : ∀ {n} →
  closeC {u = n} tt (read ℓ (return tt) (return tt))
    ≡ read (cell (suffixFin {n} {1} zero)) (return tt) (return tt)
closeRead = cong (op lookup _) (funExt λ { true → refl ; false → refl })

-- freshness: no renaming identifies x : Loc bound before the allocation
-- with ℓ (Properties.apart)
open import RMRCanonicity.PRACBPVGeneric.Properties lsLocks lsOps using (apart; locks-mono; subInner)

freshness : ∀ {Θ'} (ρ : Ren ((∅ 0 , 𝕃) #ℓ) Θ') → ¬ (renNe ρ (up here) ≡ renNe ρ (nm ∋z))
freshness ρ = apart ρ here ∋z

-- no renaming deletes the fresh lock that opn needs to open x : 𝕃 ⊸ 𝕃
-- (Properties.locks-mono) ...
noUnlock : ∀ {w w'} → Ren ((∅ w , 𝕃⊸ 𝕃) #ℓ) (∅ w' , 𝕃⊸ 𝕃) → ⊥
noUnlock ρ = locks-mono ρ

-- ... nor does any substitution: close deletes only a lock directly after
-- the root, as in afterNew-ℓ (Properties.subInner)
noUnlockSub : ∀ {w w'} → Sub ((∅ w , 𝕃⊸ 𝕃) #ℓ) (∅ w' , 𝕃⊸ 𝕃) → ⊥
noUnlockSub σ = subInner σ
