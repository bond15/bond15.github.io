-- Guarded type theory's later modality: tick is a LOCK KIND, step an operation.
--
--   W = ω^op: objects ℕ (steps remaining), a map n → m iff m ≤ n.
--   ONE lock kind, tick:
--     arity     E 0 = 0 (no position),  E (n + 1) = y n
--               (the scope is defined only where a position exists)
--     strength  n + 1 → n
--     names     none
--   ▸ A = [ tick ] A,  ⟦▸A⟧(0) = 1,  ⟦▸A⟧(n + 1) = ⟦A⟧(n).
--   ONE operation, step: no parameters, one slot behind [tick].
--
-- No fixpoints: ▸, next, delay, and the Fitch discipline.
module RMRCanonicity.PRACBPVGeneric.Instances.Guarded where

------------------------------------------------------------------------
-- ω^op

data _≤ω_ : ℕ → ℕ → Type where
  z≤ : ∀ {n} → zero ≤ω n
  s≤s : ∀ {m n} → m ≤ω n → suc m ≤ω suc n

isProp≤ω : ∀ {m n} → isProp (m ≤ω n)
isProp≤ω z≤ z≤ = refl
isProp≤ω (s≤s p) (s≤s q) = cong s≤s (isProp≤ω p q)

≤ω-refl : ∀ {n} → n ≤ω n
≤ω-refl {zero} = z≤
≤ω-refl {suc n} = s≤s ≤ω-refl

≤ω-trans : ∀ {k m n} → k ≤ω m → m ≤ω n → k ≤ω n
≤ω-trans z≤ q = z≤
≤ω-trans (s≤s p) (s≤s q) = s≤s (≤ω-trans p q)

≤ω-suc : ∀ n → n ≤ω suc n
≤ω-suc zero = z≤
≤ω-suc (suc n) = s≤s (≤ω-suc n)

-- a map n → m is a proof that m ≤ n
Omega : Category ℓ-zero ℓ-zero
Omega .ob = ℕ
Omega .Hom[_,_] n m = m ≤ω n
Omega .id = ≤ω-refl
Omega ._⋆_ f g = ≤ω-trans g f
Omega .⋆IdL f = isProp≤ω _ _
Omega .⋆IdR f = isProp≤ω _ _
Omega .⋆Assoc f g h = isProp≤ω _ _
Omega .isSetHom = isProp→isSet isProp≤ω

------------------------------------------------------------------------
-- the lock kind tick

data GBase : Type where

-- a tick exists at n + 1, not at 0
PosΩ : ℕ → Type
PosΩ zero = ⊥
PosΩ (suc n) = Unit

isPropPosΩ : ∀ n → isProp (PosΩ n)
isPropPosΩ zero = isProp⊥
isPropPosΩ (suc n) = isPropUnit

-- behind the tick at n + 1 the world is n
scΩ : ∀ n → PosΩ n → ℕ
scΩ (suc n) _ = n

tickOb : ℕ → Fam Omega .ob
tickOb n = (PosΩ n ,Σ isProp→isSet (isPropPosΩ n)) ,Σ scΩ n

-- positions and maps between them are propositions, so all laws are free
isPropTickHom : ∀ {m n} → isProp (Fam Omega [ tickOb m , tickOb n ])
isPropTickHom {m} {n} = isPropΣ (isPropΠ λ _ → isPropPosΩ n) λ _ → isPropΠ λ _ → isProp≤ω

Etick : Functor (Omega ^op) (Fam Omega)
Etick .F-ob = tickOb
Etick .F-hom z≤ = (λ ()) ,Σ (λ ())
Etick .F-hom (s≤s f) = (λ _ → tt) ,Σ (λ _ → f)
Etick .F-id = isPropTickHom _ _
Etick .F-seq f g = isPropTickHom _ _

-- the strength n + 1 → n
strTick : NatTrans Etick (Yo Omega)
strTick .N-ob n = (λ _ → tt) ,Σ ιΩ n
  where
  ιΩ : ∀ n (p : PosΩ n) → scΩ n p ≤ω n
  ιΩ (suc n) _ = ≤ω-suc n
strTick .N-hom f = isPropΣ (isPropΠ λ _ → isPropUnit) (λ _ → isPropΠ λ _ → isProp≤ω) _ _

GConst : GBase → Functor Omega (SET ℓ-zero)
GConst ()

data GLk : Type where
  tick : GLk

gLocks : LockSig Omega
gLocks .LockSig.Base = GBase
gLocks .LockSig.Const = GConst
gLocks .LockSig.Lk = GLk
gLocks .LockSig.kind tick = record
  { E = Etick ; str = strTick ; names = [] ; gen = noNames Omega GConst Etick }

------------------------------------------------------------------------
-- the operation step

data GOp : Type where
  step : GOp

gOps : OpSig gLocks
gOps .OpSig.Op = GOp
gOps .OpSig.isSetOp = isProp→isSet λ { step step → refl }
gOps .OpSig.params step = []
gOps .OpSig.Ar step = Unit
gOps .OpSig.isSetAr step = isProp→isSet isPropUnit
gOps .OpSig.slot step _ = tick ∷ []

-- the syntax

open import RMRCanonicity.PRACBPVGeneric.Syntax gLocks gOps public
open import RMRCanonicity.PRACBPVGeneric.Properties gLocks gOps using (locks-mono; subInner)

infixr 7 ▸_

▸_ : ValTy → ValTy
▸ A = [ tick ] A

-- next : A → ▸ A, the unit of the modality (the transpose of the strength)
next : ∀ {Θ A} → Val Θ A → Val Θ (▸ A)
next = box tick

-- delay M: one step, then M one world later
delay : ∀ {Θ B} → Comp Θ B → Comp Θ B
delay M = op step []ᵥ λ _ p → ↑C p M

-- ▸ U B → B: step, then force the later thunk
later : ∀ {Θ B} → Val Θ (▸ U B) → Comp Θ B
later V = op step []ᵥ λ _ p → force (openV V p)

-- applicative structure, opening both arguments behind one tick
_⊛_ : ∀ {Θ A B} → Val Θ (▸ U (A ⇒ B)) → Val Θ (▸ A) → Val Θ (▸ U B)
f ⊛ x = shut λ p → thunk (app (force (openV f p)) (openV x p))

-- a later variable opens behind a tick ...
openLater : ∀ {n A} → Val ((∅ (suc n) , ▸ A) ⧀⟨ tick , tt ⟩) A
openLater = ne (opn here)

-- ... but not without one: in x : ▸ A every neutral has type ▸ A
noOpenWithoutTick : ∀ {n A B} → Ne (∅ n , ▸ A) B → B ≡ ▸ A
noOpenWithoutTick here = refl
noOpenWithoutTick (wkv ())

-- behind a tick the world is one step earlier
tick-world : ∀ {n} → world (∅ (suc n) ⧀⟨ tick , tt ⟩) ≡ n
tick-world = refl

-- at world 0 there are no ticks: delay drops its continuation, next is trivial
delay₀ : ∀ {B} (M : Comp (∅ 0) B) → delay M ≡ op step []ᵥ (λ _ ())
delay₀ M = cong (op step []ᵥ) (funExt λ _ → funExt λ ())

next₀ : next {Θ = ∅ 0} tt ≡ shut λ ()
next₀ = cong shut (funExt λ ())

-- next then open is the identity, up to crossing the tick
next-open : ∀ {n A} (V : Val (∅ (suc n)) A) → openV (next V) tt ≡ ↑V tt V
next-open V = refl

-- no renaming deletes the tick that opn needs to open x : ▸ A
-- (Properties.locks-mono) ...
noUntick : ∀ {n m A} → Ren ((∅ (suc n) , ▸ A) ⧀⟨ tick , tt ⟩) (∅ m , ▸ A) → ⊥
noUntick ρ = locks-mono ρ

-- ... nor does any substitution: close deletes only a tick directly after
-- the root (Properties.subInner)
noUntickSub : ∀ {n m A} → Sub ((∅ (suc n) , ▸ A) ⧀⟨ tick , tt ⟩) (∅ m , ▸ A) → ⊥
noUntickSub σ = subInner σ

-- ... and no renaming moves a variable from before a tick to after it
noTimeTravel : ∀ {n A} {p : PosΩ n} →
  Ren ((∅ n , A) ⧀⟨ tick , p ⟩) ((∅ n ⧀⟨ tick , p ⟩) , A) → ⊥
noTimeTravel {n} {A} (drop ρ) = noLockToLock ρ
  where
  noVarToRoot : Ren (∅ n , A) (∅ n) → ⊥
  noVarToRoot ()

  noLockToRoot : ∀ {p'} → Ren ((∅ n , A) ⧀⟨ tick , p' ⟩) (∅ n) → ⊥
  noLockToRoot ()

  noLockToLock : ∀ {p' p''} → Ren ((∅ n , A) ⧀⟨ tick , p' ⟩) (∅ n ⧀⟨ tick , p'' ⟩) → ⊥
  noLockToLock (keepL ρ' _) = noVarToRoot ρ'
  noLockToLock (dropL ρ' _) = noLockToRoot ρ'
