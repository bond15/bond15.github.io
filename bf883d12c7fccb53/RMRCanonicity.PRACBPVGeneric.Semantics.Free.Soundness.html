--   shutW-strength   crossing a word's locks is the polynomial's strength
--   sound⇝           head steps preserve ⟦_⟧C (pointwise)
--   den-plug         ⟦ plug S M ⟧ e = ⟦ S ⟧ (⟦ M ⟧ e , e)

-- crossing the locks of a word

-- dropW ρ is dropW idR followed by ρ
dropW-split : ∀ {Θ Δ} (ρ : Ren Θ Δ) (μ : List Lk) (ps : WPos μ (world Δ)) v
  (x : El ⟦ Δ ⧀*⟨ μ , ps ⟩ ⟧t v) →
  ⟦ dropW ρ μ ps ⟧r .N-ob v x ≡ ⟦ ρ ⟧r .N-ob v (⟦ dropW idR μ ps ⟧r .N-ob v x)
dropW-split ρ [] ps v x = refl
dropW-split ρ (κ ∷ []) p v x = refl
dropW-split ρ (κ ∷ μ@(_ ∷ _)) (p , ps) v x =
  dropW-split (dropL ρ p) μ ps v x
  ∙ cong (⟦ ρ ⟧r .N-ob v) (sym (dropW-split (dropL idR p) μ ps v x))

-- a family of continuations that reads the environment only across the
-- word's locks: shutting it moves the environment along the strength
shutW-strength : ∀ {Θ} {X Z : 𝒱} (μ : List Lk)
  (h : (ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X)
  (g : PshHom (X ×Psh ⟦ Θ ⟧t) Z)
  (h' : (ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t Z) →
  (∀ ps t x → h' ps .N-ob t x ≡ g .N-ob t (h ps .N-ob t x , ⟦ dropW idR μ ps ⟧r .N-ob t x)) →
  ∀ v e ps →
  shutW μ h' .N-ob v e ps ≡ g .N-ob _ (shutW μ h .N-ob v e ps , ⟦ Θ ⟧t .F-hom (wι μ v ps) e)
shutW-strength {Θ} [] h g h' hyp v e ps =
  hyp ps v e ∙ cong (λ y → g .N-ob v (h ps .N-ob v e , y)) (sym (funExt⁻ (⟦ Θ ⟧t .F-id) e))
shutW-strength {Θ} (κ ∷ []) h g h' hyp v e q =
  hyp _ _ (K.Over.inj κ (πt Θ) e q)
  ∙ cong (λ y → g .N-ob _ (h _ .N-ob _ (K.Over.inj κ (πt Θ) e q) , y))
      (funExt⁻ (⟦ Θ ⟧t .F-id) _)
shutW-strength {Θ} {X} {Z} (κ ∷ μ@(_ ∷ _)) h g h' hyp v e (q , ps) =
  shutW-strength {Θ ⧀⟨ κ , q₀ ⟩} μ (λ ps → h (q₀ , ps)) g' (λ ps → h' (q₀ , ps))
    (λ ps t x → hyp (q₀ , ps) t x
      ∙ cong (λ y → g .N-ob t (h (q₀ , ps) .N-ob t x , y)) (dropW-split (dropL {Θ} idR q₀) μ ps t x))
    _ (K.Over.inj κ (πt Θ) e q) ps
  ∙ cong (λ y → g .N-ob _ (shutW μ (λ ps → h (q₀ , ps)) .N-ob _ (K.Over.inj κ (πt Θ) e q) ps , y))
      (cong (λ m → ⟦ Θ ⟧t .F-hom m (⟦ Θ ⟧t .F-hom (K.ι κ v q) e)) (W.⋆IdL _)
       ∙ sym (funExt⁻ (⟦ Θ ⟧t .F-seq (K.ι κ v q) _) e))
  where
  q₀ = K.pos κ (πt Θ .N-ob v e) q
  g' : PshHom (X ×Psh ⟦ Θ ⧀⟨ κ , q₀ ⟩ ⟧t) Z
  g' = ×PshIntro (π₁ _ _) (π₂ _ _ ⋆PshHom ⟦ dropL {Θ} idR q₀ ⟧r) ⋆PshHom g

-- head steps

sound⇝ : ∀ {Θ B} {M M' : Comp Θ B} → M ⇝ M' → ∀ v e → ⟦ M ⟧C .N-ob v e ≡ ⟦ M' ⟧C .N-ob v e
sound⇝ {Θ} {B} (β-to {A} {V} {N}) v e =
  bind-η ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ N ⟧C v (⟦ V ⟧V .N-ob v e) e
  ∙ sym (den-sub₁ N V v e)
sound⇝ β-force v e = refl
sound⇝ (β-case {V = V} {V'} {M}) v e = sym (den-sub₂ M V V' v e)
sound⇝ (β-case₊-inl {V = V} {M}) v e = sym (den-sub₁ M V v e)
sound⇝ (β-case₊-inr {V = V} {N = N}) v e = sym (den-sub₁ N V v e)
sound⇝ {Θ} (β-app {M = M} {V}) v e =
  cong (λ y → ⟦ M ⟧C .N-ob v (y , ⟦ V ⟧V .N-ob v e)) (funExt⁻ (⟦ Θ ⟧t .F-id) e)
  ∙ sym (den-sub₁ M V v e)
sound⇝ {Θ} {B} (op-to {A} o vs {k} {N}) v e =
  bind-op ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ N ⟧C v (o , cs) ts e
  ∙ cong (λ c → Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob v ((o , cs) , c))
      (funExt λ (a , ps) → sym (shutW-strength (slot o a) (λ ps → ⟦ k a ps ⟧C)
        (bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ N ⟧C)
        (λ ps → ⟦ k a ps Syn.to renC (keep (dropW idR (slot o a) ps)) N ⟧C)
        (λ ps t x → bind-pre ⟦ Θ ⟧t ⟦ Θ ⧀*⟨ slot o a , ps ⟩ ⟧t ⟦ A ⟧v ⟦ B ⟧m
          ⟦ dropW idR (slot o a) ps ⟧r ⟦ N ⟧C ⟦ renC (keep (dropW idR (slot o a) ps)) N ⟧C
          (λ u x' a' → den-renC (keep (dropW idR (slot o a) ps)) N u (x' , a'))
          t (⟦ k a ps ⟧C .N-ob t x) x)
        v e ps))
  where
  cs = ⟦ vs ⟧Vs .N-ob v e
  ts = λ ((a , ps) : Σ _ λ a → WPos (slot o a) v) → shutW (slot o a) (λ ps → ⟦ k a ps ⟧C) .N-ob v e ps
sound⇝ {Θ} {B} (op-app {A} o vs {k} {V}) v e =
  expAlgebra-id ⟦ A ⟧v ⟦ B ⟧c v (o , cs) ts (⟦ V ⟧V .N-ob v e)
  ∙ cong (λ c → Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob v ((o , cs) , c))
      (funExt λ (a , ps) →
        cong (λ y → ts (a , ps) .N-ob _ (W.id , y)) (sym (⟦ V ⟧V .N-hom _ _ _ e))
        ∙ sym (shutW-strength (slot o a) (λ ps → ⟦ k a ps ⟧C) g
            (λ ps → ⟦ Syn.app (k a ps) (renV (dropW idR (slot o a) ps) V) ⟧C)
            (λ ps t x → cong (λ y → ⟦ k a ps ⟧C .N-ob t x .N-ob t (W.id , y))
              (den-renV (dropW idR (slot o a) ps) V t x))
            v e ps))
  where
  cs = ⟦ vs ⟧Vs .N-ob v e
  ts = λ ((a , ps) : Σ _ λ a → WPos (slot o a) v) → shutW (slot o a) (λ ps → ⟦ k a ps ⟧C) .N-ob v e ps
  g : PshHom (Algebra.Carrier ⟦ A ⇒ B ⟧c ×Psh ⟦ Θ ⟧t) (Algebra.Carrier ⟦ B ⟧c)
  g = ×PshIntro (π₁ _ _) (π₂ _ _ ⋆PshHom ⟦ V ⟧V) ⋆PshHom appExp ⟦ A ⟧v (Algebra.Carrier ⟦ B ⟧c)

-- steps under a stack

den-plug : ∀ {Θ B B'} (S : Stack Θ B B') (M : Comp Θ B) v (e : El ⟦ Θ ⟧t v) →
  ⟦ plug S M ⟧C .N-ob v e ≡ ⟦ S ⟧S .N-ob v (⟦ M ⟧C .N-ob v e , e)
den-plug Syn.hole M v e = refl
den-plug {Θ} (Syn._to•_ {A = A} {B' = B'} S N) M v e =
  cong (λ t → bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B' ⟧m ⟦ N ⟧C .N-ob v (t , e)) (den-plug S M v e)
den-plug (Syn.app• S V) M v e =
  cong (λ f → f .N-ob v (W.id , ⟦ V ⟧V .N-ob v e)) (den-plug S M v e)

sound↦ : ∀ {Θ B} {M M' : Comp Θ B} → M ↦ M' → ⟦ M ⟧C ≡ ⟦ M' ⟧C
sound↦ (step S {M} {M'} r) = makePshHomPath (funExt λ v → funExt λ e →
  den-plug S M v e
  ∙ cong (λ m → ⟦ S ⟧S .N-ob v (m , e)) (sound⇝ r v e)
  ∙ sym (den-plug S M' v e))

sound↦* : ∀ {Θ B} {M M' : Comp Θ B} → M ↦* M' → ⟦ M ⟧C ≡ ⟦ M' ⟧C
sound↦* refl* = refl
sound↦* (step* r rs) = sound↦ r ∙ sound↦* rs
sound↦* {B = B} (op* {o = o} {vs} rs) =
  cong (opDen B o ⟦ vs ⟧Vs) (funExt λ a → funExt λ ps → sound↦* (rs a ps))
