-- Termination of tree reduction (Cor 6.7, Thm 6.8).
--
-- NORMAL FORMS are trees of operations, each continuation behind the
-- locks of its slot, with returned values at the leaves:
--
--   OpTree A Θ     leaf V  |  node o vs (λ a ps → t a ps)
--                  (t a ps : OpTree A (Θ ⧀*⟨ slot o a , ps ⟩))
--   reify          the computation of a tree (return, op)
--
-- A tree in the sense of the logical relation (Tree R Θ M) is a reduction
-- of M to such a tree (treeNF: forget the leaves' relation, read back ↦*,
-- with op* under the nodes).
--
-- COR 6.7 (related).  On a lock telescope every computation of type F A is
-- related: the fundamental lemma at the empty chain, a related
-- environment because every neutral of a lock telescope is a name (𝒢-LT).
--
-- THM 6.8 (termination).  On a lock telescope every computation of type
-- F A reduces to an operation tree.  Closed computations, at ∅ w, are the
-- case of interest; the continuations of their operations live at lock
-- telescopes ∅ w ⧀*⟨ … ⟩, which is why the theorem is stated for all of
-- them.  At the root the parameter values of a node are constants
-- (closeVals-consts: a closed base value is a constant, Closing.closeVals);
-- behind a lock they may also be the lock's names.

-- operation trees

data OpTree (A : ValTy) : Tele → Type where
  leaf : ∀ {Θ} → Val Θ A → OpTree A Θ
  node : ∀ {Θ} (o : Op) → Vals Θ (params o) →
    ((a : Ar o) (ps : WPos (slot o a) (world Θ)) → OpTree A (Θ ⧀*⟨ slot o a , ps ⟩)) → OpTree A Θ

reify : ∀ {A Θ} → OpTree A Θ → Comp Θ (F A)
reify (leaf V) = return V
reify (node o vs t) = op o vs λ a ps → reify (t a ps)

-- a tree of the logical relation is a reduction to an operation tree
treeNF : ∀ {A Θ} {R : (Θ : Tele) → Val Θ A → Type} {M : Comp Θ (F A)} → Tree R Θ M →
  Σ[ t ∈ OpTree A Θ ] (M ↦* reify t)
treeNF (leaf {V = V} _) = leaf V ,ˢ refl*
treeNF (node o {vs} ts) =
  node o vs (λ a ps → treeNF (ts a ps) .fst) ,ˢ op* λ a ps → treeNF (ts a ps) .snd
treeNF (back r t) = treeNF t .fst ,ˢ step* r (treeNF t .snd)

-- Corollary 6.7 and Theorem 6.8

related : ∀ {Θ A} → IsLT Θ → (M : Comp Θ (F A)) → Tree 𝒱⟦ A ⟧ Θ M
related l M = fundC M l []ᶜ (𝒢-LT l)

termination : ∀ {Θ A} → IsLT Θ → (M : Comp Θ (F A)) → Σ[ t ∈ OpTree A Θ ] (M ↦* reify t)
termination l M = treeNF (related l M)

-- at the root, the parameters of an operation are constants
closeVals-consts : ∀ {w bs} (vs : Vals (∅ w) bs) → consts (closeVals vs) ≡ vs
closeVals-consts []ᵥ = refl
closeVals-consts (const c ∷ᵥ vs) = cong (const c ∷ᵥ_) (closeVals-consts vs)
closeVals-consts (ne () ∷ᵥ vs)
