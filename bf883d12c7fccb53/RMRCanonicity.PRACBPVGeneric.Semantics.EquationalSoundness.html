-- The free-model denotation is sound for the equational theory (POPL's
-- EquationalSoundness):  M ≈c M'  ⇒  ⟦ M ⟧C = ⟦ M' ⟧C,  by induction on ≈,
-- for ANY theory with a free model, with no further hypothesis on the
-- theory than the module's: its models are closed under exponentials (ec,
-- Semantics.StrongTheory; automatic for equation-free theories and for a
-- chaotic W such as W = 1, where POPL works).
--
-- The denotation is Semantics.Free.Denotation: F A denotes the free model
-- of the theory on ⟦ A ⟧, so every ⟦ B ⟧ is a model and the axiom `theory`
-- is sound because ⟦ B ⟧ satisfies the theory (⟦ B ⟧m .snd).
--
-- THE CASES.  Pointwise (sound-v, sound-vs, sound-c: at every world v and
-- environment e), as POPL's; sound-v≡ and sound-c≡ are the equalities of
-- maps of presheaves.
--
--   congruences   the clauses of ⟦_⟧ (shutW for op, λExp for lam)
--   β laws        Free.Soundness.sound⇝ (each β law is a head step)
--   η𝟙, ηU        by definition;  η[κ]  Lock.shut-open (at a neutral)
--   η𝟘            ⟦ 𝟘 ⟧ is empty
--   η+, η×        den-sub₁, then den-subC at inlS, inrS, pairS, whose
--                 denotations put the injection / pair back (by definition)
--   η⇒            den-renC at the weakening, naturality of ⟦ M ⟧
--   ηF            den-stack-op: a stack denotes an algebra homomorphism
--                 in its hole (bind-op, expAlgebra-id); stack-bind: so it
--                 is the bind of its restriction to the unit (bind-unique,
--                 the free model's universal property)
--   op-stack      den-stack-op, and shutW-strength: the stack moved across
--                 the word's locks reads the environment along the word's
--                 strength, as the polynomial's algebra does
--   theory        den-inst: a polynomial term evaluated in the closed
--                 computations (inst) denotes its evaluation in ⟦ B ⟧
--                 (Free.Closed.opNodeC-den); at the root a computation is
--                 determined by its denotation at the identity (at-root);
--                 then ⟦ B ⟧ satisfies the theory.

-- pointwise equal maps are equal
pshPath : ∀ {X Y : 𝒱} {f g : PshHom X Y} → (∀ v x → f .N-ob v x ≡ g .N-ob v x) → f ≡ g
pshPath h = makePshHomPath (funExt λ v → funExt (h v))

----------------------------------------------------------------------
-- stacks

-- the denotation commutes with renaming a stack
den-renS : ∀ {Θ Θ' B B'} (ρ : Ren Θ Θ') (S : Stack Θ B B') v (m : El ⟦ B ⟧u v) (e : El ⟦ Θ' ⟧t v) →
  ⟦ renS ρ S ⟧S .N-ob v (m , e) ≡ ⟦ S ⟧S .N-ob v (m , ⟦ ρ ⟧r .N-ob v e)
den-renS ρ hole v m e = refl
den-renS {Θ} {Θ'} ρ (_to•_ {A} {B' = B'} S N) v m e =
  cong (λ t → bind ⟦ Θ' ⟧t ⟦ A ⟧v ⟦ B' ⟧m ⟦ renC (Syn.keep ρ) N ⟧C .N-ob v (t , e)) (den-renS ρ S v m e)
  ∙ bind-pre ⟦ Θ ⟧t ⟦ Θ' ⟧t ⟦ A ⟧v ⟦ B' ⟧m ⟦ ρ ⟧r ⟦ N ⟧C ⟦ renC (Syn.keep ρ) N ⟧C
      (λ u x' a → den-renC (Syn.keep ρ) N u (x' , a))
      v (⟦ S ⟧S .N-ob v (m , ⟦ ρ ⟧r .N-ob v e)) e
den-renS ρ (app• S V) v m e =
  cong₂ (λ f a → f .N-ob v (W.id , a)) (den-renS ρ S v m e) (den-renV ρ V v e)

-- a stack denotes an algebra homomorphism in its hole: an operation in the
-- hole is the operation of the stack's result, the environment moved along
-- the strength into each continuation
den-stack-op : ∀ {Θ B B'} (S : Stack Θ B B') v (o : SN.Shape v)
  (ts : (p : SN.Position v o) → El ⟦ B ⟧u (SN.scope v o p)) (e : El ⟦ Θ ⟧t v) →
  ⟦ S ⟧S .N-ob v (Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob v (o , ts) , e)
    ≡ Algebra.algebra ⟦ B' ⟧c .NatTrans.N-ob v
        (o , λ p → ⟦ S ⟧S .N-ob _ (ts p , ⟦ Θ ⟧t .F-hom (ι v o p) e))
den-stack-op hole v o ts e = refl
den-stack-op {Θ} (_to•_ {A} {B' = B'} S N) v o ts e =
  cong (λ t → bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B' ⟧m ⟦ N ⟧C .N-ob v (t , e)) (den-stack-op S v o ts e)
  ∙ bind-op ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B' ⟧m ⟦ N ⟧C v o _ e
den-stack-op {Θ} (app• {A = A} {B' = B'} S V) v o ts e =
  cong (λ f → f .N-ob v (W.id , ⟦ V ⟧V .N-ob v e)) (den-stack-op S v o ts e)
  ∙ expAlgebra-id ⟦ A ⟧v ⟦ B' ⟧c v o _ (⟦ V ⟧V .N-ob v e)
  ∙ cong (λ c → Algebra.algebra ⟦ B' ⟧c .NatTrans.N-ob v (o , c)) (funExt λ p →
      cong (λ a → ⟦ S ⟧S .N-ob _ (ts p , ⟦ Θ ⟧t .F-hom (ι v o p) e) .N-ob _ (W.id , a))
        (sym (⟦ V ⟧V .N-hom _ _ (ι v o p) e)))

-- F's η in stack form: a stack is algebraic in its hole (den-stack-op), so
-- it is the bind of its restriction to the unit (bind-unique), and that
-- restriction is the stack applied to return x
stack-bind : ∀ {Θ A B} (S : Stack Θ (F A) B) v (t : El ⟦ F A ⟧u v) (e : El ⟦ Θ ⟧t v) →
  ⟦ S ⟧S .N-ob v (t , e)
    ≡ bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ plug (renS (drop idR) S) (Syn.return (ne here)) ⟧C .N-ob v (t , e)
stack-bind {Θ} {A} {B} S v t e =
  bind-unique ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ S ⟧S (den-stack-op S) v t e
  ∙ bind-pre ⟦ Θ ⟧t ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m idPshHom
      ⟦ plug (renS (drop idR) S) (Syn.return (ne here)) ⟧C _
      (λ u x a → sym (den-plug (renS (drop idR) S) (Syn.return (ne here)) u (x , a)
                      ∙ den-renS (drop idR) S u (η ⟦ A ⟧v .N-ob u a) (x , a)))
      v t e

----------------------------------------------------------------------
-- the theory's equations

-- a closed environment denotes a natural environment
envOf : ∀ {X B} → ClosedEnv X B → NatTrans X ⟦ B ⟧u
envOf γ .NatTrans.N-ob u x = denClosed (γ .at x)
envOf γ .NatTrans.N-hom f = funExt λ x → cong denClosed (γ .nat f x) ∙ denClosed-reindex f (γ .at x)

-- inst denotes evaluation
den-inst : ∀ {X B u} (γ : ClosedEnv X B) (t : Term (PM.polynomial L O) X u) →
  denClosed (inst γ t) ≡ evaluate ⟦ B ⟧c (envOf γ) .NatTrans.N-ob u t
den-inst γ (var x) = refl
den-inst {B = B} {u} γ (ops (o , cs) args) =
  opNodeC-den o cs (λ a ps → inst γ (args (a , ps)))
  ∙ cong (λ ks → Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob u ((o , cs) , ks))
      (funExt λ (a , ps) → den-inst γ (args (a , ps)))

-- at the root, a computation is determined by its denotation at the identity
at-root : ∀ {u B} (M : Comp (∅ u) B) v (h : W [ u , v ]) →
  ⟦ M ⟧C .N-ob v h ≡ ⟦ B ⟧u .F-hom h (denClosed M)
at-root M v h = cong (⟦ M ⟧C .N-ob v) (sym (W.⋆IdL h)) ∙ ⟦ M ⟧C .N-hom _ _ h W.id

----------------------------------------------------------------------
-- soundness

sound-v : ∀ {Θ A} {V V' : Val Θ A} → V ≈v V' → ∀ v (e : El ⟦ Θ ⟧t v) →
  ⟦ V ⟧V .N-ob v e ≡ ⟦ V' ⟧V .N-ob v e
sound-vs : ∀ {Θ bs} {vs vs' : Vals Θ bs} → vs ≈vs vs' → ∀ v (e : El ⟦ Θ ⟧t v) →
  ⟦ vs ⟧Vs .N-ob v e ≡ ⟦ vs' ⟧Vs .N-ob v e
sound-c : ∀ {Θ B} {M M' : Comp Θ B} → M ≈c M' → ∀ v (e : El ⟦ Θ ⟧t v) →
  ⟦ M ⟧C .N-ob v e ≡ ⟦ M' ⟧C .N-ob v e

sound-v ≈refl v e = refl
sound-v (≈sym p) v e = sym (sound-v p v e)
sound-v (≈trans p p') v e = sound-v p v e ∙ sound-v p' v e
sound-v (inl≈ p) v e = cong inl (sound-v p v e)
sound-v (inr≈ p) v e = cong inr (sound-v p v e)
sound-v (pair≈ p p') v e = cong₂ _,_ (sound-v p v e) (sound-v p' v e)
sound-v (thunk≈ p) = sound-c p
sound-v (shut≈ p) v e = funExt λ q → sound-v (p _) _ _
sound-v (η𝟙 V) v e = refl
sound-v (ηU V) v e = refl
sound-v (η[κ] {Θ} {κ} (ne n)) v e i = K.Over.shut-open κ (πt Θ) ⟦ n ⟧ne i .N-ob v e
sound-v (η[κ] (shut V)) v e = refl

sound-vs []≈ v e = refl
sound-vs (p ∷≈ ps) v e = cong₂ _,_ (sound-v p v e) (sound-vs ps v e)

sound-c ≈refl v e = refl
sound-c (≈sym p) v e = sym (sound-c p v e)
sound-c (≈trans p p') v e = sound-c p v e ∙ sound-c p' v e
sound-c (return≈ {A = A} p) v e = cong (η ⟦ A ⟧v .N-ob v) (sound-v p v e)
sound-c (to≈ {Θ} {A} {B} {N = N} {N'} p p') v e i =
  bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m (pshPath {f = ⟦ N ⟧C} {⟦ N' ⟧C} (sound-c p') i) .N-ob v (sound-c p v e i , e)
sound-c (force≈ p) = sound-v p
sound-c (case≈ {V = V} {M = M} {M'} p p') v e =
  cong (λ y → ⟦ M ⟧C .N-ob v ((e , y .fst) , y .snd)) (sound-v p v e)
  ∙ sound-c p' v _
sound-c (case₊≈ {V = V} {V'} {M} {M'} {N} {N'} p p' p'') v e =
  cong (λ s → casePsh ⟦ M ⟧C ⟦ N ⟧C .N-ob v (e , s)) (sound-v p v e)
  ∙ casePsh-pt ⟦ M ⟧C ⟦ N ⟧C ⟦ M' ⟧C ⟦ N' ⟧C e e
      (λ a → sound-c p' v (e , a)) (λ a → sound-c p'' v (e , a)) (⟦ V' ⟧V .N-ob v e)
sound-c (absurd≈ {V = V} p) v e = ⊥.rec (⟦ V ⟧V .N-ob v e)
sound-c (lam≈ {A = A} {B} {M} {M'} p) v e =
  cong (λ φ → λExp ⟦ A ⟧v ⟦ B ⟧u φ .N-ob v e) (pshPath {f = ⟦ M ⟧C} {⟦ M' ⟧C} (sound-c p))
sound-c (app≈ p p') v e = cong₂ (λ f a → f .N-ob v (W.id , a)) (sound-c p v e) (sound-v p' v e)
sound-c (op≈ {B = B} {o = o} {k = k} {k'} ps pk) v e =
  cong₂ (λ cs ks → Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob v ((o , cs) , ks))
    (sound-vs ps v e)
    (funExt λ (a , qs) i →
      shutW (slot o a) (λ qs → pshPath {f = ⟦ k a qs ⟧C} {⟦ k' a qs ⟧C} (sound-c (pk a qs)) i) .N-ob v e qs)
-- β
sound-c (F-β V N) = sound⇝ (β-to {V = V} {N})
sound-c (U-β M) = sound⇝ (β-force {M = M})
sound-c (×-β V V' M) = sound⇝ (β-case {V = V} {V'} {M})
sound-c (+-β₁ V M N) = sound⇝ (β-case₊-inl {V = V} {M} {N})
sound-c (+-β₂ V M N) = sound⇝ (β-case₊-inr {V = V} {M} {N})
sound-c (⇒-β M V) = sound⇝ (β-app {M = M} {V})
-- η
sound-c (η𝟘 M V) v e = ⊥.rec (⟦ V ⟧V .N-ob v e)
sound-c (η+ M V) v e = den-sub₁ M V v e ∙ cases (⟦ V ⟧V .N-ob v e)
  where
  cases : ∀ s → ⟦ M ⟧C .N-ob v (e , s) ≡ casePsh ⟦ subC inlS M ⟧C ⟦ subC inrS M ⟧C .N-ob v (e , s)
  cases (inl a) = sym (den-subC inlS M v (e , a))
  cases (inr a) = sym (den-subC inrS M v (e , a))
sound-c (η× M V) v e =
  den-sub₁ M V v e
  ∙ sym (den-subC pairS M v ((e , ⟦ V ⟧V .N-ob v e .fst) , ⟦ V ⟧V .N-ob v e .snd))
sound-c (η⇒ M) v e = makePshHomPath (funExt λ u → funExt λ (h , a) →
  cong (λ f → f .N-ob u (W.id , a)) (den-renC (drop idR) M u (_ , a) ∙ ⟦ M ⟧C .N-hom _ _ h e)
  ∙ cong (λ m → ⟦ M ⟧C .N-ob v e .N-ob u (m , a)) (W.⋆IdR h))
sound-c (ηF S M) v e = den-plug S M v e ∙ stack-bind S v (⟦ M ⟧C .N-ob v e) e
-- algebraicity
sound-c (op-stack {Θ} {B} {B'} S o vs k) v e =
  den-plug S (Syn.op o vs k) v e
  ∙ den-stack-op S v (o , ⟦ vs ⟧Vs .N-ob v e) _ e
  ∙ cong (λ c → Algebra.algebra ⟦ B' ⟧c .NatTrans.N-ob v ((o , ⟦ vs ⟧Vs .N-ob v e) , c))
      (funExt λ (a , ps) → sym (shutW-strength (slot o a) (λ ps → ⟦ k a ps ⟧C) ⟦ S ⟧S
        (λ ps → ⟦ plug (renS (dropW idR (slot o a) ps) S) (k a ps) ⟧C)
        (λ ps t x → den-plug (renS (dropW idR (slot o a) ps) S) (k a ps) t x
                    ∙ den-renS (dropW idR (slot o a) ps) S t (⟦ k a ps ⟧C .N-ob t x) x)
        v e ps))
-- the theory
sound-c (theory {Θ} {B} e γ g) v x =
  den-renC rg (inst γ (lhs e)) v x
  ∙ at-root (inst γ (lhs e)) v h
  ∙ cong (⟦ B ⟧u .F-hom h)
      (den-inst γ (lhs e) ∙ ⟦ B ⟧m .snd e (envOf γ) ∙ sym (den-inst γ (rhs e)))
  ∙ sym (at-root (inst γ (rhs e)) v h)
  ∙ sym (den-renC rg (inst γ (rhs e)) v x)
  where
  open Machine.Equations eqs using (lhs; rhs)
  rg = root {Θ' = Θ} g
  h = ⟦ rg ⟧r .N-ob v x

-- as equalities of maps of presheaves
sound-v≡ : ∀ {Θ A} {V V' : Val Θ A} → V ≈v V' → ⟦ V ⟧V ≡ ⟦ V' ⟧V
sound-v≡ p = pshPath (sound-v p)

sound-c≡ : ∀ {Θ B} {M M' : Comp Θ B} → M ≈c M' → ⟦ M ⟧C ≡ ⟦ M' ⟧C
sound-c≡ p = pshPath (sound-c p)
