-- A deterministic runner's right module satisfies the hypotheses of
-- normalization on the configuration side (Semantics.Normalization):
--
--   runner-affine   it is affine (Affinity.ModuleAffine)
--   runner-finQ     every configuration shape has finitely many holes,
--                   numbered (isFinOrd)
--
-- Both hold because a configuration of a runner (Semantics.Runner) has
-- at most one hole: a running shape (w , σ) has one, a halted shape h has
-- none.  Two holes of one shape are then equal, whatever their origins
-- (Affinity.affine-subsingleton).  So no step of a runner copies a
-- program: the runner follows one branch of every operation node.
--
-- Data: those of Semantics.Runner (any equation-free theory, a runner
-- (S , H , δ)).  Nothing here is specific to PRACBPV.
open import Cubical.Foundations.Prelude
open import Cubical.Data.Empty using (⊥)
import Cubical.Data.Empty as ⊥
open import Cubical.Data.Unit using (isPropUnit)
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Data.Sigma using (Σ-syntax)
open import Cubical.Data.FinSet.Base using (isFinOrd)
open import Cubical.Data.FinSet.Properties using (isFinOrdUnit; isFinOrd⊥)
open import Cubical.Categories.Category using (Category)

open import RightModuleReduction.Theory using (Theory; module SignatureNotation)
open import RightModuleReduction.IndexedContainer using (IndexedContainer)

import RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel as FM
import RMRCanonicity.PRACBPVGeneric.Semantics.Runner as Runner
import RMRCanonicity.PRACBPVGeneric.Semantics.Affinity as Affinity

module RMRCanonicity.PRACBPVGeneric.Semantics.RunnerAffine {W : Category ℓ-zero ℓ-zero}
  (𝒯 : Theory W) (noEq : Theory.Equation 𝒯 → ⊥)
  (S : Category.ob W → Type) (H : Type)
  (isSetConf : isSet (Σ (Category.ob W) S)) (isSetH : isSet H)
  (δ : ∀ {w} → S w → (o : SignatureNotation.Shape (Theory.signature 𝒯) w) →
       H ⊎ (Σ[ p ∈ SignatureNotation.Position (Theory.signature 𝒯) w o ]
              S (SignatureNotation.scope (Theory.signature 𝒯) w o p)))
  where

open Runner 𝒯 noEq S H isSetConf isSetH δ using (RunQ′; runnerModule)
open IndexedContainer RunQ′ using () renaming (Shape to CShape; Pos to CPos)

-- a configuration has at most one hole
runner-subsingleton : ∀ s → isProp (CPos s .fst)
runner-subsingleton (inl _) = isPropUnit
runner-subsingleton (inr _) = ⊥.isProp⊥

-- the runner's module is affine
runner-affine : Affinity.ModuleAffine 𝒯 (FM.freeModel 𝒯 noEq) RunQ′ runnerModule
runner-affine = Affinity.affine-subsingleton 𝒯 (FM.freeModel 𝒯 noEq) RunQ′ runnerModule runner-subsingleton

-- one hole, or none
runner-finQ : ∀ s → isFinOrd (CPos s .fst)
runner-finQ (inl _) = isFinOrdUnit
runner-finQ (inr _) = isFinOrd⊥
