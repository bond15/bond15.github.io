-- Value and computation types.
--   A ::= 0 | 1 | A + A | A × A | U B | [ κ ] A | base b
--   B ::= F A | A ⇒ B
-- [ κ ] A exists only for LOCK KINDS κ (not for operations):
--   ⟦ [ κ ] A ⟧ w = Π (p : Pos κ w) . ⟦ A ⟧ (scope κ w p) = R_κ ⟦A⟧,
-- the right adjoint of the lock.  E.g. ▸ A = [ tick ] A (guarded),
-- 𝕃 ⊸ A = [ fresh ] A (local state).
module RMRCanonicity.PRACBPVGeneric.Types (Lk Base : Set) where

open import Cubical.Foundations.Prelude using (Type)

data ValTy : Type
data CompTy : Type

infixr 6 _×_
infixr 5 _⊕_
infixr 4 _⇒_
infixr 7 [_]_

data ValTy where
  𝟘 : ValTy
  𝟙 : ValTy
  _⊕_ : ValTy → ValTy → ValTy
  _×_ : ValTy → ValTy → ValTy
  U : CompTy → ValTy
  [_]_ : Lk → ValTy → ValTy
  base : Base → ValTy

data CompTy where
  F : ValTy → CompTy
  _⇒_ : ValTy → CompTy → CompTy
