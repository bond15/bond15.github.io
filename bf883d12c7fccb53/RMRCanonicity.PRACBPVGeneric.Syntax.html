-- World-indexed CBPV over a lock signature and an operation signature.
--
-- Judgements  Θ ⊢ V : A  and  Θ ⊢ M : B.  A telescope determines its world:
--
--   ∅ w             the root: the representable context y w     world = w
--   Θ , A           a variable                                   world = world Θ
--   Θ ⧀⟨ κ , p ⟩    the lock of κ at position p                  world = scope κ (world Θ) p
--
-- With no lock kinds, telescopes are ∅ w , A₁ , … , Aₙ (the old core's flat
-- w | Γ) and every slot is unlocked, so every operation keeps the world.
--
-- Neutrals are the projections and the three maps out of a lock:
--   up    Kripke access across a lock (the strength ι)
--   opn   open a [ κ ] A across exactly one κ-lock (the counit of L_κ ⊣ R_κ)
--   nm    a name bound by the lock (the generic elements gen)
--
-- Operations:  op o vs (λ a ps → M),  vs : Vals Θ (params o), and for each
-- slot a and positions ps of its lock word, M : Comp (Θ ⧀*⟨ slot o a , ps ⟩) B.  A slot
-- with the empty word has no lock at all.
--
-- Renamings and substitutions are DATA; each constructor is meant to denote
-- one structure map, so they respect locks by construction (soundness
-- against a semantics is not yet proven; Properties checks that no
-- renaming deletes a lock and no substitution deletes one except by close):
--   Ren   idR | root f | keep | drop | keepL (L_κ is a functor) | dropL (ι)
--   Sub   ⌜ ρ ⌝ | σ ,ₛ V | lockₛ σ p τ (a map out of a lock factors through
--         the lock of the target, then a renaming) | close p f (familial
--         representability: L_{κ,p} (y u) ≅ y (scope κ u p); names become
--         constants)
open import Cubical.Foundations.Prelude using (Type; ℓ-zero; fst; snd; refl; _≡_; sym; cong; funExt⁻)
open import Cubical.Categories.Category using (Category; _[_,_])
open import Cubical.Categories.Functor using (Functor)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Data.Unit using (Unit; tt)
open import Cubical.Data.Sigma using () renaming (_,_ to _,ˢ_)

open import RMRCanonicity.PRACBPVGeneric.Fam
open import RMRCanonicity.PRACBPVGeneric.Signature

----------------------------------------------------------------------
-- world maps: W with a strict identity adjoined (idW denotes W.id; toW
-- below), so that weakening and same-world substitution leave constants
-- and positions definitionally untouched

data WMap : World → World → Type where
  idW : ∀ {w} → WMap w w
  mapW : ∀ {w w'} → W [ w , w' ] → WMap w w'

infixl 6 _⋆W_

_⋆W_ : ∀ {u v w} → WMap u v → WMap v w → WMap u w
f ⋆W idW = f
idW ⋆W mapW g = mapW g
mapW f ⋆W mapW g = mapW (f W.⋆ g)

posW : (κ : Lk) {w w' : World} → WMap w w' → Pos κ w' → Pos κ w
posW κ idW p = p
posW κ (mapW f) p = pos κ f p

smapW : (κ : Lk) {w w' : World} (f : WMap w w') (p : Pos κ w') →
  WMap (scope κ w (posW κ f p)) (scope κ w' p)
smapW κ idW p = idW
smapW κ (mapW f) p = mapW (smap κ f p)

cmapW : ∀ {w w' b} → WMap w w' → Cst b w → Cst b w'
cmapW idW c = c
cmapW {b = b} (mapW f) c = Const b .F-hom f c

-- the interpretation in W (identity on objects): idW denotes W.id, and the
-- operations above agree with their W-versions
toW : ∀ {w w'} → WMap w w' → W [ w , w' ]
toW idW = W.id
toW (mapW f) = f

toW-⋆ : ∀ {u v w} (f : WMap u v) (g : WMap v w) → toW (f ⋆W g) ≡ toW f W.⋆ toW g
toW-⋆ f idW = sym (W.⋆IdR (toW f))
toW-⋆ idW (mapW g) = sym (W.⋆IdL g)
toW-⋆ (mapW f) (mapW g) = refl

posW-toW : (κ : Lk) {w w' : World} (f : WMap w w') (p : Pos κ w') → posW κ f p ≡ pos κ (toW f) p
posW-toW κ idW p = sym (cong (λ h → h .fst p) (E κ .F-id))
posW-toW κ (mapW f) p = refl

cmapW-toW : ∀ {w w' b} (f : WMap w w') (c : Cst b w) → cmapW f c ≡ Const b .F-hom (toW f) c
cmapW-toW {b = b} idW c = sym (funExt⁻ (Const b .F-id) c)
cmapW-toW (mapW f) c = refl

-- positions of a lock word, pulled back along a world map
wposW : (μ : List Lk) {w w' : World} → WMap w w' → WPos μ w' → WPos μ w
wposW [] f _ = tt
wposW (κ ∷ []) f p = posW κ f p
wposW (κ ∷ μ@(_ ∷ _)) f (p ,ˢ ps) = posW κ f p ,ˢ wposW μ (smapW κ f p) ps

----------------------------------------------------------------------
-- telescopes, with their world (induction-recursion)

infixl 5 _,_

data Tele : Type
world : Tele → World

data Tele where
  ∅ : World → Tele
  _,_ : Tele → ValTy → Tele
  _⧀⟨_,_⟩ : (Θ : Tele) (κ : Lk) → Pos κ (world Θ) → Tele

world (∅ w) = w
world (Θ , A) = world Θ
world (Θ ⧀⟨ κ , p ⟩) = scope κ (world Θ) p

-- the locks of a lock word μ, at positions ps
_⧀*⟨_,_⟩ : (Θ : Tele) (μ : List Lk) → WPos μ (world Θ) → Tele
Θ ⧀*⟨ [] , _ ⟩ = Θ
Θ ⧀*⟨ κ ∷ [] , p ⟩ = Θ ⧀⟨ κ , p ⟩
Θ ⧀*⟨ κ ∷ μ@(_ ∷ _) , (p ,ˢ ps) ⟩ = (Θ ⧀⟨ κ , p ⟩) ⧀*⟨ μ , ps ⟩

-- behind a lock word the world is the word's scope
world-⧀* : ∀ Θ (μ : List Lk) (ps : WPos μ (world Θ)) → world (Θ ⧀*⟨ μ , ps ⟩) ≡ wscope μ (world Θ) ps
world-⧀* Θ [] _ = refl
world-⧀* Θ (κ ∷ []) p = refl
world-⧀* Θ (κ ∷ μ@(_ ∷ _)) (p ,ˢ ps) = world-⧀* (Θ ⧀⟨ κ , p ⟩) μ ps

----------------------------------------------------------------------
-- neutrals

data _∋_ : List Base → Base → Type where
  ∋z : ∀ {b cs} → (b ∷ cs) ∋ b
  ∋s : ∀ {b c cs} → cs ∋ b → (c ∷ cs) ∋ b

pick : ∀ {w bs b} → bs ∋ b → Cs w bs → Cst b w
pick ∋z (c ,ˢ cs) = c
pick (∋s k) (c ,ˢ cs) = pick k cs

data Ne : Tele → ValTy → Type where
  here : ∀ {Θ A} → Ne (Θ , A) A
  wkv : ∀ {Θ A B} → Ne Θ A → Ne (Θ , B) A
  -- Kripke access across a lock (the strength)
  up : ∀ {Δ κ p A} → Ne Δ A → Ne (Δ ⧀⟨ κ , p ⟩) A
  -- open a [ κ ] A across exactly one κ-lock (the counit)
  opn : ∀ {Δ κ p A} → Ne Δ ([ κ ] A) → Ne (Δ ⧀⟨ κ , p ⟩) A
  -- a name bound by the lock (the generic elements)
  nm : ∀ {Δ κ p b} → names κ ∋ b → Ne (Δ ⧀⟨ κ , p ⟩) (base b)

----------------------------------------------------------------------
-- terms

data Val (Θ : Tele) : ValTy → Type
data Vals (Θ : Tele) : List Base → Type
data Comp (Θ : Tele) : CompTy → Type
data Stack (Θ : Tele) : CompTy → CompTy → Type

data Val Θ where
  ne : ∀ {A} → Ne Θ A → Val Θ A
  tt : Val Θ 𝟙
  inl : ∀ {A A'} → Val Θ A → Val Θ (A ⊕ A')
  inr : ∀ {A A'} → Val Θ A' → Val Θ (A ⊕ A')
  pair : ∀ {A A'} → Val Θ A → Val Θ A' → Val Θ (A × A')
  thunk : ∀ {B} → Comp Θ B → Val Θ (U B)
  -- a constant at the telescope's world (e.g. a cell)
  const : ∀ {b} → Cst b (world Θ) → Val Θ (base b)
  -- shut: a value behind the lock, at every position of κ (R_κ intro)
  shut : ∀ {κ A} → ((p : Pos κ (world Θ)) → Val (Θ ⧀⟨ κ , p ⟩) A) → Val Θ ([ κ ] A)

infixr 5 _∷ᵥ_

data Vals Θ where
  []ᵥ : Vals Θ []
  _∷ᵥ_ : ∀ {b bs} → Val Θ (base b) → Vals Θ bs → Vals Θ (b ∷ bs)

infixl 4 _to_

data Comp Θ where
  return : ∀ {A} → Val Θ A → Comp Θ (F A)
  _to_ : ∀ {A B} → Comp Θ (F A) → Comp (Θ , A) B → Comp Θ B
  force : ∀ {B} → Val Θ (U B) → Comp Θ B
  case : ∀ {A A' B} → Val Θ (A × A') → Comp (Θ , A , A') B → Comp Θ B
  -- case₊ V (x. M) (y. N): by cases on a sum; absurd V: the empty case
  case₊ : ∀ {A A' B} → Val Θ (A ⊕ A') → Comp (Θ , A) B → Comp (Θ , A') B → Comp Θ B
  absurd : ∀ {B} → Val Θ 𝟘 → Comp Θ B
  lam : ∀ {A B} → Comp (Θ , A) B → Comp Θ (A ⇒ B)
  app : ∀ {A B} → Comp Θ (A ⇒ B) → Val Θ A → Comp Θ B
  -- op o vs (λ a ps → M): typed parameters, and each slot behind its locks
  op : ∀ {B} (o : Op) → Vals Θ (params o) →
    ((a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B) → Comp Θ B

infixl 4 _to•_

data Stack Θ where
  hole : ∀ {B} → Stack Θ B B
  _to•_ : ∀ {A B B'} → Stack Θ B (F A) → Comp (Θ , A) B' → Stack Θ B B'
  app• : ∀ {A B B'} → Stack Θ B (A ⇒ B') → Val Θ A → Stack Θ B B'

plug : ∀ {Θ B B'} → Stack Θ B B' → Comp Θ B → Comp Θ B'
plug hole M = M
plug (S to• N) M = plug S M to N
plug (app• S V) M = app (plug S M) V

consts : ∀ {Θ bs} → Cs (world Θ) bs → Vals Θ bs
consts {bs = []} cs = []ᵥ
consts {bs = b ∷ bs} (c ,ˢ cs) = const c ∷ᵥ consts cs

-- open, on any value: resolves at shut (values stay β-normal)
openV : ∀ {Θ κ A} → Val Θ ([ κ ] A) → (q : Pos κ (world Θ)) → Val (Θ ⧀⟨ κ , q ⟩) A
openV (ne n) q = ne (opn n)
openV (shut V) q = V q

----------------------------------------------------------------------
-- renamings, with their world map computed

data Ren : Tele → Tele → Type
wm : ∀ {Θ Θ'} → Ren Θ Θ' → WMap (world Θ) (world Θ')

data Ren where
  idR : ∀ {Θ} → Ren Θ Θ
  -- the root maps into any telescope, along a world map (y is a functor)
  root : ∀ {w Θ'} → WMap w (world Θ') → Ren (∅ w) Θ'
  keep : ∀ {Θ Θ' A} → Ren Θ Θ' → Ren (Θ , A) (Θ' , A)
  drop : ∀ {Θ Θ' A} → Ren Θ Θ' → Ren Θ (Θ' , A)
  -- under a lock of the same kind (L_κ is a functor): positions pulled back
  keepL : ∀ {Δ Δ' κ} (ρ : Ren Δ Δ') (p : Pos κ (world Δ')) →
    Ren (Δ ⧀⟨ κ , posW κ (wm ρ) p ⟩) (Δ' ⧀⟨ κ , p ⟩)
  -- across a lock of the target, by the strength
  dropL : ∀ {Θ Δ' κ} → Ren Θ Δ' → (p : Pos κ (world Δ')) → Ren Θ (Δ' ⧀⟨ κ , p ⟩)

wm idR = idW
wm (root f) = f
wm (keep ρ) = wm ρ
wm (drop ρ) = wm ρ
wm (keepL {κ = κ} ρ p) = smapW κ (wm ρ) p
wm (dropL {Δ' = Δ'} {κ = κ} ρ p) = wm ρ ⋆W mapW (ι κ (world Δ') p)

renNe : ∀ {Θ Θ' A} → Ren Θ Θ' → Ne Θ A → Ne Θ' A
renNe idR n = n
renNe (root f) ()
renNe (keep ρ) here = here
renNe (keep ρ) (wkv n) = wkv (renNe ρ n)
renNe (drop ρ) n = wkv (renNe ρ n)
renNe (keepL ρ p) (up n) = up (renNe ρ n)
renNe (keepL ρ p) (opn n) = opn (renNe ρ n)
renNe (keepL ρ p) (nm k) = nm k
renNe (dropL ρ p) n = up (renNe ρ n)

-- under the locks of a word
keepW : ∀ {Θ Θ'} (μ : List Lk) (ρ : Ren Θ Θ') (ps : WPos μ (world Θ')) →
  Ren (Θ ⧀*⟨ μ , wposW μ (wm ρ) ps ⟩) (Θ' ⧀*⟨ μ , ps ⟩)
keepW [] ρ _ = ρ
keepW (κ ∷ []) ρ p = keepL ρ p
keepW (κ ∷ μ@(_ ∷ _)) ρ (p ,ˢ ps) = keepW μ (keepL ρ p) ps

renV : ∀ {Θ Θ' A} → Ren Θ Θ' → Val Θ A → Val Θ' A
renVs : ∀ {Θ Θ' bs} → Ren Θ Θ' → Vals Θ bs → Vals Θ' bs
renC : ∀ {Θ Θ' B} → Ren Θ Θ' → Comp Θ B → Comp Θ' B
renS : ∀ {Θ Θ' B B'} → Ren Θ Θ' → Stack Θ B B' → Stack Θ' B B'

renV ρ (ne n) = ne (renNe ρ n)
renV ρ tt = tt
renV ρ (inl V) = inl (renV ρ V)
renV ρ (inr V) = inr (renV ρ V)
renV ρ (pair V V') = pair (renV ρ V) (renV ρ V')
renV ρ (thunk M) = thunk (renC ρ M)
renV ρ (const c) = const (cmapW (wm ρ) c)
renV ρ (shut {κ = κ} V) = shut λ q → renV (keepL ρ q) (V (posW κ (wm ρ) q))

renVs ρ []ᵥ = []ᵥ
renVs ρ (V ∷ᵥ vs) = renV ρ V ∷ᵥ renVs ρ vs

renC ρ (return V) = return (renV ρ V)
renC ρ (M to N) = renC ρ M to renC (keep ρ) N
renC ρ (force V) = force (renV ρ V)
renC ρ (case V M) = case (renV ρ V) (renC (keep (keep ρ)) M)
renC ρ (case₊ V M N) = case₊ (renV ρ V) (renC (keep ρ) M) (renC (keep ρ) N)
renC ρ (absurd V) = absurd (renV ρ V)
renC ρ (lam M) = lam (renC (keep ρ) M)
renC ρ (app M V) = app (renC ρ M) (renV ρ V)
renC ρ (op o vs k) =
  op o (renVs ρ vs) λ a ps → renC (keepW (slot o a) ρ ps) (k a (wposW (slot o a) (wm ρ) ps))

renS ρ hole = hole
renS ρ (S to• N) = renS ρ S to• renC (keep ρ) N
renS ρ (app• S V) = app• (renS ρ S) (renV ρ V)

-- weakening by a variable
wkV : ∀ {Θ A B} → Val Θ A → Val (Θ , B) A
wkV = renV (drop idR)

wkC : ∀ {Θ A B} → Comp Θ B → Comp (Θ , A) B
wkC = renC (drop idR)

-- crossing a lock: earlier variables are reached with up, constants move
-- along ι
↑V : ∀ {Θ κ A} (p : Pos κ (world Θ)) → Val Θ A → Val (Θ ⧀⟨ κ , p ⟩) A
↑V p = renV (dropL idR p)

↑C : ∀ {Θ κ B} (p : Pos κ (world Θ)) → Comp Θ B → Comp (Θ ⧀⟨ κ , p ⟩) B
↑C p = renC (dropL idR p)

-- the unit of [ κ ], the transpose of the strength (for ▸ this is next)
box : ∀ {Θ A} (κ : Lk) → Val Θ A → Val Θ ([ κ ] A)
box κ V = shut λ p → ↑V p V

-- closed computations form a presheaf: reindexing along W
reindexC : ∀ {w w' B} → W [ w , w' ] → Comp (∅ w) B → Comp (∅ w') B
reindexC f = renC (root (mapW f))

----------------------------------------------------------------------
-- substitutions, with their world map computed

data Sub : Tele → Tele → Type
wmS : ∀ {Θ Θ'} → Sub Θ Θ' → WMap (world Θ) (world Θ')

infixl 5 _,ₛ_

data Sub where
  ⌜_⌝ : ∀ {Θ Θ'} → Ren Θ Θ' → Sub Θ Θ'
  _,ₛ_ : ∀ {Θ Θ' A} → Sub Θ Θ' → Val Θ' A → Sub (Θ , A) Θ'
  -- out of a lock: through a lock of the same kind in the target, then a
  -- renaming (the factorisation of a map out of L_κ Γ)
  lockₛ : ∀ {Δ Δ' Θ' κ} (σ : Sub Δ Δ') (p : Pos κ (world Δ')) →
    Ren (Δ' ⧀⟨ κ , p ⟩) Θ' → Sub (Δ ⧀⟨ κ , posW κ (wmS σ) p ⟩) Θ'
  -- a lock right after the root is a world, L_{κ,p} (y u) ≅ y (scope κ u p):
  -- its names become constants
  close : ∀ {u Θ' κ} (p : Pos κ u) → WMap (scope κ u p) (world Θ') →
    Sub (∅ u ⧀⟨ κ , p ⟩) Θ'

wmS ⌜ ρ ⌝ = wm ρ
wmS (σ ,ₛ V) = wmS σ
wmS (lockₛ {κ = κ} σ p τ) = smapW κ (wmS σ) p ⋆W wm τ
wmS (close p f) = f

subNe : ∀ {Θ Θ' A} → Sub Θ Θ' → Ne Θ A → Val Θ' A
subNe ⌜ ρ ⌝ n = ne (renNe ρ n)
subNe (σ ,ₛ V) here = V
subNe (σ ,ₛ V) (wkv n) = subNe σ n
subNe (lockₛ σ p τ) (up n) = renV τ (↑V p (subNe σ n))
subNe (lockₛ σ p τ) (opn n) = renV τ (openV (subNe σ n) p)
subNe (lockₛ σ p τ) (nm k) = ne (renNe τ (nm k))
subNe (close p f) (up ())
subNe (close p f) (opn ())
subNe (close {u = u} {κ = κ} p f) (nm k) = const (cmapW f (pick k (gen κ u p)))

-- weakening a substitution
wkS : ∀ {Θ Θ' A} → Sub Θ Θ' → Sub Θ (Θ' , A)
wkS ⌜ ρ ⌝ = ⌜ drop ρ ⌝
wkS (σ ,ₛ V) = wkS σ ,ₛ wkV V
wkS (lockₛ σ p τ) = lockₛ σ p (drop τ)
wkS (close p f) = close p f

liftS : ∀ {Θ Θ' A} → Sub Θ Θ' → Sub (Θ , A) (Θ' , A)
liftS σ = wkS σ ,ₛ ne here

-- under the locks of a word
liftWS : ∀ {Θ Θ'} (μ : List Lk) (σ : Sub Θ Θ') (ps : WPos μ (world Θ')) →
  Sub (Θ ⧀*⟨ μ , wposW μ (wmS σ) ps ⟩) (Θ' ⧀*⟨ μ , ps ⟩)
liftWS [] σ _ = σ
liftWS (κ ∷ []) σ p = lockₛ σ p idR
liftWS (κ ∷ μ@(_ ∷ _)) σ (p ,ˢ ps) = liftWS μ (lockₛ σ p idR) ps

subV : ∀ {Θ Θ' A} → Sub Θ Θ' → Val Θ A → Val Θ' A
subVs : ∀ {Θ Θ' bs} → Sub Θ Θ' → Vals Θ bs → Vals Θ' bs
subC : ∀ {Θ Θ' B} → Sub Θ Θ' → Comp Θ B → Comp Θ' B
subS : ∀ {Θ Θ' B B'} → Sub Θ Θ' → Stack Θ B B' → Stack Θ' B B'

subV σ (ne n) = subNe σ n
subV σ tt = tt
subV σ (inl V) = inl (subV σ V)
subV σ (inr V) = inr (subV σ V)
subV σ (pair V V') = pair (subV σ V) (subV σ V')
subV σ (thunk M) = thunk (subC σ M)
subV σ (const c) = const (cmapW (wmS σ) c)
subV σ (shut {κ = κ} V) = shut λ q → subV (lockₛ σ q idR) (V (posW κ (wmS σ) q))

subVs σ []ᵥ = []ᵥ
subVs σ (V ∷ᵥ vs) = subV σ V ∷ᵥ subVs σ vs

subC σ (return V) = return (subV σ V)
subC σ (M to N) = subC σ M to subC (liftS σ) N
subC σ (force V) = force (subV σ V)
subC σ (case V M) = case (subV σ V) (subC (liftS (liftS σ)) M)
subC σ (case₊ V M N) = case₊ (subV σ V) (subC (liftS σ) M) (subC (liftS σ) N)
subC σ (absurd V) = absurd (subV σ V)
subC σ (lam M) = lam (subC (liftS σ) M)
subC σ (app M V) = app (subC σ M) (subV σ V)
subC σ (op o vs k) =
  op o (subVs σ vs) λ a ps → subC (liftWS (slot o a) σ ps) (k a (wposW (slot o a) (wmS σ) ps))

subS σ hole = hole
subS σ (S to• N) = subS σ S to• subC (liftS σ) N
subS σ (app• S V) = app• (subS σ S) (subV σ V)

idS : ∀ {Θ} → Sub Θ Θ
idS = ⌜ idR ⌝

-- M[V/x]
sub₁ : ∀ {Θ A B} → Comp (Θ , A) B → Val Θ A → Comp Θ B
sub₁ M V = subC (idS ,ₛ V) M

-- M[V/x, V'/y]
sub₂ : ∀ {Θ A A' B} → Comp (Θ , A , A') B → Val Θ A → Val Θ A' → Comp Θ B
sub₂ M V V' = subC (idS ,ₛ V ,ₛ V') M

-- the machine step under one lock: a closed continuation behind a lock IS
-- a closed term at the scope world, its names the generic constants
closeC : ∀ {u κ B} (p : Pos κ u) → Comp (∅ u ⧀⟨ κ , p ⟩) B → Comp (∅ (scope κ u p)) B
closeC p = subC (close p idW)

-- the syntactic operation node on closed computations is Closing.opNodeC
