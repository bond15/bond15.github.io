-- Canonicity of the machine at Q = T (POPL's canonicity-T and
-- ground-canonicity, the configuration η M), for an equation-free theory
-- (Semantics.SelfMachine: configurations are terms over closed
-- computations, ρ = μ).
--
-- Data: those of Semantics.SelfMachine.
--
--   canonicity-T          η M reaches a final configuration (POPL's
--                         canonicity-T: some T(ret)(ṽ))
--   ground-canonicity-T   at a ground type it is unique
--   denote-ηC             the machine's denotation of η M is the term h M
--                         (Machine.Run.h: the term-algebra denotation at the
--                         identity, evaluated at the unit)
--   denote-ηC-free        at a first-order type, followed by Q (same), it
--                         is the FREE-MODEL denotation of M
--                         (Semantics.Free.Denotation at noEquations and
--                         FM.freeModel), by Free.Comparison.machine-comparison
--
-- So the machine's denotation of the initial configuration agrees with
-- the denotation the equational theory is sound for, and with the
-- final configuration's values (ConfigCanonicity.den-final).
open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Data.Sigma using (Σ-syntax; _×_)
open import Cubical.Data.FinSet.Base using (isFinOrd; isFinOrd→isFinSet)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Functor using (_⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Presheaf.Morphism.Alt using (PshHom; PshHom→NatTrans)

open import RightModuleReduction.Theory using (Term)
open import RightModuleReduction.Model using (Hom)

open import RMRCanonicity.PRACBPVGeneric.Signature
import RMRCanonicity.PRACBPVGeneric.Types as Types
-- the polynomial is always written PM.polynomial L O (see Denotation)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
import RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel as FM
import RMRCanonicity.PRACBPVGeneric.Semantics.SelfMachine as SelfMachine
import RMRCanonicity.PRACBPVGeneric.Semantics.ConfigCanonicity as ConfigCanonicity
import RMRCanonicity.PRACBPVGeneric.Semantics.StrongTheory as StrongTheory
import RMRCanonicity.PRACBPVGeneric.Semantics.Free.Denotation as FreeDen
import RMRCanonicity.PRACBPVGeneric.Semantics.Free.Comparison as Comparison

module RMRCanonicity.PRACBPVGeneric.Semantics.SelfCanonicity
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L))
  (isSetWorld : isSet (Category.ob W))
  (A : Types.ValTy (LockSig.Lk L) (LockSig.Base L))
  -- every operation has finitely many continuations at every world, numbered
  (finCont : ∀ (o : OpSig.Op O) w →
     isFinOrd (Σ[ a ∈ OpSig.Ar O o ] Notation.WPos L (OpSig.slot O o a) w))
  where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O using (F; ∅; Comp)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Denotation L O using (𝒱; El; ⟦_⟧v; ⟦_⟧C)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Ground L O using (Ground)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Machine L O using (theory; noEquations; noEquations-empty)
open import RMRCanonicity.PRACBPVGeneric.Metatheory.ClosedLaws L O isSetLk isSetBase using (closedLaws)

-- the machine at Q = T
open SelfMachine L O isSetLk isSetBase isSetWorld A finCont

-- final configurations, den-final, final-unique, and (the self module is
-- affine) normalize and ground-canonicity
module CC = ConfigCanonicity L O closedLaws noEquations
  (FM.freeModel (theory noEquations) noEquations-empty) SM.TermQ′ SM.selfModule A
open CC public using (Final; retF; denF; den-final; final-unique)
open CC.WithAffinity (λ o w → isFinOrd→isFinSet (finCont o w)) (SM.self-finQ finPos) SM.self-affine public
  using (normalize; ground-canonicity)

-- the free-model denotation for the equation-free theory, and the
-- comparison with the term-algebra one
module Fr = FreeDen L O noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  (StrongTheory.expClosed-noEquations L O)
module Cmp = Comparison L O noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  (StrongTheory.expClosed-noEquations L O)

private
  module W = Category W

----------------------------------------------------------------------
-- canonicity at the configuration η M

canonicity-T : ∀ {w} (M : Comp (∅ w) (F A)) → Σ[ ṽ ∈ Final ] ηC M ⟶* retF ṽ
canonicity-T M = normalize (ηC M)

ground-canonicity-T : Ground A → ∀ {w} (M : Comp (∅ w) (F A)) →
  Σ[ ṽ ∈ Final ] (ηC M ⟶* retF ṽ) × (∀ ṽ' → ηC M ⟶* retF ṽ' → ṽ ≡ ṽ')
ground-canonicity-T g M = ground-canonicity g (ηC M)

----------------------------------------------------------------------
-- the denotation of η M

-- ρ = μ on a configuration with one leaf: the leaf's term
denote-ηC : ∀ {w} (M : Comp (∅ w) (F A)) →
  denote (ηC M) ≡ SM.fromTerm ⟦ A ⟧v (w , h .Hom.transformation .NatTrans.N-ob w M)
denote-ηC M = refl

-- renaming the variables of a term, read as a configuration
fromTerm-map : ∀ {X Y : 𝒱} (α : NatTrans X Y) {w} (u : Term (PM.polynomial L O) X w) →
  SM.fromTerm Y (w , (SM.ER.T .fst ⟪ α ⟫) .NatTrans.N-ob w u) ≡ (SM.ER.Q ⟪ α ⟫) (SM.fromTerm X (w , u))
fromTerm-map {X} {Y} α {w} u =
  SM.conf-ext Y (SM.to-from Y _ ∙ sym (cong (w ,_) (SM.assemble-map α u)))

-- at a first-order type the machine's denotation of η M is, up to Q (same),
-- the free-model denotation of M
denote-ηC-free : (fo : Cmp.FirstOrder A) → ∀ {w} (M : Comp (∅ w) (F A)) →
  (SM.ER.Q ⟪ PshHom→NatTrans (Cmp.same fo) ⟫) (denote (ηC M))
    ≡ SM.fromTerm Fr.⟦ A ⟧v (w , Fr.⟦ M ⟧C .PshHom.N-ob w W.id)
denote-ηC-free fo {w} M =
  sym (fromTerm-map (PshHom→NatTrans (Cmp.same fo)) (h .Hom.transformation .NatTrans.N-ob w M))
  ∙ cong (λ u → SM.fromTerm Fr.⟦ A ⟧v (w , u)) (Cmp.machine-comparison fo M)
