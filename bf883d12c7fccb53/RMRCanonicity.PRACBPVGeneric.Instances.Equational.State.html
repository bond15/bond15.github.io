-- Boolean state as Q = T with the lens laws (POPL's Instances/State;
-- paper §2.1 "Theory of Finite State" for S = 2, §2.2 its free model,
-- §5.3 the state example).
--
-- W = 1, no locks.  Operations get (two continuations, one per state) and
-- set_c for each state c (one continuation); set_c is Instances.Constant's
-- put c.  The equations are the complete set (POPL's D13, Plotkin–Power):
--
--   get-set   get (λ s → set_s x)            = x
--   set-get   set_c (get x)                  = set_c (x c)
--   set-set   set_c (set_c' x)               = set_c' x
--   get-get   get (λ i → get (λ j → x i j))  = get (λ i → x i i)
--
-- Free model: (Bool × X)^Bool, as a container with shapes Bool → Bool (the
-- final state as a function of the initial one) and positions Bool (the
-- thread run from each initial state):
--
--   η x = (id , λ _ → x)
--   get k  at initial state s runs k s from s
--   set_c k at initial state s runs k from c
--
-- Configurations are T (closed computations): a run from each initial
-- state, in parallel, with ρ = μ.
--
--   stateEqs, stateFree, freeModel, Q′, state-module
--   state-affine      POPL's state-affine: get and set_c at a hole p keep
--                     the other holes and fill p from a fresh generator
--   state-finQ, state-finOp
--   state-get, state-set
--                     POPL's derived rules, at any hole p (POPL: at s₀)
--   state-sn, state-wn, state-ground-canonicity
--   toggle-run        toggle from both initial states ends in (not , return tt),
--                     the only final configuration (toggle-unique), with
--                     the free-model comparison (toggle-comparison)

-- the theory

-- a constant presheaf of variables
K : (S : Type) → isSet S → 𝒱
K S isSetS .F-ob _ = S , isSetS
K S _ .F-hom _ x = x
K S _ .F-id = refl
K S _ .F-seq _ _ = refl

getT : ∀ {X w} → (Bool → Term (PM.polynomial L₁ O₁) X w) → Term (PM.polynomial L₁ O₁) X w
getT k = ops (get , tt) λ q → k (q .fst)

setT : ∀ {X w} → Bool → Term (PM.polynomial L₁ O₁) X w → Term (PM.polynomial L₁ O₁) X w
setT c t = ops (put c , tt) λ _ → t

data StateEq : Type where
  get-set : StateEq
  set-get : Bool → StateEq
  set-set : Bool → Bool → StateEq
  get-get : StateEq

stateEqs : Machine.Equations L₁ O₁
stateEqs .Machine.Equations.Equation = StateEq
stateEqs .Machine.Equations.equationWorld _ = tt*
stateEqs .Machine.Equations.Variables get-set = K Unit isSetUnit
stateEqs .Machine.Equations.Variables (set-get c) = K Bool isSetBool
stateEqs .Machine.Equations.Variables (set-set c c') = K Unit isSetUnit
stateEqs .Machine.Equations.Variables get-get = K (Bool × Bool) (isSet× isSetBool isSetBool)
stateEqs .Machine.Equations.lhs get-set = getT λ s → setT s (var tt)
stateEqs .Machine.Equations.rhs get-set = var tt
stateEqs .Machine.Equations.lhs (set-get c) = setT c (getT var)
stateEqs .Machine.Equations.rhs (set-get c) = setT c (var c)
stateEqs .Machine.Equations.lhs (set-set c c') = setT c (setT c' (var tt))
stateEqs .Machine.Equations.rhs (set-set c c') = setT c' (var tt)
stateEqs .Machine.Equations.lhs get-get = getT λ i → getT λ j → var (i , j)
stateEqs .Machine.Equations.rhs get-get = getT λ i → var (i , i)

-- the free model (Bool × X)^Bool, presented by the container (shapes

Sh : hSet ℓ-zero
Sh = (Bool → Bool) , isSetΠ λ _ → isSetBool

Ps : (Bool → Bool) → hSet ℓ-zero
Ps _ = Bool , isSetBool

open C.Presented stateEqs Sh Ps (λ _ → _≟_)

-- get runs k s from s; set_c runs k from c
opS : ∀ {Z} (o : GSOp) → (GSArgs o → Ext Z) → Ext Z
opS get k = (λ s → k s .fst s) , (λ s → k s .snd s)
opS (put c) k = (λ _ → k tt .fst c) , (λ _ → k tt .snd c)

opS-nat : ∀ {Z Z'} (g : Z → Z') (o : GSOp) (k : GSArgs o → Ext Z) → pmap g (opS o k) ≡ opS o (λ a → pmap g (k a))
opS-nat g get k = refl
opS-nat g (put c) k = refl

open WithOps (λ s → s) opS opS-nat

-- the lens laws in any model
module Laws (N : Model) where
  Nget : (Bool → ∣ N ∣) → ∣ N ∣
  Nget k = Nalg N get k

  Nset : Bool → ∣ N ∣ → ∣ N ∣
  Nset c x = Nalg N (put c) λ _ → x

  getset : ∀ x → Nget (λ s → Nset s x) ≡ x
  getset x = N .snd get-set (naturally _ _ λ _ _ → x)

  setget : ∀ c (g : Bool → ∣ N ∣) → Nset c (Nget g) ≡ Nset c (g c)
  setget c g = N .snd (set-get c) (naturally _ _ λ _ → g)

  setset : ∀ c c' x → Nset c (Nset c' x) ≡ Nset c' x
  setset c c' x = N .snd (set-set c c') (naturally _ _ λ _ _ → x)

  getget : (g : Bool → Bool → ∣ N ∣) → Nget (λ i → Nget (λ j → g i j)) ≡ Nget (λ i → g i i)
  getget g = N .snd get-get (naturally _ _ λ _ ij → g (ij .fst) (ij .snd))

  -- get (λ _ → x) = x, derived
  getconst : ∀ x → Nget (λ _ → x) ≡ x
  getconst x = cong Nget (funExt λ _ → sym (getset x)) ∙ getget (λ i j → Nset j x) ∙ getset x

-- POPL's stateFree: (σ , γ) extends to get (λ s → set_{σ s} (f (γ s)))
stateFree : PolyFree
stateFree .PolyFree.sat X get-set env = refl
stateFree .PolyFree.sat X (set-get c) env = refl
stateFree .PolyFree.sat X (set-set c c') env = refl
stateFree .PolyFree.sat X get-get env = refl
stateFree .PolyFree.ext N f t = Laws.Nget N λ s → Laws.Nset N (t .fst s) (f (t .snd s))
stateFree .PolyFree.ext-η N f z = Laws.getset N (f z)
stateFree .PolyFree.ext-hom N f get k =
  sym (Laws.getget N λ i j → Laws.Nset N (k i .fst j) (f (k i .snd j)))
stateFree .PolyFree.ext-hom N f (put c) k =
  Laws.getconst N _
  ∙ sym (Laws.setset N c (k tt .fst c) (f (k tt .snd c)))
  ∙ sym (Laws.setget N c λ s → Laws.Nset N (k tt .fst s) (f (k tt .snd s)))
stateFree .PolyFree.ext-unique N f h hh hη t =
  hh get (λ s → opS (put (t .fst s)) λ _ → (λ s → s) , λ _ → t .snd s)
  ∙ cong (Laws.Nget N) (funExt λ s →
      hh (put (t .fst s)) (λ _ → (λ s → s) , λ _ → t .snd s)
      ∙ cong (Laws.Nset N (t .fst s)) (hη (t .snd s)))

open WithFree stateFree

-- finiteness

isFinOrdBool : isFinOrd Bool
isFinOrdBool = 2 , invEquiv SumFin2≃Bool

state-finQ : ∀ (σ : Bool → Bool) → isFinOrd Bool
state-finQ _ = isFinOrdBool

state-finOp = finOpOf λ { get → isFinOrdBool ; (put c) → isFinOrdUnit }

state-ec : StrongTheory.ExpClosed L₁ O₁ stateEqs
state-ec = StrongTheory.expClosed-chaotic L₁ O₁ chaotic stateEqs

------------------------------------------------------------------------
-- updating one thread

upd : ∀ {Z : Type} → Bool → Z → (Bool → Z) → Bool → Z
upd true z f true = z
upd true z f false = f false
upd false z f true = f true
upd false z f false = z

-- at p, or elsewhere
upd-at : ∀ {Z : Type} (p : Bool) (z : Z) (f : Bool → Z) → upd p z f p ≡ z
upd-at true z f = refl
upd-at false z f = refl

upd-other : ∀ {Z : Type} (p : Bool) (z : Z) (f : Bool → Z) (r : Bool) → ¬ r ≡ p → upd p z f r ≡ f r
upd-other true z f true ne = ⊥.rec (ne refl)
upd-other true z f false ne = refl
upd-other false z f true ne = refl
upd-other false z f false ne = ⊥.rec (ne refl)

private
  isInl : {X Y : Type} → X ⊎ Y → Bool
  isInl (inl _) = true
  isInl (inr _) = false

  inl≢inr : {X Y : Type} {x : X} {y : Y} → inl x ≡ inr y → ⊥
  inl≢inr e = true≢false (cong isInl e)

-- the machine

module _ {A : ValTy} where
  open PolyMachine L₁ O₁ isSetLk isSetBase stateEqs freeModel Q′ tt* presentation A

  -- ρ = μ
  state-module : RightModule (PS.ER.T) (PS.ER.Q)
  state-module = PS.polyModule presentation

  -- at the generic redex, the hole p holds the fresh generators ...
  hole-origin : ∀ σ p o → Σ[ q ∈ _ ] origin σ p o p ≡ inr q
  hole-origin σ p (get , cs) =
    (σ p , tt) , origin-is σ p (get , cs) p ∙ cong (λ y → y .snd (σ p) .fst) (genSpliced-hole σ p (get , cs))
  hole-origin σ p (put c , cs) =
    (tt , tt) , origin-is σ p (put c , cs) p ∙ cong (λ y → y .snd (σ p) .fst) (genSpliced-hole σ p (put c , cs))

  -- ... and every other hole its own generator
  other-origin : ∀ σ p o r (ne : ¬ r ≡ p) → origin σ p o r ≡ inl (r , ne)
  other-origin σ p o r ne = origin-is σ p o r ∙ cong (λ y → y .snd (σ r) .fst) (genSpliced-other σ p o r ne)

  -- POPL's state-affine
  state-affine : Affine
  state-affine σ p o r₁ r₂ e = cases (r₁ ≟ p) (r₂ ≟ p)
    where
    -- (by a helper, not with: the origins themselves decide r ≟ p)
    cases : Dec (r₁ ≡ p) → Dec (r₂ ≡ p) → r₁ ≡ r₂
    cases (yes e₁) (yes e₂) = e₁ ∙ sym e₂
    cases (no n₁) (no n₂) = cong fst (isEmbedding→Inj isEmbedding-inl _ _
      (sym (other-origin σ p o r₁ n₁) ∙ e ∙ other-origin σ p o r₂ n₂))
    cases (yes e₁) (no n₂) = ⊥.rec (inl≢inr
      (sym (other-origin σ p o r₂ n₂) ∙ sym e ∙ cong (origin σ p o) e₁ ∙ hole-origin σ p o .snd))
    cases (no n₁) (yes e₂) = ⊥.rec (inl≢inr
      (sym (other-origin σ p o r₁ n₁) ∙ e ∙ cong (origin σ p o) e₂ ∙ hole-origin σ p o .snd))

  open WithAffinity state-finOp state-finQ state-affine

  -- POPL's derived rule for get, at the hole p (the thread run from the
  -- initial state p, now in state σ p):
  --   ⟨… , (σ p , S[get M]) , …⟩ ⟶ ⟨… , (σ p , S[M (σ p)]) , …⟩
  state-get : (σ : Bool → Bool) (f : Bool → Comp (∅ tt*) (F A)) (p : Bool)
    (k : Bool → Unit → Comp (∅ tt*) (F A)) → f p ≡ op get []ᵥ k →
    (σ , f) ⟶ (σ , upd p (k (σ p) tt) f)
  state-get σ f p k e =
    subst ((σ , f) ⟶_) (ΣPathP (funExt (λ r → cong fst (at r (r ≟ p))) , funExt (λ r → cong snd (at r (r ≟ p)))))
      (effect-in (σ , f) p e)
    where
    g = spliced (σ , f) p ((get , tt) , λ q → k (q .fst) (q .snd)) .snd

    -- the thread run from r, and its state
    run : Bool → Bool × Comp (∅ tt*) (F A)
    run r = g r .fst (σ r) , g r .snd (σ r)

    at : ∀ r → Dec (r ≡ p) → run r ≡ (σ r , upd p (k (σ p) tt) f r)
    at r (yes er) = subst (λ r → run r ≡ (σ r , upd p (k (σ p) tt) f r)) (sym er)
      (cong (λ y → y .fst (σ p) , y .snd (σ p)) (spliced-hole (σ , f) p ((get , tt) , λ q → k (q .fst) (q .snd)))
       ∙ ΣPathP (refl , sym (upd-at p (k (σ p) tt) f)))
    at r (no ne) =
      cong (λ y → y .fst (σ r) , y .snd (σ r)) (spliced-other (σ , f) p ((get , tt) , λ q → k (q .fst) (q .snd)) r ne)
      ∙ ΣPathP (refl , sym (upd-other p (k (σ p) tt) f r ne))

  -- POPL's derived rule for set_c, at the hole p:
  --   ⟨… , (σ p , S[set_c M]) , …⟩ ⟶ ⟨… , (c , S[M]) , …⟩
  state-set : (σ : Bool → Bool) (f : Bool → Comp (∅ tt*) (F A)) (p c : Bool)
    (k : Unit → Unit → Comp (∅ tt*) (F A)) → f p ≡ op (put c) []ᵥ k →
    (σ , f) ⟶ (upd p c σ , upd p (k tt tt) f)
  state-set σ f p c k e =
    subst ((σ , f) ⟶_) (ΣPathP (funExt (λ r → cong fst (at r (r ≟ p))) , funExt (λ r → cong snd (at r (r ≟ p)))))
      (effect-in (σ , f) p e)
    where
    g = spliced (σ , f) p ((put c , tt) , λ q → k (q .fst) (q .snd)) .snd

    run : Bool → Bool × Comp (∅ tt*) (F A)
    run r = g r .fst (σ r) , g r .snd (σ r)

    at : ∀ r → Dec (r ≡ p) → run r ≡ (upd p c σ r , upd p (k tt tt) f r)
    at r (yes er) = subst (λ r → run r ≡ (upd p c σ r , upd p (k tt tt) f r)) (sym er)
      (cong (λ y → y .fst (σ p) , y .snd (σ p)) (spliced-hole (σ , f) p ((put c , tt) , λ q → k (q .fst) (q .snd)))
       ∙ ΣPathP (sym (upd-at p c σ) , sym (upd-at p (k tt tt) f)))
    at r (no ne) =
      cong (λ y → y .fst (σ r) , y .snd (σ r)) (spliced-other (σ , f) p ((put c , tt) , λ q → k (q .fst) (q .snd)) r ne)
      ∙ ΣPathP (sym (upd-other p c σ r ne) , sym (upd-other p (k tt tt) f r ne))

  state-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  state-sn = strongNormalization

  state-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  state-wn = weakNormalization

  state-ground-canonicity : Ground A → (M : Comp (∅ tt*) (F A)) →
    Σ[ ṽ ∈ Final ] (ηC M ⟶* retF ṽ) × (∀ ṽ' → ηC M ⟶* retF ṽ' → ṽ ≡ ṽ')
  state-ground-canonicity = ground-canonicity-T

-- a run: toggle, from both initial states

-- read the bit, write its negation
toggle : Comp (∅ tt*) (F 𝟙)
toggle = op get []ᵥ λ b _ → op (put (not b)) []ᵥ λ _ _ → return tt

open PolyMachine L₁ O₁ isSetLk isSetBase stateEqs freeModel Q′ tt* presentation 𝟙
open Comparing state-ec using (module Cmp; module Fr; final-comparison)

-- from each initial state s the final state is not s, and tt is returned
toggled : Final
toggled = not , λ _ → tt

toggle-run : ηC toggle ⟶* retF toggled
toggle-run =
  state-get _ _ true _ refl
  then state-get _ _ false _ refl
  then state-set _ _ true false _ refl
  then state-set _ _ false true _ refl
  then (ΣPathP (funExt (λ { true → refl ; false → refl }) , funExt λ { true → refl ; false → refl })
  by-equation done)

toggle-unique : ∀ ṽ → ηC toggle ⟶* retF ṽ → ṽ ≡ toggled
toggle-unique ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p toggle-run

toggle-comparison :
  (ER.Q ⟪ PshHom→NatTrans (Cmp.same Cmp.fo𝟙) ⟫) (denF toggled)
    ≡ from Fr.⟦ 𝟙 ⟧v (Fr.⟦ toggle ⟧C .PshHom.N-ob tt* refl)
toggle-comparison = final-comparison Cmp.fo𝟙 toggle _ toggle-run
