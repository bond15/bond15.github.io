-- The size of the logical relation's derivations, the measure of
-- normalization (C.1).
--
-- SIZE is the height of a derivation (Tree, LogicalRelation.agda): a leaf
-- has size 0, a back step adds one, and a node is one more than the
-- largest of its continuations' derivations.  The maximum ranges over the
-- continuations of the operation at the telescope's world,
--
--   Conts o w  =  Σ (a : Ar o). WPos (slot o a) w,
--
-- which is assumed finite (finOp, a module parameter).  The height, not
-- the number of nodes, is what substitution cannot increase: substituting
-- into op o vs k pulls the continuations' positions back along the
-- substitution's world map (wposW), and a pull-back may repeat a
-- continuation or miss one, but never reaches a new one (FinMax.max-pull).
-- A count of nodes would grow when positions are repeated.
--
--   size-unique         derivations of the same computation have the same
--                       size: tree reduction is deterministic (↦-det) and
--                       return, op do not step (ret-normal, op-normal)
--   size-back, size-op  a computation that steps, an operation: the size
--                       of any derivation, read off
--   size-monoTree       a derivation moved along a chain is not larger;
--                       its instances subTree (a substitution), renTree (a
--                       renaming), reindexTree (a closed computation
--                       reindexed along W)
--   closeTree           closing the locks of a continuation family
--                       (Closing.closeK) keeps derivations, and is not
--                       larger (size-closeTree)
--   size-closeK<        a closed continuation of an operation is smaller
--                       than the operation (any two derivations)
--
-- On closed computations (telescope ∅ w) every computation of type F A
-- has a derivation, rel M = Termination.related lt∅ M, so the size is a
-- function of the computation:  sz M = size (rel M).  It drops by one
-- under a tree step (sz-step), does not grow under reindexing
-- (sz-reindex), and every closed continuation of an operation is smaller
-- than the operation (sz-cont).  rel is opaque: otherwise a conversion
-- check between sz M and size (related lt∅ M) evaluates the fundamental
-- lemma on M (minutes, then out of memory, in sz-cont).

----------------------------------------------------------------------
-- the size of a derivation

-- the continuations of an operation at a world
Conts : Op → World → FinSet ℓ-zero
Conts o w = (Σ[ a ∈ Ar o ] WPos (slot o a) w) ,ˢ finOp o w

-- the height of a derivation
size : ∀ {A} {R : (Θ : Tele) → Val Θ A → Type} {Θ} {M : Comp Θ (F A)} → Tree R Θ M → ℕ
size (leaf _) = 0
size {Θ = Θ} (node o ts) = suc (max (Conts o (world Θ)) λ q → size (ts (q .fst) (q .snd)))
size (back r t) = suc (size t)

module _ {A} {R : (Θ : Tele) → Val Θ A → Type} where

  -- transporting a derivation along a path of computations keeps its size
  size-subst : ∀ {Θ} {M M' : Comp Θ (F A)} (e : M ≡ M') (t : Tree R Θ M) →
    size (subst (Tree R Θ) e t) ≡ size t
  size-subst {Θ} {M} e t =
    J (λ M' e → size (subst (Tree R Θ) e t) ≡ size t) (cong size (substRefl {B = Tree R Θ} t)) e

  -- two derivations of the same computation have the same size
  size-unique : ∀ {Θ} {M : Comp Θ (F A)} (t t' : Tree R Θ M) → size t ≡ size t'
  size-unique (leaf _) (leaf _) = refl
  size-unique (leaf _) (back r _) = ⊥.rec (ret-normal r)
  size-unique {Θ} (node o ts) (node .o ts') =
    cong suc (cong (max (Conts o (world Θ))) (funExt λ q →
      size-unique (ts (q .fst) (q .snd)) (ts' (q .fst) (q .snd))))
  size-unique (node o _) (back r _) = ⊥.rec (op-normal o r)
  size-unique (back r _) (leaf _) = ⊥.rec (ret-normal r)
  size-unique (back r _) (node o _) = ⊥.rec (op-normal o r)
  -- the two steps have the same target (↦-det)
  size-unique {Θ} (back r t) (back r' t') =
    cong suc (J (λ M₂ _ → (t₂ : Tree R Θ M₂) → size t ≡ size t₂)
                (λ t₂ → size-unique t t₂) (↦-det r r') t')

  -- a computation that steps: any derivation is one more than any
  -- derivation of its reduct
  size-back : ∀ {Θ} {M M' : Comp Θ (F A)} (t : Tree R Θ M) → M ↦ M' → (t' : Tree R Θ M') →
    size t ≡ suc (size t')
  size-back (leaf _) r t' = ⊥.rec (ret-normal r)
  size-back (node o _) r t' = ⊥.rec (op-normal o r)
  size-back {Θ} (back r₀ t₀) r t' =
    cong suc (sym (size-subst (↦-det r₀ r) t₀) ∙ size-unique (subst (Tree R Θ) (↦-det r₀ r) t₀) t')

  -- an operation: its derivations are nodes
  node-inv : ∀ {Θ} (o : Op) {vs : Vals Θ (params o)}
    {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (F A)} →
    Tree R Θ (op o vs k) → ∀ a ps → Tree R (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)
  node-inv o (node .o ts) = ts
  node-inv o (back r _) = ⊥.rec (op-normal o r)

  size-op : ∀ {Θ} (o : Op) {vs : Vals Θ (params o)}
    {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (F A)}
    (t : Tree R Θ (op o vs k)) →
    size t ≡ suc (max (Conts o (world Θ)) λ q → size (node-inv o t (q .fst) (q .snd)))
  size-op o (node .o ts) = refl
  size-op o (back r _) = ⊥.rec (op-normal o r)

----------------------------------------------------------------------
-- substitution does not increase the size

-- along a chain: a node's continuations are read at pulled-back positions
size-monoTree : ∀ {A Θ Θ'} (χ : Chain Θ Θ') {M : Comp Θ (F A)} (t : Tree 𝒱⟦ A ⟧ Θ M) →
  size (monoTree χ t) ≤ size t
size-monoTree {A} χ (leaf {V = V} v) = ≤-reflexive (size-subst {R = 𝒱⟦ A ⟧} (sym (ch-return χ V)) (leaf (mono𝒱 A χ V v)))
size-monoTree {A} {Θ} {Θ'} χ (node o {vs} {k} ts) =
  ≤-trans (≤-reflexive (size-subst {R = 𝒱⟦ A ⟧} (sym (ch-op χ o vs k)) (node o λ a ps → monoTree (wordᶜ (slot o a) χ ps) (ts a (wposᶜ (slot o a) χ ps)))))
    (suc-≤-suc
      (≤-trans
        (max-mono (Conts o (world Θ')) _ _ λ q →
          size-monoTree (wordᶜ (slot o (q .fst)) χ (q .snd)) (ts (q .fst) (wposᶜ (slot o (q .fst)) χ (q .snd))))
        (max-pull (Conts o (world Θ)) (Conts o (world Θ')) (λ q → size (ts (q .fst) (q .snd)))
          λ q → q .fst ,ˢ wposᶜ (slot o (q .fst)) χ (q .snd))))
size-monoTree χ (back r t) = suc-≤-suc (size-monoTree χ t)

module _ {A : ValTy} where

  -- along a substitution: the chain of one substitution
  subTree : ∀ {Θ Θ'} (σ : Sub Θ Θ') {M : Comp Θ (F A)} → Tree 𝒱⟦ A ⟧ Θ M → Tree 𝒱⟦ A ⟧ Θ' (subC σ M)
  subTree σ = monoTree (σ ∷ᶜ []ᶜ)

  size-subTree : ∀ {Θ Θ'} (σ : Sub Θ Θ') {M : Comp Θ (F A)} (t : Tree 𝒱⟦ A ⟧ Θ M) →
    size (subTree σ t) ≤ size t
  size-subTree σ = size-monoTree (σ ∷ᶜ []ᶜ)

  -- along a renaming
  renTree : ∀ {Θ Θ'} (ρ : Ren Θ Θ') {M : Comp Θ (F A)} → Tree 𝒱⟦ A ⟧ Θ M → Tree 𝒱⟦ A ⟧ Θ' (renC ρ M)
  renTree {Θ' = Θ'} ρ {M} t = subst (Tree 𝒱⟦ A ⟧ Θ') (sub-renC ρ M) (subTree ⌜ ρ ⌝ t)

  size-renTree : ∀ {Θ Θ'} (ρ : Ren Θ Θ') {M : Comp Θ (F A)} (t : Tree 𝒱⟦ A ⟧ Θ M) →
    size (renTree ρ t) ≤ size t
  size-renTree ρ {M} t = ≤-trans (≤-reflexive (size-subst (sub-renC ρ M) (subTree ⌜ ρ ⌝ t))) (size-subTree ⌜ ρ ⌝ t)

  -- a closed computation reindexed along W
  reindexTree : ∀ {w w'} (f : W [ w , w' ]) {M : Comp (∅ w) (F A)} →
    Tree 𝒱⟦ A ⟧ (∅ w) M → Tree 𝒱⟦ A ⟧ (∅ w') (reindexC f M)
  reindexTree f = renTree (root (mapW f))

  size-reindexTree : ∀ {w w'} (f : W [ w , w' ]) {M : Comp (∅ w) (F A)} (t : Tree 𝒱⟦ A ⟧ (∅ w) M) →
    size (reindexTree f t) ≤ size t
  size-reindexTree f = size-renTree (root (mapW f))

----------------------------------------------------------------------
-- closing the locks of a continuation family

-- the strict identity pulls positions back to themselves
wposW-idW : ∀ (μ : List Lk) {w} (ps : WPos μ w) → wposW μ idW ps ≡ ps
wposW-idW [] ps = refl
wposW-idW (κ ∷ []) p = refl
wposW-idW (κ ∷ μ@(_ ∷ _)) (p ,ˢ ps) = cong (p ,ˢ_) (wposW-idW μ ps)

module _ {A : ValTy} where

  -- Closing.closeK on derivations: close p idW, lifted under the rest of
  -- the word, outermost lock first
  closeTree : ∀ {w} (μ : List Lk) (K : (ps : WPos μ w) → Comp (∅ w ⧀*⟨ μ , ps ⟩) (F A)) →
    (∀ ps → Tree 𝒱⟦ A ⟧ (∅ w ⧀*⟨ μ , ps ⟩) (K ps)) →
    ∀ ps → Tree 𝒱⟦ A ⟧ (∅ (wscope μ w ps)) (closeK μ K ps)
  closeTree [] K ts ps = ts ps
  closeTree (κ ∷ []) K ts p = subTree (close p idW) (ts p)
  closeTree (κ ∷ κ' ∷ μ') K ts (p ,ˢ ps) =
    closeTree (κ' ∷ μ')
      (λ ps' → subC (liftWS (κ' ∷ μ') (close p idW) ps') (K (p ,ˢ wposW (κ' ∷ μ') idW ps')))
      (λ ps' → subTree (liftWS (κ' ∷ μ') (close p idW) ps') (ts (p ,ˢ wposW (κ' ∷ μ') idW ps')))
      ps

  size-closeTree : ∀ {w} (μ : List Lk) (K : (ps : WPos μ w) → Comp (∅ w ⧀*⟨ μ , ps ⟩) (F A))
    (ts : ∀ ps → Tree 𝒱⟦ A ⟧ (∅ w ⧀*⟨ μ , ps ⟩) (K ps)) (ps : WPos μ w) →
    size (closeTree μ K ts ps) ≤ size (ts ps)
  size-closeTree [] K ts ps = ≤-refl
  size-closeTree (κ ∷ []) K ts p = size-subTree (close p idW) (ts p)
  size-closeTree (κ ∷ κ' ∷ μ') K ts (p ,ˢ ps) =
    ≤-trans (size-closeTree (κ' ∷ μ') _ ts' ps)
      (≤-trans (size-subTree (liftWS (κ' ∷ μ') (close p idW) ps) (ts (p ,ˢ wposW (κ' ∷ μ') idW ps)))
        (≤-reflexive (cong (λ q → size (ts (p ,ˢ q))) (wposW-idW (κ' ∷ μ') ps))))
    where
    ts' : ∀ ps' → Tree 𝒱⟦ A ⟧ _ _
    ts' ps' = subTree (liftWS (κ' ∷ μ') (close p idW) ps') (ts (p ,ˢ wposW (κ' ∷ μ') idW ps'))

  -- any derivation of a closed continuation of an operation is smaller
  -- than any derivation of the operation: it is not larger than the
  -- continuation's own derivation (size-closeTree), which is below the
  -- node's maximum
  size-closeK< : ∀ {w} (o : Op) {vs : Vals (∅ w) (params o)}
    {k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) (F A)}
    (t : Tree 𝒱⟦ A ⟧ (∅ w) (op o vs k)) (a : Ar o) (ps : WPos (slot o a) w)
    (t' : Tree 𝒱⟦ A ⟧ (∅ (wscope (slot o a) w ps)) (closeK (slot o a) (k a) ps)) →
    size t' < size t
  size-closeK< {w} o {k = k} t a ps t' =
    ≤-trans
      (suc-≤-suc
        (≤-trans (≤-reflexive (size-unique t' (closeTree (slot o a) (k a) (ts a) ps)))
          (≤-trans (size-closeTree (slot o a) (k a) (ts a) ps)
            (max-ub (Conts o w) (λ q → size (ts (q .fst) (q .snd))) (a ,ˢ ps)))))
      (≤-reflexive (sym (size-op o t)))
    where
    ts = node-inv o t

----------------------------------------------------------------------
-- the size of a closed computation

module _ {A : ValTy} where

  -- the derivation of a closed computation.  Opaque, so that the size is
  -- never computed by unfolding the fundamental lemma: comparing
  -- size (rel M) with size (rel M′) compares M with M′ only.
  opaque
    rel : ∀ {w} (M : Comp (∅ w) (F A)) → Tree 𝒱⟦ A ⟧ (∅ w) M
    rel M = related lt∅ M

  sz : ∀ {w} → Comp (∅ w) (F A) → ℕ
  sz M = size (rel M)

  -- a tree step lowers it by one
  sz-step : ∀ {w} {M M' : Comp (∅ w) (F A)} → M ↦ M' → sz M ≡ suc (sz M')
  sz-step {M = M} {M'} r = size-back (rel M) r (rel M')

  -- reindexing does not raise it
  sz-reindex : ∀ {w w'} (f : W [ w , w' ]) (M : Comp (∅ w) (F A)) → sz (reindexC f M) ≤ sz M
  sz-reindex f M =
    ≤-trans (≤-reflexive (size-unique (rel (reindexC f M)) (reindexTree f (rel M))))
      (size-reindexTree f (rel M))

  -- every closed continuation of a closed operation is smaller than it
  sz-cont : ∀ {w} (o : Op) (vs : Vals (∅ w) (params o))
    (k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) (F A))
    (a : Ar o) (ps : WPos (slot o a) w) →
    sz (closeK (slot o a) (k a) ps) < sz (op o vs k)
  sz-cont o vs k a ps = size-closeK< o (rel (op o vs k)) a ps (rel (closeK (slot o a) (k a) ps))
