-- Strong and weak normalization, and canonicity, of the combined machine at a
-- deterministic runner.
--
-- Data: those of Semantics.RunMachine (a lock signature L, an operation
-- signature O, isSet Lk, isSet Base, a runner (S , H , δ) on the derived
-- polynomial, the result type A), and
--
--   finOp   every operation has finitely many continuations at every
--           world (Σ a. WPos (slot o a) w is a finite set)
--
-- Semantics.Normalization needs two more hypotheses, on the right module.
-- For a runner both hold (Semantics.RunnerAffine): its configurations
-- have at most one hole, so it is affine (runner-affine) and has finitely
-- many holes (runner-finQ).  So:
--
--   strongNormalization   every configuration is accessible for the
--                         converse of _⟶_: there is no infinite run
--   weakNormalization     every configuration reaches a terminal one,
--   wn-sound              with the same denotation
--
-- Terminal configurations: running σ (return V) (running-terminal) and
-- halted h (halted-terminal).
--
-- Canonicity (Semantics.ConfigCanonicity, parity stage G), re-exported:
--
--   Final, retF, denF   final configurations: a value at every hole
--   den-final           a final configuration denotes its values
--   final-unique        two final configurations reached from one
--                       configuration are equal, given a left inverse of
--                       the closed denotation
--   normalize           every configuration reaches a final one
--   ground-canonicity   at a ground type, a unique one  This module re-exports RunMachine, so an
-- instance opens one copy of the machine.
open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Data.Sum using (_⊎_)
open import Cubical.Data.Sigma using (Σ-syntax)
open import Cubical.Data.FinSet.Base using (isFinSet)
open import Cubical.Categories.Category using (Category)

open import RightModuleReduction.Theory using (module SignatureNotation)

open import RMRCanonicity.PRACBPVGeneric.Signature
import RMRCanonicity.PRACBPVGeneric.Types as Types
-- the polynomial is always written PM.polynomial L O (see Denotation)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
import RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel as FM
import RMRCanonicity.PRACBPVGeneric.Semantics.RunMachine as RunMachine
import RMRCanonicity.PRACBPVGeneric.Semantics.RunnerAffine as RunnerAffine
import RMRCanonicity.PRACBPVGeneric.Semantics.Normalization as Normalization
import RMRCanonicity.PRACBPVGeneric.Semantics.ConfigCanonicity as ConfigCanonicity

module RMRCanonicity.PRACBPVGeneric.Semantics.RunNormalization
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L))
  (S : Category.ob W → Type) (H : Type)
  (isSetConf : isSet (Σ (Category.ob W) S)) (isSetH : isSet H)
  (δ : ∀ {w} → S w → (o : SignatureNotation.Shape (PM.polynomial L O) w) →
       H ⊎ (Σ[ p ∈ SignatureNotation.Position (PM.polynomial L O) w o ]
              S (SignatureNotation.scope (PM.polynomial L O) w o p)))
  (A : Types.ValTy (LockSig.Lk L) (LockSig.Base L))
  -- every operation has finitely many continuations at every world
  (finOp : ∀ (o : OpSig.Op O) w →
     isFinSet (Σ[ a ∈ OpSig.Ar O o ] Notation.WPos L (OpSig.slot O o a) w))
  where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O using (∅; Val; return)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Machine L O using (theory; noEquations; noEquations-empty)
open import RMRCanonicity.PRACBPVGeneric.Metatheory.ClosedLaws L O isSetLk isSetBase using (closedLaws)

-- the machine at the runner
open RunMachine L O isSetLk isSetBase S H isSetConf isSetH δ A public

private
  module RA = RunnerAffine (theory noEquations) noEquations-empty S H isSetConf isSetH δ

-- normalization, with the runner's affinity and finiteness
open Normalization L O closedLaws noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  Rn.RunQ′ Rn.runnerModule A finOp RA.runner-finQ RA.runner-affine public
  using (Terminal; strongNormalization; progress; weakNormalization; wn-sound)

-- canonicity, with the runner's affinity and finiteness
module CC = ConfigCanonicity L O closedLaws noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  Rn.RunQ′ Rn.runnerModule A
open CC public using (Final; retF; denF; den-final; final-unique)
open CC.WithAffinity finOp RA.runner-finQ RA.runner-affine public using (normalize; ground-canonicity)

-- the terminal configurations
running-terminal : ∀ {w} (σ : S w) (V : Val (∅ w) A) → Terminal (running σ (return V))
running-terminal σ V _ = V , refl

halted-terminal : ∀ h → Terminal (halted h)
halted-terminal h ()
