-- Substitutions denote maps of telescopes over their world maps, and the
-- denotation commutes with substitution.
--
--   ⟦ σ ⟧ : ⟦ Θ' ⟧ → ⟦ Θ ⟧       for σ : Sub Θ Θ',  over  y (wmS σ)
--
--   ⌜ ρ ⌝           ⟦ ρ ⟧
--   σ ,ₛ V          ⟨ ⟦ σ ⟧ , ⟦ V ⟧ ⟩
--   lockₛ σ p τ     ⟦ τ ⟧ ; L_κ ⟦ σ ⟧ on the p-summand
--   close p f       ⟦ Θ' ⟧ → y (world Θ') → y (scope κ u p) ≅ p-summand of L_κ (y u)
--                   (familial representability, ν)
--
-- The neutral cases are the adjunction: up is the strength's naturality
-- (ε-Lmap, unlock-box), opn is "openV is the counit" (open-shut at shut),
-- nm is unlock-Lmap-at, and nm under close is the generic element read at
-- ν (by definition of unlock).
--
-- ⟦ idS ⟧s is the identity by definition (idS = ⌜ idR ⌝ and ⟦ idR ⟧r = id),
-- so single and double substitution are den-subC at idS ,ₛ V, with the
-- environment extended definitionally (den-sub₁, den-sub₂).

------------------------------------------------------------------------
-- the denotation of a substitution

⟦_⟧s : ∀ {Θ Θ'} → Sub Θ Θ' → PshHom ⟦ Θ' ⟧t ⟦ Θ ⟧t
sub-over : ∀ {Θ Θ'} (σ : Sub Θ Θ') v (e : El ⟦ Θ' ⟧t v) →
  πt Θ .N-ob v (⟦ σ ⟧s .N-ob v e) ≡ toW (wmS σ) W.⋆ πt Θ' .N-ob v e

⟦ ⌜ ρ ⌝ ⟧s = ⟦ ρ ⟧r
⟦ σ ,ₛ V ⟧s = ×PshIntro ⟦ σ ⟧s ⟦ V ⟧V
⟦ lockₛ {Δ} {Δ'} {Θ'} {κ} σ p τ ⟧s =
  ⟦ τ ⟧r ⋆PshHom K.Lsum κ (πt Δ') (πt Δ) ⟦ σ ⟧s (toW (wmS σ)) (sub-over σ)
    p (posW κ (wmS σ) p) (toW (smapW κ (wmS σ) p)) (pos-smapW κ (wmS σ) p)
⟦ close {u} {Θ'} {κ} p f ⟧s = πt Θ' ⋆PshHom Ymap (toW f) ⋆PshHom K.ν κ p

sub-over ⌜ ρ ⌝ = ren-over ρ
sub-over (σ ,ₛ V) v e = sub-over σ v e
sub-over {Θ' = Θ'} (lockₛ {κ = κ} σ p τ) v e =
  cong (toW (smapW κ (wmS σ) p) W.⋆_) (ren-over τ v e)
  ∙ sym (W.⋆Assoc _ _ _)
  ∙ cong (W._⋆ πt Θ' .N-ob v e) (sym (toW-⋆ (smapW κ (wmS σ) p) (Syn.wm τ)))
sub-over (close p f) v e = refl

-- weakening a substitution forgets the new component
sub-wk : ∀ {Θ Θ' A} (σ : Sub Θ Θ') v (e : El ⟦ Θ' ⟧t v) (a : El ⟦ A ⟧v v) →
  ⟦ wkS {A = A} σ ⟧s .N-ob v (e , a) ≡ ⟦ σ ⟧s .N-ob v e
sub-wk ⌜ ρ ⌝ v e a = refl
sub-wk (σ ,ₛ V) v e a = cong₂ _,_ (sub-wk σ v e a) (den-renV (drop idR) V v (e , a))
sub-wk (lockₛ σ p τ) v e a = refl
sub-wk (close p f) v e a = refl

sub-lift : ∀ {Θ Θ' A} (σ : Sub Θ Θ') v (e : El ⟦ Θ' ⟧t v) (a : El ⟦ A ⟧v v) →
  ⟦ liftS {A = A} σ ⟧s .N-ob v (e , a) ≡ (⟦ σ ⟧s .N-ob v e , a)
sub-lift σ v e a = cong (_, a) (sub-wk σ v e a)

------------------------------------------------------------------------
-- openV is the counit
--
-- ⟦ openV V q ⟧V = incl ⋆ unlock ⟦ V ⟧V, pointwise: s .fst is incl's
-- component (the summand inclusion forgets the position witness).

openV-counit : ∀ {Δ κ A} (V : Val Δ ([ κ ] A)) (q : Pos κ (world Δ)) {t}
  (s : El ⟦ Δ ⧀⟨ κ , q ⟩ ⟧t t) →
  ⟦ openV V q ⟧V .N-ob t s ≡ K.unlock κ ⟦ V ⟧V .N-ob t (s .fst)
openV-counit (Syn.ne m) q s = refl
openV-counit {Δ} {κ} (Syn.shut V) q s = K.open-shut-at κ (πt Δ) (λ p → ⟦ V p ⟧V) q s

------------------------------------------------------------------------
-- the denotation commutes with substitution

den-subNe : ∀ {Θ Θ' A} (σ : Sub Θ Θ') (n : Ne Θ A) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ subNe σ n ⟧V .N-ob v e ≡ ⟦ n ⟧ne .N-ob v (⟦ σ ⟧s .N-ob v e)
den-subNe ⌜ ρ ⌝ n = den-renNe ρ n
den-subNe (σ ,ₛ V) here v e = refl
den-subNe (σ ,ₛ V) (wkv n) v e = den-subNe σ n v e
den-subNe (lockₛ {Δ} {Δ'} {κ = κ} σ p τ) (up n) v e =
  den-renV τ (↑V p (subNe σ n)) v e
  ∙ den-renV (dropL idR p) (subNe σ n) v (⟦ τ ⟧r .N-ob v e)
  ∙ den-subNe σ n v (K.ε κ ⟦ Δ' ⟧t .N-ob v x)
  ∙ cong (⟦ n ⟧ne .N-ob v) (sym (K.ε-Lmap κ ⟦ σ ⟧s x))
  ∙ sym (K.unlock-box κ ⟦ n ⟧ne (K.Lmap κ ⟦ σ ⟧s .N-ob v x))
  where
  x = ⟦ τ ⟧r .N-ob v e .fst
den-subNe (lockₛ {κ = κ} σ p τ) (opn n) v e =
  den-renV τ (openV (subNe σ n) p) v e
  ∙ openV-counit (subNe σ n) p (⟦ τ ⟧r .N-ob v e)
  ∙ K.unlock-Lmap-at κ ⟦ σ ⟧s ⟦ n ⟧ne ⟦ subNe σ n ⟧V (den-subNe σ n) (⟦ τ ⟧r .N-ob v e .fst)
den-subNe (lockₛ {κ = κ} σ p τ) (nm k) v e =
  den-renNe τ (nm k) v e
  ∙ K.unlock-Lmap-at κ ⟦ σ ⟧s (namesR κ k) (namesR κ k) (λ _ _ → refl) (⟦ τ ⟧r .N-ob v e .fst)
den-subNe (close p f) (up ())
den-subNe (close p f) (opn ())
den-subNe {Θ' = Θ'} (close p f) (nm {b = b} k) v e =
  cong (Const b .F-hom (πt Θ' .N-ob v e)) (cmapW-toW f _)
  ∙ sym (funExt⁻ (Const b .F-seq (toW f) (πt Θ' .N-ob v e)) _)
  where open Syn using (Const)

-- continuations behind a lock word
shutW-sub : ∀ {Θ Θ'} {X : 𝒱} (μ : List Lk) (σ : Sub Θ Θ')
  (k : (ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X)
  (k' : (ps : WPos μ (world Θ')) → PshHom ⟦ Θ' ⧀*⟨ μ , ps ⟩ ⟧t X) →
  (∀ ps v e → k' ps .N-ob v e ≡ k (wposW μ (wmS σ) ps) .N-ob v (⟦ liftWS μ σ ps ⟧s .N-ob v e)) →
  ∀ v e → shutW μ k' .N-ob v e ≡ shutW μ k .N-ob v (⟦ σ ⟧s .N-ob v e)
shutW-sub [] σ k k' hyp v e = funExt λ _ → hyp _ v e
shutW-sub {Θ} {Θ'} (κ ∷ []) σ k k' hyp v e = funExt λ q →
  hyp _ _ (K.Over.inj κ (πt Θ') e q)
  ∙ sym (K.Over.copair-at κ (πt Θ) k _ _ _)
  ∙ K.Over.copair-at κ (πt Θ) k _ _ _
shutW-sub {Θ} {Θ'} (κ ∷ μ@(_ ∷ _)) σ k k' hyp v e = funExt λ (q , ps) →
  funExt⁻ (shutW-sub μ (lockₛ σ _ idR) (λ ps → k (_ , ps)) (λ ps → k' (_ , ps))
            (λ ps → hyp (_ , ps)) _ (K.Over.inj κ (πt Θ') e q)) ps
  ∙ funExt⁻ (sym (K.Over.copair-at κ (πt Θ) φ _ _ _) ∙ K.Over.copair-at κ (πt Θ) φ _ _ _) ps
  where
  φ : (p : Pos κ (world Θ)) → PshHom ⟦ Θ ⧀⟨ κ , p ⟩ ⟧t (KW.R μ _)
  φ p = shutW μ (λ ps → k (p , ps))

den-subV : ∀ {Θ Θ' A} (σ : Sub Θ Θ') (V : Val Θ A) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ subV σ V ⟧V .N-ob v e ≡ ⟦ V ⟧V .N-ob v (⟦ σ ⟧s .N-ob v e)
den-subVs : ∀ {Θ Θ' bs} (σ : Sub Θ Θ') (vs : Vals Θ bs) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ subVs σ vs ⟧Vs .N-ob v e ≡ ⟦ vs ⟧Vs .N-ob v (⟦ σ ⟧s .N-ob v e)
den-subC : ∀ {Θ Θ' B} (σ : Sub Θ Θ') (M : Comp Θ B) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ subC σ M ⟧C .N-ob v e ≡ ⟦ M ⟧C .N-ob v (⟦ σ ⟧s .N-ob v e)

den-subV σ (Syn.ne n) = den-subNe σ n
den-subV σ Syn.tt v e = refl
den-subV σ (Syn.inl V) v e = cong inl (den-subV σ V v e)
den-subV σ (Syn.inr V) v e = cong inr (den-subV σ V v e)
den-subV σ (Syn.pair V V') v e = cong₂ _,_ (den-subV σ V v e) (den-subV σ V' v e)
den-subV σ (Syn.thunk M) = den-subC σ M
den-subV {Θ} {Θ'} σ (Syn.const {b} c) v e =
  cong (Const b .F-hom (πt Θ' .N-ob v e)) (cmapW-toW (wmS σ) c)
  ∙ sym (funExt⁻ (Const b .F-seq (toW (wmS σ)) (πt Θ' .N-ob v e)) c)
  ∙ cong (λ h → Const b .F-hom h c) (sym (sub-over σ v e))
  where open Syn using (Const)
den-subV {Θ} {Θ'} σ (Syn.shut {κ} V) v e = funExt λ q →
  den-subV (lockₛ σ _ idR) (V _) _ (K.Over.inj κ (πt Θ') e q)
  ∙ sym (K.Over.copair-at κ (πt Θ) (λ p → ⟦ V p ⟧V) _ _ _)
  ∙ K.Over.copair-at κ (πt Θ) (λ p → ⟦ V p ⟧V) _ _ _

den-subVs σ Syn.[]ᵥ v e = refl
den-subVs σ (V Syn.∷ᵥ vs) v e = cong₂ _,_ (den-subV σ V v e) (den-subVs σ vs v e)

den-subC σ (Syn.return V) v e = cong var (den-subV σ V v e)
den-subC {Θ} {Θ'} σ (Syn._to_ {A} {B} M N) v e =
  cong (λ t → bind ⟦ Θ' ⟧t ⟦ A ⟧v ⟦ B ⟧c ⟦ subC (liftS σ) N ⟧C .N-ob v (t , e))
    (den-subC σ M v e)
  ∙ bind-pre ⟦ Θ ⟧t ⟦ Θ' ⟧t ⟦ A ⟧v ⟦ B ⟧c ⟦ σ ⟧s ⟦ N ⟧C ⟦ subC (liftS σ) N ⟧C
      (λ u x' a → den-subC (liftS σ) N u (x' , a) ∙ cong (⟦ N ⟧C .N-ob u) (sub-lift σ u x' a))
      v (⟦ M ⟧C .N-ob v (⟦ σ ⟧s .N-ob v e)) e
den-subC σ (Syn.force V) = den-subV σ V
den-subC σ (Syn.case V M) v e =
  den-subC (liftS (liftS σ)) M v ((e , y .fst) , y .snd)
  ∙ cong (⟦ M ⟧C .N-ob v)
      (sub-lift (liftS σ) v (e , y .fst) (y .snd) ∙ cong (_, y .snd) (sub-lift σ v e (y .fst)))
  ∙ cong (λ y → ⟦ M ⟧C .N-ob v ((⟦ σ ⟧s .N-ob v e , y .fst) , y .snd)) (den-subV σ V v e)
  where
  y = ⟦ subV σ V ⟧V .N-ob v e
den-subC σ (Syn.case₊ V M N) v e =
  cong (λ s → casePsh ⟦ subC (liftS σ) M ⟧C ⟦ subC (liftS σ) N ⟧C .N-ob v (e , s)) (den-subV σ V v e)
  ∙ casePsh-pt ⟦ subC (liftS σ) M ⟧C ⟦ subC (liftS σ) N ⟧C ⟦ M ⟧C ⟦ N ⟧C e (⟦ σ ⟧s .N-ob v e)
      (λ a → den-subC (liftS σ) M v (e , a) ∙ cong (⟦ M ⟧C .N-ob v) (sub-lift σ v e a))
      (λ a → den-subC (liftS σ) N v (e , a) ∙ cong (⟦ N ⟧C .N-ob v) (sub-lift σ v e a))
      (⟦ V ⟧V .N-ob v (⟦ σ ⟧s .N-ob v e))
den-subC σ (Syn.absurd V) v e = ⊥.rec (⟦ V ⟧V .N-ob v (⟦ σ ⟧s .N-ob v e))
den-subC σ (Syn.lam M) v e = makePshHomPath (funExt λ u → funExt λ (h , a) →
  den-subC (liftS σ) M u (_ , a)
  ∙ cong (⟦ M ⟧C .N-ob u) (sub-lift σ u _ a)
  ∙ cong (λ y → ⟦ M ⟧C .N-ob u (y , a)) (⟦ σ ⟧s .N-hom _ _ h e))
den-subC σ (Syn.app M V) v e =
  cong₂ (λ f a → f .N-ob v (W.id , a)) (den-subC σ M v e) (den-subV σ V v e)
den-subC {Θ} {Θ'} σ (Syn.op {B} o vs k) v e =
  cong (Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob v)
    (ΣPathP (ΣPathP (refl , den-subVs σ vs v e) , funExt λ (a , ps) →
      funExt⁻ (shutW-sub (slot o a) σ (k' a) (λ ps → ⟦ subC (liftWS (slot o a) σ ps) (k a _) ⟧C)
        (λ ps → den-subC (liftWS (slot o a) σ ps) (k a _)) v e) ps))
  where
  k' : ∀ a (ps : WPos (slot o a) (world Θ)) → PshHom ⟦ Θ ⧀*⟨ slot o a , ps ⟩ ⟧t (Algebra.Carrier ⟦ B ⟧c)
  k' a ps = ⟦ k a ps ⟧C

------------------------------------------------------------------------
-- single and double substitution extend the environment by the value

den-sub₁ : ∀ {Θ A B} (M : Comp (Θ ▸ A) B) (V : Val Θ A) v (e : El ⟦ Θ ⟧t v) →
  ⟦ sub₁ M V ⟧C .N-ob v e ≡ ⟦ M ⟧C .N-ob v (e , ⟦ V ⟧V .N-ob v e)
den-sub₁ M V = den-subC (idS ,ₛ V) M

den-sub₂ : ∀ {Θ A A' B} (M : Comp ((Θ ▸ A) ▸ A') B) (V : Val Θ A) (V' : Val Θ A') v
  (e : El ⟦ Θ ⟧t v) →
  ⟦ sub₂ M V V' ⟧C .N-ob v e ≡ ⟦ M ⟧C .N-ob v ((e , ⟦ V ⟧V .N-ob v e) , ⟦ V' ⟧V .N-ob v e)
den-sub₂ M V V' = den-subC (idS ,ₛ V ,ₛ V') M
