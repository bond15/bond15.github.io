-- Normalization of the Boolean global-state machine
-- (Instances.Machines.GlobalState).
--
--   finOp     get has two continuations, put b one
--   state-sn  every configuration is strongly normalizing, at every result
--             type A
--   state-wn  every configuration reaches a terminal one, at every A
--   state-ground-canonicity
--             at a ground type A, every configuration reaches a unique
--             final one (Semantics.ConfigCanonicity at the runner)
--
-- The runs toggle-run and toggle-run-false end in terminal
-- configurations (toggle-normal, toggle-false-normal), and these are the
-- only final configurations toggle reaches (toggle-unique,
-- toggle-false-unique).
module RMRCanonicity.PRACBPVGeneric.Instances.Normalization.GlobalState where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Equiv using (invEquiv)
open import Cubical.Data.Unit using (Unit; tt; tt*)
open import Cubical.Data.Sum using (inl)
open import Cubical.Data.Empty using (⊥)
open import Cubical.Data.Bool using (Bool; true; false)
open import Cubical.Data.Sigma using (Σ-syntax; _×_)
open import Cubical.Data.SumFin.Properties using (SumFin2≃Bool)
open import Cubical.Data.FinSet.Base using (isFinSet; isFinOrd; isFinOrd→isFinSet)
open import Cubical.Data.FinSet.Properties using (isFinOrdUnit)
open import Cubical.Data.FinSet.Constructors using (isFinOrdΣ)
open import Cubical.Induction.WellFounded using (Acc)
open import Cubical.Categories.Instances.Terminal using (TerminalCategory)

open import RMRCanonicity.PRACBPVGeneric.Instances.Constant
  using (noLocks; stateOps; get; put; module GlobalState)
open GlobalState hiding (_,_; _×_)
open import RMRCanonicity.PRACBPVGeneric.Instances.Machines.GlobalState
  using (isSetLk; isSetBase; S; isSetConf; isSet⊥; δ; toggle-run; toggle-run-false)
import RMRCanonicity.PRACBPVGeneric.Semantics.RunNormalization as RunNormalization
open import RMRCanonicity.PRACBPVGeneric.Semantics.Ground (noLocks TerminalCategory) stateOps
  using (Ground; g𝟙; reflect; reflect-den)

isFinOrdBool : isFinOrd Bool
isFinOrdBool = 2 , invEquiv SumFin2≃Bool

-- get continues in one of two slots, put b in one
finOp : ∀ o w → isFinSet (Σ[ a ∈ Ar o ] WPos (slot o a) w)
finOp get w = isFinOrd→isFinSet (isFinOrdΣ Bool isFinOrdBool _ λ _ → isFinOrdUnit)
finOp (put b) w = isFinOrd→isFinSet (isFinOrdΣ Unit isFinOrdUnit _ λ _ → isFinOrdUnit)

module _ {A : ValTy} where
  open RunNormalization (noLocks TerminalCategory) stateOps isSetLk isSetBase S ⊥ isSetConf isSet⊥ δ A finOp

  state-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  state-sn = strongNormalization

  state-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  state-wn = weakNormalization

  state-ground-canonicity : Ground A → ∀ x →
    Σ[ ṽ ∈ Final ] (x ⟶* retF ṽ) × (∀ ṽ' → x ⟶* retF ṽ' → ṽ ≡ ṽ')
  state-ground-canonicity = ground-canonicity

-- the runs (Instances.Machines.GlobalState) are at A = 𝟙
open RunNormalization (noLocks TerminalCategory) stateOps isSetLk isSetBase S ⊥ isSetConf isSet⊥ δ 𝟙 finOp

-- toggle from true ends returning, at state false
toggle-normal : Σ[ y ∈ Config ] (running true toggle ⟶* y) × Terminal y
toggle-normal = running false (return tt) , toggle-run , running-terminal false tt

toggle-false-normal : Σ[ y ∈ Config ] (running false toggle ⟶* y) × Terminal y
toggle-false-normal = running true (return tt) , toggle-run-false , running-terminal true tt

-- every run of toggle to a final configuration ends where toggle-run does
-- (final-unique at the ground type 𝟙)
toggle-unique : ∀ ṽ → running true toggle ⟶* retF ṽ → ṽ ≡ (inl (tt* , false) , λ _ → tt)
toggle-unique ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p toggle-run

toggle-false-unique : ∀ ṽ → running false toggle ⟶* retF ṽ → ṽ ≡ (inl (tt* , true) , λ _ → tt)
toggle-false-unique ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p toggle-run-false
