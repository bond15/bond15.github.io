-- Tree reduction: β-steps (to, force, case, case₊ at inl and inr, app;
-- absurd has none, as no closed value has type 𝟘), and frame steps that push one stack frame into
-- every continuation of an operation, through its locks.
--
--   (op o vs k) to x. N   ⇝  op o vs (λ a ps → k a ps to x. N⟨dropW ps⟩)
--   (op o vs k) V         ⇝  op o vs (λ a ps → k a ps (V⟨dropW ps⟩))
--
-- where dropW ps : Ren Θ (Θ ⧀*⟨ μ , ps ⟩) crosses the locks of the slot's
-- word (Kripke access along the composite strength).
--
-- M ↦ M' is a head step inside a stack; M ↦* M' is its reflexive-transitive
-- closure, plus congruence under op, continuation by continuation (op*).
--
-- There is no close rule.  An op's continuations sit behind the locks of
-- their slot's word, and tree reduction never removes those locks.  Closing
-- them (turning a closed op into an algebra node at the root) is a step of
-- the combined machine: Closing.closeOp and the close step of
-- Semantics.Machine.
open import Cubical.Foundations.Prelude using (Type; ℓ-zero)
open import Cubical.Categories.Category using (Category)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Data.Sigma using () renaming (_,_ to _,ˢ_)

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Reduction {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O

-- across the locks of a word (iterated dropL)
dropW : ∀ {Θ Δ} (ρ : Ren Θ Δ) (μ : List Lk) (ps : WPos μ (world Δ)) → Ren Θ (Δ ⧀*⟨ μ , ps ⟩)
dropW ρ [] _ = ρ
dropW ρ (κ ∷ []) p = dropL ρ p
dropW ρ (κ ∷ μ@(_ ∷ _)) (p ,ˢ ps) = dropW (dropL ρ p) μ ps

infix 4 _⇝_ _↦_ _↦*_

data _⇝_ {Θ : Tele} {B : CompTy} : Comp Θ B → Comp Θ B → Type where
  β-to : ∀ {A} {V : Val Θ A} {N : Comp (Θ , A) B} → (return V to N) ⇝ sub₁ N V
  β-force : ∀ {M : Comp Θ B} → force (thunk M) ⇝ M
  β-case : ∀ {A A'} {V : Val Θ A} {V' : Val Θ A'} {M : Comp (Θ , A , A') B} →
    case (pair V V') M ⇝ sub₂ M V V'
  β-case₊-inl : ∀ {A A'} {V : Val Θ A} {M : Comp (Θ , A) B} {N : Comp (Θ , A') B} →
    case₊ (inl V) M N ⇝ sub₁ M V
  β-case₊-inr : ∀ {A A'} {V : Val Θ A'} {M : Comp (Θ , A) B} {N : Comp (Θ , A') B} →
    case₊ (inr V) M N ⇝ sub₁ N V
  β-app : ∀ {A} {M : Comp (Θ , A) B} {V : Val Θ A} → app (lam M) V ⇝ sub₁ M V
  op-to : ∀ {A} (o : Op) (vs : Vals Θ (params o))
    {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (F A)}
    {N : Comp (Θ , A) B} →
    (op o vs k to N) ⇝ op o vs (λ a ps → k a ps to renC (keep (dropW idR (slot o a) ps)) N)
  op-app : ∀ {A} (o : Op) (vs : Vals Θ (params o))
    {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (A ⇒ B)}
    {V : Val Θ A} →
    app (op o vs k) V ⇝ op o vs (λ a ps → app (k a ps) (renV (dropW idR (slot o a) ps) V))

-- tree reduction: a head step inside a stack
data _↦_ {Θ : Tele} {B' : CompTy} : Comp Θ B' → Comp Θ B' → Type where
  step : ∀ {B} (S : Stack Θ B B') {M M' : Comp Θ B} → M ⇝ M' → plug S M ↦ plug S M'

-- many steps, with congruence under op continuation by continuation
data _↦*_ {Θ : Tele} {B : CompTy} : Comp Θ B → Comp Θ B → Type where
  refl* : ∀ {M} → M ↦* M
  step* : ∀ {M M' M''} → M ↦ M' → M' ↦* M'' → M ↦* M''
  op* : ∀ {o : Op} {vs : Vals Θ (params o)}
    {k k' : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B} →
    (∀ a ps → k a ps ↦* k' a ps) → op o vs k ↦* op o vs k'
