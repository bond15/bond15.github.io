-- Writer as Q = T with its equations (POPL's Instances/Writer; paper
-- §2.1 "Theory of Writer", §2.2 its free model, §5.3 the Writer example).
--
-- W = 1, no locks.  For a monoid (M, ·, ε) (a set), one operation tell_m
-- for each m : M, with one continuation, and the equations
--
--   tell_ε x = x              tell_m (tell_n x) = tell_{m · n} x
--
-- Free model: T X = M × X, tell_n (m , x) = (n · m , x), η x = (ε , x).
-- As a container: shapes M, one position each (Ext Z = Σ_{m ∈ M} (1 → Z)).
-- Configurations are T (closed computations): an output and one thread,
-- with ρ = μ (Instances.Equational.Common, Semantics.PolySelfModule).
--
--   writerEqs, writerFree, freeModel, Q′, polyModule
--   writer-affine     one hole per configuration (affine-subsingleton)
--   writer-finQ, writer-finOp
--   writer-effect     POPL's derived rule  ⟨m , tell_n M⟩ ⟶ ⟨m · n , M⟩
--   writer-sn, writer-wn, writer-ground-canonicity
--   writer-run        ⟨ε , tell_m (tell_n (return tt))⟩ ⟶* ⟨(ε · m) · n , return tt⟩,
--                     the only final configuration (writer-run-unique),
--                     whose values are read off the free-model denotation
--                     (writer-run-comparison)

------------------------------------------------------------------------
-- the theory

-- one variable
𝟏 : 𝒱
𝟏 .F-ob _ = Unit , isSetUnit
𝟏 .F-hom _ _ = tt
𝟏 .F-id = refl
𝟏 .F-seq _ _ = refl

tellT : ∀ {X w} → Out → Term (PM.polynomial L₁ O₁) X w → Term (PM.polynomial L₁ O₁) X w
tellT m t = ops (m , tt) λ _ → t

data WriterEq : Type where
  tell-ε : WriterEq
  tell-tell : Out → Out → WriterEq

-- tell_ε x = x ;  tell_m (tell_n x) = tell_{m · n} x
writerEqs : Machine.Equations L₁ O₁
writerEqs .Machine.Equations.Equation = WriterEq
writerEqs .Machine.Equations.equationWorld _ = tt*
writerEqs .Machine.Equations.Variables _ = 𝟏
writerEqs .Machine.Equations.lhs tell-ε = tellT ε (var tt)
writerEqs .Machine.Equations.rhs tell-ε = var tt
writerEqs .Machine.Equations.lhs (tell-tell m n) = tellT m (tellT n (var tt))
writerEqs .Machine.Equations.rhs (tell-tell m n) = tellT (m · n) (var tt)

------------------------------------------------------------------------
-- the free model M × X, presented by the container (shapes M, one
-- position each)

Sh : hSet ℓ-zero
Sh = Out , is-set

Ps : Out → hSet ℓ-zero
Ps _ = Unit , isSetUnit

open C.Presented writerEqs Sh Ps (λ _ _ _ → yes refl)

-- tell_n (m , x) = (n · m , x)
opW : ∀ {Z} → Out → (Unit → Ext Z) → Ext Z
opW n k = n · k tt .fst , k tt .snd

open WithOps ε opW (λ g n k → refl)

-- the laws in any model
module Laws (N : Model) where
  tell : Out → ∣ N ∣ → ∣ N ∣
  tell m x = Nalg N m λ _ → x

  tell-ε-law : ∀ x → tell ε x ≡ x
  tell-ε-law x = N .snd tell-ε (naturally _ _ λ _ _ → x)

  tell-tell-law : ∀ m n x → tell m (tell n x) ≡ tell (m · n) x
  tell-tell-law m n x = N .snd (tell-tell m n) (naturally _ _ λ _ _ → x)

-- POPL's writerFree: the extension of f is (m , x) ↦ tell_m (f x)
writerFree : PolyFree
writerFree .PolyFree.sat X tell-ε env = ΣPathP (·IdL _ , refl)
writerFree .PolyFree.sat X (tell-tell m n) env = ΣPathP (·Assoc m n _ , refl)
writerFree .PolyFree.ext N f t = Laws.tell N (t .fst) (f (t .snd tt))
writerFree .PolyFree.ext-η N f z = Laws.tell-ε-law N (f z)
writerFree .PolyFree.ext-hom N f n k = sym (Laws.tell-tell-law N n (k tt .fst) (f (k tt .snd tt)))
writerFree .PolyFree.ext-unique N f h hh hη t =
  cong h (ΣPathP (sym (·IdR (t .fst)) , refl))
  ∙ hh (t .fst) (λ _ → ε , t .snd)
  ∙ cong (Laws.tell N (t .fst)) (hη (t .snd tt))

open WithFree writerFree

------------------------------------------------------------------------
-- finiteness

writer-finQ : ∀ (s : Out) → isFinOrd Unit
writer-finQ _ = isFinOrdUnit

writer-finOp = finOpOf λ _ → isFinOrdUnit

-- every theory is closed under exponentials at W = 1
writer-ec : StrongTheory.ExpClosed L₁ O₁ writerEqs
writer-ec = StrongTheory.expClosed-chaotic L₁ O₁ chaotic writerEqs

------------------------------------------------------------------------
-- the machine

module _ {A : ValTy} where
  open PolyMachine L₁ O₁ isSetLk isSetBase writerEqs freeModel Q′ tt* presentation A

  -- ρ = μ
  writer-module : RightModule (PS.ER.T) (PS.ER.Q)
  writer-module = PS.polyModule presentation

  -- one hole per configuration
  writer-affine : Affine
  writer-affine = affine-subsingleton λ _ → isPropUnit

  open WithAffinity writer-finOp writer-finQ writer-affine

  -- an output and one thread
  conf : Out → Comp (∅ tt*) (F A) → Config
  conf m M = m , λ _ → M

  -- POPL's derived rule:  ⟨m , tell_n M⟩ ⟶ ⟨m · n , M⟩
  writer-effect : (m n : Out) (k : Unit → Unit → Comp (∅ tt*) (F A)) →
    conf m (op n []ᵥ k) ⟶ conf (m · n) (k tt tt)
  writer-effect m n k =
    subst (conf m (op n []ᵥ k) ⟶_)
      (cong out (spliced-hole (conf m (op n []ᵥ k)) tt ((n , tt) , λ q → k (q .fst) (q .snd)))
       ∙ ΣPathP (cong (m ·_) (·IdR n) , refl))
      (effect-in (conf m (op n []ᵥ k)) tt refl)
    where
    -- μ at the shape m: tell_m
    out : Ext (Comp (∅ tt*) (F A)) → Config
    out y = m · y .fst , y .snd

  writer-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  writer-sn = strongNormalization

  writer-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  writer-wn = weakNormalization

  writer-ground-canonicity : Ground A → (M : Comp (∅ tt*) (F A)) →
    Σ[ ṽ ∈ Final ] (ηC M ⟶* retF ṽ) × (∀ ṽ' → ηC M ⟶* retF ṽ' → ṽ ≡ ṽ')
  writer-ground-canonicity = ground-canonicity-T

------------------------------------------------------------------------
-- a run

open PolyMachine L₁ O₁ isSetLk isSetBase writerEqs freeModel Q′ tt* presentation 𝟙
open Comparing writer-ec using (module Cmp; module Fr; final-comparison)

-- tell_m (tell_n (return tt))
twoTells : Out → Out → Comp (∅ tt*) (F 𝟙)
twoTells m n = op m []ᵥ λ _ _ → op n []ᵥ λ _ _ → return tt

writer-run : ∀ m n → ηC (twoTells m n) ⟶* retF ((ε · m) · n , λ _ → tt)
writer-run m n = writer-effect ε m _ then writer-effect (ε · m) n _ then done

-- the only final configuration
writer-run-unique : ∀ m n ṽ → ηC (twoTells m n) ⟶* retF ṽ → ṽ ≡ ((ε · m) · n , λ _ → tt)
writer-run-unique m n ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p (writer-run m n)

-- its values, read off the free-model denotation
writer-run-comparison : ∀ m n →
  (ER.Q ⟪ PshHom→NatTrans (Cmp.same Cmp.fo𝟙) ⟫) (denF ((ε · m) · n , λ _ → tt))
    ≡ from Fr.⟦ 𝟙 ⟧v (Fr.⟦ twoTells m n ⟧C .PshHom.N-ob tt* refl)
writer-run-comparison m n = final-comparison Cmp.fo𝟙 (twoTells m n) _ (writer-run m n)
