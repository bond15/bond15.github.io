-- Theories whose models are closed under exponentials.
--
-- The free-model denotation (Semantics.Free.Denotation) interprets
-- sequencing  M to N  in an environment G by extending N along the free
-- model's universal property INTO THE EXPONENTIAL ALGEBRA  G ⇒ ⟦ B ⟧
-- (Semantics.Algebra.expAlg: an operation acts pointwise, the argument moved
-- into each continuation along the polynomial's strength ι), and it
-- interprets  A ⇒ B  by that exponential.  Both need the exponential of a
-- model to be a model again:
--
--   ExpClosed eqs  =  ∀ G B →  B satisfies the theory  →  G ⇒ B satisfies it
--
-- This is what makes the free-model monad strong for the polynomial's
-- strength (the strong monad of Moggi's semantics, which sequencing in a
-- context needs).  POPL works at W = 1, where every theory has it.  Here:
--
--   expClosed-noEquations   an equation-free theory has it, at every W
--   expClosed-chaotic       every theory has it when W is chaotic (exactly
--                           one map between any two worlds, e.g. W = 1)
--
-- At a general W it is a real condition.  The argument in G ⇒ B is moved
-- along ι, so an equation between two operations whose continuations live
-- at the same world but are reached along different maps f, g : w → w'
-- (op_f x = op_g x) can hold in B and fail in G ⇒ B for G = y w.  We have
-- not built such a lock signature; the condition is stated, not refuted.
--
-- The chaotic case.  Fix a world v and γ : G v.  An environment ρ into
-- G ⇒ B gives the environment  ρ^γ x = ρ x (id , γ moved to x's world)
-- into B (natural because all maps are equal), and evaluating a term in
-- G ⇒ B at (h , γ) is evaluating it in B at ρ^γ, moved along h
-- (eval-chaotic).  So the equation transfers from B.
open import Cubical.Foundations.Prelude

module RMRCanonicity.PRACBPVGeneric.Semantics.StrongTheory
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Semantics.Denotation L O
  using (𝒱; El; Exp; expAlg; expAlgebra-ob; ι; tpos; smapP)
open Machine L O using (Equations; theory; noEquations)

open Functor
open PshHom

private
  module W = Category W
  module SN = SignatureNotation (PM.polynomial L O)

----------------------------------------------------------------------
-- the condition

ExpClosed : Equations → Type₁
ExpClosed eqs = (G : 𝒱) (B : Algebra (PM.polynomial L O)) →
  Satisfies (theory eqs) B → Satisfies (theory eqs) (expAlg G B)

----------------------------------------------------------------------
-- an equation-free theory

expClosed-noEquations : ExpClosed noEquations
expClosed-noEquations G B sat ()

----------------------------------------------------------------------
-- a chaotic world category

record Chaotic : Type where
  field
    the : ∀ w w' → W [ w , w' ]
    unique : ∀ {w w'} (f : W [ w , w' ]) → f ≡ the w w'

module _ (ch : Chaotic) (eqs : Equations) where
  open Chaotic ch

  module _ (G : 𝒱) (B : Algebra (PM.polynomial L O)) {X : 𝒱}
    (ρ : NatTrans X (Algebra.Carrier (expAlg G B))) {v : W.ob} (γ : El G v) where

    private
      CB = Algebra.Carrier B
      algB = Algebra.algebra B .NatTrans.N-ob

    -- γ, moved to the world u
    γ-at : ∀ u → El G u
    γ-at u = G .F-hom (the v u) γ

    -- moving γ-at u along f gives γ-at u'
    γ-at-hom : ∀ {u u'} (f : W [ u , u' ]) → G .F-hom f (γ-at u) ≡ γ-at u'
    γ-at-hom {u} {u'} f = sym (funExt⁻ (G .F-seq (the v u) f) γ) ∙ cong (λ m → G .F-hom m γ) (unique _)

    -- ρ read at γ: an environment into B
    ρ^ : NatTrans X CB
    ρ^ .NatTrans.N-ob u x = ρ .NatTrans.N-ob u x .N-ob u (W.id , γ-at u)
    ρ^ .NatTrans.N-hom {u} {u'} f = funExt λ x →
      (λ i → ρ .NatTrans.N-hom f i x .N-ob u' (W.id , γ-at u'))
      ∙ cong₂ (λ m y → ρ .NatTrans.N-ob u x .N-ob u' (m , y))
          (W.⋆IdR f ∙ sym (W.⋆IdL f)) (sym (γ-at-hom f))
      ∙ ρ .NatTrans.N-ob u x .N-hom u' u f (W.id , γ-at u)

    -- evaluating in G ⇒ B at (h , γ) is evaluating in B at ρ^, moved along h
    eval-chaotic : ∀ {u} (t : Term (PM.polynomial L O) X u) {u''} (h : W [ u , u'' ]) →
      evaluate (expAlg G B) ρ .NatTrans.N-ob u t .N-ob u'' (h , γ-at u'')
        ≡ CB .F-hom h (evaluate B ρ^ .NatTrans.N-ob u t)
    eval-chaotic {u} (var x) {u''} h =
      cong₂ (λ m y → ρ .NatTrans.N-ob u x .N-ob u'' (m , y)) (sym (W.⋆IdL h)) (sym (γ-at-hom h))
      ∙ ρ .NatTrans.N-ob u x .N-hom u'' u h (W.id , γ-at u)
    eval-chaotic {u} (ops o args) {u''} h =
      expAlgebra-ob G B u _ u'' h (γ-at u'')
      ∙ cong (λ c → algB u'' (Poly.Shapes (PM.polynomial L O) .F-hom h o , c)) (funExt λ p →
          cong (λ y → evaluate (expAlg G B) ρ .NatTrans.N-ob _ (args (tpos h o p)) .N-ob _ (smapP h o p W.⋆ W.id , y))
            (γ-at-hom (ι u'' (Poly.Shapes (PM.polynomial L O) .F-hom h o) p))
          ∙ eval-chaotic (args (tpos h o p)) (smapP h o p W.⋆ W.id)
          ∙ cong (λ m → CB .F-hom m (evaluate B ρ^ .NatTrans.N-ob _ (args (tpos h o p)))) (W.⋆IdR _))
      ∙ funExt⁻ (Algebra.algebra B .NatTrans.N-hom h)
          (o , λ p → evaluate B ρ^ .NatTrans.N-ob _ (args p))

  -- every theory is closed under exponentials over a chaotic W
  expClosed-chaotic : ExpClosed eqs
  expClosed-chaotic G B sat e ρ = makePshHomPath (funExt λ v → funExt λ (h , γ) →
    cong (λ y → ev (Equations.lhs eqs e) .N-ob v (h , y)) (sym (γ-id γ))
    ∙ eval-chaotic G B ρ γ (Equations.lhs eqs e) h
    ∙ cong (Algebra.Carrier B .F-hom h) (sat e (ρ^ G B ρ γ))
    ∙ sym (eval-chaotic G B ρ γ (Equations.rhs eqs e) h)
    ∙ cong (λ y → ev (Equations.rhs eqs e) .N-ob v (h , y)) (γ-id γ))
    where
    ev = evaluate (expAlg G B) ρ .NatTrans.N-ob (Equations.equationWorld eqs e)

    γ-id : ∀ {v} (γ : El G v) → G .F-hom (the v v) γ ≡ γ
    γ-id {v} γ = cong (λ m → G .F-hom m γ) (sym (unique W.id)) ∙ funExt⁻ (G .F-id) γ
