-- The equational theory of PRACBPV over a theory (POPL's Equational).
--
-- As in POPL (its D4), terms stay a plain inductive type and the theory is
-- an inductive relation on them, at every telescope:
--
--   V ≈v V'    values        vs ≈vs vs'   parameter lists (pointwise)
--   M ≈c M'    computations
--
-- an equivalence relation, a congruence for every term former, and the
-- axioms below (the standard full set of Levy, as POPL's D5).
--
--   β   (return V) to N = N[V]           force (thunk M) = M
--       case (pair V V') M = M[V, V']    case₊ (inl V) M N = M[V]
--       case₊ (inr V) M N = N[V]         app (lam M) V = M[V]
--   η   V = tt  (V : 𝟙)                  thunk (force V) = V
--       shut (λ q. openV V q) = V  (V : [ κ ] A)
--       M[V] = absurd V  (V : 𝟘)
--       M[V] = case₊ V (M[inl x]) (M[inr y])      M[V] = case V (M[pair x y])
--       lam (app (wk M) x) = M
--       S[M] = M to S[return x]   (F, in the general stack form)
--   algebraicity   S[op o vs k] = op o vs (λ a ps. S⟨across the locks⟩[k a ps])
--   the theory     every equation, instantiated in the syntax
--
-- The lock modality.  [ κ ] A has introduction shut and elimination openV
-- (a neutral opn on variables; at shut it resolves).  Its β law
-- openV (shut V) q = V q holds by definition ([κ]-β below); its η law is
-- the axiom η[κ].  open≈ shows openV is a congruence.
--
-- There are no value eliminators (no complex values), so the η laws of 𝟘,
-- + and × are stated for computations only, in substitution form (POPL's
-- η𝟘c, η+c, η×c); their values-form instances (η𝟘v, η+v, η×v) have no
-- counterpart.  op is algebraic for every stack (POPL's op-stack): the
-- stack is moved across the locks of each slot's word (dropW), as in the
-- frame steps of Reduction; op-to and op-app are its instances (⇝→≈).
--
-- THE THEORY'S EQUATIONS.  An equation e of a theory over the polynomial
-- (Machine.Equations) is a pair of polynomial terms at the world
-- equationWorld e, over a presheaf of variables.  The syntactic model of
-- the polynomial is the closed computations, with the operation node
-- opNodeC (Closing): a polynomial term evaluates to a closed computation
-- (inst) once its variables are sent to closed computations, naturally in
-- the world (ClosedEnv: this is POPL's γ, made natural, as the semantic
-- Satisfies requires).  The axiom `theory` is that equation, moved into
-- any telescope Θ along a world map g by the root renaming:
--
--   renC (root g) (inst γ (lhs e)) ≈c renC (root g) (inst γ (rhs e))
--
-- With g the identity this is the equation itself at ∅ (equationWorld e);
-- in general it is its instance at a world reached from the equation's
-- world.  Stating it under root g makes ≈ closed under substitution
-- (Metatheory.EqSubstitution).  POPL instantiates its equations with open
-- computations; here the leaves are closed: at general W an environment
-- into open computations is not a natural transformation, so Satisfies
-- would not make such an instance sound.
--
-- Derived: ≡→≈v, ≡→≈c, ≈vs-refl, [κ]-β, open≈, op-to≈ and op-app≈
-- (algebraicity for sequencing and application), plug≈, sub₁≈, and tree
-- reduction inside ≈ (⇝→≈, ↦→≈, ↦*→≈).

----------------------------------------------------------------------
-- the theory's equations in the syntax

-- a natural assignment of closed computations to the variables
record ClosedEnv (X : 𝒱) (B : CompTy) : Type where
  field
    at : ∀ {u} → El X u → Comp (∅ u) B
    nat : ∀ {u u'} (f : W [ u , u' ]) (x : El X u) → at (X .F-hom f x) ≡ reindexC f (at x)

open ClosedEnv public

-- a polynomial term, evaluated in the closed computations (opNodeC)
inst : ∀ {X B u} → ClosedEnv X B → Term (PM.polynomial L O) X u → Comp (∅ u) B
inst γ (var x) = γ .at x
inst γ (ops (o ,ˢ cs) args) = opNodeC o cs λ a ps → inst γ (args (a ,ˢ ps))

----------------------------------------------------------------------
-- the substitutions of the η laws

-- z ↦ inl x,  z ↦ inr y,  z ↦ pair x y
inlS : ∀ {Θ A A'} → Sub (Θ , (A ⊕ A')) (Θ , A)
inlS = wkS idS ,ₛ inl (ne here)

inrS : ∀ {Θ A A'} → Sub (Θ , (A ⊕ A')) (Θ , A')
inrS = wkS idS ,ₛ inr (ne here)

pairS : ∀ {Θ A A'} → Sub (Θ , (A × A')) (Θ , A , A')
pairS = wkS (wkS idS) ,ₛ pair (ne (wkv here)) (ne here)

----------------------------------------------------------------------
-- the relation

infix 3 _≈v_ _≈vs_ _≈c_
infixr 5 _∷≈_

data _≈v_ : ∀ {Θ A} → Val Θ A → Val Θ A → Type
data _≈vs_ : ∀ {Θ bs} → Vals Θ bs → Vals Θ bs → Type
data _≈c_ : ∀ {Θ B} → Comp Θ B → Comp Θ B → Type

data _≈v_ where
  -- equivalence
  ≈refl : ∀ {Θ A} {V : Val Θ A} → V ≈v V
  ≈sym : ∀ {Θ A} {V V' : Val Θ A} → V ≈v V' → V' ≈v V
  ≈trans : ∀ {Θ A} {U V V' : Val Θ A} → U ≈v V → V ≈v V' → U ≈v V'
  -- congruence (ne, tt and const have no subterms)
  inl≈ : ∀ {Θ A A'} {V V' : Val Θ A} → V ≈v V' → inl {A' = A'} V ≈v inl V'
  inr≈ : ∀ {Θ A A'} {V V' : Val Θ A'} → V ≈v V' → inr {A = A} V ≈v inr V'
  pair≈ : ∀ {Θ A A'} {V V' : Val Θ A} {U U' : Val Θ A'} → V ≈v V' → U ≈v U' → pair V U ≈v pair V' U'
  thunk≈ : ∀ {Θ B} {M M' : Comp Θ B} → M ≈c M' → thunk M ≈v thunk M'
  shut≈ : ∀ {Θ κ A} {V V' : (p : Pos κ (world Θ)) → Val (Θ ⧀⟨ κ , p ⟩) A} →
    (∀ p → V p ≈v V' p) → shut V ≈v shut V'
  -- η
  η𝟙 : ∀ {Θ} (V : Val Θ 𝟙) → V ≈v tt
  ηU : ∀ {Θ B} (V : Val Θ (U B)) → thunk (force V) ≈v V
  η[κ] : ∀ {Θ κ A} (V : Val Θ ([ κ ] A)) → shut (λ q → openV V q) ≈v V

data _≈vs_ where
  []≈ : ∀ {Θ} → []ᵥ {Θ} ≈vs []ᵥ
  _∷≈_ : ∀ {Θ b bs} {V V' : Val Θ (base b)} {vs vs' : Vals Θ bs} →
    V ≈v V' → vs ≈vs vs' → (V ∷ᵥ vs) ≈vs (V' ∷ᵥ vs')

data _≈c_ where
  -- equivalence
  ≈refl : ∀ {Θ B} {M : Comp Θ B} → M ≈c M
  ≈sym : ∀ {Θ B} {M M' : Comp Θ B} → M ≈c M' → M' ≈c M
  ≈trans : ∀ {Θ B} {L M N : Comp Θ B} → L ≈c M → M ≈c N → L ≈c N
  -- congruence
  return≈ : ∀ {Θ A} {V V' : Val Θ A} → V ≈v V' → return V ≈c return V'
  to≈ : ∀ {Θ A B} {M M' : Comp Θ (F A)} {N N' : Comp (Θ , A) B} →
    M ≈c M' → N ≈c N' → (M to N) ≈c (M' to N')
  force≈ : ∀ {Θ B} {V V' : Val Θ (U B)} → V ≈v V' → force V ≈c force V'
  case≈ : ∀ {Θ A A' B} {V V' : Val Θ (A × A')} {M M' : Comp (Θ , A , A') B} →
    V ≈v V' → M ≈c M' → case V M ≈c case V' M'
  case₊≈ : ∀ {Θ A A' B} {V V' : Val Θ (A ⊕ A')} {M M' : Comp (Θ , A) B} {N N' : Comp (Θ , A') B} →
    V ≈v V' → M ≈c M' → N ≈c N' → case₊ V M N ≈c case₊ V' M' N'
  absurd≈ : ∀ {Θ B} {V V' : Val Θ 𝟘} → V ≈v V' → absurd {B = B} V ≈c absurd V'
  lam≈ : ∀ {Θ A B} {M M' : Comp (Θ , A) B} → M ≈c M' → lam M ≈c lam M'
  app≈ : ∀ {Θ A B} {M M' : Comp Θ (A ⇒ B)} {V V' : Val Θ A} →
    M ≈c M' → V ≈v V' → app M V ≈c app M' V'
  op≈ : ∀ {Θ B} {o : Op} {vs vs' : Vals Θ (params o)}
    {k k' : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B} →
    vs ≈vs vs' → (∀ a ps → k a ps ≈c k' a ps) → op o vs k ≈c op o vs' k'
  -- β
  F-β : ∀ {Θ A B} (V : Val Θ A) (N : Comp (Θ , A) B) → (return V to N) ≈c sub₁ N V
  U-β : ∀ {Θ B} (M : Comp Θ B) → force (thunk M) ≈c M
  ×-β : ∀ {Θ A A' B} (V : Val Θ A) (V' : Val Θ A') (M : Comp (Θ , A , A') B) →
    case (pair V V') M ≈c sub₂ M V V'
  +-β₁ : ∀ {Θ A A' B} (V : Val Θ A) (M : Comp (Θ , A) B) (N : Comp (Θ , A') B) →
    case₊ (inl V) M N ≈c sub₁ M V
  +-β₂ : ∀ {Θ A A' B} (V : Val Θ A') (M : Comp (Θ , A) B) (N : Comp (Θ , A') B) →
    case₊ (inr V) M N ≈c sub₁ N V
  ⇒-β : ∀ {Θ A B} (M : Comp (Θ , A) B) (V : Val Θ A) → app (lam M) V ≈c sub₁ M V
  -- η
  η𝟘 : ∀ {Θ B} (M : Comp (Θ , 𝟘) B) (V : Val Θ 𝟘) → sub₁ M V ≈c absurd V
  η+ : ∀ {Θ A A' B} (M : Comp (Θ , (A ⊕ A')) B) (V : Val Θ (A ⊕ A')) →
    sub₁ M V ≈c case₊ V (subC inlS M) (subC inrS M)
  η× : ∀ {Θ A A' B} (M : Comp (Θ , (A × A')) B) (V : Val Θ (A × A')) →
    sub₁ M V ≈c case V (subC pairS M)
  η⇒ : ∀ {Θ A B} (M : Comp Θ (A ⇒ B)) → lam (app (wkC M) (ne here)) ≈c M
  ηF : ∀ {Θ A B} (S : Stack Θ (F A) B) (M : Comp Θ (F A)) →
    plug S M ≈c (M to plug (renS (drop idR) S) (return (ne here)))
  -- algebraicity: every stack enters every continuation, across its locks
  op-stack : ∀ {Θ B B'} (S : Stack Θ B B') (o : Op) (vs : Vals Θ (params o))
    (k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B) →
    plug S (op o vs k) ≈c op o vs (λ a ps → plug (renS (dropW idR (slot o a) ps) S) (k a ps))
  -- the theory: each equation, at any world reached from its own
  theory : ∀ {Θ B} (e : Equation) (γ : ClosedEnv (Variables e) B)
    (g : WMap (equationWorld e) (world Θ)) →
    renC (root {Θ' = Θ} g) (inst γ (lhs e)) ≈c renC (root g) (inst γ (rhs e))

----------------------------------------------------------------------
-- reasoning helpers

≡→≈v : ∀ {Θ A} {V V' : Val Θ A} → V ≡ V' → V ≈v V'
≡→≈v {V = V} p = subst (V ≈v_) p ≈refl

≡→≈c : ∀ {Θ B} {M M' : Comp Θ B} → M ≡ M' → M ≈c M'
≡→≈c {M = M} p = subst (M ≈c_) p ≈refl

≈vs-refl : ∀ {Θ bs} (vs : Vals Θ bs) → vs ≈vs vs
≈vs-refl []ᵥ = []≈
≈vs-refl (V ∷ᵥ vs) = ≈refl ∷≈ ≈vs-refl vs

----------------------------------------------------------------------
-- the lock modality: β holds by definition, open is a congruence

[κ]-β : ∀ {Θ κ A} (V : (p : Pos κ (world Θ)) → Val (Θ ⧀⟨ κ , p ⟩) A) (q : Pos κ (world Θ)) →
  openV (shut V) q ≈v V q
[κ]-β V q = ≈refl

open≈ : ∀ {Θ κ A} {V V' : Val Θ ([ κ ] A)} → V ≈v V' → ∀ q → openV V q ≈v openV V' q
open≈ ≈refl q = ≈refl
open≈ (≈sym p) q = ≈sym (open≈ p q)
open≈ (≈trans p p') q = ≈trans (open≈ p q) (open≈ p' q)
open≈ (shut≈ p) q = p q
open≈ (η[κ] V) q = ≈refl

----------------------------------------------------------------------
-- algebraicity with respect to sequencing and application (the two
-- one-frame stacks; tree reduction's frame steps)

op-to≈ : ∀ {Θ A B} (o : Op) (vs : Vals Θ (params o))
  (k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (F A))
  (N : Comp (Θ , A) B) →
  (op o vs k to N) ≈c op o vs (λ a ps → k a ps to renC (keep (dropW idR (slot o a) ps)) N)
op-to≈ o vs k N = op-stack (hole to• N) o vs k

op-app≈ : ∀ {Θ A B} (o : Op) (vs : Vals Θ (params o))
  (k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (A ⇒ B))
  (V : Val Θ A) →
  app (op o vs k) V ≈c op o vs (λ a ps → app (k a ps) (renV (dropW idR (slot o a) ps) V))
op-app≈ o vs k V = op-stack (app• hole V) o vs k

----------------------------------------------------------------------
-- stacks, and tree reduction inside ≈

plug≈ : ∀ {Θ B B'} (S : Stack Θ B B') {M M' : Comp Θ B} → M ≈c M' → plug S M ≈c plug S M'
plug≈ hole p = p
plug≈ (S to• N) p = to≈ (plug≈ S p) ≈refl
plug≈ (app• S V) p = app≈ (plug≈ S p) ≈refl

-- substitution is a congruence in the substituted value (through ⇒-β)
sub₁≈ : ∀ {Θ A B} (M : Comp (Θ , A) B) {V V' : Val Θ A} → V ≈v V' → sub₁ M V ≈c sub₁ M V'
sub₁≈ M {V} {V'} p = ≈trans (≈sym (⇒-β M V)) (≈trans (app≈ ≈refl p) (⇒-β M V'))

-- each head step is a β law, or algebraicity at a one-frame stack
⇝→≈ : ∀ {Θ B} {M M' : Comp Θ B} → M ⇝ M' → M ≈c M'
⇝→≈ (β-to {V = V} {N}) = F-β V N
⇝→≈ (β-force {M = M}) = U-β M
⇝→≈ (β-case {V = V} {V'} {M}) = ×-β V V' M
⇝→≈ (β-case₊-inl {V = V} {M} {N}) = +-β₁ V M N
⇝→≈ (β-case₊-inr {V = V} {M} {N}) = +-β₂ V M N
⇝→≈ (β-app {M = M} {V}) = ⇒-β M V
⇝→≈ (op-to o vs {k} {N}) = op-to≈ o vs k N
⇝→≈ (op-app o vs {k} {V}) = op-app≈ o vs k V

↦→≈ : ∀ {Θ B} {M M' : Comp Θ B} → M ↦ M' → M ≈c M'
↦→≈ (step S r) = plug≈ S (⇝→≈ r)

↦*→≈ : ∀ {Θ B} {M M' : Comp Θ B} → M ↦* M' → M ≈c M'
↦*→≈ refl* = ≈refl
↦*→≈ (step* s r) = ≈trans (↦→≈ s) (↦*→≈ r)
↦*→≈ (op* {vs = vs} rs) = op≈ (≈vs-refl vs) λ a ps → ↦*→≈ (rs a ps)
