-- Q = T for a free model presented by a container (POPL's operational
-- model: T X = ⟦ p ⟧ X for a polynomial p, configurations T (closed
-- computations), the action μ), for any theory, with or without
-- equations.
--
-- Data: a theory 𝒯 with a free model (monad T), an RMR indexed container
-- Q′ (Q X = Σ_s Π_{r ∈ Pos s} X (input s r)), a world w₀, and a
-- PRESENTATION of the free model's carrier at w₀ by the container: a
-- natural isomorphism
--
--   to   : Q X → T X w₀          from : T X w₀ → Q X
--
-- (to-from, from-to, to-nat).  At W = 1 with the carrier defined as the
-- container's extension (Semantics.PolyFreeModel) both are the identity.
--
-- THE MODULE.  ρ = μ, read through the presentation:
--
--   ρ_X  =  from_X ∘ μ_X ∘ to_{T X}        (ρ-is-μ: to ∘ ρ = μ ∘ to)
--
-- and the module laws are the monad laws (idr-μ for ρ-unit, assoc-μ for
-- ρ-assoc, naturality of μ for ρ's), carried along the isomorphism.  So
-- every free model with a container presentation is a right module over
-- itself (polyModule); RightModuleReduction.RightModule.selfModule is the
-- case Q = T as an endofunctor, which RMR's machine cannot use (its
-- configurations are a container).
--
-- THE UNIT CONFIGURATION.  η-conf t = from (η t) (one thread, holding t).
-- For any algebra A and homomorphism h into the free model on Y, RMR's
-- denotation of a unit configuration is h, presented (denote-η: naturality
-- of from and of η, then μ ∘ η = id, the monad's left unit law).
open import Cubical.Foundations.Prelude
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Functor using (Functor; _∘F_; _⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Monad.Base using (IsMonad)

open import RightModuleReduction.Theory using (Theory; Algebra)
open import RightModuleReduction.Model using (FreeModel; Hom)
open import RightModuleReduction.IndexedContainer using (IndexedContainer)
open import RightModuleReduction.RightModule using (RightModule)
import RightModuleReduction.EffectReduction as EffectReduction

module RMRCanonicity.PRACBPVGeneric.Semantics.PolySelfModule {W : Category ℓ-zero ℓ-zero}
  (𝒯 : Theory W) (fm : FreeModel 𝒯) (Q′ : IndexedContainer (Category.ob W))
  (w₀ : Category.ob W)
  where

-- RMR's effect reduction for this free model and container
module ER = EffectReduction 𝒯 fm Q′
open ER using (𝒱; T; Q)

open Category
open Functor
open NatTrans

private
  T₀ = T .fst
  module Tm = IsMonad (T .snd)

  -- elements of the carrier of T X at w₀, and configurations over X
  TAt : 𝒱 .ob → Type
  TAt X = (T₀ ⟅ X ⟆ ⟅ w₀ ⟆) .fst

  Conf : 𝒱 .ob → Type
  Conf X = (Q ⟅ X ⟆) .fst

  -- the monad laws, at w₀
  μη : ∀ X (t : TAt X) → Tm.μ .N-ob X .N-ob w₀ (Tm.η .N-ob (T₀ ⟅ X ⟆) .N-ob w₀ t) ≡ t
  μη X t i = congP (λ _ → N-ob) Tm.idl-μ i X .N-ob w₀ t

  μTη : ∀ X (t : TAt X) → Tm.μ .N-ob X .N-ob w₀ ((T₀ ⟪ Tm.η .N-ob X ⟫) .N-ob w₀ t) ≡ t
  μTη X t i = congP (λ _ → N-ob) Tm.idr-μ i X .N-ob w₀ t

  μTμ : ∀ X (t : TAt (T₀ ⟅ T₀ ⟅ X ⟆ ⟆)) →
    Tm.μ .N-ob X .N-ob w₀ ((T₀ ⟪ Tm.μ .N-ob X ⟫) .N-ob w₀ t)
      ≡ Tm.μ .N-ob X .N-ob w₀ (Tm.μ .N-ob (T₀ ⟅ X ⟆) .N-ob w₀ t)
  μTμ X t i = congP (λ _ → N-ob) Tm.assoc-μ i X .N-ob w₀ t

  μ-nat : ∀ {X Y} (α : NatTrans X Y) (t : TAt (T₀ ⟅ X ⟆)) →
    Tm.μ .N-ob Y .N-ob w₀ ((T₀ ⟪ T₀ ⟪ α ⟫ ⟫) .N-ob w₀ t) ≡ (T₀ ⟪ α ⟫) .N-ob w₀ (Tm.μ .N-ob X .N-ob w₀ t)
  μ-nat α t i = Tm.μ .N-hom α i .N-ob w₀ t

  η-nat : ∀ {X Y} (α : NatTrans X Y) (x : (X ⟅ w₀ ⟆) .fst) →
    (T₀ ⟪ α ⟫) .N-ob w₀ (Tm.η .N-ob X .N-ob w₀ x) ≡ Tm.η .N-ob Y .N-ob w₀ (α .N-ob w₀ x)
  η-nat α x i = Tm.η .N-hom α (~ i) .N-ob w₀ x

------------------------------------------------------------------------
-- a presentation of the free model at w₀ by the container

record Presentation : Type₁ where
  field
    to : (X : 𝒱 .ob) → Conf X → TAt X
    from : (X : 𝒱 .ob) → TAt X → Conf X
    to-from : ∀ X t → to X (from X t) ≡ t
    from-to : ∀ X c → from X (to X c) ≡ c
    to-nat : ∀ {X Y} (α : NatTrans X Y) (c : Conf X) →
      to Y ((Q ⟪ α ⟫) c) ≡ (T₀ ⟪ α ⟫) .N-ob w₀ (to X c)

module _ (pres : Presentation) where
  open Presentation pres

  from-nat : ∀ {X Y} (α : NatTrans X Y) (t : TAt X) →
    from Y ((T₀ ⟪ α ⟫) .N-ob w₀ t) ≡ (Q ⟪ α ⟫) (from X t)
  from-nat {X} {Y} α t =
    cong (λ u → from Y ((T₀ ⟪ α ⟫) .N-ob w₀ u)) (sym (to-from X t))
    ∙ cong (from Y) (sym (to-nat α (from X t)))
    ∙ from-to Y _

  ----------------------------------------------------------------------
  -- the action ρ = μ

  act : (X : 𝒱 .ob) → Conf (T₀ ⟅ X ⟆) → Conf X
  act X c = from X (Tm.μ .N-ob X .N-ob w₀ (to (T₀ ⟅ X ⟆) c))

  -- read through the presentation, ρ is μ
  ρ-is-μ : ∀ X (c : Conf (T₀ ⟅ X ⟆)) → to X (act X c) ≡ Tm.μ .N-ob X .N-ob w₀ (to (T₀ ⟅ X ⟆) c)
  ρ-is-μ X c = to-from X _

  ρ : NatTrans (Q ∘F T₀) Q
  ρ .N-ob = act
  ρ .N-hom {X} {Y} α = funExt λ c →
    cong (λ u → from Y (Tm.μ .N-ob Y .N-ob w₀ u)) (to-nat (T₀ ⟪ α ⟫) c)
    ∙ cong (from Y) (μ-nat α (to (T₀ ⟅ X ⟆) c))
    ∙ from-nat α _

  -- the free model, presented by the container, is a right module over
  -- itself
  polyModule : RightModule T Q
  polyModule .RightModule.ρ = ρ
  polyModule .RightModule.ρ-unit X = funExt λ c →
    cong (λ u → from X (Tm.μ .N-ob X .N-ob w₀ u)) (to-nat (Tm.η .N-ob X) c)
    ∙ cong (from X) (μTη X (to X c))
    ∙ from-to X c
  polyModule .RightModule.ρ-assoc X = funExt λ c →
    cong (λ u → from X (Tm.μ .N-ob X .N-ob w₀ u)) (to-nat (Tm.μ .N-ob X) c)
    ∙ cong (from X) (μTμ X (to (T₀ ⟅ T₀ ⟅ X ⟆ ⟆) c))
    ∙ cong (λ u → from X (Tm.μ .N-ob X .N-ob w₀ u)) (sym (to-from (T₀ ⟅ X ⟆) _))

  ----------------------------------------------------------------------
  -- the unit configuration, and its denotation

  -- one thread, holding x
  η-conf : (X : 𝒱 .ob) → (X ⟅ w₀ ⟆) .fst → Conf X
  η-conf X x = from X (Tm.η .N-ob X .N-ob w₀ x)

  -- RMR's denotation of a unit configuration is h, presented
  denote-η : (alg : Algebra (Theory.signature 𝒯)) (Y : 𝒱 .ob)
    (h : Hom (Theory.signature 𝒯) alg ((FreeModel.free fm ⟅ Y ⟆) .fst))
    (x : (Algebra.Carrier alg ⟅ w₀ ⟆) .fst) →
    ER.Semantic.denote polyModule alg Y h (η-conf (Algebra.Carrier alg) x)
      ≡ from Y (h .Hom.transformation .N-ob w₀ x)
  denote-η alg Y h x =
    cong (act Y) (sym (from-nat htr (Tm.η .N-ob X .N-ob w₀ x)))
    ∙ cong (λ u → from Y (Tm.μ .N-ob Y .N-ob w₀ u))
        (to-from (T₀ ⟅ Y ⟆) _ ∙ η-nat htr x)
    ∙ cong (from Y) (μη Y (htr .N-ob w₀ x))
    where
    X = Algebra.Carrier alg
    htr = h .Hom.transformation
