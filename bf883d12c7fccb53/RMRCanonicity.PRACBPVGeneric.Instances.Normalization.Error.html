-- Normalization of the error machine (Instances.Machines.Error).
--
--   finOp     raise has no continuations
--   error-sn  every configuration is strongly normalizing
--   error-wn  every configuration reaches a terminal one (a value, or
--             raised)
--
-- The run raise-to (Instances.Machines.Error) ends in the terminal
-- configuration halted tt (raise-to-normal).
module RMRCanonicity.PRACBPVGeneric.Instances.Normalization.Error where

open import Cubical.Foundations.Prelude hiding (_,_)
open import Cubical.Data.Unit using (Unit; tt; tt*; isSetUnit)
open import Cubical.Data.Empty using (⊥)
open import Cubical.Data.Sigma using (Σ-syntax; _×_) renaming (_,_ to _,Σ_)
open import Cubical.Data.FinSet.Base using (isFinSet; isFinOrd→isFinSet)
open import Cubical.Data.FinSet.Properties using (isFinOrd⊥)
open import Cubical.Data.FinSet.Constructors using (isFinOrdΣ)
open import Cubical.Induction.WellFounded using (Acc)
open import Cubical.Categories.Instances.Terminal using (TerminalCategory)

open import RMRCanonicity.PRACBPVGeneric.Instances.Constant using (noLocks; errorOps; raise; module Error)
open Error hiding (_×_)
open import RMRCanonicity.PRACBPVGeneric.Instances.Machines.Error using (isSetLk; isSetBase; S; isSetConf; δ; raise-to)
import RMRCanonicity.PRACBPVGeneric.Semantics.RunNormalization as RunNormalization

-- raise has no continuations
finOp : ∀ o w → isFinSet (Σ[ a ∈ Ar o ] WPos (slot o a) w)
finOp raise w = isFinOrd→isFinSet (isFinOrdΣ ⊥ isFinOrd⊥ _ λ ())

module _ {A : ValTy} where
  open RunNormalization (noLocks TerminalCategory) errorOps isSetLk isSetBase S Unit isSetConf isSetUnit δ A finOp

  error-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  error-sn = strongNormalization

  error-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  error-wn = weakNormalization

  -- raise to N ends raised
  raise-to-normal : ∀ {A'} (N : Comp (∅ tt* , A') (F A)) →
    Σ[ y ∈ Config ] (running tt (raiseC to N) ⟶* y) × Terminal y
  raise-to-normal N = halted tt ,Σ raise-to N ,Σ halted-terminal tt
