-- Value and computation types form sets, when the lock kinds and the base
-- types do.
--
-- The types are an inductive type with labels 𝟘 | 𝟙 | ⊕ | × | U | [ κ ] | base b
-- (values) and F | ⇒ (computations).  We encode them as an indexed
-- W-type over the two sorts; a W-type is a set as soon as its labels are,
-- and the labels are sets because Lk and Base are (the hypotheses; LockSig
-- does not assume them).  The types are a retract of the W-type.
open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels using (isSetRetract)
open import Cubical.Data.Empty using (⊥)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.Bool using (Bool; true; false)
open import Cubical.Data.Sum using (_⊎_; inl; inr; isSet⊎)
import Cubical.Data.W.Indexed as IW

module RMRCanonicity.PRACBPVGeneric.Metatheory.TypesSet
  {Lk Base : Type} (isSetLk : isSet Lk) (isSetBase : isSet Base) where

open import RMRCanonicity.PRACBPVGeneric.Types Lk Base

private
  data Sort : Type where
    vT cT : Sort

  -- the type formers: 𝟘 | 𝟙 | ⊕ | × | U | [ κ ] | base b, and F | ⇒
  Label : Sort → Type
  Label vT = Unit ⊎ (Unit ⊎ (Unit ⊎ (Unit ⊎ (Unit ⊎ (Lk ⊎ Base)))))
  Label cT = Unit ⊎ Unit

  Arity : (s : Sort) → Label s → Type
  Arity vT (inl _) = ⊥
  Arity vT (inr (inl _)) = ⊥
  Arity vT (inr (inr (inl _))) = Bool
  Arity vT (inr (inr (inr (inl _)))) = Bool
  Arity vT (inr (inr (inr (inr (inl _))))) = Unit
  Arity vT (inr (inr (inr (inr (inr (inl _)))))) = Unit
  Arity vT (inr (inr (inr (inr (inr (inr _)))))) = ⊥
  Arity cT (inl _) = Unit
  Arity cT (inr _) = Bool

  Input : (s : Sort) (l : Label s) → Arity s l → Sort
  Input vT (inl _) ()
  Input vT (inr (inl _)) ()
  Input vT (inr (inr (inl _))) _ = vT
  Input vT (inr (inr (inr (inl _)))) _ = vT
  Input vT (inr (inr (inr (inr (inl _))))) _ = cT
  Input vT (inr (inr (inr (inr (inr (inl _)))))) _ = vT
  Input vT (inr (inr (inr (inr (inr (inr _)))))) ()
  Input cT (inl _) _ = vT
  Input cT (inr _) true = vT
  Input cT (inr _) false = cT

  Tree : Sort → Type
  Tree = IW.IW Label Arity Input

  isSetLabel : ∀ s → isSet (Label s)
  isSetLabel vT =
    isSet⊎ isSetUnit (isSet⊎ isSetUnit (isSet⊎ isSetUnit (isSet⊎ isSetUnit
      (isSet⊎ isSetUnit (isSet⊎ isSetLk isSetBase)))))
  isSetLabel cT = isSet⊎ isSetUnit isSetUnit

  encV : ValTy → Tree vT
  encC : CompTy → Tree cT
  encV 𝟘 = IW.node (inl tt) λ ()
  encV 𝟙 = IW.node (inr (inl tt)) λ ()
  encV (A ⊕ A') = IW.node (inr (inr (inl tt))) λ { true → encV A ; false → encV A' }
  encV (A × A') = IW.node (inr (inr (inr (inl tt)))) λ { true → encV A ; false → encV A' }
  encV (U B) = IW.node (inr (inr (inr (inr (inl tt))))) λ _ → encC B
  encV ([ κ ] A) = IW.node (inr (inr (inr (inr (inr (inl κ)))))) λ _ → encV A
  encV (base b) = IW.node (inr (inr (inr (inr (inr (inr b)))))) λ ()
  encC (F A) = IW.node (inl tt) λ _ → encV A
  encC (A ⇒ B) = IW.node (inr tt) λ { true → encV A ; false → encC B }

  decV : Tree vT → ValTy
  decC : Tree cT → CompTy
  decV (IW.node (inl _) _) = 𝟘
  decV (IW.node (inr (inl _)) _) = 𝟙
  decV (IW.node (inr (inr (inl _))) k) = decV (k true) ⊕ decV (k false)
  decV (IW.node (inr (inr (inr (inl _)))) k) = decV (k true) × decV (k false)
  decV (IW.node (inr (inr (inr (inr (inl _))))) k) = U (decC (k tt))
  decV (IW.node (inr (inr (inr (inr (inr (inl κ)))))) k) = [ κ ] decV (k tt)
  decV (IW.node (inr (inr (inr (inr (inr (inr b)))))) _) = base b
  decC (IW.node (inl _) k) = F (decV (k tt))
  decC (IW.node (inr _) k) = decV (k true) ⇒ decC (k false)

  retV : ∀ A → decV (encV A) ≡ A
  retC : ∀ B → decC (encC B) ≡ B
  retV 𝟘 = refl
  retV 𝟙 = refl
  retV (A ⊕ A') = cong₂ _⊕_ (retV A) (retV A')
  retV (A × A') = cong₂ _×_ (retV A) (retV A')
  retV (U B) = cong U (retC B)
  retV ([ κ ] A) = cong ([ κ ]_) (retV A)
  retV (base b) = refl
  retC (F A) = cong F (retV A)
  retC (A ⇒ B) = cong₂ _⇒_ (retV A) (retC B)

isSetValTy : isSet ValTy
isSetValTy = isSetRetract encV decV retV (IW.isOfHLevelSuc-IW 1 isSetLabel vT)

isSetCompTy : isSet CompTy
isSetCompTy = isSetRetract encC decC retC (IW.isOfHLevelSuc-IW 1 isSetLabel cT)
