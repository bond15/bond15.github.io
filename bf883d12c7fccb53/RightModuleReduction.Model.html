module _ {W : Category ℓ-zero ℓ-zero} (P : Signature W) where
  private
    C = PresheafCategory (W ^op) ℓ-zero
    Ext = Poly.Extension P {ℓX = ℓ-zero}

  -- A homomorphism commutes with interpretation of operation layers.
  Preserves : (A B : Algebra P) → NatTrans (Carrier A) (Carrier B) → Type
  Preserves A B h =
    seqTrans (algebra A) h ≡ seqTrans (Ext ⟪ h ⟫) (algebra B)

  isPropPreserves : (A B : Algebra P)
    (h : NatTrans (Carrier A) (Carrier B)) → isProp (Preserves A B h)
  isPropPreserves A B h = isSetNatTrans _ _

  record Hom (A B : Algebra P) : Type where
    field
      transformation : NatTrans (Carrier A) (Carrier B)
      preserves : Preserves A B transformation

  open Hom

  -- Equality of the underlying natural transformations determines equality
  -- of homomorphisms, since preservation is a proposition.
  Hom-path : {A B : Algebra P} {f g : Hom A B} →
    transformation f ≡ transformation g → f ≡ g
  Hom-path p i .transformation = p i
  Hom-path {A} {B} {f} {g} p i .preserves =
    isProp→PathP (λ j → isPropPreserves A B (p j))
      (preserves f) (preserves g) i

  idHom : (A : Algebra P) → Hom A A
  idHom A .transformation = idTrans (Carrier A)
  idHom A .preserves =
    C .⋆IdR (algebra A)
    ∙ sym (C .⋆IdL (algebra A))
    ∙ cong (λ h → seqTrans h (algebra A)) (sym (Ext .F-id))

  -- Composition is in diagrammatic order: first f, then g.  Keep the
  -- coherence proof opaque: clients such as free-model uniqueness compute
  -- with the transformation component, but should not normalize this long
  -- associativity path merely because the composite is used as a Hom.
  abstract
    seqHom-preserves : {A B D : Algebra P} (f : Hom A B) (g : Hom B D) →
      Preserves A D (seqTrans (transformation f) (transformation g))
    seqHom-preserves {A} {B} {D} f g =
      sym (C .⋆Assoc (algebra A) (transformation f) (transformation g))
      ∙ cong (λ h → seqTrans h (transformation g)) (preserves f)
      ∙ C .⋆Assoc (Ext ⟪ transformation f ⟫) (algebra B) (transformation g)
      ∙ cong (seqTrans (Ext ⟪ transformation f ⟫)) (preserves g)
      ∙ sym (C .⋆Assoc (Ext ⟪ transformation f ⟫)
          (Ext ⟪ transformation g ⟫) (algebra D))
      ∙ cong (λ h → seqTrans h (algebra D))
          (sym (Ext .F-seq (transformation f) (transformation g)))

  seqHom : {A B D : Algebra P} → Hom A B → Hom B D → Hom A D
  seqHom f g .transformation = seqTrans (transformation f) (transformation g)
  seqHom f g .preserves = seqHom-preserves f g

  homIsSet : (A B : Algebra P) → isSet (Hom A B)
  homIsSet A B = isSetRetract
    (λ h → transformation h , preserves h)
    (λ p → record { transformation = p .fst ; preserves = p .snd })
    (λ _ → refl)
    (isSetΣSndProp isSetNatTrans (isPropPreserves A B))

  Alg : Category (ℓ-suc ℓ-zero) ℓ-zero
  Alg .ob = Algebra P
  Alg .Hom[_,_] = Hom
  Alg .id = idHom _
  Alg ._⋆_ = seqHom
  Alg .⋆IdL f = Hom-path (C .⋆IdL (transformation f))
  Alg .⋆IdR f = Hom-path (C .⋆IdR (transformation f))
  Alg .⋆Assoc f g h = Hom-path
    (C .⋆Assoc (transformation f) (transformation g) (transformation h))
  Alg .isSetHom {x} {y} = homIsSet x y

  forgetful : Functor Alg (PresheafCategory (W ^op) ℓ-zero)
  forgetful .F-ob = Carrier
  forgetful .F-hom = transformation
  forgetful .F-id = refl
  forgetful .F-seq f g = refl

module _ {W : Category ℓ-zero ℓ-zero} (𝒯 : Theory W) where
  private
    P = 𝒯 .Theory.signature

  -- Models form the full subcategory cut out by equation satisfaction.
  Mod : Category (ℓ-suc ℓ-zero) ℓ-zero
  Mod = FullSubcategory (Alg P) (Satisfies 𝒯)

  modelInclusion : Functor Mod (Alg P)
  modelInclusion = FullInclusion (Alg P) (Satisfies 𝒯)

  modelInclusion-fullyFaithful : isFullyFaithful modelInclusion
  modelInclusion-fullyFaithful = isFullyFaithfulIncl (Alg P) (Satisfies 𝒯)

  -- Satisfaction is a proposition, so the inclusion also embeds objects.
  modelInclusion-embedding : isEmbedding (modelInclusion .F-ob)
  modelInclusion-embedding = isEmbdIncl-ob (Alg P) (isPropSatisfies 𝒯)

  modelForgetful : Functor Mod (PresheafCategory (W ^op) ℓ-zero)
  modelForgetful = forgetful P ∘F modelInclusion

  -- The library uses pairs for subcategory objects. These carry exactly
  -- the same data as the Model record in Theory.agda.
  modelObjectIso : Iso (Model 𝒯) (Mod .ob)
  modelObjectIso .Iso.fun M = M .Model.underlying , M .Model.equations
  modelObjectIso .Iso.inv (A , laws) = record
    { underlying = A ; equations = laws }
  modelObjectIso .Iso.sec _ = refl
  modelObjectIso .Iso.ret _ = refl

-- A chosen free-model functor, with its universal property as an adjunction.
record FreeModel {W : Category ℓ-zero ℓ-zero} (𝒯 : Theory W) : Type₁ where
  private
    C = PresheafCategory (W ^op) ℓ-zero

  field
    free : Functor C (Mod 𝒯)
    adjunction : UnitCounit._⊣_ free (modelForgetful 𝒯)
