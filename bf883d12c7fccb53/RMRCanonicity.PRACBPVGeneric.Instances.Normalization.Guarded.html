-- Normalization of the guarded machine (Instances.Machines.Guarded).
--
--   finOp       step has one continuation at a world n + 1, none at 0
--   guarded-sn  every configuration is strongly normalizing, at every
--               result type A
--   guarded-wn  every configuration reaches a terminal one (a value, or
--               out of fuel), at every A
--   guarded-ground-canonicity
--               at a ground type A, every configuration reaches a unique
--               final one (Semantics.ConfigCanonicity at the runner)
--
-- The runs delay-delay and delay-delay-timeout end in terminal
-- configurations (delay-delay-normal, delay-delay-timeout-normal).
module RMRCanonicity.PRACBPVGeneric.Instances.Normalization.Guarded where

open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Data.Nat using (ℕ; zero; suc)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.Sum using (inl)
open import Cubical.Data.Sigma using (Σ-syntax) renaming (_×_ to _×Σ_)
open import Cubical.Data.FinSet.Base using (isFinSet; isFinOrd; isFinOrd→isFinSet)
open import Cubical.Data.FinSet.Properties using (isFinOrdUnit; isFinOrd⊥)
open import Cubical.Data.FinSet.Constructors using (isFinOrdΣ)
open import Cubical.Induction.WellFounded using (Acc)

open import RMRCanonicity.PRACBPVGeneric.Instances.Guarded hiding (_,_)
open import RMRCanonicity.PRACBPVGeneric.Instances.Machines.Guarded
  using (isSetGLk; isSetGBase; S; isSetConf; δ; delay-delay; delay-delay-timeout)
import RMRCanonicity.PRACBPVGeneric.Semantics.RunNormalization as RunNormalization
open import RMRCanonicity.PRACBPVGeneric.Semantics.Ground gLocks gOps using (Ground; g𝟙; reflect; reflect-den)

-- a tick exists at n + 1, not at 0
isFinOrdPosΩ : ∀ n → isFinOrd (PosΩ n)
isFinOrdPosΩ zero = isFinOrd⊥
isFinOrdPosΩ (suc n) = isFinOrdUnit

-- step has one slot, behind one tick
finOp : ∀ o w → isFinSet (Σ[ a ∈ Ar o ] WPos (slot o a) w)
finOp step n = isFinOrd→isFinSet (isFinOrdΣ Unit isFinOrdUnit _ λ _ → isFinOrdPosΩ n)

module _ {A : ValTy} where
  open RunNormalization gLocks gOps isSetGLk isSetGBase S Unit isSetConf isSetUnit δ A finOp

  guarded-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  guarded-sn = strongNormalization

  guarded-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) ×Σ Terminal y
  guarded-wn = weakNormalization

  guarded-ground-canonicity : Ground A → ∀ x →
    Σ[ ṽ ∈ Final ] (x ⟶* retF ṽ) ×Σ (∀ ṽ' → x ⟶* retF ṽ' → ṽ ≡ ṽ')
  guarded-ground-canonicity = ground-canonicity

-- the runs (Instances.Machines.Guarded) are at A = 𝟙
open RunNormalization gLocks gOps isSetGLk isSetGBase S Unit isSetConf isSetUnit δ 𝟙 finOp

-- with two units of fuel: returns at world 0
delay-delay-normal : Σ[ y ∈ Config ] (running {2} tt (delay (delay (return tt))) ⟶* y) ×Σ Terminal y
delay-delay-normal = running {0} tt (return tt) , delay-delay , running-terminal tt tt

-- with one: out of fuel
delay-delay-timeout-normal : Σ[ y ∈ Config ] (running {1} tt (delay (delay (return tt))) ⟶* y) ×Σ Terminal y
delay-delay-timeout-normal = halted tt , delay-delay-timeout , halted-terminal tt

-- with two units of fuel, every run to a final configuration returns at
-- world 0 (final-unique at the ground type 𝟙)
delay-delay-unique : ∀ ṽ → running {2} tt (delay (delay (return tt))) ⟶* retF ṽ →
  ṽ ≡ (inl (0 , tt) , λ _ → tt)
delay-delay-unique ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p delay-delay
