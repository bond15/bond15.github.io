-- The machine at Q = T for error (Semantics.SelfMachine): W = 1, no
-- locks, one operation raise with no continuations, no equations.
--
-- A configuration is a term over closed computations.  Raising makes the
-- configuration the term raise: one node, no leaves (raised).  So Q = T
-- here is RMR's error configurations Q X ≅ X + 1 up to the shape of the
-- term (var or raise).
--
--   finCont         raise has no continuations
--   error-sn/-wn    strong and weak normalization at Q = T
--   raise-run       η raise ⟶* raised
--   raise-to-run    η (raise to N) ⟶* raised (a tree step, then raise)
--   …-normal        the runs end in a terminal configuration
--   raise-run-sound the run preserves the denotation (Machine.sound⟶*)
--   error-ground-canonicity-T
--                   at a ground type, η M reaches a unique final
--                   configuration (Semantics.SelfCanonicity)
--   error-denote-free
--                   at a first-order type, the machine's denotation of η M
--                   is the free-model denotation of M
module RMRCanonicity.PRACBPVGeneric.Instances.SelfModule.Error where

isSetLk : isSet ⊥
isSetLk = isProp→isSet isProp⊥

isSetBase : isSet NoBase
isSetBase = isProp→isSet λ ()

-- raise has no continuations
finCont : ∀ o w → isFinOrd (Σ[ a ∈ Ar o ] WPos (slot o a) w)
finCont raise w = isFinOrdΣ ⊥ isFinOrd⊥ _ λ ()

module _ {A : ValTy} where
  open SelfMachine (noLocks TerminalCategory) errorOps isSetLk isSetBase isSetUnit* A finCont

  error-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  error-sn = strongNormalization

  error-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  error-wn = weakNormalization

  -- the configuration raise: one node, no leaves
  raised : Config
  raised = nodeC {tt*} raise tt λ ()

  raised-terminal : Terminal raised
  raised-terminal ((() ,Σ _) ,Σ _)

  -- raise: the only thread becomes the node raise
  raise-run : ηC {tt*} raiseC ⟶* raised
  raise-run = op-root then (cong (nodeC raise tt) (funExt λ ()) by-equation done)

  -- raise to N: push the frame into raise's (no) continuations, then raise
  raise-to-run : ∀ {A'} (N : Comp (∅ tt* , A') (F A)) → ηC {tt*} (raiseC to N) ⟶* raised
  raise-to-run N =
    pure-root (step hole (op-to raise []ᵥ)) then op-root then (cong (nodeC raise tt) (funExt λ ()) by-equation done)

  raise-normal : Σ[ y ∈ Config ] (ηC {tt*} raiseC ⟶* y) × Terminal y
  raise-normal = raised ,Σ raise-run ,Σ raised-terminal

  raise-to-normal : ∀ {A'} (N : Comp (∅ tt* , A') (F A)) →
    Σ[ y ∈ Config ] (ηC {tt*} (raiseC to N) ⟶* y) × Terminal y
  raise-to-normal N = raised ,Σ raise-to-run N ,Σ raised-terminal

  raise-run-sound : denote (ηC {tt*} raiseC) ≡ denote raised
  raise-run-sound = sound⟶* raise-run

  open SelfCanonicity (noLocks TerminalCategory) errorOps isSetLk isSetBase isSetUnit* A finCont
    using (Final; retF; ground-canonicity-T; denote-ηC-free; module Fr; module Cmp)

  error-ground-canonicity-T : Ground A → (M : Comp (∅ tt*) (F A)) →
    Σ[ ṽ ∈ Final ] (ηC M ⟶* retF ṽ) × (∀ ṽ' → ηC M ⟶* retF ṽ' → ṽ ≡ ṽ')
  error-ground-canonicity-T g M = ground-canonicity-T g M

  error-denote-free : (fo : Cmp.FirstOrder A) (M : Comp (∅ tt*) (F A)) →
    (SM.ER.Q ⟪ PshHom→NatTrans (Cmp.same fo) ⟫) (denote (ηC M))
      ≡ SM.fromTerm Fr.⟦ A ⟧v (tt* ,Σ Fr.⟦ M ⟧C .PshHom.N-ob tt* refl)
  error-denote-free fo M = denote-ηC-free fo M
