-- Ground types: closed values are read back from their denotations (POPL's
-- Ground, reflect, reflect-den in ConfigNormalization; its D12).
--
--   Ground        built from 𝟘, 𝟙, ⊕, × and base types.  POPL has no base
--                 types; here a closed value of base type is a constant
--                 (no neutral lives at a root ∅ w), and ⟦ base b ⟧ at w is
--                 Const b w, so base types are ground too
--   reflect       ⟦ A ⟧ at w → closed values at ∅ w
--   reflect-den   reflect is a left inverse of the closed denotation
--                 (⟦ V ⟧ at the identity), which is all final-unique needs
--
-- [ κ ] A is not ground: a closed value shut V has V p behind a lock, where
-- the lock's names may occur.  U B and A ⇒ B are not ground (POPL's).
open import Cubical.Foundations.Prelude
open import Cubical.Data.Sum using (inl; inr)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Functor using (Functor)
open import Cubical.Categories.Presheaf.Morphism.Alt using (PshHom)

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Semantics.Ground
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O as Syn
  using (ValTy; 𝟘; 𝟙; _⊕_; base; Const; ∅; Val; ne; tt; inl; inr; pair; const)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Denotation L O using (El; ⟦_⟧v; ⟦_⟧V)

open Functor
open PshHom

private
  module W = Category W

-- the types whose closed values are read back from ⟦ A ⟧
data Ground : ValTy → Type where
  g𝟘 : Ground 𝟘
  g𝟙 : Ground 𝟙
  g⊕ : ∀ {A A'} → Ground A → Ground A' → Ground (A ⊕ A')
  g× : ∀ {A A'} → Ground A → Ground A' → Ground (A Syn.× A')
  gbase : ∀ b → Ground (base b)

-- read back a closed value
reflect : ∀ {A w} → Ground A → El ⟦ A ⟧v w → Val (∅ w) A
reflect g𝟘 ()
reflect g𝟙 _ = tt
reflect (g⊕ g g') (inl a) = inl (reflect g a)
reflect (g⊕ g g') (inr a) = inr (reflect g' a)
reflect (g× g g') (a , a') = pair (reflect g a) (reflect g' a')
reflect (gbase b) c = const c

-- reflect is a left inverse of the closed denotation
reflect-den : ∀ {A w} (g : Ground A) (V : Val (∅ w) A) → reflect g (⟦ V ⟧V .N-ob w W.id) ≡ V
reflect-den g (ne ())
reflect-den g𝟙 tt = refl
reflect-den (g⊕ g g') (inl V) = cong inl (reflect-den g V)
reflect-den (g⊕ g g') (inr V) = cong inr (reflect-den g' V)
reflect-den (g× g g') (pair V V') = cong₂ pair (reflect-den g V) (reflect-den g' V')
reflect-den (gbase b) (const c) = cong const (funExt⁻ (Const b .F-id) c)
