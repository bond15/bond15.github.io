-- The combined machine at Q = T (Semantics.SelfModule): configurations are
-- terms whose leaves hold closed computations, and ρ = μ.  This is the
-- POPL setting, for an equation-free theory.
--
-- Data: a lock signature L and an operation signature O, isSet Lk and
-- isSet Base (for the syntactic laws), worlds forming a set, the result
-- type A, and
--
--   finCont   every operation has finitely many continuations at every
--             world, numbered (Σ a. WPos (slot o a) w is isFinOrd)
--
-- From finCont: the polynomial's positions are discrete (so terms form a
-- configuration container), finOp, and finQ (a term has finitely many
-- leaves).  The self module is affine (SelfModule.self-affine).  So
-- Semantics.Normalization applies:
--
--   strongNormalization, weakNormalization, wn-sound
--
-- Configurations and rules at the root:
--
--   ηC M           the term with one leaf, holding M (η M)
--   nodeC o cs ks  one operation node, its leaves holding ks
--   pure-root      M ↦ M' : ηC M ⟶ ηC M'
--   op-root        ηC (op o vs k) ⟶ nodeC o (closeVals vs) (closeK …):
--                  the op is closed and becomes a node of the
--                  configuration; its continuations are new threads
--   op-in          the same at any leaf of any configuration (the target
--                  is RMR's step, read off with SelfModule.conf-ext)
open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Data.Empty using (⊥)
import Cubical.Data.Empty as ⊥
open import Cubical.Data.Unit using (Unit; tt)
open import Cubical.Data.Sigma using (ΣPathP; Σ-syntax)
open import Cubical.Data.FinSet.Base using (isFinOrd; isFinOrd→isFinSet)
open import Cubical.Data.FinSet.Properties using (isFinSet→Discrete)
open import Cubical.Relation.Nullary using (Discrete; Dec; yes; no)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Functor using (_⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Monad.Base using (IsMonad)

open import RightModuleReduction.Theory using (Term; var; ops; module SignatureNotation)
open import RightModuleReduction.IndexedContainer
  using (IndexedContainer; plug; plug-hole; plug-remaining)

open import RMRCanonicity.PRACBPVGeneric.Signature
import RMRCanonicity.PRACBPVGeneric.Types as Types
-- the polynomial is always written PM.polynomial L O (see Denotation)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
import RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel as FM
import RMRCanonicity.PRACBPVGeneric.Semantics.SelfModule as SelfModule
import RMRCanonicity.PRACBPVGeneric.Semantics.Normalization as Normalization

module RMRCanonicity.PRACBPVGeneric.Semantics.SelfMachine
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L))
  (isSetWorld : isSet (Category.ob W))
  (A : Types.ValTy (LockSig.Lk L) (LockSig.Base L))
  -- every operation has finitely many continuations at every world, numbered
  (finCont : ∀ (o : OpSig.Op O) w →
     isFinOrd (Σ[ a ∈ OpSig.Ar O o ] Notation.WPos L (OpSig.slot O o a) w))
  where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
  using (F; ∅; _⧀*⟨_,_⟩; Comp; Vals; op; Op; Ar; slot; params; Cs; WPos; wscope)
open import RMRCanonicity.PRACBPVGeneric.Closing L O using (closeVals; closeK)
open import RMRCanonicity.PRACBPVGeneric.Reduction L O using (_↦_)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Machine L O
open import RMRCanonicity.PRACBPVGeneric.Metatheory.ClosedLaws L O isSetLk isSetBase using (closedLaws)

private
  module W = Category W
  module P = SignatureNotation (PM.polynomial L O)

-- the polynomial's positions at (o , cs) are the continuations of o
discretePos : ∀ w (o : P.Shape w) → Discrete (P.Position w o)
discretePos w (o , cs) = isFinSet→Discrete (isFinOrd→isFinSet (finCont o w))

finPos : ∀ w (o : P.Shape w) → isFinOrd (P.Position w o)
finPos w (o , cs) = finCont o w

-- the term model as configurations, with ρ = μ
module SM = SelfModule (theory noEquations) noEquations-empty isSetWorld discretePos

-- the machine with the proved syntactic laws, configurations of terms, and
-- result type A
open WithLaws closedLaws public
open Run noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  SM.TermQ′ SM.selfModule A public

-- normalization: finOp and finQ from finCont, affinity of the self module
open Normalization L O closedLaws noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  SM.TermQ′ SM.selfModule A (λ o w → isFinOrd→isFinSet (finCont o w)) (SM.self-finQ finPos) SM.self-affine public
  using (Terminal; strongNormalization; progress; weakNormalization; wn-sound)

private
  X = CompPsh (F A)

------------------------------------------------------------------------
-- configurations

-- η M: one leaf, holding M
ηC : ∀ {w} → Comp (∅ w) (F A) → Config
ηC {w} M = (w , var tt) , λ _ → M

-- one operation node, its leaves holding the closed continuations ks
nodeC : ∀ {w} (o : Op) (cs : Cs w (params o)) →
  ((a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ (wscope (slot o a) w ps)) (F A)) → Config
nodeC {w} o cs ks = (w , ops (o , cs) λ _ → var tt) , λ ((a , ps) , _) → ks a ps

------------------------------------------------------------------------
-- the rules at the root

-- the one-hole context of ηC
rootCtx : ∀ w → Ctxt
rootCtx w = ((w , var tt) , tt) , λ _ ne → ⊥.rec (ne refl)

plug-root : ∀ {w} (M : Comp (∅ w) (F A)) → plug SM.TermQ′ X (rootCtx w) M ≡ ηC M
plug-root {w} M = ΣPathP (refl , funExt λ _ → plug-hole SM.TermQ′ X (rootCtx w) M)

-- a tree step in the only thread
pure-root : ∀ {w} {M M' : Comp (∅ w) (F A)} → M ↦ M' → ηC M ⟶ ηC M'
pure-root {w} {M} {M'} r = subst2 _⟶_ (plug-root M) (plug-root M') (pure (rootCtx w) r)

-- a closed op in the only thread: it becomes the configuration's node
op-root : ∀ {w} {o : Op} {vs : Vals (∅ w) (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) (F A)} →
  ηC (op o vs k) ⟶ nodeC o (closeVals vs) (λ a → closeK (slot o a) (k a))
op-root {w} {o} {vs} {k} =
  subst2 _⟶_ (plug-root (op o vs k))
    (cong (λ f → SM.act X ((w , var tt) , f)) (funExt λ _ →
      plug-hole SM.TermQ′ (ER.T .fst ⟅ X ⟆)
        ((ER.∂Q ⟪ IsMonad.η (ER.T .snd) .NatTrans.N-ob X ⟫) (rootCtx w))
        (ops (o , closeVals vs) λ (a , ps) → var (closeK (slot o a) (k a) ps))))
    (effect (rootCtx w) o vs k)

------------------------------------------------------------------------
-- the rules at any leaf

-- unplugging a leaf and plugging it back
ctxAt : (x : Config) → IndexedContainer.Pos SM.TermQ′ (x .fst) .fst → Ctxt
ctxAt (s , f) p = (s , p) , λ r _ → f r

plug-unplug : (x : Config) (p : IndexedContainer.Pos SM.TermQ′ (x .fst) .fst) →
  plug SM.TermQ′ X (ctxAt x p) (x .snd p) ≡ x
plug-unplug (s , f) p = ΣPathP (refl , funExt λ r → at r (SM.discreteLeaf (s .snd) r p))
  where
  at : (r : IndexedContainer.Pos SM.TermQ′ s .fst) → Dec (r ≡ p) →
    plug SM.TermQ′ X (ctxAt (s , f) p) (f p) .snd r ≡ f r
  at r (yes e) = subst (λ r → plug SM.TermQ′ X (ctxAt (s , f) p) (f p) .snd r ≡ f r) (sym e)
    (plug-hole SM.TermQ′ X (ctxAt (s , f) p) (f p))
  at r (no ne) = plug-remaining SM.TermQ′ X (ctxAt (s , f) p) (f p) r ne

private
  -- a step from an equal configuration
  from : ∀ {x x' y} → x ≡ x' → x ⟶ y → x' ⟶ y
  from {y = y} e st = subst (_⟶ y) e st

-- a pure step at a leaf
pure-in : (x : Config) (p : IndexedContainer.Pos SM.TermQ′ (x .fst) .fst) {M' : Comp (∅ _) (F A)} →
  x .snd p ↦ M' → x ⟶ plug SM.TermQ′ X (ctxAt x p) M'
pure-in x p r = from (plug-unplug x p) (pure (ctxAt x p) r)

-- an effect step at a leaf holding a closed op
op-in : (x : Config) (p : IndexedContainer.Pos SM.TermQ′ (x .fst) .fst) {o : Op}
  {vs : Vals (∅ _) (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) _) → Comp (∅ _ ⧀*⟨ slot o a , ps ⟩) (F A)} →
  x .snd p ≡ op o vs k →
  x ⟶ ER.step SM.selfModule (CompAlg (F A)) (ctxAt x p) ((o , closeVals vs) , λ (a , ps) → closeK (slot o a) (k a) ps)
op-in x p {o} {vs} {k} e =
  from (cong (plug SM.TermQ′ X (ctxAt x p)) (sym e) ∙ plug-unplug x p) (effect (ctxAt x p) o vs k)
