-- The Kripke logical relation for termination (6.2, 6.4, Lemma 6.5).
--
-- LOCK TELESCOPES.  A lock telescope is a root followed by locks only,
-- ∅ w ⧀⟨ κ₁ , p₁ ⟩ ⋯ ⧀⟨ κₙ , pₙ ⟩ (IsLT): the telescopes of closed terms and
-- of the continuations of closed operations.  It has no variables, but it
-- has neutrals: the names its locks bind, reached across later locks,
-- up^j (nm k).  All of them have base type (LT-ne), so the relation can
-- relate every neutral of a lock telescope, which is what makes the empty
-- chain a related environment (𝒢-LT).
--
-- THE RELATION is defined at every telescope.  Its Kripke quantifiers
-- (at U B and A ⇒ B) range over chains (Chain.agda) into lock telescopes:
--
--   Tree R Θ M     M reduces to a tree of operations with R-related leaves:
--                    leaf   R Θ V                        →  Tree R Θ (return V)
--                    node   ∀ a ps. Tree R (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)
--                                                        →  Tree R Θ (op o vs k)
--                    back   M ↦ M' , Tree R Θ M'         →  Tree R Θ M
--   𝒱⟦ 𝟘 ⟧ Θ V = ⊥
--   𝒱⟦ 𝟙 ⟧ Θ tt = ⊤                 𝒱⟦ 𝟙 ⟧ Θ (ne n) = ⊥
--   𝒱⟦ A ⊕ A' ⟧ Θ (inl V) = 𝒱⟦ A ⟧ Θ V    (inr V') = 𝒱⟦ A' ⟧ Θ V'    (ne n) = ⊥
--   𝒱⟦ A × A' ⟧ Θ (pair V V') = 𝒱⟦ A ⟧ Θ V × 𝒱⟦ A' ⟧ Θ V'     (ne n) = ⊥
--   𝒱⟦ U B ⟧ Θ V = ∀ Θ' lock telescope, χ : Chain Θ Θ'.  𝒞⟦ B ⟧ Θ' (force (chV χ V))
--   𝒱⟦ [ κ ] A ⟧ Θ V = ∀ q.  𝒱⟦ A ⟧ (Θ ⧀⟨ κ , q ⟩) (openV V q)
--   𝒱⟦ base b ⟧ Θ V = ⊤             (constants and names alike)
--   𝒞⟦ F A ⟧ = Tree 𝒱⟦ A ⟧
--   𝒞⟦ A ⇒ B ⟧ Θ M = ∀ Θ' lock telescope, χ, V.  𝒱⟦ A ⟧ Θ' V → 𝒞⟦ B ⟧ Θ' (app (chC χ M) V)
--   𝒢 χ = ∀ n : Ne Θ A.  𝒱⟦ A ⟧ Θ' (chV χ (ne n))         (χ : Chain Θ Θ')
--
-- [ κ ] A is R_κ: a boxed value is related when it is related behind the
-- lock, at every position.  Its elimination is opn (a neutral), so in an
-- environment the open case of 𝒢 is exactly this clause.
--
-- LEMMA 6.5.
--   mono𝒱, mono𝒞    the relation is stable under any chain (Kripke
--                    monotonicity); mono𝒱R for a renaming
--   back𝒞           anti-reduction
--   op𝒞             an op whose continuations are related is related; at
--                    A ⇒ B the frame is pushed in (op-app) and the argument
--                    moved across the word's locks (mono𝒱R along dropW)
--   seq𝒞            M to N, for M related and N related under a Kripke
--                    hypothesis; at a node the continuation is moved across
--                    the word's locks (sub-liftS-ren)
--   case𝒞           case on a related pair
--   case₊𝒞          case₊ on a related injection (the neutral case is ⊥)
--   𝒢-LT, mono𝒢, 𝒢-lock, 𝒢-word, 𝒢-β, 𝒢-β₂   related environments
open import Cubical.Foundations.Prelude hiding (Sub; _,_; comp)
open import Cubical.Data.Unit using (Unit) renaming (tt to ⋆)
open import Cubical.Data.Empty as ⊥ using (⊥)

-- 6.2 lock telescopes

data IsLT : Tele → Type where
  lt∅ : ∀ {w} → IsLT (∅ w)
  lt⧀ : ∀ {Θ κ p} → IsLT Θ → IsLT (Θ ⧀⟨ κ , p ⟩)

IsLT-⧀* : ∀ {Θ} → IsLT Θ → (μ : List Lk) (ps : WPos μ (world Θ)) → IsLT (Θ ⧀*⟨ μ , ps ⟩)
IsLT-⧀* l [] ps = l
IsLT-⧀* l (κ ∷ []) p = lt⧀ l
IsLT-⧀* l (κ ∷ μ@(_ ∷ _)) (p ,ˢ ps) = IsLT-⧀* (lt⧀ l) μ ps

private
  isBase : ValTy → Type
  isBase 𝟘 = ⊥
  isBase 𝟙 = ⊥
  isBase (A ⊕ A') = ⊥
  isBase (A × A') = ⊥
  isBase (U B) = ⊥
  isBase ([ κ ] A) = ⊥
  isBase (base b) = Unit

-- every neutral of a lock telescope is a name, up^j (nm k): of base type
LT-ne : ∀ {Θ A} → IsLT Θ → Ne Θ A → Σ[ b ∈ Base ] (A ≡ base b)
LT-ne lt∅ ()
LT-ne (lt⧀ l) (up n) = LT-ne l n
LT-ne (lt⧀ l) (opn n) = ⊥.rec (subst isBase (sym (LT-ne l n .snd)) ⋆)
LT-ne (lt⧀ l) (nm {b = b} k) = b ,ˢ refl

-- 6.4 the relation

data Tree {A : ValTy} (R : (Θ : Tele) → Val Θ A → Type) : (Θ : Tele) → Comp Θ (F A) → Type where
  leaf : ∀ {Θ} {V : Val Θ A} → R Θ V → Tree R Θ (return V)
  node : ∀ {Θ} (o : Op) {vs : Vals Θ (params o)}
    {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) (F A)} →
    (∀ a ps → Tree R (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)) → Tree R Θ (op o vs k)
  back : ∀ {Θ} {M M' : Comp Θ (F A)} → M ↦ M' → Tree R Θ M' → Tree R Θ M

𝒱⟦_⟧ : (A : ValTy) (Θ : Tele) → Val Θ A → Type
𝒞⟦_⟧ : (B : CompTy) (Θ : Tele) → Comp Θ B → Type

𝒱⟦ 𝟘 ⟧ Θ V = ⊥
𝒱⟦ 𝟙 ⟧ Θ (ne n) = ⊥
𝒱⟦ 𝟙 ⟧ Θ tt = Unit
𝒱⟦ A ⊕ A' ⟧ Θ (ne n) = ⊥
𝒱⟦ A ⊕ A' ⟧ Θ (inl V) = 𝒱⟦ A ⟧ Θ V
𝒱⟦ A ⊕ A' ⟧ Θ (inr V) = 𝒱⟦ A' ⟧ Θ V
𝒱⟦ A × A' ⟧ Θ (ne n) = ⊥
𝒱⟦ A × A' ⟧ Θ (pair V V') = 𝒱⟦ A ⟧ Θ V ×ˢ 𝒱⟦ A' ⟧ Θ V'
𝒱⟦ U B ⟧ Θ V = ∀ {Θ'} → IsLT Θ' → (χ : Chain Θ Θ') → 𝒞⟦ B ⟧ Θ' (force (chV χ V))
𝒱⟦ [ κ ] A ⟧ Θ V = ∀ q → 𝒱⟦ A ⟧ (Θ ⧀⟨ κ , q ⟩) (openV V q)
𝒱⟦ base b ⟧ Θ V = Unit

𝒞⟦ F A ⟧ = Tree 𝒱⟦ A ⟧
𝒞⟦ A ⇒ B ⟧ Θ M = ∀ {Θ'} → IsLT Θ' → (χ : Chain Θ Θ') (V : Val Θ' A) → 𝒱⟦ A ⟧ Θ' V →
  𝒞⟦ B ⟧ Θ' (app (chC χ M) V)

-- related environments: every neutral goes to a related value
𝒢 : ∀ {Θ Θ'} → Chain Θ Θ' → Type
𝒢 {Θ} {Θ'} χ = ∀ {A} (n : Ne Θ A) → 𝒱⟦ A ⟧ Θ' (chV χ (ne n))

-- 6.5 monotonicity

mono𝒱 : ∀ A {Θ Θ'} (χ : Chain Θ Θ') (V : Val Θ A) → 𝒱⟦ A ⟧ Θ V → 𝒱⟦ A ⟧ Θ' (chV χ V)
mono𝒱 𝟘 χ V ()
mono𝒱 𝟙 χ (ne n) ()
mono𝒱 𝟙 χ tt _ = subst (𝒱⟦ 𝟙 ⟧ _) (sym (ch-tt χ)) ⋆
mono𝒱 (A ⊕ A') χ (ne n) ()
mono𝒱 (A ⊕ A') χ (inl V) v = subst (𝒱⟦ A ⊕ A' ⟧ _) (sym (ch-inl χ V)) (mono𝒱 A χ V v)
mono𝒱 (A ⊕ A') χ (inr V) v = subst (𝒱⟦ A ⊕ A' ⟧ _) (sym (ch-inr χ V)) (mono𝒱 A' χ V v)
mono𝒱 (A × A') χ (ne n) ()
mono𝒱 (A × A') χ (pair V V') (v ,ˢ v') =
  subst (𝒱⟦ A × A' ⟧ _) (sym (ch-pair χ V V')) (mono𝒱 A χ V v ,ˢ mono𝒱 A' χ V' v')
mono𝒱 (U B) χ V h l χ' = subst (λ X → 𝒞⟦ B ⟧ _ (force X)) (ch-++V χ χ' V) (h l (χ ++ᶜ χ'))
mono𝒱 ([ κ ] A) χ V h q =
  subst (𝒱⟦ A ⟧ _) (ch-openV χ q V) (mono𝒱 A (lockᶜ κ χ q) (openV V (posᶜ κ χ q)) (h (posᶜ κ χ q)))
mono𝒱 (base b) χ V _ = ⋆

-- along a renaming
mono𝒱R : ∀ A {Θ Θ'} (ρ : Ren Θ Θ') (V : Val Θ A) → 𝒱⟦ A ⟧ Θ V → 𝒱⟦ A ⟧ Θ' (renV ρ V)
mono𝒱R A ρ V v = subst (𝒱⟦ A ⟧ _) (sub-renV ρ V) (mono𝒱 A (⌜ ρ ⌝ ∷ᶜ []ᶜ) V v)

monoTree : ∀ {A Θ Θ'} (χ : Chain Θ Θ') {M : Comp Θ (F A)} → Tree 𝒱⟦ A ⟧ Θ M →
  Tree 𝒱⟦ A ⟧ Θ' (chC χ M)
monoTree {A} {Θ' = Θ'} χ (leaf {V = V} v) =
  subst (Tree 𝒱⟦ A ⟧ Θ') (sym (ch-return χ V)) (leaf (mono𝒱 A χ V v))
monoTree {A} {Θ' = Θ'} χ (node o {vs} {k} ts) =
  subst (Tree 𝒱⟦ A ⟧ Θ') (sym (ch-op χ o vs k))
    (node o λ a ps → monoTree (wordᶜ (slot o a) χ ps) (ts a (wposᶜ (slot o a) χ ps)))
monoTree χ (back r t) = back (ch-↦ χ r) (monoTree χ t)

mono𝒞 : ∀ B {Θ Θ'} (χ : Chain Θ Θ') (M : Comp Θ B) → 𝒞⟦ B ⟧ Θ M → 𝒞⟦ B ⟧ Θ' (chC χ M)
mono𝒞 (F A) χ M t = monoTree χ t
mono𝒞 (A ⇒ B) χ M h l χ' V v = subst (λ X → 𝒞⟦ B ⟧ _ (app X V)) (ch-++C χ χ' M) (h l (χ ++ᶜ χ') V v)

-- 6.5 anti-reduction, operations, sequencing, case, case₊

back𝒞 : ∀ B {Θ} {M M' : Comp Θ B} → M ↦ M' → 𝒞⟦ B ⟧ Θ M' → 𝒞⟦ B ⟧ Θ M
back𝒞 (F A) r t = back r t
back𝒞 (A ⇒ B) r h l χ V v = back𝒞 B (↦-app V (ch-↦ χ r)) (h l χ V v)

op𝒞 : ∀ B {Θ} (o : Op) {vs : Vals Θ (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B} →
  (∀ a ps → 𝒞⟦ B ⟧ (Θ ⧀*⟨ slot o a , ps ⟩) (k a ps)) → 𝒞⟦ B ⟧ Θ (op o vs k)
op𝒞 (F A) o h = node o h
op𝒞 (A ⇒ B) o {vs} {k} h {Θ'} l χ V v =
  subst (λ X → 𝒞⟦ B ⟧ Θ' (app X V)) (sym (ch-op χ o vs k))
    (back𝒞 B (step hole (op-app o (chVs χ vs)))
      (op𝒞 B o λ a ps →
        h a (wposᶜ (slot o a) χ ps) (IsLT-⧀* l (slot o a) ps) (wordᶜ (slot o a) χ ps)
          (renV (dropW idR (slot o a) ps) V) (mono𝒱R A (dropW idR (slot o a) ps) V v)))

-- the Kripke hypothesis on the continuation of a sequencing
SeqHyp : ∀ {Θ A} B → Comp (Θ , A) B → Type
SeqHyp {Θ} {A} B N = ∀ {Θ'} → IsLT Θ' → (χ : Chain Θ Θ') (V : Val Θ' A) → 𝒱⟦ A ⟧ Θ' V →
  𝒞⟦ B ⟧ Θ' (sub₁ (chC (liftᶜ χ) N) V)

seq𝒞 : ∀ {Θ A B} {M : Comp Θ (F A)} {N : Comp (Θ , A) B} → IsLT Θ →
  Tree 𝒱⟦ A ⟧ Θ M → SeqHyp B N → 𝒞⟦ B ⟧ Θ (M to N)
seq𝒞 {B = B} l (leaf {V = V} v) hN = back𝒞 B (step hole β-to) (hN l []ᶜ V v)
seq𝒞 {B = B} {N = N} l (node o {vs} ts) hN =
  back𝒞 B (step hole (op-to o vs))
    (op𝒞 B o λ a ps → seq𝒞 (IsLT-⧀* l (slot o a) ps) (ts a ps) λ {Θ'} l' χ' V v →
      subst (λ X → 𝒞⟦ B ⟧ Θ' (sub₁ (chC (liftᶜ χ') X) V))
        (sub-liftS-ren (dropW idR (slot o a) ps) N)
        (hN l' (⌜ dropW idR (slot o a) ps ⌝ ∷ᶜ χ') V v))
seq𝒞 {B = B} {N = N} l (back r t) hN = back𝒞 B (↦-to N r) (seq𝒞 l t hN)

case𝒞 : ∀ {Θ A A' B} {M : Comp (Θ , A , A') B} (X : Val Θ (A × A')) → 𝒱⟦ A × A' ⟧ Θ X →
  (∀ V V' → 𝒱⟦ A ⟧ Θ V → 𝒱⟦ A' ⟧ Θ V' → 𝒞⟦ B ⟧ Θ (sub₂ M V V')) → 𝒞⟦ B ⟧ Θ (case X M)
case𝒞 (ne n) () h
case𝒞 {B = B} (pair V V') (v ,ˢ v') h = back𝒞 B (step hole β-case) (h V V' v v')

case₊𝒞 : ∀ {Θ A A' B} {M : Comp (Θ , A) B} {N : Comp (Θ , A') B} (X : Val Θ (A ⊕ A')) →
  𝒱⟦ A ⊕ A' ⟧ Θ X → (∀ V → 𝒱⟦ A ⟧ Θ V → 𝒞⟦ B ⟧ Θ (sub₁ M V)) →
  (∀ V' → 𝒱⟦ A' ⟧ Θ V' → 𝒞⟦ B ⟧ Θ (sub₁ N V')) → 𝒞⟦ B ⟧ Θ (case₊ X M N)
case₊𝒞 (ne n) () hM hN
case₊𝒞 {B = B} (inl V) v hM hN = back𝒞 B (step hole β-case₊-inl) (hM V v)
case₊𝒞 {B = B} (inr V) v hM hN = back𝒞 B (step hole β-case₊-inr) (hN V v)

-- related environments

-- on a lock telescope every neutral is a name, hence related: the empty
-- chain is a related environment
𝒢-LT : ∀ {Θ} → IsLT Θ → 𝒢 ([]ᶜ {Θ})
𝒢-LT {Θ} l n =
  subst (λ A → (m : Ne Θ A) → 𝒱⟦ A ⟧ Θ (ne m)) (sym (LT-ne l n .snd)) (λ _ → ⋆) n

mono𝒢 : ∀ {Θ Θ' Θ''} (χ : Chain Θ Θ') → 𝒢 χ → (χ' : Chain Θ' Θ'') → 𝒢 (χ ++ᶜ χ')
mono𝒢 χ G χ' {A} n = subst (𝒱⟦ A ⟧ _) (sym (ch-++V χ χ' (ne n))) (mono𝒱 A χ' (chV χ (ne n)) (G n))

-- under a lock: up crosses it (monotonicity along dropL), opn is the
-- [ κ ] clause, names are of base type
𝒢-lock : ∀ {Θ Θ'} (κ : Lk) (χ : Chain Θ Θ') (q : Pos κ (world Θ')) → 𝒢 χ → 𝒢 (lockᶜ κ χ q)
𝒢-lock κ χ q G {A} (up n) =
  subst (𝒱⟦ A ⟧ _) (sym (ch-↑ χ q (ne n))) (mono𝒱R A (dropL idR q) (chV χ (ne n)) (G n))
𝒢-lock κ χ q G {A} (opn n) = subst (𝒱⟦ A ⟧ _) (sym (ch-openV χ q (ne n))) (G n q)
𝒢-lock κ χ q G (nm k) = ⋆

-- under the locks of a word, outermost lock first
𝒢-word : ∀ {Θ Θ'} (μ : List Lk) (χ : Chain Θ Θ') (ps : WPos μ (world Θ')) → 𝒢 χ → 𝒢 (wordᶜ μ χ ps)
𝒢-word [] χ ps G = subst 𝒢 (sym (wordᶜ-[] χ ps)) G
𝒢-word (κ ∷ []) χ p G = 𝒢-lock κ χ p G
𝒢-word (κ ∷ κ' ∷ μ) χ (p ,ˢ ps) G =
  transport (λ i → 𝒢 (wordᶜ-cons κ κ' μ χ p ps (~ i)))
    (𝒢-word (κ' ∷ μ) (lockᶜ κ χ p) ps (𝒢-lock κ χ p G))

-- under a variable, then a related value for it
𝒢-β : ∀ {Θ Θ' A} (χ : Chain Θ Θ') {V : Val Θ' A} → 𝒢 χ → 𝒱⟦ A ⟧ Θ' V →
  𝒢 (liftᶜ χ ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ)
𝒢-β {A = A} χ {V} G v here = subst (𝒱⟦ A ⟧ _) (sym (ch-β-here χ V)) v
𝒢-β χ {V} G v {B} (wkv n) = subst (𝒱⟦ B ⟧ _) (sym (ch-β χ V n)) (G n)

-- under two variables, then related values for them (case)
𝒢-β₂ : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') {V : Val Θ' A} {V' : Val Θ' A'} → 𝒢 χ →
  𝒱⟦ A ⟧ Θ' V → 𝒱⟦ A' ⟧ Θ' V' → 𝒢 (liftᶜ (liftᶜ χ) ++ᶜ (idS ,ₛ V ,ₛ V') ∷ᶜ []ᶜ)
𝒢-β₂ {A' = A'} χ {V} {V'} G v v' here = subst (𝒱⟦ A' ⟧ _) (sym (ch-β₂-snd χ V V')) v'
𝒢-β₂ {A = A} χ {V} {V'} G v v' (wkv here) = subst (𝒱⟦ A ⟧ _) (sym (ch-β₂-fst χ V V')) v
𝒢-β₂ χ {V} {V'} G v v' {B} (wkv (wkv n)) = subst (𝒱⟦ B ⟧ _) (sym (ch-β₂ χ V V' n)) (G n)
