-- The combined machine at a deterministic runner (math spec §10, Cor 10.2).
--
-- Data: a lock signature L and an operation signature O, isSet Lk and
-- isSet Base (the hypotheses of the syntactic laws), a runner (S , H , δ)
-- on the derived polynomial PM.polynomial L O, and the result type A.
--
-- It instantiates Semantics.Machine with
--
--   the theory           theory noEquations (the polynomial, no equations)
--   its free model       FreeModel.freeModel: terms, T X = TermPsh X
--   the container        Runner.RunQ′:  Q X = Σ_w S w × X w + H
--   the right module     Runner.runnerModule:  ρ = run
--   the syntactic laws   Metatheory.ClosedLaws.closedLaws (proved)
--
-- and derives the machine's rules at running configurations
-- (configurations are running σ M or halted h):
--
--   pure-at    M ↦ M' : a tree step in the running program
--   op-step    a closed op is closed (Closing.closeOp) and δ decides the
--              next configuration (next), in one machine step
--   op-at      op-step as a one-step run
--
-- Corollaries of machine soundness (Thm 9.1), for a closed program M:
--
--   denote-running   denote (running σ M) = run σ (h M)          (refl)
--   h-is-denClosed   h M = denClosed M                   (evaluate-var)
--   run-return       running σ M ⟶* running σ' (return V) implies
--                    run σ (denClosed M) ≡ the running configuration
--                    (w' , σ') holding ⟦ V ⟧ at the identity
--   run-halt         running σ M ⟶* halted h implies
--                    run σ (denClosed M) ≡ the halted configuration h
--
-- So every machine run computes the program's denotation, run by the
-- runner.
open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Data.Empty using (⊥)
open import Cubical.Data.Sum using (_⊎_)
open import Cubical.Data.Sigma using (Σ-syntax)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Monad.Base using (IsMonad)
open import Cubical.Categories.Presheaf.Morphism.Alt using (PshHom)

open import RightModuleReduction.Theory using (module SignatureNotation; var; ops)
open import RightModuleReduction.Model using (Hom)

open import RMRCanonicity.PRACBPVGeneric.Signature
import RMRCanonicity.PRACBPVGeneric.Types as Types
-- the polynomial is always written PM.polynomial L O (see Denotation)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
import RMRCanonicity.PRACBPVGeneric.Semantics.FreeModel as FM
import RMRCanonicity.PRACBPVGeneric.Semantics.Runner as Runner

module RMRCanonicity.PRACBPVGeneric.Semantics.RunMachine
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L))
  (S : Category.ob W → Type) (H : Type)
  (isSetConf : isSet (Σ (Category.ob W) S)) (isSetH : isSet H)
  (δ : ∀ {w} → S w → (o : SignatureNotation.Shape (PM.polynomial L O) w) →
       H ⊎ (Σ[ p ∈ SignatureNotation.Position (PM.polynomial L O) w o ]
              S (SignatureNotation.scope (PM.polynomial L O) w o p)))
  (A : Types.ValTy (LockSig.Lk L) (LockSig.Base L))
  where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
  using (F; ∅; _⧀*⟨_,_⟩; Comp; Val; Vals; op; return; Op; Ar; slot; params; Cs; WPos; wscope)
open import RMRCanonicity.PRACBPVGeneric.Closing L O using (closeVals; closeK)
open import RMRCanonicity.PRACBPVGeneric.Reduction L O using (_↦_)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Denotation L O using (⟦_⟧v; ⟦_⟧V) public
open import RMRCanonicity.PRACBPVGeneric.Semantics.Closed L O using (denClosed) public
open import RMRCanonicity.PRACBPVGeneric.Semantics.Machine L O
open import RMRCanonicity.PRACBPVGeneric.Metatheory.ClosedLaws L O isSetLk isSetBase using (closedLaws)

private
  module W = Category W

-- the runner on the derived polynomial, for the equation-free theory
module Rn = Runner (theory noEquations) noEquations-empty S H isSetConf isSetH δ
open Rn public using (run)

-- the machine with the proved syntactic laws, this runner, and result type A
open WithLaws closedLaws public
open Run noEquations (FM.freeModel (theory noEquations) noEquations-empty)
  Rn.RunQ′ Rn.runnerModule A public

------------------------------------------------------------------------
-- configurations

-- running from state σ (at w), the closed program M in the hole
running : ∀ {w} → S w → Comp (∅ w) (F A) → Config
running = Rn.running (CompPsh (F A))

-- halted with outcome h
halted : H → Config
halted = Rn.halted (CompPsh (F A))

-- what δ makes of an operation node with closed continuations ks
next : ∀ {w} → S w → (o : Op) (cs : Cs w (params o)) →
  ((a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ (wscope (slot o a) w ps)) (F A)) → Config
next σ o cs ks = Rn.next (CompPsh (F A)) σ (o , cs) λ (a , ps) → ks a ps

------------------------------------------------------------------------
-- the derived rules

-- a tree step in the running program
pure-at : ∀ {w} {σ : S w} {M M' : Comp (∅ w) (F A)} → M ↦ M' → running σ M ⟶ running σ M'
pure-at {σ = σ} {M} {M'} r =
  subst2 _⟶_ (Rn.plug-context (CompPsh (F A)) σ M) (Rn.plug-context (CompPsh (F A)) σ M')
    (pure (Rn.context (CompPsh (F A)) σ) r)

-- a closed op: close its parameters and continuations, and let δ decide
-- the next configuration (the machine's fused effect step, transported
-- along the plug-hole equations of the running context)
op-step : ∀ {w} {σ : S w} {o : Op} {vs : Vals (∅ w) (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) (F A)} →
  running σ (op o vs k) ⟶ next σ o (closeVals vs) (λ a → closeK (slot o a) (k a))
op-step {σ = σ} {o} {vs} {k} =
  subst2 _⟶_ (Rn.plug-context (CompPsh (F A)) σ (op o vs k))
    (cong (Rn.act (CompPsh (F A)))
      (Rn.plug-mapped-context (IsMonad.η (Rn.ER.T .snd) .NatTrans.N-ob (CompPsh (F A))) σ
        (ops (o , closeVals vs) λ (a , ps) → var (closeK (slot o a) (k a) ps))))
    (effect (Rn.context (CompPsh (F A)) σ) o vs k)

-- op-step as a run
op-at : ∀ {w} {σ : S w} {o : Op} {vs : Vals (∅ w) (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) (F A)} →
  running σ (op o vs k) ⟶* next σ o (closeVals vs) (λ a → closeK (slot o a) (k a))
op-at = op-step then done

-- runs compose
infixr 5 _then*_

_then*_ : ∀ {x y z} → x ⟶* y → y ⟶* z → x ⟶* z
done then* rs = rs
(r then rs) then* rs' = r then (rs then* rs')
(p by-equation rs) then* rs' = p by-equation (rs then* rs')

------------------------------------------------------------------------
-- corollaries of machine soundness

-- a running configuration denotes the run of its program's denotation
denote-running : ∀ {w} (σ : S w) (M : Comp (∅ w) (F A)) →
  denote (running σ M) ≡ run σ (h .Hom.transformation .NatTrans.N-ob w M)
denote-running σ M = refl

-- into the term model, h is denClosed
h-is-denClosed : ∀ {w} (M : Comp (∅ w) (F A)) → h .Hom.transformation .NatTrans.N-ob w M ≡ denClosed M
h-is-denClosed {w} M = FM.evaluate-var (theory noEquations) noEquations-empty ⟦ A ⟧v w (denClosed M)

-- a run that returns: the runner runs the denotation to the same state
-- and the returned value's denotation
run-return : ∀ {w w'} {σ : S w} {σ' : S w'} {M : Comp (∅ w) (F A)} {V : Val (∅ w') A} →
  running σ M ⟶* running σ' (return V) →
  run σ (denClosed M) ≡ Rn.running ⟦ A ⟧v σ' (⟦ V ⟧V .PshHom.N-ob w' W.id)
run-return {σ = σ} {M = M} r = cong (run σ) (sym (h-is-denClosed M)) ∙ sound⟶* r

-- a run that halts: the runner halts on the denotation with the same outcome
run-halt : ∀ {w} {σ : S w} {M : Comp (∅ w) (F A)} {out : H} →
  running σ M ⟶* halted out → run σ (denClosed M) ≡ Rn.halted ⟦ A ⟧v out
run-halt {σ = σ} {M = M} r = cong (run σ) (sym (h-is-denClosed M)) ∙ sound⟶* r
