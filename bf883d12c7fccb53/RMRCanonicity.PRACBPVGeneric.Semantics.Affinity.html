-- Affinity of a right module, read off the module at the generic redex
-- (Yoneda).
--
-- Data: any theory 𝒯 with a free model (monad T), an RMR indexed
-- container Q′ (Q X = Σ_s Π_{r ∈ Pos s} X (input s r)) and a right
-- T-module R (ρ : Q ∘ T ⇒ Q).  Nothing here is specific to PRACBPV.
--
-- RMR's effect step at a one-hole context C = ((s , p) , a) and an
-- operation o in the hole with arguments k is
--
--   stepPsh X C (o , k)  =  ρ_X (plug C' (o (η ∘ k)))      (C' = C with η on the other holes)
--
-- and it only needs a carrier presheaf X (step-agrees: RMR's step is it
-- at the algebra's carrier).  It is natural in X (stepPsh-natural).
--
-- THE GENERIC REDEX.  Fix s, p, o.  The SOURCES of a redex are the other
-- holes r ≠ p of s and the positions q of o; each sits at a world
-- (srcWorld).  Gen is the coproduct of the representables at the sources,
-- Gen = Σ_σ W [ srcWorld σ , − ], and the generic redex fills each source
-- σ with its own generator (σ , id).  Its step is a configuration of Gen:
--
--   generic s p o     a shape (composite), and at each of its holes r′
--                     an element (origin r′ , moved r′) of Gen: the
--                     source the hole comes from and a world map out of
--                     the source's world
--
-- By naturality every actual step is the generic one, renamed along the
-- map env : Gen ⇒ X that sends a generator to its filler (represent): the
-- result has shape composite s p o, and its hole r′ holds the origin's
-- filler moved along moved r′ (composite-filler, by refl).
--
-- MODULE AFFINITY (ModuleAffine): at every generic redex, distinct holes
-- of the result have distinct origins.  Then no step of the module copies
-- a filler; the fillers are moved, dropped, or new.  It is a property of
-- ρ alone (on the generic configurations), not of a syntax.
-- Configurations with at most one hole are affine (affine-subsingleton).
open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels
open import Cubical.Data.Sigma using (ΣPathP)
open import Cubical.Data.Sum using (_⊎_; inl; inr; isSet⊎)
open import Cubical.Relation.Nullary using (¬_; isProp¬)
open import Cubical.Categories.Category using (Category; _^op; _[_,_])
open import Cubical.Categories.Functor using (Functor; _⟅_⟆; _⟪_⟫; _∘F_)
open import Cubical.Categories.NaturalTransformation using (NatTrans; seqTrans)
open import Cubical.Categories.Monad.Base using (Monad; IsMonad)
open import Polynomial.Presheaf.Discrete.Base using (Poly)
open import RightModuleReduction.Theory using (Theory; Algebra; module SignatureNotation)
open import RightModuleReduction.Model using (FreeModel; Hom)
open import RightModuleReduction.IndexedContainer
  using (IndexedContainer; Extension; Contexts; plug; plug-map)

module RMRCanonicity.PRACBPVGeneric.Semantics.Affinity
  {W : Category ℓ-zero ℓ-zero} (𝒯 : Theory W) (freeModel : FreeModel 𝒯)
  (Q′ : IndexedContainer (Category.ob W)) where

import RightModuleReduction.EffectReduction 𝒯 freeModel Q′ as ER
open ER using (𝒱; Q; ∂Q; T)

open Category
open Functor
open NatTrans
open IndexedContainer Q′ using () renaming (Shape to CShape; Pos to CPos; input to inQ)

private
  P = Theory.signature 𝒯
  open SignatureNotation P

  Σ𝒯 : Functor 𝒱 𝒱
  Σ𝒯 = Poly.Extension P {ℓX = ℓ-zero}

  T₀ = T .fst
  module Tm = IsMonad (T .snd)
  F = FreeModel.free freeModel

  η : (X : 𝒱 .ob) → NatTrans X (T₀ ⟅ X ⟆)
  η X = Tm.η .N-ob X

  -- the free model's operations on generators lifted by η
  freeOp : (X : 𝒱 .ob) → NatTrans (Σ𝒯 ⟅ X ⟆) (T₀ ⟅ X ⟆)
  freeOp X = seqTrans (Σ𝒯 ⟪ η X ⟫) (Algebra.algebra ((F ⟅ X ⟆) .fst))

module _ (R : RightModule T Q) where
  open RightModule R

  ----------------------------------------------------------------------
  -- the effect step needs only a carrier presheaf

  stepPsh : (X : 𝒱 .ob) (C@((s , p) , a) : (∂Q ⟅ X ⟆) .fst) →
    (Σ𝒯 ⟅ X ⟆ ⟅ inQ s p ⟆) .fst → (Q ⟅ X ⟆) .fst
  stepPsh X C@((s , p) , a) ok =
    ρ .N-ob X (plug Q′ (T₀ ⟅ X ⟆) ((∂Q ⟪ η X ⟫) C) (freeOp X .N-ob (inQ s p) ok))

  -- RMR's step is stepPsh at the algebra's carrier, definitionally
  step-agrees : (alg : Algebra P) (C@((s , p) , a) : (∂Q ⟅ Algebra.Carrier alg ⟆) .fst)
    (ok : (Σ𝒯 ⟅ Algebra.Carrier alg ⟆ ⟅ inQ s p ⟆) .fst) →
    ER.step R alg C ok ≡ stepPsh (Algebra.Carrier alg) C ok
  step-agrees alg C ok = refl

  private
    η-nat : {X Y : 𝒱 .ob} (f : NatTrans X Y) →
      seqTrans (η X) (T₀ ⟪ f ⟫) ≡ seqTrans f (η Y)
    η-nat f = sym (Tm.η .N-hom f)

    op-nat : {X Y : 𝒱 .ob} (f : NatTrans X Y) (w : W .ob) (ok : (Σ𝒯 ⟅ X ⟆ ⟅ w ⟆) .fst) →
      (T₀ ⟪ f ⟫) .N-ob w (freeOp X .N-ob w ok) ≡ freeOp Y .N-ob w ((Σ𝒯 ⟪ f ⟫) .N-ob w ok)
    op-nat {X} {Y} f w ok =
      cong (λ h → h .N-ob w ((Σ𝒯 ⟪ η X ⟫) .N-ob w ok)) ((F ⟪ f ⟫) .Hom.preserves)
      ∙ cong (λ g → Algebra.algebra ((F ⟅ Y ⟆) .fst) .N-ob w ((Σ𝒯 ⟪ g ⟫) .N-ob w ok)) (η-nat f)

  -- naturality of the effect step in the carrier
  stepPsh-natural : {X Y : 𝒱 .ob} (f : NatTrans X Y)
    (C@((s , p) , a) : (∂Q ⟅ X ⟆) .fst) (ok : (Σ𝒯 ⟅ X ⟆ ⟅ inQ s p ⟆) .fst) →
    (Q ⟪ f ⟫) (stepPsh X C ok) ≡ stepPsh Y ((∂Q ⟪ f ⟫) C) ((Σ𝒯 ⟪ f ⟫) .N-ob (inQ s p) ok)
  stepPsh-natural {X} {Y} f C@((s , p) , a) ok =
    sym (funExt⁻ (ρ .N-hom f) _)
    ∙ cong (ρ .N-ob Y) (plug-map Q′ (T₀ ⟅ X ⟆) (T₀ ⟪ f ⟫) ((∂Q ⟪ η X ⟫) C) (freeOp X .N-ob (inQ s p) ok))
    ∙ (λ i → ρ .N-ob Y (plug Q′ (T₀ ⟅ Y ⟆) ((∂Q ⟪ η-nat f i ⟫) C) (op-nat f (inQ s p) ok i)))

  ----------------------------------------------------------------------
  -- the sources of a redex (s , p , o), and its generic redex

  module _ (s : CShape .fst) (p : CPos s .fst) (o : Shape (inQ s p)) where
    -- another hole, or a position of the operation
    Source : Type
    Source = (Σ[ r ∈ CPos s .fst ] ¬ r ≡ p) ⊎ Position (inQ s p) o

    srcWorld : Source → W .ob
    srcWorld (inl (r , _)) = inQ s r
    srcWorld (inr q) = scope (inQ s p) o q

    isSetSource : isSet Source
    isSetSource = isSet⊎ (isSetΣ (CPos s .snd) (λ _ → isProp→isSet (isProp¬ _)))
      (P .Poly.Pos .F-ob (inQ s p , o) .snd)

    -- the coproduct of representables W [ srcWorld σ , − ]
    Gen : 𝒱 .ob
    Gen .F-ob w = (Σ[ σ ∈ Source ] W [ srcWorld σ , w ]) ,
      isSetΣ isSetSource (λ _ → W .isSetHom)
    Gen .F-hom f (σ , g) = σ , W ._⋆_ g f
    Gen .F-id = funExt λ (σ , g) → ΣPathP (refl , W .⋆IdR g)
    Gen .F-seq f f′ = funExt λ (σ , g) → ΣPathP (refl , sym (W .⋆Assoc g f f′))

    -- each source filled with its own generator
    genCtx : (∂Q ⟅ Gen ⟆) .fst
    genCtx = (s , p) , λ r ne → inl (r , ne) , W .id

    genArgs : (Σ𝒯 ⟅ Gen ⟆ ⟅ inQ s p ⟆) .fst
    genArgs = o , λ q → inr q , W .id

    -- the generic step
    generic : (Q ⟅ Gen ⟆) .fst
    generic = stepPsh Gen genCtx genArgs

    composite : CShape .fst
    composite = generic .fst

    -- where a hole of the result comes from, and the world map it is
    -- moved along
    origin : CPos composite .fst → Source
    origin r′ = generic .snd r′ .fst

    moved : (r′ : CPos composite .fst) → W [ srcWorld (origin r′) , inQ composite r′ ]
    moved r′ = generic .snd r′ .snd

  -- no generic redex uses a source twice
  ModuleAffine : Type
  ModuleAffine = ∀ s p o (r₁ r₂ : CPos (composite s p o) .fst) →
    origin s p o r₁ ≡ origin s p o r₂ → r₁ ≡ r₂

  ----------------------------------------------------------------------
  -- every actual step is the generic one, renamed (Yoneda)

  module _ (X : 𝒱 .ob) (C@((s , p) , a) : (∂Q ⟅ X ⟆) .fst)
    (ok@(o , k) : (Σ𝒯 ⟅ X ⟆ ⟅ inQ s p ⟆) .fst) where

    -- the filler at a source
    srcFill : (σ : Source s p o) → (X ⟅ srcWorld s p o σ ⟆) .fst
    srcFill (inl (r , ne)) = a r ne
    srcFill (inr q) = k q

    env : NatTrans (Gen s p o) X
    env .N-ob w (σ , g) = (X ⟪ g ⟫) (srcFill σ)
    env .N-hom f = funExt λ (σ , g) → funExt⁻ (X .F-seq g f) (srcFill σ)

    represent : stepPsh X C ok ≡ (Q ⟪ env ⟫) (generic s p o)
    represent =
      (λ i → stepPsh X ((s , p) , λ r ne → X .F-id (~ i) (a r ne))
                       (o , λ q → X .F-id (~ i) (k q)))
      ∙ sym (stepPsh-natural env (genCtx s p o) (genArgs s p o))

    -- the renamed generic step, read off definitionally
    composite-shape : ((Q ⟪ env ⟫) (generic s p o)) .fst ≡ composite s p o
    composite-shape = refl

    composite-filler : (r′ : CPos (composite s p o) .fst) →
      ((Q ⟪ env ⟫) (generic s p o)) .snd r′ ≡ (X ⟪ moved s p o r′ ⟫) (srcFill (origin s p o r′))
    composite-filler r′ = refl

  ----------------------------------------------------------------------
  -- configurations with at most one hole are affine

  affine-subsingleton : (∀ s → isProp (CPos s .fst)) → ModuleAffine
  affine-subsingleton prop s p o r₁ r₂ _ = prop (composite s p o) r₁ r₂
