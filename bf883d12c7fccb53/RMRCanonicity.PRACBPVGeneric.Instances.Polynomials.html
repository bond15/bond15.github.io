-- The RMR polynomials, with strength, of the instances, and some read-back.
--
-- (Strengths are stated at  Strength (polynomial L O), the form of the
-- definition: restating them at a separately named polynomial makes Agda
-- compare the two category structures by unfolding.)
module RMRCanonicity.PRACBPVGeneric.Instances.Polynomials where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Nat using (ℕ; _+_)
open import Cubical.Data.Unit using (Unit; tt)
open import Cubical.Data.Empty using (⊥)
open import Cubical.Data.Bool using (Bool; true; false)
open import Cubical.Data.FinData using (Fin)
open import Cubical.Data.Sigma using (_×_)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Instances.Terminal using (TerminalCategory)
open import Polynomial.Presheaf.FinBijective.LocalState.Injections using (Inj; extendInjection)
open import RightModuleReduction.Theory using (Signature; module SignatureNotation)

open import RMRCanonicity.PRACBPVGeneric.Instances.LocalState using (lsLocks; lsOps; lookup; new)
open import RMRCanonicity.PRACBPVGeneric.Instances.Guarded using (Omega; gLocks; gOps; step)
open import RMRCanonicity.PRACBPVGeneric.Instances.Constant using (noLocks; stateOps)
import RMRCanonicity.PRACBPVGeneric.Polynomial as P

------------------------------------------------------------------------
-- local state

localStatePoly : Signature Inj
localStatePoly = P.polynomial lsLocks lsOps

localStateStrength : P.Strength lsLocks lsOps (P.polynomial lsLocks lsOps)
localStateStrength = P.strength lsLocks lsOps

private
  module LS = SignatureNotation (P.polynomial lsLocks lsOps)

-- new b at world n: its slot lives at n + 1, reached by the prefix inclusion
newScope : ∀ n b → LS.scope n (new b , tt) (tt , tt) ≡ n + 1
newScope n b = refl

newStrength : ∀ n b →
  P.strength lsLocks lsOps .NatTrans.N-ob ((n , (new b , tt)) , (tt , tt)) ≡ extendInjection {n} {1}
newStrength n b = refl

-- lookup's two slots stay at world n (no lock)
lookupScope : ∀ n (i : Fin n) (a : Bool) → LS.scope n (lookup , (i , tt)) (a , tt) ≡ n
lookupScope n i a = refl

------------------------------------------------------------------------
-- guarded

guardedPoly : Signature Omega
guardedPoly = P.polynomial gLocks gOps

guardedStrength : P.Strength gLocks gOps (P.polynomial gLocks gOps)
guardedStrength = P.strength gLocks gOps

-- at world 0 step has no position
noStepAt0 : SignatureNotation.Position (P.polynomial gLocks gOps) 0 (step , tt) → ⊥
noStepAt0 (_ , ())

------------------------------------------------------------------------
-- Boolean global state

statePoly : Signature TerminalCategory
statePoly = P.polynomial (noLocks TerminalCategory) stateOps

stateStrength : P.Strength (noLocks TerminalCategory) stateOps (P.polynomial (noLocks TerminalCategory) stateOps)
stateStrength = P.strength (noLocks TerminalCategory) stateOps
