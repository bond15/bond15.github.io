RefPsh : ConfigPsh .ob
RefPsh .F-ob (n , _) = Fin n , isSetFin
RefPsh .F-hom f = f .fst .fst
RefPsh .F-id = refl
RefPsh .F-seq _ _ = refl

BoolPsh : ConfigPsh .ob
BoolPsh = Constant _ _ (Bool , isSetBool)

UnitPsh : ConfigPsh .ob
UnitPsh = Constant _ _ (Unit , isSetUnit)

updateStore : ∀ {n} → Fin n → Bool → (Fin n → Bool) → Fin n → Bool
updateStore i value σ j with discreteFin j i
... | yes _ = value
... | no _ = σ j

updateStore-same : ∀ {n} (i : Fin n) (value : Bool)
  (σ : Fin n → Bool) → updateStore i value σ i ≡ value
updateStore-same i value σ with discreteFin i i
... | yes _ = refl
... | no i≠i = ⊥.rec (i≠i refl)

updateStore-different : ∀ {n} (i : Fin n) (value : Bool)
  (σ : Fin n → Bool) (j : Fin n) → ¬ j ≡ i →
  updateStore i value σ j ≡ σ j
updateStore-different i value σ j j≠i with discreteFin j i
... | yes j≡i = ⊥.rec (j≠i j≡i)
... | no _ = refl

freshCell : (n : ℕ) → Fin (n + 1)
freshCell n = FinSumChar.fun n 1 (inr zero)

emptyStore : Fin 0 → Bool
emptyStore i = ⊥.rec (¬Fin0 i)

zeroStore : ∀ {n} → (Fin n → Bool) → Fin (n + 0) → Bool
zeroStore σ = σ ++Fin emptyStore

updateStore-map : ∀ {n n'} {σ : Fin n → Bool}
    {σ' : Fin n' → Bool} (f : Config [ (n , σ) , (n' , σ') ])
    (i : Fin n) (value : Bool) (x : Fin n) →
    updateStore (f .fst .fst i) value σ' (f .fst .fst x)
      ≡ updateStore i value σ x
updateStore-map f i value x
  with discreteFin (f .fst .fst x) (f .fst .fst i) | discreteFin x i
... | yes _ | yes _ = refl
... | yes p | no x≠i =
  ⊥.rec (x≠i (isEmbedding→Inj (f .fst .snd) x i p))
... | no fx≠fi | yes p =
  ⊥.rec (fx≠fi (cong (f .fst .fst) p))
... | no _ | no _ = funExt⁻ (f .snd) x

reindexStore-append : ∀ {n n' k}
    (rho : Injection n n')
    (source : Fin n → Bool)
    (target environment : Fin n' → Bool)
    (preserves : (λ x → target (rho .fst x)) ≡ source)
    (offImage : (y : Fin n') → ¬ ImageFiber rho y →
      environment y ≡ target y)
    (privateSuffix : Fin k → Bool) →
    reindexStore rho (source ++Fin privateSuffix) environment
      ≡ target ++Fin privateSuffix
reindexStore-append {n} {n'} {k} rho source target environment
    preserves offImage privateSuffix = funExt point
  where
    lhs = reindexStore rho (source ++Fin privateSuffix) environment
    rhs = target ++Fin privateSuffix

    point : (z : Fin (n' + k)) → lhs z ≡ rhs z
    point z = helper (FinSumChar.inv n' k z) refl
      where
      helper : (u : Fin n' ⊎ Fin k) →
        FinSumChar.inv n' k z ≡ u → lhs z ≡ rhs z
      helper (inl y) p with image? rho y
      ... | yes (x , q) =
        let z≡prefix-y =
              sym (FinSumChar.sec n' k z)
              ∙ cong (FinSumChar.fun n' k) p
        in cong lhs z≡prefix-y
          ∙ cong lhs (cong prefixFin (sym q))
          ∙ reindexStore-image rho (source ++Fin privateSuffix)
              environment x
          ∙ sym (FinSumChar.++FinInl n k source privateSuffix x)
          ∙ sym (funExt⁻ preserves x)
          ∙ cong target q
          ∙ FinSumChar.++FinInl n' k target privateSuffix y
          ∙ cong rhs (sym z≡prefix-y)
      ... | no not-image =
        let z≡prefix-y =
              sym (FinSumChar.sec n' k z)
              ∙ cong (FinSumChar.fun n' k) p
        in cong lhs z≡prefix-y
          ∙ reindexStore-complement rho (source ++Fin privateSuffix)
              environment y not-image
          ∙ offImage y not-image
          ∙ FinSumChar.++FinInl n' k target privateSuffix y
          ∙ cong rhs (sym z≡prefix-y)
      helper (inr c) p =
        let z≡suffix-c =
              sym (FinSumChar.sec n' k z)
              ∙ cong (FinSumChar.fun n' k) p
        in cong lhs z≡suffix-c
          ∙ reindexStore-private rho (source ++Fin privateSuffix)
              environment c
          ∙ sym (FinSumChar.++FinInr n k source privateSuffix c)
          ∙ FinSumChar.++FinInr n' k target privateSuffix c
          ∙ cong rhs (sym z≡suffix-c)

reindexStore-zero : ∀ {n n'} {σ : Fin n → Bool}
    {σ' : Fin n' → Bool} (f : Config [ (n , σ) , (n' , σ') ]) →
    reindexStore (f .fst) (zeroStore σ) σ' ≡ zeroStore σ'
reindexStore-zero {σ = σ} {σ' = σ'} f =
  reindexStore-append (f .fst) σ σ' σ' (f .snd)
    (λ _ _ → refl) emptyStore

reindexStore-update : ∀ {n n'} {σ : Fin n → Bool}
  {σ' : Fin n' → Bool} (f : Config [ (n , σ) , (n' , σ') ])
  (i : Fin n) (value : Bool) →
  reindexStore (f .fst) (zeroStore (updateStore i value σ)) σ'
    ≡ zeroStore (updateStore (f .fst .fst i) value σ')
reindexStore-update {σ = σ} {σ' = σ'} f i value =
  reindexStore-append
    (f .fst)
    (updateStore i value σ)
    (updateStore (f .fst .fst i) value σ')
    σ'
    (funExt (updateStore-map {σ = σ} {σ' = σ'} f i value))
    (λ y not-image →
      sym (updateStore-different (f .fst .fst i) value σ' y
        (λ y≡fi → not-image (i , sym y≡fi))))
    emptyStore
