-- Tree reduction is stable under substitution and renaming (5.6).
--
--   sub-plug   subC σ (plug S M) ≡ plug (subS σ S) (subC σ M)
--   sub-⇝      M ⇝ M' → subC σ M ⇝ subC σ M'
--   sub-↦      M ↦ M' → subC σ M ↦ subC σ M'
--   sub-↦*     M ↦* M' → subC σ M ↦* subC σ M'
--   ren-⇝, ren-↦, ren-↦*   the same for renamings (via sub-ren)
--
-- The β-steps are the substitution lemmas sub-sub₁ / sub-sub₂.  The frame
-- steps push a frame across the locks of a slot's word (dropW); under σ
-- that is sub-dropW (op-to) and sub-dropWV (op-app): crossing the word
-- then substituting under it (liftWS) is substituting then crossing.
-- The reducts are moved along these equations with subst; nothing
-- computes on the transported steps.
open import Cubical.Foundations.Prelude hiding (Sub; _,_; comp)
open import Cubical.Categories.Category using (Category)

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Metatheory.SubstReduction
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
open import RMRCanonicity.PRACBPVGeneric.Reduction L O
open import RMRCanonicity.PRACBPVGeneric.Metatheory.Substitution L O
  using (sub-sub₁; sub-sub₂; sub-dropW; sub-dropWV; sub-renC)

sub-plug : ∀ {Θ Θ' B B'} (σ : Sub Θ Θ') (S : Stack Θ B B') (M : Comp Θ B) →
  subC σ (plug S M) ≡ plug (subS σ S) (subC σ M)
sub-plug σ hole M = refl
sub-plug σ (S to• N) M = cong (_to subC (liftS σ) N) (sub-plug σ S M)
sub-plug σ (app• S V) M = cong (λ X → app X (subV σ V)) (sub-plug σ S M)

sub-⇝ : ∀ {Θ Θ' B} (σ : Sub Θ Θ') {M M' : Comp Θ B} → M ⇝ M' → subC σ M ⇝ subC σ M'
sub-⇝ σ (β-to {V = V} {N}) =
  subst (subC σ (return V to N) ⇝_) (sym (sub-sub₁ σ N V)) β-to
sub-⇝ σ β-force = β-force
sub-⇝ σ (β-case {V = V} {V'} {M}) =
  subst (subC σ (case (pair V V') M) ⇝_) (sym (sub-sub₂ σ M V V')) β-case
sub-⇝ σ (β-case₊-inl {V = V} {M} {N}) =
  subst (subC σ (case₊ (inl V) M N) ⇝_) (sym (sub-sub₁ σ M V)) β-case₊-inl
sub-⇝ σ (β-case₊-inr {V = V} {M} {N}) =
  subst (subC σ (case₊ (inr V) M N) ⇝_) (sym (sub-sub₁ σ N V)) β-case₊-inr
sub-⇝ σ (β-app {M = M} {V}) =
  subst (subC σ (app (lam M) V) ⇝_) (sym (sub-sub₁ σ M V)) β-app
sub-⇝ σ (op-to o vs {k} {N}) =
  subst (subC σ (op o vs k to N) ⇝_)
    (cong (op o (subVs σ vs)) (funExt λ a → funExt λ ps →
      cong (subC (liftWS (slot o a) σ ps) (k a (wposW (slot o a) (wmS σ) ps)) to_)
        (sym (sub-dropW σ (slot o a) ps N))))
    (op-to o (subVs σ vs))
sub-⇝ σ (op-app o vs {k} {V}) =
  subst (subC σ (app (op o vs k) V) ⇝_)
    (cong (op o (subVs σ vs)) (funExt λ a → funExt λ ps →
      cong (app (subC (liftWS (slot o a) σ ps) (k a (wposW (slot o a) (wmS σ) ps))))
        (sym (sub-dropWV σ (slot o a) ps V))))
    (op-app o (subVs σ vs))

sub-↦ : ∀ {Θ Θ' B} (σ : Sub Θ Θ') {M M' : Comp Θ B} → M ↦ M' → subC σ M ↦ subC σ M'
sub-↦ σ (step S {M} {M'} r) =
  subst2 _↦_ (sym (sub-plug σ S M)) (sym (sub-plug σ S M')) (step (subS σ S) (sub-⇝ σ r))

sub-↦* : ∀ {Θ Θ' B} (σ : Sub Θ Θ') {M M' : Comp Θ B} → M ↦* M' → subC σ M ↦* subC σ M'
sub-↦* σ refl* = refl*
sub-↦* σ (step* s r) = step* (sub-↦ σ s) (sub-↦* σ r)
sub-↦* σ (op* {o = o} rs) =
  op* λ a ps → sub-↦* (liftWS (slot o a) σ ps) (rs a (wposW (slot o a) (wmS σ) ps))

ren-⇝ : ∀ {Θ Θ' B} (ρ : Ren Θ Θ') {M M' : Comp Θ B} → M ⇝ M' → renC ρ M ⇝ renC ρ M'
ren-⇝ ρ {M} {M'} r = subst2 _⇝_ (sub-renC ρ M) (sub-renC ρ M') (sub-⇝ ⌜ ρ ⌝ r)

ren-↦ : ∀ {Θ Θ' B} (ρ : Ren Θ Θ') {M M' : Comp Θ B} → M ↦ M' → renC ρ M ↦ renC ρ M'
ren-↦ ρ {M} {M'} s = subst2 _↦_ (sub-renC ρ M) (sub-renC ρ M') (sub-↦ ⌜ ρ ⌝ s)

ren-↦* : ∀ {Θ Θ' B} (ρ : Ren Θ Θ') {M M' : Comp Θ B} → M ↦* M' → renC ρ M ↦* renC ρ M'
ren-↦* ρ {M} {M'} r = subst2 _↦*_ (sub-renC ρ M) (sub-renC ρ M') (sub-↦* ⌜ ρ ⌝ r)
