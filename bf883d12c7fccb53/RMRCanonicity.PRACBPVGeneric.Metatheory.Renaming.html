-- The renaming theory of the syntax: identity, fusion and extensionality
-- (Thms 8.2-8.4).
--
-- WHY HETEROGENEOUS.  A renaming acts on a term through two things only:
-- its action on neutrals (renNe ρ) and its world map (toW (wm ρ)).  Under
-- a lock, renV ρ (shut V) reads V at the pulled-back position
-- posW κ (wm ρ) q, so two renamings with the same world map read V at two
-- positions that are equal only propositionally (pos κ W.id q ≡ q is the
-- arity's F-id, pos κ (f ⋆ g) q ≡ pos κ f (pos κ g q) its F-seq).  The
-- renamed subterms then live over different telescopes.  So every theorem
-- is stated over an ARBITRARY path of source telescopes e, with a path of
-- terms over it:
--
--   Agree e ρ₁ ρ₂            ρ₁ and ρ₂ agree on neutrals and on world maps
--                            (over e)
--   Composite e ρ₁ ρ₂ ρ₁₂    ρ₁ then ρ₂ agrees with ρ₁₂ (over e)
--
--   ren-agree-id   Agree e ρ idR → t ≡[ e ] t' → renX ρ t ≡ t'
--   ren-fuse       Composite e ρ₁ ρ₂ ρ₁₂ → t ≡[ e ] t' →
--                    renX ρ₂ (renX ρ₁ t) ≡ renX ρ₁₂ t'
--   ren-id         renX idR t ≡ t                          (Thm 8.3)
--   ren-ext        Agree e ρ₁ ρ₂ → t₁ ≡[ e ] t₂ → renX ρ₁ t₁ ≡ renX ρ₂ t₂
--                                                          (Thm 8.2)
-- for X = V (values), Vs (value lists) and C (computations); ren-fuse is
-- Thm 8.4.
--
-- METHOD.  Each theorem first eliminates the telescope path and the term
-- path together (J-tele), which leaves the same telescope and the same
-- term, and then inducts on the term.  Inside the induction the only
-- non-trivial paths are the ones the closure lemmas build:
--
--   keep     the same path, one variable further  (AgreeId-keep,
--            Composite-keep)
--   keepL    a path of POSITIONS (AgreeId-pt, Composite-pt), built in
--            Ey κ w x = Σ_p W [ scope p , x ], the elements of the arity
--            E κ w at x: the pulled-back position together with the scope
--            map, so the first component moves the telescope and the
--            second is the new world map  (AgreeId-keepL, Composite-keepL)
--   keepW    the keepL paths along a lock word, outermost lock first
--            (AgreeId-pts / -keepW, Composite-pts / -keepW)
--   root     out of the root ∅ u there are no neutrals, so agreement is
--            agreement of world maps (Yoneda: y u is free on one element)
--            (Agree-root, Composite-root)
--
-- Only agreement WITH THE IDENTITY needs closure lemmas: ren-ext is
-- fusion with idR (Agree→Composite) followed by ren-id.
--
-- Transport is never computed on Ne, Val or Comp (indexed families); J is
-- only used to produce paths, never reduced.

-- elements of a lock kind's arity

-- the elements at x of Ey κ w = Σ_{p ∈ Pos κ w} y (scope κ w p) (the
-- arity E κ w as a presheaf, math spec §1): a position at w with a map out
-- of its scope
Ey : Lk → World → World → Type
Ey κ w x = Σ[ p ∈ Pos κ w ] W [ scope κ w p , x ]

-- the arity on a map f, at q: (pos f q , smap f q)
pt : ∀ κ {w w'} → W [ w , w' ] → (q : Pos κ w') → Ey κ w (scope κ w' q)
pt κ f q = pos κ f q ,ˢ smap κ f q

-- the same, on a syntactic world map (posW, smapW)
ptW : ∀ κ {w w'} → WMap w w' → (q : Pos κ w') → Ey κ w (scope κ w' q)
ptW κ g q = posW κ g q ,ˢ toW (smapW κ g q)

-- the arity's functor laws, at one position
pt-id : ∀ κ {w} (q : Pos κ w) → pt κ W.id q ≡ (q ,ˢ W.id)
pt-id κ q i = E κ .F-id i .fst q ,ˢ E κ .F-id i .snd q

pt-seq : ∀ κ {u v w} (f : W [ u , v ]) (g : W [ v , w ]) (q : Pos κ w) →
  pt κ (f W.⋆ g) q ≡ (pos κ f (pos κ g q) ,ˢ smap κ f (pos κ g q) W.⋆ smap κ g q)
pt-seq κ f g q i = E κ .F-seq g f i .fst q ,ˢ E κ .F-seq g f i .snd q

-- the syntactic world maps denote their W-maps (the strict identity idW
-- denotes W.id only up to the arity's F-id)
ptW-toW : ∀ κ {w w'} (g : WMap w w') (q : Pos κ w') → ptW κ g q ≡ pt κ (toW g) q
ptW-toW κ idW q = sym (pt-id κ q)
ptW-toW κ (mapW f) q = refl

-- pulling back along g₂ then g₁ is pulling back along their composite
ptW-seq : ∀ κ {w₁ w₂ w₃} (g₁ : WMap w₁ w₂) (g₂ : WMap w₂ w₃) (q : Pos κ w₃) →
  Path (Ey κ w₁ (scope κ w₃ q))
    (posW κ g₁ (posW κ g₂ q) ,ˢ toW (smapW κ g₁ (posW κ g₂ q)) W.⋆ toW (smapW κ g₂ q))
    (pt κ (toW g₁ W.⋆ toW g₂) q)
ptW-seq κ g₁ g₂ q =
  cong (λ y → y .fst ,ˢ y .snd W.⋆ toW (smapW κ g₂ q)) (ptW-toW κ g₁ (posW κ g₂ q))
  ∙ cong (λ y → pos κ (toW g₁) (y .fst) ,ˢ smap κ (toW g₁) (y .fst) W.⋆ y .snd) (ptW-toW κ g₂ q)
  ∙ sym (pt-seq κ (toW g₁) (toW g₂) q)

cmapW-seq : ∀ {w₁ w₂ w₃ b} (g₁ : WMap w₁ w₂) (g₂ : WMap w₂ w₃) (g₁₂ : WMap w₁ w₃) →
  toW g₁ W.⋆ toW g₂ ≡ toW g₁₂ → (c : Cst b w₁) → cmapW g₂ (cmapW g₁ c) ≡ cmapW g₁₂ c
cmapW-seq {b = b} g₁ g₂ g₁₂ r c =
  cmapW-toW g₂ (cmapW g₁ c)
  ∙ cong (Const b .F-hom (toW g₂)) (cmapW-toW g₁ c)
  ∙ sym (funExt⁻ (Const b .F-seq (toW g₁) (toW g₂)) c)
  ∙ cong (λ h → Const b .F-hom h c) r
  ∙ sym (cmapW-toW g₁₂ c)

cmapW-id : ∀ {w b} (g : WMap w w) → toW g ≡ W.id → (c : Cst b w) → cmapW g c ≡ c
cmapW-id {b = b} g r c =
  cmapW-toW g c ∙ cong (λ h → Const b .F-hom h c) r ∙ funExt⁻ (Const b .F-id) c

----------------------------------------------------------------------
-- neutrals under keep and keepL

-- the root has no neutrals
Ne∅ : ∀ {w A} → Ne (∅ w) A → ⊥
Ne∅ ()

ne∅ : ∀ {u u' Θ' A} (e : u ≡ u') (f₀ : Ne (∅ u) A → Ne Θ' A) (f₁ : Ne (∅ u') A → Ne Θ' A) →
  PathP (λ i → Ne (∅ (e i)) A → Ne Θ' A) f₀ f₁
ne∅ e f₀ f₁ = isProp→PathP (λ i → isPropΠ λ n → ⊥.rec (Ne∅ n)) f₀ f₁

-- keep and keepL act on neutrals through the action under them
liftNe, : ∀ {Δ Δ' B} → (∀ {A} → Ne Δ A → Ne Δ' A) → ∀ {A} → Ne (Δ , B) A → Ne (Δ' , B) A
liftNe, f here = here
liftNe, f (wkv n) = wkv (f n)

liftNe⧀ : ∀ {Δ Δ'} κ {p q} → (∀ {A} → Ne Δ A → Ne Δ' A) →
  ∀ {A} → Ne (Δ ⧀⟨ κ , p ⟩) A → Ne (Δ' ⧀⟨ κ , q ⟩) A
liftNe⧀ κ f (up n) = up (f n)
liftNe⧀ κ f (opn n) = opn (f n)
liftNe⧀ κ f (nm k) = nm k

private
  renNe-keep : ∀ {Δ Δ' B A} (ρ : Ren Δ Δ') (n : Ne (Δ , B) A) →
    renNe (keep ρ) n ≡ liftNe, (renNe ρ) n
  renNe-keep ρ here = refl
  renNe-keep ρ (wkv n) = refl

  renNe-keep² : ∀ {Δ₁ Δ₂ Δ₃ B A} (ρ₁ : Ren Δ₁ Δ₂) (ρ₂ : Ren Δ₂ Δ₃) (n : Ne (Δ₁ , B) A) →
    renNe (keep ρ₂) (renNe (keep ρ₁) n) ≡ liftNe, (λ m → renNe ρ₂ (renNe ρ₁ m)) n
  renNe-keep² ρ₁ ρ₂ here = refl
  renNe-keep² ρ₁ ρ₂ (wkv n) = refl

  liftNe,-id : ∀ {Δ B A} (n : Ne (Δ , B) A) → liftNe, (λ m → m) n ≡ n
  liftNe,-id here = refl
  liftNe,-id (wkv n) = refl

  renNe-keepL : ∀ {Δ Δ' κ A} (ρ : Ren Δ Δ') (q : Pos κ (world Δ'))
    (n : Ne (Δ ⧀⟨ κ , posW κ (wm ρ) q ⟩) A) → renNe (keepL ρ q) n ≡ liftNe⧀ κ (renNe ρ) n
  renNe-keepL ρ q (up n) = refl
  renNe-keepL ρ q (opn n) = refl
  renNe-keepL ρ q (nm k) = refl

  renNe-keepL² : ∀ {Δ₁ Δ₂ Δ₃ κ A} (ρ₁ : Ren Δ₁ Δ₂) (ρ₂ : Ren Δ₂ Δ₃) (q : Pos κ (world Δ₃))
    (n : Ne (Δ₁ ⧀⟨ κ , posW κ (wm ρ₁) (posW κ (wm ρ₂) q) ⟩) A) →
    renNe (keepL ρ₂ q) (renNe (keepL ρ₁ (posW κ (wm ρ₂) q)) n)
      ≡ liftNe⧀ κ (λ m → renNe ρ₂ (renNe ρ₁ m)) n
  renNe-keepL² ρ₁ ρ₂ q (up n) = refl
  renNe-keepL² ρ₁ ρ₂ q (opn n) = refl
  renNe-keepL² ρ₁ ρ₂ q (nm k) = refl

  liftNe⧀-id : ∀ {Δ κ q A} (n : Ne (Δ ⧀⟨ κ , q ⟩) A) → liftNe⧀ κ (λ m → m) n ≡ n
  liftNe⧀-id (up n) = refl
  liftNe⧀-id (opn n) = refl
  liftNe⧀-id (nm k) = refl

-- agreement and composites, over a path of source telescopes

record Agree {Θ₁ Θ₂ Θ'} (e : Θ₁ ≡ Θ₂) (ρ₁ : Ren Θ₁ Θ') (ρ₂ : Ren Θ₂ Θ') : Type where
  field
    on-ne : ∀ {A} → PathP (λ i → Ne (e i) A → Ne Θ' A) (renNe ρ₁) (renNe ρ₂)
    on-world : PathP (λ i → W [ world (e i) , world Θ' ]) (toW (wm ρ₁)) (toW (wm ρ₂))

record Composite {Θ₁ Θ₁' Θ₂ Θ₃} (e : Θ₁ ≡ Θ₁')
  (ρ₁ : Ren Θ₁ Θ₂) (ρ₂ : Ren Θ₂ Θ₃) (ρ₁₂ : Ren Θ₁' Θ₃) : Type where
  field
    on-ne : ∀ {A} → PathP (λ i → Ne (e i) A → Ne Θ₃ A) (λ n → renNe ρ₂ (renNe ρ₁ n)) (renNe ρ₁₂)
    on-world : PathP (λ i → W [ world (e i) , world Θ₃ ])
      (toW (wm ρ₁) W.⋆ toW (wm ρ₂)) (toW (wm ρ₁₂))

----------------------------------------------------------------------
-- agreeing with the identity: closure under keep, keepL, keepW

Agree-idR : ∀ {Θ} → Agree refl (idR {Θ}) idR
Agree-idR .on-ne = refl
Agree-idR .on-world = refl

module _ {Θ₁ Θ₂} {e : Θ₁ ≡ Θ₂} {ρ : Ren Θ₁ Θ₂} where

  AgreeId-keep : ∀ {B} → Agree e ρ idR → Agree (λ i → e i , B) (keep ρ) idR
  AgreeId-keep ag .on-ne =
    funExt (renNe-keep ρ) ◁ (λ i → liftNe, (λ {A} → ag .on-ne {A} i)) ▷ funExt liftNe,-id
  AgreeId-keep ag .on-world = ag .on-world

  -- the position path under keepL: (posW κ (wm ρ) q , smapW …) to (q , id)
  AgreeId-pt : ∀ {κ} → Agree e ρ idR → (q : Pos κ (world Θ₂)) →
    PathP (λ i → Ey κ (world (e i)) (scope κ (world Θ₂) q)) (ptW κ (wm ρ) q) (q ,ˢ W.id)
  AgreeId-pt {κ} ag q = ptW-toW κ (wm ρ) q ◁ (λ i → pt κ (ag .on-world i) q) ▷ pt-id κ q

  AgreeId-keepL : ∀ {κ} (ag : Agree e ρ idR) (q : Pos κ (world Θ₂)) →
    Agree (λ i → e i ⧀⟨ κ , AgreeId-pt ag q i .fst ⟩) (keepL ρ q) idR
  AgreeId-keepL {κ} ag q .on-ne =
    funExt (renNe-keepL ρ q)
    ◁ (λ i → liftNe⧀ κ {p = AgreeId-pt ag q i .fst} {q = q} (λ {A} → ag .on-ne {A} i))
    ▷ funExt liftNe⧀-id
  AgreeId-keepL ag q .on-world i = AgreeId-pt ag q i .snd

-- along a word: the positions, and the agreement over them
AgreeId-pts : ∀ {Θ₁ Θ₂} {e : Θ₁ ≡ Θ₂} {ρ : Ren Θ₁ Θ₂} (μ : List Lk) → Agree e ρ idR →
  (ps : WPos μ (world Θ₂)) → PathP (λ i → WPos μ (world (e i))) (wposW μ (wm ρ) ps) ps
AgreeId-pts [] ag ps = refl
AgreeId-pts (κ ∷ []) ag p i = AgreeId-pt ag p i .fst
AgreeId-pts (κ ∷ μ@(_ ∷ _)) ag (p ,ˢ ps) i =
  AgreeId-pt ag p i .fst ,ˢ AgreeId-pts μ (AgreeId-keepL ag p) ps i

AgreeId-keepW : ∀ {Θ₁ Θ₂} {e : Θ₁ ≡ Θ₂} {ρ : Ren Θ₁ Θ₂} (μ : List Lk) (ag : Agree e ρ idR)
  (ps : WPos μ (world Θ₂)) →
  Agree (λ i → e i ⧀*⟨ μ , AgreeId-pts μ ag ps i ⟩) (keepW μ ρ ps) idR
AgreeId-keepW [] ag ps = ag
AgreeId-keepW (κ ∷ []) ag p = AgreeId-keepL ag p
AgreeId-keepW (κ ∷ μ@(_ ∷ _)) ag (p ,ˢ ps) = AgreeId-keepW μ (AgreeId-keepL ag p) ps

-- out of the root, agreement is agreement of world maps
Agree-root : ∀ {u u' Θ'} (e : u ≡ u') {ρ₁ : Ren (∅ u) Θ'} {ρ₂ : Ren (∅ u') Θ'} →
  PathP (λ i → W [ e i , world Θ' ]) (toW (wm ρ₁)) (toW (wm ρ₂)) → Agree (cong ∅ e) ρ₁ ρ₂
Agree-root e r .on-ne = ne∅ e _ _
Agree-root e r .on-world = r

----------------------------------------------------------------------
-- composites: closure under keep, keepL, keepW

module _ {Θ₁ Θ₁' Θ₂ Θ₃} {e : Θ₁ ≡ Θ₁'}
  {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁' Θ₃} where

  Composite-keep : ∀ {B} → Composite e ρ₁ ρ₂ ρ₁₂ →
    Composite (λ i → e i , B) (keep ρ₁) (keep ρ₂) (keep ρ₁₂)
  Composite-keep c .on-ne =
    funExt (renNe-keep² ρ₁ ρ₂)
    ◁ (λ i → liftNe, (λ {A} → c .on-ne {A} i))
    ▷ sym (funExt (renNe-keep ρ₁₂))
  Composite-keep c .on-world = c .on-world

  -- the position path under keepL: pulling back along ρ₂ then ρ₁, to
  -- pulling back along ρ₁₂ (the arity's F-seq, then the world maps' path)
  Composite-pt : ∀ {κ} → Composite e ρ₁ ρ₂ ρ₁₂ → (q : Pos κ (world Θ₃)) →
    PathP (λ i → Ey κ (world (e i)) (scope κ (world Θ₃) q))
      (posW κ (wm ρ₁) (posW κ (wm ρ₂) q) ,ˢ
        toW (smapW κ (wm ρ₁) (posW κ (wm ρ₂) q)) W.⋆ toW (smapW κ (wm ρ₂) q))
      (ptW κ (wm ρ₁₂) q)
  Composite-pt {κ} c q =
    ptW-seq κ (wm ρ₁) (wm ρ₂) q ◁ (λ i → pt κ (c .on-world i) q) ▷ sym (ptW-toW κ (wm ρ₁₂) q)

  Composite-keepL : ∀ {κ} (c : Composite e ρ₁ ρ₂ ρ₁₂) (q : Pos κ (world Θ₃)) →
    Composite (λ i → e i ⧀⟨ κ , Composite-pt c q i .fst ⟩)
      (keepL ρ₁ (posW κ (wm ρ₂) q)) (keepL ρ₂ q) (keepL ρ₁₂ q)
  Composite-keepL {κ} c q .on-ne =
    funExt (renNe-keepL² ρ₁ ρ₂ q)
    ◁ (λ i → liftNe⧀ κ {p = Composite-pt c q i .fst} {q = q} (λ {A} → c .on-ne {A} i))
    ▷ sym (funExt (renNe-keepL ρ₁₂ q))
  Composite-keepL c q .on-world i = Composite-pt c q i .snd

-- along a word: the positions, and the composite over them
Composite-pts : ∀ {Θ₁ Θ₁' Θ₂ Θ₃} {e : Θ₁ ≡ Θ₁'}
  {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁' Θ₃} (μ : List Lk) → Composite e ρ₁ ρ₂ ρ₁₂ → (ps : WPos μ (world Θ₃)) →
  PathP (λ i → WPos μ (world (e i))) (wposW μ (wm ρ₁) (wposW μ (wm ρ₂) ps)) (wposW μ (wm ρ₁₂) ps)
Composite-pts [] c ps = refl
Composite-pts (κ ∷ []) c p i = Composite-pt c p i .fst
Composite-pts (κ ∷ μ@(_ ∷ _)) c (p ,ˢ ps) i =
  Composite-pt c p i .fst ,ˢ Composite-pts μ (Composite-keepL c p) ps i

Composite-keepW : ∀ {Θ₁ Θ₁' Θ₂ Θ₃} {e : Θ₁ ≡ Θ₁'}
  {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁' Θ₃}
  (μ : List Lk) (c : Composite e ρ₁ ρ₂ ρ₁₂) (ps : WPos μ (world Θ₃)) →
  Composite (λ i → e i ⧀*⟨ μ , Composite-pts μ c ps i ⟩)
    (keepW μ ρ₁ (wposW μ (wm ρ₂) ps)) (keepW μ ρ₂ ps) (keepW μ ρ₁₂ ps)
Composite-keepW [] c ps = c
Composite-keepW (κ ∷ []) c p = Composite-keepL c p
Composite-keepW (κ ∷ μ@(_ ∷ _)) c (p ,ˢ ps) = Composite-keepW μ (Composite-keepL c p) ps

-- out of the root, a composite is a composite of world maps
Composite-root : ∀ {u u' Θ₂ Θ₃} (e : u ≡ u')
  {ρ₁ : Ren (∅ u) Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren (∅ u') Θ₃} →
  PathP (λ i → W [ e i , world Θ₃ ]) (toW (wm ρ₁) W.⋆ toW (wm ρ₂)) (toW (wm ρ₁₂)) →
  Composite (cong ∅ e) ρ₁ ρ₂ ρ₁₂
Composite-root e r .on-ne = ne∅ e _ _
Composite-root e r .on-world = r

-- agreement is a composite with the identity
Agree→Composite : ∀ {Θ₁ Θ₂ Θ'} {e : Θ₁ ≡ Θ₂} {ρ₁ : Ren Θ₁ Θ'} {ρ₂ : Ren Θ₂ Θ'} →
  Agree e ρ₁ ρ₂ → Composite e ρ₁ idR ρ₂
Agree→Composite ag .on-ne = ag .on-ne
Agree→Composite {ρ₁ = ρ₁} ag .on-world = W.⋆IdR (toW (wm ρ₁)) ◁ ag .on-world

----------------------------------------------------------------------
-- eliminating a path of telescopes with a path of terms over it

J-tele : (X : Tele → Type) {Θ : Tele} {t : X Θ}
  (P : (Θ' : Tele) → Θ ≡ Θ' → X Θ' → Type) → P Θ refl t →
  ∀ {Θ'} (e : Θ ≡ Θ') {t' : X Θ'} → PathP (λ i → X (e i)) t t' → P Θ' e t'
J-tele X P d e tp = J (λ x q → P (x .fst) (cong fst q) (x .snd)) d (ΣPathP (e ,ˢ tp))

----------------------------------------------------------------------
-- a renaming that agrees with the identity acts as the identity

ren-agree-idV : ∀ {Θ₁ Θ₂ A} {e : Θ₁ ≡ Θ₂} {ρ : Ren Θ₁ Θ₂} → Agree e ρ idR →
  (t : Val Θ₁ A) {t' : Val Θ₂ A} → PathP (λ i → Val (e i) A) t t' → renV ρ t ≡ t'
ren-agree-idVs : ∀ {Θ₁ Θ₂ bs} {e : Θ₁ ≡ Θ₂} {ρ : Ren Θ₁ Θ₂} → Agree e ρ idR →
  (t : Vals Θ₁ bs) {t' : Vals Θ₂ bs} → PathP (λ i → Vals (e i) bs) t t' → renVs ρ t ≡ t'
ren-agree-idC : ∀ {Θ₁ Θ₂ B} {e : Θ₁ ≡ Θ₂} {ρ : Ren Θ₁ Θ₂} → Agree e ρ idR →
  (t : Comp Θ₁ B) {t' : Comp Θ₂ B} → PathP (λ i → Comp (e i) B) t t' → renC ρ t ≡ t'

-- the same telescope and the same term: induction on the term
private
  idV : ∀ {Θ A} {ρ : Ren Θ Θ} → Agree refl ρ idR → (t : Val Θ A) → renV ρ t ≡ t
  idVs : ∀ {Θ bs} {ρ : Ren Θ Θ} → Agree refl ρ idR → (t : Vals Θ bs) → renVs ρ t ≡ t
  idC : ∀ {Θ B} {ρ : Ren Θ Θ} → Agree refl ρ idR → (t : Comp Θ B) → renC ρ t ≡ t

  idV ag (ne n) i = ne (ag .on-ne i n)
  idV ag tt = refl
  idV ag (inl V) = cong inl (idV ag V)
  idV ag (inr V) = cong inr (idV ag V)
  idV ag (pair V V') = cong₂ pair (idV ag V) (idV ag V')
  idV ag (thunk M) = cong thunk (idC ag M)
  idV {ρ = ρ} ag (const c) = cong const (cmapW-id (wm ρ) (ag .on-world) c)
  idV ag (shut V) = cong shut (funExt λ q →
    ren-agree-idV (AgreeId-keepL ag q) (V _) (λ i → V (AgreeId-pt ag q i .fst)))

  idVs ag []ᵥ = refl
  idVs ag (V ∷ᵥ vs) = cong₂ _∷ᵥ_ (idV ag V) (idVs ag vs)

  idC ag (return V) = cong return (idV ag V)
  idC ag (M to N) = cong₂ _to_ (idC ag M) (idC (AgreeId-keep ag) N)
  idC ag (force V) = cong force (idV ag V)
  idC ag (case V M) = cong₂ case (idV ag V) (idC (AgreeId-keep (AgreeId-keep ag)) M)
  idC ag (case₊ V M N) =
    cong₃ case₊ (idV ag V) (idC (AgreeId-keep ag) M) (idC (AgreeId-keep ag) N)
  idC ag (absurd V) = cong absurd (idV ag V)
  idC ag (lam M) = cong lam (idC (AgreeId-keep ag) M)
  idC ag (app M V) = cong₂ app (idC ag M) (idV ag V)
  idC ag (op o vs k) = cong₂ (op o) (idVs ag vs) (funExt λ a → funExt λ ps →
    ren-agree-idC (AgreeId-keepW (slot o a) ag ps) (k a _) (λ i → k a (AgreeId-pts (slot o a) ag ps i)))

ren-agree-idV {Θ₁} {A = A} ag t tp =
  J-tele (λ Θ → Val Θ A) (λ Θ' e' t' → (ρ' : Ren Θ₁ Θ') → Agree e' ρ' idR → renV ρ' t ≡ t')
    (λ ρ' ag' → idV ag' t) _ tp _ ag
ren-agree-idVs {Θ₁} {bs = bs} ag t tp =
  J-tele (λ Θ → Vals Θ bs) (λ Θ' e' t' → (ρ' : Ren Θ₁ Θ') → Agree e' ρ' idR → renVs ρ' t ≡ t')
    (λ ρ' ag' → idVs ag' t) _ tp _ ag
ren-agree-idC {Θ₁} {B = B} ag t tp =
  J-tele (λ Θ → Comp Θ B) (λ Θ' e' t' → (ρ' : Ren Θ₁ Θ') → Agree e' ρ' idR → renC ρ' t ≡ t')
    (λ ρ' ag' → idC ag' t) _ tp _ ag

-- Thm 8.3: the identity renaming
ren-idV : ∀ {Θ A} (t : Val Θ A) → renV idR t ≡ t
ren-idV t = ren-agree-idV Agree-idR t refl

ren-idVs : ∀ {Θ bs} (t : Vals Θ bs) → renVs idR t ≡ t
ren-idVs t = ren-agree-idVs Agree-idR t refl

ren-idC : ∀ {Θ B} (t : Comp Θ B) → renC idR t ≡ t
ren-idC t = ren-agree-idC Agree-idR t refl

-- Thm 8.4: fusion

ren-fuseV : ∀ {Θ₁ Θ₁' Θ₂ Θ₃ A} {e : Θ₁ ≡ Θ₁'}
  {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁' Θ₃} → Composite e ρ₁ ρ₂ ρ₁₂ →
  (t : Val Θ₁ A) {t' : Val Θ₁' A} → PathP (λ i → Val (e i) A) t t' →
  renV ρ₂ (renV ρ₁ t) ≡ renV ρ₁₂ t'
ren-fuseVs : ∀ {Θ₁ Θ₁' Θ₂ Θ₃ bs} {e : Θ₁ ≡ Θ₁'}
  {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁' Θ₃} → Composite e ρ₁ ρ₂ ρ₁₂ →
  (t : Vals Θ₁ bs) {t' : Vals Θ₁' bs} → PathP (λ i → Vals (e i) bs) t t' →
  renVs ρ₂ (renVs ρ₁ t) ≡ renVs ρ₁₂ t'
ren-fuseC : ∀ {Θ₁ Θ₁' Θ₂ Θ₃ B} {e : Θ₁ ≡ Θ₁'}
  {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁' Θ₃} → Composite e ρ₁ ρ₂ ρ₁₂ →
  (t : Comp Θ₁ B) {t' : Comp Θ₁' B} → PathP (λ i → Comp (e i) B) t t' →
  renC ρ₂ (renC ρ₁ t) ≡ renC ρ₁₂ t'

private
  fuseV : ∀ {Θ₁ Θ₂ Θ₃ A} {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁ Θ₃} →
    Composite refl ρ₁ ρ₂ ρ₁₂ → (t : Val Θ₁ A) → renV ρ₂ (renV ρ₁ t) ≡ renV ρ₁₂ t
  fuseVs : ∀ {Θ₁ Θ₂ Θ₃ bs} {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁ Θ₃} →
    Composite refl ρ₁ ρ₂ ρ₁₂ → (t : Vals Θ₁ bs) → renVs ρ₂ (renVs ρ₁ t) ≡ renVs ρ₁₂ t
  fuseC : ∀ {Θ₁ Θ₂ Θ₃ B} {ρ₁ : Ren Θ₁ Θ₂} {ρ₂ : Ren Θ₂ Θ₃} {ρ₁₂ : Ren Θ₁ Θ₃} →
    Composite refl ρ₁ ρ₂ ρ₁₂ → (t : Comp Θ₁ B) → renC ρ₂ (renC ρ₁ t) ≡ renC ρ₁₂ t

  fuseV c (ne n) i = ne (c .on-ne i n)
  fuseV c tt = refl
  fuseV c (inl V) = cong inl (fuseV c V)
  fuseV c (inr V) = cong inr (fuseV c V)
  fuseV c (pair V V') = cong₂ pair (fuseV c V) (fuseV c V')
  fuseV c (thunk M) = cong thunk (fuseC c M)
  fuseV {ρ₁ = ρ₁} {ρ₂} {ρ₁₂} c (const x) =
    cong const (cmapW-seq (wm ρ₁) (wm ρ₂) (wm ρ₁₂) (c .on-world) x)
  fuseV c (shut V) = cong shut (funExt λ q →
    ren-fuseV (Composite-keepL c q) (V _) (λ i → V (Composite-pt c q i .fst)))

  fuseVs c []ᵥ = refl
  fuseVs c (V ∷ᵥ vs) = cong₂ _∷ᵥ_ (fuseV c V) (fuseVs c vs)

  fuseC c (return V) = cong return (fuseV c V)
  fuseC c (M to N) = cong₂ _to_ (fuseC c M) (fuseC (Composite-keep c) N)
  fuseC c (force V) = cong force (fuseV c V)
  fuseC c (case V M) = cong₂ case (fuseV c V) (fuseC (Composite-keep (Composite-keep c)) M)
  fuseC c (case₊ V M N) =
    cong₃ case₊ (fuseV c V) (fuseC (Composite-keep c) M) (fuseC (Composite-keep c) N)
  fuseC c (absurd V) = cong absurd (fuseV c V)
  fuseC c (lam M) = cong lam (fuseC (Composite-keep c) M)
  fuseC c (app M V) = cong₂ app (fuseC c M) (fuseV c V)
  fuseC c (op o vs k) = cong₂ (op o) (fuseVs c vs) (funExt λ a → funExt λ ps →
    ren-fuseC (Composite-keepW (slot o a) c ps) (k a _) (λ i → k a (Composite-pts (slot o a) c ps i)))

ren-fuseV {Θ₁} {Θ₃ = Θ₃} {A = A} {ρ₁ = ρ₁} {ρ₂} c t tp =
  J-tele (λ Θ → Val Θ A)
    (λ Θ' e' t' → (ρ : Ren Θ' Θ₃) → Composite e' ρ₁ ρ₂ ρ → renV ρ₂ (renV ρ₁ t) ≡ renV ρ t')
    (λ ρ c' → fuseV c' t) _ tp _ c
ren-fuseVs {Θ₁} {Θ₃ = Θ₃} {bs = bs} {ρ₁ = ρ₁} {ρ₂} c t tp =
  J-tele (λ Θ → Vals Θ bs)
    (λ Θ' e' t' → (ρ : Ren Θ' Θ₃) → Composite e' ρ₁ ρ₂ ρ → renVs ρ₂ (renVs ρ₁ t) ≡ renVs ρ t')
    (λ ρ c' → fuseVs c' t) _ tp _ c
ren-fuseC {Θ₁} {Θ₃ = Θ₃} {B = B} {ρ₁ = ρ₁} {ρ₂} c t tp =
  J-tele (λ Θ → Comp Θ B)
    (λ Θ' e' t' → (ρ : Ren Θ' Θ₃) → Composite e' ρ₁ ρ₂ ρ → renC ρ₂ (renC ρ₁ t) ≡ renC ρ t')
    (λ ρ c' → fuseC c' t) _ tp _ c

-- Thm 8.2: extensionality (fusion with the identity, then ren-id)

ren-extV : ∀ {Θ₁ Θ₂ Θ' A} {e : Θ₁ ≡ Θ₂} {ρ₁ : Ren Θ₁ Θ'} {ρ₂ : Ren Θ₂ Θ'} → Agree e ρ₁ ρ₂ →
  (t₁ : Val Θ₁ A) {t₂ : Val Θ₂ A} → PathP (λ i → Val (e i) A) t₁ t₂ → renV ρ₁ t₁ ≡ renV ρ₂ t₂
ren-extV {ρ₁ = ρ₁} ag t₁ tp = sym (ren-idV (renV ρ₁ t₁)) ∙ ren-fuseV (Agree→Composite ag) t₁ tp

ren-extVs : ∀ {Θ₁ Θ₂ Θ' bs} {e : Θ₁ ≡ Θ₂} {ρ₁ : Ren Θ₁ Θ'} {ρ₂ : Ren Θ₂ Θ'} → Agree e ρ₁ ρ₂ →
  (t₁ : Vals Θ₁ bs) {t₂ : Vals Θ₂ bs} → PathP (λ i → Vals (e i) bs) t₁ t₂ → renVs ρ₁ t₁ ≡ renVs ρ₂ t₂
ren-extVs {ρ₁ = ρ₁} ag t₁ tp = sym (ren-idVs (renVs ρ₁ t₁)) ∙ ren-fuseVs (Agree→Composite ag) t₁ tp

ren-extC : ∀ {Θ₁ Θ₂ Θ' B} {e : Θ₁ ≡ Θ₂} {ρ₁ : Ren Θ₁ Θ'} {ρ₂ : Ren Θ₂ Θ'} → Agree e ρ₁ ρ₂ →
  (t₁ : Comp Θ₁ B) {t₂ : Comp Θ₂ B} → PathP (λ i → Comp (e i) B) t₁ t₂ → renC ρ₁ t₁ ≡ renC ρ₂ t₂
ren-extC {ρ₁ = ρ₁} ag t₁ tp = sym (ren-idC (renC ρ₁ t₁)) ∙ ren-fuseC (Agree→Composite ag) t₁ tp
