module Polynomial.Presheaf.FinBijective.LocalState.Config
  {v : Level} (V : hSet v) where

open Category
open Functor
open Iso

------------------------------------------------------------------------
-- Configurations
------------------------------------------------------------------------

Store : ℕ → Type v
Store n = Fin n → V .fst

isSetStore : (n : ℕ) → isSet (Store n)
isSetStore n = isSet→ (V .snd)

------------------------------------------------------------------------
-- Reassociation
------------------------------------------------------------------------

assocFinIso : ∀ n k l → Iso (Fin ((n + k) + l)) (Fin (n + (k + l)))
assocFinIso n k l =
  compIso
    (invIso (equivToIso (FinSumChar.Equiv (n + k) l)))
    (compIso
      (⊎Iso (invIso (equivToIso (FinSumChar.Equiv n k))) idIso)
      (compIso
        ⊎-assoc-Iso
        (compIso
          (⊎Iso idIso (equivToIso (FinSumChar.Equiv k l)))
          (equivToIso (FinSumChar.Equiv n (k + l))))))

assocStore : ∀ {n k l} →
  Store ((n + k) + l) → Store (n + (k + l))
assocStore {n} {k} {l} τ i = τ (assocFinIso n k l .inv i)

assocInjection : ∀ n k l → Injection ((n + k) + l) (n + (k + l))
assocInjection n k l =
  Equiv→Embedding (isoToEquiv (assocFinIso n k l))

-- Regard a store on n as a store on the canonically equivalent n + 0.
extendStore0 : ∀ {n} → Store n → Store (n + 0)
extendStore0 {n} σ i =
  Sum.rec σ (λ j → ⊥.rec (¬Fin0 j)) (FinSumChar.inv n 0 i)

------------------------------------------------------------------------
-- Configurations as the category of elements of the store presheaf
------------------------------------------------------------------------

-- Worlds are finite sets of allocated cells and structural maps are
-- injections.  Stores pull back by restriction along an injection.
StoreP : Presheaf Inj v
StoreP .F-ob n = Store n , isSetStore n
StoreP .F-hom ρ τ i = τ (ρ .fst i)
StoreP .F-id = refl
StoreP .F-seq _ _ = refl

-- Hence a configuration morphism (n,σ) → (m,τ) is precisely an
-- injection ρ : n ↪ m whose restriction of τ is σ.
Config : Category v v
Config = Contravariant.∫ StoreP

-- The forgetful functor exhibits Config as a total category over Inj.
Config→Inj : Functor Config Inj
Config→Inj = TotalCategory.Fst

configHom≡ : ∀ {q q'} {f g : Config [ q , q' ]} →
  f .fst ≡ g .fst → f ≡ g
configHom≡ {q} {q'} =
  Contravariant.ElementHom≡ StoreP {o1 = q} {o2 = q'}

------------------------------------------------------------------------
-- Canonical configuration isomorphisms
------------------------------------------------------------------------

-- The canonical configuration map from left- to right-associated worlds.
assocConfigHom : ∀ {n k l} (τ : Store ((n + k) + l)) →
  Config
    [ ((n + k) + l , τ)
    , (n + (k + l) , assocStore {n = n} {k = k} {l = l} τ)
    ]
assocConfigHom {n} {k} {l} τ =
  assocInjection n k l ,
  funExt (λ i → cong τ (assocFinIso n k l .ret i))

-- The canonical configuration map n → n + 0.
extendConfig0 : ∀ {n} (σ : Store n) →
  Config [ (n , σ) , (n + 0 , extendStore0 σ) ]
extendConfig0 {n} σ =
  extendInjection , funExt λ i →
    cong (extendStore0 σ) refl
    ∙ cong
        (Sum.rec σ (λ j → ⊥.rec (¬Fin0 j)))
        (FinSumChar.ret n 0 (inl i))

ConfigPsh : Category _ _
ConfigPsh = PresheafCategory (Config ^op) v
