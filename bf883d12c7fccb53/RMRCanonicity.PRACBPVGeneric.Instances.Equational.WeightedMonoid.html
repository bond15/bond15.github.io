-- The weighted monoid as Q = T with its equations (POPL's
-- Instances/WeightedMonoid and WeightedMonoidRules; paper §2.1 "Theory of
-- Weighted Monoids", §2.2 its free model, §5.3 the weighted-monoid
-- example).  Its AFFINITY (POPL's WeightedMonoidAffine) is deferred by user
-- decision, so neither SN, WN nor ground canonicity is stated here: they
-- need it (Semantics.Normalization).  Machine soundness and the uniqueness
-- of final configurations (final-unique) need no affinity and are here.
--
-- W = 1, no locks.  Over a monoid of weights (R, ·, 1r) (a set): a nullary
-- unit e, a binary mul ⊗ (its slots true, false: left, right) and a unary
-- act_r for each r : R, with
--
--   x ⊗ (y ⊗ z) = (x ⊗ y) ⊗ z     e ⊗ x = x     x ⊗ e = x
--   act_1 x = x     act_r (act_r' x) = act_{r · r'} x
--   act_r (x ⊗ y) = act_r x ⊗ act_r y     act_r e = e
--
-- Free model: weighted lists List (R × X), e = [], ⊗ = append,
-- act_r = scale the weights by r, η x = [(1 , x)].  As a container
-- (POPL's): shapes Σ n. (Fin n → R), positions Fin n, so a configuration
-- is a list of weighted threads; the operations are computed on lists
-- through fromL / toL.  ρ = μ (Instances.Equational.Common,
-- Semantics.PolySelfModule).
--
--   wmEqs, wmFree, freeModel, Q′, wm-module
--   wm-act, wm-unit, wm-mul
--                     POPL's derived rules, at the first thread:
--                       ⟨(r , act_r' M) ∣ rest⟩ ⟶ ⟨(r · r' , M) ∣ rest⟩
--                       ⟨(r , e) ∣ rest⟩       ⟶ ⟨rest⟩
--                       ⟨(r , M ⊗ M') ∣ rest⟩  ⟶ ⟨(r , M) ∣ (r , M') ∣ rest⟩
--   wm-run            act_r (return tt ⊗ return tt) reaches two weighted
--                     returns; wm-run-sound, wm-run-unique (final-unique),
--                     wm-run-comparison (the free-model denotation)

data WOp : Type where
  unit mul : WOp
  act : R → WOp

WAr : WOp → Type
WAr unit = ⊥
WAr mul = Bool
WAr (act _) = Unit

private
  enc : WOp → (Unit ⊎ Unit) ⊎ R
  enc unit = inl (inl tt)
  enc mul = inl (inr tt)
  enc (act r) = inr r

  dec : (Unit ⊎ Unit) ⊎ R → WOp
  dec (inl (inl _)) = unit
  dec (inl (inr _)) = mul
  dec (inr r) = act r

  dec-enc : ∀ o → dec (enc o) ≡ o
  dec-enc unit = refl
  dec-enc mul = refl
  dec-enc (act r) = refl

isSetWOp : isSet WOp
isSetWOp = isSetRetract enc dec dec-enc (isSet⊎ (isSet⊎ isSetUnit isSetUnit) is-set)

isSetWAr : ∀ o → isSet (WAr o)
isSetWAr unit = isProp→isSet isProp⊥
isSetWAr mul = isSetBool
isSetWAr (act _) = isSetUnit

-- a constant presheaf of variables
K : (n : ℕ) → 𝒱
K n .F-ob _ = Fin n , isSetFin
K n .F-hom _ x = x
K n .F-id = refl
K n .F-seq _ _ = refl

eT : ∀ {X w} → Term (PM.polynomial L₁ O₁) X w
eT = ops (unit , tt) λ q → ⊥.rec (q .fst)

_⊗T_ : ∀ {X w} → Term (PM.polynomial L₁ O₁) X w → Term (PM.polynomial L₁ O₁) X w → Term (PM.polynomial L₁ O₁) X w
a ⊗T b = ops (mul , tt) λ q → pick (q .fst) a b

actT : ∀ {X w} → R → Term (PM.polynomial L₁ O₁) X w → Term (PM.polynomial L₁ O₁) X w
actT r a = ops (act r , tt) λ _ → a

data WEq : Type where
  e-assoc e-unitl e-unitr e-act1 : WEq
  e-actact : R → R → WEq
  e-actdist e-acte : R → WEq

private
  x₀ x₁ x₂ : ∀ {w} → Term (PM.polynomial L₁ O₁) (K 3) w
  x₀ = var zero
  x₁ = var (suc zero)
  x₂ = var (suc (suc zero))

wmEqs : Machine.Equations L₁ O₁
wmEqs .Machine.Equations.Equation = WEq
wmEqs .Machine.Equations.equationWorld _ = tt*
wmEqs .Machine.Equations.Variables _ = K 3
wmEqs .Machine.Equations.lhs e-assoc = x₀ ⊗T (x₁ ⊗T x₂)
wmEqs .Machine.Equations.rhs e-assoc = (x₀ ⊗T x₁) ⊗T x₂
wmEqs .Machine.Equations.lhs e-unitl = eT ⊗T x₀
wmEqs .Machine.Equations.rhs e-unitl = x₀
wmEqs .Machine.Equations.lhs e-unitr = x₀ ⊗T eT
wmEqs .Machine.Equations.rhs e-unitr = x₀
wmEqs .Machine.Equations.lhs e-act1 = actT 1r x₀
wmEqs .Machine.Equations.rhs e-act1 = x₀
wmEqs .Machine.Equations.lhs (e-actact r r') = actT r (actT r' x₀)
wmEqs .Machine.Equations.rhs (e-actact r r') = actT (r · r') x₀
wmEqs .Machine.Equations.lhs (e-actdist r) = actT r (x₀ ⊗T x₁)
wmEqs .Machine.Equations.rhs (e-actdist r) = actT r x₀ ⊗T actT r x₁
wmEqs .Machine.Equations.lhs (e-acte r) = actT r eT
wmEqs .Machine.Equations.rhs (e-acte r) = eT

-- weighted lists, and the container presentation Σ n. (Fin n → R)

L : Type → Type
L X = List (R × X)

scale : ∀ {X : Type} → R → R × X → R × X
scale r (q , y) = r · q , y

opsL : ∀ {X : Type} (o : WOp) → (WAr o → L X) → L X
opsL unit a = []
opsL mul a = a true ++ a false
opsL (act r) a = map (scale r) (a tt)

-- POPL's ⟦ wmPoly ⟧
Tw : Type → Type
Tw Z = Σ[ s ∈ Σ ℕ (λ n → Fin n → R) ] (Fin (s .fst) → Z)

cons : ∀ {X : Type} {n} → X → (Fin n → X) → Fin (suc n) → X
cons x f zero = x
cons x f (suc i) = f i

cons-η : ∀ {X : Type} {n} (f : Fin (suc n) → X) → cons (f zero) (λ i → f (suc i)) ≡ f
cons-η f = funExt λ { zero → refl ; (suc i) → refl }

none : ∀ {X : Type} → Fin 0 → X
none ()

-- the map on threads (Common's pmap)
twmap : ∀ {X Y : Type} → (X → Y) → Tw X → Tw Y
twmap f t = t .fst , λ r → f (t .snd r)

-- The bijection between weighted lists and the container is OPAQUE:
-- unfolding it inside comparisons copies the list three times per layer
-- (length, weights, threads), and Agda's conversion then takes minutes and
-- gigabytes.  Everything about it is stated as the lemmas below.
opaque
  toL' : ∀ {X : Type} (n : ℕ) → (Fin n → R) → (Fin n → X) → L X
  toL' zero w γ = []
  toL' (suc n) w γ = (w zero , γ zero) ∷ toL' n (λ i → w (suc i)) (λ i → γ (suc i))

  toL : ∀ {X : Type} → Tw X → L X
  toL t = toL' (t .fst .fst) (t .fst .snd) (t .snd)

  fromL : ∀ {X : Type} → L X → Tw X
  fromL [] = (0 , none) , none
  fromL ((r , y) ∷ l) = (suc (fromL l .fst .fst) , cons r (fromL l .fst .snd)) , cons y (fromL l .snd)

  toL-fromL : ∀ {X : Type} (l : L X) → toL (fromL l) ≡ l
  toL-fromL [] = refl
  toL-fromL ((r , y) ∷ l) = cong ((r , y) ∷_) (toL-fromL l)

  fromL-toL' : ∀ {X : Type} (n : ℕ) (w : Fin n → R) (γ : Fin n → X) → fromL (toL' n w γ) ≡ ((n , w) , γ)
  fromL-toL' zero w γ = ΣPathP (ΣPathP (refl , funExt λ ()) , funExt λ ())
  fromL-toL' {X} (suc n) w γ =
    cong g (fromL-toL' n (λ i → w (suc i)) (λ i → γ (suc i))) ∙ ΣPathP (ΣPathP (refl , cons-η w) , cons-η γ)
    where
    g : Tw X → Tw X
    g u = (suc (u .fst .fst) , cons (w zero) (u .fst .snd)) , cons (γ zero) (u .snd)

  fromL-toL : ∀ {X : Type} (t : Tw X) → fromL (toL t) ≡ t
  fromL-toL t = fromL-toL' (t .fst .fst) (t .fst .snd) (t .snd)

  -- reading the first thread off
  toL-suc : ∀ {X : Type} (n : ℕ) (w : Fin (suc n) → R) (g : Fin (suc n) → X) →
    toL ((suc n , w) , g) ≡ (w zero , g zero) ∷ toL ((n , λ i → w (suc i)) , λ i → g (suc i))
  toL-suc n w g = refl

  toL-zero : ∀ {X : Type} (w : Fin 0 → R) (g : Fin 0 → X) → toL ((0 , w) , g) ≡ []
  toL-zero w g = refl

  -- one thread
  toL-one : ∀ {X : Type} (r : R) (y : X) → toL ((1 , λ _ → r) , λ _ → y) ≡ (r , y) ∷ []
  toL-one r y = refl

  fromL-∷ : ∀ {X : Type} (r : R) (y : X) (t : Tw X) →
    fromL ((r , y) ∷ toL t) ≡ ((suc (t .fst .fst) , cons r (t .fst .snd)) , cons y (t .snd))
  fromL-∷ {X} r y t = cong g (fromL-toL t)
    where
    g : Tw X → Tw X
    g u = (suc (u .fst .fst) , cons r (u .fst .snd)) , cons y (u .snd)

-- the operations on Tw, through lists
opW : ∀ {Z} (o : WOp) → (WAr o → Tw Z) → Tw Z
opW o k = fromL (opsL o λ a → toL (k a))

------------------------------------------------------------------------
-- list lemmas (POPL's)

private
  ++-assoc' : {X : Type} (a b c : List X) → (a ++ b) ++ c ≡ a ++ (b ++ c)
  ++-assoc' [] b c = refl
  ++-assoc' (y ∷ a) b c = cong (y ∷_) (++-assoc' a b c)

  ++-unit' : {X : Type} (a : List X) → a ++ [] ≡ a
  ++-unit' [] = refl
  ++-unit' (y ∷ a) = cong (y ∷_) (++-unit' a)

  map-++' : {X Y : Type} (f : X → Y) (a b : List X) → map f (a ++ b) ≡ map f a ++ map f b
  map-++' f [] b = refl
  map-++' f (y ∷ a) b = cong (f y ∷_) (map-++' f a b)

  map-scale1 : {X : Type} (a : L X) → map (scale 1r) a ≡ a
  map-scale1 [] = refl
  map-scale1 ((q , y) ∷ a) = cong₂ _∷_ (ΣPathP (·IdL q , refl)) (map-scale1 a)

  map-scale² : {X : Type} (r r' : R) (a : L X) → map (scale r) (map (scale r') a) ≡ map (scale (r · r')) a
  map-scale² r r' [] = refl
  map-scale² r r' ((q , y) ∷ a) = cong₂ _∷_ (ΣPathP (·Assoc r r' q , refl)) (map-scale² r r' a)

  second : {X Y : Type} → (X → Y) → R × X → R × Y
  second f (q , y) = q , f y

  map-second-scale : {X Y : Type} (f : X → Y) (r : R) (a : L X) →
    map (second f) (map (scale r) a) ≡ map (scale r) (map (second f) a)
  map-second-scale f r [] = refl
  map-second-scale f r ((q , y) ∷ a) = cong (_ ∷_) (map-second-scale f r a)

  opsL-nat : {X Y : Type} (f : X → Y) (o : WOp) (a : WAr o → L X) →
    map (second f) (opsL o a) ≡ opsL o (λ j → map (second f) (a j))
  opsL-nat f unit a = refl
  opsL-nat f mul a = map-++' (second f) (a true) (a false)
  opsL-nat f (act r) a = map-second-scale f r (a tt)

opaque
  unfolding toL' toL fromL

  pmap-fromL : {X Y : Type} (f : X → Y) (l : L X) → twmap f (fromL l) ≡ fromL (map (second f) l)
  pmap-fromL f [] = ΣPathP (refl , funExt λ ())
  pmap-fromL {Y = Y} f ((r , y) ∷ l) =
    ΣPathP (refl , funExt λ { zero → refl ; (suc j) → refl }) ∙ cong g (pmap-fromL f l)
    where
    g : Tw Y → Tw Y
    g u = (suc (u .fst .fst) , cons r (u .fst .snd)) , cons (f y) (u .snd)

  toL-pmap : {X Y : Type} (f : X → Y) (t : Tw X) → toL (twmap f t) ≡ map (second f) (toL t)
  toL-pmap f t = go (t .fst .fst) (t .fst .snd) (t .snd)
    where
    go : (n : ℕ) (w : Fin n → R) (γ : Fin n → _) → toL' n w (λ i → f (γ i)) ≡ map (second f) (toL' n w γ)
    go zero w γ = refl
    go (suc n) w γ = cong (_ ∷_) (go n (λ i → w (suc i)) (λ i → γ (suc i)))

opW-nat : ∀ {Z Z'} (g : Z → Z') (o : WOp) (k : WAr o → Tw Z) → pmap g (opW o k) ≡ opW o (λ a → pmap g (k a))
opW-nat g o k =
  pmap-fromL g _
  ∙ cong fromL (opsL-nat g o _ ∙ cong (opsL o) (funExt λ j → sym (toL-pmap g (k j))))

three : ∀ {X : Type} → X → X → X → Fin 3 → X
three a b c zero = a
three a b c (suc zero) = b
three a b c (suc (suc zero)) = c

-- the laws in any model
module NLaws (N : Model) where
  Nunit : ∣ N ∣
  Nunit = Nalg N unit ⊥.rec

  Nmul : ∣ N ∣ → ∣ N ∣ → ∣ N ∣
  Nmul a b = Nalg N mul λ j → pick j a b

  Nact : R → ∣ N ∣ → ∣ N ∣
  Nact r a = Nalg N (act r) λ _ → a

  -- mul on a family is mul on its two members
  split : (h : Bool → ∣ N ∣) → Nalg N mul h ≡ Nmul (h true) (h false)
  split h = cong (Nalg N mul) (funExt λ { true → refl ; false → refl })

  private
    ev : (Fin 3 → ∣ N ∣) → Term (PM.polynomial L₁ O₁) (K 3) tt* → ∣ N ∣
    ev γ t = evaluate (N .fst) (naturally _ _ λ _ → γ) .NatTrans.N-ob tt* t

    law : ∀ e (γ : Fin 3 → ∣ N ∣) → ev γ (Machine.Equations.lhs wmEqs e) ≡ ev γ (Machine.Equations.rhs wmEqs e)
    law e γ = N .snd e (naturally _ _ λ _ → γ)

    unitE : (γ : Fin 3 → ∣ N ∣) → Nunit ≡ ev γ eT
    unitE γ = cong (Nalg N unit) (funExt λ ())

  N-assoc : ∀ a b c → Nmul a (Nmul b c) ≡ Nmul (Nmul a b) c
  N-assoc a b c =
    cong (Nmul a) (sym (split _)) ∙ sym (split _) ∙ law e-assoc (three a b c) ∙ split _
    ∙ cong (λ z → Nmul z c) (split _)

  N-unitl : ∀ a → Nmul Nunit a ≡ a
  N-unitl a = cong (λ z → Nmul z a) (unitE (λ _ → a)) ∙ sym (split _) ∙ law e-unitl (λ _ → a)

  N-unitr : ∀ a → Nmul a Nunit ≡ a
  N-unitr a = cong (Nmul a) (unitE (λ _ → a)) ∙ sym (split _) ∙ law e-unitr (λ _ → a)

  N-act1 : ∀ a → Nact 1r a ≡ a
  N-act1 a = law e-act1 (λ _ → a)

  N-actact : ∀ r r' a → Nact r (Nact r' a) ≡ Nact (r · r') a
  N-actact r r' a = law (e-actact r r') (λ _ → a)

  N-actdist : ∀ r a b → Nact r (Nmul a b) ≡ Nmul (Nact r a) (Nact r b)
  N-actdist r a b = cong (Nact r) (sym (split _)) ∙ law (e-actdist r) (three a b a) ∙ split _

  N-acte : ∀ r → Nact r Nunit ≡ Nunit
  N-acte r = cong (Nact r) (unitE (λ _ → Nunit)) ∙ law (e-acte r) (λ _ → Nunit) ∙ sym (unitE (λ _ → Nunit))

  -- POPL's !_f on lists
  extL : ∀ {Z : Type} → (Z → ∣ N ∣) → L Z → ∣ N ∣
  extL f [] = Nunit
  extL f ((r , y) ∷ l) = Nmul (Nact r (f y)) (extL f l)

  module _ {Z : Type} (f : Z → ∣ N ∣) where
    extL-++ : (a b : L Z) → extL f (a ++ b) ≡ Nmul (extL f a) (extL f b)
    extL-++ [] b = sym (N-unitl _)
    extL-++ ((r , y) ∷ a) b = cong (Nmul _) (extL-++ a b) ∙ N-assoc _ _ _

    extL-map : (r : R) (a : L Z) → extL f (map (scale r) a) ≡ Nact r (extL f a)
    extL-map r [] = sym (N-acte r)
    extL-map r ((q , y) ∷ a) =
      cong₂ Nmul (sym (N-actact r q (f y))) (extL-map r a) ∙ sym (N-actdist r _ _)

-- the three variables of an environment
module _ (X : 𝒱) (env : NatTrans (K 3) (Carrier X)) where
  γ0 γ1 γ2 : Ext (X .F-ob tt* .fst)
  γ0 = env .NatTrans.N-ob tt* zero
  γ1 = env .NatTrans.N-ob tt* (suc zero)
  γ2 = env .NatTrans.N-ob tt* (suc (suc zero))

-- POPL's wmFree: the extension of f is extL f, read on lists
wmFree : PolyFree
-- (equation by equation: evaluation unfolds on these terms; the
-- satisfaction is the list algebra's, read through fromL / toL)
wmFree .PolyFree.sat X e-assoc env =
  cong fromL (cong (toL (γ0 X env) ++_) (toL-fromL (toL (γ1 X env) ++ toL (γ2 X env)))
              ∙ sym (++-assoc' (toL (γ0 X env)) (toL (γ1 X env)) (toL (γ2 X env)))
              ∙ cong (_++ toL (γ2 X env)) (sym (toL-fromL (toL (γ0 X env) ++ toL (γ1 X env)))))
wmFree .PolyFree.sat X e-unitl env =
  cong fromL (cong (_++ toL (γ0 X env)) (toL-fromL [])) ∙ fromL-toL (γ0 X env)
wmFree .PolyFree.sat X e-unitr env =
  cong fromL (cong (toL (γ0 X env) ++_) (toL-fromL []) ∙ ++-unit' (toL (γ0 X env))) ∙ fromL-toL (γ0 X env)
wmFree .PolyFree.sat X e-act1 env = cong fromL (map-scale1 (toL (γ0 X env))) ∙ fromL-toL (γ0 X env)
wmFree .PolyFree.sat X (e-actact r r') env =
  cong fromL (cong (map (scale r)) (toL-fromL (map (scale r') (toL (γ0 X env))))
              ∙ map-scale² r r' (toL (γ0 X env)))
wmFree .PolyFree.sat X (e-actdist r) env =
  cong fromL (cong (map (scale r)) (toL-fromL (toL (γ0 X env) ++ toL (γ1 X env)))
              ∙ map-++' (scale r) (toL (γ0 X env)) (toL (γ1 X env))
              ∙ cong₂ _++_ (sym (toL-fromL (map (scale r) (toL (γ0 X env)))))
                           (sym (toL-fromL (map (scale r) (toL (γ1 X env))))))
wmFree .PolyFree.sat X (e-acte r) env = cong fromL (cong (map (scale r)) (toL-fromL []))
wmFree .PolyFree.ext N f t = NLaws.extL N f (toL t)
wmFree .PolyFree.ext-η N f z =
  cong (NLaws.extL N f) (toL-one 1r z) ∙ NLaws.N-unitr N _ ∙ NLaws.N-act1 N _
wmFree .PolyFree.ext-hom N f unit k =
  cong (NLaws.extL N f) (toL-fromL []) ∙ cong (Nalg N unit) (funExt λ ())
wmFree .PolyFree.ext-hom N f mul k =
  cong (NLaws.extL N f) (toL-fromL (toL (k true) ++ toL (k false)))
  ∙ NLaws.extL-++ N f (toL (k true)) (toL (k false))
  ∙ sym (NLaws.split N _)
wmFree .PolyFree.ext-hom N f (act r) k =
  cong (NLaws.extL N f) (toL-fromL (map (scale r) (toL (k tt)))) ∙ NLaws.extL-map N f r (toL (k tt))
wmFree .PolyFree.ext-unique N {Z} f h hh hη t = cong h (sym (fromL-toL t)) ∙ onL (toL t)
  where
  open NLaws N

  onL : (l : L Z) → h (fromL l) ≡ extL f l
  onL [] = hh unit ⊥.rec ∙ cong (Nalg N unit) (funExt λ ())
  onL ((r , y) ∷ l) =
    cong h (cong fromL (cong₂ _∷_ (ΣPathP (sym (·IdR r) , refl)) (sym (toL-fromL l))
              ∙ cong (λ a → map (scale r) a ++ toL (fromL l)) (sym (toL-one 1r y))
              ∙ cong (_++ toL (fromL l)) (sym (toL-fromL (map (scale r) (toL ((1 , λ _ → 1r) , λ _ → y)))))))
    ∙ hh mul (λ j → pick j (opW (act r) λ _ → (1 , λ _ → 1r) , λ _ → y) (fromL l))
    ∙ split _
    ∙ cong₂ Nmul (hh (act r) (λ _ → (1 , λ _ → 1r) , λ _ → y) ∙ cong (Nact r) (hη y)) (onL l)

-- the machine (no affinity)

module _ {A : ValTy} where
  open PolyMachine L₁ O₁ isSetLk isSetBase wmEqs freeModel Q′ tt* presentation A

  -- ρ = μ
  wm-module : RightModule (PS.ER.T) (PS.ER.Q)
  wm-module = PS.polyModule presentation

  -- the operations of the free model on closed computations
  -- (written free ⟅ CompPsh (F A) ⟆, never through an alias)
  open NLaws (free ⟅ CompPsh (F A) ⟆) using () renaming (Nmul to Nmul′; Nact to Nact′; extL to extL′)

  private
    -- μ at a configuration of units: the threads with their weights
    -- (POPL's rest)
    rest : (n : ℕ) (ws : Fin n → R) (γ : Fin n → Comp (∅ tt*) (F A)) →
      toL (extL′ (λ t → t) (toL ((n , ws) , λ i → (1 , λ _ → 1r) , λ _ → γ i))) ≡ toL ((n , ws) , γ)
    rest zero ws γ =
      cong (λ l → toL (extL′ (λ t → t) l)) (toL-zero ws _) ∙ toL-fromL [] ∙ sym (toL-zero ws γ)
    rest (suc n) ws γ =
      cong (λ l → toL (extL′ (λ t → t) l)) (toL-suc n ws _)
      ∙ toL-fromL (toL (Nact′ (ws zero) ((1 , λ _ → 1r) , λ _ → γ zero))
                   ++ toL (extL′ (λ t → t) (toL ((n , λ i → ws (suc i)) , λ i → (1 , λ _ → 1r) , λ _ → γ (suc i)))))
      ∙ cong₂ _++_
          (toL-fromL (map (scale (ws zero)) (toL ((1 , λ _ → 1r) , λ _ → γ zero)))
           ∙ cong (map (scale (ws zero))) (toL-one 1r (γ zero))
           ∙ cong (_∷ []) (ΣPathP (·IdR (ws zero) , refl)))
          (rest n (λ i → ws (suc i)) (λ i → γ (suc i)))
      ∙ sym (toL-suc n ws γ)

  -- the effect step at the first thread of a configuration: μ puts the
  -- operation's weighted list, scaled by the thread's weight, in front of
  -- the other threads
  first-step : (r : R) (o : WOp) (k : WAr o → Unit → Comp (∅ tt*) (F A))
    (n : ℕ) (ws : Fin n → R) (γ : Fin n → Comp (∅ tt*) (F A)) →
    ((suc n , cons r ws) , cons (op o []ᵥ k) γ) ⟶
      fromL (map (scale r) (toL (opW o λ a → (1 , λ _ → 1r) , λ _ → k a tt)) ++ toL ((n , ws) , γ))
  first-step r o k n ws γ =
    subst (((suc n , cons r ws) , cons (op o []ᵥ k) γ) ⟶_)
      (cong out gpath
       ∙ cong (λ l → extL′ (λ t → t) l) (toL-suc n (cons r ws) _)
       ∙ cong fromL (cong₂ _++_ (toL-fromL (map (scale r) (toL (opW o λ a → (1 , λ _ → 1r) , λ _ → k a tt))))
                                (rest n ws γ)))
      (effect-in x zero refl)
    where
    x : Config
    x = (suc n , cons r ws) , cons (op o []ᵥ k) γ

    ok = (o , tt) , λ q → k (q .fst) (q .snd)

    -- the operation at the first thread, the units of the others
    gpath : spliced x zero ok .snd
      ≡ cons (opW o λ a → (1 , λ _ → 1r) , λ _ → k a tt) (λ i → (1 , λ _ → 1r) , λ _ → γ i)
    gpath = funExt λ { zero → spliced-hole x zero ok ; (suc i) → spliced-other x zero ok (suc i) snotz }

    out : (Fin (suc n) → Ext (Comp (∅ tt*) (F A))) → Ext (Comp (∅ tt*) (F A))
    out g = extL′ (λ t → t) (toL ((suc n , cons r ws) , g))

  -- POPL's derived rules, at the first thread
  --   ⟨(r , act_r' M) ∣ rest⟩ ⟶ ⟨(r · r' , M) ∣ rest⟩
  wm-act : (r r' : R) (k : Unit → Unit → Comp (∅ tt*) (F A))
    (n : ℕ) (ws : Fin n → R) (γ : Fin n → Comp (∅ tt*) (F A)) →
    ((suc n , cons r ws) , cons (op (act r') []ᵥ k) γ) ⟶ ((suc n , cons (r · r') ws) , cons (k tt tt) γ)
  wm-act r r' k n ws γ = subst (((suc n , cons r ws) , cons (op (act r') []ᵥ k) γ) ⟶_)
    (cong (λ l → fromL (map (scale r) l ++ toL ((n , ws) , γ)))
       (toL-fromL (map (scale r') (toL ((1 , λ _ → 1r) , λ _ → k tt tt))) ∙ cong (map (scale r')) (toL-one 1r (k tt tt)))
     ∙ fromL-∷ (r · (r' · 1r)) (k tt tt) ((n , ws) , γ)
     ∙ cong {B = λ _ → Config} (λ q → (suc n , cons q ws) , cons (k tt tt) γ) (cong (r ·_) (·IdR r')))
    (first-step r (act r') k n ws γ)

  --   ⟨(r , e) ∣ rest⟩ ⟶ ⟨rest⟩
  wm-unit : (r : R) (k : ⊥ → Unit → Comp (∅ tt*) (F A))
    (n : ℕ) (ws : Fin n → R) (γ : Fin n → Comp (∅ tt*) (F A)) →
    ((suc n , cons r ws) , cons (op unit []ᵥ k) γ) ⟶ ((n , ws) , γ)
  wm-unit r k n ws γ = subst (((suc n , cons r ws) , cons (op unit []ᵥ k) γ) ⟶_)
    (cong (λ l → fromL (map (scale r) l ++ toL ((n , ws) , γ))) (toL-fromL [])
     ∙ fromL-toL ((n , ws) , γ))
    (first-step r unit k n ws γ)

  --   ⟨(r , M ⊗ M') ∣ rest⟩ ⟶ ⟨(r , M) ∣ (r , M') ∣ rest⟩
  wm-mul : (r : R) (k : Bool → Unit → Comp (∅ tt*) (F A))
    (n : ℕ) (ws : Fin n → R) (γ : Fin n → Comp (∅ tt*) (F A)) →
    ((suc n , cons r ws) , cons (op mul []ᵥ k) γ) ⟶
      ((suc (suc n) , cons r (cons r ws)) , cons (k true tt) (cons (k false tt) γ))
  wm-mul r k n ws γ = subst (((suc n , cons r ws) , cons (op mul []ᵥ k) γ) ⟶_)
    (cong (λ l → fromL (map (scale r) l ++ toL ((n , ws) , γ)))
       (toL-fromL (toL ((1 , λ _ → 1r) , λ _ → k true tt) ++ toL ((1 , λ _ → 1r) , λ _ → k false tt))
        ∙ cong₂ _++_ (toL-one 1r (k true tt)) (toL-one 1r (k false tt)))
     ∙ cong (λ l → fromL ((r · 1r , k true tt) ∷ l)) (sym (toL-suc n (cons (r · 1r) ws) (cons (k false tt) γ)))
     ∙ fromL-∷ (r · 1r) (k true tt) ((suc n , cons (r · 1r) ws) , cons (k false tt) γ)
     ∙ cong {B = λ _ → Config} (λ q → (suc (suc n) , cons q (cons q ws)) , cons (k true tt) (cons (k false tt) γ))
         (·IdR r))
    (first-step r mul k n ws γ)

-- a run

-- act_r (return tt ⊗ return tt)
actPair : R → Comp (∅ tt*) (F 𝟙)
actPair r = op (act r) []ᵥ λ _ _ → op mul []ᵥ λ _ _ → return tt

open PolyMachine L₁ O₁ isSetLk isSetBase wmEqs freeModel Q′ tt* presentation 𝟙
open Comparing wm-ec using (module Cmp; module Fr; final-comparison)

-- two threads of weight 1r · r, both returning tt
twoReturns : R → Final
twoReturns r = (2 , cons (1r · r) (cons (1r · r) none)) , λ _ → tt

wm-run : ∀ r → ηC (actPair r) ⟶* retF (twoReturns r)
wm-run r =
  ΣPathP (ΣPathP (refl , funExt λ { zero → refl }) , funExt λ { zero → refl })
  by-equation wm-act 1r r _ 0 none none
  then wm-mul (1r · r) _ 0 none none
  then (ΣPathP (refl , funExt λ { zero → refl ; (suc zero) → refl }) by-equation done)

wm-run-sound : ∀ r → denote (ηC (actPair r)) ≡ denote (retF (twoReturns r))
wm-run-sound r = sound⟶* (wm-run r)

wm-run-unique : ∀ r ṽ → ηC (actPair r) ⟶* retF ṽ → ṽ ≡ twoReturns r
wm-run-unique r ṽ p = final-unique (reflect g𝟙) (reflect-den g𝟙) ṽ _ p (wm-run r)

wm-run-comparison : ∀ r →
  (ER.Q ⟪ PshHom→NatTrans (Cmp.same Cmp.fo𝟙) ⟫) (denF (twoReturns r))
    ≡ from Fr.⟦ 𝟙 ⟧v (Fr.⟦ actPair r ⟧C .PshHom.N-ob tt* refl)
wm-run-comparison r = final-comparison Cmp.fo𝟙 (actPair r) _ (wm-run r)
