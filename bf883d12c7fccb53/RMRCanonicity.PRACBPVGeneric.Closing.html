-- Closed computations and closing the locks of a closed operation
-- (syntax only).
--
-- A closed computation lives at the root: Comp (∅ w) B.  An operation
-- node of RMR has closed continuations at the scopes of its positions,
-- whereas a closed op o vs k has its continuations behind the locks of
-- their slots.  This module goes between the two.
--
--   opNodeC o cs ks   the syntactic operation node: an op whose parameters
--                     are constants and whose continuations are closed
--                     computations at the scopes, moved under the slot's
--                     locks by the root (wsW: the scope and the world
--                     behind the locks are the same world)
--   closeVals vs      a closed list of base values is a list of constants
--                     (a closed neutral does not exist)
--   closeK μ K ps     a continuation family behind the locks of a word,
--                     closed outermost lock first (close p idW: names
--                     become constants, familial representability).  It
--                     reads the family at wposW μ idW ps, which is ps only
--                     propositionally, so no term is transported.
--   closeOp o vs k    = opNodeC o (closeVals vs) (closeK on every slot)
--
-- ClosedLaws collects the syntactic facts the machine needs: closed
-- computations form a set, a presheaf under reindexC, and opNodeC is
-- natural.  They are proved from renaming extensionality and fusion
-- (Metatheory); the machine (Semantics.Machine) takes them as a parameter.
open import Cubical.Foundations.Prelude using (Type; ℓ-zero; _≡_; isSet)
open import Cubical.Categories.Category using (Category; _[_,_])
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Data.Unit using (tt)
open import Cubical.Data.Sigma using () renaming (_,_ to _,ˢ_)

open import RMRCanonicity.PRACBPVGeneric.Fam
open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Closing {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O

private
  module W = Category W

----------------------------------------------------------------------
-- the syntactic operation node

-- the world behind a lock word is its scope (a strict identity on each
-- word, so no transport): the root of a closed continuation
wsW : (μ : List Lk) (Θ : Tele) (ps : WPos μ (world Θ)) →
  WMap (wscope μ (world Θ) ps) (world (Θ ⧀*⟨ μ , ps ⟩))
wsW [] Θ ps = idW
wsW (κ ∷ []) Θ p = idW
wsW (κ ∷ μ@(_ ∷ _)) Θ (p ,ˢ ps) = wsW μ (Θ ⧀⟨ κ , p ⟩) ps

-- op o cs, with closed continuations at the scopes moved under the locks
opNodeC : ∀ {w B} (o : Op) → Cs w (params o) →
  ((a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ (wscope (slot o a) w ps)) B) → Comp (∅ w) B
opNodeC {w} o cs ks = op o (consts cs) λ a ps → renC (root (wsW (slot o a) (∅ w) ps)) (ks a ps)

----------------------------------------------------------------------
-- closing a closed operation

-- a closed value of base type is a constant
closeVals : ∀ {w bs} → Vals (∅ w) bs → Cs w bs
closeVals []ᵥ = tt
closeVals (const c ∷ᵥ vs) = c ,ˢ closeVals vs
closeVals (ne () ∷ᵥ vs)

-- close the locks of a word, outermost lock first
closeK : ∀ {w B} (μ : List Lk) → ((ps : WPos μ w) → Comp (∅ w ⧀*⟨ μ , ps ⟩) B) →
  (ps : WPos μ w) → Comp (∅ (wscope μ w ps)) B
closeK [] K ps = K ps
closeK (κ ∷ []) K p = closeC p (K p)
closeK (κ ∷ κ' ∷ μ') K (p ,ˢ ps) =
  closeK (κ' ∷ μ')
    (λ ps' → subC (liftWS (κ' ∷ μ') (close p idW) ps') (K (p ,ˢ wposW (κ' ∷ μ') idW ps')))
    ps

-- a closed operation as an operation node
closeOp : ∀ {w B} (o : Op) → Vals (∅ w) (params o) →
  ((a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) B) → Comp (∅ w) B
closeOp o vs k = opNodeC o (closeVals vs) (λ a → closeK (slot o a) (k a))

----------------------------------------------------------------------
-- the syntactic laws of closed computations

-- a lock word's arity on a map of W: positions pulled back, scope maps
wpos : (μ : List Lk) {w w' : World} → W [ w , w' ] → WPos μ w' → WPos μ w
wpos μ = Arity.pos W (wordE μ)

wsmap : (μ : List Lk) {w w' : World} (f : W [ w , w' ]) (ps : WPos μ w') →
  W [ wscope μ w (wpos μ f ps) , wscope μ w' ps ]
wsmap μ = Arity.smap W (wordE μ)

record ClosedLaws : Type where
  field
    isSetComp : ∀ {w B} → isSet (Comp (∅ w) B)
    -- closed computations form a presheaf under reindexC
    reindexC-id : ∀ {w B} (M : Comp (∅ w) B) → reindexC (W.id {w}) M ≡ M
    reindexC-seq : ∀ {u v w B} (f : W [ u , v ]) (g : W [ v , w ]) (M : Comp (∅ u) B) →
      reindexC (f W.⋆ g) M ≡ reindexC g (reindexC f M)
    -- the operation node is natural: reindexing moves the parameters and
    -- reindexes each continuation along the word's scope map
    reindexC-opNodeC : ∀ {w w' B} (f : W [ w , w' ]) (o : Op) (cs : Cs w (params o))
      (ks : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ (wscope (slot o a) w ps)) B) →
      reindexC f (opNodeC o cs ks)
        ≡ opNodeC o (cmaps W Const (params o) f cs)
            (λ a ps → reindexC (wsmap (slot o a) f ps) (ks a (wpos (slot o a) f ps)))
