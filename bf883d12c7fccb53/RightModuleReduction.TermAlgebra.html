{-# OPTIONS --no-allow-unsolved-metas --no-lossy-unification #-}

module RightModuleReduction.TermAlgebra where

open import Cubical.Foundations.Prelude
open import Cubical.Categories.Category using (Category; _^op)
open import Cubical.Categories.Functor using (_⟅_⟆)
open import Cubical.Categories.NaturalTransformation using (NatTrans; makeNatTransPath)
open import Cubical.Categories.Presheaf.Base using (Presheaf)
open import RightModuleReduction.Theory
open import RightModuleReduction.Model using (Hom)

open NatTrans

-- Operation nodes in the unquotiented term presheaf.
TermAlg : {W : Category ℓ-zero ℓ-zero} (Σ : Signature W) →
  Presheaf (W ^op) ℓ-zero → Algebra Σ
TermAlg Σ X .Algebra.Carrier = TermPsh Σ X
TermAlg Σ X .Algebra.algebra .N-ob w (o , k) = ops o k
TermAlg Σ X .Algebra.algebra .N-hom f = refl

-- Interpreting variables in an algebra extends to an algebra homomorphism.
evaluateHom : {W : Category ℓ-zero ℓ-zero} {Σ : Signature W}
  {X : Presheaf (W ^op) ℓ-zero} (A : Algebra Σ) →
  NatTrans X (A .Algebra.Carrier) → Hom Σ (TermAlg Σ X) A
evaluateHom A env .Hom.transformation = evaluate A env
evaluateHom A env .Hom.preserves = makeNatTransPath refl
