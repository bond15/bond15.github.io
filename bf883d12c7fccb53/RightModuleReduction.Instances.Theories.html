One : Category ℓ-zero ℓ-zero
One = TerminalCategory

-- Ordinary signatures: operation shapes with set-valued arities.
-- Every argument stays at the output world.
private
  module ConstantSignature {W : Category ℓ-zero ℓ-zero}
    (Ops : hSet ℓ-zero) (Args : Ops .fst → hSet ℓ-zero) where

    Shapes : Functor ((W ^op) ^op) (SET ℓ-zero)
    Shapes = Constant _ _ Ops

    B = Contravariant.∫ Shapes
    Tags = DiscreteCategory (Ops .fst , isSet→isGroupoid (Ops .snd))

    tag : Functor B Tags
    tag .F-ob (_ , o) = o
    tag .F-hom (_ , p) = sym p
    tag .F-id = Ops .snd _ _ _ _
    tag .F-seq _ _ = Ops .snd _ _ _ _

    positions : Functor B (SET ℓ-zero)
    positions = DiscFunc Args ∘F tag

    polynomial : Signature W
    polynomial .Poly.Shapes = Shapes
    polynomial .Poly.Pos = positions
    polynomial .Poly.s = TotalCategory.Fst ∘F Covariant.ForgetElements positions

{-
Error, on a carrier A.

Operations:
  1. raise : A (no arguments)

Laws:
  None.
-}

module Error where
  signature : Signature One
  signature = ConstantSignature.polynomial
    (Unit , isSetUnit) (λ _ → Empty.⊥ , (isProp→isSet λ ()))

  raiseT : {X : Presheaf (One ^op) ℓ-zero} → Term signature X tt*
  raiseT = ops tt (λ ())

  theory : Theory One
  theory = record
    { signature = signature
    ; Equation = Empty.⊥
    ; equationWorld = λ ()
    ; Variables = λ ()
    ; lhs = λ ()
    ; rhs = λ ()
    }

{-
Reader, with environments E and carrier A.

Operations:
  1. ask : (E → A) → A

Laws (x : A, k : E → E → A):
  1. ask-const:
       ask (λ _ → x) = x
  2. ask-ask:
       ask (λ e → ask (λ e′ → k e e′)) = ask (λ e → k e e)
-}

module Reader (Environments : hSet ℓ-zero) where
  private E = Environments .fst

  signature : Signature One
  signature = ConstantSignature.polynomial
    (Unit , isSetUnit) (λ _ → Environments)

  askT : {X : Presheaf (One ^op) ℓ-zero} →
    (E → Term signature X tt*) → Term signature X tt*
  askT = ops tt

  data Law : Type where
    ask-const ask-ask : Law

  variables : Law → hSet ℓ-zero
  variables ask-const = Unit , isSetUnit
  variables ask-ask = (E × E) , isSet× (Environments .snd) (Environments .snd)

  lhs rhs : (e : Law) → Term signature (Constant _ _ (variables e)) tt*
  lhs ask-const = askT (λ _ → var tt)
  lhs ask-ask = askT (λ e → askT (λ e′ → var (e , e′)))
  rhs ask-const = var tt
  rhs ask-ask = askT (λ e → var (e , e))

  theory : Theory One
  theory = record
    { signature = signature ; Equation = Law ; equationWorld = λ _ → tt*
    ; Variables = λ e → Constant _ _ (variables e) ; lhs = lhs ; rhs = rhs }

{-
Writer, with output monoid (M, ε, ·) and carrier A.

Operations:
  1. tell m : A → A (one operation for each m : M)

Laws (m n : M, x : A):
  1. tell-ε:
       tell ε x = x
  2. tell-tell:
       tell m (tell n x) = tell (m · n) x
-}

module Writer (M : Monoid ℓ-zero) where
  open MonoidStr (M .snd) using (ε; _·_; is-set)
  private Output = M .fst

  signature : Signature One
  signature = ConstantSignature.polynomial
    (Output , is-set) (λ _ → Unit , isSetUnit)

  tellT : {X : Presheaf (One ^op) ℓ-zero} → Output →
    Term signature X tt* → Term signature X tt*
  tellT m t = ops m (λ _ → t)

  data Law : Type where
    tell-ε : Law
    tell-tell : Output → Output → Law

  lhs rhs : Law → Term signature (Constant _ _ (Unit , isSetUnit)) tt*
  lhs tell-ε = tellT ε (var tt)
  lhs (tell-tell m n) = tellT m (tellT n (var tt))
  rhs tell-ε = var tt
  rhs (tell-tell m n) = tellT (m · n) (var tt)

  theory : Theory One
  theory = record
    { signature = signature ; Equation = Law ; equationWorld = λ _ → tt*
    ; Variables = λ _ → Constant _ _ (Unit , isSetUnit) ; lhs = lhs ; rhs = rhs }

{-
List nondeterminism, on a carrier A.

Operations:
  1. fail : A (no arguments)
  2. or : A → A → A (ordered binary choice)

Laws (x y z : A):
  1. fail-or:
       or fail x = x
  2. or-fail:
       or x fail = x
  3. or-assoc:
       or x (or y z) = or (or x y) z

These are the monoid laws; choice is neither required to commute nor
required to be idempotent.
-}

module List where
  private
    Args : Bool → hSet ℓ-zero
    Args false = Empty.⊥ , (isProp→isSet λ ())
    Args true = Bool , isSetBool

  signature : Signature One
  signature = ConstantSignature.polynomial (Bool , isSetBool) Args

  failT : {X : Presheaf (One ^op) ℓ-zero} → Term signature X tt*
  failT = ops false (λ ())

  orT : {X : Presheaf (One ^op) ℓ-zero} →
    Term signature X tt* → Term signature X tt* → Term signature X tt*
  orT l r = ops true λ { true → l ; false → r }

  data Three : Type where
    x y z : Three

  private
    encodeThree : Three → Bool ⊎ Unit
    encodeThree x = inl false
    encodeThree y = inl true
    encodeThree z = inr tt
    decodeThree : Bool ⊎ Unit → Three
    decodeThree (inl false) = x
    decodeThree (inl true) = y
    decodeThree (inr _) = z
    retractThree : (v : Three) → decodeThree (encodeThree v) ≡ v
    retractThree x = refl
    retractThree y = refl
    retractThree z = refl
    isSetThree : isSet Three
    isSetThree = isSetRetract encodeThree decodeThree retractThree
      (isSet⊎ isSetBool isSetUnit)

  data Law : Type where
    fail-or or-fail or-assoc : Law

  variables : Law → hSet ℓ-zero
  variables fail-or = Unit , isSetUnit
  variables or-fail = Unit , isSetUnit
  variables or-assoc = Three , isSetThree

  lhs rhs : (e : Law) → Term signature (Constant _ _ (variables e)) tt*
  lhs fail-or = orT failT (var tt)
  lhs or-fail = orT (var tt) failT
  lhs or-assoc = orT (var x) (orT (var y) (var z))
  rhs fail-or = var tt
  rhs or-fail = var tt
  rhs or-assoc = orT (orT (var x) (var y)) (var z)

  theory : Theory One
  theory = record
    { signature = signature ; Equation = Law ; equationWorld = λ _ → tt*
    ; Variables = λ e → Constant _ _ (variables e) ; lhs = lhs ; rhs = rhs }

{-
Global state, with an arbitrary set of stores S and carrier A.

Operations:
  1. get : (S → A) → A
  2. put s : A → A (one operation for each s : S)

Laws (s s′ : S, x : A, k of the indicated arity):
  1. get-put:
       get (λ s → put s x) = x
  2. get-get:
       get (λ s → get (λ s′ → k s s′)) = get (λ s → k s s)
  3. put-put:
       put s (put s′ x) = put s′ x
  4. put-get:
       put s (get k) = put s (k s)
-}

module GlobalState (Stores : hSet ℓ-zero) where
  private
    S = Stores .fst
    Ops : hSet ℓ-zero
    Ops = (Unit ⊎ S) , isSet⊎ isSetUnit (Stores .snd)

    Args : Ops .fst → hSet ℓ-zero
    Args (inl _) = Stores
    Args (inr _) = Unit , isSetUnit

  signature : Signature One
  signature = ConstantSignature.polynomial Ops Args

  getT : {X : Presheaf (One ^op) ℓ-zero} →
    (S → Term signature X tt*) → Term signature X tt*
  getT = ops (inl tt)

  putT : {X : Presheaf (One ^op) ℓ-zero} → S →
    Term signature X tt* → Term signature X tt*
  putT s t = ops (inr s) (λ _ → t)

  data Law : Type where
    get-put get-get : Law
    put-put : S → S → Law
    put-get : S → Law

  variables : Law → hSet ℓ-zero
  variables get-put = Unit , isSetUnit
  variables get-get = (S × S) , isSet× (Stores .snd) (Stores .snd)
  variables (put-put _ _) = Unit , isSetUnit
  variables (put-get _) = Stores

  lhs rhs : (e : Law) → Term signature (Constant _ _ (variables e)) tt*
  lhs get-put = getT (λ s → putT s (var tt))
  lhs get-get = getT (λ s → getT (λ s′ → var (s , s′)))
  lhs (put-put s s′) = putT s (putT s′ (var tt))
  lhs (put-get s) = putT s (getT (λ s′ → var s′))
  rhs get-put = var tt
  rhs get-get = getT (λ s → var (s , s))
  rhs (put-put s s′) = putT s′ (var tt)
  rhs (put-get s) = putT s (var s)

  theory : Theory One
  theory = record
    { signature = signature ; Equation = Law ; equationWorld = λ _ → tt*
    ; Variables = λ e → Constant _ _ (variables e) ; lhs = lhs ; rhs = rhs }

{-
Boolean local state, on a world-indexed carrier A : Inj → Set.
Here n is a world, i j : Fin n, and b c : Bool.

Operations:
  1. lookup i : (Bool → A(n)) → A(n)
  2. update i b : A(n) → A(n)
  3. new b : A(n + 1) → A(n)

In the laws, get, set, and alloc denote lookup, update, and new.
The term constructors below are named getT, setT, and allocT.
Write fresh = freshCell n and old j = prefixFin j.
Variables x and branch values k live at n in laws 1–7, and at n + 1
in laws 8–11. Every displayed equation has result world n.

Laws:
   1. get-set-current:
        get i (λ b → set i b x) = x
   2. set-get-same:
        set i b (get i k) = set i b (k b)
   3. set-set-same:
        set i b (set i c x) = set i c x
   4. get-get-same:
        get i (λ b → get i (λ c → k b c)) = get i (λ b → k b b)
   5. get-get-commute:
        get i (λ b → get j (λ c → k b c))
          = get j (λ c → get i (λ b → k b c))
   6. set-set-commute, assuming i ≠ j:
        set i b (set j c x) = set j c (set i b x)
   7. set-get-commute, assuming i ≠ j:
        set i b (get j k) = get j (λ c → set i b (k c))
   8. alloc-set-fresh:
        alloc b (set fresh c x) = alloc c x
   9. alloc-get-fresh:
        alloc b (get fresh k) = alloc b (k b)
  10. alloc-set-old:
        alloc b (set (old j) c x) = set j c (alloc b x)
  11. alloc-get-old:
        alloc b (get (old j) k) = get j (λ c → alloc b (k c))
-}

module LocalState where
  data Op : Type where
    lookup update new : Op

  private
    encodeOp : Op → Bool ⊎ Unit
    encodeOp lookup = inl true
    encodeOp update = inl false
    encodeOp new = inr tt

    decodeOp : Bool ⊎ Unit → Op
    decodeOp (inl true) = lookup
    decodeOp (inl false) = update
    decodeOp (inr _) = new

    decode-encodeOp : (o : Op) → decodeOp (encodeOp o) ≡ o
    decode-encodeOp lookup = refl
    decode-encodeOp update = refl
    decode-encodeOp new = refl

    isSetOp : isSet Op
    isSetOp = isSetRetract encodeOp decodeOp decode-encodeOp
      (isSet⊎ isSetBool isSetUnit)

    Parameters : Op → ℕ → hSet ℓ-zero
    Parameters lookup n = Fin n , isSetFin
    Parameters update n = (Fin n × Bool) , isSet× isSetFin isSetBool
    Parameters new n = Bool , isSetBool

    renameParameter : (o : Op) {n m : ℕ} → Inj [ n , m ] →
      Parameters o n .fst → Parameters o m .fst
    renameParameter lookup ρ i = ρ .fst i
    renameParameter update ρ (i , b) = ρ .fst i , b
    renameParameter new ρ b = b

    Shapes : Functor ((Inj ^op) ^op) (SET ℓ-zero)
    Shapes .F-ob n = (Σ[ o ∈ Op ] Parameters o n .fst) ,
      isSetΣ isSetOp (λ o → Parameters o n .snd)
    Shapes .F-hom ρ (o , p) = o , renameParameter o ρ p
    Shapes .F-id = funExt λ
      { (lookup , _) → refl ; (update , _) → refl ; (new , _) → refl }
    Shapes .F-seq ρ θ = funExt λ
      { (lookup , _) → refl ; (update , _) → refl ; (new , _) → refl }

    B = Contravariant.∫ Shapes
    Tags = DiscreteCategory (Op , isSet→isGroupoid isSetOp)

    tag : Functor B Tags
    tag .F-ob (_ , o , _) = o
    tag .F-hom (_ , p) = sym (cong fst p)
    tag .F-id = isSetOp _ _ _ _
    tag .F-seq _ _ = isSetOp _ _ _ _

    Args : Op → hSet ℓ-zero
    Args lookup = Bool , isSetBool
    Args update = Unit , isSetUnit
    Args new = Unit , isSetUnit

    positions : Functor B (SET ℓ-zero)
    positions = DiscFunc Args ∘F tag

    -- Allocation extends a renaming by the fresh last cell.
    extendWorld : Functor (Inj ^op) (Inj ^op)
    extendWorld .F-ob n = n + 1
    extendWorld .F-hom ρ = extendRight ρ 1
    extendWorld .F-id = extendRight-id
    extendWorld .F-seq ρ θ = sym (extendRight-seq θ ρ)

    argumentWorld : Op → Functor (Inj ^op) (Inj ^op)
    argumentWorld lookup = Id
    argumentWorld update = Id
    argumentWorld new = extendWorld

    worldFunctors : Functor B (FUNCTOR (Inj ^op) (Inj ^op))
    worldFunctors = DiscFunc argumentWorld ∘F tag

    result : Functor B (Inj ^op)
    result = appF (Inj ^op) (Inj ^op) ∘F
      (worldFunctors ,F TotalCategory.Fst)

  signature : Signature Inj
  signature .Poly.Shapes = Shapes
  signature .Poly.Pos = positions
  signature .Poly.s = result ∘F Covariant.ForgetElements positions

  -- The polynomial's structural action is the expected renaming of
  -- locations, with the fresh cell fixed in an allocation continuation.
  rename-layer : (A : Functor ((Inj ^op) ^op) (SET ℓ-zero)) {n m : ℕ}
    (ρ : Inj [ n , m ]) (o : Op) (p : Parameters o n .fst)
    (k : Args o .fst → (A ⟅ argumentWorld o ⟅ n ⟆ ⟆) .fst) →
    (Poly.Extension signature ⟅ A ⟆) .F-hom ρ ((o , p) , k) ≡
    ((o , renameParameter o ρ p) ,
      (λ e → A .F-hom (argumentWorld o .F-hom ρ) (k e)))
  rename-layer A {n} {m} ρ o p k = ΣPathP (refl , funExt λ e →
    cong₂ (λ (f : Inj [ argumentWorld o ⟅ n ⟆ , argumentWorld o ⟅ m ⟆ ]) x → A .F-hom f x)
      (cong (λ (α : NatTrans (argumentWorld o) (argumentWorld o)) → (Inj ._⋆_) (α .NatTrans.N-ob n)
          (argumentWorld o .F-hom ρ))
        (substRefl {B = λ o′ → NatTrans (argumentWorld o) (argumentWorld o′)} {x = o}
          (idTrans (argumentWorld o)))
        ∙ Inj .⋆IdL (argumentWorld o .F-hom ρ))
      (cong k (funExt⁻
        (substRefl {B = λ o′ → Args o .fst → Args o′ .fst} {x = o} (λ x → x)) e)))

  getT : {X : Presheaf (Inj ^op) ℓ-zero} {n : ℕ} → Fin n →
    (Bool → Term signature X n) → Term signature X n
  getT i k = ops (lookup , i) k

  setT : {X : Presheaf (Inj ^op) ℓ-zero} {n : ℕ} → Fin n → Bool →
    Term signature X n → Term signature X n
  setT i b t = ops (update , i , b) (λ _ → t)

  allocT : {X : Presheaf (Inj ^op) ℓ-zero} {n : ℕ} → Bool →
    Term signature X (n + 1) → Term signature X n
  allocT b t = ops (new , b) (λ _ → t)

  freshCell : (n : ℕ) → Fin (n + 1)
  freshCell n = suffixFin {n = n} zero

  -- Freely generated variables at world n: every injection gives a renamed copy.
  VarsAt : ℕ → hSet ℓ-zero → Presheaf (Inj ^op) ℓ-zero
  VarsAt n V .F-ob m = (V .fst × Inj [ n , m ]) , isSet× (V .snd) (Inj .isSetHom)
  VarsAt n V .F-hom f (a , g) = a , Inj ._⋆_ g f
  VarsAt n V .F-id = funExt λ (a , g) → cong (a ,_) (Inj .⋆IdR g)
  VarsAt n V .F-seq f g = funExt λ (a , h) → cong (a ,_) (sym (Inj .⋆Assoc h f g))

  at : {n : ℕ} {V : Type} → V → V × Inj [ n , n ]
  at a = a , Inj .id

  assignment : {n : ℕ} {V : hSet ℓ-zero} (A : Presheaf (Inj ^op) ℓ-zero) →
    (V .fst → (A ⟅ n ⟆) .fst) → NatTrans (VarsAt n V) A
  assignment A env .N-ob m (a , f) = A .F-hom f (env a)
  assignment A env .N-hom g = funExt λ (a , f) → funExt⁻ (A .F-seq f g) (env a)

  data Law : Type where
    get-set-current : (n : ℕ) → Fin n → Law
    set-get-same : (n : ℕ) → Fin n → Bool → Law
    set-set-same : (n : ℕ) → Fin n → Bool → Bool → Law
    get-get-same : (n : ℕ) → Fin n → Law
    get-get-commute : (n : ℕ) → Fin n → Fin n → Law
    set-set-commute : (n : ℕ) (i j : Fin n) → ¬ i ≡ j → Bool → Bool → Law
    set-get-commute : (n : ℕ) (i j : Fin n) → ¬ i ≡ j → Bool → Law
    alloc-set-fresh : (n : ℕ) → Bool → Bool → Law
    alloc-get-fresh : (n : ℕ) → Bool → Law
    alloc-set-old : (n : ℕ) → Fin n → Bool → Bool → Law
    alloc-get-old : (n : ℕ) → Fin n → Bool → Law

  lawWorld : Law → ℕ
  lawWorld (get-set-current n _) = n
  lawWorld (set-get-same n _ _) = n
  lawWorld (set-set-same n _ _ _) = n
  lawWorld (get-get-same n _) = n
  lawWorld (get-get-commute n _ _) = n
  lawWorld (set-set-commute n _ _ _ _ _) = n
  lawWorld (set-get-commute n _ _ _ _) = n
  lawWorld (alloc-set-fresh n _ _) = n
  lawWorld (alloc-get-fresh n _) = n
  lawWorld (alloc-set-old n _ _ _) = n
  lawWorld (alloc-get-old n _ _) = n

  variables : Law → Presheaf (Inj ^op) ℓ-zero
  variables (get-set-current n _) = VarsAt n (Unit , isSetUnit)
  variables (set-get-same n _ _) = VarsAt n (Bool , isSetBool)
  variables (set-set-same n _ _ _) = VarsAt n (Unit , isSetUnit)
  variables (get-get-same n _) = VarsAt n ((Bool × Bool) , isSet× isSetBool isSetBool)
  variables (get-get-commute n _ _) = VarsAt n ((Bool × Bool) , isSet× isSetBool isSetBool)
  variables (set-set-commute n _ _ _ _ _) = VarsAt n (Unit , isSetUnit)
  variables (set-get-commute n _ _ _ _) = VarsAt n (Bool , isSetBool)
  variables (alloc-set-fresh n _ _) = VarsAt (n + 1) (Unit , isSetUnit)
  variables (alloc-get-fresh n _) = VarsAt (n + 1) (Bool , isSetBool)
  variables (alloc-set-old n _ _ _) = VarsAt (n + 1) (Unit , isSetUnit)
  variables (alloc-get-old n _ _) = VarsAt (n + 1) (Bool , isSetBool)

  lhs rhs : (e : Law) → Term signature (variables e) (lawWorld e)
  lhs (get-set-current n i) = getT i (λ b → setT i b (var (at tt)))
  lhs (set-get-same n i b) = setT i b (getT i (λ c → var (at c)))
  lhs (set-set-same n i b c) = setT i b (setT i c (var (at tt)))
  lhs (get-get-same n i) = getT i (λ b → getT i (λ c → var (at (b , c))))
  lhs (get-get-commute n i j) = getT i (λ b → getT j (λ c → var (at (b , c))))
  lhs (set-set-commute n i j _ b c) = setT i b (setT j c (var (at tt)))
  lhs (set-get-commute n i j _ b) = setT i b (getT j (λ c → var (at c)))
  lhs (alloc-set-fresh n b c) = allocT b (setT (freshCell n) c (var (at tt)))
  lhs (alloc-get-fresh n b) = allocT b (getT (freshCell n) (λ c → var (at c)))
  lhs (alloc-set-old n j b c) = allocT b (setT (prefixFin j) c (var (at tt)))
  lhs (alloc-get-old n j b) = allocT b (getT (prefixFin j) (λ c → var (at c)))
  rhs (get-set-current n i) = var (at tt)
  rhs (set-get-same n i b) = setT i b (var (at b))
  rhs (set-set-same n i b c) = setT i c (var (at tt))
  rhs (get-get-same n i) = getT i (λ b → var (at (b , b)))
  rhs (get-get-commute n i j) = getT j (λ c → getT i (λ b → var (at (b , c))))
  rhs (set-set-commute n i j _ b c) = setT j c (setT i b (var (at tt)))
  rhs (set-get-commute n i j _ b) = getT j (λ c → setT i b (var (at c)))
  rhs (alloc-set-fresh n b c) = allocT c (var (at tt))
  rhs (alloc-get-fresh n b) = allocT b (var (at b))
  rhs (alloc-set-old n j b c) = setT j c (allocT b (var (at tt)))
  rhs (alloc-get-old n j b) = getT j (λ c → allocT b (var (at c)))

  theory : Theory Inj
  theory = record
    { signature = signature ; Equation = Law ; equationWorld = lawWorld
    ; Variables = variables ; lhs = lhs ; rhs = rhs }
