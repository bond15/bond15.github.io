-- Properties of the structural renamings and substitutions.
--
-- Every renaming is injective on neutrals (it has a left inverse).  So a
-- renaming never identifies a variable bound before a lock with a name
-- bound by that lock (freshness: ρ x ≠ ρ ℓ).  With renamings as arbitrary
-- functions on neutrals (the first version) this fails: a function may
-- send x to ℓ.
--
-- Locks are never deleted: a renaming does not decrease the number of
-- locks (locks-mono), and a substitution loses at most the locks directly
-- after the root (subInner, a counting bound; only close removes a lock).  The instances' noUnlock, noUntick
-- and their Sub versions are special cases.
open import Cubical.Foundations.Prelude using (Type; ℓ-zero; _≡_; refl; cong; sym; _∙_)
open import Cubical.Categories.Category using (Category)
open import Cubical.Data.Nat using (ℕ; zero; suc)
open import Cubical.Data.Nat.Order.Recursive using (_≤_; ≤-refl; ≤-trans)
open import Cubical.Data.Maybe using (Maybe; nothing; just; map-Maybe)
open import Cubical.Data.Maybe.Properties using (just-inj; ¬just≡nothing)
import Cubical.Data.Empty as ⊥
open import Cubical.Relation.Nullary using (¬_)

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Properties {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O

-- a left inverse of each renaming on neutrals
unren : ∀ {Θ Θ' A} → Ren Θ Θ' → Ne Θ' A → Maybe (Ne Θ A)
unren idR m = just m
unren (root f) m = nothing
unren (keep ρ) here = just here
unren (keep ρ) (wkv m) = map-Maybe wkv (unren ρ m)
unren (drop ρ) here = nothing
unren (drop ρ) (wkv m) = unren ρ m
unren (keepL ρ p) (up m) = map-Maybe up (unren ρ m)
unren (keepL ρ p) (opn m) = map-Maybe opn (unren ρ m)
unren (keepL ρ p) (nm k) = just (nm k)
unren (dropL ρ p) (up m) = unren ρ m
unren (dropL ρ p) (opn m) = nothing
unren (dropL ρ p) (nm k) = nothing

unren-ren : ∀ {Θ Θ' A} (ρ : Ren Θ Θ') (n : Ne Θ A) → unren ρ (renNe ρ n) ≡ just n
unren-ren idR n = refl
unren-ren (root f) ()
unren-ren (keep ρ) here = refl
unren-ren (keep ρ) (wkv n) = cong (map-Maybe wkv) (unren-ren ρ n)
unren-ren (drop ρ) n = unren-ren ρ n
unren-ren (keepL ρ p) (up n) = cong (map-Maybe up) (unren-ren ρ n)
unren-ren (keepL ρ p) (opn n) = cong (map-Maybe opn) (unren-ren ρ n)
unren-ren (keepL ρ p) (nm k) = refl
unren-ren (dropL ρ p) n = unren-ren ρ n

renNe-inj : ∀ {Θ Θ' A} (ρ : Ren Θ Θ') (n n' : Ne Θ A) → renNe ρ n ≡ renNe ρ n' → n ≡ n'
renNe-inj ρ n n' q =
  just-inj _ _ (sym (unren-ren ρ n) ∙ cong (unren ρ) q ∙ unren-ren ρ n')

private
  unup : ∀ {Δ κ p A} → Ne (Δ ⧀⟨ κ , p ⟩) A → Maybe (Ne Δ A)
  unup (up n) = just n
  unup (opn n) = nothing
  unup (nm k) = nothing

-- freshness: a variable from before a lock is never renamed to the same
-- neutral as a name bound by that lock
apart : ∀ {Δ κ p Θ' b} (ρ : Ren (Δ ⧀⟨ κ , p ⟩) Θ') (n : Ne Δ (base b)) (k : names κ ∋ b) →
  ¬ (renNe ρ (up n) ≡ renNe ρ (nm k))
apart ρ n k q = ¬just≡nothing (cong unup (renNe-inj ρ (up n) (nm k) q))

----------------------------------------------------------------------
-- locks are never deleted

-- the number of locks in a telescope
locks : Tele → ℕ
locks (∅ w) = 0
locks (Θ , A) = locks Θ
locks (Θ ⧀⟨ κ , p ⟩) = suc (locks Θ)

-- the locks NOT directly after the root (the ones close cannot remove)
inner : Tele → ℕ
inner (∅ w) = 0
inner (Θ , A) = inner Θ
inner (∅ w ⧀⟨ κ , p ⟩) = 0
inner ((Θ , A) ⧀⟨ κ , p ⟩) = suc (inner Θ)
inner ((Θ ⧀⟨ κ' , p' ⟩) ⧀⟨ κ , p ⟩) = suc (inner (Θ ⧀⟨ κ' , p' ⟩))

private
  ≤-step : ∀ m n → m ≤ n → m ≤ suc n
  ≤-step zero n _ = _
  ≤-step (suc m) (suc n) h = ≤-step m n h

  inner≤locks : ∀ Θ → inner Θ ≤ locks Θ
  inner≤locks (∅ w) = _
  inner≤locks (Θ , A) = inner≤locks Θ
  inner≤locks (∅ w ⧀⟨ κ , p ⟩) = _
  inner≤locks ((Θ , A) ⧀⟨ κ , p ⟩) = inner≤locks Θ
  inner≤locks ((Θ ⧀⟨ κ' , p' ⟩) ⧀⟨ κ , p ⟩) = inner≤locks (Θ ⧀⟨ κ' , p' ⟩)

  inner-lock : ∀ Δ κ p → inner (Δ ⧀⟨ κ , p ⟩) ≤ suc (inner Δ)
  inner-lock (∅ w) κ p = _
  inner-lock (Θ , A) κ p = ≤-refl (inner Θ)
  inner-lock (Θ ⧀⟨ κ' , p' ⟩) κ p = ≤-refl (inner (Θ ⧀⟨ κ' , p' ⟩))

-- a renaming never deletes a lock
locks-mono : ∀ {Θ Θ'} → Ren Θ Θ' → locks Θ ≤ locks Θ'
locks-mono {Θ} idR = ≤-refl (locks Θ)
locks-mono (root f) = _
locks-mono (keep ρ) = locks-mono ρ
locks-mono (drop ρ) = locks-mono ρ
locks-mono (keepL ρ p) = locks-mono ρ
locks-mono {Θ} (dropL {Δ' = Δ'} ρ p) = ≤-step (locks Θ) (locks Δ') (locks-mono ρ)

-- a substitution loses at most the locks directly after the root (counting bound)
subInner : ∀ {Θ Θ'} → Sub Θ Θ' → inner Θ ≤ locks Θ'
subInner {Θ} {Θ'} ⌜ ρ ⌝ = ≤-trans {inner Θ} {locks Θ} {locks Θ'} (inner≤locks Θ) (locks-mono ρ)
subInner (σ ,ₛ V) = subInner σ
subInner {Θ' = Θ'} (lockₛ {Δ = Δ} {Δ' = Δ'} {κ = κ} σ p τ) =
  ≤-trans {inner (Δ ⧀⟨ κ , q ⟩)} {suc (inner Δ)} {locks Θ'} (inner-lock Δ κ q)
    (≤-trans {suc (inner Δ)} {suc (locks Δ')} {locks Θ'} (subInner σ) (locks-mono τ))
  where q = posW κ (wmS σ) p
subInner (close p f) = _

-- no renaming removes a lock after a root (the semantic map exists, but it
-- sends the names to constants: it is the substitution close)
noUndo : ∀ {w w' κ p} → Ren (∅ w ⧀⟨ κ , p ⟩) (∅ w') → ⊥.⊥
noUndo ρ = locks-mono ρ
