-- Neutrals, values, value lists and computations form sets, when the lock
-- kinds and the base types do (Thm 8.1).
--
-- The hypotheses are needed because terms store types (to, case and app
-- store the type of the bound or applied value) and names store a base
-- type (nm k : Ne _ (base b) with k : names κ ∋ b).
--
--   bs ∋ b      a retract of ⊥ (at []) and (b' ≡ b) ⊎ (bs ∋ b)
--   Ne Θ A      a retract of a code defined by recursion on the telescope
--               (∅: none; a variable: here or wkv; a lock: up, opn or nm)
--   Val, Vals,  one indexed W-type over the sorts val Θ A | vals Θ bs |
--   Comp        comp Θ B.  Its labels are the constructors with their
--               non-recursive data (a neutral, a constant, a type, an
--               operation), so they are sets; its arities are the
--               subterms, and op's subterms are its parameter list and its
--               continuations at every slot and every position of the
--               slot's lock word.  Only the labels need to be sets.
open import Cubical.Foundations.Prelude hiding (_,_; comp)
open import Cubical.Foundations.HLevels using (isSetRetract; isSetΣ; isSet×)
open import Cubical.Data.Empty as ⊥ using (⊥; isProp⊥)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.Bool using (Bool; true; false; isSetBool)
open import Cubical.Data.Sum using (_⊎_; inl; inr; isSet⊎)
open import Cubical.Data.Sigma using (Σ-syntax) renaming (_,_ to _,ˢ_; _×_ to _×ˢ_)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Functor using (Functor)
import Cubical.Data.W.Indexed as IW

open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Metatheory.TermsSet
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L)) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
open import RMRCanonicity.PRACBPVGeneric.Metatheory.TypesSet isSetLk isSetBase public

open Functor

----------------------------------------------------------------------
-- names: membership in a list of base types

private
  ∋Code : List Base → Base → Type
  ∋Code [] b = ⊥
  ∋Code (b' ∷ bs) b = (b' ≡ b) ⊎ (bs ∋ b)

  enc∋ : ∀ {bs b} → bs ∋ b → ∋Code bs b
  enc∋ ∋z = inl refl
  enc∋ (∋s k) = inr k

  dec∋ : ∀ {bs b} → ∋Code bs b → bs ∋ b
  dec∋ {b' ∷ bs} (inl r) = subst ((b' ∷ bs) ∋_) r ∋z
  dec∋ {b' ∷ bs} (inr k) = ∋s k

  ret∋ : ∀ {bs b} (k : bs ∋ b) → dec∋ (enc∋ k) ≡ k
  ret∋ {b ∷ bs} ∋z = substRefl {B = (b ∷ bs) ∋_} ∋z
  ret∋ (∋s k) = refl

isSet∋ : ∀ {bs b} → isSet (bs ∋ b)
isSet∋ {[]} = isSetRetract enc∋ dec∋ ret∋ λ ()
isSet∋ {b' ∷ bs} {b} =
  isSetRetract enc∋ dec∋ ret∋ (isSet⊎ (isProp→isSet (isSetBase b' b)) isSet∋)

----------------------------------------------------------------------
-- neutrals, by recursion on the telescope

private
  NeCode : Tele → ValTy → Type
  NeCode (∅ w) A = ⊥
  NeCode (Θ , B) A = (B ≡ A) ⊎ Ne Θ A
  NeCode (Δ ⧀⟨ κ , p ⟩) A =
    Ne Δ A ⊎ (Ne Δ ([ κ ] A) ⊎ (Σ[ b ∈ Base ] (names κ ∋ b) ×ˢ (base b ≡ A)))

  encNe : ∀ {Θ A} → Ne Θ A → NeCode Θ A
  encNe here = inl refl
  encNe (wkv n) = inr n
  encNe (up n) = inl n
  encNe (opn n) = inr (inl n)
  encNe (nm {b = b} k) = inr (inr (b ,ˢ k ,ˢ refl))

  decNe : ∀ {Θ A} → NeCode Θ A → Ne Θ A
  decNe {Θ , B} (inl r) = subst (Ne (Θ , B)) r here
  decNe {Θ , B} (inr n) = wkv n
  decNe {Δ ⧀⟨ κ , p ⟩} (inl n) = up n
  decNe {Δ ⧀⟨ κ , p ⟩} (inr (inl n)) = opn n
  decNe {Δ ⧀⟨ κ , p ⟩} (inr (inr (b ,ˢ k ,ˢ r))) = subst (Ne (Δ ⧀⟨ κ , p ⟩)) r (nm k)

  retNe : ∀ {Θ A} (n : Ne Θ A) → decNe (encNe n) ≡ n
  retNe {Θ , B} here = substRefl {B = Ne (Θ , B)} here
  retNe (wkv n) = refl
  retNe (up n) = refl
  retNe (opn n) = refl
  retNe {Δ ⧀⟨ κ , p ⟩} (nm k) = substRefl {B = Ne (Δ ⧀⟨ κ , p ⟩)} (nm k)

isSetNe : ∀ {Θ A} → isSet (Ne Θ A)
isSetNe {∅ w} = isSetRetract encNe decNe retNe λ ()
isSetNe {Θ , B} {A} =
  isSetRetract encNe decNe retNe (isSet⊎ (isProp→isSet (isSetValTy B A)) isSetNe)
isSetNe {Δ ⧀⟨ κ , p ⟩} {A} =
  isSetRetract encNe decNe retNe
    (isSet⊎ isSetNe (isSet⊎ isSetNe
      (isSetΣ isSetBase λ b → isSet× isSet∋ (isProp→isSet (isSetValTy (base b) A)))))

----------------------------------------------------------------------
-- values, value lists and computations: one indexed W-type

private
  -- the sorts of the syntax, each at a telescope
  data Ix : Type where
    val : Tele → ValTy → Ix
    vals : Tele → List Base → Ix
    comp : Tele → CompTy → Ix

  -- the introduction form of a value type, with its non-recursive data
  -- (a constant for base b, the injection for A ⊕ A': true is inl; 𝟘 has
  -- none)
  IntroV : Tele → ValTy → Type
  IntroV Θ 𝟘 = ⊥
  IntroV Θ (A ⊕ A') = Bool
  IntroV Θ (base b) = Cst b (world Θ)
  IntroV Θ _ = Unit

  -- a computation is an introduction form (return or lam, by the type),
  -- or to (with the bound type), force, case (with the two bound types),
  -- app (with the argument type), op o, case₊ (with the two summands), or
  -- absurd
  Label : Ix → Type
  Label (val Θ A) = Ne Θ A ⊎ IntroV Θ A
  Label (vals Θ bs) = Unit
  Label (comp Θ B) =
    Unit ⊎ (ValTy ⊎ (Unit ⊎ ((ValTy ×ˢ ValTy) ⊎ (ValTy ⊎ (Op ⊎ ((ValTy ×ˢ ValTy) ⊎ Unit))))))

  Arity : (x : Ix) → Label x → Type
  Arity (val Θ A) (inl _) = ⊥
  Arity (val Θ 𝟙) (inr _) = ⊥
  Arity (val Θ 𝟘) (inr ())
  Arity (val Θ (A ⊕ A')) (inr _) = Unit
  Arity (val Θ (A × A')) (inr _) = Bool
  Arity (val Θ (U B)) (inr _) = Unit
  Arity (val Θ ([ κ ] A)) (inr _) = Pos κ (world Θ)
  Arity (val Θ (base b)) (inr _) = ⊥
  Arity (vals Θ []) _ = ⊥
  Arity (vals Θ (b ∷ bs)) _ = Bool
  Arity (comp Θ B) (inl _) = Unit
  Arity (comp Θ B) (inr (inl _)) = Bool
  Arity (comp Θ B) (inr (inr (inl _))) = Unit
  Arity (comp Θ B) (inr (inr (inr (inl _)))) = Bool
  Arity (comp Θ B) (inr (inr (inr (inr (inl _))))) = Bool
  Arity (comp Θ B) (inr (inr (inr (inr (inr (inl o)))))) =
    Unit ⊎ (Σ[ a ∈ Ar o ] WPos (slot o a) (world Θ))
  Arity (comp Θ B) (inr (inr (inr (inr (inr (inr (inl _))))))) = Unit ⊎ Bool
  Arity (comp Θ B) (inr (inr (inr (inr (inr (inr (inr _))))))) = Unit

  -- the subterm of return V or lam M
  introInput : Tele → CompTy → Ix
  introInput Θ (F A) = val Θ A
  introInput Θ (A ⇒ B) = comp (Θ , A) B

  Input : (x : Ix) (l : Label x) → Arity x l → Ix
  Input (val Θ A) (inl _) ()
  Input (val Θ 𝟙) (inr _) ()
  Input (val Θ 𝟘) (inr ()) _
  Input (val Θ (A ⊕ A')) (inr true) _ = val Θ A
  Input (val Θ (A ⊕ A')) (inr false) _ = val Θ A'
  Input (val Θ (A × A')) (inr _) true = val Θ A
  Input (val Θ (A × A')) (inr _) false = val Θ A'
  Input (val Θ (U B)) (inr _) _ = comp Θ B
  Input (val Θ ([ κ ] A)) (inr _) p = val (Θ ⧀⟨ κ , p ⟩) A
  Input (val Θ (base b)) (inr _) ()
  Input (vals Θ []) _ ()
  Input (vals Θ (b ∷ bs)) _ true = val Θ (base b)
  Input (vals Θ (b ∷ bs)) _ false = vals Θ bs
  Input (comp Θ B) (inl _) _ = introInput Θ B
  Input (comp Θ B) (inr (inl A)) true = comp Θ (F A)
  Input (comp Θ B) (inr (inl A)) false = comp (Θ , A) B
  Input (comp Θ B) (inr (inr (inl _))) _ = val Θ (U B)
  Input (comp Θ B) (inr (inr (inr (inl (A ,ˢ A'))))) true = val Θ (A × A')
  Input (comp Θ B) (inr (inr (inr (inl (A ,ˢ A'))))) false = comp (Θ , A , A') B
  Input (comp Θ B) (inr (inr (inr (inr (inl A))))) true = comp Θ (A ⇒ B)
  Input (comp Θ B) (inr (inr (inr (inr (inl A))))) false = val Θ A
  Input (comp Θ B) (inr (inr (inr (inr (inr (inl o)))))) (inl _) = vals Θ (params o)
  Input (comp Θ B) (inr (inr (inr (inr (inr (inl o)))))) (inr (a ,ˢ ps)) =
    comp (Θ ⧀*⟨ slot o a , ps ⟩) B
  Input (comp Θ B) (inr (inr (inr (inr (inr (inr (inl (A ,ˢ A')))))))) (inl _) = val Θ (A ⊕ A')
  Input (comp Θ B) (inr (inr (inr (inr (inr (inr (inl (A ,ˢ A')))))))) (inr true) = comp (Θ , A) B
  Input (comp Θ B) (inr (inr (inr (inr (inr (inr (inl (A ,ˢ A')))))))) (inr false) = comp (Θ , A') B
  Input (comp Θ B) (inr (inr (inr (inr (inr (inr (inr _))))))) _ = val Θ 𝟘

  Tree : Ix → Type
  Tree = IW.IW Label Arity Input

  isSetIntroV : ∀ Θ A → isSet (IntroV Θ A)
  isSetIntroV Θ 𝟘 = isProp→isSet isProp⊥
  isSetIntroV Θ 𝟙 = isSetUnit
  isSetIntroV Θ (A ⊕ A') = isSetBool
  isSetIntroV Θ (A × A') = isSetUnit
  isSetIntroV Θ (U B) = isSetUnit
  isSetIntroV Θ ([ κ ] A) = isSetUnit
  isSetIntroV Θ (base b) = Const b .F-ob (world Θ) .snd

  isSetLabel : ∀ x → isSet (Label x)
  isSetLabel (val Θ A) = isSet⊎ isSetNe (isSetIntroV Θ A)
  isSetLabel (vals Θ bs) = isSetUnit
  isSetLabel (comp Θ B) =
    isSet⊎ isSetUnit (isSet⊎ isSetValTy (isSet⊎ isSetUnit
      (isSet⊎ (isSet× isSetValTy isSetValTy) (isSet⊎ isSetValTy
        (isSet⊎ isSetOp (isSet⊎ (isSet× isSetValTy isSetValTy) isSetUnit))))))

  encV : ∀ {Θ A} → Val Θ A → Tree (val Θ A)
  encVs : ∀ {Θ bs} → Vals Θ bs → Tree (vals Θ bs)
  encC : ∀ {Θ B} → Comp Θ B → Tree (comp Θ B)

  encV (ne n) = IW.node (inl n) λ ()
  encV tt = IW.node (inr tt) λ ()
  encV (inl V) = IW.node (inr true) λ _ → encV V
  encV (inr V) = IW.node (inr false) λ _ → encV V
  encV (pair V V') = IW.node (inr tt) λ { true → encV V ; false → encV V' }
  encV (thunk M) = IW.node (inr tt) λ _ → encC M
  encV (const c) = IW.node (inr c) λ ()
  encV (shut V) = IW.node (inr tt) λ p → encV (V p)

  encVs []ᵥ = IW.node tt λ ()
  encVs (V ∷ᵥ vs) = IW.node tt λ { true → encV V ; false → encVs vs }

  encC (return V) = IW.node (inl tt) λ _ → encV V
  encC (lam M) = IW.node (inl tt) λ _ → encC M
  encC (_to_ {A = A} M N) = IW.node (inr (inl A)) λ { true → encC M ; false → encC N }
  encC (force V) = IW.node (inr (inr (inl tt))) λ _ → encV V
  encC (case {A = A} {A' = A'} V M) =
    IW.node (inr (inr (inr (inl (A ,ˢ A'))))) λ { true → encV V ; false → encC M }
  encC (app {A = A} M V) =
    IW.node (inr (inr (inr (inr (inl A))))) λ { true → encC M ; false → encV V }
  encC (op o vs k) =
    IW.node (inr (inr (inr (inr (inr (inl o)))))) λ { (inl _) → encVs vs ; (inr (a ,ˢ ps)) → encC (k a ps) }
  encC (case₊ {A = A} {A' = A'} V M N) =
    IW.node (inr (inr (inr (inr (inr (inr (inl (A ,ˢ A')))))))) λ
      { (inl _) → encV V ; (inr true) → encC M ; (inr false) → encC N }
  encC (absurd V) = IW.node (inr (inr (inr (inr (inr (inr (inr tt))))))) λ _ → encV V

  decV : ∀ {Θ A} → Tree (val Θ A) → Val Θ A
  decVs : ∀ {Θ bs} → Tree (vals Θ bs) → Vals Θ bs
  decC : ∀ {Θ B} → Tree (comp Θ B) → Comp Θ B
  decIntro : ∀ {Θ} B → ((x : Unit) → Tree (introInput Θ B)) → Comp Θ B

  decV (IW.node (inl n) _) = ne n
  decV {A = 𝟘} (IW.node (inr ()) _)
  decV {A = 𝟙} (IW.node (inr _) _) = tt
  decV {A = A ⊕ A'} (IW.node (inr true) k) = inl (decV (k tt))
  decV {A = A ⊕ A'} (IW.node (inr false) k) = inr (decV (k tt))
  decV {A = A × A'} (IW.node (inr _) k) = pair (decV (k true)) (decV (k false))
  decV {A = U B} (IW.node (inr _) k) = thunk (decC (k tt))
  decV {A = [ κ ] A} (IW.node (inr _) k) = shut λ p → decV (k p)
  decV {A = base b} (IW.node (inr c) _) = const c

  decVs {bs = []} _ = []ᵥ
  decVs {bs = b ∷ bs} (IW.node _ k) = decV (k true) ∷ᵥ decVs (k false)

  -- return or lam, by the type (a separate function, so that decC's other
  -- clauses do not split on the type)
  decIntro (F A) k = return (decV (k tt))
  decIntro (A ⇒ B) k = lam (decC (k tt))

  decC {B = B} (IW.node (inl _) k) = decIntro B k
  decC (IW.node (inr (inl A)) k) = decC (k true) to decC (k false)
  decC (IW.node (inr (inr (inl _))) k) = force (decV (k tt))
  decC (IW.node (inr (inr (inr (inl (A ,ˢ A'))))) k) = case (decV (k true)) (decC (k false))
  decC (IW.node (inr (inr (inr (inr (inl A))))) k) = app (decC (k true)) (decV (k false))
  decC (IW.node (inr (inr (inr (inr (inr (inl o)))))) k) =
    op o (decVs (k (inl tt))) λ a ps → decC (k (inr (a ,ˢ ps)))
  decC (IW.node (inr (inr (inr (inr (inr (inr (inl (A ,ˢ A')))))))) k) =
    case₊ (decV (k (inl tt))) (decC (k (inr true))) (decC (k (inr false)))
  decC (IW.node (inr (inr (inr (inr (inr (inr (inr _))))))) k) = absurd (decV (k tt))

  retV : ∀ {Θ A} (V : Val Θ A) → decV (encV V) ≡ V
  retVs : ∀ {Θ bs} (vs : Vals Θ bs) → decVs (encVs vs) ≡ vs
  retC : ∀ {Θ B} (M : Comp Θ B) → decC (encC M) ≡ M

  retV (ne n) = refl
  retV tt = refl
  retV (inl V) = cong inl (retV V)
  retV (inr V) = cong inr (retV V)
  retV (pair V V') = cong₂ pair (retV V) (retV V')
  retV (thunk M) = cong thunk (retC M)
  retV (const c) = refl
  retV (shut V) = cong shut (funExt λ p → retV (V p))

  retVs []ᵥ = refl
  retVs (V ∷ᵥ vs) = cong₂ _∷ᵥ_ (retV V) (retVs vs)

  retC (return V) = cong return (retV V)
  retC (lam M) = cong lam (retC M)
  retC (M to N) = cong₂ _to_ (retC M) (retC N)
  retC (force V) = cong force (retV V)
  retC (case V M) = cong₂ case (retV V) (retC M)
  retC (app M V) = cong₂ app (retC M) (retV V)
  retC (op o vs k) = cong₂ (op o) (retVs vs) (funExt λ a → funExt λ ps → retC (k a ps))
  retC (case₊ V M N) = cong₃ case₊ (retV V) (retC M) (retC N)
  retC (absurd V) = cong absurd (retV V)

isSetVal : ∀ {Θ A} → isSet (Val Θ A)
isSetVal {Θ} {A} = isSetRetract encV decV retV (IW.isOfHLevelSuc-IW 1 isSetLabel (val Θ A))

isSetVals : ∀ {Θ bs} → isSet (Vals Θ bs)
isSetVals {Θ} {bs} = isSetRetract encVs decVs retVs (IW.isOfHLevelSuc-IW 1 isSetLabel (vals Θ bs))

isSetComp : ∀ {Θ B} → isSet (Comp Θ B)
isSetComp {Θ} {B} = isSetRetract encC decC retC (IW.isOfHLevelSuc-IW 1 isSetLabel (comp Θ B))
