-- The old examples (error, global state, ...): operations with plain
-- arities and no world structure.  The lock signature is EMPTY (no base
-- types, no lock kinds) and every slot has the empty lock word.  So there
-- are no modal types and no locks: every telescope is ∅ w , A₁ , … , Aₙ,
-- exactly the old core's w | Γ (Flat below).
module RMRCanonicity.PRACBPVGeneric.Instances.Constant where

open import Cubical.Foundations.Prelude hiding (_,_)
open import Cubical.Foundations.HLevels using (isSetRetract)
open import Cubical.Data.Sum using (_⊎_; inl; inr; isSet⊎)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.Empty using (⊥; isProp⊥)
open import Cubical.Data.Bool using (Bool; true; false; isSetBool)
open import Cubical.Data.List using ([])
open import Cubical.Categories.Category using (Category)
open import Cubical.Categories.Instances.Terminal using (TerminalCategory)

open import RMRCanonicity.PRACBPVGeneric.Signature

data NoBase : Type where

module _ (W : Category ℓ-zero ℓ-zero) where
  noLocks : LockSig W
  noLocks .LockSig.Base = NoBase
  noLocks .LockSig.Const ()
  noLocks .LockSig.Lk = ⊥
  noLocks .LockSig.kind ()

  -- operations with constant arities: no parameters, every slot unlocked
  constOps : (Ops : Type) → isSet Ops → (Args : Ops → Type) → (∀ o → isSet (Args o)) →
    OpSig noLocks
  constOps Ops isSetOps Args isSetArgs .OpSig.Op = Ops
  constOps Ops isSetOps Args isSetArgs .OpSig.isSetOp = isSetOps
  constOps Ops isSetOps Args isSetArgs .OpSig.params o = []
  constOps Ops isSetOps Args isSetArgs .OpSig.Ar = Args
  constOps Ops isSetOps Args isSetArgs .OpSig.isSetAr = isSetArgs
  constOps Ops isSetOps Args isSetArgs .OpSig.slot o a = []

------------------------------------------------------------------------
-- error

data ErrOp : Type where
  raise : ErrOp

errorOps : OpSig (noLocks TerminalCategory)
errorOps = constOps TerminalCategory ErrOp (isProp→isSet λ { raise raise → refl })
  (λ _ → ⊥) (λ _ → isProp→isSet isProp⊥)

module Error where
  open import RMRCanonicity.PRACBPVGeneric.Syntax (noLocks TerminalCategory) errorOps public

  raiseC : ∀ {Θ B} → Comp Θ B
  raiseC = op raise []ᵥ λ ()

------------------------------------------------------------------------
-- Boolean global state

data GSOp : Type where
  get : GSOp
  put : Bool → GSOp

GSArgs : GSOp → Type
GSArgs get = Bool
GSArgs (put b) = Unit

private
  enc : GSOp → Unit ⊎ Bool
  enc get = inl tt
  enc (put b) = inr b

  dec : Unit ⊎ Bool → GSOp
  dec (inl _) = get
  dec (inr b) = put b

  dec-enc : ∀ o → dec (enc o) ≡ o
  dec-enc get = refl
  dec-enc (put b) = refl

isSetGSOp : isSet GSOp
isSetGSOp = isSetRetract enc dec dec-enc (isSet⊎ isSetUnit isSetBool)

isSetGSArgs : ∀ o → isSet (GSArgs o)
isSetGSArgs get = isSetBool
isSetGSArgs (put b) = isSetUnit

stateOps : OpSig (noLocks TerminalCategory)
stateOps = constOps TerminalCategory GSOp isSetGSOp GSArgs isSetGSArgs

module GlobalState where
  open import RMRCanonicity.PRACBPVGeneric.Syntax (noLocks TerminalCategory) stateOps public

  toggle : ∀ {Θ} → Comp Θ (F 𝟙)
  toggle = op get []ᵥ λ
    { true _ → op (put false) []ᵥ λ _ _ → return tt
    ; false _ → op (put true) []ᵥ λ _ _ → return tt }

  -- a continuation sees the variables directly: no lock, no up
  getThen : ∀ {Θ A} → Comp (Θ , A) (F A)
  getThen = op get []ᵥ λ _ _ → return (ne here)

  -- with no lock kinds every telescope is a root followed by variables
  data Flat : Tele → Type where
    root∅ : ∀ {w} → Flat (∅ w)
    var : ∀ {Θ A} → Flat Θ → Flat (Θ , A)

  flat : ∀ Θ → Flat Θ
  flat (∅ w) = root∅
  flat (Θ , A) = var (flat Θ)
  flat (Θ ⧀⟨ () , p ⟩)
