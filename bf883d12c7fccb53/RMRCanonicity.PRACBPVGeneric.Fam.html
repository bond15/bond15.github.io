-- Familial arities: functors into sums of representables.
--
-- Presheaves are covariant functors W → Set; the representable at u is
-- y u = W [ u , - ].  Fam W is the free coproduct completion of W^op,
-- i.e. (by Yoneda) the full subcategory of presheaves on the sums of
-- representables:
--
--   objects    (I , X)  with I a set and X : I → ob W     ("Σ_{i∈I} y (X i)")
--   morphisms  (I , X) → (J , Y)  =  Σ (φ : I → J) Π_i W [ Y (φ i) , X i ]
--
-- A FAMILIAL ARITY is a functor  E : W^op → Fam W,
--
--   E w = Σ_{p ∈ Pos w} y (scope w p),
--
-- so on f : w → w' it is (pos f , smap f): positions go backwards, and each
-- position p' at w' gets a map  scope w (pos f p') → scope w' p'.  The scope
-- depends on the position (scope : ∫ Pos → W).
--
--   R E X w = Π_{p ∈ Pos w} X (scope w p) = Psh (E w , X)   the right adjoint
--   L E Γ   = ∫^w Γ w × E w                                 its left adjoint (the lock)
--   L E (y u) = E u                                         familial representability
--
-- A STRONG familial arity also has a strength  str : E ⇒ y  (components
-- ι w p : w → scope w p), i.e. it is an object of the slice [W^op , Fam W] / y.
-- That slice has a unit (y), a composite (E ⊙ E') and sums (Σᴬ), and these
-- are all the structure the syntax and the RMR polynomial use.
module RMRCanonicity.PRACBPVGeneric.Fam where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels using (hSet; isSetΣ; isSetΠ; isSet→; isSet×)
open import Cubical.Data.Sigma using (ΣPathP; _×_)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Categories.Category using (Category; _[_,_]; _^op)
open import Cubical.Categories.Functor using (Functor)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Instances.Sets using (SET)

open Category
open Functor
open NatTrans

module _ (W : Category ℓ-zero ℓ-zero) where
  private
    module W = Category W

  Fam : Category (ℓ-suc ℓ-zero) ℓ-zero
  Fam .ob = Σ[ I ∈ hSet ℓ-zero ] (I .fst → W.ob)
  Fam .Hom[_,_] (I , X) (J , Y) = Σ[ φ ∈ (I .fst → J .fst) ] ((i : I .fst) → W [ Y (φ i) , X i ])
  Fam .id = (λ i → i) , (λ i → W.id)
  Fam ._⋆_ (φ , κ) (ψ , ν) = (λ i → ψ (φ i)) , (λ i → ν (φ i) W.⋆ κ i)
  Fam .⋆IdL (φ , κ) = ΣPathP (refl , funExt λ i → W.⋆IdR (κ i))
  Fam .⋆IdR (φ , κ) = ΣPathP (refl , funExt λ i → W.⋆IdL (κ i))
  Fam .⋆Assoc (φ , κ) (ψ , ν) (χ , μ) =
    ΣPathP (refl , funExt λ i → sym (W.⋆Assoc (μ (ψ (φ i))) (ν (φ i)) (κ i)))
  Fam .isSetHom {_} {J , _} = isSetΣ (isSet→ (J .snd)) (λ _ → isSetΠ λ _ → W.isSetHom)

  -- composition and identity with explicit objects (Fam's hom-types unfold
  -- to Σ-types, so its implicit objects cannot be inferred)
  seqF : ∀ x y z → Fam [ x , y ] → Fam [ y , z ] → Fam [ x , z ]
  seqF x y z f g = Fam ._⋆_ {x} {y} {z} f g

  idF : ∀ x → Fam [ x , x ]
  idF x = Fam .id {x}

  -- the Yoneda embedding, into one-summand families: Yo w = y w
  Yo : Functor (W ^op) Fam
  Yo .F-ob w = (Unit , isSetUnit) , (λ _ → w)
  Yo .F-hom f = (λ _ → tt) , (λ _ → f)
  Yo .F-id = refl
  Yo .F-seq f g = refl

  -- a global element of a presheaf (a map from the terminal presheaf)
  record Global (X : Functor W (SET ℓ-zero)) : Type where
    field
      el : ∀ w → X .F-ob w .fst
      nat : ∀ {w w'} (f : W [ w , w' ]) → X .F-hom f (el w) ≡ el w'

  -- the data of a familial arity, read off E
  module Arity (E : Functor (W ^op) Fam) where
    Pos : W.ob → Type
    Pos w = E .F-ob w .fst .fst

    isSetPos : ∀ w → isSet (Pos w)
    isSetPos w = E .F-ob w .fst .snd

    scope : (w : W.ob) → Pos w → W.ob
    scope w = E .F-ob w .snd

    pos : ∀ {w w'} → W [ w , w' ] → Pos w' → Pos w
    pos f = E .F-hom f .fst

    smap : ∀ {w w'} (f : W [ w , w' ]) (p : Pos w') → W [ scope w (pos f p) , scope w' p ]
    smap f = E .F-hom f .snd

    -- the right adjoint: R X w = Π_{p ∈ Pos w} X (scope w p) = Psh (E w , X)
    R : Functor W (SET ℓ-zero) → Functor W (SET ℓ-zero)
    R X .F-ob w = ((p : Pos w) → X .F-ob (scope w p) .fst) , isSetΠ (λ p → X .F-ob (scope w p) .snd)
    R X .F-hom f k p = X .F-hom (smap f p) (k (pos f p))
    R X .F-id = funExt λ k → funExt λ p →
      (λ i → X .F-hom (E .F-id i .snd p) (k (E .F-id i .fst p))) ∙ funExt⁻ (X .F-id) (k p)
    R X .F-seq f g = funExt λ k → funExt λ p →
      (λ i → X .F-hom (E .F-seq g f i .snd p) (k (E .F-seq g f i .fst p)))
      ∙ funExt⁻ (X .F-seq _ _) _

  -- the components of a strength E ⇒ y
  ιOf : {E : Functor (W ^op) Fam} → NatTrans E Yo →
    ∀ w (p : Arity.Pos E w) → W [ w , Arity.scope E w p ]
  ιOf σ w = σ .N-ob w .snd

  ----------------------------------------------------------------------
  -- composite of familial arities:  (E ⊙ E') w = Σ_{p ∈ E w} E' (scope w p)
  -- (Kleisli composition for the free coproduct completion; on presheaves
  -- it is the composite of the locks, L_{E ⊙ E'} = L_{E'} ∘ L_E)

  module _ (E E' : Functor (W ^op) Fam) where
    private
      module E = Arity E
      module E' = Arity E'

    _⊙_ : Functor (W ^op) Fam
    _⊙_ .F-ob w =
      ((Σ[ p ∈ E.Pos w ] E'.Pos (E.scope w p)) , isSetΣ (E.isSetPos w) (λ p → E'.isSetPos _)) ,
      λ x → E'.scope (E.scope w (x .fst)) (x .snd)
    _⊙_ .F-hom f =
      (λ x → E.pos f (x .fst) , E'.pos (E.smap f (x .fst)) (x .snd)) ,
      (λ x → E'.smap (E.smap f (x .fst)) (x .snd))
    _⊙_ .F-id {w} =
      (λ k → (λ x → E .F-id k .fst (x .fst) , E' .F-hom (E .F-id k .snd (x .fst)) .fst (x .snd)) ,
             (λ x → E' .F-hom (E .F-id k .snd (x .fst)) .snd (x .snd)))
      ∙ (λ k → (λ x → x .fst , E' .F-id k .fst (x .snd)) ,
               (λ x → E' .F-id k .snd (x .snd)))
    _⊙_ .F-seq f g =
      (λ k → (λ x → E .F-seq f g k .fst (x .fst) ,
                    E' .F-hom (E .F-seq f g k .snd (x .fst)) .fst (x .snd)) ,
             (λ x → E' .F-hom (E .F-seq f g k .snd (x .fst)) .snd (x .snd)))
      ∙ (λ k → (λ x → E.pos g (E.pos f (x .fst)) ,
                      E' .F-seq (E.smap f (x .fst)) (E.smap g (E.pos f (x .fst))) k .fst (x .snd)) ,
               (λ x → E' .F-seq (E.smap f (x .fst)) (E.smap g (E.pos f (x .fst))) k .snd (x .snd)))


  -- strengths compose: ι_{E ⊙ E'} (p , q) = ι_E p ⋆ ι_{E'} q
  module _ {E E' : Functor (W ^op) Fam} (σ : NatTrans E Yo) (σ' : NatTrans E' Yo) where
    private
      module E = Arity E
      module E' = Arity E'

      -- naturality of a strength, at one position
      nat : ∀ {F : Functor (W ^op) Fam} (τ : NatTrans F Yo) {w w'} (f : W [ w , w' ])
        (p : Arity.Pos F w') →
        ιOf τ w (Arity.pos F f p) W.⋆ Arity.smap F f p ≡ f W.⋆ ιOf τ w' p
      nat τ f p i = τ .N-hom f i .snd p

    _⊙ˢ_ : NatTrans (E ⊙ E') Yo
    _⊙ˢ_ .N-ob w = (λ _ → tt) , λ x → ιOf σ w (x .fst) W.⋆ ιOf σ' (E.scope w (x .fst)) (x .snd)
    _⊙ˢ_ .N-hom {w'} {w} f = ΣPathP (refl , funExt λ x →
      let p = x .fst ; q = x .snd ; g = E.smap f p in
        W.⋆Assoc (ιOf σ w (E.pos f p)) (ιOf σ' _ (E'.pos g q)) (E'.smap g q)
      ∙ cong (ιOf σ w (E.pos f p) W.⋆_) (nat σ' g q)
      ∙ sym (W.⋆Assoc (ιOf σ w (E.pos f p)) g (ιOf σ' _ q))
      ∙ cong (W._⋆ ιOf σ' _ q) (nat σ f p)
      ∙ W.⋆Assoc f (ιOf σ w' p) (ιOf σ' _ q))

  ----------------------------------------------------------------------
  -- sums of familial arities over a set:  (Σᴬ A E) w = Σ_{a ∈ A} E_a w

  module _ (A : hSet ℓ-zero) (E : A .fst → Functor (W ^op) Fam) where
    Σᴬ : Functor (W ^op) Fam
    Σᴬ .F-ob w =
      ((Σ[ a ∈ A .fst ] Arity.Pos (E a) w) , isSetΣ (A .snd) (λ a → Arity.isSetPos (E a) w)) ,
      λ x → Arity.scope (E (x .fst)) w (x .snd)
    Σᴬ .F-hom f =
      (λ x → x .fst , Arity.pos (E (x .fst)) f (x .snd)) ,
      (λ x → Arity.smap (E (x .fst)) f (x .snd))
    Σᴬ .F-id k =
      (λ x → x .fst , E (x .fst) .F-id k .fst (x .snd)) ,
      (λ x → E (x .fst) .F-id k .snd (x .snd))
    Σᴬ .F-seq f g k =
      (λ x → x .fst , E (x .fst) .F-seq f g k .fst (x .snd)) ,
      (λ x → E (x .fst) .F-seq f g k .snd (x .snd))

    Σˢ : (∀ a → NatTrans (E a) Yo) → NatTrans Σᴬ Yo
    Σˢ σ .N-ob w = (λ _ → tt) , λ x → ιOf (σ (x .fst)) w (x .snd)
    Σˢ σ .N-hom f = ΣPathP (refl , funExt λ x i → σ (x .fst) .N-hom f i .snd (x .snd))

  ----------------------------------------------------------------------
  -- lists of constants (parameters and bound names)

  module _ {Base : Type} (Const : Base → Functor W (SET ℓ-zero)) where

    Csts : W.ob → List Base → Type
    Csts w [] = Unit
    Csts w (b ∷ bs) = Const b .F-ob w .fst × Csts w bs

    cmaps : ∀ {w w'} (bs : List Base) → W [ w , w' ] → Csts w bs → Csts w' bs
    cmaps [] f cs = cs
    cmaps (b ∷ bs) f (c , cs) = Const b .F-hom f c , cmaps bs f cs

    isSetCsts : ∀ w bs → isSet (Csts w bs)
    isSetCsts w [] = isSetUnit
    isSetCsts w (b ∷ bs) = isSet× (Const b .F-ob w .snd) (isSetCsts w bs)

    cmaps-id : ∀ {w} bs (cs : Csts w bs) → cmaps bs W.id cs ≡ cs
    cmaps-id [] cs = refl
    cmaps-id (b ∷ bs) (c , cs) = ΣPathP (funExt⁻ (Const b .F-id) c , cmaps-id bs cs)

    cmaps-seq : ∀ {u v w} bs (f : W [ u , v ]) (g : W [ v , w ]) (cs : Csts u bs) →
      cmaps bs (f W.⋆ g) cs ≡ cmaps bs g (cmaps bs f cs)
    cmaps-seq [] f g cs = refl
    cmaps-seq (b ∷ bs) f g (c , cs) =
      ΣPathP (funExt⁻ (Const b .F-seq f g) c , cmaps-seq bs f g cs)

    CstsF : List Base → Functor W (SET ℓ-zero)
    CstsF bs .F-ob w = Csts w bs , isSetCsts w bs
    CstsF bs .F-hom f = cmaps bs f
    CstsF bs .F-id = funExt (cmaps-id bs)
    CstsF bs .F-seq f g = funExt (cmaps-seq bs f g)

    -- a lock that binds no names
    noNames : (E : Functor (W ^op) Fam) → Global (Arity.R E (CstsF []))
    noNames E .Global.el w p = tt
    noNames E .Global.nat f = refl
