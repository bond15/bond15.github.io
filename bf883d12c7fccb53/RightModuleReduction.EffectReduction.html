open import Cubical.Categories.Category using (Category; _^op)
open import Cubical.Categories.Functor using (Functor; _∘F_; _⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans; seqTrans)
open import Cubical.Categories.Presheaf.Base using (PresheafCategory)
open import Cubical.Categories.Monad.Base using (Monad; IsMonad)
open import Cubical.Categories.Adjoint using (module UnitCounit)
open import Cubical.Categories.Adjoint.Monad using (MonadFromAdjunction)
open import Polynomial.Presheaf.Discrete.Base
  using (Poly)
open import Cubical.Categories.Instances.Sets using (SET)
open import RightModuleReduction.IndexedContainer
  using (IndexedContainer; Extension; Contexts; plug; plug-map)
open import RightModuleReduction.Theory using (Theory; Algebra)
open import RightModuleReduction.Model using (FreeModel; modelForgetful; Hom; seqHom)
open import RightModuleReduction.RightModule using (RightModule)

module RightModuleReduction.EffectReduction
  {W : Category ℓ-zero ℓ-zero}
  (𝒯 : Theory W)
  (freeModel : FreeModel 𝒯)
  (Q′ : IndexedContainer (Category.ob W))
  where

open Category
open Functor
open NatTrans

-- Computations are world-indexed; configurations are sets.
𝒱 : Category (ℓ-suc ℓ-zero) ℓ-zero
𝒱 = PresheafCategory (W ^op) ℓ-zero

open IndexedContainer Q′ using () renaming (input to inQ)

-- This can be generalized from indexed containers to a FinBijPoly (W ^op) (K ^op),
-- giving Q : [W,Set] → [K,Set] with configurations indexed by output worlds.
Q : Functor 𝒱 (SET ℓ-zero)
Q = Extension Q′

∂Q : Functor 𝒱 (SET ℓ-zero)
∂Q = Contexts Q′

-- The free-model monad on covariant W-indexed families.
T : Monad 𝒱
T = (modelForgetful 𝒯 ∘F FreeModel.free freeModel) ,
    MonadFromAdjunction (FreeModel.free freeModel) (modelForgetful 𝒯)
      (FreeModel.adjunction freeModel)

module _
  (R : RightModule T Q)
  (alg : Algebra (Theory.signature 𝒯))
  where
  open RightModule R public

  A : 𝒱 .ob
  A = Algebra.Carrier alg

  private
    Σ : Functor 𝒱 𝒱
    Σ = Poly.Extension (Theory.signature 𝒯) {ℓX = ℓ-zero}

    algA : NatTrans (Σ ⟅ A ⟆) A
    algA = Algebra.algebra alg

    TA : 𝒱 .ob
    TA = T .fst ⟅ A ⟆

    ηA : NatTrans A TA
    ηA = IsMonad.η (T .snd) .N-ob A

    -- The supplied free model gives the interpretation of operations in TA.
    freeAlg : Algebra (Theory.signature 𝒯)
    freeAlg = (FreeModel.free freeModel ⟅ A ⟆) .fst

    -- Lift the arguments through η, then apply the free model's operation.
    freeOperation : NatTrans (Σ ⟅ A ⟆) TA
    freeOperation = seqTrans (Σ ⟪ ηA ⟫) (Algebra.algebra freeAlg)

    interp : (w : W .ob) → (Σ ⟅ A ⟆ ⟅ w ⟆) .fst → (TA ⟅ w ⟆) .fst
    interp = freeOperation .N-ob

  -- Lift the other fillers, plug the free operation, and apply the module action.
  step : (C@((s , p) , a) : (∂Q ⟅ A ⟆) .fst) →
    (Σ ⟅ A ⟆ ⟅ inQ s p ⟆) .fst →
    (Q ⟅ A ⟆) .fst
  step C@((s , p) , a) op+Args = ρ .N-ob A
    (plug Q′ TA
      ((∂Q ⟪ ηA ⟫) C)
      (interp (inQ s p) op+Args))

  -- The left-hand side interprets the operation in the source algebra.
  redex : (C@((s , p) , a) : (∂Q ⟅ A ⟆) .fst) →
    (Σ ⟅ A ⟆ ⟅ inQ s p ⟆) .fst →
    (Q ⟅ A ⟆) .fst
  redex C@((s , p) , a) op+Args =
    plug Q′ A C
      (algA .N-ob (inQ s p) op+Args)

  infix 4 _↦_ _↦*_

  data _↦_ : (Q ⟅ A ⟆) .fst →
             (Q ⟅ A ⟆) .fst → Type where
    effect-step : (C@((s , p) , a) : (∂Q ⟅ A ⟆) .fst)
      (op+Args : (Σ ⟅ A ⟆ ⟅ inQ s p ⟆) .fst) →
      redex C op+Args ↦ step C op+Args

  -- Finite runs, allowing equality rewrites between effect steps.
  data _↦*_ : (Q ⟅ A ⟆) .fst →
              (Q ⟅ A ⟆) .fst → Type where
    done : {x : (Q ⟅ A ⟆) .fst} → x ↦* x
    _then_ : {x y z : (Q ⟅ A ⟆) .fst} →
      x ↦ y → y ↦* z → x ↦* z
    _by-equation_ : {x y z : (Q ⟅ A ⟆) .fst} →
      x ≡ y → y ↦* z → x ↦* z

  infixr 5 _then_ _by-equation_

  -- A denotation preserves the signature operations and takes values in
  -- a chosen free model. The source algebra need not satisfy the equations.
  module Semantic (Y : 𝒱 .ob)
    (h : Hom (Theory.signature 𝒯) alg
      ((FreeModel.free freeModel ⟅ Y ⟆) .fst)) where

    private
      F = FreeModel.free freeModel
      module Adj = UnitCounit._⊣_ (FreeModel.adjunction freeModel)

      TY : 𝒱 .ob
      TY = T .fst ⟅ Y ⟆

      htr : NatTrans A TY
      htr = h .Hom.transformation

      μY : NatTrans (T .fst ⟅ TY ⟆) TY
      μY = IsMonad.μ (T .snd) .N-ob Y

      αY : NatTrans (Σ ⟅ TY ⟆) TY
      αY = Algebra.algebra ((F ⟅ Y ⟆) .fst)

      -- The free extension is a model homomorphism: F h followed by ε.
      bindHom : Hom (Theory.signature 𝒯) freeAlg ((F ⟅ Y ⟆) .fst)
      bindHom = seqHom (Theory.signature 𝒯)
        (F ⟪ htr ⟫) (Adj.ε .N-ob (F ⟅ Y ⟆))

      bind : NatTrans TA TY
      bind = bindHom .Hom.transformation

      -- Extending h and then restricting to generators recovers h.
      bind-unit : seqTrans ηA bind ≡ htr
      bind-unit =
        sym (𝒱 .⋆Assoc ηA (T .fst ⟪ htr ⟫) μY)
        ∙ cong (λ f → seqTrans f μY) (sym (Adj.η .N-hom htr))
        ∙ 𝒱 .⋆Assoc htr (Adj.η .N-ob TY) μY
        ∙ cong (seqTrans htr) (Adj.Δ₂ (F ⟅ Y ⟆))
        ∙ 𝒱 .⋆IdR htr

      targetOperation : NatTrans (Σ ⟅ A ⟆) TY
      targetOperation = seqTrans (Σ ⟪ htr ⟫) αY

      -- The free extension interprets a lifted operation using h on its arguments.
      bind-operation : seqTrans freeOperation bind ≡ targetOperation
      bind-operation =
        𝒱 .⋆Assoc (Σ ⟪ ηA ⟫) (Algebra.algebra freeAlg) bind
        ∙ cong (seqTrans (Σ ⟪ ηA ⟫)) (bindHom .Hom.preserves)
        ∙ sym (𝒱 .⋆Assoc (Σ ⟪ ηA ⟫) (Σ ⟪ bind ⟫) αY)
        ∙ cong (λ f → seqTrans f αY) (sym (Σ .F-seq ηA bind))
        ∙ cong (λ f → seqTrans (Σ ⟪ f ⟫) αY) bind-unit

    -- First interpret each computation, then apply the configuration action.
    denote : (Q ⟅ A ⟆) .fst → (Q ⟅ Y ⟆) .fst
    denote x = ρ .N-ob Y ((Q ⟪ htr ⟫) x)

    private
      -- Naturality and associativity of the module action allow the two
      -- successive actions to be combined into the free extension of h.
      action-bind : (z : (Q ⟅ TA ⟆) .fst) →
        denote (ρ .N-ob A z) ≡
        ρ .N-ob Y ((Q ⟪ bind ⟫) z)
      action-bind z =
        cong (ρ .N-ob Y)
          (cong (λ f → f z) (sym (ρ .N-hom htr)))
        ∙ cong
            (λ f → f ((Q ⟪ T .fst ⟪ htr ⟫ ⟫) z))
            (sym (ρ-assoc Y))
        ∙ cong (ρ .N-ob Y)
            (cong (λ f → f z)
              (sym (Q .F-seq (T .fst ⟪ htr ⟫) μY)))

      -- Both sides of the rule denote this configuration.
      target : (C@((s , p) , a) : (∂Q ⟅ A ⟆) .fst) →
        (Σ ⟅ A ⟆ ⟅ inQ s p ⟆) .fst →
        (Q ⟅ Y ⟆) .fst
      target C@((s , p) , a) op+Args = ρ .N-ob Y
        (plug Q′ TY ((∂Q ⟪ htr ⟫) C)
          (targetOperation .N-ob (inQ s p) op+Args))

      denote-redex : (C@((s , p) , a) : (∂Q ⟅ A ⟆) .fst)
        (op+Args : (Σ ⟅ A ⟆ ⟅ inQ s p ⟆) .fst) →
        denote (redex C op+Args) ≡ target C op+Args
      denote-redex C@((s , p) , a) op+Args =
        cong (ρ .N-ob Y)
          (plug-map Q′ A htr C
            (algA .N-ob (inQ s p) op+Args))
        ∙ cong
            (λ x → ρ .N-ob Y
              (plug Q′ TY ((∂Q ⟪ htr ⟫) C) x))
            (cong (λ f → f .N-ob (inQ s p) op+Args) (h .Hom.preserves))

      denote-step : (C@((s , p) , a) : (∂Q ⟅ A ⟆) .fst)
        (op+Args : (Σ ⟅ A ⟆ ⟅ inQ s p ⟆) .fst) →
        denote (step C op+Args) ≡ target C op+Args
      denote-step C@((s , p) , a) op+Args =
        action-bind
          (plug Q′ TA ((∂Q ⟪ ηA ⟫) C)
            (interp (inQ s p) op+Args))
        ∙ cong (ρ .N-ob Y)
            (plug-map Q′ TA bind ((∂Q ⟪ ηA ⟫) C)
              (interp (inQ s p) op+Args))
        ∙ (λ i → ρ .N-ob Y
            (plug Q′ TY ((∂Q ⟪ bind-unit i ⟫) C)
              (bind .N-ob (inQ s p)
                (interp (inQ s p) op+Args))))
        ∙ cong
            (λ x → ρ .N-ob Y
              (plug Q′ TY ((∂Q ⟪ htr ⟫) C) x))
            (cong (λ f → f .N-ob (inQ s p) op+Args) bind-operation)

    sound↦ : {x y : (Q ⟅ A ⟆) .fst} →
      x ↦ y → denote x ≡ denote y
    sound↦ (effect-step C@((s , p) , a) op+Args) =
      denote-redex C op+Args ∙ sym (denote-step C op+Args)

    sound↦* : {x y : (Q ⟅ A ⟆) .fst} →
      x ↦* y → denote x ≡ denote y
    sound↦* done = refl
    sound↦* (r then rs) = sound↦ r ∙ sound↦* rs
    sound↦* (p by-equation rs) =
      cong denote p ∙ sound↦* rs
