-- The denotational semantics in a FREE MODEL of a theory (POPL's
-- Denotation 𝒯 fm), next to Semantics.Denotation (which interprets F A by
-- the term algebra and is not changed).
--
-- Data: a theory over the polynomial (eqs), a free model of it (fm), and
-- the closure of its models under exponentials (ec, Semantics.StrongTheory;
-- automatic for equation-free theories and for chaotic W, e.g. W = 1).
--
--   Types       as Semantics.Denotation, except
--               ⟦ F A ⟧ = T ⟦ A ⟧, the free model on ⟦ A ⟧
--               ⟦ A ⇒ B ⟧ = ⟦ A ⟧ ⇒ ⟦ B ⟧, the exponential model (ec)
--               so every ⟦ B ⟧ is a MODEL of the theory (⟦_⟧m)
--   return V    η ∘ ⟦ V ⟧, the free model's unit
--   M to N      bind: N extended along the universal property of T ⟦ A ⟧
--               into the exponential model ⟦ Θ ⟧ ⇒ ⟦ B ⟧, then applied
--   the rest    as Semantics.Denotation (the same text)
--
-- bind's interface is the old one (bind-pre, bind-η, bind-op), proved from
-- the universal property (ext, ext-η, ext-unique) instead of by induction
-- on terms, plus bind-unique: a map algebraic in its first argument is the
-- bind of its restriction to η (F's η law in stack form).
--
-- Everything that does not depend on the interpretation of types (the
-- locks K, KW, constants, opNode, the exponential algebra) is reused from
-- Semantics.Denotation.

------------------------------------------------------------------------
-- the free model

-- models of the theory (algebras that satisfy it)
Model : Type₁
Model = Category.ob (Mod 𝒯)

-- the carrier of a model
∣_∣ : Model → 𝒱
∣ M ∣ = Algebra.Carrier (M .fst)

-- the free model on X, and its carrier
T : 𝒱 → Model
T X = FreeModel.free fm ⟅ X ⟆

private
  module Adj = UnitCounit._⊣_ (FreeModel.adjunction fm)
  module Bij = NaturalBijection._⊣_
    (adj→adj' (FreeModel.free fm) (modelForgetful 𝒯) (FreeModel.adjunction fm))

-- the unit
η : (X : 𝒱) → PshHom X ∣ T X ∣
η X = NatTrans→PshHom (Adj.η .NatTrans.N-ob X)

-- the universal property: extend an environment into a model
ext : ∀ {X} (M : Model) → NatTrans X ∣ M ∣ → Hom (PM.polynomial L O) (T X .fst) (M .fst)
ext {X} M env = Bij._♯ {X} {M} env

ext-η : ∀ {X} (M : Model) (env : NatTrans X ∣ M ∣) (v : W.ob) (x : El X v) →
  ext M env .transformation .NatTrans.N-ob v (η X .N-ob v x) ≡ env .NatTrans.N-ob v x
ext-η {X} M env v x i = Bij.adjIso {X} {M} .Iso.sec env i .NatTrans.N-ob v x

ext-unique : ∀ {X} (M : Model) (env : NatTrans X ∣ M ∣) (h : Hom (PM.polynomial L O) (T X .fst) (M .fst)) →
  (∀ v x → h .transformation .NatTrans.N-ob v (η X .N-ob v x) ≡ env .NatTrans.N-ob v x) →
  h ≡ ext M env
ext-unique {X} M env h p =
  sym (Bij.adjIso {X} {M} .Iso.ret h) ∙ cong (Bij._♯ {X} {M}) (makeNatTransPath (funExt λ v → funExt (p v)))

-- a homomorphism, pointwise at an operation
hom-op : ∀ {A B : Algebra (PM.polynomial L O)} (h : Hom (PM.polynomial L O) A B) (v : W.ob) (o : SN.Shape v)
  (ts : (p : SN.Position v o) → El (Algebra.Carrier A) (SN.scope v o p)) →
  h .transformation .NatTrans.N-ob v (Algebra.algebra A .NatTrans.N-ob v (o , ts))
    ≡ Algebra.algebra B .NatTrans.N-ob v (o , λ p → h .transformation .NatTrans.N-ob _ (ts p))
hom-op h v o ts i = h .preserves i .NatTrans.N-ob v (o , ts)

------------------------------------------------------------------------
-- the exponential model, and precomposition

expModel : 𝒱 → Model → Model
expModel G M = expAlg G (M .fst) , ec G (M .fst) (M .snd)

-- precomposing with φ in the environment: a homomorphism of exponentials
preExp : ∀ {G G' : 𝒱} (B : Algebra (PM.polynomial L O)) (φ : PshHom G' G) →
  Hom (PM.polynomial L O) (expAlg G B) (expAlg G' B)
preExp {G} {G'} B φ .transformation .NatTrans.N-ob w e .N-ob v (h , x') = e .N-ob v (h , φ .N-ob v x')
preExp {G} {G'} B φ .transformation .NatTrans.N-ob w e .N-hom u v f (h , x') =
  cong (λ y → e .N-ob u (_ , y)) (φ .N-hom u v f x') ∙ e .N-hom u v f (h , φ .N-ob v x')
preExp {G} {G'} B φ .transformation .NatTrans.N-hom f = funExt λ e → makePshHomPath refl
preExp {G} {G'} B φ .preserves = makeNatTransPath (funExt λ w → funExt λ (o , ks) →
  makePshHomPath (funExt λ v → funExt λ (g , x') →
    expAlgebra-ob G B w (o , ks) v g (φ .N-ob v x')
    ∙ cong (λ c → Algebra.algebra B .NatTrans.N-ob v (Poly.Shapes (PM.polynomial L O) .F-hom g o , c))
        (funExt λ p → cong (λ y → ks (tpos g o p) .N-ob _ (smapP g o p W.⋆ W.id , y))
          (sym (φ .N-hom _ _ (ι v (Poly.Shapes (PM.polynomial L O) .F-hom g o) p) x')))
    ∙ sym (expAlgebra-ob G' B w (o , λ p → preExp B φ .transformation .NatTrans.N-ob _ (ks p)) v g x')))

------------------------------------------------------------------------
-- sequencing

bindHom : (G X : 𝒱) (M : Model) → PshHom (G ×Psh X) ∣ M ∣ →
  Hom (PM.polynomial L O) (T X .fst) (expAlg G (M .fst))
bindHom G X M N = ext (expModel G M) (bindEnv G X (M .fst) N)

opaque
  bind : (G X : 𝒱) (M : Model) → PshHom (G ×Psh X) ∣ M ∣ → PshHom (∣ T X ∣ ×Psh G) ∣ M ∣
  bind G X M N =
    (NatTrans→PshHom (bindHom G X M N .transformation) ×PshHom idPshHom) ⋆PshHom appExp G ∣ M ∣

  -- bind is the extension, applied at the identity
  bind-at : (G X : 𝒱) (M : Model) (N : PshHom (G ×Psh X) ∣ M ∣) (v : W.ob) (t : El ∣ T X ∣ v) (x : El G v) →
    bind G X M N .N-ob v (t , x) ≡ bindHom G X M N .transformation .NatTrans.N-ob v t .N-ob v (W.id , x)
  bind-at G X M N v t x = refl

-- bind's interface (as Semantics.Algebra's): changing the environment,
-- the unit, and operations
bind-pre : (G G' X : 𝒱) (M : Model) (φ : PshHom G' G)
  (N : PshHom (G ×Psh X) ∣ M ∣) (N' : PshHom (G' ×Psh X) ∣ M ∣)
  (hyp : ∀ v (x' : El G' v) (a : El X v) → N' .N-ob v (x' , a) ≡ N .N-ob v (φ .N-ob v x' , a))
  (v : W.ob) (t : El ∣ T X ∣ v) (x' : El G' v) →
  bind G' X M N' .N-ob v (t , x') ≡ bind G X M N .N-ob v (t , φ .N-ob v x')
bind-pre G G' X M φ N N' hyp v t x' =
  bind-at G' X M N' v t x'
  ∙ (λ i → same i .transformation .NatTrans.N-ob v t .N-ob v (W.id , x'))
  ∙ sym (bind-at G X M N v t (φ .N-ob v x'))
  where
  same : bindHom G' X M N' ≡ seqHom (PM.polynomial L O) (bindHom G X M N) (preExp (M .fst) φ)
  same = sym (ext-unique (expModel G' M) (bindEnv G' X (M .fst) N')
    (seqHom (PM.polynomial L O) (bindHom G X M N) (preExp (M .fst) φ))
    λ w a → makePshHomPath (funExt λ u → funExt λ (h , x'') →
      (λ i → ext-η (expModel G M) (bindEnv G X (M .fst) N) w a i .N-ob u (h , φ .N-ob u x''))
      ∙ sym (hyp u x'' (X .F-hom h a))))

bind-η : (G X : 𝒱) (M : Model) (N : PshHom (G ×Psh X) ∣ M ∣) (v : W.ob) (a : El X v) (x : El G v) →
  bind G X M N .N-ob v (η X .N-ob v a , x) ≡ N .N-ob v (x , a)
bind-η G X M N v a x =
  bind-at G X M N v (η X .N-ob v a) x
  ∙ (λ i → ext-η (expModel G M) (bindEnv G X (M .fst) N) v a i .N-ob v (W.id , x))
  ∙ cong (λ a' → N .N-ob v (x , a')) (funExt⁻ (X .F-id) a)

bind-op : (G X : 𝒱) (M : Model) (N : PshHom (G ×Psh X) ∣ M ∣)
  (v : W.ob) (o : SN.Shape v) (ts : (p : SN.Position v o) → El ∣ T X ∣ (SN.scope v o p)) (x : El G v) →
  bind G X M N .N-ob v (Algebra.algebra (T X .fst) .NatTrans.N-ob v (o , ts) , x)
    ≡ Algebra.algebra (M .fst) .NatTrans.N-ob v
        (o , λ p → bind G X M N .N-ob (SN.scope v o p) (ts p , G .F-hom (ι v o p) x))
bind-op G X M N v o ts x =
  bind-at G X M N v _ x
  ∙ cong (λ f → f .N-ob v (W.id , x)) (hom-op (bindHom G X M N) v o ts)
  ∙ expAlgebra-id G (M .fst) v o _ x
  ∙ cong (λ c → Algebra.algebra (M .fst) .NatTrans.N-ob v (o , c))
      (funExt λ p → sym (bind-at G X M N _ (ts p) (G .F-hom (ι v o p) x)))

-- a map that is algebraic in its first argument is the bind of its
-- restriction to the unit
bind-unique : (G X : 𝒱) (M : Model) (α : PshHom (∣ T X ∣ ×Psh G) ∣ M ∣) →
  (∀ v (o : SN.Shape v) (ts : (p : SN.Position v o) → El ∣ T X ∣ (SN.scope v o p)) (x : El G v) →
    α .N-ob v (Algebra.algebra (T X .fst) .NatTrans.N-ob v (o , ts) , x)
      ≡ Algebra.algebra (M .fst) .NatTrans.N-ob v (o , λ p → α .N-ob _ (ts p , G .F-hom (ι v o p) x))) →
  ∀ v (t : El ∣ T X ∣ v) (x : El G v) →
  α .N-ob v (t , x) ≡ bind G X M (×PshIntro (π₂ _ _ ⋆PshHom η X) (π₁ _ _) ⋆PshHom α) .N-ob v (t , x)
bind-unique G X M α αop v t x =
  cong (λ t' → α .N-ob v (t' , x)) (sym (funExt⁻ (∣ T X ∣ .F-id) t))
  ∙ (λ i → same i .transformation .NatTrans.N-ob v t .N-ob v (W.id , x))
  ∙ sym (bind-at G X M N v t x)
  where
  N = ×PshIntro (π₂ _ _ ⋆PshHom η X) (π₁ _ _) ⋆PshHom α
  TXa = Algebra.algebra (T X .fst) .NatTrans.N-ob
  Ma = Algebra.algebra (M .fst) .NatTrans.N-ob

  -- α curried: a homomorphism into the exponential
  αc : Hom (PM.polynomial L O) (T X .fst) (expAlg G (M .fst))
  αc .transformation = PshHom→NatTrans (λExp G ∣ M ∣ α)
  αc .preserves = makeNatTransPath (funExt λ w → funExt λ (o , ts) →
    makePshHomPath (funExt λ u → funExt λ (h , y) →
      cong (λ t' → α .N-ob u (t' , y))
        (sym (funExt⁻ (Algebra.algebra (T X .fst) .NatTrans.N-hom h) (o , ts)))
      ∙ αop u _ _ y
      ∙ cong (λ c → Ma u (Poly.Shapes (PM.polynomial L O) .F-hom h o , c)) (funExt λ p →
          cong (λ m → α .N-ob _ (∣ T X ∣ .F-hom m (ts (tpos h o p)) ,
                                 G .F-hom (ι u (Poly.Shapes (PM.polynomial L O) .F-hom h o) p) y))
            (sym (W.⋆IdR _)))
      ∙ sym (expAlgebra-ob G (M .fst) w (o , λ p → λExp G ∣ M ∣ α .N-ob _ (ts p)) u h y)))

  same : αc ≡ bindHom G X M N
  same = ext-unique (expModel G M) (bindEnv G X (M .fst) N) αc λ w a →
    makePshHomPath (funExt λ u → funExt λ (h , y) →
      cong (λ t' → α .N-ob u (t' , y)) (sym (η X .N-hom _ _ h a)))

------------------------------------------------------------------------
-- types: every computation type denotes a model

⟦_⟧v : ValTy → 𝒱
⟦_⟧m : CompTy → Model

⟦ 𝟘 ⟧v = EmptyPsh
⟦ 𝟙 ⟧v = UnitPsh
⟦ A ⊕ A' ⟧v = ⟦ A ⟧v ⊎Psh ⟦ A' ⟧v
⟦ A Syn.× A' ⟧v = ⟦ A ⟧v ×Psh ⟦ A' ⟧v
⟦ U B ⟧v = ∣ ⟦ B ⟧m ∣
⟦ [ κ ] A ⟧v = K.R κ ⟦ A ⟧v
⟦ base b ⟧v = Cst b

⟦ F A ⟧m = T ⟦ A ⟧v
⟦ A ⇒ B ⟧m = expModel ⟦ A ⟧v ⟦ B ⟧m

-- the algebra, and its carrier
⟦_⟧c : CompTy → Algebra (PM.polynomial L O)
⟦ B ⟧c = ⟦ B ⟧m .fst

⟦_⟧u : CompTy → 𝒱
⟦ B ⟧u = ∣ ⟦ B ⟧m ∣

------------------------------------------------------------------------
-- telescopes: a presheaf over the representable of the world

⟦_⟧t : Tele → 𝒱
πt : (Θ : Tele) → PshHom ⟦ Θ ⟧t (Y (world Θ))

⟦ ∅ w ⟧t = Y w
⟦ Θ ▸ A ⟧t = ⟦ Θ ⟧t ×Psh ⟦ A ⟧v
⟦ Θ ⧀⟨ κ , p ⟩ ⟧t = K.Over.Summand κ (πt Θ) p

πt (∅ w) = idPshHom
πt (Θ ▸ A) = π₁ _ _ ⋆PshHom πt Θ
πt (Θ ⧀⟨ κ , p ⟩) = K.Over.πS κ (πt Θ) p

-- the summand inclusion of a lock
incl : ∀ {Δ κ p} → PshHom ⟦ Δ ⧀⟨ κ , p ⟩ ⟧t (K.L κ ⟦ Δ ⟧t)
incl {Δ} {κ} {p} = K.Over.inc κ (πt Δ) p

------------------------------------------------------------------------
-- neutrals

⟦_⟧ne : ∀ {Θ A} → Ne Θ A → PshHom ⟦ Θ ⟧t ⟦ A ⟧v
⟦ here ⟧ne = π₂ _ _
⟦ wkv n ⟧ne = π₁ _ _ ⋆PshHom ⟦ n ⟧ne
⟦ up {Δ} {κ} {p} {A} n ⟧ne = incl {Δ} {κ} {p} ⋆PshHom K.unlock κ (⟦ n ⟧ne ⋆PshHom K.box κ ⟦ A ⟧v)
⟦ opn {Δ} {κ} {p} n ⟧ne = incl {Δ} {κ} {p} ⋆PshHom K.unlock κ ⟦ n ⟧ne
⟦ nm {Δ} {κ} {p} k ⟧ne = incl {Δ} {κ} {p} ⋆PshHom K.unlock κ (namesR κ k)

------------------------------------------------------------------------
-- continuations behind a lock word: R_μ's introduction rule, iterated

shutW : ∀ {Θ} {X : 𝒱} (μ : List Lk) →
  ((ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X) → PshHom ⟦ Θ ⟧t (KW.R μ X)
shutW [] k = k tt ⋆PshHom unitR
shutW {Θ} (κ ∷ []) k = K.Over.shut κ (πt Θ) k
shutW {Θ} (κ ∷ μ@(κ' ∷ μ')) k =
  K.Over.shut κ (πt Θ) (λ p → shutW {Θ ⧀⟨ κ , p ⟩} μ (λ ps → k (p , ps))) ⋆PshHom uncurryR κ κ' μ'

-- the denotation of op o vs k, from those of vs and of the continuations
opDen : ∀ {Θ} (B : CompTy) (o : Op) → PshHom ⟦ Θ ⟧t (Par (params o)) →
  ((a : Ar o) (ps : WPos (slot o a) (world Θ)) → PshHom ⟦ Θ ⧀*⟨ slot o a , ps ⟩ ⟧t ⟦ B ⟧u) →
  PshHom ⟦ Θ ⟧t ⟦ B ⟧u
opDen {Θ} B o vs ks =
  opNode o vs (λ a → shutW {Θ} (slot o a) (ks a)) ⋆PshHom NatTrans→PshHom (Algebra.algebra ⟦ B ⟧c)

------------------------------------------------------------------------
-- values and computations: maps out of the telescope

⟦_⟧V : ∀ {Θ A} → Val Θ A → PshHom ⟦ Θ ⟧t ⟦ A ⟧v
⟦_⟧Vs : ∀ {Θ bs} → Vals Θ bs → PshHom ⟦ Θ ⟧t (Par bs)
⟦_⟧C : ∀ {Θ B} → Comp Θ B → PshHom ⟦ Θ ⟧t ⟦ B ⟧u

⟦ Syn.ne n ⟧V = ⟦ n ⟧ne
⟦ Syn.tt ⟧V = UnitPsh-intro
⟦ Syn.inl {A} {A'} V ⟧V = ⟦ V ⟧V ⋆PshHom inlPsh ⟦ A ⟧v ⟦ A' ⟧v
⟦ Syn.inr {A} {A'} V ⟧V = ⟦ V ⟧V ⋆PshHom inrPsh ⟦ A ⟧v ⟦ A' ⟧v
⟦ Syn.pair V V' ⟧V = ×PshIntro ⟦ V ⟧V ⟦ V' ⟧V
⟦ Syn.thunk M ⟧V = ⟦ M ⟧C
⟦_⟧V {Θ} (Syn.const {b} c) = πt Θ ⋆PshHom yoneda (Cst b) c
⟦_⟧V {Θ} (Syn.shut {κ} V) = K.Over.shut κ (πt Θ) (λ p → ⟦ V p ⟧V)

⟦ Syn.[]ᵥ ⟧Vs .N-ob v e = tt
⟦ Syn.[]ᵥ ⟧Vs .N-hom c c' f e = refl
⟦ V Syn.∷ᵥ vs ⟧Vs .N-ob v e = ⟦ V ⟧V .N-ob v e , ⟦ vs ⟧Vs .N-ob v e
⟦ V Syn.∷ᵥ vs ⟧Vs .N-hom c c' f e = ΣPathP (⟦ V ⟧V .N-hom c c' f e , ⟦ vs ⟧Vs .N-hom c c' f e)

-- the free model's unit
⟦ Syn.return {A} V ⟧C = ⟦ V ⟧V ⋆PshHom η ⟦ A ⟧v
-- the free model's universal property, into the exponential model
⟦_⟧C {Θ} (Syn._to_ {A} {B} M N) =
  ×PshIntro ⟦ M ⟧C idPshHom ⋆PshHom bind ⟦ Θ ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ N ⟧C
⟦ Syn.force V ⟧C = ⟦ V ⟧V
⟦ Syn.case V M ⟧C =
  ×PshIntro (×PshIntro idPshHom (⟦ V ⟧V ⋆PshHom π₁ _ _)) (⟦ V ⟧V ⋆PshHom π₂ _ _) ⋆PshHom ⟦ M ⟧C
⟦ Syn.case₊ V M N ⟧C = ×PshIntro idPshHom ⟦ V ⟧V ⋆PshHom casePsh ⟦ M ⟧C ⟦ N ⟧C
⟦ Syn.absurd {B} V ⟧C = ⟦ V ⟧V ⋆PshHom EmptyPsh-rec ⟦ B ⟧u
⟦ Syn.lam {A = A} {B} M ⟧C = λExp ⟦ A ⟧v ⟦ B ⟧u ⟦ M ⟧C
⟦ Syn.app {A = A} {B} M V ⟧C = ×PshIntro ⟦ M ⟧C ⟦ V ⟧V ⋆PshHom appExp ⟦ A ⟧v ⟦ B ⟧u
⟦_⟧C {Θ} (Syn.op {B} o vs k) = opDen B o ⟦ vs ⟧Vs (λ a ps → ⟦ k a ps ⟧C)

------------------------------------------------------------------------
-- stacks: maps U ⟦ B ⟧ × ⟦ Θ ⟧ → U ⟦ B' ⟧

⟦_⟧S : ∀ {Θ B B'} → Stack Θ B B' → PshHom (⟦ B ⟧u ×Psh ⟦ Θ ⟧t) ⟦ B' ⟧u
⟦ Syn.hole ⟧S = π₁ _ _
⟦_⟧S {Θ} (Syn._to•_ {A} {B' = B'} S N) =
