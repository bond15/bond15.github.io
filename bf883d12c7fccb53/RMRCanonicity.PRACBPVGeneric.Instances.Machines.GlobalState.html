-- The Boolean global-state machine (W = 1, no locks; get with two slots,
-- put b with one).
--
--   S _ = Bool                  the state
--   H = ⊥                       never halts
--   δ s get     = continue in slot s, state unchanged
--   δ s (put b) = continue in the slot, new state b
--
-- so Q X ≅ Bool × X(tt), RMR's state configurations.  The run: toggle
-- (read the bit and write its negation) from true ends in false.
module RMRCanonicity.PRACBPVGeneric.Instances.Machines.GlobalState where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels using (isSetΣ)
open import Cubical.Data.Unit using (Unit; tt; Unit*; tt*)
open import Cubical.Data.Unit.Properties using (isSetUnit*)
open import Cubical.Data.Empty using (⊥; isProp⊥)
open import Cubical.Data.Bool using (Bool; true; false; isSetBool)
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Data.Sigma using (Σ-syntax)
open import Cubical.Categories.Instances.Terminal using (TerminalCategory)

open import RightModuleReduction.Theory using (module SignatureNotation)

open import RMRCanonicity.PRACBPVGeneric.Instances.Constant
  using (NoBase; noLocks; stateOps; get; put; module GlobalState)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
open GlobalState hiding (_,_)
import RMRCanonicity.PRACBPVGeneric.Semantics.RunMachine as RunMachine

-- shapes, positions and scopes of the derived polynomial
module P = SignatureNotation (PM.polynomial (noLocks TerminalCategory) stateOps)

------------------------------------------------------------------------
-- the runner

isSetLk : isSet ⊥
isSetLk = isProp→isSet isProp⊥

isSetBase : isSet NoBase
isSetBase = isProp→isSet λ ()

-- the state: one bit
S : Unit* {ℓ-zero} → Type
S _ = Bool

isSetConf : isSet (Σ (Unit* {ℓ-zero}) S)
isSetConf = isSetΣ isSetUnit* λ _ → isSetBool

isSet⊥ : isSet ⊥
isSet⊥ = isProp→isSet isProp⊥

-- get continues in the slot named by the state; put b overwrites it
δ : ∀ {w} → S w → (o : P.Shape w) → ⊥ ⊎ (Σ[ p ∈ P.Position w o ] S (P.scope w o p))
δ s (get , tt) = inr ((s , tt) , s)
δ s (put b , tt) = inr ((tt , tt) , b)

------------------------------------------------------------------------
-- runs

open RunMachine (noLocks TerminalCategory) stateOps isSetLk isSetBase S ⊥ isSetConf isSet⊥ δ 𝟙

-- toggle from true: get (reads true), then put false
toggle-run : running true toggle ⟶* running false (return tt)
toggle-run = op-at then* op-at

-- and from false: get (reads false), then put true
toggle-run-false : running false toggle ⟶* running true (return tt)
toggle-run-false = op-at then* op-at

-- the runner runs toggle's denotation to the same final state
toggle-run-sound : run true (denClosed toggle) ≡ Rn.running ⟦ 𝟙 ⟧v false tt
toggle-run-sound = run-return toggle-run

toggle-run-false-sound : run false (denClosed toggle) ≡ Rn.running ⟦ 𝟙 ⟧v true tt
toggle-run-false-sound = run-return toggle-run-false

-- the same equations by computation (refl), independently of the machine
toggle-run-den : run true (denClosed toggle) ≡ Rn.running ⟦ 𝟙 ⟧v false tt
toggle-run-den = refl

toggle-run-false-den : run false (denClosed toggle) ≡ Rn.running ⟦ 𝟙 ⟧v true tt
toggle-run-false-den = refl
