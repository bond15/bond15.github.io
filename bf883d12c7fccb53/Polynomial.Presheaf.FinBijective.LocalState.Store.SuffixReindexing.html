module Polynomial.Presheaf.FinBijective.LocalState.Store.SuffixReindexing
  {v : Level} (V : hSet v) where

open import Polynomial.Presheaf.FinBijective.LocalState.Config V
open Category

-- Canonical inclusions of the public prefix and private suffix.
private
  mergePublic : ∀ {n n' k} (rho : Injection n n') →
    Store (n + k) → Store n' → Store n'
  mergePublic rho tau sigma' y with image? rho y
  ... | yes (x , _) = tau (prefixFin x)
  ... | no _ = sigma' y

  mergePublic-image : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') (x : Fin n) →
    mergePublic rho tau sigma' (rho .fst x) ≡ tau (prefixFin x)
  mergePublic-image rho tau sigma' x with image? rho (rho .fst x)
  ... | yes (x' , path) = cong (λ i → tau (prefixFin i))
    (isEmbedding→Inj (rho .snd) x' x path)
  ... | no not-image = ⊥.rec (not-image (x , refl))

  mergePublic-complement : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') (y : Fin n') →
    ¬ ImageFiber rho y → mergePublic rho tau sigma' y ≡ sigma' y
  mergePublic-complement rho tau sigma' y not-image with image? rho y
  ... | yes in-image = ⊥.rec (not-image in-image)
  ... | no _ = refl

  privateStore : ∀ {n k} → Store (n + k) → Store k
  privateStore tau i = tau (suffixFin i)

-- Reindex the public part of a result store along rho and retain its private
-- suffix verbatim.
reindexStore : ∀ {n n' k} (rho : Injection n n') →
  Store (n + k) → Store n' → Store (n' + k)
reindexStore rho tau sigma' =
  mergePublic rho tau sigma' ++Fin privateStore tau

private
  reindexStore-public : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') (y : Fin n') →
    reindexStore rho tau sigma' (prefixFin y)
    ≡ mergePublic rho tau sigma' y
  reindexStore-public {n' = n'} {k = k} rho tau sigma' y =
    sym (FinSumChar.++FinInl n' k _ _ y)

abstract
  reindexStore-image : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') (x : Fin n) →
    reindexStore rho tau sigma' (prefixFin (rho .fst x))
    ≡ tau (prefixFin x)
  reindexStore-image rho tau sigma' x =
    reindexStore-public rho tau sigma' (rho .fst x)
    ∙ mergePublic-image rho tau sigma' x

  reindexStore-complement : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') (y : Fin n') →
    ¬ ImageFiber rho y →
    reindexStore rho tau sigma' (prefixFin y) ≡ sigma' y
  reindexStore-complement rho tau sigma' y not-image =
    reindexStore-public rho tau sigma' y
    ∙ mergePublic-complement rho tau sigma' y not-image

  reindexStore-private : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') (c : Fin k) →
    reindexStore rho tau sigma' (suffixFin c) ≡ tau (suffixFin c)
  reindexStore-private {n' = n'} {k = k} rho tau sigma' c =
    sym (FinSumChar.++FinInr n' k _ _ c)

private
  abstract
    reindexStore-preserves : ∀ {n n' k} (rho : Injection n n')
      (tau : Store (n + k)) (sigma' : Store n') →
      (λ z → reindexStore rho tau sigma' (extendRight rho k .fst z)) ≡ tau
    reindexStore-preserves {n} {n'} {k} rho tau sigma' =
      finSumFunExt
        (λ x →
          cong (reindexStore rho tau sigma')
            (extendRight-map rho k (inl x))
          ∙ reindexStore-image rho tau sigma' x)
        (λ c →
          cong (reindexStore rho tau sigma')
            (extendRight-map rho k (inr c))
          ∙ reindexStore-private rho tau sigma' c)

-- The result configuration map uses rho on the public cells and the identity
-- on the private suffix.
reindexResultHom : ∀ {n n' k} (rho : Injection n n')
  (tau : Store (n + k)) (sigma' : Store n') →
  Config [ (n + k , tau) , (n' + k , reindexStore rho tau sigma') ]
reindexResultHom rho tau sigma' =
  extendRight rho _ , reindexStore-preserves rho tau sigma'

abstract
  reindexResultHom-fst : ∀ {n n' k} (rho : Injection n n')
    (tau : Store (n + k)) (sigma' : Store n') →
    reindexResultHom rho tau sigma' .fst ≡ extendRight rho k
  reindexResultHom-fst rho tau sigma' = refl

  reindexStore-id : ∀ {n k} (tau : Store (n + k)) (sigma : Store n) →
    reindexStore idInjection tau sigma ≡ tau
  reindexStore-id tau sigma =
    finSumFunExt
      (reindexStore-image idInjection tau sigma)
      (reindexStore-private idInjection tau sigma)

private
  mergePublic-seq : ∀ {n n' n'' k}
    {sigma : Store n} {sigma' : Store n'} {sigma'' : Store n''}
    (r : Config [ (n , sigma) , (n' , sigma') ])
    (s : Config [ (n' , sigma') , (n'' , sigma'') ])
    (tau : Store (n + k)) →
    mergePublic (s .fst) (reindexStore (r .fst) tau sigma') sigma''
    ≡ mergePublic (composeInjection (r .fst) (s .fst)) tau sigma''
  mergePublic-seq {sigma' = sigma'} {sigma'' = sigma''} r s tau =
    funExt point
    where
    rho = r .fst
    kappa = s .fst
    inner = reindexStore rho tau sigma'
    lhs = mergePublic kappa inner sigma''
    rhs = mergePublic (composeInjection rho kappa) tau sigma''

    middle : (y : Fin _) → mergePublic rho tau sigma' y ≡ rhs (kappa .fst y)
    middle y with image? rho y
    ... | yes (x , path) =
      sym (mergePublic-image (composeInjection rho kappa) tau sigma'' x)
      ∙ cong rhs (cong (kappa .fst) path)
    ... | no not-rho =
      sym (funExt⁻ (s .snd) y)
      ∙ sym (mergePublic-complement (composeInjection rho kappa) tau sigma''
        (kappa .fst y) (λ (x , path) →
          not-rho (x , isEmbedding→Inj (kappa .snd) (rho .fst x) y path)))

    point : (z : Fin _) → lhs z ≡ rhs z
    point z with image? kappa z
    ... | yes (y , path) =
      reindexStore-public rho tau sigma' y
      ∙ middle y
      ∙ cong rhs path
    ... | no not-kappa =
      sym (mergePublic-complement (composeInjection rho kappa) tau sigma'' z
        (λ (x , path) → not-kappa (rho .fst x , path)))

abstract
  reindexStore-seq : ∀ {n n' n'' k}
    {sigma : Store n} {sigma' : Store n'} {sigma'' : Store n''}
    (r : Config [ (n , sigma) , (n' , sigma') ])
    (s : Config [ (n' , sigma') , (n'' , sigma'') ])
    (tau : Store (n + k)) →
    reindexStore (s .fst) (reindexStore (r .fst) tau sigma') sigma''
    ≡ reindexStore (composeInjection (r .fst) (s .fst)) tau sigma''
  reindexStore-seq
      {n} {n'} {n''} {k} {sigma} {sigma'} {sigma''} r s tau =
    finSumFunExt prefix suffix
    where
    lhs : Store (n'' + k)
    lhs = reindexStore (s .fst) (reindexStore (r .fst) tau _) _

    rhs : Store (n'' + k)
    rhs = reindexStore (composeInjection (r .fst) (s .fst)) tau _

    prefix : (y : Fin n'') → lhs (prefixFin y) ≡ rhs (prefixFin y)
    prefix y =
      reindexStore-public (s .fst) (reindexStore (r .fst) tau _) _ y
      ∙ funExt⁻ (mergePublic-seq r s tau) y
      ∙ sym (reindexStore-public
          (composeInjection (r .fst) (s .fst)) tau _ y)

    suffix : (c : Fin k) → lhs (suffixFin c) ≡ rhs (suffixFin c)
    suffix c =
      reindexStore-private (s .fst) (reindexStore (r .fst) tau _) _ c
      ∙ reindexStore-private (r .fst) tau _ c
      ∙ sym (reindexStore-private
          (composeInjection (r .fst) (s .fst)) tau _ c)
