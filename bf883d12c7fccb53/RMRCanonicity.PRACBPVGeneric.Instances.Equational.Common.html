-- Free models presented by a container at W = 1 (POPL's
-- Instances/Common, mkOperationalModel, and its FreeModel record).
--
-- Data: plain operations (Op, Ar: no locks, no parameters, every slot
-- unlocked, as Instances.Constant.constOps), a theory eqs over their
-- polynomial, and a container: shapes Sh, positions Ps (discrete), so
--
--   Ext Z = Σ_{s ∈ Sh} (Ps s → Z)            POPL's ⟦ p ⟧ Z
--
-- with a unit shape ηs and operations opP on every Ext Z, natural in Z
-- (opP-nat).  The free model's carrier on X is Ext (X tt) (Carrier, an
-- algebra polyAlg), its unit x ↦ (ηs , λ _ → x).  Given that the algebras
-- satisfy the theory and the universal property, stated at the level of
-- types as POPL states it (PolyFree: ext, ext-η, ext-hom, ext-unique),
-- this module builds (Presented eqs Sh Ps decPs, then WithOps ηs opP
-- opP-nat, then WithFree pf)
--
--   freeModel     RMR's FreeModel (free functor and adjunction,
--                 RightModuleReduction.FreeAdjunction)
--   Q′            the configuration container (input = the one world)
--   presentation  Q X = T X tt, both directions the identity (refl), for
--                 Semantics.PolySelfModule (ρ = μ) and Semantics.PolyMachine
--
-- (The module ρ = μ is not restated here: an instance gets it inside its
-- application of PolyMachine.  Checking PolySelfModule.polyModule against a
-- type at a concrete free model unifies the result type before the
-- arguments, and unfolds the free model; inside PolyMachine the free model
-- is a parameter.)
--
-- Also: chaotic (W = 1 is chaotic, so every theory is closed under
-- exponentials, StrongTheory.expClosed-chaotic), finOpOf (finitely many
-- continuations from finitely many slots), isSetLk, isSetBase.

module RMRCanonicity.PRACBPVGeneric.Instances.Equational.Common
  (Op : Type) (isSetOp : isSet Op) (Ar : Op → Type) (isSetAr : ∀ o → isSet (Ar o)) where

open Functor
open NatTrans

-- one world, no locks, plain operations
𝟙W : Category ℓ-zero ℓ-zero
𝟙W = TerminalCategory

L₁ : LockSig 𝟙W
L₁ = noLocks 𝟙W

O₁ : OpSig L₁
O₁ = constOps 𝟙W Op isSetOp Ar isSetAr

isSetLk : isSet ⊥
isSetLk = isProp→isSet isProp⊥

isSetBase : isSet NoBase
isSetBase = isProp→isSet λ ()

-- every theory over this polynomial is closed under exponentials
chaotic : StrongTheory.Chaotic L₁ O₁
chaotic .StrongTheory.Chaotic.the w w' = isPropUnit* w w'
chaotic .StrongTheory.Chaotic.unique f = isSetUnit* _ _ f _

-- finitely many continuations, from finitely many slots
finOpOf : (∀ o → isFinOrd (Ar o)) → ∀ (o : Op) w →
  isFinSet (Σ[ a ∈ Ar o ] Notation.WPos L₁ (OpSig.slot O₁ o a) w)
finOpOf finAr o w = isFinOrd→isFinSet (isFinOrdΣ (Ar o) (finAr o) _ λ _ → isFinOrdUnit)

private
  𝒞 = PresheafCategory (𝟙W ^op) ℓ-zero

  El : 𝒞 .Category.ob → Type
  El X = X .F-ob tt* .fst

module Presented (eqs : Machine.Equations L₁ O₁)
  -- the container
  (Sh : hSet ℓ-zero) (Ps : Sh .fst → hSet ℓ-zero) (decPs : ∀ s → Discrete (Ps s .fst))
  where

  -- (the theory is always written Machine.theory L₁ O₁ eqs, never through
  -- an alias: an alias is not syntactically the instances' type, and
  -- comparing the two unfolds the free-model monad)
  -- POPL's ⟦ p ⟧ Z, and its action on maps
  Ext : Type → Type
  Ext Z = Σ[ s ∈ Sh .fst ] (Ps s .fst → Z)

  pmap : ∀ {Z Z'} → (Z → Z') → Ext Z → Ext Z'
  pmap g t = t .fst , λ r → g (t .snd r)

  -- models, read at the one world
  Model : Type₁
  Model = Mod (Machine.theory L₁ O₁ eqs) .Category.ob

  ∣_∣ : Model → Type
  ∣ N ∣ = El (Algebra.Carrier (N .fst))

  -- an operation of a model, its continuations indexed by the slots
  Nalg : (N : Model) (o : Op) → (Ar o → ∣ N ∣) → ∣ N ∣
  Nalg N o k = Algebra.algebra (N .fst) .N-ob tt* ((o , tt) , λ q → k (q .fst))

  -- the configuration container: every hole at the one world
  Q′ : IndexedContainer Unit*
  Q′ .IndexedContainer.Shape = Sh
  Q′ .IndexedContainer.Pos = Ps
  Q′ .IndexedContainer.decPos = decPs
  Q′ .IndexedContainer.input _ _ = tt*

  module WithOps (ηs : Sh .fst)
    (opP : ∀ {Z} (o : Op) → (Ar o → Ext Z) → Ext Z)
    (opP-nat : ∀ {Z Z'} (g : Z → Z') (o : Op) (k : Ar o → Ext Z) →
       pmap g (opP o k) ≡ opP o (λ a → pmap g (k a)))
    where

    -- the carrier on X: Ext (X tt), moved by X's action
    Carrier : 𝒞 .Category.ob → 𝒞 .Category.ob
    Carrier X .F-ob w = Ext (X .F-ob w .fst) , isSetΣ (Sh .snd) (λ s → isSetΠ λ _ → X .F-ob w .snd)
    Carrier X .F-hom f = pmap (X .F-hom f)
    Carrier X .F-id = funExt λ t → ΣPathP (refl , funExt λ r → funExt⁻ (X .F-id) (t .snd r))
    Carrier X .F-seq f g = funExt λ t → ΣPathP (refl , funExt λ r → funExt⁻ (X .F-seq f g) (t .snd r))

    polyAlg : 𝒞 .Category.ob → Algebra (PM.polynomial L₁ O₁)
    polyAlg X .Algebra.Carrier = Carrier X
    polyAlg X .Algebra.algebra = naturally _ _ λ w ok → opP (ok .fst .fst) λ a → ok .snd (a , tt)

    -- the satisfaction and universal property, at the level of types
    record PolyFree : Type₁ where
      field
        sat : ∀ X → Satisfies (Machine.theory L₁ O₁ eqs) (polyAlg X)
        ext : (N : Model) {Z : Type} → (Z → ∣ N ∣) → Ext Z → ∣ N ∣
        ext-η : ∀ N {Z} (f : Z → ∣ N ∣) (z : Z) → ext N f (ηs , λ _ → z) ≡ f z
        ext-hom : ∀ N {Z} (f : Z → ∣ N ∣) (o : Op) (k : Ar o → Ext Z) →
          ext N f (opP o k) ≡ Nalg N o (λ a → ext N f (k a))
        ext-unique : ∀ N {Z} (f : Z → ∣ N ∣) (h : Ext Z → ∣ N ∣) →
          (∀ o k → h (opP o k) ≡ Nalg N o (λ a → h (k a))) →
          (∀ z → h (ηs , λ _ → z) ≡ f z) → ∀ t → h t ≡ ext N f t

    module WithFree (pf : PolyFree) where
      open PolyFree pf

      fmap : ∀ {X Y} → NatTrans X Y → Hom (PM.polynomial L₁ O₁) (polyAlg X) (polyAlg Y)
      fmap α .Hom.transformation = naturally _ _ λ w → pmap (α .N-ob w)
      fmap α .Hom.preserves = makeNatTransPath (funExt λ w → funExt λ ok →
        opP-nat (α .N-ob w) (ok .fst .fst) λ a → ok .snd (a , tt))

      free : Functor 𝒞 (Mod (Machine.theory L₁ O₁ eqs))
      free .F-ob X = polyAlg X , sat X
      free .F-hom = fmap
      free .F-id = Hom-path (PM.polynomial L₁ O₁) (makeNatTransPath refl)
      free .F-seq f g = Hom-path (PM.polynomial L₁ O₁) (makeNatTransPath refl)

      ηP : (X : 𝒞 .Category.ob) → NatTrans X (Carrier X)
      ηP X = naturally _ _ λ w x → ηs , λ _ → x

      unit : NatTrans 𝟙⟨ 𝒞 ⟩ (modelForgetful (Machine.theory L₁ O₁ eqs) ∘F free)
      unit .N-ob = ηP
      unit .N-hom α = makeNatTransPath refl

      extend : ∀ {X} {N : Model} → NatTrans X (Algebra.Carrier (N .fst)) → Hom (PM.polynomial L₁ O₁) (polyAlg X) (N .fst)
      extend {N = N} f .Hom.transformation = naturally _ _ λ w → ext N (f .N-ob w)
      extend {N = N} f .Hom.preserves = makeNatTransPath (funExt λ w → funExt λ ok →
        ext-hom N (f .N-ob w) (ok .fst .fst) λ a → ok .snd (a , tt))

      module UP = FA.FromUniversalProperty 𝒞 (Mod (Machine.theory L₁ O₁ eqs)) free (modelForgetful (Machine.theory L₁ O₁ eqs)) unit
        (λ {X} {N} f → extend {X} {N} f)
        (λ {X} {N} f → makeNatTransPath (funExt λ w → funExt λ x → ext-η N (f .N-ob w) x))
        (λ {X} {N} f h p → Hom-path (PM.polynomial L₁ O₁) (makeNatTransPath (funExt λ w → funExt λ t →
          ext-unique N (f .N-ob w) (h .Hom.transformation .N-ob w)
            (λ o k i → h .Hom.preserves i .N-ob w ((o , tt) , λ q → k (q .fst)))
            (λ z i → p i .N-ob w z) t)))

      -- the free model, carried by the container's extension
      freeModel : FreeModel (Machine.theory L₁ O₁ eqs)
      freeModel .FreeModel.free = free
      freeModel .FreeModel.adjunction = UP.adjunction

      -- the carrier at the one world IS the configurations
      presentation : PolySelfModule.Presentation (Machine.theory L₁ O₁ eqs) freeModel Q′ tt*
      presentation .PolySelfModule.Presentation.to X c = c
      presentation .PolySelfModule.Presentation.from X t = t
      presentation .PolySelfModule.Presentation.to-from X t = refl
      presentation .PolySelfModule.Presentation.from-to X c = refl
      presentation .PolySelfModule.Presentation.to-nat α c = refl
