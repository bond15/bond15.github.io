-- Canonical forms denote, and are unique in the free model (POPL's
-- EqCanonicity, with its correction C3).
--
-- A canonical form is an operation tree t (Termination.OpTree): nodes
-- op o vs, each continuation behind the locks of its slot, and values of
-- lock telescopes at the leaves.  Its image in the free model of the
-- theory (fm),
--
--   ⌊ t ⌋ : ⟦ Θ ⟧ → T ⟦ A ⟧      leaf V ↦ η ∘ ⟦ V ⟧,  node o vs t ↦ the node of T,
--                                its continuations shut through their words,
--
-- is built from the free model's unit and operations only (POPL's
-- ⌊ t ⌋ = eval ops (η ∘ ⟦ V ⟧)).  At a root ∅ w it is an element of
-- T ⟦ A ⟧ w (⌊ t ⌋₀, at the identity; POPL's environment ρ∅).  The
-- denotation is the free-model one (Semantics.Free.Denotation), sound for
-- the theory (EquationalSoundness) under the module's ec.
--
--   den-reify               ⟦ reify t ⟧ = ⌊ t ⌋
--   canonical-form-denotes  M ≈ reify t  →  ⟦ M ⟧ = ⌊ t ⌋     (and ⌊ t ⌋₀ at a root)
--   canonical-form-unique   M ≈ reify t ,  M ≈ reify t'  →  ⌊ t ⌋ = ⌊ t' ⌋
--   canonicity              on a lock telescope a canonical form exists and
--                           computes the denotation
--
-- C3 (as POPL): uniqueness holds in the free model, not of the tree t
-- itself; with equations two different trees can be ≈ (writer:
-- tell_e (return V) ≈ return V for the unit e), and then ⌊_⌋ identifies
-- them because T ⟦ A ⟧ satisfies the theory.

  (eqs : Machine.Equations L O) (fm : FreeModel (Machine.theory L O eqs))
  (ec : StrongTheory.ExpClosed L O eqs) where

open PshHom

private
  module W = Category W

----------------------------------------------------------------------
-- the image of a canonical form in the free model
-- (⟦ F A ⟧c is T ⟦ A ⟧ .fst by definition; it is written ⟦ F A ⟧c, the
-- form ⟦ op … ⟧C unfolds to, so that den-reify compares syntactically)

⌊_⌋ : ∀ {A Θ} → OpTree A Θ → PshHom ⟦ Θ ⟧t ⟦ F A ⟧u
⌊_⌋ {A} (leaf V) = ⟦ V ⟧V ⋆PshHom η ⟦ A ⟧v
⌊_⌋ {A} {Θ} (node o vs t) =
  opNode o ⟦ vs ⟧Vs (λ a → shutW {Θ} (slot o a) (λ ps → ⌊ t a ps ⌋))
  ⋆PshHom NatTrans→PshHom (Algebra.algebra ⟦ F A ⟧c)

-- at a root, an element of T ⟦ A ⟧ w
⌊_⌋₀ : ∀ {A w} → OpTree A (∅ w) → El ⟦ F A ⟧u w
⌊_⌋₀ {w = w} t = ⌊ t ⌋ .N-ob w W.id

den-reify : ∀ {A Θ} (t : OpTree A Θ) → ⟦ reify t ⟧C ≡ ⌊ t ⌋
den-reify (leaf V) = refl
den-reify {A} {Θ} (node o vs t) =
  cong (λ ks → opNode o ⟦ vs ⟧Vs (λ a → shutW {Θ} (slot o a) (ks a))
               ⋆PshHom NatTrans→PshHom (Algebra.algebra ⟦ F A ⟧c))
    (funExt λ a → funExt λ ps → den-reify (t a ps))

----------------------------------------------------------------------
-- canonical forms denote, and are unique in the free model

canonical-form-denotes : ∀ {A Θ} (M : Comp Θ (F A)) (t : OpTree A Θ) →
  M ≈c reify t → ⟦ M ⟧C ≡ ⌊ t ⌋
canonical-form-denotes M t p = sound-c≡ p ∙ den-reify t

-- at a root: the canonical form computes the closed denotation
canonical-form-denotes₀ : ∀ {A w} (M : Comp (∅ w) (F A)) (t : OpTree A (∅ w)) →
  M ≈c reify t → denClosed M ≡ ⌊ t ⌋₀
canonical-form-denotes₀ {w = w} M t p = cong (λ φ → φ .N-ob w W.id) (canonical-form-denotes M t p)

-- C3: uniqueness in the free model
canonical-form-unique : ∀ {A Θ} (M : Comp Θ (F A)) (t t' : OpTree A Θ) →
  M ≈c reify t → M ≈c reify t' → ⌊ t ⌋ ≡ ⌊ t' ⌋
canonical-form-unique M t t' p p' =
  sym (canonical-form-denotes M t p) ∙ canonical-form-denotes M t' p'

-- on a lock telescope a canonical form exists, and computes the denotation
canonicity : ∀ {Θ A} → IsLT Θ → (M : Comp Θ (F A)) →
  Σ[ t ∈ OpTree A Θ ] (M ≈c reify t) × (⟦ M ⟧C ≡ ⌊ t ⌋)
canonicity l M =
  equational-canonicity l M .fst ,
  equational-canonicity l M .snd ,
  canonical-form-denotes M _ (equational-canonicity l M .snd)
