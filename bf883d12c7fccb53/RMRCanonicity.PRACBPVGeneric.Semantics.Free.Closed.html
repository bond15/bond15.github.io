-- same proofs over Semantics.Free.Denotation, at every computation type B

--                                  naturally in the world

------------------------------------------------------------------------
-- the denotation of a closed computation

denClosed : ∀ {w B} → Comp (∅ w) B → El ⟦ B ⟧u w
denClosed {w} M = ⟦ M ⟧C .N-ob w W.id

-- natural in the world: reindexing is the presheaf action (Lemma 7.1)
denClosed-reindex : ∀ {w w' B} (f : W [ w , w' ]) (M : Comp (∅ w) B) →
  denClosed (reindexC f M) ≡ ⟦ B ⟧u .F-hom f (denClosed M)
denClosed-reindex f M =
  den-renC (root (mapW f)) M _ W.id
  ∙ cong (⟦ M ⟧C .N-ob _) (W.⋆IdR f ∙ sym (W.⋆IdL f))
  ∙ ⟦ M ⟧C .N-hom _ _ f W.id

------------------------------------------------------------------------
-- the syntactic operation node (Lemma 7.2)

-- the constants of the shape, read at the identity
consts-id : ∀ {w} bs (cs : Cs w bs) → ⟦ consts {Θ = ∅ w} cs ⟧Vs .N-ob w W.id ≡ cs
consts-id [] cs = refl
consts-id {w} (b ∷ bs) (c , cs) =
  ΣPathP (funExt⁻ (Syn.Const b .F-id) c , consts-id bs cs)

-- a continuation family that reads the environment only through the root
-- of a closed term: shutW reads it at the position and map E μ (π e)
shutW-root : ∀ {Θ} {X : 𝒱} (μ : List Lk)
  (G : (ps : WPos μ (world Θ)) → PshHom (Y (wscope μ (world Θ) ps)) X)
  (k : (ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X) →
  (∀ ps t y → k ps .N-ob t y ≡ G ps .N-ob t (toW (wsW μ Θ ps) W.⋆ πt (Θ ⧀*⟨ μ , ps ⟩) .N-ob t y)) →
  ∀ v e ps → shutW μ k .N-ob v e ps
    ≡ G (KW.pos μ (πt Θ .N-ob v e) ps) .N-ob _ (KW.smap μ (πt Θ .N-ob v e) ps)
shutW-root [] G k hyp v e ps = hyp _ v e ∙ cong (G _ .N-ob v) (W.⋆IdL _)
shutW-root {Θ} (κ ∷ []) G k hyp v e q =
  hyp _ _ (K.Over.inj κ (πt Θ) e q) ∙ cong (G _ .N-ob _) (W.⋆IdL _)
shutW-root {Θ} (κ ∷ μ@(_ ∷ _)) G k hyp v e (q , ps) =
  shutW-root {Θ ⧀⟨ κ , _ ⟩} μ (λ ps → G (_ , ps)) (λ ps → k (_ , ps)) (λ ps → hyp (_ , ps))
    _ (K.Over.inj κ (πt Θ) e q) ps

-- the operation node denotes the algebra's node, at every type
opNodeC-den : ∀ {w B} (o : Op) (cs : Cs w (params o))
  (ks : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ (wscope (slot o a) w ps)) B) →
  denClosed (opNodeC o cs ks)
    ≡ Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob w ((o , cs) , λ (a , ps) → denClosed (ks a ps))
opNodeC-den {w} {B} o cs ks i =
  Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob w ((o , consts-id (params o) cs i) , λ (a , ps) → arg a ps i)
  where
  rt : ∀ a (ps : WPos (slot o a) w) → Syn.Ren (∅ (wscope (slot o a) w ps)) (∅ w ⧀*⟨ slot o a , ps ⟩)
  rt a ps = root (wsW (slot o a) (∅ w) ps)

  arg : ∀ a ps → shutW (slot o a) (λ ps → ⟦ renC (rt a ps) (ks a ps) ⟧C) .N-ob w W.id ps
               ≡ denClosed (ks a ps)
  arg a ps =
    shutW-root {∅ w} (slot o a) (λ ps → ⟦ ks a ps ⟧C) _
      (λ ps t y → den-renC (rt a ps) (ks a ps) t y) w W.id ps
    ∙ (λ i → ⟦ ks a (wordE (slot o a) .F-id i .fst ps) ⟧C .N-ob _ (wordE (slot o a) .F-id i .snd ps))

------------------------------------------------------------------------
-- closing (Lemmas 7.3, 7.4, Cor 7.5)

-- a closed base value denotes its constant
closeVals-den : ∀ {w bs} (vs : Vals (∅ w) bs) → ⟦ vs ⟧Vs .N-ob w W.id ≡ closeVals vs
closeVals-den []ᵥ = refl
closeVals-den (const {b} c ∷ᵥ vs) = cong₂ _,_ (funExt⁻ (Syn.Const b .F-id) c) (closeVals-den vs)
closeVals-den (ne () ∷ᵥ vs)

-- at the root, the R-introduction's unit is ν at the identity, for any
-- family over the summands (copair-at twice: no path over positions)
inj-ν : ∀ {w} {X : 𝒱} (κ : Lk) (φ : (p : Pos κ w) → PshHom ⟦ ∅ w ⧀⟨ κ , p ⟩ ⟧t X) (q : Pos κ w) →
  φ (K.pos κ W.id q) .N-ob _ (K.Over.inj κ (πt (∅ w)) W.id q)
    ≡ φ q .N-ob _ (K.ν κ q .N-ob _ (W.id W.⋆ W.id))
inj-ν κ φ q =
  sym (K.Over.copair-at κ (πt (∅ _)) φ _ _ _)
  ∙ K.Over.copair-at κ (πt (∅ _)) φ _ _ (K.ν κ q .N-ob _ W.id .snd .snd)
  ∙ cong (λ m → φ q .N-ob _ (K.ν κ q .N-ob _ m)) (sym (W.⋆IdL W.id))

-- closing a word's locks, outermost first, is familial representability
-- at the root
closeK-den : ∀ {w B} (μ : List Lk) (K' : (ps : WPos μ w) → Comp (∅ w ⧀*⟨ μ , ps ⟩) B) (ps : WPos μ w) →
  shutW {∅ w} μ (λ ps → ⟦ K' ps ⟧C) .N-ob w W.id ps ≡ denClosed (closeK μ K' ps)
closeK-den [] K' ps = refl
closeK-den {w} (κ ∷ []) K' q =
  inj-ν κ (λ p → ⟦ K' p ⟧C) q
  ∙ sym (den-subC (close q idW) (K' q) _ W.id)
closeK-den {w} {B} (κ ∷ μ@(κ' ∷ μ')) K' (q , ps) =
  funExt⁻ (inj-ν κ (λ p → shutW {∅ w ⧀⟨ κ , p ⟩} μ (λ ps → ⟦ K' (p , ps) ⟧C)) q) ps
  ∙ sym (funExt⁻ (shutW-sub μ (close q idW) (λ ps → ⟦ K' (q , ps) ⟧C) (λ ps → ⟦ K'' ps ⟧C)
          (λ ps → den-subC (liftWS μ (close q idW) ps) (K' (q , wposW μ idW ps))) _ W.id) ps)
  ∙ closeK-den μ K'' ps
  where
  K'' : (ps : WPos μ (scope κ w q)) → Comp (∅ (scope κ w q) ⧀*⟨ μ , ps ⟩) B
  K'' ps = subC (liftWS μ (close q idW) ps) (K' (q , wposW μ idW ps))

-- the machine's close step does not change the denotation
close-den : ∀ {w B} (o : Op) (vs : Vals (∅ w) (params o))
  (k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) B) →
  denClosed (op o vs k) ≡ denClosed (closeOp o vs k)
close-den {w} {B} o vs k =
  (λ i → Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob w
           ((o , closeVals-den vs i) , λ (a , ps) → closeK-den (slot o a) (k a) ps i))
  ∙ sym (opNodeC-den o (closeVals vs) (λ a → closeK (slot o a) (k a)))
