-- The local-state machine (W = Inj, one lock kind fresh; Instances.LocalState).
--
--   S n = Store n = Fin n → Bool     a store with n cells
--   H = ⊥                            never halts
--   δ σ (lookup , c)    continue in slot σ c, same store
--   δ σ (update b , c)  continue, cell c overwritten with b
--   δ σ (new b)         continue at n + 1 (the scope of [fresh]), the store
--                       extended by a cell initialised to b
--   (the cases for the test operations newIf / new₂ are commented out)
--
-- so Q X = Σ_n Store n × X n, RMR's heap configurations.  No Plotkin-power
-- reassociation is needed: the runner reads the child at n + 1 directly.
--
-- The machine closes new's lock with close tt idW, which turns the name ℓ
-- into the constant suffixFin {n} {1} zero = freshCell n: the cell the
-- runner has just initialised.

module P = SignatureNotation (PM.polynomial lsLocks lsOps)

------------------------------------------------------------------------
-- the runner

isSetLSLk : isSet LSLk
isSetLSLk = isProp→isSet λ { fresh fresh → refl }

isSetLSBase : isSet LSBase
isSetLSBase = isProp→isSet λ { Loc Loc → refl }

isSetConf : isSet (Σ ℕ Store)
isSetConf = isSetΣ isSetℕ isSetStore

isSet⊥ : isSet ⊥
isSet⊥ = isProp→isSet isProp⊥

-- a new cell, initialised to b
init : Bool → Store 1
init b _ = b

-- -- newIf b at cell c: allocate (slot true) only if c holds true
-- newIf-δ : ∀ {n} (b : Bool) (cs : Cs n (Loc ∷ [])) → Store n → Bool →
--   ⊥ ⊎ (Σ[ p ∈ P.Position n (newIf b , cs) ] Store (P.scope n (newIf b , cs) p))
-- newIf-δ b cs σ true = inr ((true , tt) , σ ++Fin init b)
-- newIf-δ b cs σ false = inr ((false , tt) , σ)

δ : ∀ {n} → Store n → (o : P.Shape n) → ⊥ ⊎ (Σ[ p ∈ P.Position n o ] Store (P.scope n o p))
δ σ (lookup , (c , tt)) = inr ((σ c , tt) , σ)
δ σ (update b , (c , tt)) = inr ((tt , tt) , updateStore c b σ)
δ σ (new b , tt) = inr ((tt , tt) , σ ++Fin init b)
-- δ σ (newIf b , (c , tt)) = newIf-δ b (c , tt) σ (σ c)
-- δ σ (new₂ b , tt) = inr ((tt , (tt , tt)) , (σ ++Fin init b) ++Fin init b)

-- programs returning a location

module ReturnsLoc where
  open RunMachine lsLocks lsOps isSetLSLk isSetLSBase Store ⊥ isSetConf isSet⊥ δ 𝕃

  -- new ℓ := false; ℓ := true; return ℓ
  allocWrite : Comp (∅ 0) (F 𝕃)
  allocWrite = newℓ false (write ℓ true (return ℓ))

  -- the store after the run: one cell, overwritten with true
  σ₁ : Store 1
  σ₁ = updateStore (freshCell 0) true (emptyStore ++Fin init false)

  -- close new (ℓ becomes the fresh cell), allocate; close the write, write
  alloc-write : running {0} emptyStore allocWrite ⟶* running {1} σ₁ (return (cell (freshCell 0)))
  alloc-write = op-at then* op-at

  alloc-write-sound : run emptyStore (denClosed allocWrite) ≡ Rn.running ⟦ 𝕃 ⟧v σ₁ (freshCell 0)
  alloc-write-sound = run-return alloc-write
