-- The syntactic metatheory (syntax only: no Semantics module is imported,
-- except that the equational theory takes a theory over the polynomial,
-- Semantics.Machine.Equations, as its parameter):
--
--   TypesSet     ValTy and CompTy are sets
--   TermsSet     bs ∋ b, Ne, Val, Vals and Comp are sets (Thm 8.1)
--   Renaming     renaming identity, fusion and extensionality over paths
--                of telescopes (Thms 8.2-8.4)
--   Substitution substituting a renaming, identity, the three fusions and
--                extensionality of substitution (Thms 5.1-5.5), and the
--                one-variable and one-lock laws (5.6)
--   SubstReduction  tree reduction is stable under substitution and
--                renaming (5.6)
--   Determinism  tree reduction is deterministic; return and op are
--                normal (6.1)
--   Chain        formal composites of substitutions and their laws (6.3)
--   LogicalRelation  lock telescopes, the Kripke logical relation and its
--                closure properties (6.2, 6.4, Lemma 6.5)
--   Fundamental  the compatibility lemmas and the fundamental lemma
--                (Thm 6.6)
--   Termination  every computation of type F A on a lock telescope reduces
--                to an operation tree (Cor 6.7, Thm 6.8)
--   TreeNormalization  strong normalization of tree reduction: every
--                computation of type F A on a lock telescope is accessible
--                for the converse of ↦ (parity stage G)
--   ClosedLaws   closedLaws : Closing.ClosedLaws (Cor 8.5)
--   TreeSize     the height of a derivation of the logical relation: unique,
--                not raised by substitution or by closing locks; the size
--                sz of a closed computation (Stage C)
--   Equational   the equational theory ≈ over a theory: equivalence,
--                congruence, β and η (stack-form F-η, the lock modality),
--                algebraicity of operations, the theory's equations
--                (parity stage F)
--   EqSubstitution  ≈ is closed under substitution and chains
--   EqLogicalRelation  the equational logical relation (≈-closed), its
--                fundamental theorem, and equational canonicity
--
-- Every module that needs a set of types takes isSet Lk and isSet Base as
-- hypotheses (LockSig does not assume them).
module RMRCanonicity.PRACBPVGeneric.Metatheory.Everything where

open import Cubical.Foundations.Prelude using (isSet; isProp→isSet; refl)

import RMRCanonicity.PRACBPVGeneric.Metatheory.TypesSet

import RMRCanonicity.PRACBPVGeneric.Metatheory.Determinism
import RMRCanonicity.PRACBPVGeneric.Metatheory.Chain
import RMRCanonicity.PRACBPVGeneric.Metatheory.LogicalRelation
import RMRCanonicity.PRACBPVGeneric.Metatheory.Fundamental
import RMRCanonicity.PRACBPVGeneric.Metatheory.Termination
import RMRCanonicity.PRACBPVGeneric.Metatheory.TreeNormalization

-- a test instantiation: local state (both sets have one element)
import RMRCanonicity.PRACBPVGeneric.Closing as Closing
open import RMRCanonicity.PRACBPVGeneric.Instances.LocalState using (LSLk; fresh; LSBase; Loc; lsLocks; lsOps)

private
  isSetLSLk : isSet LSLk
  isSetLSLk = isProp→isSet λ { fresh fresh → refl }

  isSetLSBase : isSet LSBase
  isSetLSBase = isProp→isSet λ { Loc Loc → refl }

  lsClosedLaws : Closing.ClosedLaws lsLocks lsOps
  lsClosedLaws = RMRCanonicity.PRACBPVGeneric.Metatheory.ClosedLaws.closedLaws lsLocks lsOps isSetLSLk isSetLSBase
