{-# OPTIONS --no-allow-unsolved-metas --no-lossy-unification #-}

module RightModuleReduction.Instances.Terminal where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.HLevels
open import Cubical.Categories.Category using (Category; _^op)
open import Cubical.Categories.Functor using (Functor; _⟅_⟆)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Presheaf.Base using (PresheafCategory)
open import RightModuleReduction.Instances.Theories using (One)

open Category
open Functor
open NatTrans

-- Covariant families on the terminal category.
𝒱 : Category (ℓ-suc ℓ-zero) ℓ-zero
𝒱 = PresheafCategory (One ^op) ℓ-zero

-- On the terminal category, component maps automatically form a natural map.
naturally : (X Y : 𝒱 .ob) →
  ((w : One .ob) → (X ⟅ w ⟆) .fst → (Y ⟅ w ⟆) .fst) → NatTrans X Y
naturally X Y f .N-ob = f
naturally X Y f .N-hom {x = w} p = J
  (λ w′ p → (λ x → f w′ (X .F-hom p x)) ≡
    (λ x → Y .F-hom p (f w x)))
  (funExt λ x → cong (f w) (funExt⁻ (X .F-id) x)
    ∙ sym (funExt⁻ (Y .F-id) (f w x))) p
