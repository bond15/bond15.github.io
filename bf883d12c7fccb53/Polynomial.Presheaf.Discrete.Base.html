{-# OPTIONS --no-allow-unsolved-metas --no-lossy-unification #-}

module Polynomial.Presheaf.Discrete.Base where

-- Polynomials with arbitrary set-valued positions and arbitrary position maps.
-- This construction is independent of any particular local-state monad.

open import Cubical.Foundations.Prelude hiding (I ; J)
open import Cubical.Foundations.HLevels
open import Cubical.Data.Sigma
open import Cubical.Categories.Category
open import Cubical.Categories.Functor
open import Cubical.Categories.NaturalTransformation
open import Cubical.Categories.Instances.Sets
open import Cubical.Categories.Instances.Opposite
open import Cubical.Categories.Instances.Elements
import Cubical.Categories.Instances.TotalCategory as TotalCategory
open import Cubical.Categories.Presheaf.Base
open import Cubical.Categories.Presheaf.Constructions.Reindex using (reindPshF)

open Category
open Functor
open NatTrans

-- A polynomial I <- E -> B -> J with discrete fibres.
-- The category-of-elements presentations
-- supply the chosen discrete (op)fibration structure.  Bijective transport
-- is not used by this extension construction, only by FinBijPoly's hole API.
record Poly
    {ℓI ℓI' ℓJ ℓJ' ℓShape ℓPos : Level}
    (I : Category ℓI ℓI')
    (J : Category ℓJ ℓJ')
    : Type
        (ℓ-max
          (ℓ-max (ℓ-max ℓI ℓI') (ℓ-max ℓJ ℓJ'))
          (ℓ-suc (ℓ-max ℓShape ℓPos)))
    where

  field
    Shapes : Functor (J ^op) (SET ℓShape)

  -- Shapes and their projection to the output category.
  B : Category _ _
  B = Contravariant.∫ Shapes

  t : Functor B J
  t = TotalCategory.Fst

  field
    Pos : Functor B (SET ℓPos)

  -- Positions and their projection to shapes.
  PositionFamily : Functor B (SET ℓPos)
  PositionFamily = Pos

  E : Category _ _
  E = Covariant.∫ PositionFamily

  p : Functor E B
  p = Covariant.ForgetElements PositionFamily

  field
    s : Functor E I

  Positions : B .ob → Type ℓPos
  Positions b = Pos .F-ob b .fst

  transportPosition :
    {b b' : B .ob} → B [ b , b' ] → Positions b → Positions b'
  transportPosition f = Pos .F-hom f

  transportPosition-id : {b : B .ob} →
    transportPosition (B .id {b}) ≡ (λ x → x)
  transportPosition-id = Pos .F-id

  transportPosition-seq :
    {b b' b'' : B .ob} (f : B [ b , b' ]) (g : B [ b' , b'' ]) →
    transportPosition ((B ⋆ f) g) ≡
    (λ x → transportPosition g (transportPosition f x))
  transportPosition-seq f g = Pos .F-seq f g

  private
    liftPosition : ∀ {b b' : B .ob} (f : B [ b , b' ])
      (e : Positions b) →
      E [ (b , e) , (b' , transportPosition f e) ]
    liftPosition f e = f , refl

  ----------------------------------------------------------------------
  -- Direct extension: Sigma_t Pi_p Delta_s
  --
  -- Since p is presented as a split discrete opfibration, Pi_p is the
  -- honest dependent product over Positions b.  Since t is presented as a
  -- split discrete fibration, Sigma_t is the honest dependent sum over
  -- Shapes j.  No end, coend, Ran, Lan, or quotient occurs below.

  Δs : ∀ {ℓX} →
    Functor (PresheafCategory I ℓX) (PresheafCategory E ℓX)
  Δs = reindPshF s

  Πp-ob : ∀ {ℓX} →
    PresheafCategory E ℓX .ob →
    PresheafCategory B (ℓ-max ℓPos ℓX) .ob
  Πp-ob X .F-ob b .fst =
    (e : Positions b) → X .F-ob (b , e) .fst
  Πp-ob X .F-ob b .snd =
    isSetΠ (λ e → X .F-ob (b , e) .snd)
  Πp-ob X .F-hom f values e =
    X .F-hom (liftPosition f e)
      (values (transportPosition f e))
  Πp-ob X .F-id {b} = funExt λ values → funExt λ e →
    deform values e
    ∙ cong
        (λ h → X .F-hom h (values e))
        (Covariant.ElementHom≡ PositionFamily refl)
    ∙ funExt⁻ (X .F-id) (values e)
    where
    positionPath : (e : Positions b) →
      transportPosition (B .id) e ≡ e
    positionPath e = funExt⁻ transportPosition-id e

    deform : (values : (e : Positions b) → X .F-ob (b , e) .fst)
      (e : Positions b) →
      X .F-hom (liftPosition (B .id) e)
          (values (transportPosition (B .id) e))
      ≡ X .F-hom (B .id , positionPath e) (values e)
    deform values e k =
      X .F-hom
        (B .id , (λ i → positionPath e (i ∧ k)))
        (values (positionPath e k))

  Πp-ob X .F-seq {b} {b'} {b''} f g =
    funExt λ values → funExt λ e →
      deform values e
      ∙ cong
          (λ h → X .F-hom h
            (values (transportPosition f (transportPosition g e))))
          (Covariant.ElementHom≡ PositionFamily refl)
      ∙ funExt⁻
          (X .F-seq (liftPosition f (transportPosition g e))
            (liftPosition g e))
          (values (transportPosition f (transportPosition g e)))
    where
    positionPath : (e : Positions b'') →
      transportPosition ((B ⋆ g) f) e ≡
      transportPosition f (transportPosition g e)
    positionPath e = funExt⁻ (transportPosition-seq g f) e

    deform :
      (values : (e : Positions b) → X .F-ob (b , e) .fst)
      (e : Positions b'') →
      X .F-hom (liftPosition ((B ⋆ g) f) e)
        (values (transportPosition ((B ⋆ g) f) e))
      ≡ X .F-hom ((B ⋆ g) f , positionPath e)
        (values (transportPosition f (transportPosition g e)))
    deform values e k =
      X .F-hom
        ((B ⋆ g) f , (λ i → positionPath e (i ∧ k)))
        (values (positionPath e k))

  Πp-hom : ∀ {ℓX} {X Y : PresheafCategory E ℓX .ob} →
    NatTrans X Y → NatTrans (Πp-ob X) (Πp-ob Y)
  Πp-hom α .N-ob b values e = α .N-ob (b , e) (values e)
  Πp-hom α .N-hom f = funExt λ values → funExt λ e →
    funExt⁻ (α .N-hom (liftPosition f e))
      (values (transportPosition f e))

  Πp : ∀ {ℓX} →
    Functor (PresheafCategory E ℓX)
      (PresheafCategory B (ℓ-max ℓPos ℓX))
  Πp .F-ob = Πp-ob
  Πp .F-hom = Πp-hom
  Πp .F-id = makeNatTransPath refl
  Πp .F-seq α β = makeNatTransPath refl

  ShapesAt : J .ob → Type ℓShape
  ShapesAt j = Shapes .F-ob j .fst

  reindexShape : {j j' : J .ob} →
    J [ j , j' ] → ShapesAt j' → ShapesAt j
  reindexShape f = Shapes .F-hom f

  reindexShape-id : {j : J .ob} →
    reindexShape (J .id {j}) ≡ (λ b → b)
  reindexShape-id = Shapes .F-id

  reindexShape-seq :
    {j j' j'' : J .ob} (f : J [ j , j' ]) (g : J [ j' , j'' ]) →
    reindexShape ((J ⋆ f) g) ≡
    (λ b → reindexShape f (reindexShape g b))
  reindexShape-seq f g = Shapes .F-seq g f

  private
    liftShape : ∀ {j j' : J .ob} (f : J [ j , j' ])
      (b : ShapesAt j') →
      B [ (j , reindexShape f b) , (j' , b) ]
    liftShape f b = f , refl

  Σt-ob : ∀ {ℓX} →
    PresheafCategory B ℓX .ob →
    PresheafCategory J (ℓ-max ℓShape ℓX) .ob
  Σt-ob X .F-ob j .fst =
    Σ[ b ∈ ShapesAt j ] X .F-ob (j , b) .fst
  Σt-ob X .F-ob j .snd =
    isSetΣ (Shapes .F-ob j .snd)
      (λ b → X .F-ob (j , b) .snd)
  Σt-ob X .F-hom f (b , value) =
    reindexShape f b , X .F-hom (liftShape f b) value
  Σt-ob X .F-id {j} = funExt λ (b , value) →
    deform b value
    ∙ ΣPathP (refl , funExt⁻ (X .F-id) value)
    where
    shapePath : (b : ShapesAt j) → reindexShape (J .id) b ≡ b
    shapePath b = funExt⁻ reindexShape-id b

    objectPath : (b : ShapesAt j) →
      (j , reindexShape (J .id) b) ≡ (j , b)
    objectPath b = ΣPathP (refl , shapePath b)

    homPath : (b : ShapesAt j) →
      PathP
        (λ k → B [ objectPath b k , (j , b) ])
        (liftShape (J .id) b)
        (B .id)
    homPath b =
      Contravariant.ElementHomPathP Shapes
        (liftShape (J .id) b) (B .id)
        (objectPath b) refl refl

    deform : (b : ShapesAt j) (value : X .F-ob (j , b) .fst) →
      ( reindexShape (J .id) b
      , X .F-hom (liftShape (J .id) b) value )
      ≡ (b , X .F-hom (B .id) value)
    deform b value =
      ΣPathP
        ( shapePath b
        , (λ k → X .F-hom (homPath b k) value) )
  Σt-ob X .F-seq {x} {y} {z} f g =
    funExt λ (b , value) →
      deform b value
      ∙ ΣPathP
          ( refl
          , funExt⁻
              (X .F-seq
                (liftShape f b)
                (liftShape g (reindexShape f b)))
              value )
    where
    shapePath : (b : ShapesAt x) →
      reindexShape ((J ⋆ g) f) b ≡
      reindexShape g (reindexShape f b)
    shapePath b = funExt⁻ (reindexShape-seq g f) b

    objectPath : (b : ShapesAt x) →
      (z , reindexShape ((J ⋆ g) f) b) ≡
      (z , reindexShape g (reindexShape f b))
    objectPath b = ΣPathP (refl , shapePath b)

    compositeLift : (b : ShapesAt x) →
      B [ (z , reindexShape g (reindexShape f b)) , (x , b) ]
    compositeLift b =
      (B ⋆ liftShape g (reindexShape f b)) (liftShape f b)

    homPath : (b : ShapesAt x) →
      PathP
        (λ k → B [ objectPath b k , (x , b) ])
        (liftShape ((J ⋆ g) f) b)
        (compositeLift b)
    homPath b =
      Contravariant.ElementHomPathP Shapes
        (liftShape ((J ⋆ g) f) b) (compositeLift b)
        (objectPath b) refl refl

    deform : (b : ShapesAt x) (value : X .F-ob (x , b) .fst) →
      ( reindexShape ((J ⋆ g) f) b
      , X .F-hom (liftShape ((J ⋆ g) f) b) value )
      ≡
      ( reindexShape g (reindexShape f b)
      , X .F-hom (compositeLift b) value )
    deform b value =
      ΣPathP
        ( shapePath b
        , (λ k → X .F-hom (homPath b k) value) )

  Σt-hom : ∀ {ℓX} {X Y : PresheafCategory B ℓX .ob} →
    NatTrans X Y → NatTrans (Σt-ob X) (Σt-ob Y)
  Σt-hom α .N-ob j (b , value) =
    b , α .N-ob (j , b) value
  Σt-hom α .N-hom f = funExt λ (b , value) →
    ΣPathP
      ( refl
      , funExt⁻ (α .N-hom (liftShape f b)) value )

  Σt : ∀ {ℓX} →
    Functor (PresheafCategory B ℓX)
      (PresheafCategory J (ℓ-max ℓShape ℓX))
  Σt .F-ob = Σt-ob
  Σt .F-hom = Σt-hom
  Σt .F-id = makeNatTransPath refl
  Σt .F-seq α β = makeNatTransPath refl

  Extension : ∀ {ℓX} →
    Functor (PresheafCategory I ℓX)
      (PresheafCategory J
        (ℓ-max ℓShape (ℓ-max ℓPos ℓX)))
  Extension = (Σt ∘F Πp) ∘F Δs
