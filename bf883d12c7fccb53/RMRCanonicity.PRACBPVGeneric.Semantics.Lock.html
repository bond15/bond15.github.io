-- The lock of a strong familial arity E (with strength ι), taken literally:
-- the left adjoint L ⊣ R of
--
--   R X w = Π_{q ∈ Pos w} X (scope w q)
--
-- is the coend  L Γ = ∫^v Γ v × E v,  a SET QUOTIENT: at t its elements are
--
--   ⟪ v , q , g , e ⟫      q ∈ Pos v,  g : scope v q → t,  e ∈ Γ v
--   ⟪ v , q , g , Γ f e ⟫ = ⟪ v' , pos f q , smap f q ⋆ g , e ⟫     (f : v' → v)
--
-- (L = Lan_scope ∘ π^* for scope , π : ∫ Pos → W, and R = Ran_π ∘ scope^*).
--
-- Everything else is the adjunction.  The two transposes are
--   unlock φ    : L Γ → X    for φ : Γ → R X     ⟪ v , q , g , e ⟫ ↦ X g (φ e q)
--   transpose ψ : Γ → R X    for ψ : L Γ → X     e ↦ λ q → ψ ⟪ v , q , id , e ⟫
-- and every map OUT of a lock that the syntax uses is an unlock:
--   ε = unlock box       the strength (box e q = Γ (ι v q) e)        ↔ up
--   unlock φ             the counit composed with L φ                 ↔ opn
--   unlock names         the generic elements                         ↔ nm
--
-- Over a world.  If Γ lies over y w (π : Γ → y w), then L Γ lies over
-- L (y w) ≅ E w = Σ_p y (scope w p) (familial representability), via Lπ.
-- The p-SUMMAND of L Γ is the pullback of Lπ along the p-th injection
--     Summand p t = { (x , k) ∈ L Γ t × W [ scope w p , t ] | Lπ x = (p , k) },
-- which lies over y (scope w p) by the second projection.  L Γ is the sum
-- of its summands: maps out of it are families of maps out of the summands
-- (copair), and shut = transpose ∘ copair is R's introduction rule.
--
-- What is proved (the math spec's Props 1.1–1.5):
--   1.1 L ⊣ R as a hom-set bijection, natural in Γ:
--       unlock-transpose, transpose-unlock, unlock-Lmap
--   1.2 L Γ is the sum of its summands: copair-β, and copair-at (any witness)
--   1.3 R's introduction is shut = transpose ∘ copair (shut-is), with β open-shut, η shut-open
--   1.4 familial representability on summands: ν and πS inverse (πS-ν by refl, ν-πS)
--   1.5 structural maps: unlock-box, ε-over, ε-Lmap, Lsum over E f (Lπ-Lmap), unlock-Lmap-at

module RMRCanonicity.PRACBPVGeneric.Semantics.Lock
  {W : Category ℓ-zero ℓ-zero} (E : Functor (W ^op) (Fam W)) (str : NatTrans E (Yo W)) where

open import RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf W

open Functor
open PshHom
open Arity W E public using (Pos; isSetPos; scope; pos; smap)

private
  module W = Category W

------------------------------------------------------------------------
-- the arity's laws, at a position

ι : ∀ w (p : Pos w) → W [ w , scope w p ]
ι = ιOf W str

pos-id : ∀ {w} (p : Pos w) → pos W.id p ≡ p
pos-id p i = E .F-id i .fst p

smap-id : ∀ {w} (p : Pos w) →
  PathP (λ i → W [ scope w (pos-id p i) , scope w p ]) (smap W.id p) W.id
smap-id p i = E .F-id i .snd p

pos-seq : ∀ {u v w} (h : W [ u , v ]) (f : W [ v , w ]) (q : Pos w) →
  pos (h W.⋆ f) q ≡ pos h (pos f q)
pos-seq h f q i = E .F-seq f h i .fst q

smap-seq : ∀ {u v w} (h : W [ u , v ]) (f : W [ v , w ]) (q : Pos w) →
  PathP (λ i → W [ scope u (pos-seq h f q i) , scope w q ])
    (smap (h W.⋆ f) q) (smap h (pos f q) W.⋆ smap f q)
smap-seq h f q i = E .F-seq f h i .snd q

-- naturality of the strength
ι-nat : ∀ {w w'} (f : W [ w , w' ]) (p : Pos w') →
  ι w (pos f p) W.⋆ smap f p ≡ f W.⋆ ι w' p
ι-nat f p i = str .NatTrans.N-hom f i .snd p

------------------------------------------------------------------------
-- the right adjoint

R : 𝒱 → 𝒱
R X .F-ob w = ((p : Pos w) → El X (scope w p)) , isSetΠ λ p → X .F-ob _ .snd
R X .F-hom f r p = X .F-hom (smap f p) (r (pos f p))
R X .F-id = funExt λ r → funExt λ p →
  (λ i → X .F-hom (smap-id p i) (r (pos-id p i))) ∙ funExt⁻ (X .F-id) (r p)
R X .F-seq f g = funExt λ r → funExt λ p →
  (λ i → X .F-hom (smap-seq f g p i) (r (pos-seq f g p i)))
  ∙ funExt⁻ (X .F-seq _ _) _

Rmap : ∀ {X X'} → PshHom X X' → PshHom (R X) (R X')
Rmap α .N-ob w r p = α .N-ob _ (r p)
Rmap α .N-hom c c' f r = funExt λ p → α .N-hom _ _ (smap f p) (r (pos f p))

------------------------------------------------------------------------
-- the lock: a coend

module _ (Γ : 𝒱) where
  Raw : W.ob → Type
  Raw t = Σ[ v ∈ W.ob ] Σ[ q ∈ Pos v ] (W [ scope v q , t ] × El Γ v)

  data Shift (t : W.ob) : Raw t → Raw t → Type where
    shift : ∀ {v v'} (f : W [ v' , v ]) (q : Pos v) (g : W [ scope v q , t ]) (e : El Γ v') →
      Shift t (v , q , g , Γ .F-hom f e) (v' , pos f q , smap f q W.⋆ g , e)

  -- the elements of L Γ at t (LEl, not to clash with the syntax's lock kinds)
  LEl : W.ob → Type
  LEl t = Raw t / Shift t

  -- a point of the lock, with its type fixed (for use under cong)
  pt : ∀ {t} → Raw t → LEl t
  pt = ⟪_⟫

  private
    mapL : ∀ {t t'} → W [ t , t' ] → LEl t → LEl t'
    mapL h = rec squash/ (λ (v , q , g , e) → ⟪ v , q , g W.⋆ h , e ⟫)
      λ { _ _ (shift {v' = v'} f q g e) →
        eq/ _ _ (shift f q (g W.⋆ h) e)
        ∙ cong (λ m → pt (v' , pos f q , m , e)) (sym (W.⋆Assoc (smap f q) g h)) }

  L : 𝒱
  L .F-ob t = LEl t , squash/
  L .F-hom = mapL
  L .F-id = funExt (elimProp (λ _ → squash/ _ _) λ (v , q , g , e) →
    cong (λ m → pt (v , q , m , e)) (W.⋆IdR g))
  L .F-seq f g = funExt (elimProp (λ _ → squash/ _ _) λ (v , q , k , e) →
    cong (λ m → pt (v , q , m , e)) (sym (W.⋆Assoc k f g)))

-- L is a functor
Lmap : ∀ {Γ Δ} → PshHom Γ Δ → PshHom (L Γ) (L Δ)
Lmap α .N-ob t = rec squash/ (λ (v , q , g , e) → ⟪ v , q , g , α .N-ob v e ⟫)
  λ { _ _ (shift {v = v} f q g e) →
    cong (λ x → pt _ (v , q , g , x)) (α .N-hom _ _ f e) ∙ eq/ _ _ (shift f q g (α .N-ob _ e)) }
Lmap α .N-hom c c' h = elimProp (λ _ → squash/ _ _) λ _ → refl

------------------------------------------------------------------------
-- the adjunction L ⊣ R, by its two transposes

module _ {Γ X : 𝒱} where
  unlock : PshHom Γ (R X) → PshHom (L Γ) X
  unlock φ .N-ob t = rec (X .F-ob t .snd) (λ (v , q , g , e) → X .F-hom g (φ .N-ob v e q))
    λ { _ _ (shift f q g e) →
      cong (X .F-hom g) (funExt⁻ (φ .N-hom _ _ f e) q)
      ∙ sym (funExt⁻ (X .F-seq (smap f q) g) _) }
  unlock φ .N-hom c c' h = elimProp (λ _ → X .F-ob c .snd _ _) λ (v , q , g , e) →
    funExt⁻ (X .F-seq g h) _

  transpose : PshHom (L Γ) X → PshHom Γ (R X)
  transpose ψ .N-ob v e q = ψ .N-ob (scope v q) ⟪ v , q , W.id , e ⟫
  transpose ψ .N-hom c c' f e = funExt λ q →
    cong (ψ .N-ob _) (eq/ _ _ (shift f q W.id e))
    ∙ cong (λ m → ψ .N-ob _ (pt Γ (c' , pos f q , m , e))) (W.⋆IdR _ ∙ sym (W.⋆IdL _))
    ∙ ψ .N-hom _ _ (smap f q) ⟪ c' , pos f q , W.id , e ⟫

  -- the two transposes are inverse
  unlock-transpose : (ψ : PshHom (L Γ) X) → unlock (transpose ψ) ≡ ψ
  unlock-transpose ψ = makePshHomPath (funExt λ t → funExt (elimProp (λ _ → X .F-ob t .snd _ _)
    λ (v , q , g , e) →
      sym (ψ .N-hom _ _ g ⟪ v , q , W.id , e ⟫)
      ∙ cong (λ m → ψ .N-ob t (pt Γ (v , q , m , e))) (W.⋆IdL g)))

  transpose-unlock : (φ : PshHom Γ (R X)) → transpose (unlock φ) ≡ φ
  transpose-unlock φ = makePshHomPath (funExt λ v → funExt λ e → funExt λ q →
    funExt⁻ (X .F-id) (φ .N-ob v e q))

-- unlock is natural in Γ: precomposing L α is precomposing α
unlock-Lmap : ∀ {Γ Δ X} (α : PshHom Γ Δ) (φ : PshHom Δ (R X)) →
  Lmap α ⋆PshHom unlock φ ≡ unlock (α ⋆PshHom φ)
unlock-Lmap {X = X} α φ = makePshHomPath (funExt λ t → funExt (elimProp
  (λ _ → X .F-ob t .snd _ _) λ _ → refl))

-- the unit of [ κ ] induced by the strength, and its transpose: the
-- strength of the lock, L Γ → Γ (Kripke access across the lock)
box : (Γ : 𝒱) → PshHom Γ (R Γ)
box Γ .N-ob v e q = Γ .F-hom (ι v q) e
box Γ .N-hom c c' f e = funExt λ q →
  sym (funExt⁻ (Γ .F-seq f (ι c q)) e)
  ∙ cong (λ m → Γ .F-hom m e) (sym (ι-nat f q))
  ∙ funExt⁻ (Γ .F-seq _ _) e

ε : (Γ : 𝒱) → PshHom (L Γ) Γ
ε Γ = unlock (box Γ)

------------------------------------------------------------------------
-- over a world: familial representability and the summands

-- E w as a presheaf:  Σ_{p ∈ Pos w} y (scope w p)
Ey : W.ob → 𝒱
Ey w .F-ob t = (Σ[ p ∈ Pos w ] W [ scope w p , t ]) , isSetΣ (isSetPos w) λ _ → W.isSetHom
Ey w .F-hom h (p , k) = p , k W.⋆ h
Ey w .F-id = funExt λ (p , k) → cong (p ,_) (W.⋆IdR k)
Ey w .F-seq f g = funExt λ (p , k) → cong (p ,_) (sym (W.⋆Assoc k f g))

-- the generic element y w → R (E w), whose transpose L (y w) → E w is the
-- familial-representability iso
genE : (w : W.ob) → PshHom (Y w) (R (Ey w))
genE w .N-ob v h q = pos h q , smap h q
genE w .N-hom c c' f h = funExt λ q → ΣPathP (pos-seq h f q , smap-seq h f q)

module Over {w : W.ob} {Γ : 𝒱} (π : PshHom Γ (Y w)) where
  -- L Γ over E w
  Lπ : PshHom (L Γ) (Ey w)
  Lπ = unlock (π ⋆PshHom genE w)

  -- the p-summand: the pullback of Lπ along the p-th injection y (scope w p) → E w
  SumEl : Pos w → W.ob → Type
  SumEl p t = Σ[ x ∈ El (L Γ) t ] Σ[ k ∈ W [ scope w p , t ] ] (Lπ .N-ob t x ≡ (p , k))

  -- elements of a summand are determined by their first two components
  summand≡ : ∀ {p t} {x x' : El (L Γ) t} {k k' : W [ scope w p , t ]}
    {r : Lπ .N-ob t x ≡ (p , k)} {r' : Lπ .N-ob t x' ≡ (p , k')} →
    x ≡ x' → k ≡ k' → Path (SumEl p t) (x , k , r) (x' , k' , r')
  summand≡ {t = t} px pk = ΣPathP (px , ΣPathP (pk ,
    isProp→PathP (λ i → Ey w .F-ob t .snd _ _) _ _))

  Summand : Pos w → 𝒱
  Summand p .F-ob t = SumEl p t ,
    isSetΣ squash/ λ x → isSetΣ W.isSetHom λ k → isProp→isSet (Ey w .F-ob t .snd _ _)
  Summand p .F-hom h (x , k , r) =
    L Γ .F-hom h x , k W.⋆ h , Lπ .N-hom _ _ h x ∙ cong (Ey w .F-hom h) r
  Summand p .F-id = funExt λ (x , k , r) →
    summand≡ (funExt⁻ (L Γ .F-id) x) (W.⋆IdR k)
  Summand p .F-seq f g = funExt λ (x , k , r) →
    summand≡ (funExt⁻ (L Γ .F-seq f g) x) (sym (W.⋆Assoc k f g))

  inc : ∀ p → PshHom (Summand p) (L Γ)
  inc p .N-ob t s = s .fst
  inc p .N-hom c c' h s = refl

  -- the summand lies over y (scope w p)
  πS : ∀ p → PshHom (Summand p) (Y (scope w p))
  πS p .N-ob t s = s .snd .fst
  πS p .N-hom c c' h s = refl

  -- L Γ is the sum of its summands: a family of maps out of the summands
  -- is a map out of L Γ (an element is sent by the map of its summand)
  module _ {X : 𝒱} (φ : (p : Pos w) → PshHom (Summand p) X) where
    copair-ob : ∀ t → El (L Γ) t → El X t
    copair-ob t x = φ (Lπ .N-ob t x .fst) .N-ob t (x , Lπ .N-ob t x .snd , refl)

    -- at any element, with any witness of its summand
    copair-at : ∀ {t} (x : El (L Γ) t) (pk : El (Ey w) t) (r : Lπ .N-ob t x ≡ pk) →
      copair-ob t x ≡ φ (pk .fst) .N-ob t (x , pk .snd , r)
    copair-at x pk r i = φ (r i .fst) .N-ob _ (x , r i .snd , λ j → r (i ∧ j))

    copair : PshHom (L Γ) X
    copair .N-ob = copair-ob
    copair .N-hom c c' h x =
      copair-at (L Γ .F-hom h x) (Ey w .F-hom h (Lπ .N-ob c' x))
        (Lπ .N-hom _ _ h x ∙ cong (Ey w .F-hom h) refl)
      ∙ φ _ .N-hom c c' h (x , Lπ .N-ob c' x .snd , refl)

  -- the copairing restricted to the p-summand is the p-th map
  copair-β : ∀ {X} (φ : (p : Pos w) → PshHom (Summand p) X) p →
    inc p ⋆PshHom copair φ ≡ φ p
  copair-β φ p = makePshHomPath (funExt λ t → funExt λ (x , k , r) →
    copair-at φ x (p , k) r)

  -- the unit at a position: the environment e, put behind the lock at q
  -- (it lands in the summand of the position q pulls back to)
  inj : ∀ {v} (e : El Γ v) (q : Pos v) → El (Summand (pos (π .N-ob v e) q)) (scope v q)
  inj {v} e q = ⟪ v , q , W.id , e ⟫ , smap (π .N-ob v e) q , cong (_ ,_) (W.⋆IdR _)

  -- R's introduction rule: a map out of every summand
  shut : ∀ {X} → ((p : Pos w) → PshHom (Summand p) X) → PshHom Γ (R X)
  shut φ .N-ob v e q = φ (pos (π .N-ob v e) q) .N-ob _ (inj e q)
  shut {X} φ .N-hom c c' f e = funExt λ q →
    shut-transpose (Γ .F-hom f e) q
    ∙ funExt⁻ (transpose (copair φ) .N-hom c c' f e) q
    ∙ cong (X .F-hom (smap f q)) (sym (shut-transpose e (pos f q)))
    where
    shut-transpose : ∀ {v} (e : El Γ v) q → φ _ .N-ob _ (inj e q) ≡ copair φ .N-ob _ ⟪ v , q , W.id , e ⟫
    shut-transpose e q = sym (copair-at φ _ _ _)

  -- shut is transpose ∘ copair
  shut-is : ∀ {X} (φ : (p : Pos w) → PshHom (Summand p) X) → shut φ ≡ transpose (copair φ)
  shut-is φ = makePshHomPath (funExt λ v → funExt λ e → funExt λ q → sym (copair-at φ _ _ _))

  -- β: opening a shut value at p is its p-th component
  open-shut : ∀ {X} (φ : (p : Pos w) → PshHom (Summand p) X) p →
    inc p ⋆PshHom unlock (shut φ) ≡ φ p
  open-shut φ p =
    cong (λ s → inc p ⋆PshHom unlock s) (shut-is φ)
    ∙ cong (inc p ⋆PshHom_) (unlock-transpose (copair φ))
    ∙ copair-β φ p

  -- η: shutting the openings of a value gives the value back
  shut-open : ∀ {X} (ψ : PshHom Γ (R X)) → shut (λ p → inc p ⋆PshHom unlock ψ) ≡ ψ
  shut-open {X} ψ = makePshHomPath (funExt λ v → funExt λ e → funExt λ q →
    funExt⁻ (X .F-id) (ψ .N-ob v e q))

-- familial representability, on the p-summand:
--   y (scope u p) ≅ Summand p   (for Γ = y u),  ν and πS
module _ {u : W.ob} where
  open Over {u} {Y u} idPshHom

  ν : ∀ p → PshHom (Y (scope u p)) (Summand p)
  ν p .N-ob t k = ⟪ u , p , k , W.id ⟫ , k ,
    ΣPathP (pos-id p , (λ i → smap-id p i W.⋆ k) ▷ W.⋆IdL k)
  ν p .N-hom c c' h k = summand≡ refl refl

  πS-ν : ∀ p {t} (k : W [ scope u p , t ]) → πS p .N-ob t (ν p .N-ob t k) ≡ k
  πS-ν p k = refl

  ν-πS : ∀ p {t} (s : El (Summand p) t) → ν p .N-ob t (πS p .N-ob t s) ≡ s
  ν-πS p {t} (x , k , r) = summand≡ (goal x k r) refl
    where
    goal : ∀ x k (r : Lπ .N-ob t x ≡ (p , k)) → pt (Y u) (u , p , k , W.id) ≡ x
    goal = elimProp (λ _ → isPropΠ2 λ _ _ → squash/ _ _) λ (v , q , g , h) k r →
      sym (cong (λ (pk : El (Ey u) t) → pt (Y u) (u , pk .fst , pk .snd , W.id)) r)
      ∙ sym (eq/ _ _ (shift h q g W.id))
      ∙ cong (λ m → pt (Y u) (v , q , g , m)) (W.⋆IdL h)

------------------------------------------------------------------------
-- lemmas for the structural maps

-- E on morphisms, as a map of presheaves  E w' → E w  (for f : w → w')
Epull : ∀ {w w'} → W [ w , w' ] → PshHom (Ey w') (Ey w)
Epull f .N-ob t (p , k) = pos f p , smap f p W.⋆ k
Epull f .N-hom c c' h (p , k) = cong (pos f p ,_) (sym (W.⋆Assoc _ k h))

-- unlock at a point moved by L α
unlock-Lmap-at : ∀ {Γ Δ X} (α : PshHom Γ Δ) (φ : PshHom Δ (R X)) (φ' : PshHom Γ (R X)) →
  (∀ v e → φ' .N-ob v e ≡ φ .N-ob v (α .N-ob v e)) →
  ∀ {t} (x : El (L Γ) t) → unlock φ' .N-ob t x ≡ unlock φ .N-ob t (Lmap α .N-ob t x)
unlock-Lmap-at {X = X} α φ φ' hyp {t} = elimProp (λ _ → X .F-ob t .snd _ _)
  λ (v , q , g , e) → cong (λ r → X .F-hom g (r q)) (hyp v e)

-- unlocking a boxed map is precomposing the strength
unlock-box : ∀ {Γ X} (φ : PshHom Γ X) {t} (x : El (L Γ) t) →
  unlock (φ ⋆PshHom box X) .N-ob t x ≡ φ .N-ob t (ε Γ .N-ob t x)
unlock-box {Γ} {X} φ {t} = elimProp (λ _ → X .F-ob t .snd _ _) λ (v , q , g , e) →
  cong (X .F-hom g) (sym (φ .N-hom _ _ (ι v q) e)) ∙ sym (φ .N-hom _ _ g _)

-- L of a map over f : w → w' lies over E f
Lπ-Lmap : ∀ {w w' Γ Δ} (πΓ : PshHom Γ (Y w')) (πΔ : PshHom Δ (Y w))
  (α : PshHom Γ Δ) (f : W [ w , w' ]) →
  (∀ v e → πΔ .N-ob v (α .N-ob v e) ≡ f W.⋆ πΓ .N-ob v e) →
  ∀ {t} (x : El (L Γ) t) →
  Over.Lπ πΔ .N-ob t (Lmap α .N-ob t x) ≡ Epull f .N-ob t (Over.Lπ πΓ .N-ob t x)
Lπ-Lmap {w} πΓ πΔ α f over {t} = elimProp (λ _ → Ey w .F-ob t .snd _ _)
  λ (v , q , g , e) →
    cong (λ h → pos h q , smap h q W.⋆ g) (over v e)
    ∙ ΣPathP (pos-seq f (πΓ .N-ob v e) q ,
        (λ i → smap-seq f (πΓ .N-ob v e) q i W.⋆ g)
        ▷ W.⋆Assoc (smap f _) (smap (πΓ .N-ob v e) q) g)

module _ {w : W.ob} {Γ : 𝒱} (π : PshHom Γ (Y w)) where
  open Over π

  -- the strength of the p-summand lies over ι w p
  ε-over : ∀ {p t} (s : El (Summand p) t) →
    π .N-ob t (ε Γ .N-ob t (s .fst)) ≡ ι w p W.⋆ s .snd .fst
  ε-over {p} {t} (x , k , r) = goal x k r
    where
    goal : ∀ x k (r : Lπ .N-ob t x ≡ (p , k)) → π .N-ob t (ε Γ .N-ob t x) ≡ ι w p W.⋆ k
    goal = elimProp (λ _ → isPropΠ2 λ _ _ → W.isSetHom _ _) λ (v , q , g , e) k r →
      π .N-hom _ _ g _
      ∙ cong (W._⋆ g) (π .N-hom _ _ (ι v q) e)
      ∙ cong (W._⋆ g) (sym (ι-nat (π .N-ob v e) q))
      ∙ W.⋆Assoc _ _ g
      ∙ cong (λ (pk : El (Ey w) t) → ι w (pk .fst) W.⋆ pk .snd) r

-- L α on summands.  For α over f : w → w', the p-summand goes to the
-- p'-summand, where (p' , m) is E f at p (given up to a path, so that a
-- strict identity on world maps can be used)
module _ {w w' : W.ob} {Γ Δ : 𝒱} (πΓ : PshHom Γ (Y w')) (πΔ : PshHom Δ (Y w))
  (α : PshHom Γ Δ) (f : W [ w , w' ])
  (over : ∀ v e → πΔ .N-ob v (α .N-ob v e) ≡ f W.⋆ πΓ .N-ob v e) where

  Lsum : (p : Pos w') (p' : Pos w) (m : W [ scope w p' , scope w' p ]) →
    Path (El (Ey w) (scope w' p)) (pos f p , smap f p) (p' , m) →
    PshHom (Over.Summand πΓ p) (Over.Summand πΔ p')
  Lsum p p' m eq .N-ob t (x , k , r) =
    Lmap α .N-ob t x , m W.⋆ k ,
    Lπ-Lmap πΓ πΔ α f over x
    ∙ cong (Epull f .N-ob t) r
    ∙ cong (λ (pm : El (Ey w) (scope w' p)) → pm .fst , pm .snd W.⋆ k) eq
  Lsum p p' m eq .N-hom c c' h (x , k , r) =
    Over.summand≡ πΔ (Lmap α .N-hom c c' h x) (sym (W.⋆Assoc m k h))

-- the strength is natural
ε-Lmap : ∀ {Γ Δ} (α : PshHom Γ Δ) {t} (x : El (L Γ) t) →
  ε Δ .N-ob t (Lmap α .N-ob t x) ≡ α .N-ob t (ε Γ .N-ob t x)
ε-Lmap {Γ} {Δ} α x =
  sym (unlock-Lmap-at α (box Δ) (α ⋆PshHom box Δ) (λ _ _ → refl) x) ∙ unlock-box α x

-- opening a value of R X is unlocking it; at shut it is β
open-shut-at : ∀ {w Γ X} (π : PshHom Γ (Y w)) (φ : (p : Pos w) → PshHom (Over.Summand π p) X)
  p {t} (s : El (Over.Summand π p) t) →
  φ p .N-ob t s ≡ unlock (Over.shut π φ) .N-ob t (s .fst)
open-shut-at π φ p {t} s i = Over.open-shut π φ p (~ i) .N-ob t s
