-- The combined machine and its soundness (Thm 9.1).
--
-- Data: a theory over the derived polynomial (Equations; possibly none),
-- a free model of it (monad T), an RMR indexed container Q′ with a right
-- T-module ρ : Q ∘ T ⇒ Q, and a value type A.  Configurations are
-- Q (closed F A computations); a one-hole context Ctxt has a hole at a
-- world.  There are two steps:
--
--   pure     a tree step (Reduction._↦_) in the hole
--   effect   a closed op o vs k in the hole is closed and run in one
--            step: its parameters become constants, its continuations'
--            locks are closed (Closing.closeOp; names become constants),
--            and RMR's effect step (ER.step) runs the resulting
--            operation node (Closing.opNodeC) through the module
--
-- Closing and running are fused on purpose.  As a separate step, close
-- fired on its own output: closeOp o vs k is opNodeC …, which is itself
-- an op, so close could fire again forever and the machine was not
-- strongly normalizing.  Now every op in the hole is consumed by the
-- module in the step that closes it.
--
-- Soundness: denote = ρ ∘ Q h, where h sends a closed computation to its
-- denotation at the identity world map (denClosed, a term over ⟦ A ⟧) and
-- then into the free model.  Pure steps preserve h by tree soundness
-- (Soundness.sound↦); an effect step is closing, which preserves h by
-- familial representability (Closed.close-den), followed by RMR's effect
-- step, which preserves denote by ER.Semantic.sound↦.
--
-- Closed computations form the P-algebra CompAlg (opNodeC is its
-- operation) given the syntactic laws Closing.ClosedLaws (module WithLaws);
-- denClosed is then a P-homomorphism (hClosed, Thm 8.6: Lemmas 7.1, 7.2).

-- theories over the derived polynomial

record Equations : Type₁ where
  field
    Equation : Type
    equationWorld : Equation → W.ob
    Variables : Equation → 𝒱
    lhs rhs : (e : Equation) → Term (PM.polynomial L O) (Variables e) (equationWorld e)

theory : Equations → Theory W
theory eqs .Theory.signature = PM.polynomial L O
theory eqs .Theory.Equation = Equations.Equation eqs
theory eqs .Theory.equationWorld = Equations.equationWorld eqs
theory eqs .Theory.Variables = Equations.Variables eqs
theory eqs .Theory.lhs = Equations.lhs eqs
theory eqs .Theory.rhs = Equations.rhs eqs

-- the equation-free theory
noEquations : Equations
noEquations .Equations.Equation = ⊥
noEquations .Equations.equationWorld ()
noEquations .Equations.Variables ()
noEquations .Equations.lhs ()
noEquations .Equations.rhs ()

noEquations-empty : Theory.Equation (theory noEquations) → ⊥
noEquations-empty ()

------------------------------------------------------------------------
-- closed computations as an algebra, given the syntactic laws

module WithLaws (laws : ClosedLaws) where
  open ClosedLaws laws

  -- closed computations of type B: a presheaf under reindexC
  CompPsh : CompTy → 𝒱
  CompPsh B .F-ob w = Comp (∅ w) B , isSetComp
  CompPsh B .F-hom f = reindexC f
  CompPsh B .F-id = funExt reindexC-id
  CompPsh B .F-seq f g = funExt (reindexC-seq f g)

  -- the syntactic operation node, on an element of the extension
  opC : ∀ {B w} → El (Ext (CompPsh B)) w → Comp (∅ w) B
  opC ((o , cs) , ks) = opNodeC o cs (λ a ps → ks (a , ps))

  -- naturality: the polynomial reindexes positions and scopes by the
  -- word's arity (pos-is, smap-is), then reindexC-opNodeC.  For each i,
  -- reindexC (smap-is … i) (ks (pos-is … i)) has one fixed type, so the
  -- first path is homogeneous.
  opC-nat : ∀ {B w w'} (f : W [ w , w' ]) (x : El (Ext (CompPsh B)) w) →
    opC (Ext (CompPsh B) .F-hom f x) ≡ reindexC f (opC x)
  opC-nat f ((o , cs) , ks) =
    (λ i → opNodeC o (cmaps W Const (params o) f cs) λ a ps →
       reindexC (PM.smap-is L O f o cs (a , ps) i) (ks (PM.pos-is L O f o cs (a , ps) i)))
    ∙ sym (reindexC-opNodeC f o cs (λ a ps → ks (a , ps)))

  CompAlg : CompTy → Algebra (PM.polynomial L O)
  CompAlg B .Algebra.Carrier = CompPsh B
  CompAlg B .Algebra.algebra .NatTrans.N-ob w = opC
  CompAlg B .Algebra.algebra .NatTrans.N-hom f = funExt (opC-nat f)

  -- denClosed is a homomorphism into the term algebra (Thm 8.6)
  hClosed : (A : ValTy) → Hom (PM.polynomial L O) (CompAlg (F A)) (TermAlg (PM.polynomial L O) ⟦ A ⟧v)
  hClosed A .Hom.transformation .NatTrans.N-ob w = denClosed
  hClosed A .Hom.transformation .NatTrans.N-hom f = funExt (denClosed-reindex f)
  hClosed A .Hom.preserves = makeNatTransPath (funExt λ w → funExt λ ((o , cs) , ks) →
    opNodeC-den o cs (λ a ps → ks (a , ps)))

  ----------------------------------------------------------------------
  -- the machine: a theory with a free model, a configuration container,
  -- a right module, and the value type of the result

  module Run (eqs : Equations) (freeModel : FreeModel (theory eqs))
    (Q′ : IndexedContainer W.ob)
    (R : RightModule (EffectReduction.T (theory eqs) freeModel Q′)
                     (EffectReduction.Q (theory eqs) freeModel Q′))
    (A : ValTy) where

    module ER = EffectReduction (theory eqs) freeModel Q′
    open ER using (Q; ∂Q)
    open IndexedContainer Q′ using () renaming (input to inQ)

    private
      module Adj = UnitCounit._⊣_ (FreeModel.adjunction freeModel)

    -- the free model on ⟦ A ⟧
    freeA : Algebra (PM.polynomial L O)
    freeA = (FreeModel.free freeModel ⟅ ⟦ A ⟧v ⟆) .fst

    -- closed computations into the free model: denClosed, then the unit
    h : Hom (PM.polynomial L O) (CompAlg (F A)) freeA
    h = seqHom (PM.polynomial L O) (hClosed A) (evaluateHom freeA (Adj.η .NatTrans.N-ob ⟦ A ⟧v))

    open ER.Semantic R (CompAlg (F A)) ⟦ A ⟧v h public
      using (denote) renaming (sound↦ to sound-effect)

    Config : Type
    Config = (Q ⟅ CompPsh (F A) ⟆) .fst

    Ctxt : Type
    Ctxt = (∂Q ⟅ CompPsh (F A) ⟆) .fst

    hole-world : Ctxt → W.ob
    hole-world ((s , p) , _) = inQ s p

    -- equal denotations at the identity give equal h-images
    h-cong : ∀ {w} {M M' : Comp (∅ w) (F A)} → denClosed M ≡ denClosed M' →
      h .Hom.transformation .NatTrans.N-ob w M ≡ h .Hom.transformation .NatTrans.N-ob w M'
    h-cong {w} e = cong (evaluate freeA (Adj.η .NatTrans.N-ob ⟦ A ⟧v) .NatTrans.N-ob w) e

    -- a step in the hole that h does not see
    hole-step : (C : Ctxt) {M M' : Comp (∅ (hole-world C)) (F A)} →
      h .Hom.transformation .NatTrans.N-ob _ M ≡ h .Hom.transformation .NatTrans.N-ob _ M' →
      denote (plug Q′ (CompPsh (F A)) C M) ≡ denote (plug Q′ (CompPsh (F A)) C M')
    hole-step C {M} {M'} e = cong (RightModule.ρ R .NatTrans.N-ob ⟦ A ⟧v)
      (plug-map Q′ (CompPsh (F A)) (h .Hom.transformation) C M
       ∙ cong (plug Q′ (Algebra.Carrier freeA) ((∂Q ⟪ h .Hom.transformation ⟫) C)) e
       ∙ sym (plug-map Q′ (CompPsh (F A)) (h .Hom.transformation) C M'))

    infix 4 _⟶_ _⟶*_

    data _⟶_ : Config → Config → Type where
      -- a tree step in the hole
      pure : (C : Ctxt) {M M' : Comp (∅ (hole-world C)) (F A)} →
        M ↦ M' → plug Q′ (CompPsh (F A)) C M ⟶ plug Q′ (CompPsh (F A)) C M'
      -- a closed op in the hole: close its parameters and continuations,
      -- then run the closed operation node by RMR's effect step (the
      -- target is ER.step at that node; ER.redex at the same node is
      -- plug C (closeOp o vs k), definitionally)
      effect : (C : Ctxt) (o : Op) (vs : Vals (∅ (hole-world C)) (params o))
        (k : (a : Ar o) (ps : WPos (slot o a) (hole-world C)) →
             Comp (∅ (hole-world C) ⧀*⟨ slot o a , ps ⟩) (F A)) →
        plug Q′ (CompPsh (F A)) C (op o vs k) ⟶
          ER.step R (CompAlg (F A)) C ((o , closeVals vs) , λ (a , ps) → closeK (slot o a) (k a) ps)

    data _⟶*_ : Config → Config → Type where
      done : ∀ {x} → x ⟶* x
      _then_ : ∀ {x y z} → x ⟶ y → y ⟶* z → x ⟶* z
      _by-equation_ : ∀ {x y z} → x ≡ y → y ⟶* z → x ⟶* z

    infixr 5 _then_ _by-equation_

    -- Thm 9.1: machine steps preserve the denotation
    sound⟶ : ∀ {x y} → x ⟶ y → denote x ≡ denote y
    -- (⟦_⟧C is not injective: the computations are passed explicitly)
    sound⟶ (pure C {M} {M'} r) =
      hole-step C {M} {M'}
        (h-cong {M = M} {M'} (cong (λ φ → φ .N-ob (hole-world C) W.id) (sound-tree r)))
    sound⟶ (effect C o vs k) =
      hole-step C {op o vs k} {closeOp o vs k}
        (h-cong {M = op o vs k} {closeOp o vs k} (close-den o vs k))
      ∙ sound-effect (ER.effect-step C ((o , closeVals vs) , λ (a , ps) → closeK (slot o a) (k a) ps))

    sound⟶* : ∀ {x y} → x ⟶* y → denote x ≡ denote y
    sound⟶* done = refl
    sound⟶* (r then rs) = sound⟶ r ∙ sound⟶* rs
    sound⟶* (p by-equation rs) = cong denote p ∙ sound⟶* rs
