-- The multiset order on finite families of natural numbers, in counting
-- form (pure mathematics, for the normalization measure).
--
-- A finite family h : X → ℕ (X a finite set) is a finite multiset of
-- natural numbers.  Its COUNTING FUNCTION is
--
--     count X h n  =  #{ x ∈ X | n ≤ h x }          (how many are ≥ n)
--
-- and the order on counting functions is
--
--     G' ⊏ G  :=  Σ m.  G' m < G m  ×  ∀ n > m. G' n ≤ G n.
--
-- On counting functions of finite multisets this is the Dershowitz–Manna
-- multiset order of (ℕ , <) (remark, not formalised: take m the largest
-- height at which the exact multiplicities differ).
--
-- Replacement (count-update, count-replace).  Remove one element p of X
-- and add elements each of height < h p; keep the others, possibly
-- lowered, without repeating any of them (an injection).  Then the
-- counting function decreases: at every n ≥ h p the count does not grow
-- and at n = h p it drops by p itself.  So every one-step replacement
-- (one element by finitely many smaller ones) is a ⊏-step.
--
-- Well-foundedness (⊏-acc).  A counting function vanishes above the
-- largest height H (count-vanishes).  For functions vanishing above H,
-- G' ⊏ G implies that G' vanishes above H too and that G' is below G in
-- the lexicographic order on (G H , G (H-1) , … , G 0) (⊏→lex), and the
-- lexicographic order on ℕ^(H+1) is well founded (lex-wf: an induction on
-- the first component with an inner induction on the rest).
--
--   atLeast n m             the indicator of n ≤ m
--   count                   the counting function
--   _⊏_, Vanishes           the order; vanishing above H
--   count-vanishes          a bounded family has a vanishing count
--   count-update            one element lowered, the others not raised
--   count-replace           one element replaced by smaller ones, the
--                           others moved injectively and not raised
--   LexBelow, lex-wf        the lexicographic order on G H , … , G 0
--   ⊏→lex, ⊏-acc            ⊏ is well founded on vanishing functions

----------------------------------------------------------------------
-- the counting function

private
  ind : {P : Type} → Dec P → ℕ
  ind (yes _) = 1
  ind (no _) = 0

-- the indicator of n ≤ m
atLeast : ℕ → ℕ → ℕ
atLeast n m = ind (≤Dec n m)

atLeast-yes : ∀ {n m} → n ≤ m → atLeast n m ≡ 1
atLeast-yes {n} {m} le with ≤Dec n m
... | yes _ = refl
... | no ¬le = ⊥.rec (¬le le)

atLeast-no : ∀ {n m} → m < n → atLeast n m ≡ 0
atLeast-no {n} {m} lt with ≤Dec n m
... | yes le = ⊥.rec (<-asym lt le)
... | no _ = refl

atLeast-mono : ∀ n {m m'} → m ≤ m' → atLeast n m ≤ atLeast n m'
atLeast-mono n {m} {m'} le with ≤Dec n m
... | yes nm = ≤-reflexive (sym (atLeast-yes (≤-trans nm le)))
... | no _ = zero-≤

-- how many elements of the family have height at least n
count : (X : FinSet ℓ) → (X .fst → ℕ) → ℕ → ℕ
count X h n = sum X λ x → atLeast n (h x)

----------------------------------------------------------------------
-- the order, and vanishing above a bound

infix 4 _⊏_

_⊏_ : (ℕ → ℕ) → (ℕ → ℕ) → Type
G' ⊏ G = Σ[ m ∈ ℕ ] (G' m < G m) × (∀ n → m < n → G' n ≤ G n)

Vanishes : ℕ → (ℕ → ℕ) → Type
Vanishes H G = ∀ n → H < n → G n ≡ 0

count-vanishes : (X : FinSet ℓ) (h : X .fst → ℕ) (H : ℕ) → (∀ x → h x ≤ H) → Vanishes H (count X h)
count-vanishes X h H bound n H<n = sum-zero X _ λ x → atLeast-no (≤<-trans (bound x) H<n)

----------------------------------------------------------------------
-- replacement

module _ (X : FinSet ℓ) (dec : Discrete (X .fst)) (p : X .fst) (h : X .fst → ℕ) where

  private
    -- the count without p
    rest : ℕ → ℕ
    rest n = sum (Minus X dec p) λ r → atLeast n (h (r .fst))

    -- a function G' that at every n ≥ h p is at most the count without p
    -- is ⊏-below the count (with m = h p)
    ⊏-intro : (G' : ℕ → ℕ) → (∀ n → h p ≤ n → G' n ≤ rest n) → G' ⊏ count X h
    ⊏-intro G' bound = h p , lt , above
      where
      total : ∀ n → h p ≤ n → G' n + atLeast n (h p) ≤ count X h n
      total n le = ≤-trans (≤-+k (bound n le))
        (≤-reflexive (+-comm (rest n) _ ∙ sym (sum-split X dec p λ x → atLeast n (h x))))

      lt : G' (h p) < count X h (h p)
      lt = subst (_≤ count X h (h p)) (cong (G' (h p) +_) (atLeast-yes ≤-refl) ∙ +-comm (G' (h p)) 1)
        (total (h p) ≤-refl)

      above : ∀ n → h p < n → G' n ≤ count X h n
      above n lt' = ≤-trans ≤SumLeft (total n (<-weaken lt'))

  -- one element lowered, the others not raised
  count-update : (h' : X .fst → ℕ) → h' p < h p → (∀ r → ¬ r ≡ p → h' r ≤ h r) →
    count X h' ⊏ count X h
  count-update h' lt others = ⊏-intro (count X h') λ n le →
    subst (_≤ rest n)
      (sym (sum-split X dec p (λ x → atLeast n (h' x)) ∙ cong (_+ rest' n) (atLeast-no (<≤-trans lt le))))
      (sum≤ (Minus X dec p) (λ r → atLeast n (h' (r .fst))) (λ r → atLeast n (h (r .fst)))
        λ r → atLeast-mono n (others (r .fst) (r .snd)))
    where
    rest' : ℕ → ℕ
    rest' n = sum (Minus X dec p) λ r → atLeast n (h' (r .fst))

  -- the height of an origin: a remaining element of X, or an element of
  -- a new family wt
  srcHeight : {S : FinSet ℓ} → (S .fst → ℕ) → Minus X dec p .fst ⊎ S .fst → ℕ
  srcHeight wt (inl r) = h (r .fst)
  srcHeight wt (inr c) = wt c

  -- one element replaced by elements of a finite family wt, each below
  -- it: every element of Y comes from a remaining element of X or from wt
  -- (org, injective) and is not higher than its origin
  module _ (Y S : FinSet ℓ) (h' : Y .fst → ℕ) (wt : S .fst → ℕ) (wt< : ∀ c → wt c < h p)
    (org : Y .fst → Minus X dec p .fst ⊎ S .fst)
    (org-inj : ∀ y y' → org y ≡ org y' → y ≡ y') where

    count-replace : (∀ y → h' y ≤ srcHeight {S = S} wt (org y)) → count Y h' ⊏ count X h
    count-replace below = ⊏-intro (count Y h') bound
      where
      bound : ∀ n → h p ≤ n → count Y h' n ≤ rest n
      bound n le =
        ≤-trans (sum≤ Y _ _ λ y → atLeast-mono n (below y))
          (≤-trans (sum-inj Y (_ , isFinSet⊎ (Minus X dec p) S) org org-inj g)
            (≤-reflexive (sum⊎ (Minus X dec p) S g
              ∙ cong (rest n +_) (sum-zero S _ λ c → atLeast-no (<≤-trans (wt< c) le))
              ∙ +-zero (rest n))))
        where
        g : Minus X dec p .fst ⊎ S .fst → ℕ
        g s = atLeast n (srcHeight {S = S} wt s)

----------------------------------------------------------------------
-- well-foundedness

-- the lexicographic order on G H , G (H-1) , … , G 0
LexBelow : ℕ → (ℕ → ℕ) → (ℕ → ℕ) → Type
LexBelow zero G' G = G' 0 < G 0
LexBelow (suc H) G' G = (G' (suc H) < G (suc H)) ⊎ ((G' (suc H) ≡ G (suc H)) × LexBelow H G' G)

-- An induction on the first component n = G (suc H), with an inner
-- induction on the rest.  The two inductions are on the Acc arguments,
-- passed explicitly so that the termination checker sees the
-- lexicographic decrease (H , Acc n , Acc rest).
lex-wf : ∀ H → WellFounded (LexBelow H)

lex-acc : ∀ H n → Acc _<_ n → ∀ G → G (suc H) ≡ n → Acc (LexBelow H) G → Acc (LexBelow (suc H)) G

lex-below : ∀ H n → Acc _<_ n → ∀ G → G (suc H) ≡ n → Acc (LexBelow H) G →
  ∀ G' → LexBelow (suc H) G' G → Acc (LexBelow (suc H)) G'

lex-acc H n an G e aG = acc (lex-below H n an G e aG)

-- the first component dropped: the outer hypothesis
lex-below H n (acc r) G e aG G' (inl lt) =
  lex-acc H (G' (suc H)) (r _ (subst (G' (suc H) <_) e lt)) G' refl (lex-wf H G')
-- the first component kept: the inner hypothesis
lex-below H n an G e (acc rG) G' (inr (e' , l)) = lex-acc H n an G' (e' ∙ e) (rG G' l)

lex-wf zero G = go (G 0) (<-wellfounded (G 0)) G refl
  where
  go : ∀ n → Acc _<_ n → ∀ G → G 0 ≡ n → Acc (LexBelow zero) G
  go n (acc r) G e = acc λ G' lt → go (G' 0) (r (G' 0) (subst (G' 0 <_) e lt)) G' refl
lex-wf (suc H) G = lex-acc H (G (suc H)) (<-wellfounded _) G refl (lex-wf H G)

-- strict at m, not above it, up to H: lexicographically below
lex-intro : ∀ H m {G' G} → m ≤ H → (∀ n → m < n → G' n ≤ G n) → G' m < G m → LexBelow H G' G
lex-intro zero m {G'} {G} m≤0 above lt = subst (λ k → G' k < G k) (≤0→≡0 m≤0) lt
lex-intro (suc H) m {G'} {G} m≤ above lt with ≤-split m≤
... | inr m≡ = inl (subst (λ k → G' k < G k) m≡ lt)
... | inl m< with ≤-split (above (suc H) m<)
...   | inl top< = inl top<
...   | inr top≡ = inr (top≡ , lex-intro H m (pred-≤-pred m<) above lt)

-- below a vanishing function, ⊏ is lexicographic and keeps vanishing
⊏→lex : ∀ H {G' G} → Vanishes H G → G' ⊏ G → Vanishes H G' × LexBelow H G' G
⊏→lex H {G'} {G} van (m , lt , above) = van' , lex-intro H m m≤H above lt
  where
  -- G m > 0, so m is at most H
  m≤H : m ≤ H
  m≤H with ≤Dec m H
  ... | yes le = le
  ... | no ¬le = ⊥.rec (¬-<-zero (subst (G' m <_) (van m (<-asym' λ m<sH → ¬le (pred-≤-pred m<sH))) lt))

  van' : Vanishes H G'
  van' n H<n = ≤0→≡0 (subst (G' n ≤_) (van n H<n) (above n (≤<-trans m≤H H<n)))

-- every function vanishing above some H is accessible for ⊏
⊏-acc : ∀ H G → Vanishes H G → Acc (LexBelow H) G → Acc _⊏_ G
⊏-acc H G van (acc r) = acc λ G' lt →
  let (van' , l) = ⊏→lex H van lt in ⊏-acc H G' van' (r G' l)

⊏-vanishing-acc : ∀ H G → Vanishes H G → Acc _⊏_ G
⊏-vanishing-acc H G van = ⊏-acc H G van (lex-wf H G)
