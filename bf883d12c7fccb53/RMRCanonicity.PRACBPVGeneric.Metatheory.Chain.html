-- Chains: formal composites of substitutions (6.3).
--
-- WHY.  The logical relation is Kripke: a related thunk or function must
-- stay related after any further substitution, and those compose.  But
-- data substitutions have no composite with exactly the right source.
-- Under a lock, subV σ reads a body at the pulled-back position
-- posW κ (wmS σ) q, so σ₁ then σ₂ reads it at
-- posW κ (wmS σ₁) (posW κ (wmS σ₂) q), and a single substitution with world
-- map wmS σ₁ ⋆ wmS σ₂ reads it at a position equal to that one only
-- propositionally (the arity's F-seq).  So the Kripke structure ranges over
-- FORMAL composites, lists of substitutions, acting by iterated
-- substitution:
--
--   Chain Θ₁ Θ₃     []ᶜ  |  σ ∷ᶜ χ   (σ : Sub Θ₁ Θ₂, χ : Chain Θ₂ Θ₃)
--   chV, chVs, chC  the action: substitute σ, then act by χ
--   χ ++ᶜ χ'        composition (ch-++: the action of a composite is the
--                   composite of the actions, by definition at each step)
--
-- Every structural operation on substitutions extends to chains, one
-- substitution at a time:
--
--   liftᶜ χ          under a variable (liftS on each)
--   wposᶜ μ χ ps     positions of a lock word pulled back along the chain
--   wordᶜ μ χ ps     under the locks of a word (liftWS on each)
--   posᶜ, lockᶜ      the one-lock case of wposᶜ, wordᶜ
--
-- and a chain commutes with every term former (ch-return, ch-to, …,
-- ch-inl, ch-inr, ch-case₊, ch-absurd, ch-op, ch-shut) and with the one-variable and one-lock operations
-- (ch-wkV, ch-↑, ch-openV, ch-β).  Each law is by induction on the chain,
-- and each step is the corresponding law for ONE substitution (Thm 5.6):
-- the term formers are the clauses of subC, the others are wkS-ne,
-- sub-liftS-wkV, sub-↑, sub-openV, sub-wkV.  ch-↦ is sub-↦ iterated.
--
-- Over a word of two or more locks, wordᶜ (κ ∷ μ) χ (p , ps) is
-- wordᶜ μ (lockᶜ κ χ p) ps, but only over a path of positions
-- (wposᶜ-cons, wordᶜ-cons): each step of the path is definitional, the
-- path itself is not (χ is a variable).
open import Cubical.Foundations.Prelude hiding (Sub; _,_; comp)
open import Cubical.Data.Sigma using () renaming (_,_ to _,ˢ_)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Categories.Category using (Category)

open import RMRCanonicity.PRACBPVGeneric.Signature

----------------------------------------------------------------------
-- chains and their action

infixr 5 _∷ᶜ_ _++ᶜ_

data Chain : Tele → Tele → Type where
  []ᶜ : ∀ {Θ} → Chain Θ Θ
  _∷ᶜ_ : ∀ {Θ₁ Θ₂ Θ₃} → Sub Θ₁ Θ₂ → Chain Θ₂ Θ₃ → Chain Θ₁ Θ₃

_++ᶜ_ : ∀ {Θ₁ Θ₂ Θ₃} → Chain Θ₁ Θ₂ → Chain Θ₂ Θ₃ → Chain Θ₁ Θ₃
[]ᶜ ++ᶜ χ' = χ'
(σ ∷ᶜ χ) ++ᶜ χ' = σ ∷ᶜ (χ ++ᶜ χ')

chV : ∀ {Θ Θ' A} → Chain Θ Θ' → Val Θ A → Val Θ' A
chV []ᶜ V = V
chV (σ ∷ᶜ χ) V = chV χ (subV σ V)

chVs : ∀ {Θ Θ' bs} → Chain Θ Θ' → Vals Θ bs → Vals Θ' bs
chVs []ᶜ vs = vs
chVs (σ ∷ᶜ χ) vs = chVs χ (subVs σ vs)

chC : ∀ {Θ Θ' B} → Chain Θ Θ' → Comp Θ B → Comp Θ' B
chC []ᶜ M = M
chC (σ ∷ᶜ χ) M = chC χ (subC σ M)

-- under a variable
liftᶜ : ∀ {Θ Θ' A} → Chain Θ Θ' → Chain (Θ , A) (Θ' , A)
liftᶜ []ᶜ = []ᶜ
liftᶜ (σ ∷ᶜ χ) = liftS σ ∷ᶜ liftᶜ χ

-- positions of a lock word, pulled back along the chain
wposᶜ : ∀ {Θ Θ'} (μ : List Lk) → Chain Θ Θ' → WPos μ (world Θ') → WPos μ (world Θ)
wposᶜ μ []ᶜ ps = ps
wposᶜ μ (σ ∷ᶜ χ) ps = wposW μ (wmS σ) (wposᶜ μ χ ps)

-- under the locks of a word
wordᶜ : ∀ {Θ Θ'} (μ : List Lk) (χ : Chain Θ Θ') (ps : WPos μ (world Θ')) →
  Chain (Θ ⧀*⟨ μ , wposᶜ μ χ ps ⟩) (Θ' ⧀*⟨ μ , ps ⟩)
wordᶜ μ []ᶜ ps = []ᶜ
wordᶜ μ (σ ∷ᶜ χ) ps = liftWS μ σ (wposᶜ μ χ ps) ∷ᶜ wordᶜ μ χ ps

-- under one lock: lockᶜ κ (σ ∷ᶜ χ) q = lockₛ σ (posᶜ κ χ q) idR ∷ᶜ lockᶜ κ χ q
posᶜ : ∀ {Θ Θ'} (κ : Lk) → Chain Θ Θ' → Pos κ (world Θ') → Pos κ (world Θ)
posᶜ κ = wposᶜ (κ ∷ [])

lockᶜ : ∀ {Θ Θ'} (κ : Lk) (χ : Chain Θ Θ') (q : Pos κ (world Θ')) →
  Chain (Θ ⧀⟨ κ , posᶜ κ χ q ⟩) (Θ' ⧀⟨ κ , q ⟩)
lockᶜ κ = wordᶜ (κ ∷ [])

----------------------------------------------------------------------
-- composition

ch-++V : ∀ {Θ₁ Θ₂ Θ₃ A} (χ : Chain Θ₁ Θ₂) (χ' : Chain Θ₂ Θ₃) (V : Val Θ₁ A) →
  chV (χ ++ᶜ χ') V ≡ chV χ' (chV χ V)
ch-++V []ᶜ χ' V = refl
ch-++V (σ ∷ᶜ χ) χ' V = ch-++V χ χ' (subV σ V)

ch-++C : ∀ {Θ₁ Θ₂ Θ₃ B} (χ : Chain Θ₁ Θ₂) (χ' : Chain Θ₂ Θ₃) (M : Comp Θ₁ B) →
  chC (χ ++ᶜ χ') M ≡ chC χ' (chC χ M)
ch-++C []ᶜ χ' M = refl
ch-++C (σ ∷ᶜ χ) χ' M = ch-++C χ χ' (subC σ M)

liftᶜ-++ : ∀ {Θ₁ Θ₂ Θ₃ A} (χ : Chain Θ₁ Θ₂) (χ' : Chain Θ₂ Θ₃) →
  liftᶜ {A = A} (χ ++ᶜ χ') ≡ liftᶜ χ ++ᶜ liftᶜ χ'
liftᶜ-++ []ᶜ χ' = refl
liftᶜ-++ (σ ∷ᶜ χ) χ' = cong (liftS σ ∷ᶜ_) (liftᶜ-++ χ χ')

----------------------------------------------------------------------
-- a chain commutes with the term formers

ch-tt : ∀ {Θ Θ'} (χ : Chain Θ Θ') → chV χ tt ≡ tt
ch-tt []ᶜ = refl
ch-tt (σ ∷ᶜ χ) = ch-tt χ

ch-inl : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') (V : Val Θ A) → chV χ (inl {A' = A'} V) ≡ inl (chV χ V)
ch-inl []ᶜ V = refl
ch-inl (σ ∷ᶜ χ) V = ch-inl χ (subV σ V)

ch-inr : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') (V : Val Θ A') → chV χ (inr {A = A} V) ≡ inr (chV χ V)
ch-inr []ᶜ V = refl
ch-inr (σ ∷ᶜ χ) V = ch-inr χ (subV σ V)

ch-pair : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') (V : Val Θ A) (V' : Val Θ A') →
  chV χ (pair V V') ≡ pair (chV χ V) (chV χ V')
ch-pair []ᶜ V V' = refl
ch-pair (σ ∷ᶜ χ) V V' = ch-pair χ (subV σ V) (subV σ V')

ch-thunk : ∀ {Θ Θ' B} (χ : Chain Θ Θ') (M : Comp Θ B) → chV χ (thunk M) ≡ thunk (chC χ M)
ch-thunk []ᶜ M = refl
ch-thunk (σ ∷ᶜ χ) M = ch-thunk χ (subC σ M)

ch-shut : ∀ {Θ Θ' κ A} (χ : Chain Θ Θ') (V : (p : Pos κ (world Θ)) → Val (Θ ⧀⟨ κ , p ⟩) A) →
  chV χ (shut V) ≡ shut λ q → chV (lockᶜ κ χ q) (V (posᶜ κ χ q))
ch-shut []ᶜ V = refl
ch-shut {κ = κ} (σ ∷ᶜ χ) V = ch-shut χ λ q → subV (lockₛ σ q idR) (V (posW κ (wmS σ) q))

ch-return : ∀ {Θ Θ' A} (χ : Chain Θ Θ') (V : Val Θ A) → chC χ (return V) ≡ return (chV χ V)
ch-return []ᶜ V = refl
ch-return (σ ∷ᶜ χ) V = ch-return χ (subV σ V)

ch-to : ∀ {Θ Θ' A B} (χ : Chain Θ Θ') (M : Comp Θ (F A)) (N : Comp (Θ , A) B) →
  chC χ (M to N) ≡ (chC χ M to chC (liftᶜ χ) N)
ch-to []ᶜ M N = refl
ch-to (σ ∷ᶜ χ) M N = ch-to χ (subC σ M) (subC (liftS σ) N)

ch-force : ∀ {Θ Θ' B} (χ : Chain Θ Θ') (V : Val Θ (U B)) → chC χ (force V) ≡ force (chV χ V)
ch-force []ᶜ V = refl
ch-force (σ ∷ᶜ χ) V = ch-force χ (subV σ V)

ch-case : ∀ {Θ Θ' A A' B} (χ : Chain Θ Θ') (V : Val Θ (A × A')) (M : Comp (Θ , A , A') B) →
  chC χ (case V M) ≡ case (chV χ V) (chC (liftᶜ (liftᶜ χ)) M)
ch-case []ᶜ V M = refl
ch-case (σ ∷ᶜ χ) V M = ch-case χ (subV σ V) (subC (liftS (liftS σ)) M)

ch-case₊ : ∀ {Θ Θ' A A' B} (χ : Chain Θ Θ') (V : Val Θ (A ⊕ A')) (M : Comp (Θ , A) B)
  (N : Comp (Θ , A') B) → chC χ (case₊ V M N) ≡ case₊ (chV χ V) (chC (liftᶜ χ) M) (chC (liftᶜ χ) N)
ch-case₊ []ᶜ V M N = refl
ch-case₊ (σ ∷ᶜ χ) V M N = ch-case₊ χ (subV σ V) (subC (liftS σ) M) (subC (liftS σ) N)

ch-absurd : ∀ {Θ Θ' B} (χ : Chain Θ Θ') (V : Val Θ 𝟘) → chC χ (absurd {B = B} V) ≡ absurd (chV χ V)
ch-absurd []ᶜ V = refl
ch-absurd (σ ∷ᶜ χ) V = ch-absurd χ (subV σ V)

ch-lam : ∀ {Θ Θ' A B} (χ : Chain Θ Θ') (M : Comp (Θ , A) B) → chC χ (lam M) ≡ lam (chC (liftᶜ χ) M)
ch-lam []ᶜ M = refl
ch-lam (σ ∷ᶜ χ) M = ch-lam χ (subC (liftS σ) M)

ch-app : ∀ {Θ Θ' A B} (χ : Chain Θ Θ') (M : Comp Θ (A ⇒ B)) (V : Val Θ A) →
  chC χ (app M V) ≡ app (chC χ M) (chV χ V)
ch-app []ᶜ M V = refl
ch-app (σ ∷ᶜ χ) M V = ch-app χ (subC σ M) (subV σ V)

ch-op : ∀ {Θ Θ' B} (χ : Chain Θ Θ') (o : Op) (vs : Vals Θ (params o))
  (k : (a : Ar o) (ps : WPos (slot o a) (world Θ)) → Comp (Θ ⧀*⟨ slot o a , ps ⟩) B) →
  chC χ (op o vs k)
    ≡ op o (chVs χ vs) λ a ps → chC (wordᶜ (slot o a) χ ps) (k a (wposᶜ (slot o a) χ ps))
ch-op []ᶜ o vs k = refl
ch-op (σ ∷ᶜ χ) o vs k =
  ch-op χ o (subVs σ vs) λ a ps → subC (liftWS (slot o a) σ ps) (k a (wposW (slot o a) (wmS σ) ps))

----------------------------------------------------------------------
-- a chain commutes with crossing and opening a lock

ch-↑ : ∀ {Θ Θ' κ A} (χ : Chain Θ Θ') (q : Pos κ (world Θ')) (V : Val Θ A) →
  chV (lockᶜ κ χ q) (↑V (posᶜ κ χ q) V) ≡ ↑V q (chV χ V)
ch-↑ []ᶜ q V = refl
ch-↑ {κ = κ} (σ ∷ᶜ χ) q V =
  cong (chV (lockᶜ κ χ q)) (sub-↑ σ (posᶜ κ χ q) V) ∙ ch-↑ χ q (subV σ V)

ch-openV : ∀ {Θ Θ' κ A} (χ : Chain Θ Θ') (q : Pos κ (world Θ')) (V : Val Θ ([ κ ] A)) →
  chV (lockᶜ κ χ q) (openV V (posᶜ κ χ q)) ≡ openV (chV χ V) q
ch-openV []ᶜ q V = refl
ch-openV {κ = κ} (σ ∷ᶜ χ) q V =
  cong (chV (lockᶜ κ χ q)) (sub-openV σ (posᶜ κ χ q) V) ∙ ch-openV χ q (subV σ V)

----------------------------------------------------------------------
-- under a variable, and the β-substitution

ch-wkV : ∀ {Θ Θ' A B} (χ : Chain Θ Θ') (V : Val Θ A) →
  chV (liftᶜ {A = B} χ) (wkV V) ≡ wkV (chV χ V)
ch-wkV []ᶜ V = refl
ch-wkV (σ ∷ᶜ χ) V = cong (chV (liftᶜ χ)) (sub-liftS-wkV σ V) ∙ ch-wkV χ (subV σ V)

ch-lift-here : ∀ {Θ Θ' A} (χ : Chain Θ Θ') → chV (liftᶜ {A = A} χ) (ne here) ≡ ne here
ch-lift-here []ᶜ = refl
ch-lift-here (σ ∷ᶜ χ) = ch-lift-here χ

-- (ne (wkv n) is wkV (ne n) by definition)
ch-lift-wkv : ∀ {Θ Θ' A B} (χ : Chain Θ Θ') (n : Ne Θ A) →
  chV (liftᶜ {A = B} χ) (ne (wkv n)) ≡ wkV (chV χ (ne n))
ch-lift-wkv χ n = ch-wkV χ (ne n)

-- under a variable, then substitute V for it
ch-β-here : ∀ {Θ Θ' A} (χ : Chain Θ Θ') (V : Val Θ' A) →
  chV (liftᶜ χ ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ) (ne here) ≡ V
ch-β-here χ V = ch-++V (liftᶜ χ) _ (ne here) ∙ cong (subV (idS ,ₛ V)) (ch-lift-here χ)

ch-β : ∀ {Θ Θ' A B} (χ : Chain Θ Θ') (V : Val Θ' B) (n : Ne Θ A) →
  chV (liftᶜ χ ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ) (ne (wkv n)) ≡ chV χ (ne n)
ch-β χ V n =
  ch-++V (liftᶜ χ) _ (ne (wkv n))
  ∙ cong (subV (idS ,ₛ V)) (ch-lift-wkv χ n)
  ∙ sub-wkV V (chV χ (ne n))

-- the same under two variables (case)
ch-β₂-snd : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') (V : Val Θ' A) (V' : Val Θ' A') →
  chV (liftᶜ (liftᶜ χ) ++ᶜ (idS ,ₛ V ,ₛ V') ∷ᶜ []ᶜ) (ne here) ≡ V'
ch-β₂-snd χ V V' =
  ch-++V (liftᶜ (liftᶜ χ)) _ (ne here) ∙ cong (subV (idS ,ₛ V ,ₛ V')) (ch-lift-here (liftᶜ χ))

ch-β₂-fst : ∀ {Θ Θ' A A'} (χ : Chain Θ Θ') (V : Val Θ' A) (V' : Val Θ' A') →
  chV (liftᶜ (liftᶜ χ) ++ᶜ (idS ,ₛ V ,ₛ V') ∷ᶜ []ᶜ) (ne (wkv here)) ≡ V
ch-β₂-fst χ V V' =
  ch-++V (liftᶜ (liftᶜ χ)) _ (ne (wkv here))
  ∙ cong (subV (idS ,ₛ V ,ₛ V')) (ch-lift-wkv (liftᶜ χ) here ∙ cong wkV (ch-lift-here χ))

ch-β₂ : ∀ {Θ Θ' A A' B} (χ : Chain Θ Θ') (V : Val Θ' A) (V' : Val Θ' A') (n : Ne Θ B) →
  chV (liftᶜ (liftᶜ χ) ++ᶜ (idS ,ₛ V ,ₛ V') ∷ᶜ []ᶜ) (ne (wkv (wkv n))) ≡ chV χ (ne n)
ch-β₂ χ V V' n =
  ch-++V (liftᶜ (liftᶜ χ)) _ (ne (wkv (wkv n)))
  ∙ cong (subV (idS ,ₛ V ,ₛ V'))
      (ch-lift-wkv (liftᶜ χ) (wkv n) ∙ cong wkV (ch-lift-wkv χ n))
  ∙ sub-cons-wkV (idS ,ₛ V) V' (wkV (chV χ (ne n)))
  ∙ sub-wkV V (chV χ (ne n))

-- the body of a binder, after χ then a further chain χ', then β
ch-sub₁ : ∀ {Θ₁ Θ₂ Θ₃ A B} (χ : Chain Θ₁ Θ₂) (χ' : Chain Θ₂ Θ₃) (N : Comp (Θ₁ , A) B) (V : Val Θ₃ A) →
  chC (liftᶜ (χ ++ᶜ χ') ++ᶜ (idS ,ₛ V) ∷ᶜ []ᶜ) N ≡ sub₁ (chC (liftᶜ χ') (chC (liftᶜ χ) N)) V
ch-sub₁ χ χ' N V =
  ch-++C (liftᶜ (χ ++ᶜ χ')) _ N
  ∙ cong (λ X → sub₁ X V) (cong (λ c → chC c N) (liftᶜ-++ χ χ') ∙ ch-++C (liftᶜ χ) (liftᶜ χ') N)

-- a renaming under a variable, as a substitution
sub-liftS-ren : ∀ {Θ Θ' A B} (ρ : Ren Θ Θ') (N : Comp (Θ , A) B) →
  subC (liftS ⌜ ρ ⌝) N ≡ renC (keep ρ) N
sub-liftS-ren ρ N = sub-agree-renC (AgreeSR-keep (AgreeSR-ren ρ)) N refl

----------------------------------------------------------------------
-- under the locks of a word

-- the empty word changes nothing
wordᶜ-[] : ∀ {Θ Θ'} (χ : Chain Θ Θ') (ps : WPos [] (world Θ')) → wordᶜ [] χ ps ≡ χ
wordᶜ-[] []ᶜ ps = refl
wordᶜ-[] (σ ∷ᶜ χ) ps = cong (σ ∷ᶜ_) (wordᶜ-[] χ ps)

-- a word of two or more locks: the first lock, then the rest
wposᶜ-cons : ∀ {Θ Θ'} (κ κ' : Lk) (μ : List Lk) (χ : Chain Θ Θ') (p : Pos κ (world Θ'))
  (ps : WPos (κ' ∷ μ) (scope κ (world Θ') p)) →
  wposᶜ (κ ∷ κ' ∷ μ) χ (p ,ˢ ps) ≡ (posᶜ κ χ p ,ˢ wposᶜ (κ' ∷ μ) (lockᶜ κ χ p) ps)
wposᶜ-cons κ κ' μ []ᶜ p ps = refl
wposᶜ-cons κ κ' μ (σ ∷ᶜ χ) p ps = cong (wposW (κ ∷ κ' ∷ μ) (wmS σ)) (wposᶜ-cons κ κ' μ χ p ps)

wordᶜ-cons : ∀ {Θ Θ'} (κ κ' : Lk) (μ : List Lk) (χ : Chain Θ Θ') (p : Pos κ (world Θ'))
  (ps : WPos (κ' ∷ μ) (scope κ (world Θ') p)) →
  PathP (λ i → Chain (Θ ⧀*⟨ κ ∷ κ' ∷ μ , wposᶜ-cons κ κ' μ χ p ps i ⟩) ((Θ' ⧀⟨ κ , p ⟩) ⧀*⟨ κ' ∷ μ , ps ⟩))
    (wordᶜ (κ ∷ κ' ∷ μ) χ (p ,ˢ ps)) (wordᶜ (κ' ∷ μ) (lockᶜ κ χ p) ps)
wordᶜ-cons κ κ' μ []ᶜ p ps = refl
wordᶜ-cons κ κ' μ (σ ∷ᶜ χ) p ps i =
  liftWS (κ ∷ κ' ∷ μ) σ (wposᶜ-cons κ κ' μ χ p ps i) ∷ᶜ wordᶜ-cons κ κ' μ χ p ps i

----------------------------------------------------------------------
-- reduction

ch-↦ : ∀ {Θ Θ' B} (χ : Chain Θ Θ') {M M' : Comp Θ B} → M ↦ M' → chC χ M ↦ chC χ M'
ch-↦ []ᶜ r = r
ch-↦ (σ ∷ᶜ χ) r = ch-↦ χ (sub-↦ σ r)
