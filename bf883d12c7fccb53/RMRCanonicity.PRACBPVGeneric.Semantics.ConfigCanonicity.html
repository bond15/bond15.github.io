-- Canonicity of the machine (POPL's ConfigNormalization: normalize-T,
-- den-final, final-unique, ground-canonicity; its D12), generic in the
-- right module.
--
-- Data: those of the machine (Semantics/Machine.agda): the syntactic laws,
-- a theory with a free model, a configuration container Q′, a right
-- module R over the free-model monad T, the result type A.
--
-- FINAL CONFIGURATIONS.  A final configuration is a shape of Q′ with a
-- closed value at every hole (Final; POPL's ṽ : T (CVal A)), returned at
-- every hole (retF; POPL's T(ret)(ṽ)).  Its values' closed denotations
-- form an element of Q ⟦ A ⟧ (denF; POPL's T(⟦_⟧)(ṽ)).
--
--   den-final      ⟦ retF ṽ ⟧ = denF ṽ: each returned value denotes η of
--                  its value, and ρ ∘ Q η = id is the module's unit law.
--                  So this holds in ANY right module (POPL: Q = T, μ ∘ T η)
--   final-unique   if ⟦ · ⟧ has a left inverse r on closed values of type
--                  A, two final configurations reached from the same
--                  configuration are equal: their denotations are, by
--                  machine soundness (Machine.sound⟶*), and Q r reads the
--                  values back.  No affinity is needed (as POPL notes)
--
-- With the hypotheses of Semantics.Normalization (finOp, finQ, affinity),
-- in the module WithAffinity:
--
--   normalize          every configuration reaches a final one (POPL's
--                      normalize-T; weakNormalization, its terminal
--                      configuration read as retF ṽ)
--   ground-canonicity  at a ground type (Semantics.Ground) the final
--                      configuration is unique (final-unique at
--                      reflect, reflect-den)
--
-- POPL's canonicity-T, the case of the configuration η M, is stated for
-- Q = T in Semantics.SelfCanonicity (ηC M).

module RMRCanonicity.PRACBPVGeneric.Semantics.ConfigCanonicity
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (laws : Closing.ClosedLaws L O)
  (eqs : Machine.Equations L O) (freeModel : FreeModel (Machine.theory L O eqs))
  (Q′ : IndexedContainer (Category.ob W))
  (R : RightModule (EffectReduction.T (Machine.theory L O eqs) freeModel Q′)
                   (EffectReduction.Q (Machine.theory L O eqs) freeModel Q′))
  (A : Types.ValTy (LockSig.Lk L) (LockSig.Base L))
  where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O using (∅; Val; return; Op; Ar; slot)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Denotation L O using (El; ⟦_⟧v; ⟦_⟧V)
open import RMRCanonicity.PRACBPVGeneric.Semantics.Ground L O using (Ground; reflect; reflect-den)
open Machine.WithLaws L O laws
open Run eqs freeModel Q′ R A
open IndexedContainer Q′ using () renaming (Shape to CShape; Pos to CPos; input to inQ)

open PshHom

private
  module W = Category W

----------------------------------------------------------------------
-- final configurations

-- a closed value at every hole
Final : Type
Final = Σ[ s ∈ CShape .fst ] ((r : CPos s .fst) → Val (∅ (inQ s r)) A)

-- the configuration returning them
retF : Final → Config
retF (s , vs) = s , λ r → return (vs r)

-- their closed denotations, an element of Q ⟦ A ⟧
denF : Final → (ER.Q ⟅ ⟦ A ⟧v ⟆) .fst
denF (s , vs) = s , λ r → ⟦ vs r ⟧V .N-ob (inQ s r) W.id

-- the denotation of a final configuration: ρ ∘ Q η = id
den-final : (ṽ : Final) → denote (retF ṽ) ≡ denF ṽ
den-final ṽ = funExt⁻ (RightModule.ρ-unit R ⟦ A ⟧v) (denF ṽ)

-- uniqueness of the final configuration, given a left inverse r of the
-- closed denotation at A (D12)
final-unique : (r : ∀ {w} → El ⟦ A ⟧v w → Val (∅ w) A) →
  (∀ {w} (V : Val (∅ w) A) → r (⟦ V ⟧V .N-ob w W.id) ≡ V) →
  ∀ {x} (ṽ₁ ṽ₂ : Final) → x ⟶* retF ṽ₁ → x ⟶* retF ṽ₂ → ṽ₁ ≡ ṽ₂
final-unique r retr ṽ₁ ṽ₂ p₁ p₂ = sym (readback ṽ₁) ∙ cong mapR same ∙ readback ṽ₂
  where
  same : denF ṽ₁ ≡ denF ṽ₂
  same = sym (den-final ṽ₁) ∙ sym (sound⟶* p₁) ∙ sound⟶* p₂ ∙ den-final ṽ₂

  -- Q r
  mapR : (ER.Q ⟅ ⟦ A ⟧v ⟆) .fst → Final
  mapR (s , as) = s , λ p → r (as p)

  readback : ∀ ṽ → mapR (denF ṽ) ≡ ṽ
  readback (s , vs) = ΣPathP (refl , funExt λ p → retr (vs p))

-- existence, under the hypotheses of Semantics.Normalization

module WithAffinity
  -- every operation has finitely many continuations at every world
  (finOp : ∀ (o : Op) w → isFinSet (Σ[ a ∈ Ar o ] Notation.WPos L (slot o a) w))
  -- every configuration shape has finitely many holes, numbered
  (finQ : ∀ s → isFinOrd (CPos s .fst))
  -- the module is affine
  (aff : Affinity.ModuleAffine (Machine.theory L O eqs) freeModel Q′ R)
  where

  open Normalization L O laws eqs freeModel Q′ R A finOp finQ aff
    using (Terminal; weakNormalization)

  -- a terminal configuration is final
  terminal-final : (y : Config) → Terminal y → Σ[ ṽ ∈ Final ] y ≡ retF ṽ
  terminal-final y t = (y .fst , λ r → t r .fst) , ΣPathP (refl , funExt λ r → t r .snd)

  -- every configuration reaches a final one
  normalize : ∀ x → Σ[ ṽ ∈ Final ] x ⟶* retF ṽ
  normalize x =
    terminal-final y t .fst , subst (x ⟶*_) (terminal-final y t .snd) rs
    where
    y = weakNormalization x .fst
    rs = weakNormalization x .snd .fst
    t = weakNormalization x .snd .snd

  -- at a ground type, it is unique
  ground-canonicity : Ground A → ∀ x →
    Σ[ ṽ ∈ Final ] (x ⟶* retF ṽ) × (∀ ṽ' → x ⟶* retF ṽ' → ṽ ≡ ṽ')
  ground-canonicity g x =
    normalize x .fst , normalize x .snd ,
    λ ṽ' p' → final-unique (reflect g) (reflect-den g) (normalize x .fst) ṽ' (normalize x .snd) p'
