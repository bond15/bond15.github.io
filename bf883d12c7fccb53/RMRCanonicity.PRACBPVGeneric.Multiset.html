-- One-step Dershowitz–Manna on lists (Thm 2.2 of the build spec).
--
-- Fix a relation _<_ on A.  A list ys is one step below a list xs when ys
-- arises from xs by replacing ONE element x by a finite list `new` of
-- elements each strictly below x, in place:
--
--     ys ≺ₘ xs  :=  Σ pre post x new.
--                     xs ≡ pre ++ x ∷ post   ×   ys ≡ pre ++ new ++ post
--                     × every element of new is < x.
--
-- This is the one-step, order-preserving replacement relation.  Its
-- transitive closure, read on the underlying multisets, lies inside the
-- Dershowitz–Manna multiset order (remark only; nothing here needs
-- permutations).
--
-- Theorem (Dershowitz–Manna, one-step form).  If _<_ is well founded, so is
-- _≺ₘ_.  Proof: by a double induction.  The outer induction is on x with
-- a generalised hypothesis that quantifies over the tail:
--
--     acc-cons : Acc x → ∀ ys → Acc ys → Acc (x ∷ ys).
--
-- The inner induction is on Acc ys.  Take zs ≺ₘ x ∷ ys.
--   * The replaced element is the head x: then zs = new ++ ys with every
--     element of new below x.  Induction on new; each y ∷ (new' ++ ys)
--     is accessible by the OUTER hypothesis at y < x.
--   * The replaced element is in the tail: then zs = x ∷ ys' with
--     ys' ≺ₘ ys, and the INNER hypothesis applies.
-- Finally the empty list has nothing below it, and an induction on the
-- list gives acc-all; dm-wf follows.
--
--   All, all, All-++         lists all of whose elements satisfy P
--   _≺ₘ_                     one-step replacement
--   ≺ₘ-replace, ≺ₘ-single    its constructor forms
--   acc-cons (with acc-new, below-cons), acc-[], acc-all, dm-wf
--   dm-wf-ℕ                  the smoke test at (ℕ , <)
module RMRCanonicity.PRACBPVGeneric.Multiset where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Sigma
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.Nat.Order using (_<_; <-wellfounded)
open import Cubical.Data.List.Base using (List; []; _∷_; _++_; [_])
open import Cubical.Data.List.Properties using (cons-inj₁; cons-inj₂; ¬nil≡cons)
import Cubical.Data.Empty as ⊥
open import Cubical.Induction.WellFounded using (Acc; acc; WellFounded)

private
  variable
    ℓ ℓ' ℓ'' : Level

-- lists all of whose elements satisfy P
data All {A : Type ℓ} (P : A → Type ℓ') : List A → Type (ℓ-max ℓ ℓ') where
  []  : All P []
  _∷_ : ∀ {x xs} → P x → All P xs → All P (x ∷ xs)

module _ {A : Type ℓ} {P : A → Type ℓ'} where

  all : (∀ x → P x) → ∀ xs → All P xs
  all p [] = []
  all p (x ∷ xs) = p x ∷ all p xs

  All-++ : ∀ {xs ys} → All P xs → All P ys → All P (xs ++ ys)
  All-++ [] q = q
  All-++ (p ∷ ps) q = p ∷ All-++ ps q

module _ {A : Type ℓ} (_<_ : A → A → Type ℓ') where

  -- one-step, order-preserving replacement of one element by smaller ones
  _≺ₘ_ : List A → List A → Type (ℓ-max ℓ ℓ')
  ys ≺ₘ xs =
    Σ[ pre ∈ List A ] Σ[ post ∈ List A ] Σ[ x ∈ A ] Σ[ new ∈ List A ]
      (xs ≡ pre ++ x ∷ post) × (ys ≡ pre ++ new ++ post) × All (_< x) new

module _ {A : Type ℓ} {_<_ : A → A → Type ℓ'} where

  private
    _≺_ = _≺ₘ_ _<_

  ≺ₘ-replace : ∀ pre post {x} {new} → All (_< x) new →
    (pre ++ new ++ post) ≺ (pre ++ x ∷ post)
  ≺ₘ-replace pre post {x} {new} a = pre , post , x , new , refl , refl , a

  ≺ₘ-single : ∀ pre post {x y} → y < x → (pre ++ y ∷ post) ≺ (pre ++ x ∷ post)
  ≺ₘ-single pre post y<x = ≺ₘ-replace pre post (y<x ∷ [])

  -- nothing is below the empty list
  acc-[] : Acc _≺_ []
  acc-[] = acc λ { zs ([] , post , x , new , p , _) → ⊥.rec (¬nil≡cons p)
                 ; zs ((_ ∷ _) , post , x , new , p , _) → ⊥.rec (¬nil≡cons p) }

  -- The generalised outer hypothesis quantifies over the tail.  The two
  -- inductions are on the Acc arguments, passed explicitly so that the
  -- termination checker sees the lexicographic decrease (Acc x, Acc ys).
  acc-cons : ∀ {x} → Acc _<_ x → ∀ ys → Acc _≺_ ys → Acc _≺_ (x ∷ ys)

  -- the head x was replaced by `new`: induction on new, each element
  -- accessible by the outer hypothesis
  acc-new : ∀ {x ys} → Acc _<_ x → Acc _≺_ ys → ∀ new → All (_< x) new → Acc _≺_ (new ++ ys)

  -- everything one step below x ∷ ys is accessible
  below-cons : ∀ {x ys} → Acc _<_ x → Acc _≺_ ys → ∀ zs → zs ≺ (x ∷ ys) → Acc _≺_ zs

  acc-cons ax ys ays = acc (below-cons ax ays)

  acc-new ax ays [] [] = ays
  acc-new {ys = ys} ax@(acc rx) ays (y ∷ new') (y<x ∷ a) =
    acc-cons (rx y y<x) (new' ++ ys) (acc-new ax ays new' a)

  -- the replaced element is the head
  below-cons ax ays zs ([] , post , x' , new , p , q , a) =
    subst (Acc _≺_)
      (cong (new ++_) (cons-inj₂ p) ∙ sym q)
      (acc-new ax ays new (subst (λ z → All (_< z) new) (sym (cons-inj₁ p)) a))
  -- the replaced element is in the tail: the inner hypothesis
  below-cons ax (acc rys) zs ((x₀ ∷ pre') , post , x' , new , p , q , a) =
    subst (Acc _≺_)
      (cong (_∷ (pre' ++ new ++ post)) (cons-inj₁ p) ∙ sym q)
      (acc-cons ax (pre' ++ new ++ post)
        (rys _ (pre' , post , x' , new , cons-inj₂ p , refl , a)))

  acc-all : ∀ {xs} → All (Acc _<_) xs → Acc _≺_ xs
  acc-all [] = acc-[]
  acc-all {x ∷ xs} (ax ∷ axs) = acc-cons ax xs (acc-all axs)

  -- Theorem 2.2 (one-step Dershowitz–Manna on lists)
  dm-wf : WellFounded _<_ → WellFounded _≺_
  dm-wf wf xs = acc-all (all wf xs)

-- smoke test: the instance at the natural numbers
dm-wf-ℕ : WellFounded (_≺ₘ_ {A = ℕ} _<_)
dm-wf-ℕ = dm-wf <-wellfounded
