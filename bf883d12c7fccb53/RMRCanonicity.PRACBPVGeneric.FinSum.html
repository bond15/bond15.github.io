-- Sums of ℕ-valued functions over finite sets, and search in a finite
-- family (pure mathematics, for the normalization measure).
--
-- For a finite set X and f : X → ℕ, sum X f is the cubical library's
-- Σ_{x ∈ X} f x (the cardinality of Σ_x Fin (f x)).  The facts used:
--
--   sum-ext        pointwise equal functions have equal sums
--   sum-zero       a sum of zeros is zero
--   sum-inj        a sum over an injection into Y is at most the sum over Y
--   Minus X p      X without the point p (X discrete)
--   sum-split      Σ_X f = f p + Σ_{X - p} f
--   search         on Fin n, either every element is good or one is bad
--                  (constructive: the first bad one in the enumeration)
--
-- Monotonicity (sum≤) and sums over a coproduct (sum⊎) are the library's.
module RMRCanonicity.PRACBPVGeneric.FinSum where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Equiv using (_≃_)
open import Cubical.Foundations.Isomorphism using (iso; isoToEquiv)
open import Cubical.Foundations.Function using (_∘_)
open import Cubical.Functions.Embedding using (injEmbedding)
open import Cubical.HITs.PropositionalTruncation using (∣_∣₁)
open import Cubical.Data.Empty as ⊥ using ()
open import Cubical.Data.Nat using (ℕ; zero; suc; _+_)
open import Cubical.Data.Nat.Order using (_≤_)
open import Cubical.Data.Unit using (Unit*; tt*; tt)
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Data.Sigma using (Σ-syntax; ΣPathP; Σ-cong-equiv-fst)
open import Cubical.Data.SumFin using (Fin)
open import Cubical.Data.FinSet.Base using (FinSet; isFinSet→isSet)
open import Cubical.Data.FinSet.Properties using (isFinSetFin; isDecProp→isFinSet)
open import Cubical.Data.FinSet.Constructors using (isFinSetΣ; isFinSet⊎)
open import Cubical.Data.FinSet.Induction using (𝟙)
open import Cubical.Data.FinSet.Cardinality using (sum; sum⊎; sum𝟙; sumConst; cardEquiv; card↪Inequality')
open import Cubical.Relation.Nullary using (¬_; Discrete; Dec; yes; no; isProp¬; Dec¬)

open import Cubical.Data.FinSet.Cardinality using (sum; sum≤; sum⊎) public

private
  variable
    ℓ : Level

sum-ext : (X : FinSet ℓ) {f g : X .fst → ℕ} → (∀ x → f x ≡ g x) → sum X f ≡ sum X g
sum-ext X h = cong (sum X) (funExt h)

sum-zero : (X : FinSet ℓ) (f : X .fst → ℕ) → (∀ x → f x ≡ 0) → sum X f ≡ 0
sum-zero X f h = sumConst X f 0 h

-- sums are invariant under equivalence
private
  sum-≃ : (X Y : FinSet ℓ) (e : X .fst ≃ Y .fst) (g : Y .fst → ℕ) →
    sum X (λ x → g (e .fst x)) ≡ sum Y g
  sum-≃ X Y e g =
    cardEquiv (_ , isFinSetΣ X (λ x → Fin (g (e .fst x)) , isFinSetFin))
              (_ , isFinSetΣ Y (λ y → Fin (g y) , isFinSetFin))
              ∣ Σ-cong-equiv-fst e ∣₁

-- a sum over an injection is at most the full sum
sum-inj : (X Y : FinSet ℓ) (e : X .fst → Y .fst) →
  (∀ x x' → e x ≡ e x' → x ≡ x') → (g : Y .fst → ℕ) →
  sum X (g ∘ e) ≤ sum Y g
sum-inj X Y e inj g =
  card↪Inequality' (_ , isFinSetΣ X (λ x → Fin (g (e x)) , isFinSetFin))
                   (_ , isFinSetΣ Y (λ y → Fin (g y) , isFinSetFin))
                   (λ (x , i) → e x , i)
                   (injEmbedding (isFinSet→isSet (isFinSetΣ Y (λ y → Fin (g y) , isFinSetFin))) inj')
  where
  inj' : ∀ {a b : Σ[ x ∈ X .fst ] Fin (g (e x))} →
    (e (a .fst) , a .snd) ≡ (e (b .fst) , b .snd) → a ≡ b
  inj' {x , i} {x' , i'} q = ΣPathP (xx ,
    subst (λ (π : e x ≡ e x') → PathP (λ j → Fin (g (π j))) i i')
      (isFinSet→isSet (Y .snd) _ _ (cong fst q) (cong e xx))
      (cong snd q))
    where xx = inj x x' (cong fst q)

------------------------------------------------------------------------
-- removing a point of a discrete finite set

module _ (X : FinSet ℓ) (dec : Discrete (X .fst)) (p : X .fst) where

  Minus : FinSet ℓ
  Minus = (Σ[ r ∈ X .fst ] ¬ r ≡ p) ,
    isFinSetΣ X (λ r → (¬ r ≡ p) , isDecProp→isFinSet (isProp¬ _) (Dec¬ (dec r p)))

  private
    from : Unit* {ℓ} ⊎ Minus .fst → X .fst
    from (inl _) = p
    from (inr (r , _)) = r

    rest : (r : X .fst) → ¬ r ≡ p → Unit* {ℓ} ⊎ Minus .fst
    rest r ne = inr (r , ne)

    to' : (r : X .fst) → Dec (r ≡ p) → Unit* {ℓ} ⊎ Minus .fst
    to' r (yes _) = inl tt*
    to' r (no ne) = rest r ne

    from-to : (r : X .fst) (d : Dec (r ≡ p)) → from (to' r d) ≡ r
    from-to r (yes e) = sym e
    from-to r (no _) = refl

    to-from : (y : Unit* ⊎ Minus .fst) → to' (from y) (dec (from y) p) ≡ y
    to-from (inl _) with dec p p
    ... | yes _ = refl
    ... | no ne = ⊥.rec (ne refl)
    to-from (inr (r , ne)) with dec r p
    ... | yes e = ⊥.rec (ne e)
    ... | no ne' = cong (rest r) (isProp¬ _ ne' ne)

    split≃ : (Unit* ⊎ Minus .fst) ≃ X .fst
    split≃ = isoToEquiv (iso from (λ r → to' r (dec r p)) (λ r → from-to r (dec r p)) to-from)

  -- the sum is the value at p plus the sum over the rest
  sum-split : (f : X .fst → ℕ) → sum X f ≡ f p + sum Minus (f ∘ fst)
  sum-split f =
    sym (sum-≃ (_ , isFinSet⊎ 𝟙 Minus) X split≃ f)
    ∙ sum⊎ 𝟙 Minus (f ∘ from)
    ∙ cong (_+ sum Minus (f ∘ fst)) (sum𝟙 (λ (_ : Unit* {ℓ}) → f p))

-- search a finite family for a bad element

search : ∀ n (Good Bad : Fin n → Type ℓ) → (∀ i → Good i ⊎ Bad i) →
  (∀ i → Good i) ⊎ (Σ[ i ∈ Fin n ] Bad i)
search zero Good Bad c = inl λ ()
search (suc n) Good Bad c = found (c (inl tt)) (search n (Good ∘ inr) (Bad ∘ inr) (c ∘ inr))
  where
  found : Good (inl tt) ⊎ Bad (inl tt) →
    (∀ i → Good (inr i)) ⊎ (Σ[ i ∈ Fin n ] Bad (inr i)) →
    (∀ i → Good i) ⊎ (Σ[ i ∈ Fin (suc n) ] Bad i)
  found (inr b) _ = inr (inl tt , b)
  found (inl g) (inl all) = inl λ { (inl tt) → g ; (inr i) → all i }
  found (inl g) (inr (i , b)) = inr (inr i , b)
