-- E κ at a world map, read with the strict identity
pos-smapW : ∀ κ {w w'} (f : WMap w w') (p : Pos κ w') →
  Path (El (K.Ey κ w) (scope κ w' p)) (K.pos κ (toW f) p , K.smap κ (toW f) p)
    (posW κ f p , toW (smapW κ f p))
pos-smapW κ idW p = ΣPathP (K.pos-id κ p , K.smap-id κ p)
pos-smapW κ (mapW f) p = refl

------------------------------------------------------------------------
-- the denotation of a renaming

⟦_⟧r : ∀ {Θ Θ'} → Ren Θ Θ' → PshHom ⟦ Θ' ⟧t ⟦ Θ ⟧t
ren-over : ∀ {Θ Θ'} (ρ : Ren Θ Θ') v (e : El ⟦ Θ' ⟧t v) →
  πt Θ .N-ob v (⟦ ρ ⟧r .N-ob v e) ≡ toW (wm ρ) W.⋆ πt Θ' .N-ob v e

⟦ idR ⟧r = idPshHom
⟦ root {Θ' = Θ'} f ⟧r = πt Θ' ⋆PshHom Ymap (toW f)
⟦ keep ρ ⟧r = ×PshIntro (π₁ _ _ ⋆PshHom ⟦ ρ ⟧r) (π₂ _ _)
⟦ drop ρ ⟧r = π₁ _ _ ⋆PshHom ⟦ ρ ⟧r
⟦ keepL {Δ} {Δ'} {κ} ρ p ⟧r =
  K.Lsum κ (πt Δ') (πt Δ) ⟦ ρ ⟧r (toW (wm ρ)) (ren-over ρ)
    p (posW κ (wm ρ) p) (toW (smapW κ (wm ρ) p)) (pos-smapW κ (wm ρ) p)
⟦ dropL {Δ' = Δ'} {κ} ρ p ⟧r = incl {Δ'} {κ} {p} ⋆PshHom K.ε κ ⟦ Δ' ⟧t ⋆PshHom ⟦ ρ ⟧r

ren-over idR v e = sym (W.⋆IdL _)
ren-over (root f) v e = refl
ren-over (keep ρ) v e = ren-over ρ v (e .fst)
ren-over (drop ρ) v e = ren-over ρ v (e .fst)
ren-over (keepL ρ p) v e = refl
ren-over (dropL {Δ' = Δ'} {κ} ρ p) v s =
  ren-over ρ v _
  ∙ cong (toW (wm ρ) W.⋆_) (K.ε-over κ (πt Δ') s)
  ∙ sym (W.⋆Assoc _ _ _)
  ∙ cong (W._⋆ s .snd .fst) (sym (toW-⋆ (wm ρ) (mapW (K.ι κ (world Δ') p))))

-- the denotation commutes with renaming

den-renNe : ∀ {Θ Θ' A} (ρ : Ren Θ Θ') (n : Ne Θ A) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ renNe ρ n ⟧ne .N-ob v e ≡ ⟦ n ⟧ne .N-ob v (⟦ ρ ⟧r .N-ob v e)
den-renNe idR n v e = refl
den-renNe (root f) () v e
den-renNe (keep ρ) here v e = refl
den-renNe (keep ρ) (wkv n) v e = den-renNe ρ n v (e .fst)
den-renNe (drop ρ) n v e = den-renNe ρ n v (e .fst)
den-renNe (keepL {κ = κ} ρ p) (up {A = A} n) v (x , _) =
  K.unlock-Lmap-at κ ⟦ ρ ⟧r (⟦ n ⟧ne ⋆PshHom K.box κ ⟦ A ⟧v) (⟦ renNe ρ n ⟧ne ⋆PshHom K.box κ ⟦ A ⟧v)
    (λ v e → cong (K.box κ ⟦ A ⟧v .N-ob v) (den-renNe ρ n v e)) x
den-renNe (keepL {κ = κ} ρ p) (opn n) v (x , _) =
  K.unlock-Lmap-at κ ⟦ ρ ⟧r ⟦ n ⟧ne ⟦ renNe ρ n ⟧ne (den-renNe ρ n) x
den-renNe (keepL {κ = κ} ρ p) (nm k) v (x , _) =
  K.unlock-Lmap-at κ ⟦ ρ ⟧r (namesR κ k) (namesR κ k) (λ _ _ → refl) x
den-renNe (dropL {Δ' = Δ'} {κ} ρ p) n v (x , _) =
  K.unlock-box κ ⟦ renNe ρ n ⟧ne x ∙ den-renNe ρ n v (K.ε κ ⟦ Δ' ⟧t .N-ob v x)

-- continuations behind a lock word: shutW commutes with renaming
shutW-ren : ∀ {Θ Θ'} {X : 𝒱} (μ : List Lk) (ρ : Ren Θ Θ')
  (k : (ps : WPos μ (world Θ)) → PshHom ⟦ Θ ⧀*⟨ μ , ps ⟩ ⟧t X)
  (k' : (ps : WPos μ (world Θ')) → PshHom ⟦ Θ' ⧀*⟨ μ , ps ⟩ ⟧t X) →
  (∀ ps v e → k' ps .N-ob v e ≡ k (wposW μ (wm ρ) ps) .N-ob v (⟦ keepW μ ρ ps ⟧r .N-ob v e)) →
  ∀ v e → shutW μ k' .N-ob v e ≡ shutW μ k .N-ob v (⟦ ρ ⟧r .N-ob v e)
shutW-ren [] ρ k k' hyp v e = funExt λ _ → hyp _ v e
shutW-ren {Θ} {Θ'} (κ ∷ []) ρ k k' hyp v e = funExt λ q →
  hyp _ _ (K.Over.inj κ (πt Θ') e q)
  ∙ sym (K.Over.copair-at κ (πt Θ) k _ _ _)
  ∙ K.Over.copair-at κ (πt Θ) k _ _ _
shutW-ren {Θ} {Θ'} (κ ∷ μ@(_ ∷ _)) ρ k k' hyp v e = funExt λ (q , ps) →
  funExt⁻ (shutW-ren μ (keepL ρ _) (λ ps → k (_ , ps)) (λ ps → k' (_ , ps))
            (λ ps → hyp (_ , ps)) _ (K.Over.inj κ (πt Θ') e q)) ps
  ∙ funExt⁻ (sym (K.Over.copair-at κ (πt Θ) φ _ _ _) ∙ K.Over.copair-at κ (πt Θ) φ _ _ _) ps
  where
  φ : (p : Pos κ (world Θ)) → PshHom ⟦ Θ ⧀⟨ κ , p ⟩ ⟧t (KW.R μ _)
  φ p = shutW μ (λ ps → k (p , ps))

den-renV : ∀ {Θ Θ' A} (ρ : Ren Θ Θ') (V : Val Θ A) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ renV ρ V ⟧V .N-ob v e ≡ ⟦ V ⟧V .N-ob v (⟦ ρ ⟧r .N-ob v e)
den-renVs : ∀ {Θ Θ' bs} (ρ : Ren Θ Θ') (vs : Vals Θ bs) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ renVs ρ vs ⟧Vs .N-ob v e ≡ ⟦ vs ⟧Vs .N-ob v (⟦ ρ ⟧r .N-ob v e)
den-renC : ∀ {Θ Θ' B} (ρ : Ren Θ Θ') (M : Comp Θ B) v (e : El ⟦ Θ' ⟧t v) →
  ⟦ renC ρ M ⟧C .N-ob v e ≡ ⟦ M ⟧C .N-ob v (⟦ ρ ⟧r .N-ob v e)

den-renV ρ (Syn.ne n) = den-renNe ρ n
den-renV ρ Syn.tt v e = refl
den-renV ρ (Syn.inl V) v e = cong inl (den-renV ρ V v e)
den-renV ρ (Syn.inr V) v e = cong inr (den-renV ρ V v e)
den-renV ρ (Syn.pair V V') v e = cong₂ _,_ (den-renV ρ V v e) (den-renV ρ V' v e)
den-renV ρ (Syn.thunk M) = den-renC ρ M
den-renV {Θ} {Θ'} ρ (Syn.const {b} c) v e =
  cong (Const b .F-hom (πt Θ' .N-ob v e)) (cmapW-toW (wm ρ) c)
  ∙ sym (funExt⁻ (Const b .F-seq (toW (wm ρ)) (πt Θ' .N-ob v e)) c)
  ∙ cong (λ h → Const b .F-hom h c) (sym (ren-over ρ v e))
  where open Syn using (Const)
den-renV {Θ} {Θ'} ρ (Syn.shut {κ} V) v e = funExt λ q →
  den-renV (keepL ρ _) (V _) _ (K.Over.inj κ (πt Θ') e q)
  ∙ sym (K.Over.copair-at κ (πt Θ) (λ p → ⟦ V p ⟧V) _ _ _)
  ∙ K.Over.copair-at κ (πt Θ) (λ p → ⟦ V p ⟧V) _ _ _

den-renVs ρ Syn.[]ᵥ v e = refl
den-renVs ρ (V Syn.∷ᵥ vs) v e = cong₂ _,_ (den-renV ρ V v e) (den-renVs ρ vs v e)

den-renC ρ (Syn.return {A} V) v e = cong (η ⟦ A ⟧v .N-ob v) (den-renV ρ V v e)
den-renC {Θ} {Θ'} ρ (Syn._to_ {A} {B} M N) v e =
  cong (λ t → bind ⟦ Θ' ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ renC (keep ρ) N ⟧C .N-ob v (t , e))
    (den-renC ρ M v e)
  ∙ bind-pre ⟦ Θ ⟧t ⟦ Θ' ⟧t ⟦ A ⟧v ⟦ B ⟧m ⟦ ρ ⟧r ⟦ N ⟧C ⟦ renC (keep ρ) N ⟧C
      (λ u x' a → den-renC (keep ρ) N u (x' , a))
      v (⟦ M ⟧C .N-ob v (⟦ ρ ⟧r .N-ob v e)) e
den-renC ρ (Syn.force V) = den-renV ρ V
den-renC ρ (Syn.case V M) v e =
  den-renC (keep (keep ρ)) M v _
  ∙ cong (λ y → ⟦ M ⟧C .N-ob v ((⟦ ρ ⟧r .N-ob v e , y .fst) , y .snd)) (den-renV ρ V v e)
den-renC ρ (Syn.case₊ V M N) v e =
  cong (λ s → casePsh ⟦ renC (keep ρ) M ⟧C ⟦ renC (keep ρ) N ⟧C .N-ob v (e , s)) (den-renV ρ V v e)
  ∙ casePsh-pt ⟦ renC (keep ρ) M ⟧C ⟦ renC (keep ρ) N ⟧C ⟦ M ⟧C ⟦ N ⟧C e (⟦ ρ ⟧r .N-ob v e)
      (λ a → den-renC (keep ρ) M v (e , a)) (λ a → den-renC (keep ρ) N v (e , a))
      (⟦ V ⟧V .N-ob v (⟦ ρ ⟧r .N-ob v e))
den-renC ρ (Syn.absurd V) v e = ⊥.rec (⟦ V ⟧V .N-ob v (⟦ ρ ⟧r .N-ob v e))
den-renC ρ (Syn.lam M) v e = makePshHomPath (funExt λ u → funExt λ (h , a) →
  den-renC (keep ρ) M u (_ , a)
  ∙ cong (λ y → ⟦ M ⟧C .N-ob u (y , a)) (⟦ ρ ⟧r .N-hom _ _ h e))
den-renC ρ (Syn.app M V) v e =
  cong₂ (λ f a → f .N-ob v (W.id , a)) (den-renC ρ M v e) (den-renV ρ V v e)
den-renC {Θ} {Θ'} ρ (Syn.op {B} o vs k) v e =
  cong (Algebra.algebra ⟦ B ⟧c .NatTrans.N-ob v)
    (ΣPathP (ΣPathP (refl , den-renVs ρ vs v e) , funExt λ (a , ps) →
      funExt⁻ (shutW-ren (slot o a) ρ (k' a) (λ ps → ⟦ renC (keepW (slot o a) ρ ps) (k a _) ⟧C)
        (λ ps → den-renC (keepW (slot o a) ρ ps) (k a _)) v e) ps))
  where
  k' : ∀ a (ps : WPos (slot o a) (world Θ)) → PshHom ⟦ Θ ⧀*⟨ slot o a , ps ⟩ ⟧t (Algebra.Carrier ⟦ B ⟧c)
  k' a ps = ⟦ k a ps ⟧C
