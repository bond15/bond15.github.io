-- Strong and weak normalization of the combined machine under module
-- affinity (C.1-C.3).
--
-- Data: those of the machine (Semantics/Machine.agda): the syntactic laws,
-- a theory with a free model, a configuration container Q′, a right module
-- R, the result type A.  Hypotheses:
--
--   finOp   every operation has finitely many continuations at every
--           world (Σ a. WPos (slot o a) w is a finite set)
--   finQ    every configuration shape has finitely many holes, with a
--           chosen numbering (isFinOrd; the numbering only picks which
--           hole progress steps, see below)
--   aff     the module is affine (Affinity.ModuleAffine): at the generic
--           redex distinct holes of the result have distinct origins
--
-- THE MEASURE.  Every closed computation M : Comp (∅ w) (F A) has a
-- derivation of the logical relation (Termination.related), and all its
-- derivations have the same height, sz M (TreeSize).  A configuration is
-- measured by the multiset of the sizes of its holes, written as its
-- counting function (FinCount.count: n ↦ how many holes have size ≥ n),
-- ordered by the multiset order _⊏_.
--
-- C.1 (decrease).  Every machine step lowers the measure.
--   pure     the hole's computation steps, so its size drops by one
--            (sz-step); the other holes are unchanged (count-update)
--   effect   the hole holds op o vs k, of size 1 + the largest of its
--            continuations.  The step's result is the generic step renamed
--            (Affinity.represent): each hole of the result comes from a
--            source, injectively (aff), and holds the source's filler
--            reindexed.  A source is another hole (size not raised by
--            reindexing, sz-reindex) or a continuation, closed
--            (Closing.closeK) and reindexed: smaller than op o vs k
--            (sz-cont).  So one element of the multiset is replaced by
--            smaller ones and the others are moved injectively, not raised
--            (count-replace).
--   Neither case needs the module to keep the number of holes, or the
--   world maps it moves fillers along to be bijective on positions: the
--   size is a height, which reindexing cannot raise.
--
-- C.2 (strongNormalization).  Every configuration of closed computations
-- is accessible for the converse of _⟶_: its measure vanishes above the
-- largest size of a hole, and _⊏_ is well founded there
-- (FinCount.⊏-vanishing-acc).
--
-- C.3.  A configuration is terminal when every hole returns a value.
-- progress: every configuration is terminal or steps (the first hole, in
-- finQ's numbering, whose derivation is not a leaf: a back step is a pure
-- step, a node an effect step).  weakNormalization: every configuration
-- reaches a terminal one (by C.2 and progress), and wn-sound: with the
-- same denotation (Machine.sound⟶*).

private
  module Aff = Affinity (theory eqs) freeModel Q′

-- the holes of a shape, a finite set
HoleSet : CShape .fst → FinSet ℓ-zero
HoleSet s = CPos s .fst , isFinOrd→isFinSet (finQ s)

----------------------------------------------------------------------
-- the measure

-- the multiset of the sizes of the holes, as a counting function
measure : Config → ℕ → ℕ
measure x = count (HoleSet (x .fst)) λ r → sz (x .snd r)

-- the result y of an effect step, presented: each hole of y comes from a
-- source (org, injective) and holds its filler src (org r), reindexed
record Presented (s : CShape .fst) (p : CPos s .fst) (o : Shape (inQ s p))
  (src : (σ : Aff.Source R s p o) → Comp (∅ (Aff.srcWorld R s p o σ)) (F A))
  (y : Config) : Type where
  field
    org : CPos (y .fst) .fst → Aff.Source R s p o
    org-inj : ∀ r₁ r₂ → org r₁ ≡ org r₂ → r₁ ≡ r₂
    mv : ∀ r → W [ Aff.srcWorld R s p o (org r) , inQ (y .fst) r ]
    fl : ∀ r → y .snd r ≡ reindexC (mv r) (src (org r))
open Presented

-- C.1: every step lowers the measure
decrease : ∀ {x y} → x ⟶ y → measure y ⊏ measure x
decrease (pure C@((s , p) , a) {M} {M'} st) =
  count-update (HoleSet s) (decPos s) p (λ r → sz (plugQ Q′ (CompPsh (F A)) C M .snd r)) (λ r → sz (plugQ Q′ (CompPsh (F A)) C M' .snd r))
    (subst2 _<_ (cong sz (sym (plug-hole Q′ (CompPsh (F A)) C M'))) (cong sz (sym (plug-hole Q′ (CompPsh (F A)) C M)))
      (≤-reflexive (sym (sz-step st))))
    λ r ne → ≤-reflexive (cong sz (plug-remaining Q′ (CompPsh (F A)) C M' r ne ∙ sym (plug-remaining Q′ (CompPsh (F A)) C M r ne)))
decrease (effect C@((s , p) , a) o vs k) =
  count-replace (HoleSet s) (decPos s) p before (HoleSet (y .fst)) S (λ r → sz (y .snd r)) wt wt<
    (pr .org) (pr .org-inj) bound
  where
  w = inQ s p
  S = Conts o w

  -- the closed continuations and the closed operation node's arguments
  ks : (q : S .fst) → Comp (∅ _) (F A)
  ks q = closeK (slot o (q .fst)) (k (q .fst)) (q .snd)

  o′ : Shape w
  o′ = o , closeVals vs

  y = ER.step R (CompAlg (F A)) C (o′ , ks)

  src = Aff.srcFill R (CompPsh (F A)) C (o′ , ks)

  -- the generic step renamed, presented by its origins
  prG : Presented s p o′ src ((ER.Q ⟪ Aff.env R (CompPsh (F A)) C (o′ , ks) ⟫) (Aff.generic R s p o′))
  prG .org = Aff.origin R s p o′
  prG .org-inj = aff s p o′
  prG .mv = Aff.moved R s p o′
  prG .fl r = refl

  pr : Presented s p o′ src y
  pr = subst (Presented s p o′ src) (sym (Aff.represent R (CompPsh (F A)) C (o′ , ks))) prG

  -- the sizes before the step, and of the continuations
  before : CPos s .fst → ℕ
  before r = sz (plugQ Q′ (CompPsh (F A)) C (op o vs k) .snd r)

  wt : S .fst → ℕ
  wt q = sz (ks q)

  wt< : ∀ q → wt q < before p
  wt< q = subst (wt q <_) (cong sz (sym (plug-hole Q′ (CompPsh (F A)) C (op o vs k)))) (sz-cont o vs k (q .fst) (q .snd))

  -- a source's filler is no larger than the source
  src≤ : ∀ σ → sz (src σ) ≤ srcHeight (HoleSet s) (decPos s) p before {S = S} wt σ
  src≤ (inl (r , ne)) = ≤-reflexive (cong sz (sym (plug-remaining Q′ (CompPsh (F A)) C (op o vs k) r ne)))
  src≤ (inr q) = ≤-refl

  bound : ∀ r → sz (y .snd r) ≤ srcHeight (HoleSet s) (decPos s) p before {S = S} wt (pr .org r)
  bound r =
    ≤-trans (≤-reflexive (cong sz (pr .fl r)))
      (≤-trans (sz-reindex (pr .mv r) (src (pr .org r))) (src≤ (pr .org r)))

----------------------------------------------------------------------
-- C.2: strong normalization

private
  sn-acc : ∀ x → Acc _⊏_ (measure x) → Acc (λ y x → x ⟶ y) x
  sn-acc x (acc r) = acc λ y st → sn-acc y (r (measure y) (decrease st))

  -- the largest size of a hole
  largest : Config → ℕ
  largest x = max (HoleSet (x .fst)) λ r → sz (x .snd r)

strongNormalization : ∀ x → Acc (λ y x → x ⟶ y) x
strongNormalization x =
  sn-acc x (⊏-vanishing-acc (largest x) (measure x)
    (count-vanishes (HoleSet (x .fst)) (λ r → sz (x .snd r)) (largest x)
      (max-ub (HoleSet (x .fst)) λ r → sz (x .snd r))))

----------------------------------------------------------------------
-- C.3: progress and weak normalization

-- every hole returns a value
Terminal : Config → Type
Terminal x = ∀ r → Σ[ V ∈ Val (∅ (inQ (x .fst) r)) A ] (x .snd r ≡ return V)

-- what a derivation says about a closed computation: it returns, steps,
-- or is an operation
Returns : ∀ {w} → Comp (∅ w) (F A) → Type
Returns {w} M = Σ[ V ∈ Val (∅ w) A ] (M ≡ return V)

record IsOp {w} (M : Comp (∅ w) (F A)) : Type where
  constructor isOp
  field
    o : Op
    vs : Vals (∅ w) (params o)
    k : (a : Ar o) (ps : WPos (slot o a) w) → Comp (∅ w ⧀*⟨ slot o a , ps ⟩) (F A)
    is : M ≡ op o vs k

Active : ∀ {w} → Comp (∅ w) (F A) → Type
Active {w} M = (Σ[ M' ∈ Comp (∅ w) (F A) ] M ↦ M') ⊎ IsOp M

classify : ∀ {w} {M : Comp (∅ w) (F A)} → Tree 𝒱⟦ A ⟧ (∅ w) M → Returns M ⊎ Active M
classify (leaf {V = V} _) = inl (V , refl)
classify (back st _) = inr (inl (_ , st))
classify (node o {vs} {k} _) = inr (inr (isOp o vs k refl))

private
  -- search the holes, in finQ's numbering, for a bad one
  holes : (s : CShape .fst) (Good Bad : CPos s .fst → Type) → (∀ r → Good r ⊎ Bad r) →
    (∀ r → Good r) ⊎ (Σ[ r ∈ CPos s .fst ] Bad r)
  holes s Good Bad c = found (search _ (Good ∘ invEq e) (Bad ∘ invEq e) (c ∘ invEq e))
    where
    e = finQ s .snd

    found : (∀ i → Good (invEq e i)) ⊎ (Σ[ i ∈ _ ] Bad (invEq e i)) →
      (∀ r → Good r) ⊎ (Σ[ r ∈ CPos s .fst ] Bad r)
    found (inl all) = inl λ r → subst Good (retEq e r) (all (equivFun e r))
    found (inr (i , b)) = inr (invEq e i , b)

  -- unplugging a hole and plugging it back
  ctxAt : (x : Config) → CPos (x .fst) .fst → Ctxt
  ctxAt (s , fill) p = (s , p) , λ r _ → fill r

  plug-unplug : (x : Config) (p : CPos (x .fst) .fst) → plugQ Q′ (CompPsh (F A)) (ctxAt x p) (x .snd p) ≡ x
  plug-unplug (s , fill) p = ΣPathP (refl , funExt λ r → at r (decPos s r p))
    where
    C = ctxAt (s , fill) p

    at : (r : CPos s .fst) → Dec (r ≡ p) → plugQ Q′ (CompPsh (F A)) C (fill p) .snd r ≡ fill r
    at r (yes e) =
      subst (λ r → plugQ Q′ (CompPsh (F A)) C (fill p) .snd r ≡ fill r) (sym e) (plug-hole Q′ (CompPsh (F A)) C (fill p))
    at r (no ne) = plug-remaining Q′ (CompPsh (F A)) C (fill p) r ne

  -- a step from an equal configuration
  from : ∀ {x x' y} → x ≡ x' → x ⟶ y → x' ⟶ y
  from {y = y} e st = subst (_⟶ y) e st

-- every configuration is terminal or steps
progress : ∀ x → Terminal x ⊎ (Σ[ y ∈ Config ] (x ⟶ y))
progress x@(s , fill) = found (holes s (λ r → Returns (fill r)) (λ r → Active (fill r))
                                 λ r → classify (rel (fill r)))
  where
  found : (∀ r → Returns (fill r)) ⊎ (Σ[ p ∈ CPos s .fst ] Active (fill p)) →
    Terminal x ⊎ (Σ[ y ∈ Config ] (x ⟶ y))
  found (inl ret) = inl ret
  -- a tree step in hole p
  found (inr (p , inl (M' , st))) = inr (_ , from (plug-unplug x p) (pure (ctxAt x p) st))
  -- an operation in hole p
  found (inr (p , inr (isOp o vs k e))) =
    inr (_ , from (cong (plugQ Q′ (CompPsh (F A)) (ctxAt x p)) (sym e) ∙ plug-unplug x p)
               (effect (ctxAt x p) o vs k))

private
  wn-acc : ∀ x → Acc (λ y x → x ⟶ y) x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
  wn-acc x (acc r) with progress x
  ... | inl t = x , done , t
  ... | inr (y , st) = let (z , rs , t) = wn-acc y (r y st) in z , (st then rs) , t

-- every configuration reaches a terminal one
weakNormalization : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) × Terminal y
weakNormalization x = wn-acc x (strongNormalization x)

-- and its normal form has the same denotation
wn-sound : ∀ x → denote x ≡ denote (weakNormalization x .fst)
wn-sound x = sound⟶* (weakNormalization x .snd .fst)
