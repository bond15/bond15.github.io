-- The RMR polynomial WITH STRENGTH of a lock signature and an operation
-- signature.
--
-- Step 1 (generic).  A functor G : B → Fam W is the positions-and-scopes
-- part of a polynomial over B: its Grothendieck construction gives
--     Pos = (index set of G b),      s (b , i) = (i-th summand of G b),
-- and a natural transformation G ⇒ Yo ∘ π (maps from each G b into a
-- representable) gives a strength  NatTrans s (t ∘ p).
--
-- Step 2 (this signature).  Shapes w = Σ_o Par_o w (operation, typed
-- parameters).  G sends a shape (w , (o , cs)) to E o w, where
--     E o = Σᴬ_{a ∈ Ar o} E (slot o a)          (a sum of composites)
-- is the arity of o: G forgets the parameters (base change along the
-- typing Σ_o Par_o → Op) and applies the arity.  The strength is the sum
-- of the words' strengths, which are composites of the lock kinds' ι.
--
-- Read back (all refl): shapes, positions (Σ (a ∈ Ar o) . WPos (slot o a) w),
-- scopes (the word's scope) and the strength's components (the word's
-- composite ι).  On morphisms the polynomial's reindexing at a canonical
-- lift agrees with the arity's pos / smap up to the transport along the
-- (refl) operation path (pos-is, smap-is).

open import Cubical.Categories.Category using (Category; _[_,_]; _^op)
open import Cubical.Categories.Functor using (Functor; _∘F_)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Instances.BinProduct using (_,F_)
open import Cubical.Categories.Instances.Sets using (SET)
open import Cubical.Categories.Instances.Discrete using (DiscreteCategory; DiscFunc)
open import Cubical.Categories.Instances.Functors using (FUNCTOR; appF)
open import Cubical.Categories.Instances.Elements using (module Contravariant; module Covariant)
import Cubical.Categories.Instances.TotalCategory as TotalCategory
open import Polynomial.Presheaf.Discrete.Base using (Poly)

module RMRCanonicity.PRACBPVGeneric.Polynomial {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L) where

open Category
open Functor
open NatTrans
open Notation L
open OpSig O

private
  module W = Category W

-- the strength of an RMR signature (as in RMRCanonicity.Strength)
Strength : Signature W → Type
Strength P = NatTrans (Poly.s P) (Poly.t P ∘F Poly.p P)

------------------------------------------------------------------------
-- Step 1: the Grothendieck construction of a family-valued functor

module Groth (Shapes : Functor ((W ^op) ^op) (SET ℓ-zero))
  (G : Functor (Contravariant.∫ Shapes) (Fam W)) where
  B : Category ℓ-zero ℓ-zero
  B = Contravariant.∫ Shapes

  private
    X : (b : B .ob) → G .F-ob b .fst .fst → W.ob
    X b = G .F-ob b .snd

    isSetIx : (b : B .ob) → isSet (G .F-ob b .fst .fst)
    isSetIx b = G .F-ob b .fst .snd

  PosG : Functor B (SET ℓ-zero)
  PosG .F-ob b = G .F-ob b .fst
  PosG .F-hom g = G .F-hom g .fst
  PosG .F-id = cong fst (G .F-id)
  PosG .F-seq f g = cong fst (G .F-seq f g)

  -- the scope map of a morphism of positions
  sHom : ∀ {b b'} (g : B [ b , b' ]) {i i'} → G .F-hom g .fst i ≡ i' → W [ X b' i' , X b i ]
  sHom {b} {b'} g {i} pp = subst (λ j → W [ X b' j , X b i ]) pp (G .F-hom g .snd i)

  private
    sHom-id : ∀ {b} {i} (m : Fam W [ G .F-ob b , G .F-ob b ]) → m ≡ idF W (G .F-ob b) →
      (pp : m .fst i ≡ i) → subst (λ j → W [ X b j , X b i ]) pp (m .snd i) ≡ W.id
    sHom-id {b} {i} m e =
      J (λ m' _ → (pp : m' .fst i ≡ i) →
           subst (λ j → W [ X b j , X b i ]) pp (m' .snd i) ≡ W.id)
        (λ pp → cong (λ q → subst (λ j → W [ X b j , X b i ]) q W.id) (isSetIx b i i pp refl)
                ∙ substRefl {B = λ j → W [ X b j , X b i ]} W.id)
        (sym e)

    module _ {b b' b'' : B .ob} (g : B [ b , b' ]) (h : B [ b' , b'' ]) (i : G .F-ob b .fst .fst) where
      private
        φg = G .F-hom g .fst
        φh = G .F-hom h .fst

      κ : W [ X b'' (φh (φg i)) , X b i ]
      κ = G .F-hom h .snd (φg i) W.⋆ G .F-hom g .snd i

      step3 : (path : φh (φg i) ≡ φh (φg i)) →
        subst (λ j → W [ X b'' j , X b i ]) path κ ≡ sHom h {φg i} refl W.⋆ sHom g {i} refl
      step3 path =
        cong (λ q → subst (λ j → W [ X b'' j , X b i ]) q κ) (isSetIx b'' _ _ path refl)
        ∙ substRefl {B = λ j → W [ X b'' j , X b i ]} κ
        ∙ cong₂ W._⋆_
            (sym (substRefl {B = λ j → W [ X b'' j , X b' (φg i) ]} (G .F-hom h .snd (φg i))))
            (sym (substRefl {B = λ j → W [ X b' j , X b i ]} (G .F-hom g .snd i)))

      step : (i' : G .F-ob b' .fst .fst) (pp : φg i ≡ i')
        (i'' : G .F-ob b'' .fst .fst) (pp' : φh i' ≡ i'')
        (path : φh (φg i) ≡ i'') →
        subst (λ j → W [ X b'' j , X b i ]) path κ ≡ sHom h {i'} pp' W.⋆ sHom g {i} pp
      step i' pp =
        J (λ i' pp → (i'' : G .F-ob b'' .fst .fst) (pp' : φh i' ≡ i'')
              (path : φh (φg i) ≡ i'') →
              subst (λ j → W [ X b'' j , X b i ]) path κ ≡ sHom h {i'} pp' W.⋆ sHom g {i} pp)
          (λ i'' pp' → J (λ i'' pp' → (path : φh (φg i) ≡ i'') →
                         subst (λ j → W [ X b'' j , X b i ]) path κ ≡ sHom h {φg i} pp' W.⋆ sHom g {i} refl)
                     step3 pp')
          pp

      sHom-seq : (m : Fam W [ G .F-ob b , G .F-ob b'' ]) →
        m ≡ seqF W (G .F-ob b) (G .F-ob b') (G .F-ob b'') (G .F-hom g) (G .F-hom h) →
        (i' : G .F-ob b' .fst .fst) (pp : φg i ≡ i')
        (i'' : G .F-ob b'' .fst .fst) (pp' : φh i' ≡ i'')
        (path : m .fst i ≡ i'') →
        subst (λ j → W [ X b'' j , X b i ]) path (m .snd i) ≡ sHom h {i'} pp' W.⋆ sHom g {i} pp
      sHom-seq m e i' pp i'' pp' =
        J (λ m' _ → (path : m' .fst i ≡ i'') →
             subst (λ j → W [ X b'' j , X b i ]) path (m' .snd i) ≡ sHom h {i'} pp' W.⋆ sHom g {i} pp)
          (step i' pp i'' pp')
          (sym e)

  sG : Functor (Covariant.∫ PosG) (W ^op)
  sG .F-ob (b , i) = X b i
  sG .F-hom (g , pp) = sHom g pp
  sG .F-id {b , i} = sHom-id {b} {i} (G .F-hom (B .id)) (G .F-id) _
  sG .F-seq {b , i} {b' , i'} {b'' , i''} (g , pp) (h , pp') =
    sHom-seq g h i (G .F-hom (B ._⋆_ g h)) (G .F-seq g h) i' pp i'' pp' _

  -- the polynomial: shapes, and the Grothendieck construction of G
  poly : Signature W
  poly = record { Shapes = Shapes ; Pos = PosG ; s = sG }

  -- a natural map from G into representables is a strength
  module _ (σ : NatTrans G (Yo W ∘F TotalCategory.Fst)) where
    strG : Strength poly
    strG .N-ob (b , i) = σ .N-ob b .snd i
    strG .N-hom {b , i} {b' , _} (g , pp) =
      J (λ i' pp → σ .N-ob b' .snd i' W.⋆ sHom g pp ≡ g .fst W.⋆ σ .N-ob b .snd i)
        (cong (σ .N-ob b' .snd (G .F-hom g .fst i) W.⋆_)
              (substRefl {B = λ j → W [ X b' j , X b i ]} (G .F-hom g .snd i))
         ∙ (λ k → σ .N-hom g k .snd i))
        pp

------------------------------------------------------------------------
-- Step 2: the polynomial of the signature

-- the arity of an operation: the sum, over its slots, of the composite
-- arities of their lock words; and its strength
opE : Op → Functor (W ^op) (Fam W)
opE o = Σᴬ W (Ar o , isSetAr o) (λ a → wordE (slot o a))

opStr : (o : Op) → NatTrans (opE o) (Yo W)
opStr o = Σˢ W (Ar o , isSetAr o) (λ a → wordE (slot o a)) (λ a → wordStr (slot o a))

-- shapes: an operation and its parameter constants
ShapesP : Functor ((W ^op) ^op) (SET ℓ-zero)
ShapesP .F-ob w =
  (Σ[ o ∈ Op ] Cs w (params o)) , isSetΣ isSetOp (λ o → isSetCsts W Const w (params o))
ShapesP .F-hom f (o , cs) = o , cmaps W Const (params o) f cs
ShapesP .F-id = funExt λ (o , cs) → ΣPathP (refl , cmaps-id W Const (params o) cs)
ShapesP .F-seq f g = funExt λ (o , cs) → ΣPathP (refl , cmaps-seq W Const (params o) f g cs)

BP : Category ℓ-zero ℓ-zero
BP = Contravariant.∫ ShapesP

Ops : Category ℓ-zero ℓ-zero
Ops = DiscreteCategory (Op , isSet→isGroupoid isSetOp)

-- forget the parameters (the typing of shapes)
opF : Functor BP Ops
opF .F-ob (_ , (o , _)) = o
opF .F-hom (_ , π) = sym (cong fst π)
opF .F-id = isSetOp _ _ _ _
opF .F-seq _ _ = isSetOp _ _ _ _

EF : Functor Ops (FUNCTOR (W ^op) (Fam W))
EF = DiscFunc opE

-- the arity of each shape
GP : Functor BP (Fam W)
GP = appF (W ^op) (Fam W) ∘F ((EF ∘F opF) ,F TotalCategory.Fst)

private
  module _ (o : Op) {w w' : W.ob} (f : (W ^op) [ w , w' ]) where
    -- E o f followed by a natural transformation α at w', then the strength at w'
    opLHS : ∀ {o'} → NatTrans (opE o) (opE o') → Fam W [ opE o .F-ob w , Yo W .F-ob w' ]
    opLHS {o'} α =
      seqF W (opE o .F-ob w) (opE o' .F-ob w') (Yo W .F-ob w')
        (seqF W (opE o .F-ob w) (opE o .F-ob w') (opE o' .F-ob w') (opE o .F-hom f) (α .N-ob w'))
        (opStr o' .N-ob w')

    opRHS : Fam W [ opE o .F-ob w , Yo W .F-ob w' ]
    opRHS = seqF W (opE o .F-ob w) (Yo W .F-ob w) (Yo W .F-ob w') (opStr o .N-ob w) (Yo W .F-hom f)

  opNat : ∀ {o o'} (q : o ≡ o') {w w'} (f : (W ^op) [ w , w' ]) →
    opLHS o f (EF .F-hom q) ≡ opRHS o f
  opNat {o} q =
    J (λ o' q → ∀ {w w'} (f : (W ^op) [ w , w' ]) → opLHS o f (EF .F-hom q) ≡ opRHS o f)
      (λ {w} {w'} f →
         cong (opLHS o f) (EF .F-id {o})
         ∙ cong (λ m → seqF W (opE o .F-ob w) (opE o .F-ob w') (Yo W .F-ob w') m (opStr o .N-ob w'))
                (Fam W .⋆IdR {opE o .F-ob w} {opE o .F-ob w'} (opE o .F-hom f))
         ∙ opStr o .N-hom f)
      q

-- the strengths of the operations, assembled
σP : NatTrans GP (Yo W ∘F TotalCategory.Fst)
σP .N-ob (w , (o , _)) = opStr o .N-ob w
σP .N-hom (f , sp) = opNat (sym (cong fst sp)) f

polynomial : Signature W
polynomial = Groth.poly ShapesP GP

-- (by copatterns: assigning Groth.strG ShapesP GP σP to this type directly
-- makes Agda compare the two category structures by unfolding)
strength : Strength polynomial
strength .N-ob x = Groth.strG ShapesP GP σP .N-ob x
strength .N-hom f = Groth.strG ShapesP GP σP .N-hom f

------------------------------------------------------------------------
-- read back

open SignatureNotation polynomial using (Shape; Position)

shape-is : ∀ w → Shape w ≡ (Σ[ o ∈ Op ] Cs w (params o))
shape-is w = refl

-- a position is a slot and positions for the locks of its word
position-is : ∀ w (o : Op) (cs : Cs w (params o)) →
  Position w (o , cs) ≡ (Σ[ a ∈ Ar o ] WPos (slot o a) w)
position-is w o cs = refl

-- the scope is the word's scope: it depends on the position
scope-is : ∀ w (o : Op) (cs : Cs w (params o)) (a : Ar o) (ps : WPos (slot o a) w) →
  SignatureNotation.scope polynomial w (o , cs) (a , ps) ≡ wscope (slot o a) w ps
scope-is w o cs a ps = refl

-- the strength's components are the words' composite ι
strength-is : ∀ w (o : Op) (cs : Cs w (params o)) (a : Ar o) (ps : WPos (slot o a) w) →
  strength .N-ob ((w , (o , cs)) , (a , ps)) ≡ wι (slot o a) w ps
strength-is w o cs a ps = refl

-- at a canonical lift, the arity of the polynomial is E o f
-- (up to the transport along the refl operation path)
GP-canonical : ∀ {w w'} (f : W [ w , w' ]) (o : Op) (cs : Cs w (params o)) →
  GP .F-hom {x = (w' , (o , cmaps W Const (params o) f cs))} {y = (w , (o , cs))} (f , refl)
    ≡ opE o .F-hom f
GP-canonical {w} {w'} f o cs =
  cong (λ (α : NatTrans (opE o) (opE o)) →
          seqF W (opE o .F-ob w') (opE o .F-ob w) (opE o .F-ob w) (opE o .F-hom f) (α .N-ob w))
       (EF .F-id {o})
  ∙ Fam W .⋆IdR {opE o .F-ob w'} {opE o .F-ob w} (opE o .F-hom f)

-- ... so it reindexes positions by the arity's pos ...
pos-is : ∀ {w w'} (f : W [ w , w' ]) (o : Op) (cs : Cs w (params o))
  (p : Arity.Pos W (opE o) w') →
  Poly.transportPosition polynomial {b = (w' , (o , cmaps W Const (params o) f cs))}
    {b' = (w , (o , cs))} (f , refl) p ≡ Arity.pos W (opE o) f p
pos-is f o cs p k = GP-canonical f o cs k .fst p

-- ... and maps scopes by the arity's smap
smap-is : ∀ {w w'} (f : W [ w , w' ]) (o : Op) (cs : Cs w (params o))
  (p : Arity.Pos W (opE o) w') →
  PathP (λ k → W [ Arity.scope W (opE o) w (pos-is f o cs p k) , Arity.scope W (opE o) w' p ])
    (Poly.s polynomial .F-hom
       {x = ((w' , (o , cmaps W Const (params o) f cs)) , p)}
       {y = ((w , (o , cs)) , Poly.transportPosition polynomial
               {b = (w' , (o , cmaps W Const (params o) f cs))} {b' = (w , (o , cs))} (f , refl) p)}
       ((f , refl) , refl))
    (Arity.smap W (opE o) f p)
smap-is {w} {w'} f o cs p =
  substRefl {B = λ j → W [ Arity.scope W (opE o) w j , Arity.scope W (opE o) w' p ]}
    (GP .F-hom {x = (w' , (o , cmaps W Const (params o) f cs))} {y = (w , (o , cs))} (f , refl) .snd p)
  ◁ (λ k → GP-canonical f o cs k .snd p)
