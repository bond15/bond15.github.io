{-# OPTIONS --no-allow-unsolved-metas --no-lossy-unification #-}

module RightModuleReduction.RightModule where

open import Cubical.Foundations.Prelude
open import Cubical.Categories.Category using (Category; seq')
open import Cubical.Categories.Functor using (Functor; _∘F_; _⟅_⟆; _⟪_⟫)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Monad.Base using (Monad; IsMonad)

open Category
open NatTrans

-- A right action of T on Q, valued in any category D.
record RightModule {ℓC ℓC′ ℓD ℓD′}
  {C : Category ℓC ℓC′} {D : Category ℓD ℓD′}
  (T : Monad C) (Q : Functor C D)
  : Type (ℓ-max (ℓ-max ℓC ℓC′) (ℓ-max ℓD ℓD′)) where
  private
    T₀ = T .fst
    module T = IsMonad (T .snd)

  field
    ρ : NatTrans (Q ∘F T₀) Q

    -- Q X → Q (T X) → Q X is the identity.
    ρ-unit : (X : C .ob) →
      Q ⟪ T.η .N-ob X ⟫ ⋆⟨ D ⟩ ρ .N-ob X ≡ D .id

    -- The two actions Q (T (T X)) → Q X agree.
    ρ-assoc : (X : C .ob) →
      Q ⟪ T.μ .N-ob X ⟫ ⋆⟨ D ⟩ ρ .N-ob X ≡
      ρ .N-ob (T₀ ⟅ X ⟆) ⋆⟨ D ⟩ ρ .N-ob X

-- Every monad is a right module over itself; its action is multiplication.
-- This is the syntactic/POPL module.  A concrete machine semantics still
-- needs a different configuration functor Q and an action Q T ⇒ Q.
selfModule : {ℓC ℓC′ : Level} {C : Category ℓC ℓC′} (T : Monad C) →
  RightModule T (T .fst)
selfModule {C = C} T .RightModule.ρ = IsMonad.μ (T .snd)
selfModule {C = C} T .RightModule.ρ-unit X =
  funExt⁻ (congP (λ i → N-ob) (IsMonad.idr-μ (T .snd))) X
selfModule {C = C} T .RightModule.ρ-assoc X =
  funExt⁻ (congP (λ i → N-ob) (IsMonad.assoc-μ (T .snd))) X
