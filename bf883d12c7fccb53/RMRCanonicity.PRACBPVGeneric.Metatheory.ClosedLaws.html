-- The syntactic laws of closed computations (Cor 8.5), discharging
-- Closing.ClosedLaws:
--
--   isSetComp          closed computations form a set (TermsSet)
--   reindexC-id        reindexC W.id M ≡ M: the root map along W.id agrees
--                      with the identity (ren-agree-id)
--   reindexC-seq       reindexC (f ⋆ g) M ≡ reindexC g (reindexC f M):
--                      two root maps compose to one (ren-fuse)
--   reindexC-opNodeC   the operation node is natural: the parameter list
--                      by renVs-consts, and each continuation by fusing
--                      both sides into ONE root renaming out of the scope
--                      wscope (slot o a) w (wpos (slot o a) f ps)
--
-- In the last law the left side reads the continuation at the
-- syntactically pulled-back word position wposW μ (mapW f) ps and the right
-- side at the arity's wpos μ f ps; keepW-wsW relates them, together with
-- the world maps (the word's scope map, through the strict identities wsW).
--
-- So closed computations of type B form a presheaf (reindexC) and, with
-- opNodeC, a P-algebra (Semantics.Machine.WithLaws).
open import Cubical.Foundations.Prelude hiding (_,_; comp)
open import Cubical.Data.Unit using (tt)
open import Cubical.Data.Sigma using (Σ-syntax) renaming (_,_ to _,ˢ_)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Categories.Category using (Category; _[_,_])
open import Cubical.Categories.Functor using (Functor)

open import RMRCanonicity.PRACBPVGeneric.Fam using (cmaps)
open import RMRCanonicity.PRACBPVGeneric.Signature

module RMRCanonicity.PRACBPVGeneric.Metatheory.ClosedLaws
  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L)) where

open import RMRCanonicity.PRACBPVGeneric.Syntax L O
open import RMRCanonicity.PRACBPVGeneric.Closing L O
  using (ClosedLaws; module ClosedLaws; wsW; opNodeC; wpos; wsmap)

  using () renaming (isSetComp to isSetTerms)

private
  module W = Category W

----------------------------------------------------------------------
-- reindexing constants

renVs-consts : ∀ {w w'} (f : W [ w , w' ]) (bs : List Base) (cs : Cs w bs) →
  renVs (root (mapW f)) (consts {∅ w} cs) ≡ consts {∅ w'} (cmaps W Const bs f cs)
renVs-consts f [] cs = refl
renVs-consts f (b ∷ bs) (c ,ˢ cs) = cong (const (Const b .F-hom f c) ∷ᵥ_) (renVs-consts f bs cs)

----------------------------------------------------------------------
-- keepW under a word, against the word's arity

-- For ρ : Ren Θ Θ', the word position keepW μ ρ reads at, with the world
-- map from the word's scope through keepW, is the arity's
-- (wpos μ f ps , wsmap μ f ps) at f = toW (wm ρ), followed by the strict
-- identity wsW.  An equation in Σ_{qs} W [ wscope μ (world Θ) qs , - ], so
-- that the positions and the world maps move together.
keepW-wsW : ∀ {Θ Θ'} (μ : List Lk) (ρ : Ren Θ Θ') (ps : WPos μ (world Θ')) →
  Path (Σ[ qs ∈ WPos μ (world Θ) ] W [ wscope μ (world Θ) qs , world (Θ' ⧀*⟨ μ , ps ⟩) ])
    (wposW μ (wm ρ) ps ,ˢ toW (wsW μ Θ (wposW μ (wm ρ) ps)) W.⋆ toW (wm (keepW μ ρ ps)))
    (wpos μ (toW (wm ρ)) ps ,ˢ wsmap μ (toW (wm ρ)) ps W.⋆ toW (wsW μ Θ' ps))
keepW-wsW [] ρ ps i = tt ,ˢ (W.⋆IdL (toW (wm ρ)) ∙ sym (W.⋆IdR (toW (wm ρ)))) i
keepW-wsW (κ ∷ []) ρ p =
  cong (λ y → y .fst ,ˢ W.id W.⋆ y .snd) (ptW-toW κ (wm ρ) p)
  ∙ (λ i → pos κ (toW (wm ρ)) p ,ˢ
       (W.⋆IdL (smap κ (toW (wm ρ)) p) ∙ sym (W.⋆IdR (smap κ (toW (wm ρ)) p))) i)
keepW-wsW {Θ} {Θ'} (κ ∷ κ' ∷ μ) ρ (p ,ˢ ps) =
  -- the word's tail, under the first lock (keepL ρ p), then the first lock
  (λ i → (posW κ (wm ρ) p ,ˢ keepW-wsW (κ' ∷ μ) (keepL ρ p) ps i .fst) ,ˢ
         keepW-wsW (κ' ∷ μ) (keepL ρ p) ps i .snd)
  ∙ cong under (ptW-toW κ (wm ρ) p)
  where
    under : Ey κ (world Θ) (scope κ (world Θ') p) →
      Σ[ qs ∈ WPos (κ ∷ κ' ∷ μ) (world Θ) ]
        W [ wscope (κ ∷ κ' ∷ μ) (world Θ) qs , world (Θ' ⧀*⟨ κ ∷ κ' ∷ μ , (p ,ˢ ps) ⟩) ]
    under (p' ,ˢ h) =
      (p' ,ˢ wpos (κ' ∷ μ) h ps) ,ˢ wsmap (κ' ∷ μ) h ps W.⋆ toW (wsW (κ' ∷ μ) (Θ' ⧀⟨ κ , p ⟩) ps)

----------------------------------------------------------------------
-- the laws

closedLaws : ClosedLaws
closedLaws .ClosedLaws.isSetComp = isSetTerms
closedLaws .ClosedLaws.reindexC-id M = ren-agree-idC (Agree-root refl refl) M refl
closedLaws .ClosedLaws.reindexC-seq f g M = sym (ren-fuseC (Composite-root refl refl) M refl)
closedLaws .ClosedLaws.reindexC-opNodeC {w} {w'} f o cs ks =
  cong₂ (op o) (renVs-consts f (params o) cs)
    (funExt λ a → funExt λ ps → continuation (slot o a) (ks a) ps)
  where
    -- each continuation: both sides fuse to ONE root renaming out of the
    -- scope, along G (the word's scope map, then the strict identity wsW)
    continuation : ∀ {B} (μ : List Lk) (k : (qs : WPos μ w) → Comp (∅ (wscope μ w qs)) B)
      (ps : WPos μ w') →
      renC (keepW μ (root (mapW f)) ps)
        (renC (root (wsW μ (∅ w) (wposW μ (mapW f) ps))) (k (wposW μ (mapW f) ps)))
      ≡ renC (root (wsW μ (∅ w') ps)) (renC (root (mapW (wsmap μ f ps))) (k (wpos μ f ps)))
    continuation μ k ps =
      ren-fuseC (Composite-root (λ i → wscope μ w (Y i .fst)) {ρ₁₂ = root G} (λ i → Y i .snd))
        (k (wposW μ (mapW f) ps)) (λ i → k (Y i .fst))
      ∙ sym (ren-fuseC (Composite-root refl {ρ₁₂ = root G} G-is) (k (wpos μ f ps)) refl)
      where
        G : WMap (wscope μ w (wpos μ f ps)) (world (∅ w' ⧀*⟨ μ , ps ⟩))
        G = mapW (wsmap μ f ps) ⋆W wsW μ (∅ w') ps

        G-is : toW (mapW (wsmap μ f ps)) W.⋆ toW (wsW μ (∅ w') ps) ≡ toW G
        G-is = sym (toW-⋆ (mapW (wsmap μ f ps)) (wsW μ (∅ w') ps))

        -- the left side reads k at wposW μ (mapW f) ps; that position, with
        -- the world map from its scope through keepW, is (wpos μ f ps , G)
        Y = keepW-wsW μ (root (mapW f)) ps ∙ cong (wpos μ f ps ,ˢ_) G-is
