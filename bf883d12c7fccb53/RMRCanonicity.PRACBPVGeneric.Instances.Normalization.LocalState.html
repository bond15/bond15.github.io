-- Normalization of the local-state machine (Instances.Machines.LocalState).
--
--   finOp     lookup has two continuations, update b and new b one each
--   local-sn  every configuration is strongly normalizing
--   local-wn  every configuration reaches a terminal one
--
-- The run alloc-write ends in a terminal configuration
-- (alloc-write-normal).
module RMRCanonicity.PRACBPVGeneric.Instances.Normalization.LocalState where

open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Foundations.Equiv using (invEquiv)
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.Unit using (Unit; tt)
open import Cubical.Data.Empty using (⊥)
open import Cubical.Data.Bool using (Bool; isSetBool)
open import Cubical.Data.Sigma using (Σ-syntax) renaming (_×_ to _×Σ_)
open import Cubical.Data.SumFin.Properties using (SumFin2≃Bool)
open import Cubical.Data.FinSet.Base using (isFinSet; isFinOrd; isFinOrd→isFinSet)
open import Cubical.Data.FinSet.Properties using (isFinOrdUnit)
open import Cubical.Data.FinSet.Constructors using (isFinOrdΣ)
open import Cubical.Induction.WellFounded using (Acc)

open import Polynomial.Presheaf.FinBijective.LocalState.Config (Bool , isSetBool) using (Store)
open import Polynomial.Presheaf.FinBijective.LocalState.Store.Operations using (freshCell; emptyStore)

open import RMRCanonicity.PRACBPVGeneric.Instances.LocalState hiding (_,_)
open import RMRCanonicity.PRACBPVGeneric.Instances.Machines.LocalState
  using (isSetLSLk; isSetLSBase; isSetConf; isSet⊥; δ; module ReturnsLoc)
open ReturnsLoc using (allocWrite; σ₁; alloc-write)
import RMRCanonicity.PRACBPVGeneric.Semantics.RunNormalization as RunNormalization

isFinOrdBool : isFinOrd Bool
isFinOrdBool = 2 , invEquiv SumFin2≃Bool

-- lookup continues in one of two slots, update b and new b in one
finOp : ∀ o w → isFinSet (Σ[ a ∈ Ar o ] WPos (slot o a) w)
finOp lookup n = isFinOrd→isFinSet (isFinOrdΣ Bool isFinOrdBool _ λ _ → isFinOrdUnit)
finOp (update b) n = isFinOrd→isFinSet (isFinOrdΣ Unit isFinOrdUnit _ λ _ → isFinOrdUnit)
finOp (new b) n = isFinOrd→isFinSet (isFinOrdΣ Unit isFinOrdUnit _ λ _ → isFinOrdUnit)

module _ {A : ValTy} where
  open RunNormalization lsLocks lsOps isSetLSLk isSetLSBase Store ⊥ isSetConf isSet⊥ δ A finOp

  local-sn : ∀ x → Acc (λ y x → x ⟶ y) x
  local-sn = strongNormalization

  local-wn : ∀ x → Σ[ y ∈ Config ] (x ⟶* y) ×Σ Terminal y
  local-wn = weakNormalization

open RunNormalization lsLocks lsOps isSetLSLk isSetLSBase Store ⊥ isSetConf isSet⊥ δ 𝕃 finOp
  using (Config; _⟶*_; Terminal; running; running-terminal)

-- new ℓ := false; ℓ := true; return ℓ ends returning the new cell
alloc-write-normal : Σ[ y ∈ Config ] (running {0} emptyStore allocWrite ⟶* y) ×Σ Terminal y
alloc-write-normal = running {1} σ₁ (return (cell (freshCell 0))) , alloc-write , running-terminal σ₁ (cell (freshCell 0))
