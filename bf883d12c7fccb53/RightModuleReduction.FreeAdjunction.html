{-# OPTIONS --no-allow-unsolved-metas --no-lossy-unification #-}

-- Package a chosen universal arrow into a free/forgetful adjunction.
module RightModuleReduction.FreeAdjunction where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Isomorphism using (Iso)
open import Cubical.Categories.Category using (Category; _[_,_]; seq')
open import Cubical.Categories.Functor using
  (Functor; _⟅_⟆; _⟪_⟫; _∘F_; 𝟙⟨_⟩)
open import Cubical.Categories.NaturalTransformation using (NatTrans)
open import Cubical.Categories.Adjoint using
  (module NaturalBijection; module UnitCounit; adj'→adj)

open Category
open Functor
open NatTrans

module FromUniversalProperty
  {c c′ d d′ : Level} (C : Category c c′) (D : Category d d′)
  (F : Functor C D) (U : Functor D C)
  (η : NatTrans (𝟙⟨ C ⟩) (U ∘F F))
  (extend : {X : C .ob} {B : D .ob} → C [ X , U ⟅ B ⟆ ] → D [ F ⟅ X ⟆ , B ])
  (β : {X : C .ob} {B : D .ob} (f : C [ X , U ⟅ B ⟆ ]) →
    seq' C (η .N-ob X) (U ⟪ extend f ⟫) ≡ f)
  (unique : {X : C .ob} {B : D .ob} (f : C [ X , U ⟅ B ⟆ ])
    (h : D [ F ⟅ X ⟆ , B ]) →
    seq' C (η .N-ob X) (U ⟪ h ⟫) ≡ f → h ≡ extend f)
  where

  homIso : {X : C .ob} {B : D .ob} →
    Iso (D [ F ⟅ X ⟆ , B ]) (C [ X , U ⟅ B ⟆ ])
  homIso {X} .Iso.fun h = seq' C (η .N-ob X) (U ⟪ h ⟫)
  homIso .Iso.inv = extend
  homIso .Iso.sec = β
  homIso {X} .Iso.ret h =
    sym (unique (seq' C (η .N-ob X) (U ⟪ h ⟫)) h refl)

  abstract
    natural-C : {X′ X : C .ob} {B : D .ob}
      (f : C [ X , U ⟅ B ⟆ ]) (g : C [ X′ , X ]) →
      extend (seq' C g f) ≡ seq' D (F ⟪ g ⟫) (extend f)
    natural-C {X′} {X} f g = sym (unique (seq' C g f)
      (seq' D (F ⟪ g ⟫) (extend f))
      (cong (seq' C (η .N-ob X′)) (U .F-seq (F ⟪ g ⟫) (extend f))
      ∙ sym (C .⋆Assoc (η .N-ob X′) (U ⟪ F ⟪ g ⟫ ⟫) (U ⟪ extend f ⟫))
      ∙ cong (λ h → seq' C h (U ⟪ extend f ⟫)) (sym (η .N-hom g))
      ∙ C .⋆Assoc g (η .N-ob X) (U ⟪ extend f ⟫)
      ∙ cong (seq' C g) (β f)))

  bijection : NaturalBijection._⊣_ F U
  bijection .NaturalBijection._⊣_.adjIso = homIso
  bijection .NaturalBijection._⊣_.adjNatInD {c = X} h k =
    cong (seq' C (η .N-ob X)) (U .F-seq h k)
    ∙ sym (C .⋆Assoc (η .N-ob X) (U ⟪ h ⟫) (U ⟪ k ⟫))
  bijection .NaturalBijection._⊣_.adjNatInC = natural-C

  adjunction : UnitCounit._⊣_ F U
  adjunction = adj'→adj F U bijection
