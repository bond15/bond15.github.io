-- Presheaves on worlds, as in RMR: covariant functors W → Set, i.e.
-- Presheaf (W ^op).  The few constructions the semantics needs, written by
-- hand so that their actions are one step (no functor layers to unfold).
--
--   Y w          the representable y w = W [ w , - ]
--   fromFun X    a functor W → Set (base types, parameters) as a presheaf
--   Exp G Q      the exponential: (G ⇒ Q) w = 𝒱 [ y w × G , Q ]
--                with λExp (curry) and appExp (evaluation)
--   EmptyPsh     the initial presheaf (empty at every world), EmptyPsh-rec
--   X ⊎Psh Y     the coproduct, pointwise, with inlPsh / inrPsh; casePsh is
--                the copairing in context, Γ × (X + Y) → Z (presheaves
--                are distributive: Γ × (X + Y) ≅ Γ × X + Γ × Y pointwise)

open import Cubical.Foundations.Prelude
open import Cubical.Data.Empty as ⊥ using (⊥; isProp⊥)
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr; isSet⊎)
open import Cubical.Categories.Category using (Category; _^op; _[_,_])
open import Cubical.Categories.Functor using (Functor)
open import Cubical.Categories.Instances.Sets using (SET)
open import Cubical.Categories.Presheaf.Base using (Presheaf)
open import Cubical.Categories.Presheaf.Constructions.BinProduct using (_×Psh_)
open import Cubical.Categories.Presheaf.Morphism.Alt
  using (PshHom; _⋆PshHom_; makePshHomPath; isSetPshHom)

module RMRCanonicity.PRACBPVGeneric.Semantics.Presheaf (W : Category ℓ-zero ℓ-zero) where

open Functor
open PshHom

private
  module W = Category W

module _ where

  𝒱 : Type₁
  𝒱 = Presheaf (W ^op) ℓ-zero

  -- the elements of a presheaf at a world
  El : 𝒱 → W.ob → Type
  El X v = X .F-ob v .fst

  -- the representable presheaf y w = W [ w , - ]
  Y : W.ob → 𝒱
  Y w .F-ob v = W [ w , v ] , W.isSetHom
  Y w .F-hom f h = h W.⋆ f
  Y w .F-id = funExt W.⋆IdR
  Y w .F-seq f g = funExt λ h → sym (W.⋆Assoc h f g)

  -- y on morphisms: precomposition
  Ymap : ∀ {w w'} → W [ w , w' ] → PshHom (Y w') (Y w)
  Ymap f .N-ob v h = f W.⋆ h
  Ymap f .N-hom u v g h = sym (W.⋆Assoc f h g)

  -- a functor W → Set (constants of a base type, parameters) as a presheaf
  fromFun : Functor W (SET ℓ-zero) → 𝒱
  fromFun X .F-ob = X .F-ob
  fromFun X .F-hom = X .F-hom
  fromFun X .F-id = X .F-id
  fromFun X .F-seq = X .F-seq

  -- Yoneda: an element at w is a map out of y w
  yoneda : (X : 𝒱) {w : W.ob} → El X w → PshHom (Y w) X
  yoneda X x .N-ob v h = X .F-hom h x
  yoneda X x .N-hom u v g h = funExt⁻ (X .F-seq h g) x

  ----------------------------------------------------------------------
  -- the exponential (copied from the old core's Denotation)

  Env : W.ob → 𝒱 → 𝒱
  Env w G = Y w ×Psh G

  -- precompose the world with f : w → w'
  reWorld : ∀ {w w'} (G : 𝒱) (f : W [ w , w' ]) → PshHom (Env w' G) (Env w G)
  reWorld G f .N-ob v (h , γ) = f W.⋆ h , γ
  reWorld G f .N-hom u v g (h , γ) = cong (λ m → m , G .F-hom g γ) (sym (W.⋆Assoc f h g))

  Exp : 𝒱 → 𝒱 → 𝒱
  Exp G Q .F-ob w = PshHom (Env w G) Q , isSetPshHom _ _
  Exp G Q .F-hom f e = reWorld G f ⋆PshHom e
  Exp G Q .F-id = funExt λ e → makePshHomPath (funExt λ v → funExt λ { (h , γ) →
    cong (λ m → e .N-ob v (m , γ)) (W.⋆IdL h) })
  Exp G Q .F-seq f g = funExt λ e → makePshHomPath (funExt λ v → funExt λ { (h , γ) →
    cong (λ m → e .N-ob v (m , γ)) (W.⋆Assoc f g h) })

  λExp : (G Q : 𝒱) {S : 𝒱} → PshHom (S ×Psh G) Q → PshHom S (Exp G Q)
  λExp G Q {S} α .N-ob w s .N-ob v (h , γ) = α .N-ob v (S .F-hom h s , γ)
  λExp G Q {S} α .N-ob w s .N-hom u v g (h , γ) =
    cong (λ y → α .N-ob u (y , G .F-hom g γ)) (funExt⁻ (S .F-seq h g) s)
    ∙ α .N-hom u v g (S .F-hom h s , γ)
  λExp G Q {S} α .N-hom u v g s = makePshHomPath (funExt λ v' → funExt λ { (h , γ) →
    cong (λ y → α .N-ob v' (y , γ)) (sym (funExt⁻ (S .F-seq g h) s)) })

  appExp : (G Q : 𝒱) → PshHom (Exp G Q ×Psh G) Q
  appExp G Q .N-ob v (e , γ) = e .N-ob v (W.id , γ)
  appExp G Q .N-hom u v g (e , γ) =
    cong (λ m → e .N-ob u (m , G .F-hom g γ)) (W.⋆IdR g ∙ sym (W.⋆IdL g))
    ∙ e .N-hom u v g (W.id , γ)

  ----------------------------------------------------------------------
  -- the initial presheaf and binary coproducts, pointwise

  EmptyPsh : 𝒱
  EmptyPsh .F-ob v = ⊥ , isProp→isSet isProp⊥
  EmptyPsh .F-hom f x = x
  EmptyPsh .F-id = refl
  EmptyPsh .F-seq f g = refl

  EmptyPsh-rec : (X : 𝒱) → PshHom EmptyPsh X
  EmptyPsh-rec X .N-ob v = ⊥.rec
  EmptyPsh-rec X .N-hom u v f x = ⊥.rec x

  infixr 5 _⊎Psh_

  _⊎Psh_ : 𝒱 → 𝒱 → 𝒱
  (X ⊎Psh Y) .F-ob v = (El X v ⊎ El Y v) , isSet⊎ (X .F-ob v .snd) (Y .F-ob v .snd)
  (X ⊎Psh Y) .F-hom f = ⊎.map (X .F-hom f) (Y .F-hom f)
  (X ⊎Psh Y) .F-id = funExt λ
    { (inl x) → cong inl (funExt⁻ (X .F-id) x)
    ; (inr y) → cong inr (funExt⁻ (Y .F-id) y) }
  (X ⊎Psh Y) .F-seq f g = funExt λ
    { (inl x) → cong inl (funExt⁻ (X .F-seq f g) x)
    ; (inr y) → cong inr (funExt⁻ (Y .F-seq f g) y) }

  inlPsh : (X Y : 𝒱) → PshHom X (X ⊎Psh Y)
  inlPsh X Y .N-ob v = inl
  inlPsh X Y .N-hom u v f x = refl

  inrPsh : (X Y : 𝒱) → PshHom Y (X ⊎Psh Y)
  inrPsh X Y .N-ob v = inr
  inrPsh X Y .N-hom u v f y = refl

  -- the copairing in a context Γ
  casePsh : {Γ X Y Z : 𝒱} → PshHom (Γ ×Psh X) Z → PshHom (Γ ×Psh Y) Z →
    PshHom (Γ ×Psh (X ⊎Psh Y)) Z
  casePsh α β .N-ob v (γ , inl x) = α .N-ob v (γ , x)
  casePsh α β .N-ob v (γ , inr y) = β .N-ob v (γ , y)
  casePsh α β .N-hom u v f (γ , inl x) = α .N-hom u v f (γ , x)
  casePsh α β .N-hom u v f (γ , inr y) = β .N-hom u v f (γ , y)

  -- two copairings agree at a point when their branches do
  casePsh-pt : {Γ Γ' X Y Z : 𝒱} (α' : PshHom (Γ' ×Psh X) Z) (β' : PshHom (Γ' ×Psh Y) Z)
    (α : PshHom (Γ ×Psh X) Z) (β : PshHom (Γ ×Psh Y) Z) {v : W.ob} (γ' : El Γ' v) (γ : El Γ v) →
    (∀ x → α' .N-ob v (γ' , x) ≡ α .N-ob v (γ , x)) →
    (∀ y → β' .N-ob v (γ' , y) ≡ β .N-ob v (γ , y)) →
    ∀ s → casePsh α' β' .N-ob v (γ' , s) ≡ casePsh α β .N-ob v (γ , s)
  casePsh-pt α' β' α β γ' γ hα hβ (inl x) = hα x
  casePsh-pt α' β' α β γ' γ hα hβ (inr y) = hβ y
