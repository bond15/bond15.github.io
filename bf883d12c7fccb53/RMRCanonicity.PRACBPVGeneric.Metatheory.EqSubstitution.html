-- The equational theory is closed under substitution, hence under chains.
--
--   sub≈v, sub≈vs, sub≈c   V ≈ V'  →  subX σ V ≈ subX σ V'
--   ch≈v, ch≈vs, ch≈c      the same for a chain (one substitution at a time)
--
-- The equational logical relation needs this: its Kripke clauses (at U B
-- and A ⇒ B) move a term along chains, and its closure under ≈ must
-- survive that.
--
-- By induction on the derivation.  Congruences go under the lifted
-- substitution, exactly as subX does.  Every other axiom is the same axiom
-- at the substituted terms, up to a law of substitution:
--
--   β laws       sub-⇝ (SubstReduction): each β law is a head step (⇝→≈)
--   η𝟘, η+, η×   sub-sub₁, and the η substitutions commute with liftS
--                (sub-inlS, sub-inrS, sub-pairS: both sides fuse to one
--                substitution, 5.4)
--   η⇒           sub-liftS-wkC (weakening commutes with liftS)
--   η[κ]         sub-openV
--   ηF           sub-plug, and sub-wkS (a weakened stack under liftS)
--   op-stack     sub-plug, and sub-dropWS (a stack moved across a word's
--                locks, then substituted under them; sub-dropW per frame)
--   theory       sub-root: a closed term moved in by root g, then σ, is
--                moved in by root (g ⋆W wmS σ) (5.2: ∅ u has no neutrals)
open import Cubical.Foundations.Prelude hiding (Sub; _,_; comp)
open import Cubical.Data.Sigma using () renaming (_,_ to _,ˢ_)
open import Cubical.Data.List using (List; []; _∷_)
open import Cubical.Categories.Category using (Category)

----------------------------------------------------------------------
-- laws of substitution for the axioms

-- a closed term moved in by the root, then substituted
sub-root : ∀ {u Θ Θ' B} (σ : Sub Θ Θ') (g : WMap u (world Θ)) (M : Comp (∅ u) B) →
  subC σ (renC (root g) M) ≡ renC (root (g ⋆W wmS σ)) M
sub-root σ g M = sub-ren-fuseC c M refl ∙ sub-renC (root (g ⋆W wmS σ)) M
  where
    c : CompRS refl (root g) σ ⌜ root (g ⋆W wmS σ) ⌝
    c .on-ne = funExt λ ()
    c .on-world = sym (toW-⋆ g (wmS σ))

-- weakening commutes with liftS (sub-liftS-wkV's two composites)
private
  wk₁ : ∀ {Θ Θ' B} (σ : Sub Θ Θ') → CompRS refl (drop {A = B} idR) (liftS σ) (wkS σ)
  wk₁ σ .on-ne = refl
  wk₁ σ .on-world = W.⋆IdL _

  wk₂ : ∀ {Θ Θ' B} (σ : Sub Θ Θ') → CompSR refl σ (drop {A = B} idR) (wkS σ)
  wk₂ σ .on-ne = funExt λ n → sym (wkS-ne σ n)
  wk₂ σ .on-world = W.⋆IdR _ ∙ cong toW (sym (wmS-wkS σ))

sub-liftS-wkC : ∀ {Θ Θ' A B} (σ : Sub Θ Θ') (M : Comp Θ B) →
  subC (liftS {A = A} σ) (wkC M) ≡ wkC (subC σ M)
sub-liftS-wkC σ M = sub-ren-fuseC (wk₁ σ) M refl ∙ sym (ren-sub-fuseC (wk₂ σ) M refl)

-- a weakened stack, substituted under liftS
sub-wkS : ∀ {Θ Θ' A B B'} (σ : Sub Θ Θ') (S : Stack Θ B B') →
  subS (liftS {A = A} σ) (renS (drop idR) S) ≡ renS (drop idR) (subS σ S)
sub-wkS σ hole = refl
sub-wkS σ (S to• N) = cong₂ _to•_ (sub-wkS σ S)
  (sub-ren-fuseC (CompRS-keep (wk₁ σ)) N refl ∙ sym (ren-sub-fuseC (CompSR-keep (wk₂ σ)) N refl))
sub-wkS σ (app• S V) = cong₂ app• (sub-wkS σ S) (sub-liftS-wkV σ V)

-- a stack moved across a word's locks, then substituted under them
sub-dropWS : ∀ {Θ Θ' B B'} (σ : Sub Θ Θ') (μ : List Lk) (ps : WPos μ (world Θ')) (S : Stack Θ B B') →
  subS (liftWS μ σ ps) (renS (dropW idR μ (wposW μ (wmS σ) ps)) S) ≡ renS (dropW idR μ ps) (subS σ S)
sub-dropWS σ μ ps hole = refl
sub-dropWS σ μ ps (S to• N) = cong₂ _to•_ (sub-dropWS σ μ ps S) (sub-dropW σ μ ps N)
sub-dropWS σ μ ps (app• S V) = cong₂ app• (sub-dropWS σ μ ps S) (sub-dropWV σ μ ps V)

-- the η substitutions commute with liftS: both composites are one
-- substitution (wkS σ ,ₛ inl x, and so on)
private
  -- wkV after a cons with a weakened identity
  wk-cons : ∀ {Θ A C B} (X : Val Θ B) (V : Val (Θ , A) C) →
    subV (wkS {A = A} idS ,ₛ V) (wkV {B = C} X) ≡ wkV X
  wk-cons X V = sub-cons-wkV (wkS idS) V X ∙ sub-renV (drop idR) X

sub-inlS : ∀ {Θ Θ' A A' B} (σ : Sub Θ Θ') (M : Comp (Θ , (A ⊕ A')) B) →
  subC (liftS σ) (subC inlS M) ≡ subC inlS (subC (liftS σ) M)
sub-inlS {A = A} σ M = sub-fuseC c₁ M refl ∙ sym (sub-fuseC c₂ M refl)
  where
    c₁ : CompSS refl inlS (liftS {A = A} σ) (wkS σ ,ₛ inl (ne here))
    c₁ .on-ne = funExt λ { here → refl ; (wkv n) → refl }
    c₁ .on-world = W.⋆IdL _
    c₂ : CompSS refl (liftS σ) inlS (wkS σ ,ₛ inl (ne here))
    c₂ .on-ne = funExt λ
      { here → refl
      ; (wkv n) → cong (subV inlS) (wkS-ne σ n) ∙ wk-cons (subNe σ n) _ ∙ sym (wkS-ne σ n) }
    c₂ .on-world = W.⋆IdR _ ∙ cong toW (wmS-wkS σ ∙ sym (wmS-wkS σ))

sub-inrS : ∀ {Θ Θ' A A' B} (σ : Sub Θ Θ') (M : Comp (Θ , (A ⊕ A')) B) →
  subC (liftS σ) (subC inrS M) ≡ subC inrS (subC (liftS σ) M)
sub-inrS {A' = A'} σ M = sub-fuseC c₁ M refl ∙ sym (sub-fuseC c₂ M refl)
  where
    c₁ : CompSS refl inrS (liftS {A = A'} σ) (wkS σ ,ₛ inr (ne here))
    c₁ .on-ne = funExt λ { here → refl ; (wkv n) → refl }
    c₁ .on-world = W.⋆IdL _
    c₂ : CompSS refl (liftS σ) inrS (wkS σ ,ₛ inr (ne here))
    c₂ .on-ne = funExt λ
      { here → refl
      ; (wkv n) → cong (subV inrS) (wkS-ne σ n) ∙ wk-cons (subNe σ n) _ ∙ sym (wkS-ne σ n) }
    c₂ .on-world = W.⋆IdR _ ∙ cong toW (wmS-wkS σ ∙ sym (wmS-wkS σ))

sub-pairS : ∀ {Θ Θ' A A' B} (σ : Sub Θ Θ') (M : Comp (Θ , (A × A')) B) →
  subC (liftS (liftS σ)) (subC pairS M) ≡ subC pairS (subC (liftS σ) M)
sub-pairS {Θ' = Θ'} {A} {A'} σ M = sub-fuseC c₁ M refl ∙ sym (sub-fuseC c₂ M refl)
  where
    σ₁₂ : Sub (_ , (A × A')) (Θ' , A , A')
    σ₁₂ = wkS (wkS σ) ,ₛ pair (ne (wkv here)) (ne here)

    -- two weakenings are one renaming
    wk² : ∀ {C} (X : Val Θ' C) → renV (drop (drop idR)) X ≡ wkV {B = A'} (wkV {B = A} X)
    wk² X = sym (ren-fuseV c X refl)
      where
        c : Composite refl (drop {A = A} idR) (drop {A = A'} idR) (drop (drop idR))
        c .on-ne = refl
        c .on-world = W.⋆IdL _

    c₁ : CompSS refl pairS (liftS (liftS σ)) σ₁₂
    c₁ .on-ne = funExt λ { here → refl ; (wkv n) → refl }
    c₁ .on-world = W.⋆IdL _
    c₂ : CompSS refl (liftS σ) pairS σ₁₂
    c₂ .on-ne = funExt λ
      { here → refl
      ; (wkv n) →
          cong (subV pairS) (wkS-ne σ n)
          ∙ sub-cons-wkV (wkS (wkS idS)) _ (subNe σ n)
          ∙ sub-renV (drop (drop idR)) (subNe σ n)
          ∙ wk² (subNe σ n)
          ∙ sym (wkS-ne (wkS σ) n ∙ cong wkV (wkS-ne σ n)) }
    c₂ .on-world = W.⋆IdR _ ∙ cong toW (wmS-wkS σ ∙ sym (wmS-wkS (wkS σ) ∙ wmS-wkS σ))

----------------------------------------------------------------------
-- closure under substitution

sub≈v : ∀ {Θ Θ' A} (σ : Sub Θ Θ') {V V' : Val Θ A} → V ≈v V' → subV σ V ≈v subV σ V'
sub≈vs : ∀ {Θ Θ' bs} (σ : Sub Θ Θ') {vs vs' : Vals Θ bs} → vs ≈vs vs' → subVs σ vs ≈vs subVs σ vs'
sub≈c : ∀ {Θ Θ' B} (σ : Sub Θ Θ') {M M' : Comp Θ B} → M ≈c M' → subC σ M ≈c subC σ M'

sub≈v σ ≈refl = ≈refl
sub≈v σ (≈sym p) = ≈sym (sub≈v σ p)
sub≈v σ (≈trans p p') = ≈trans (sub≈v σ p) (sub≈v σ p')
sub≈v σ (inl≈ p) = inl≈ (sub≈v σ p)
sub≈v σ (inr≈ p) = inr≈ (sub≈v σ p)
sub≈v σ (pair≈ p p') = pair≈ (sub≈v σ p) (sub≈v σ p')
sub≈v σ (thunk≈ p) = thunk≈ (sub≈c σ p)
sub≈v σ (shut≈ {κ = κ} p) = shut≈ λ q → sub≈v (lockₛ σ q idR) (p (posW κ (wmS σ) q))
sub≈v σ (η𝟙 V) = η𝟙 (subV σ V)
sub≈v σ (ηU V) = ηU (subV σ V)
sub≈v σ (η[κ] V) = ≈trans (shut≈ λ q → ≡→≈v (sub-openV σ q V)) (η[κ] (subV σ V))

sub≈vs σ []≈ = []≈
sub≈vs σ (p ∷≈ ps) = sub≈v σ p ∷≈ sub≈vs σ ps

sub≈c σ ≈refl = ≈refl
sub≈c σ (≈sym p) = ≈sym (sub≈c σ p)
sub≈c σ (≈trans p p') = ≈trans (sub≈c σ p) (sub≈c σ p')
sub≈c σ (return≈ p) = return≈ (sub≈v σ p)
sub≈c σ (to≈ p p') = to≈ (sub≈c σ p) (sub≈c (liftS σ) p')
sub≈c σ (force≈ p) = force≈ (sub≈v σ p)
sub≈c σ (case≈ p p') = case≈ (sub≈v σ p) (sub≈c (liftS (liftS σ)) p')
sub≈c σ (case₊≈ p p' p'') = case₊≈ (sub≈v σ p) (sub≈c (liftS σ) p') (sub≈c (liftS σ) p'')
sub≈c σ (absurd≈ p) = absurd≈ (sub≈v σ p)
sub≈c σ (lam≈ p) = lam≈ (sub≈c (liftS σ) p)
sub≈c σ (app≈ p p') = app≈ (sub≈c σ p) (sub≈v σ p')
sub≈c σ (op≈ {o = o} ps pk) =
  op≈ (sub≈vs σ ps) λ a qs → sub≈c (liftWS (slot o a) σ qs) (pk a (wposW (slot o a) (wmS σ) qs))
sub≈c σ (F-β V N) = ⇝→≈ (sub-⇝ σ (β-to {V = V} {N}))
sub≈c σ (U-β M) = ⇝→≈ (sub-⇝ σ (β-force {M = M}))
sub≈c σ (×-β V V' M) = ⇝→≈ (sub-⇝ σ (β-case {V = V} {V'} {M}))
sub≈c σ (+-β₁ V M N) = ⇝→≈ (sub-⇝ σ (β-case₊-inl {V = V} {M} {N}))
sub≈c σ (+-β₂ V M N) = ⇝→≈ (sub-⇝ σ (β-case₊-inr {V = V} {M} {N}))
sub≈c σ (⇒-β M V) = ⇝→≈ (sub-⇝ σ (β-app {M = M} {V}))
sub≈c σ (η𝟘 M V) = ≈trans (≡→≈c (sub-sub₁ σ M V)) (η𝟘 (subC (liftS σ) M) (subV σ V))
sub≈c σ (η+ M V) =
  ≈trans (≡→≈c (sub-sub₁ σ M V))
    (≈trans (η+ (subC (liftS σ) M) (subV σ V))
      (≡→≈c (cong₂ (case₊ (subV σ V)) (sym (sub-inlS σ M)) (sym (sub-inrS σ M)))))
sub≈c σ (η× M V) =
  ≈trans (≡→≈c (sub-sub₁ σ M V))
    (≈trans (η× (subC (liftS σ) M) (subV σ V))
      (≡→≈c (cong (case (subV σ V)) (sym (sub-pairS σ M)))))
sub≈c σ (η⇒ M) = ≈trans (lam≈ (app≈ (≡→≈c (sub-liftS-wkC σ M)) ≈refl)) (η⇒ (subC σ M))
sub≈c σ (ηF S M) =
  ≈trans (≡→≈c (sub-plug σ S M))
    (≈trans (ηF (subS σ S) (subC σ M))
      (≡→≈c (cong (subC σ M to_)
        (sym (sub-plug (liftS σ) (renS (drop idR) S) (return (ne here))
              ∙ cong (λ S' → plug S' (return (ne here))) (sub-wkS σ S))))))
sub≈c σ (op-stack S o vs k) =
  ≈trans (≡→≈c (sub-plug σ S (op o vs k)))
    (≈trans (op-stack (subS σ S) o (subVs σ vs) _)
      (≡→≈c (cong (op o (subVs σ vs)) (funExt λ a → funExt λ ps →
        sym (sub-plug (liftWS (slot o a) σ ps)
               (renS (dropW idR (slot o a) (wposW (slot o a) (wmS σ) ps)) S)
               (k a (wposW (slot o a) (wmS σ) ps))
             ∙ cong (λ S' → plug S' (subC (liftWS (slot o a) σ ps) (k a (wposW (slot o a) (wmS σ) ps))))
                 (sub-dropWS σ (slot o a) ps S))))))
sub≈c σ (theory e γ g) =
  ≈trans (≡→≈c (sub-root σ g _)) (≈trans (theory e γ (g ⋆W wmS σ)) (≡→≈c (sym (sub-root σ g _))))

----------------------------------------------------------------------
-- closure under chains

ch≈v : ∀ {Θ Θ' A} (χ : Chain Θ Θ') {V V' : Val Θ A} → V ≈v V' → chV χ V ≈v chV χ V'
ch≈v []ᶜ p = p
ch≈v (σ ∷ᶜ χ) p = ch≈v χ (sub≈v σ p)

ch≈vs : ∀ {Θ Θ' bs} (χ : Chain Θ Θ') {vs vs' : Vals Θ bs} → vs ≈vs vs' → chVs χ vs ≈vs chVs χ vs'
ch≈vs []ᶜ p = p
ch≈vs (σ ∷ᶜ χ) p = ch≈vs χ (sub≈vs σ p)

ch≈c : ∀ {Θ Θ' B} (χ : Chain Θ Θ') {M M' : Comp Θ B} → M ≈c M' → chC χ M ≈c chC χ M'
ch≈c []ᶜ p = p
ch≈c (σ ∷ᶜ χ) p = ch≈c χ (sub≈c σ p)
