-- The guarded machine (W = ω^op, one lock kind tick; Instances.Guarded):
-- the world is the fuel left.
--
--   S n = 1                     nothing but the fuel (the world) is kept
--   H = 1                       one outcome: out of fuel (timeout)
--   δ {n + 1} step = continue in the slot, at n = the scope of [tick]
--   δ {0}     step = halt: there is no tick at 0
--
-- so Q X ≅ (Σ_n X n) + 1.  A step costs one unit of fuel; delay M is a
-- step followed by M one world later.
--
-- The lock itself is the evident functor: GuardedLock.earlierIso shows
-- L_tick Γ t ≅ Γ (t + 1), the earlier modality ◀.
module RMRCanonicity.PRACBPVGeneric.Instances.Machines.Guarded where

open import Cubical.Foundations.Prelude hiding (Sub)
open import Cubical.Foundations.HLevels using (isSetΣ)
open import Cubical.Data.Nat using (ℕ; zero; suc; isSetℕ)
open import Cubical.Data.Unit using (Unit; tt; isSetUnit)
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Data.Sigma using (Σ-syntax)

open import RightModuleReduction.Theory using (var; ops; module SignatureNotation)

open import RMRCanonicity.PRACBPVGeneric.Instances.Guarded hiding (_,_)
import RMRCanonicity.PRACBPVGeneric.Polynomial as PM
import RMRCanonicity.PRACBPVGeneric.Semantics.RunMachine as RunMachine
open import RMRCanonicity.PRACBPVGeneric.Semantics.GuardedLock using (earlierIso) public

-- shapes, positions and scopes of the derived polynomial
module P = SignatureNotation (PM.polynomial gLocks gOps)

------------------------------------------------------------------------
-- the runner

isSetGLk : isSet GLk
isSetGLk = isProp→isSet λ { tick tick → refl }

isSetGBase : isSet GBase
isSetGBase = isProp→isSet λ ()

-- no state beyond the fuel
S : ℕ → Type
S _ = Unit

isSetConf : isSet (Σ ℕ S)
isSetConf = isSetΣ isSetℕ λ _ → isSetUnit

-- a step spends one unit of fuel; with none left it times out
δ : ∀ {n} → S n → (o : P.Shape n) → Unit ⊎ (Σ[ p ∈ P.Position n o ] S (P.scope n o p))
δ {zero} _ (step , tt) = inl tt
δ {suc m} _ (step , tt) = inr ((tt , tt) , tt)

------------------------------------------------------------------------
-- runs

open RunMachine gLocks gOps isSetGLk isSetGBase S Unit isSetConf isSetUnit δ 𝟙

-- two steps with two units of fuel: the program returns at world 0
delay-delay : running {2} tt (delay (delay (return tt))) ⟶* running {0} tt (return tt)
delay-delay = op-at then* op-at

-- two steps with one unit of fuel: the second step times out
delay-delay-timeout : running {1} tt (delay (delay (return tt))) ⟶* halted tt
delay-delay-timeout = op-at then* op-at

delay-delay-sound : run tt (denClosed (delay {Θ = ∅ 2} (delay (return tt)))) ≡ Rn.running ⟦ 𝟙 ⟧v {0} tt tt
delay-delay-sound = run-return delay-delay

delay-delay-timeout-sound : run tt (denClosed (delay {Θ = ∅ 1} (delay (return tt)))) ≡ Rn.halted ⟦ 𝟙 ⟧v tt
delay-delay-timeout-sound = run-halt delay-delay-timeout

-- the same equations by computation (refl), independently of the machine
delay-delay-den : run tt (denClosed (delay {Θ = ∅ 2} (delay (return tt)))) ≡ Rn.running ⟦ 𝟙 ⟧v {0} tt tt
delay-delay-den = refl

delay-delay-timeout-den : run tt (denClosed (delay {Θ = ∅ 1} (delay (return tt)))) ≡ Rn.halted ⟦ 𝟙 ⟧v tt
delay-delay-timeout-den = refl

-- refl probe (judge-feas Compute.agda): the denotation computes
delayRet : ∀ {n} →
  denClosed (delay {Θ = ∅ (suc n)} (return tt)) ≡ ops (step , tt) (λ _ → var tt)
delayRet = refl
