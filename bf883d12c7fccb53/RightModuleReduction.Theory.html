-- Presheaves on W^op are covariant world-indexed families.
Signature : Category ℓ-zero ℓ-zero → Type₁
Signature W = Poly {ℓShape = ℓ-zero} {ℓPos = ℓ-zero} (W ^op) (W ^op)

module SignatureNotation {W : Category ℓ-zero ℓ-zero} (P : Signature W) where
  Shape : W .ob → Type
  Shape = Poly.ShapesAt P

  Position : (w : W .ob) → Shape w → Type
  Position w o = Poly.Positions P (w , o)

  scope : (w : W .ob) (o : Shape w) → Position w o → W .ob
  scope w o k = Poly.s P .F-ob ((w , o) , k)

module _ {W : Category ℓ-zero ℓ-zero} (P : Signature W) where
  open SignatureNotation P

  𝓦 : Category (ℓ-suc ℓ-zero) ℓ-zero
  𝓦 = PresheafCategory (W ^op) ℓ-zero
  -- Terms over a presheaf of variables; world reindexing is defined recursively.
  data Term (X : 𝓦 .ob) : W .ob → Type where
    var : {w : W .ob} → X .F-ob w .fst → Term X w
    ops : {w : W .ob} (o : Shape w) →
      ((k : Position w o) → Term X (scope w o k)) → Term X w

  reind : {w w′ : W .ob} {X : 𝓦 .ob} →
    W [ w , w′ ] → Term X w → Term X w′
  reind {X = X} f (var x) = var (X .F-hom f x)
  reind {w} {w′} f (ops s k) =
    let s′ = P .Poly.Shapes .F-hom f s
        shapeMap : Poly.B P [ (w′ , s′) , (w , s) ]
        shapeMap = f , refl
    in ops s′ λ p →
      reind (P .Poly.s .F-hom (shapeMap , refl))
        (k (Poly.transportPosition P shapeMap p))

  private
    module Sethood (X : 𝓦 .ob) where
      Labels : W .ob → Type
      Labels w = (X ⟅ w ⟆) .fst ⊎ Shape w

      Arity : (w : W .ob) → Labels w → Type
      Arity w (inl _) = ⊥
      Arity w (inr o) = Position w o

      input : (w : W .ob) (o : Labels w) → Arity w o → W .ob
      input w (inl _) ()
      input w (inr o) k = scope w o k

      Tree = IW.IW Labels Arity input

      encode : {w : W .ob} → Term X w → Tree w
      encode (var x) = IW.node (inl x) (λ ())
      encode (ops o k) = IW.node (inr o) (λ p → encode (k p))

      decode : {w : W .ob} → Tree w → Term X w
      decode (IW.node (inl x) _) = var x
      decode (IW.node (inr o) k) = ops o (λ p → decode (k p))

      retract : {w : W .ob} (t : Term X w) → decode (encode t) ≡ t
      retract (var x) = refl
      retract (ops o k) = cong (ops o) (funExt λ p → retract (k p))

      set : (w : W .ob) → isSet (Term X w)
      set w = isSetRetract encode decode retract
        (IW.isOfHLevelSuc-IW 1
          (λ v → isSet⊎ ((X ⟅ v ⟆) .snd) ((P .Poly.Shapes ⟅ v ⟆) .snd)) w)

    B = Poly.B P
    E = Poly.E P
    Pos = P .Poly.Pos
    sorts = P .Poly.s

    Args : (X : 𝓦 .ob) → B .ob → Type
    Args X b = (p : Poly.Positions P b) → Term X (sorts ⟅ b , p ⟆)

    liftPosition : {b b′ : B .ob} (f : B [ b , b′ ]) (p : Poly.Positions P b) →
      E [ (b , p) , (b′ , Poly.transportPosition P f p) ]
    liftPosition f p = f , refl

    mapArgs : {X : 𝓦 .ob} {b b′ : B .ob} → B [ b , b′ ] → Args X b′ → Args X b
    mapArgs f k p = reind (sorts .F-hom (liftPosition f p)) (k (Poly.transportPosition P f p))

    args-id : {X : 𝓦 .ob} {b : B .ob} (k : Args X b) →
      ((p : Poly.Positions P b) → reind (W .id) (k p) ≡ k p) →
      mapArgs (B .id) k ≡ k
    args-id {X} {b} k ih = funExt λ p →
      deform p
      ∙ cong (λ h → reind (sorts .F-hom h) (k p))
          (Covariant.ElementHom≡ Pos refl)
      ∙ cong (λ h → reind h (k p)) (sorts .F-id)
      ∙ ih p
      where
      positionPath : (p : Poly.Positions P b) → Poly.transportPosition P (B .id) p ≡ p
      positionPath p = funExt⁻ (Pos .F-id) p

      deform : (p : Poly.Positions P b) → mapArgs (B .id) k p ≡
        reind (sorts .F-hom (B .id , positionPath p)) (k p)
      deform p i = reind
        (sorts .F-hom (B .id , λ j → positionPath p (i ∧ j)))
        (k (positionPath p i))

    args-seq : {X : 𝓦 .ob} {b b′ b″ : B .ob}
      (f : B [ b , b′ ]) (g : B [ b′ , b″ ]) (k : Args X b″) →
      ((p : Poly.Positions P b″) {v w : W .ob}
        (h : W [ sorts ⟅ b″ , p ⟆ , v ]) (j : W [ v , w ]) →
        reind (W ._⋆_ h j) (k p) ≡ reind j (reind h (k p))) →
      mapArgs (B ._⋆_ f g) k ≡ mapArgs f (mapArgs g k)
    args-seq {X} {b} {b′} {b″} f g k ih = funExt λ p →
      deform p
      ∙ cong (λ h → reind (sorts .F-hom h) (k (Pos .F-hom g (Pos .F-hom f p))))
          (Covariant.ElementHom≡ Pos refl)
      ∙ cong (λ h → reind h (k (Pos .F-hom g (Pos .F-hom f p))))
          (sorts .F-seq (liftPosition f p) (liftPosition g (Pos .F-hom f p)))
      ∙ ih (Pos .F-hom g (Pos .F-hom f p))
          (sorts .F-hom (liftPosition g (Pos .F-hom f p))) (sorts .F-hom (liftPosition f p))
      where
      positionPath : (p : Poly.Positions P b) →
        Pos .F-hom (B ._⋆_ f g) p ≡ Pos .F-hom g (Pos .F-hom f p)
      positionPath p = funExt⁻ (Pos .F-seq f g) p

      deform : (p : Poly.Positions P b) → mapArgs (B ._⋆_ f g) k p ≡
        reind (sorts .F-hom (B ._⋆_ f g , positionPath p))
          (k (Pos .F-hom g (Pos .F-hom f p)))
      deform p i = reind
        (sorts .F-hom (B ._⋆_ f g , λ j → positionPath p (i ∧ j)))
        (k (positionPath p i))

    reind-id : {X : 𝓦 .ob} {w : W .ob} (t : Term X w) → reind (W .id) t ≡ t
    reind-id {X} (var x) = cong var (funExt⁻ (X .F-id) x)
    reind-id {X} {w} (ops o k) = deform ∙ cong (ops o) (args-id k (λ p → reind-id (k p)))
      where
      shapePath : P .Poly.Shapes .F-hom (W .id) o ≡ o
      shapePath = funExt⁻ (P .Poly.Shapes .F-id) o

      shapeLift : PathP (λ i → B [ (w , shapePath i) , (w , o) ])
        (W .id , refl) (B .id)
      shapeLift = Contravariant.ElementHomPathP (P .Poly.Shapes)
        (W .id , refl) (B .id) (λ i → w , shapePath i) refl refl

      deform : reind (W .id) (ops o k) ≡ ops o (mapArgs (B .id) k)
      deform i = ops (shapePath i) (mapArgs (shapeLift i) k)

    reind-seq : {X : 𝓦 .ob} {u : W .ob} (t : Term X u)
      {v w : W .ob} (f : W [ u , v ]) (g : W [ v , w ]) →
      reind (W ._⋆_ f g) t ≡ reind g (reind f t)
    reind-seq {X} (var x) f g = cong var (funExt⁻ (X .F-seq f g) x)
    reind-seq {X} {u} (ops o k) {v} {w} f g = deform
      ∙ cong (ops _) (args-seq (g , refl) (f , refl) k (λ p → reind-seq (k p)))
      where
      shapePath : P .Poly.Shapes .F-hom (W ._⋆_ f g) o ≡
        P .Poly.Shapes .F-hom g (P .Poly.Shapes .F-hom f o)
      shapePath = funExt⁻ (P .Poly.Shapes .F-seq f g) o

      shapeLift : PathP (λ i → B [ (w , shapePath i) , (u , o) ])
        (W ._⋆_ f g , refl) (B ._⋆_ (g , refl) (f , refl))
      shapeLift = Contravariant.ElementHomPathP (P .Poly.Shapes)
        (W ._⋆_ f g , refl) (B ._⋆_ (g , refl) (f , refl))
        (λ i → w , shapePath i) refl refl

      deform : reind (W ._⋆_ f g) (ops o k) ≡
        ops (P .Poly.Shapes .F-hom g (P .Poly.Shapes .F-hom f o))
          (mapArgs (B ._⋆_ (g , refl) (f , refl)) k)
      deform i = ops (shapePath i) (mapArgs (shapeLift i) k)

  TermPsh : 𝓦 .ob → 𝓦 .ob
  TermPsh X .F-ob w = Term X w , Sethood.set X w
  TermPsh X .F-hom = reind
  TermPsh X .F-id = funExt reind-id
  TermPsh X .F-seq f g = funExt λ t → reind-seq t f g


record Theory (W : Category ℓ-zero ℓ-zero) : Type₁ where
  field
    signature : Signature W
    Equation : Type
    equationWorld : Equation → W .ob
    Variables : Equation → Presheaf (W ^op) ℓ-zero
    lhs rhs : (e : Equation) →
      Term signature (Variables e) (equationWorld e)

-- A natural interpretation of the signature's operation layers.
record Algebra {W : Category ℓ-zero ℓ-zero} (P : Signature W) : Type₁ where
  field
    Carrier : Presheaf (W ^op) ℓ-zero
    algebra : NatTrans (Poly.Extension P ⟅ Carrier ⟆) Carrier

module _ {W : Category ℓ-zero ℓ-zero} {P : Signature W} (A : Algebra P) where
  private
    carrier = A .Algebra.Carrier

  private
    evaluateAt : {X : Presheaf (W ^op) ℓ-zero} → NatTrans X carrier →
      {w : W .ob} → Term P X w → (carrier ⟅ w ⟆) .fst
    evaluateAt env {w} (var x) = env .N-ob w x
    evaluateAt env {w} (ops o args) =
      A .Algebra.algebra .N-ob w (o , λ k → evaluateAt env (args k))

    evaluate-natural : {X : Presheaf (W ^op) ℓ-zero} (env : NatTrans X carrier)
      {w w′ : W .ob} (f : W [ w , w′ ]) (t : Term P X w) →
      evaluateAt env (reind P f t) ≡ carrier .F-hom f (evaluateAt env t)
    evaluate-natural env f (var x) = funExt⁻ (env .N-hom f) x
    evaluate-natural env {w} {w′} f (ops o args) =
      cong (A .Algebra.algebra .N-ob w′)
        (ΣPathP (refl , funExt λ p →
          evaluate-natural env (P .Poly.s .F-hom ((f , refl) , refl))
            (args (Poly.transportPosition P (f , refl) p))))
      ∙ funExt⁻ (A .Algebra.algebra .N-hom f)
          (o , λ k → evaluateAt env (args k))

  -- Extend a natural variable assignment to a natural interpretation of terms.
  evaluate : {X : Presheaf (W ^op) ℓ-zero} →
    NatTrans X carrier → NatTrans (TermPsh P X) carrier
  evaluate env .N-ob w = evaluateAt env
  evaluate env .N-hom f = funExt λ t → evaluate-natural env f t

module _ {W : Category ℓ-zero ℓ-zero} (𝒯 : Theory W) where
  open Theory 𝒯

  -- Every equation holds under every natural assignment of its variables.
  Satisfies : Algebra signature → Type
  Satisfies A = (e : Equation)
    (env : NatTrans (Variables e) (A .Algebra.Carrier)) →
    evaluate A env .N-ob (equationWorld e) (lhs e) ≡
    evaluate A env .N-ob (equationWorld e) (rhs e)

  isPropSatisfies : (A : Algebra signature) → isProp (Satisfies A)
  isPropSatisfies A = isPropΠ λ e → isPropΠ λ env →
    (A .Algebra.Carrier ⟅ equationWorld e ⟆) .snd _ _

  record Model : Type₁ where
    field
      underlying : Algebra signature
      equations : Satisfies underlying
