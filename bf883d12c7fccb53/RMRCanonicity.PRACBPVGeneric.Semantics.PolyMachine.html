-- The combined machine at Q = T for a free model presented by a container
-- (Semantics.PolySelfModule), for a theory WITH equations: POPL's
-- configuration reduction ↦T and its ConfigNormalization, generic in the
-- theory and the presentation.
--
-- Data: a lock signature L and an operation signature O, isSet Lk and
-- isSet Base (for the syntactic laws), a theory eqs with a free model fm,
-- a container Q′ presenting fm's carrier at a world w₀ (pres), and the
-- result type A.  The right module is ρ = μ (PolySelfModule.polyModule).
--
--   ηC M            the unit configuration: one thread, holding M
--   denote-ηC       its denotation is h M, presented (PolySelfModule.denote-η)
--   ctxAt, plug-unplug, pure-in, op-in, effect-in, spliced
--   Affine, affine-subsingleton, and the generic redex of ρ = μ (Gen,
--                   origin, genSpliced, origin-is) for affinity proofs
--                   the machine's two steps at any hole of any
--                   configuration (op-in's target is RMR's step there)
--   Final, retF, denF, den-final, final-unique
--                   Semantics.ConfigCanonicity at this module (no affinity
--                   needed)
--
-- module Comparing ec (models closed under exponentials):
--
--   denote-ηC-free     at a first-order A, the machine's denotation of
--                      η M, renamed along same, is the FREE-MODEL
--                      denotation of M, presented
--                      (Free.Comparison.machine-comparison)
--   final-comparison   so is the denotation of every final configuration
--                      η M reaches: the values the machine returns are
--                      read off the free-model denotation of M
--
-- module WithAffinity finOp finQ aff (Semantics.Normalization):
--
--   strongNormalization, progress, weakNormalization, wn-sound,
--   normalize, ground-canonicity, canonicity-T (η M reaches a final
--   configuration), ground-canonicity-T (at a ground type, uniquely)

  {W : Category ℓ-zero ℓ-zero} (L : LockSig W) (O : OpSig L)
  (isSetLk : isSet (LockSig.Lk L)) (isSetBase : isSet (LockSig.Base L))

  (Q′ : IndexedContainer (Category.ob W)) (w₀ : Category.ob W)
  (pres : PolySelfModule.Presentation (Machine.theory L O eqs) fm Q′ w₀)
  (A : Types.ValTy (LockSig.Lk L) (LockSig.Base L))
  where

-- ρ = μ
module PS = PolySelfModule (Machine.theory L O eqs) fm Q′ w₀
open PolySelfModule.Presentation pres public using (to; from)

-- the machine with the proved syntactic laws, this module, and result
-- type A
open Machine.WithLaws L O closedLaws public
open Run eqs fm Q′ (PS.polyModule pres) A public

open IndexedContainer Q′ using () renaming (Shape to CShape; Pos to CPos)

private
  X = CompPsh (F A)

------------------------------------------------------------------------
-- the unit configuration

ηC : Comp (∅ w₀) (F A) → Config
ηC M = PS.η-conf pres X M

denote-ηC : (M : Comp (∅ w₀) (F A)) →
  denote (ηC M) ≡ from ⟦ A ⟧v (h .Hom.transformation .NatTrans.N-ob w₀ M)
denote-ηC M = PS.denote-η pres (CompAlg (F A)) ⟦ A ⟧v h M

------------------------------------------------------------------------
-- the steps at any hole

-- unplugging a hole
ctxAt : (x : Config) → CPos (x .fst) .fst → Ctxt
ctxAt (s , f) p = (s , p) , λ r _ → f r

-- ... and plugging it back
plug-unplug : (x : Config) (p : CPos (x .fst) .fst) → plug Q′ X (ctxAt x p) (x .snd p) ≡ x
plug-unplug (s , f) p = ΣPathP (refl , funExt λ r → at r (IndexedContainer.decPos Q′ s r p))
  where
  at : (r : CPos s .fst) → Dec (r ≡ p) → plug Q′ X (ctxAt (s , f) p) (f p) .snd r ≡ f r
  at r (yes e) = subst (λ r → plug Q′ X (ctxAt (s , f) p) (f p) .snd r ≡ f r) (sym e)
    (plug-hole Q′ X (ctxAt (s , f) p) (f p))
  at r (no ne) = plug-remaining Q′ X (ctxAt (s , f) p) (f p) r ne

private
  from≡ : ∀ {x x' y} → x ≡ x' → x ⟶ y → x' ⟶ y
  from≡ {y = y} e st = subst (_⟶ y) e st

-- a tree step at a hole
pure-in : (x : Config) (p : CPos (x .fst) .fst) {M' : Comp (∅ _) (F A)} →
  x .snd p ↦ M' → x ⟶ plug Q′ X (ctxAt x p) M'
pure-in x p r = from≡ (plug-unplug x p) (pure (ctxAt x p) r)

-- an effect step at a hole holding a closed op
op-in : (x : Config) (p : CPos (x .fst) .fst) {o : Op}
  {vs : Vals (∅ _) (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) _) → Comp (∅ _ ⧀*⟨ slot o a , ps ⟩) (F A)} →
  x .snd p ≡ op o vs k →
  x ⟶ ER.step (PS.polyModule pres) (CompAlg (F A)) (ctxAt x p)
        ((o , closeVals vs) , λ (a , ps) → closeK (slot o a) (k a) ps)
op-in x p {o} {vs} {k} e =
  from≡ (cong (plug Q′ X (ctxAt x p)) (sym e) ∙ plug-unplug x p) (effect (ctxAt x p) o vs k)

------------------------------------------------------------------------
-- what RMR's effect step computes

private
  T₀ = ER.T .fst
  ηX = IsMonad.η (ER.T .snd) .NatTrans.N-ob X
  μX = IsMonad.μ (ER.T .snd) .NatTrans.N-ob X
  Σ𝒯 = Poly.Extension (PM.polynomial L O) {ℓX = ℓ-zero}

-- the free model's operation, over the units of the continuations
opT : ∀ {w} → (Σ𝒯 ⟅ X ⟆ ⟅ w ⟆) .fst → (T₀ ⟅ X ⟆ ⟅ w ⟆) .fst
opT {w} ok = Algebra.algebra ((FreeModel.free fm ⟅ X ⟆) .fst) .NatTrans.N-ob w ((Σ𝒯 ⟪ ηX ⟫) .NatTrans.N-ob w ok)

-- the configuration over T X that the step multiplies: the operation at
-- the hole p, the unit of its computation at every other hole
spliced : (x : Config) (p : CPos (x .fst) .fst) →
  (Σ𝒯 ⟅ X ⟆ ⟅ IndexedContainer.input Q′ (x .fst) p ⟆) .fst → (ER.Q ⟅ T₀ ⟅ X ⟆ ⟆) .fst
spliced x p ok = plug Q′ (T₀ ⟅ X ⟆) ((ER.∂Q ⟪ ηX ⟫) (ctxAt x p)) (opT ok)

spliced-hole : (x : Config) (p : CPos (x .fst) .fst) → ∀ ok → spliced x p ok .snd p ≡ opT ok
spliced-hole x p ok = plug-hole Q′ (T₀ ⟅ X ⟆) ((ER.∂Q ⟪ ηX ⟫) (ctxAt x p)) (opT ok)

spliced-other : (x : Config) (p : CPos (x .fst) .fst) → ∀ ok (r : CPos (x .fst) .fst) → ¬ r ≡ p →
  spliced x p ok .snd r ≡ ηX .NatTrans.N-ob _ (x .snd r)
spliced-other x p ok r ne = plug-remaining Q′ (T₀ ⟅ X ⟆) ((ER.∂Q ⟪ ηX ⟫) (ctxAt x p)) (opT ok) r ne

-- an effect step at a hole holding a closed op: μ of the spliced
-- configuration
effect-in : (x : Config) (p : CPos (x .fst) .fst) {o : Op}
  {vs : Vals (∅ _) (params o)}
  {k : (a : Ar o) (ps : WPos (slot o a) _) → Comp (∅ _ ⧀*⟨ slot o a , ps ⟩) (F A)} →
  x .snd p ≡ op o vs k →
  x ⟶ from X (μX .NatTrans.N-ob w₀
         (to (T₀ ⟅ X ⟆) (spliced x p ((o , closeVals vs) , λ (a , ps) → closeK (slot o a) (k a) ps))))
effect-in x p e = op-in x p e

module CC = ConfigCanonicity L O closedLaws eqs fm Q′ (PS.polyModule pres) A
open CC public using (Final; retF; denF; den-final; final-unique)

------------------------------------------------------------------------
-- the comparison with the free-model denotation

module Comparing (ec : StrongTheory.ExpClosed L O eqs) where
  module Fr = FreeDen L O eqs fm ec
  module Cmp = Comparison L O eqs fm ec

  -- the machine's denotation of η M, renamed along same, is the
  -- free-model denotation of M
  denote-ηC-free : (fo : Cmp.FirstOrder A) (M : Comp (∅ w₀) (F A)) →
    (ER.Q ⟪ PshHom→NatTrans (Cmp.same fo) ⟫) (denote (ηC M))
      ≡ from Fr.⟦ A ⟧v (Fr.⟦ M ⟧C .PshHom.N-ob w₀ W.id)
  denote-ηC-free fo M =
    cong (ER.Q ⟪ PshHom→NatTrans (Cmp.same fo) ⟫) (denote-ηC M)
    ∙ sym (PS.from-nat pres (PshHom→NatTrans (Cmp.same fo)) _)
    ∙ cong (from Fr.⟦ A ⟧v) (Cmp.machine-comparison fo M)

  -- the values of a final configuration that η M reaches, renamed along
  -- same, are the free-model denotation of M
  final-comparison : (fo : Cmp.FirstOrder A) (M : Comp (∅ w₀) (F A)) (ṽ : Final) →
    ηC M ⟶* retF ṽ →
    (ER.Q ⟪ PshHom→NatTrans (Cmp.same fo) ⟫) (denF ṽ)
      ≡ from Fr.⟦ A ⟧v (Fr.⟦ M ⟧C .PshHom.N-ob w₀ W.id)
  final-comparison fo M ṽ run =
    cong (ER.Q ⟪ PshHom→NatTrans (Cmp.same fo) ⟫) (sym (den-final ṽ) ∙ sym (sound⟶* run))
    ∙ denote-ηC-free fo M

------------------------------------------------------------------------
-- affinity when every configuration has at most one hole

-- module affinity of ρ = μ (Semantics.Affinity).  An instance states its
-- affinity as Affine, never by restating the right-hand side: elaborating
-- PS.polyModule pres against Affinity's form of the module type at a
-- concrete free model unfolds the free model.
Affine : Type
Affine = Affinity.ModuleAffine (Machine.theory L O eqs) fm Q′ (PS.polyModule pres)

affine-subsingleton : (∀ s → isProp (CPos s .fst)) → Affine
affine-subsingleton = Affinity.affine-subsingleton (Machine.theory L O eqs) fm Q′ (PS.polyModule pres)

-- the generic redex (s , p , o) of ρ = μ, for proofs of Affine at an
-- instance (which never restates Affinity's names at the module)
module _ (s : CShape .fst) (p : CPos s .fst) (o : SN.Shape (IndexedContainer.input Q′ s p)) where
  private
    module Aff = Affinity (Machine.theory L O eqs) fm Q′
    R = PS.polyModule pres

  -- the generators: the other holes and the operation's positions
  Gen : ER.𝒱 .Category.ob
  Gen = Aff.Gen R s p o

  -- the hole of the result a filler comes from
  origin : CPos (Aff.composite R s p o) .fst → Aff.Source R s p o
  origin = Aff.origin R s p o

  private
    ηG = IsMonad.η (ER.T .snd) .NatTrans.N-ob Gen
    μG = IsMonad.μ (ER.T .snd) .NatTrans.N-ob Gen

  -- the generic configuration over T Gen that the step multiplies: the
  -- operation over the fresh generators at p, the unit of its own
  -- generator at every other hole
  genSpliced : (ER.Q ⟅ T₀ ⟅ Gen ⟆ ⟆) .fst
  genSpliced = plug Q′ (T₀ ⟅ Gen ⟆) ((ER.∂Q ⟪ ηG ⟫) (Aff.genCtx R s p o))
    (Algebra.algebra ((FreeModel.free fm ⟅ Gen ⟆) .fst) .NatTrans.N-ob _
      ((Σ𝒯 ⟪ ηG ⟫) .NatTrans.N-ob _ (Aff.genArgs R s p o)))

  -- the origins are read off μ of it
  origin-is : ∀ r → origin r ≡ from Gen (μG .NatTrans.N-ob w₀ (to (T₀ ⟅ Gen ⟆) genSpliced)) .snd r .fst
  origin-is r = refl

  genSpliced-hole : genSpliced .snd p ≡
    Algebra.algebra ((FreeModel.free fm ⟅ Gen ⟆) .fst) .NatTrans.N-ob _
      ((Σ𝒯 ⟪ ηG ⟫) .NatTrans.N-ob _ (Aff.genArgs R s p o))
  genSpliced-hole = plug-hole Q′ (T₀ ⟅ Gen ⟆) ((ER.∂Q ⟪ ηG ⟫) (Aff.genCtx R s p o)) _

  genSpliced-other : ∀ r (ne : ¬ r ≡ p) →
    genSpliced .snd r ≡ ηG .NatTrans.N-ob _ (Aff.genCtx R s p o .snd r ne)
  genSpliced-other r ne = plug-remaining Q′ (T₀ ⟅ Gen ⟆) ((ER.∂Q ⟪ ηG ⟫) (Aff.genCtx R s p o)) _ r ne

-- normalization and canonicity, under the hypotheses of
-- Semantics.Normalization

module WithAffinity
  -- every operation has finitely many continuations at every world
  (finOp : ∀ (o : Op) w → isFinSet (Σ[ a ∈ Ar o ] Notation.WPos L (slot o a) w))
  -- every configuration shape has finitely many holes, numbered
  (finQ : ∀ s → isFinOrd (CPos s .fst))
  -- the module is affine
  (aff : Affine)
  where

  open Normalization L O closedLaws eqs fm Q′ (PS.polyModule pres) A finOp finQ aff public
    using (Terminal; strongNormalization; progress; weakNormalization; wn-sound)
  open CC.WithAffinity finOp finQ aff public using (normalize; ground-canonicity)

  -- η M reaches a final configuration (POPL's canonicity-T)
  canonicity-T : (M : Comp (∅ w₀) (F A)) → Σ[ ṽ ∈ Final ] ηC M ⟶* retF ṽ
  canonicity-T M = normalize (ηC M)

  -- at a ground type, a unique one
  ground-canonicity-T : Ground A → (M : Comp (∅ w₀) (F A)) →
    Σ[ ṽ ∈ Final ] (ηC M ⟶* retF ṽ) × (∀ ṽ' → ηC M ⟶* retF ṽ' → ṽ ≡ ṽ')
  ground-canonicity-T g M = ground-canonicity g (ηC M)
