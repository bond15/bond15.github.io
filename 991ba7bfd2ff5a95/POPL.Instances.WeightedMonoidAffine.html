{-
  The weighted-monoid operational model is affine (paper §5.4, Example:
  "Writer and weighted monoid are affine as all reductions preserve the
  number of subterms").

  The shape length n is a variable, so ρ_{s,i,op} cannot be computed
  symbolically.  We count occurrences instead.  T(Y) is interpreted in the
  model of occurrence counts Y → ℕ (⊗ is +, e is 0, act_r is the
  identity), so ext δ (s, ρ) y = #{p | ρ p = y}.  In step(s,i,op) each
  variable occurs at most once:
    • a hole variable inl(q) occurs once, at position q;
    • the argument variables inr(j) occur at most once, inside ⟦op⟧(η y⃗).
  A function whose fibres have at most one element is injective.
-}

------------------------------------------------------------------------
-- Counting

module _ {Y : Type} (dec : Discrete Y) where
  δ' : {a b : Y} → Dec (a ≡ b) → ℕ
  δ' (yes _) = 1
  δ' (no _)  = 0

  -- δ a b = 1 if a = b, else 0
  δ : Y → Y → ℕ
  δ a b = δ' (dec a b)

  δ-yes : {a b : Y} → a ≡ b → δ a b ≡ 1
  δ-yes {a} {b} p = go (dec a b)
    where
    go : (d : Dec (a ≡ b)) → δ' d ≡ 1
    go (yes _) = refl
    go (no ¬p) = ⊥.rec (¬p p)

  δ-no : {a b : Y} → ¬ a ≡ b → δ a b ≡ 0
  δ-no {a} {b} ¬p = go (dec a b)
    where
    go : (d : Dec (a ≡ b)) → δ' d ≡ 0
    go (yes p) = ⊥.rec (¬p p)
    go (no _)  = refl

  δ≤1 : (a b : Y) → δ a b ≤ 1
  δ≤1 a b = go (dec a b)
    where
    go : (d : Dec (a ≡ b)) → δ' d ≤ 1
    go (yes _) = ≤-refl
    go (no _)  = zero-≤

  -- Number of positions of f hitting y.
  count : {n : ℕ} → (Fin n → Y) → Y → ℕ
  count {n} f y = sumFin n (λ p → δ (f p) y)

  count-zero : {n : ℕ} (f : Fin n → Y) (y : Y) → (∀ p → ¬ f p ≡ y) → count f y ≡ 0
  count-zero {zero}  f y ne = refl
  count-zero {suc n} f y ne =
    cong₂ _+_ (δ-no (ne zero)) (count-zero (f ∘ suc) y (ne ∘ suc))

  -- injective ⇒ every fibre has at most one element
  inj→count≤1 : {n : ℕ} (f : Fin n → Y) → (∀ p q → f p ≡ f q → p ≡ q) → ∀ y → count f y ≤ 1
  inj→count≤1 {zero}  f inj y = zero-≤
  inj→count≤1 {suc n} f inj y = go (dec (f zero) y)
    where
    go : Dec (f zero ≡ y) → count f y ≤ 1
    go (yes e) = subst (_≤ 1) (sym (cong₂ _+_ (δ-yes e) (count-zero (f ∘ suc) y
                   λ p q → fsnotz (inj (suc p) zero (q ∙ sym e))))) ≤-refl
    go (no ¬e) = subst (_≤ 1) (sym (cong (_+ count (f ∘ suc) y) (δ-no ¬e)))
                   (inj→count≤1 (f ∘ suc) (λ p q r → injSucFin (inj (suc p) (suc q) r)) y)

  -- every fibre has at most one element ⇒ injective
  count≤1→inj : {n : ℕ} (f : Fin n → Y) → (∀ y → count f y ≤ 1) → ∀ p q → f p ≡ f q → p ≡ q
  count≤1→inj {zero}  f c p q e = ⊥.rec (¬Fin0 p)
  count≤1→inj {suc m} f c p q e with discreteFin q p
  ... | yes q≡p = sym q≡p
  ... | no q≢p  = ⊥.rec (¬-<-zero (≤-k+-cancel {k = 1} two≤one))
    where
    g : Fin (suc m) → ℕ
    g r = δ (f r) (f p)
    q' = punchOut p q q≢p
    two≤ : 2 ≤ count f (f p)
    two≤ = subst (2 ≤_) (sym (sum-punch m g p))
             (≤-+-≤ (≤-reflexive (sym (δ-yes refl)))
               (≤-trans (≤-reflexive (sym (δ-yes (cong f (punchIn-punchOut p q q≢p) ∙ sym e))))
                        (sum-split m (g ∘ punchIn p) q')))
    two≤one : 1 + 1 ≤ 1 + 0
    two≤one = ≤-trans two≤ (c (f p))

-- Weighted monoids are affine

module _ (R : Type) (mon : MonoidOn R) where
  open MonoidOn mon renaming (e to 1r; _⊗_ to _·_)
  open Model
  open FreeModel (wmFree R mon)
  open import POPL.ConfigReduction (wmTheory R mon) (wmOM R mon)
    using (Affine; ρ; step; C-fresh; ⌈_⌉)
  open import POPL.ConfigNormalization (wmTheory R mon) (wmOM R mon)
    using (plug-sum; holeSum; holeIn)

  private
    sigR = sigWM R mon
    T' = T R mon

    -- The occurrence-count model on Y → ℕ.
    countModel : (Y : Type) → Model (wmTheory R mon)
    countModel Y .Carrier = Y → ℕ
    countModel Y .alg unit    a = λ _ → 0
    countModel Y .alg mul     a = λ y → a zero y + a (suc zero) y
    countModel Y .alg (act r) a = a zero
    countModel Y .sat e-assoc γ = funExt λ y → +-assoc (γ zero y) _ _
    countModel Y .sat e-unitl γ = refl
    countModel Y .sat e-unitr γ = funExt λ y → +-zero (γ zero y)
    countModel Y .sat e-act1  γ = refl
    countModel Y .sat (e-actact r r') γ = refl
    countModel Y .sat (e-actdist r) γ = refl
    countModel Y .sat (e-acte r) γ = refl

    -- ext into the count model is a sum over positions.
    ext-count : {X Y : Type} (h : X → Y → ℕ) (t : T' X) (y : Y)
      → ext (countModel Y) h t y ≡ sumFin _ (λ p → h (t .snd p) y)
    ext-count h ((n , w) , γ) y = go n w γ
      where
      go : (n : ℕ) (w : Fin n → R) (γ : Fin n → _)
        → extL R mon (countModel _) h (toL' R mon n w γ) y ≡ sumFin n (λ p → h (γ p) y)
      go zero    w γ = refl
      go (suc n) w γ = cong (h (γ zero) y +_) (go n (w ∘ suc) (γ ∘ suc))

  wm-affine : Affine
  wm-affine (zero , w) i o = ⊥.rec (¬Fin0 i)
  wm-affine s@(suc m , w) i o = count≤1→inj decY (ρ s i o) bound
    where
    n = suc m
    b = Signature.ar sigR o

    Y : Type
    Y = Fin∖ n i ⊎ Fin b

    decFin∖ : Discrete (Fin∖ n i)
    decFin∖ (j , ne) (j' , ne') with discreteFin j j'
    ... | yes p = yes (ΣPathP (p , isProp→PathP (λ _ → isProp¬ _) ne ne'))
    ... | no ¬p = no (¬p ∘ cong fst)

    decY : Discrete Y
    decY = discrete⊎ decFin∖ discreteFin

    occ : T' Y → Y → ℕ
    occ = ext (countModel Y) (δ decY)

    occ-η : (y' y : Y) → occ (η y') y ≡ δ decY y' y
    occ-η y' y = cong (λ f → f y) (ext-η (countModel Y) (δ decY) y')

    -- X = T(σ₂)(⌈op⌉)
    X : T' Y
    X = map inr ⌈ o ⌉

    occ-X : (y : Y) → occ X y ≡ countModel Y .alg o (λ j → δ decY (inr j)) y
    occ-X y = cong (λ f → f y)
      ( ext-map (countModel Y) (δ decY) inr ⌈ o ⌉
      ∙ ext-hom (countModel Y) (δ decY ∘ inr) .pres o η
      ∙ cong (countModel Y .alg o) (funExt λ j → ext-η (countModel Y) (δ decY ∘ inr) j))

    holes : Y → ℕ
    holes y = sumFin m (λ k → δ decY (inl (holeIn i k)) y)

    -- occurrences in step(s,i,op) = occurrences in X + one per hole
    occ-step : (y : Y) → count decY (ρ s i o) y ≡ occ X y + holes y
    occ-step y =
      sym (ext-count (δ decY) (step s i o) y)
      ∙ cong (λ f → f y) (ext-μ (countModel Y) (δ decY) (∂map (η ∘ inl) (C-fresh s i) ⟪ X ⟫))
      ∙ ext-count occ (∂map (η ∘ inl) (C-fresh s i) ⟪ X ⟫) y
      ∙ plug-sum (∂map (η ∘ inl) (C-fresh s i)) X (λ t → occ t y)
      ∙ cong (occ X y +_) (sum-cong m λ k → occ-η (inl (holeIn i k)) y)

    isInl : Y → Bool
    isInl (inl _) = true
    isInl (inr _) = false

    inl≢inr : {a : Fin∖ n i} {c : Fin b} → ¬ (_≡_ {A = Y} (inl a) (inr c))
    inl≢inr p = true≢false (cong isInl p)

    inl-inj : {a a' : Fin∖ n i} → _≡_ {A = Y} (inl a) (inl a') → a ≡ a'
    inl-inj {a} p = cong (⊎.rec (λ z → z) (λ _ → a)) p

    inr-inj : {c c' : Fin b} → _≡_ {A = Y} (inr c) (inr c') → c ≡ c'
    inr-inj {c} p = cong (⊎.rec (λ _ → c) (λ z → z)) p

    -- each hole variable occurs once
    holes≤1 : (y : Y) → holes y ≤ 1
    holes≤1 y = inj→count≤1 decY (λ k → inl (holeIn i k))
      (λ k k' p → punchIn-inj i k k' (cong fst (inl-inj p))) y

    holes-inr : (c : Fin b) → holes (inr c) ≡ 0
    holes-inr c = count-zero decY (λ k → inl (holeIn i k)) (inr c) (λ k → inl≢inr)

    -- argument variables occur at most once in ⟦op⟧(η y⃗), and no hole
    -- variable occurs there
    X-inl : (a : Fin∖ n i) → occ X (inl a) ≡ 0
    X-inl a = occ-X (inl a) ∙ go o
      where
      Y' : WOp R → Type
      Y' o' = Fin∖ n i ⊎ Fin (Signature.ar sigR o')
      d' : (o' : WOp R) → Discrete (Y' o')
      d' o' = discrete⊎ decFin∖ discreteFin
      lr : (o' : WOp R) {c : Fin (Signature.ar sigR o')} → ¬ (_≡_ {A = Y' o'} (inr c) (inl a))
      lr o' p = true≢false (cong (⊎.rec (λ _ → true) (λ _ → false)) (sym p))
      go : (o' : WOp R) → countModel (Y' o') .alg o' (λ j → δ (d' o') (inr j)) (inl a) ≡ 0
      go unit    = refl
      go mul     = cong₂ _+_ (δ-no (d' mul) (lr mul {zero})) (δ-no (d' mul) (lr mul {suc zero}))
      go (act r) = δ-no (d' (act r)) (lr (act r) {zero})

    X-inr : (c : Fin b) → occ X (inr c) ≤ 1
    X-inr c = subst (_≤ 1) (sym (occ-X (inr c))) (go o c)
      where
      Y' : WOp R → Type
      Y' o' = Fin∖ n i ⊎ Fin (Signature.ar sigR o')
      d' : (o' : WOp R) → Discrete (Y' o')
      d' o' = discrete⊎ decFin∖ discreteFin
      out : {k : ℕ} → Fin (suc k) ⊎ Fin (suc k) → Fin (suc k)
      out = ⊎.rec (λ _ → zero) (λ z → z)
      zero≢one : ¬ (_≡_ {A = Y' mul} (inr zero) (inr (suc zero)))
      zero≢one p = fznots (cong (⊎.rec (λ _ → zero) (λ z → z)) p)
      go : (o' : WOp R) (c : Fin (Signature.ar sigR o'))
        → countModel (Y' o') .alg o' (λ j → δ (d' o') (inr j)) (inr c) ≤ 1
      go unit    ()
      go mul     zero       =
        subst (_≤ 1) (sym (cong₂ _+_ (δ-yes (d' mul) {a = inr zero} refl)
                                     (δ-no (d' mul) {a = inr (suc zero)} λ p → zero≢one (sym p)))) ≤-refl
      go mul     (suc zero) =
        subst (_≤ 1) (sym (cong₂ _+_ (δ-no (d' mul) {a = inr zero} zero≢one)
                                     (δ-yes (d' mul) {a = inr (suc zero)} refl))) ≤-refl
      go (act r) c = δ≤1 (d' (act r)) (inr zero) (inr c)

    bound : ∀ y → count decY (ρ s i o) y ≤ 1
    bound (inl a) = subst (_≤ 1) (sym (occ-step (inl a) ∙ cong (_+ holes (inl a)) (X-inl a))) (holes≤1 (inl a))
    bound (inr c) = subst (_≤ 1) (sym (occ-step (inr c) ∙ cong (occ X (inr c) +_) (holes-inr c) ∙ +-zero _)) (X-inr c)
