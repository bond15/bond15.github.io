{-
  DEVIATION D15: the monad-container definition of an operational model
  (POPL.MonadContainer.OperationalModel) is equivalent to the paper's
  definition, a free model monad whose functor is polynomial
  (POPL.OperationalModel, used by the main development).

    toPoly   : OperationalModel → Paper.OperationalModel
    fromPoly : Paper.OperationalModel → OperationalModel

  Both directions keep the polynomial and preserve η, μ and the
  operations (up to paths).  Going around fromPoly ∘ toPoly gives back the
  same e and •, and the same generic effects.

  fromPoly reads the container structure off the free model monad:
    e      = the shape of η(tt)
    s • v  = the shape of μ(G), with ↖ and ↗ the positions of μ(G),
  where G = (s, λp. (v p, λm. (p, m))) is the generic element of
  T(T(Σ p. P(v p))).  By naturality of μ, μ of any element is the image
  of μ(G), which is the container formula.
-}
module POPL.MonadContainer.Equiv where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.FinData using (Fin)

open import POPL.Theory
open import POPL.FreeModel
open import POPL.Polynomial renaming (plug to _⟪_⟫)
open import POPL.MonadContainer.Base
open import POPL.MonadContainer.OperationalModel
import POPL.OperationalModel as Paper

module _ (𝒯 : Theory) where
  open Theory 𝒯
  open Model

  ------------------------------------------------------------------------
  -- Monad container ⇒ paper's definition

  toPoly : OperationalModel 𝒯 → Paper.OperationalModel 𝒯
  toPoly OM = record
    { poly     = OperationalModel.poly OM
    ; free     = OperationalModel.free OM
    ; map-poly = OperationalModel.map-poly OM
    }

  ------------------------------------------------------------------------
  -- Paper's definition ⇒ monad container

  module FromPoly (PF : Paper.OperationalModel 𝒯) where
    open Paper.OperationalModel PF

    private
      Pos : poly .Sh → Type
      Pos s = Fin (poly .P s)

      -- the generic element of T(T(Σ p. P(v p)))
      G : (s : poly .Sh) (v : Pos s → poly .Sh) → ⟦ poly ⟧ (⟦ poly ⟧ (Σ[ q ∈ Pos s ] Pos (v q)))
      G s v = s , λ q → v q , λ m → q , m

    cops : MCOps poly
    cops = record
      { e   = η tt .fst
      ; _•_ = λ s v → μ (G s v) .fst
      ; ↖   = λ s v q → μ (G s v) .snd q .fst
      ; ↗   = λ s v q → μ (G s v) .snd q .snd
      }

    η-agree : {X : Type} (x : X) → η x ≡ ηᶜ cops x
    η-agree x = sym (map-η (λ _ → x) tt) ∙ map-poly (λ _ → x) (η tt)

    μ-agree : {X : Type} (t : ⟦ poly ⟧ (⟦ poly ⟧ X)) → μ t ≡ μᶜ cops t
    μ-agree (s , γ) =
      cong μ (sym mm) ∙ sym (μ-natural h (G s v)) ∙ map-poly h (μ (G s v))
      where
      v = λ q → γ q .fst
      h : (Σ[ q ∈ Pos s ] Pos (v q)) → _
      h (q , m) = γ q .snd m
      mm : map (map h) (G s v) ≡ (s , γ)
      mm = map-poly (map h) (G s v) ∙ cong (λ f → pmap f (G s v)) (funExt (map-poly h))

    laws : MonadLaws cops
    laws = record
      { μη    = λ t → sym (μ-agree _) ∙ cong μ (sym (η-agree t)) ∙ μ-η t
      ; μmapη = λ t →
          sym (μ-agree _)
          ∙ cong μ (cong (λ f → pmap f t) (funExt λ x → sym (η-agree x)) ∙ sym (map-poly η t))
          ∙ μ-mapη t
      ; μμ    = λ t →
          sym (μ-agree _) ∙ cong μ (sym (μ-agree t)) ∙ μ-assoc t
          ∙ cong μ (map-poly μ t ∙ cong (λ f → pmap f t) (funExt μ-agree))
          ∙ μ-agree _
      }

    mc : MonadContainer
    mc = record { poly = poly ; cops = cops ; laws = laws }

    -- ⌈op⌉ = ⟦op⟧(η y₁, …)
    gen : (o : Op) → ⟦ poly ⟧ (Fin (ar o))
    gen o = ops o η

    opsC : {X : Type} → AlgStr sig (⟦ poly ⟧ X)
    opsC o args = μᶜ cops (pmap args (gen o))

    -- algebraic operations from generic effects
    ops-agree : {X : Type} (o : Op) (args : Fin (ar o) → ⟦ poly ⟧ X) → opsC o args ≡ ops o args
    ops-agree o args =
      sym (μ-agree _)
      ∙ cong μ (sym (map-poly args (ops o η)))
      ∙ cong μ (map-hom args .pres o η)
      ∙ cong (λ a → μ (ops o a)) (funExt λ j → map-η args j)
      ∙ μ-hom .pres o (η ∘ args)
      ∙ cong (ops o) (funExt λ j → μ-η (args j))

    opsC≡ops : {X : Type} → opsC {X} ≡ ops
    opsC≡ops = funExt λ o → funExt λ args → ops-agree o args

    om : OperationalModel 𝒯
    om = record
      { mc        = mc
      ; gen       = gen
      ; opsᶜ-sat  = subst (Sat 𝒯) (sym opsC≡ops) ops-sat
      ; extᶜ      = ext
      ; extᶜ-η    = λ M f x → cong (ext M f) (sym (η-agree x)) ∙ ext-η M f x
      ; extᶜ-hom  = λ M f → ishom λ o args →
          cong (ext M f) (ops-agree o args) ∙ ext-hom M f .pres o args
      ; extᶜ-uniq = λ M f h hh hη → ext-uniq M f h
          (ishom λ o args → cong h (sym (ops-agree o args)) ∙ hh .pres o args)
          (λ x → cong h (η-agree x) ∙ hη x)
      }

  fromPoly : Paper.OperationalModel 𝒯 → OperationalModel 𝒯
  fromPoly PF = FromPoly.om PF

  ------------------------------------------------------------------------
  -- Agreement

  module _ (PF : Paper.OperationalModel 𝒯) where
    private
      module P = Paper.OperationalModel PF
      module O = OperationalModel (fromPoly PF)

    fromPoly-η : {X : Type} (x : X) → O.η x ≡ P.η x
    fromPoly-η x = sym (FromPoly.η-agree PF x)

    fromPoly-μ : {X : Type} (t : _) → O.μₘ {X} t ≡ P.μ t
    fromPoly-μ t = sym (FromPoly.μ-agree PF t)

    fromPoly-ops : {X : Type} (o : Op) (args : _) → O.ops {X} o args ≡ P.ops o args
    fromPoly-ops = FromPoly.ops-agree PF

  module _ (OM : OperationalModel 𝒯) where
    private
      module O  = OperationalModel OM
      module O' = OperationalModel (fromPoly (toPoly OM))

    -- toPoly keeps η, μ and the operations on the nose (it keeps the free
    -- model); going around recovers the container structure.
    roundtrip-e : O'.e ≡ O.e
    roundtrip-e = refl

    roundtrip-• : (s : O.poly .Sh) (v : _) → s O'.• v ≡ s O.• v
    roundtrip-• s v = cong fst (sym (O.μₘ≡μ _))

    roundtrip-gen : (o : Op) → O'.gen o ≡ O.gen o
    roundtrip-gen o = O.μmapη (O.gen o)

  -- The Effect rule's right-hand side μ((∂η C)⟪⟦op⟧(η k⃗)⟫) is the same,
  -- computed in fromPoly PF or in PF.  (Used to transfer the instances'
  -- derived rules.)
  module _ (PF : Paper.OperationalModel 𝒯) where
    private
      module P = Paper.OperationalModel PF
      module O = OperationalModel (fromPoly PF)

    effect-rhs-agree : {X : Type} (C : ∂ P.poly X) (o : Op) (k : Fin (ar o) → X)
      → O.μₘ ((∂map O.η C) ⟪ O.ops o (λ j → O.η (k j)) ⟫)
        ≡ P.μ ((∂map P.η C) ⟪ P.ops o (λ j → P.η (k j)) ⟫)
    effect-rhs-agree (s , i , γ) o k =
      fromPoly-μ PF _
      ∙ cong₂ (λ C x → P.μ (C ⟪ x ⟫))
          (ΣPathP (refl , ΣPathP (refl , funExt λ x → fromPoly-η PF (γ x))))
          (fromPoly-ops PF o _ ∙ cong (P.ops o) (funExt λ j → fromPoly-η PF (k j)))
