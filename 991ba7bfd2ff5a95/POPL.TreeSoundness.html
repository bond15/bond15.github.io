{-
  Paper §4.1, Theorem (soundness of ↦tree):
      M ↦tree M'  ⇒  ⟦M⟧ = ⟦M'⟧          for closed M, M' : Comp(B).

  The paper argues that each reduction is a provable equation of CBPV⁺
  and the denotations agree.  We prove it directly from the denotation
  lemmas; CBPV(𝒯) terms are denoted by the same ⟦-⟧ as CBPV⁺ terms (the
  syntax is shared, see POPL.Types).
-}
open import Cubical.Foundations.Prelude using (Type)
open import POPL.Theory
open import POPL.FreeModel

module POPL.TreeSoundness (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.FinData using (Fin; discreteFin)
open import Cubical.Relation.Nullary

open Theory 𝒯
open Model
open FreeModel fm

open import POPL.Fin
open import POPL.Syntax sig
open import POPL.Plug sig
open import POPL.Denotation 𝒯 fm
open import POPL.DenotationLemmas 𝒯 fm
open import POPL.TreeReduction sig

private variable
  B : CTy

sound-red : {N N' : CComp B} → N ↦red N' → ⟦ N ⟧C ρ∅ ≡ ⟦ N' ⟧C ρ∅
sound-red (bind-ret {B = B} V N) =
  ext-η ⟦ B ⟧c (λ a → ⟦ N ⟧C (ρ∅ ▹ a)) (⟦ V ⟧V ρ∅) ∙ sym (den-sub₀ ρ∅ N V)
sound-red (force-thunk M)     = refl
sound-red (case+-inl V M₁ M₂) = sym (den-sub₀ ρ∅ M₁ V)
sound-red (case+-inr V M₁ M₂) = sym (den-sub₀ ρ∅ M₂ V)
sound-red (case×-pair V W M)  = sym (den-sub₁ ρ∅ M V W)
sound-red (app-lam M V)       = sym (den-sub₀ ρ∅ M V)
sound-red (π₁-pair M N)       = refl
sound-red (π₂-pair M N)       = refl

sound-h : {M M' : CComp B} → M ↦h M' → ⟦ M ⟧C ρ∅ ≡ ⟦ M' ⟧C ρ∅
sound-h (stk S {N} {N'} r) =
  den-plug S N ρ∅ ∙ cong (⟦ S ⟧S ρ∅) (sound-red r) ∙ sym (den-plug S N' ρ∅)
sound-h (opS S p o k) = den-plug-op S o k ρ∅

-- Theorem (soundness of ↦tree)
sound↦ : {M M' : CComp B} → M ↦ M' → ⟦ M ⟧C ρ∅ ≡ ⟦ M' ⟧C ρ∅
sound↦ (head h) = sound-h h
sound↦ {B = B} (opArg o k j {N} s) =
  cong (⟦ B ⟧c .alg o) (funExt λ i → args i (discreteFin i j))
  where
  args : (i : Fin (ar o)) → Dec (i ≡ j) → ⟦ k i ⟧C ρ∅ ≡ ⟦ (k [ j ≔ N ]) i ⟧C ρ∅
  args i (yes p) = cong (λ l → ⟦ k l ⟧C ρ∅) p ∙ sound↦ s
                 ∙ cong (λ X → ⟦ X ⟧C ρ∅) (sym (upd-yes k j N p))
  args i (no ¬p) = cong (λ X → ⟦ X ⟧C ρ∅) (sym (upd-no k j N ¬p))

sound↦* : {M M' : CComp B} → M ↦* M' → ⟦ M ⟧C ρ∅ ≡ ⟦ M' ⟧C ρ∅
sound↦* done    = refl
sound↦* (s ∷ p) = sound↦ s ∙ sound↦* p
