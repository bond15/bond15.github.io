-- Sums over holes

holeIn : {m : ℕ} (i : Fin (suc m)) → Fin m → Fin∖ (suc m) i
holeIn i k = punchIn i k , punchIn-avoid i k

-- Σ over the positions other than i.
holeSum : (n : ℕ) (i : Fin n) → (Fin∖ n i → ℕ) → ℕ
holeSum zero    i g = ⊥.rec (¬Fin0 i)
holeSum (suc m) i g = sumFin m (g ∘ holeIn i)

fill-other : {n : ℕ} (i : Fin n) (γ : Fin∖ n i → X) (x : X) (j : Fin n) (ne : ¬ j ≡ i)
  → fillAt i γ x j ≡ γ (j , ne)
fill-other i γ x j ne with discreteFin j i
... | yes p   = ⊥.rec (ne p)
... | no ne'  = cong (λ q → γ (j , q)) (isProp¬ _ ne' ne)

-- Σ over C⟪x⟫ = g x + Σ over the rest of C.
plug-sum : (C : ∂ poly X) (x : X) (g : X → ℕ)
  → sumFin _ (g ∘ (C ⟪ x ⟫) .snd) ≡ g x + holeSum _ (C .snd .fst) (g ∘ C .snd .snd)
plug-sum {X = X} (s , i , γ) x g = go (poly .P s) i γ
  where
  go : (n : ℕ) (i : Fin n) (γ : Fin∖ n i → X)
     → sumFin n (g ∘ fillAt i γ x) ≡ g x + holeSum n i (g ∘ γ)
  go zero    i γ = ⊥.rec (¬Fin0 i)
  go (suc m) i γ =
    sum-punch m (g ∘ fillAt i γ x) i
    ∙ cong₂ _+_ (cong g (fill-hole i γ x))
        (sum-cong m λ k → cong g (fill-other i γ x (punchIn i k) (punchIn-avoid i k)))

-- Σ along an injective renaming ρ into the holes and the arguments.
step-sum : (n : ℕ) (i : Fin n) {a b : ℕ}
  (ρ' : Fin a → Fin∖ n i ⊎ Fin b) → isInjective ρ'
  → (γ : Fin∖ n i → X) (δ : Fin b → X) (g : X → ℕ)
  → sumFin a (g ∘ ⊎.rec γ δ ∘ ρ') ≤ holeSum n i (g ∘ γ) + sumFin b (g ∘ δ)
step-sum zero    i ρ' inj γ δ g = ⊥.rec (¬Fin0 i)
step-sum (suc m) i {a} {b} ρ' inj γ δ g =
  subst (_≤ holeSum (suc m) i (g ∘ γ) + sumFin b (g ∘ δ)) (sym lhs-eq)
    (subst (sumFin a (G ∘ dec ∘ enc ∘ ρ') ≤_) rhs-eq
      (sum-inj a (m + b) (enc ∘ ρ') inj-encρ (G ∘ dec)))
  where
  G : Fin∖ (suc m) i ⊎ Fin b → ℕ
  G = g ∘ ⊎.rec γ δ
  enc : Fin∖ (suc m) i ⊎ Fin b → Fin (m + b)
  enc (inl (j , ne)) = finL (punchOut i j ne)
  enc (inr k)        = finR {m = m} k
  inlF : Fin∖ (suc m) i → Fin∖ (suc m) i ⊎ Fin b
  inlF = inl
  dec : Fin (m + b) → Fin∖ (suc m) i ⊎ Fin b
  dec = ⊎.map (holeIn i) (λ y → y) ∘ splitFin {m = m}
  dec-enc : ∀ x → dec (enc x) ≡ x
  dec-enc (inl (j , ne)) =
    cong (⊎.map (holeIn i) (λ y → y)) (splitFin-L {b = b} (punchOut i j ne))
    ∙ cong inlF (ΣPathP (punchIn-punchOut i j ne , isProp→PathP (λ _ → isProp¬ _) _ _))
  dec-enc (inr k) = cong (⊎.map (holeIn i) (λ y → y)) (splitFin-R {m = m} k)
  inj-encρ : ∀ x y → enc (ρ' x) ≡ enc (ρ' y) → x ≡ y
  inj-encρ x y p = inj x y (sym (dec-enc (ρ' x)) ∙ cong dec p ∙ dec-enc (ρ' y))
  lhs-eq : sumFin a (g ∘ ⊎.rec γ δ ∘ ρ') ≡ sumFin a (G ∘ dec ∘ enc ∘ ρ')
  lhs-eq = sum-cong a λ x → cong G (sym (dec-enc (ρ' x)))
  rhs-eq : sumFin (m + b) (G ∘ dec) ≡ holeSum (suc m) i (g ∘ γ) + sumFin b (g ∘ δ)
  rhs-eq = sum-+ m b (G ∘ dec)
      ∙ cong₂ _+_ (sum-cong m λ k → cong (G ∘ ⊎.map (holeIn i) (λ y → y)) (splitFin-L {b = b} k))
                  (sum-cong b λ k → cong (G ∘ ⊎.map (holeIn i) (λ y → y)) (splitFin-R {m = m} k))

-- The metric on configurations (Def. "Termination Metric")

‖_‖T : Config A → ℕ
‖ t ‖T = sumFin _ (λ p → ‖ t .snd p ‖)

-- ‖op(K⃗)‖ = 1 + Σ ‖Kⱼ‖
metric-op : (o : Op) (k : Fin (ar o) → CComp (F A))
  → ‖ op o k ‖ ≡ suc (sumFin (ar o) (λ j → ‖ k j ‖))
metric-op o k = irr (related (op o k)) (op o (λ j → related (k j)))

-- Σⱼ ‖S[Mⱼ]‖ < ‖S[op(M⃗)]‖
stack-op-metric : (S : CStack B (F A)) (o : Op) (k : Fin (ar o) → CComp B)
  → sumFin (ar o) (λ j → ‖ plug S (k j) ‖) < ‖ plug S (op o k) ‖
stack-op-metric S o k with emptyView S
... | isEmpty      = subst (sumFin (ar o) (λ j → ‖ k j ‖) <_) (sym (metric-op o k)) ≤-refl
... | isNonEmpty p =
  <-trans (subst (sumFin (ar o) (λ j → ‖ plug S (k j) ‖) <_)
             (sym (metric-op o (λ j → plug S (k j)))) ≤-refl)
          (metric-decrease (head (opS S p o k)))

-- Every ↦T step decreases the metric, for an affine operational model.
decrease-T : Affine → {t t' : Config A} → t ↦T t' → ‖ t' ‖T < ‖ t ‖T
decrease-T aff (pure C@(s , i , γ) {M} {M'} st) =
  subst2 _<_ (sym (plug-sum C M' ‖_‖)) (sym (plug-sum C M ‖_‖))
    (<-+k (metric-decrease st))
decrease-T aff (effect C@(s , i , γ) S o k) =
  subst (_< ‖ C ⟪ plug S (op o k) ⟫ ‖T) (sym rhs-eq-metric)
    (≤<-trans
      (step-sum (poly .P s) i (ρ s i o) (aff s i o) γ (λ j → plug S (k j)) ‖_‖)
      (subst (holeSum _ i (‖_‖ ∘ γ) + sumFin (ar o) (λ j → ‖ plug S (k j) ‖) <_)
        (+-comm (holeSum _ i (‖_‖ ∘ γ)) ‖ plug S (op o k) ‖
          ∙ sym (plug-sum C (plug S (op o k)) ‖_‖))
        (<-k+ (stack-op-metric S o k))))
  where
  rhs-eq-path : μₘ ((∂map η C) ⟪ ops o (λ j → η (plug S (k j))) ⟫)
           ≡ pmap (⊎.rec γ (λ j → plug S (k j))) (step s i o)
  rhs-eq-path = effect-via-step s i γ S o k ∙ map-poly _ (step s i o)
  rhs-eq-metric : ‖ μₘ ((∂map η C) ⟪ ops o (λ j → η (plug S (k j))) ⟫) ‖T
             ≡ sumFin _ (‖_‖ ∘ ⊎.rec γ (λ j → plug S (k j)) ∘ ρ s i o)
  rhs-eq-metric = cong ‖_‖T rhs-eq-path

-- Theorem (Strong normalization for ↦T)

_↤T_ : Config A → Config A → Type
t' ↤T t = t ↦T t'

strong-normalization-T : Affine → (t : Config A) → Acc _↤T_ t
strong-normalization-T aff = acc-measure ‖_‖T (decrease-T aff)

-- Lemma (Progress), for configurations: a configuration is T(ret)(ṽ),
-- or it steps.
private
  StepsIn : CComp (F A) → Type
  StepsIn {A} M = (C : ∂ poly (CComp (F A))) → Σ[ t' ∈ Config A ] C ⟪ M ⟫ ↦T t'

  classify : {M : CComp (F A)} → 𝒞⟦ F A ⟧ M → (Σ[ V ∈ CVal A ] ret V ≡ M) ⊎ StepsIn M
  classify (ret {V = V} v) = inl (V , refl)
  classify (op o {k} cs)   = inr λ C → _ , effect C • o k
  classify (back n s c)    = inr λ C → _ , pure C s

  search : (n : ℕ) {P Q : Fin n → Type} → (∀ p → P p ⊎ Q p) → (∀ p → P p) ⊎ (Σ[ p ∈ Fin n ] Q p)
  search zero    d = inl λ ()
  search (suc n) d with d zero | search n (d ∘ suc)
  ... | inr q | _             = inr (zero , q)
  ... | inl p | inr (j , q)   = inr (suc j , q)
  ... | inl p | inl ps        = inl λ { zero → p ; (suc j) → ps j }

progress-T : (t : Config A)
  → (Σ[ ṽ ∈ T (CVal A) ] t ≡ pmap ret ṽ) ⊎ (Σ[ t' ∈ Config A ] t ↦T t')
progress-T (s , γ) with search (poly .P s) (λ p → classify (related (γ p)))
... | inl rets = inl ((s , λ p → rets p .fst) , ΣPathP (refl , funExt λ p → sym (rets p .snd)))
... | inr (p , steps) =
  inr (steps (unplug (s , γ) p) .fst
      , subst (_↦T steps (unplug (s , γ) p) .fst) (plug-unplug (s , γ) p)
          (steps (unplug (s , γ) p) .snd))

-- A configuration with no step is T(ret)(ṽ).
terminal-T : (t : Config A) → (∀ t' → ¬ t ↦T t')
  → Σ[ ṽ ∈ T (CVal A) ] t ≡ pmap ret ṽ
terminal-T t stuck with progress-T t
... | inl r         = r
... | inr (t' , s)  = ⊥.rec (stuck t' s)

-- Theorem (Strong normalization for ↦T), existence part:
-- every configuration reduces to some T(ret)(ṽ).
normalize-T : Affine → (t : Config A) → Σ[ ṽ ∈ T (CVal A) ] t ↦T* pmap ret ṽ
normalize-T aff t = go t (strong-normalization-T aff t)
  where
  go : (t : Config _) → Acc _↤T_ t → Σ[ ṽ ∈ T (CVal _) ] t ↦T* pmap ret ṽ
  go t (acc rec) with progress-T t
  ... | inl (ṽ , p)  = ṽ , subst (_↦T* pmap ret ṽ) (sym p) done
  ... | inr (t' , s) = let (ṽ , q) = go t' (rec t' s) in ṽ , s ∷ q

-- Canonicity is the case t = η(M).
canonicity-T : Affine → (M : CComp (F A)) → Σ[ ṽ ∈ T (CVal A) ] η M ↦T* pmap ret ṽ
canonicity-T aff M = normalize-T aff (η M)

-- Lemma (uniqueness of the final configuration; D12)

-- ⟦T(ret)(ṽ)⟧ = T(⟦_⟧v)(ṽ)
den-final : (ṽ : T (CVal A)) → ⟦ pmap ret ṽ ⟧T ≡ pmap (λ V → ⟦ V ⟧V ρ∅) ṽ
den-final ṽ =
  cong (μ ∘ map den-c) (sym (map-poly ret ṽ))
  ∙ cong μ (sym (map-∘ den-c ret ṽ))
  ∙ cong μ (map-∘ η (λ V → ⟦ V ⟧V ρ∅) ṽ)
  ∙ μ-mapη _
  ∙ map-poly _ ṽ

final-unique : (r : ⟦ A ⟧v → CVal A) → (∀ V → r (⟦ V ⟧V ρ∅) ≡ V)
  → (M : CComp (F A)) (ṽ₁ ṽ₂ : T (CVal A))
  → η M ↦T* pmap ret ṽ₁ → η M ↦T* pmap ret ṽ₂ → ṽ₁ ≡ ṽ₂
final-unique r retr M ṽ₁ ṽ₂ p₁ p₂ =
  sym (readback ṽ₁) ∙ cong (pmap r) same ∙ readback ṽ₂
  where
  same : pmap (λ V → ⟦ V ⟧V ρ∅) ṽ₁ ≡ pmap (λ V → ⟦ V ⟧V ρ∅) ṽ₂
  same = sym (den-final ṽ₁) ∙ sym (sound-T* p₁) ∙ sound-T* p₂ ∙ den-final ṽ₂
  readback : (ṽ : T (CVal _)) → pmap r (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ) ≡ ṽ
  readback (s , γ) = ΣPathP (refl , funExt λ p → retr (γ p))

-- Ground types: those whose closed values are read back from ⟦A⟧.
data Ground : VTy → Type where
  g𝟘 : Ground 𝟘
  g𝟙 : Ground 𝟙
  g⊕ : {A A' : VTy} → Ground A → Ground A' → Ground (A ⊕ A')
  g⊗ : {A A' : VTy} → Ground A → Ground A' → Ground (A ⊗ A')

reflect : Ground A → ⟦ A ⟧v → CVal A
reflect g𝟘 ()
reflect g𝟙 _ = ⋆
reflect (g⊕ g g') (inl a) = inl (reflect g a)
reflect (g⊕ g g') (inr a) = inr (reflect g' a)
reflect (g⊗ g g') (a , a') = vpair (reflect g a) (reflect g' a')

reflect-den : (g : Ground A) (V : CVal A) → reflect g (⟦ V ⟧V ρ∅) ≡ V
reflect-den g (var ())
reflect-den g (absurd W) = ⊥.rec (⟦ W ⟧V ρ∅)
reflect-den g𝟙 ⋆ = refl
reflect-den (g⊕ g g') (inl V) = cong inl (reflect-den g V)
reflect-den (g⊕ g g') (inr V) = cong inr (reflect-den g' V)
reflect-den (g⊗ g g') (vpair V W) = cong₂ vpair (reflect-den g V) (reflect-den g' W)

-- For ground A: η(M) reduces to a unique T(ret)(ṽ).
ground-canonicity : Affine → Ground A → (M : CComp (F A))
  → Σ[ ṽ ∈ T (CVal A) ] (η M ↦T* pmap ret ṽ)
      × (∀ ṽ' → η M ↦T* pmap ret ṽ' → ṽ ≡ ṽ')
ground-canonicity aff g M =
  let (ṽ , p) = canonicity-T aff M in
  ṽ , p , λ ṽ' p' → final-unique (reflect g) (reflect-den g) M ṽ ṽ' p p'
