{-
  ======================================================================
  FUTURE WORK.  This module is an example, not part of the paper.
  ======================================================================

  Combining effects with a distributive law of monadic containers
  (Purdy and Damato, "Distributive Laws of Monadic Containers",
  arXiv:2503.17191).

  A distributive law γ : T S ⇒ S T makes the composite S ∘ T a monad
  (Beck), and the composite of two containers is a container.  Purdy and
  Damato show (their Example 18 and Lemma 19) that the exceptions
  container (⊤ + E) ◁ Tr has exactly one distributive law with every
  monadic container S ◁ P, giving errors under any effect:
      γ : S X + E → S (X + E),   γ (inl s) = S(inl)(s),   γ (inr e) = η(inr e).

  Here we work this out for S = writer over a monoid (M, ⊗, e):
    • the distributive law γ and Beck's four axioms;
    • the composite monad W ∘ Ex, X ↦ M × (X + E), as a monad container;
      it is isomorphic to the composite monad given by γ, with η and μ
      preserved (φ, φ-η, φ-μ);
    • the combined theory: tell_m (unary) and raise_e (nullary), with the
      writer equations and no others.  Its free model is M × (X + E), so
      the container is an operational model (POPL.MonadContainer);
    • affinity, and the derived Effect rules, which show the combination
      at work: raising keeps the log.
          ⟨m, S[tell_n(M)]⟩ ↦T ⟨m ⊗ n, S[M]⟩
          ⟨m, S[raise_e]⟩   ↦T ⟨m, raise e⟩

  Future work: build the composite operational model from a distributive
  law in general, identify the combined theory (both signatures plus the
  equations induced by the law), and show that affinity is preserved.
-}
module POPL.FutureWork.WriterWithErrors where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin; zero; suc)

open import POPL.Theory
open import POPL.Polynomial renaming (plug to _⟪_⟫)
open import POPL.MonadContainer.Base
open import POPL.MonadContainer.OperationalModel
open import POPL.Instances.Writer using (MonoidOn; isPropFin1; fin1-const)
open import POPL.Instances.Common using (inj-from-prop)

module _ (M E : Type) (mon : MonoidOn M) where
  open MonoidOn mon renaming (e to ε)

  ----------------------------------------------------------------------
  -- The two monads, and the distributive law (Purdy–Damato, Example 18)

  W : Type → Type
  W X = M × X

  ηW : {X : Type} → X → W X
  ηW x = ε , x

  μW : {X : Type} → W (W X) → W X
  μW (m , n , x) = m ⊗ n , x

  mapW : {X Y : Type} → (X → Y) → W X → W Y
  mapW f (m , x) = m , f x

  Ex : Type → Type
  Ex X = X ⊎ E

  ηE : {X : Type} → X → Ex X
  ηE = inl

  μE : {X : Type} → Ex (Ex X) → Ex X
  μE (inl y) = y
  μE (inr e) = inr e

  mapE : {X Y : Type} → (X → Y) → Ex X → Ex Y
  mapE f (inl x) = inl (f x)
  mapE f (inr e) = inr e

  -- γ : Ex ∘ W ⇒ W ∘ Ex
  γ : {X : Type} → Ex (W X) → W (Ex X)
  γ (inl (m , x)) = m , inl x
  γ (inr e)       = ε , inr e

  -- Beck's axioms
  γ-ηE : {X : Type} (w : W X) → γ (ηE w) ≡ mapW ηE w
  γ-ηE w = refl

  γ-ηW : {X : Type} (y : Ex X) → γ (mapE ηW y) ≡ ηW y
  γ-ηW (inl x) = refl
  γ-ηW (inr e) = refl

  γ-μE : {X : Type} (t : Ex (Ex (W X))) → γ (μE t) ≡ mapW μE (γ (mapE γ t))
  γ-μE (inl (inl w)) = refl
  γ-μE (inl (inr e)) = refl
  γ-μE (inr e)       = refl

  γ-μW : {X : Type} (t : Ex (W (W X))) → γ (mapE μW t) ≡ μW (mapW γ (γ t))
  γ-μW (inl (m , n , x)) = refl
  γ-μW (inr e)           = ΣPathP (sym (unitʳ ε) , refl)

  -- The composite monad W ∘ Ex given by γ (Beck):
  --   η = η_W ∘ η_E,   μ = W(μ_E) ∘ μ_W ∘ W(γ)
  ηWE : {X : Type} → X → W (Ex X)
  ηWE x = ηW (ηE x)

  μWE : {X : Type} → W (Ex (W (Ex X))) → W (Ex X)
  μWE t = mapW μE (μW (mapW γ t))

  ----------------------------------------------------------------------
  -- The composite as a monad container

  -- running (one position) or raised e (no positions)
  Pz : Unit ⊎ E → ℕ
  Pz (inl _) = 1
  Pz (inr _) = 0

  poly : Poly
  poly = record { Sh = M × (Unit ⊎ E) ; P = λ s → Pz (s .snd) }

  T : Type → Type
  T = ⟦ poly ⟧

  private
    _•'_ : (s : poly .Sh) → (Fin (Pz (s .snd)) → poly .Sh) → poly .Sh
    (m , inl _) •' v = m ⊗ v zero .fst , v zero .snd
    (m , inr e) •' v = m , inr e

    ↖' : (s : poly .Sh) (v : Fin (Pz (s .snd)) → poly .Sh)
       → Fin (Pz ((s •' v) .snd)) → Fin (Pz (s .snd))
    ↖' (m , inl _) v q = zero

    ↗' : (s : poly .Sh) (v : Fin (Pz (s .snd)) → poly .Sh)
         (q : Fin (Pz ((s •' v) .snd))) → Fin (Pz (v (↖' s v q) .snd))
    ↗' (m , inl _) v q = q

  cops : MCOps poly
  cops = record { e = ε , inl tt ; _•_ = _•'_ ; ↖ = ↖' ; ↗ = ↗' }

  private
    noArgs : {X : Type} → Fin 0 → X
    noArgs ()

    fin0 : {X : Type} (g : Fin 0 → X) → g ≡ noArgs
    fin0 g = funExt λ ()

  laws : MonadLaws cops
  laws = record { μη = μη' ; μmapη = μmapη' ; μμ = μμ' }
    where
    μη' : {X : Type} (t : T X) → μᶜ cops (ηᶜ cops t) ≡ t
    μη' ((m , inl _) , g) = ΣPathP (ΣPathP (unitˡ m , refl) , refl)
    μη' ((m , inr e) , g) = ΣPathP (ΣPathP (unitˡ m , refl) , funExt λ ())
    μmapη' : {X : Type} (t : T X) → μᶜ cops (pmap (ηᶜ cops) t) ≡ t
    μmapη' ((m , inl _) , g) = ΣPathP (ΣPathP (unitʳ m , refl) , sym (fin1-const g))
    μmapη' ((m , inr e) , g) = ΣPathP (refl , funExt λ ())
    μμ' : {X : Type} (t : T (T (T X))) → μᶜ cops (μᶜ cops t) ≡ μᶜ cops (pmap (μᶜ cops) t)
    μμ' ((m , inr e) , g) = ΣPathP (refl , funExt λ ())
    μμ' ((m , inl _) , g) with g zero
    ... | ((n , inr e) , h) = ΣPathP (refl , funExt λ ())
    ... | ((n , inl _) , h) with h zero
    ...   | ((k , z) , x) = ΣPathP (ΣPathP (assoc m n k , refl) , refl)

  mc : MonadContainer
  mc = record { poly = poly ; cops = cops ; laws = laws }

  ----------------------------------------------------------------------
  -- The container is the composite monad W ∘ Ex given by γ

  φ : {X : Type} → T X → W (Ex X)
  φ ((m , inl _) , g) = m , inl (g zero)
  φ ((m , inr e) , g) = m , inr e

  ψ : {X : Type} → W (Ex X) → T X
  ψ (m , inl x) = (m , inl tt) , λ _ → x
  ψ (m , inr e) = (m , inr e) , noArgs

  φψ : {X : Type} (w : W (Ex X)) → φ (ψ w) ≡ w
  φψ (m , inl x) = refl
  φψ (m , inr e) = refl

  ψφ : {X : Type} (t : T X) → ψ (φ t) ≡ t
  ψφ ((m , inl _) , g) = ΣPathP (refl , sym (fin1-const g))
  ψφ ((m , inr e) , g) = ΣPathP (refl , sym (fin0 g))

  φ-η : {X : Type} (x : X) → φ (ηᶜ cops x) ≡ ηWE x
  φ-η x = refl

  -- φ ∘ μ = μ_{W∘Ex} ∘ (φ ⋆ φ)
  φ-μ : {X : Type} (t : T (T X)) → φ (μᶜ cops t) ≡ μWE (mapW (mapE φ) (φ t))
  φ-μ ((m , inr e) , g) = ΣPathP (sym (unitʳ m) , refl)
  φ-μ ((m , inl _) , g) with g zero
  ... | ((n , inl _) , h) = refl
  ... | ((n , inr e) , h) = refl

  ----------------------------------------------------------------------
  -- The combined theory: writer plus raise

  data WEOp : Type where
    tell  : M → WEOp
    raise : E → WEOp

  sigWE : Signature
  sigWE = record { Op = WEOp ; ar = λ { (tell _) → 1 ; (raise _) → 0 } }

  private
    x₀ : Term sigWE (Fin 1)
    x₀ = var zero
    tellT : M → Term sigWE (Fin 1) → Term sigWE (Fin 1)
    tellT m t = op (tell m) (λ _ → t)

  -- tell_e(t) = t ;  tell_m(tell_n(t)) = tell_{m⊗n}(t) ; nothing about raise
  theory : Theory
  theory = record
    { sig = sigWE
    ; Eqn = Unit ⊎ (M × M)
    ; eqn = λ
      { (inl _)       → record { ctx = 1 ; lhs = tellT ε x₀ ; rhs = x₀ }
      ; (inr (m , n)) → record { ctx = 1 ; lhs = tellT m (tellT n x₀) ; rhs = tellT (m ⊗ n) x₀ } }
    }

  open Model

  gen : (o : WEOp) → T (Fin (Signature.ar sigWE o))
  gen (tell n)  = (n , inl tt) , λ _ → zero
  gen (raise e) = (ε , inr e) , noArgs

  private
    opsC : {X : Type} → AlgStr sigWE (T X)
    opsC o args = μᶜ cops (pmap args (gen o))

    satC : {X : Type} → Sat theory (opsC {X})
    satC (inl _)       γ = ΣPathP (ΣPathP (unitˡ _ , refl) , refl)
    satC (inr (m , n)) γ = ΣPathP (ΣPathP (sym (assoc m n _) , refl) , refl)

    module _ (N : Model theory) where
      Ntell : M → N .Carrier → N .Carrier
      Ntell m x = N .alg (tell m) (λ _ → x)
      Nraise : E → N .Carrier
      Nraise e = N .alg (raise e) noArgs
      tell-ε : ∀ x → Ntell ε x ≡ x
      tell-ε x = N .sat (inl tt) (λ _ → x)
      tell-tell : ∀ m n x → Ntell m (Ntell n x) ≡ Ntell (m ⊗ n) x
      tell-tell m n x = N .sat (inr (m , n)) (λ _ → x)

    extC : {X : Type} (N : Model theory) → (X → N .Carrier) → T X → N .Carrier
    extC N f ((m , inl _) , g) = Ntell N m (f (g zero))
    extC N f ((m , inr e) , g) = Ntell N m (Nraise N e)

    hom-tell : {X : Type} (N : Model theory) (f : X → N .Carrier) (n : M) (a : T X)
      → extC N f (opsC (tell n) (λ _ → a)) ≡ Ntell N n (extC N f a)
    hom-tell N f n ((m , inl _) , g) = sym (tell-tell N n m _)
    hom-tell N f n ((m , inr e) , g) = sym (tell-tell N n m _)

    homC : {X : Type} (N : Model theory) (f : X → N .Carrier)
      → IsHom sigWE opsC (N .alg) (extC N f)
    homC N f .pres (tell n) args =
      hom-tell N f n (args zero) ∙ cong (N .alg (tell n)) (sym (fin1-const (λ j → extC N f (args j))))
    homC N f .pres (raise e) args =
      tell-ε N (Nraise N e) ∙ cong (N .alg (raise e)) (sym (fin0 _))

    uniqC : {X : Type} (N : Model theory) (f : X → N .Carrier) (h : T X → N .Carrier)
      → IsHom sigWE opsC (N .alg) h → (∀ x → h (ηᶜ cops x) ≡ f x) → ∀ t → h t ≡ extC N f t
    uniqC N f h hh hη ((m , inl _) , g) =
      cong h (ΣPathP (ΣPathP (sym (unitʳ m) , refl) , fin1-const g))
      ∙ hh .pres (tell m) (λ _ → ηᶜ cops (g zero))
      ∙ cong (λ x → Ntell N m x) (hη (g zero))
    uniqC N f h hh hη ((m , inr e) , g) =
      cong h (ΣPathP (ΣPathP (sym (unitʳ m) , refl) , funExt λ ()))
      ∙ hh .pres (tell m) (λ _ → opsC (raise e) noArgs)
      ∙ cong (λ x → Ntell N m x) (hh .pres (raise e) noArgs ∙ cong (N .alg (raise e)) (fin0 _))

  -- The operational model (as a monad container) of the combined theory.
  OM : OperationalModel theory
  OM = record
    { mc        = mc
    ; gen       = gen
    ; opsᶜ-sat  = satC
    ; extᶜ      = extC
    ; extᶜ-η    = λ N f x → tell-ε N (f x)
    ; extᶜ-hom  = homC
    ; extᶜ-uniq = uniqC
    }

  open import POPL.MonadContainer.ConfigReduction theory OM using (Affine; ρ; _∘[_]_)

  -- Affine: every shape has at most one position.
  affine : Affine
  affine s i o = inj-from-prop (props (s ∘[ i ] o)) (ρ s i o)
    where
    props : (s : poly .Sh) → isProp (Fin (poly .P s))
    props (m , inl _) = isPropFin1
    props (m , inr _) = λ ()

  ----------------------------------------------------------------------
  -- The derived Effect rules (right-hand sides as in
  -- POPL.MonadContainer.ConfigReduction, for C = ⟨m, •⟩)

  private module O = OperationalModel OM

  -- ⟨m, S[tell_n(M)]⟩ ↦T ⟨m ⊗ n, S[M]⟩
  rule-tell : {X : Type} (m n : M) (γ' : Fin∖ 1 zero → X) (k : Fin 1 → X)
    → O.μₘ ((∂map O.η ((m , inl tt) , zero , γ')) ⟪ O.ops (tell n) (λ j → O.η (k j)) ⟫)
      ≡ ((m ⊗ n , inl tt) , λ _ → k zero)
  rule-tell m n γ' k = ΣPathP (ΣPathP (cong (m ⊗_) (unitʳ n) , refl) , refl)

  -- ⟨m, S[raise_e]⟩ ↦T ⟨m, raise e⟩: the log m is kept.
  rule-raise : {X : Type} (m : M) (e : E) (γ' : Fin∖ 1 zero → X) (k : Fin 0 → X)
    → O.μₘ ((∂map O.η ((m , inl tt) , zero , γ')) ⟪ O.ops (raise e) (λ j → O.η (k j)) ⟫)
      ≡ ((m , inr e) , noArgs)
  rule-raise m e γ' k = ΣPathP (ΣPathP (unitʳ m , refl) , funExt λ ())
