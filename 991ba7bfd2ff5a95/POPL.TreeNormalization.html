{-
  Paper §4.3 corollaries and §5.4 Def. "Termination Metric" for ↦tree.

    termination : ∀ M : Comp(F A). ∃ t : Term_Σ(Val(A)). M ↦* reify(t)
    SN          : every ↦tree reduction sequence from M : Comp(F A) is finite
    |_|         : the termination metric on Comp(F A), strictly decreased by ↦tree

  The metric (§5.4) is defined on derivations of 𝒞⟦F A⟧ and shown not to
  depend on the derivation.  This uses determinism of steps from
  non-operations (POPL.TreeReduction.det).

  CORRECTION C7: Def. "Termination Metric" sets |M| = max_{M ↦ N} |N|
  for M ≠ op(N⃗).  With that definition a pure step does not decrease the
  metric, but the proof of Thm "Strong normalization for ↦T" says it does.
  We set |M| = 1 + |N| for the unique N with M ↦ N.  The max is over
  one element, by determinism.

  DEVIATION D9: the paper proves strong normalization of ↦tree (Cor. in
  §4.3) by induction on 𝒞⟦F A⟧ and a classical argument on infinite
  sequences.  We state SN constructively, as accessibility (Acc) of M for
  the converse of ↦tree, and derive it from the decreasing metric.
-}

------------------------------------------------------------------------
-- Multi-step reduction under operations

-- Unambiguous names for the constructors (cong needs a typed function).
opC : (o : Op) → (Fin (ar o) → CComp B) → CComp B
opC o k = op o k

sucℕ : ℕ → ℕ
sucℕ = suc

↦*-≡ : {M M' N N' : CComp B} → M ≡ M' → N ≡ N' → M ↦* N → M' ↦* N'
↦*-≡ p q = subst2 _↦*_ p q

steps-at : (o : Op) (f : Fin (ar o) → CComp B) (j : Fin (ar o)) {M N : CComp B}
  → f j ≡ M → M ↦* N → op o f ↦* op o (f [ j ≔ N ])
steps-at o f j e done =
  ↦*-≡ refl (cong (opC o) (sym (upd-same f j) ∙ cong (f [ j ≔_]) e)) done
steps-at o f j {N = N} e (_∷_ {M = M₁} s p) =
  opArg o f j (subst (_↦ M₁) (sym e) s)
  ∷ ↦*-≡ refl (cong (opC o) (upd-upd f j M₁ N))
      (steps-at o (f [ j ≔ M₁ ]) j (upd-yes f j M₁ refl) p)

-- A context with n argument slots, along which argument reductions lift.
record ArgCtx (n : ℕ) (B B' : CTy) : Type where
  field
    C     : (Fin n → CComp B) → CComp B'
    lift* : (f : Fin n → CComp B) (j : Fin n) {N : CComp B}
          → f j ↦* N → C f ↦* C (f [ j ≔ N ])

open ArgCtx

-- Reduce all arguments, one position at a time.
parallel : (n : ℕ) (K : ArgCtx n B B') (k k' : Fin n → CComp B)
  → (∀ j → k j ↦* k' j) → K .C k ↦* K .C k'
parallel zero K k k' ps =
  ↦*-≡ refl (cong (K .C) (funExt λ ())) done
parallel (suc n) K k k' ps =
  K .lift* k zero (ps zero)
  ++* ↦*-≡ (cong (K .C) (sym (upd-zero k (k' zero)))) (cong (K .C) (cons-η k'))
        (parallel n K' (k ∘ suc) (k' ∘ suc) (ps ∘ suc))
  where
  K' : ArgCtx n _ _
  K' .C g = K .C (cons (k' zero) g)
  K' .lift* g j p =
    ↦*-≡ refl (cong (K .C) (upd-cons-suc (k' zero) g j _)) (K .lift* (cons (k' zero) g) (suc j) p)

op-cong* : (o : Op) (k k' : Fin (ar o) → CComp B) → (∀ j → k j ↦* k' j) → op o k ↦* op o k'
op-cong* {B = B} o = parallel (ar o) K
  where
  K : ArgCtx (ar o) B B
  K .C           = op o
  K .lift* f j p = steps-at o f j refl p

------------------------------------------------------------------------
-- Termination (Corollary "Termination", §4.3)

reify : Term Sg (CVal A) → CComp (F A)
reify (var V)   = ret V
reify (op o ts) = op o (λ j → reify (ts j))

wn : {M : CComp (F A)} → 𝒞⟦ F A ⟧ M → Σ[ t ∈ Term Sg (CVal A) ] M ↦* reify t
wn (ret {V = V} v) = var V , done
wn (op o {k} cs)   =
  op o (λ j → wn (cs j) .fst) , op-cong* o k _ (λ j → wn (cs j) .snd)
wn (back n s c)    = wn c .fst , s ∷ wn c .snd

termination : (M : CComp (F A)) → Σ[ t ∈ Term Sg (CVal A) ] M ↦* reify t
termination M = wn (related M)

------------------------------------------------------------------------
-- The termination metric (Def. "Termination Metric", §5.4; C7)

size : {P : CVal A → Type} {M : CComp (F A)} → FPred P M → ℕ
size (ret _)      = 0
size (op o cs)    = suc (sumFin _ (λ j → size (cs j)))
size (back _ _ c) = suc (size c)

-- The size does not depend on the derivation.
irr  : {P : CVal A → Type} {M : CComp (F A)} (d₁ d₂ : FPred P M) → size d₁ ≡ size d₂
irr≡ : {P : CVal A → Type} {M₁ M₂ : CComp (F A)} → M₁ ≡ M₂
     → (d₁ : FPred P M₁) (d₂ : FPred P M₂) → size d₁ ≡ size d₂

irr (ret _)       (ret _)       = refl
irr (ret _)       (back _ s _)  = ⊥.rec (ret-no-step s)
irr (op o cs)     (op o cs')    = cong sucℕ (sum-cong _ λ j → irr (cs j) (cs' j))
irr (op o cs)     (back () _ _)
irr (back _ s _)  (ret _)       = ⊥.rec (ret-no-step s)
irr (back () _ _) (op _ _)
irr (back n s c)  (back n' s' c') = cong sucℕ (irr≡ (det n s s') c c')

irr≡ {P = P} {M₁ = M₁} p d₁ d₂ =
  J (λ M₂ _ → (d₂ : FPred P M₂) → size d₁ ≡ size d₂) (irr d₁) p d₂

-- Every step strictly decreases the size.
decrease : {P : CVal A → Type} {M M' : CComp (F A)}
  → (d : FPred P M) (d' : FPred P M') → M ↦ M' → size d' < size d
decrease (ret _) d' s = ⊥.rec (ret-no-step s)
decrease (op o cs) d' (head h) = ⊥.rec (op-no-hstep h)
decrease {P = P} (op o {k} cs) (op o cs') (opArg o k j {N} s') =
  suc-≤-suc (sum-< _ (λ i → size (cs' i)) (λ i → size (cs i)) j
    (λ i ne → irr≡ (upd-no k j N ne) (cs' i) (cs i))
    (subst (_< size (cs j)) (sym (irr≡ (upd-yes k j N refl) (cs' j) dN))
      (decrease (cs j) dN s')))
  where
  dN : FPred P N
  dN = subst (FPred P) (upd-yes k j N refl) (cs' j)
decrease (op o cs) (back () _ _) (opArg o k j s')
decrease (back n s₀ c) d' s =
  subst (_< suc (size c)) (sym (irr≡ (det n s s₀) d' c)) ≤-refl

-- |M|
‖_‖ : CComp (F A) → ℕ
‖ M ‖ = size (related M)

metric-decrease : {M M' : CComp (F A)} → M ↦ M' → ‖ M' ‖ < ‖ M ‖
metric-decrease {M = M} {M'} s = decrease (related M) (related M') s

------------------------------------------------------------------------
-- Strong normalization of ↦tree (Corollary, §4.3; D9)

acc-measure : {X : Type} {R : X → X → Type} (m : X → ℕ)
  → (∀ {x y} → R y x → m y < m x) → ∀ x → Acc R x
acc-measure {R = R} m dec x = go x (<-wellfounded (m x))
  where
  go : ∀ x → Acc _<_ (m x) → Acc R x
  go x (acc rec) = acc λ y r → go y (rec (m y) (dec r))

-- Reduction, converse: R M' M  iff  M ↦ M'.
_↤_ : CComp B → CComp B → Type
M' ↤ M = M ↦ M'

strong-normalization : (M : CComp (F A)) → Acc _↤_ M
strong-normalization = acc-measure ‖_‖ metric-decrease
