{-
  Paper §4.1, Fig. 5: tree reduction ↦tree on closed CBPV(𝒯) computations.

    primitive redexes  M ↦red M'
    S[M] ↦ S[M']                    if M ↦red M'
    S[op(M⃗)] ↦ op(S[M⃗])            if S ≠ •
    op(…,Mᵢ,…) ↦ op(…,Mᵢ',…)        if Mᵢ ↦ Mᵢ'

  We name the first two rules (the ones not under an operation) head
  steps, ↦h.

  DEVIATION D8: Fig. 5 puts both the redexes and the rule S[M] ↦ S[M']
  for S ≠ • in ↦tree.  We have the single rule S[M] ↦ S[M'] for every S,
  including •, which gives the same relation without overlap.

  CORRECTION C4: Fig. 5 lists the case× redex twice and has no rule for
  π₁ ⟨M , N⟩.  We include π₁⟨M,N⟩ ↦ M.

  We also prove determinism of head steps, which the paper uses without
  proof (§4 "only one reduction applies", §5 Def. "Termination Metric").
  Head reduction is computed by a function `hs`, and every head step
  agrees with it.
-}

private variable
  A A' : VTy
  B B' B'' B₁ B₂ : CTy

-- Val(A), Comp(B), closed stacks: closed CBPV(𝒯) terms.
CVal : VTy → Type
CVal A = Val cbpv ε A

CComp : CTy → Type
CComp B = Comp cbpv ε B

CStack : CTy → CTy → Type
CStack B B' = Stack cbpv ε B B'

------------------------------------------------------------------------
-- The relations

infix 4 _↦red_ _↦h_ _↦_ _↦*_

data _↦red_ : CComp B → CComp B → Type where
  bind-ret    : (V : CVal A) (N : Comp cbpv (ε ▸ A) B) → bind (ret V) N ↦red N [ V ]₀
  force-thunk : (M : CComp B) → force (thunk M) ↦red M
  case+-inl   : (V : CVal A) (M₁ : Comp cbpv (ε ▸ A) B) (M₂ : Comp cbpv (ε ▸ A') B)
              → case+ (inl V) M₁ M₂ ↦red M₁ [ V ]₀
  case+-inr   : (V : CVal A') (M₁ : Comp cbpv (ε ▸ A) B) (M₂ : Comp cbpv (ε ▸ A') B)
              → case+ (inr V) M₁ M₂ ↦red M₂ [ V ]₀
  case×-pair  : (V : CVal A) (W : CVal A') (M : Comp cbpv (ε ▸ A ▸ A') B)
              → case× (vpair V W) M ↦red M [ V , W ]₁
  app-lam     : (M : Comp cbpv (ε ▸ A) B) (V : CVal A) → app (lam M) V ↦red M [ V ]₀
  π₁-pair     : (M : CComp B₁) (N : CComp B₂) → π₁ (cpair M N) ↦red M
  π₂-pair     : (M : CComp B₁) (N : CComp B₂) → π₂ (cpair M N) ↦red N

-- S ≠ •
NonEmptyS : CStack B B' → Type
NonEmptyS • = ⊥
NonEmptyS _ = Unit

-- M ≠ op(N⃗)
NotOp : CComp B → Type
NotOp (op o k) = ⊥
NotOp _        = Unit

data _↦h_ : CComp B → CComp B → Type where
  stk : (S : CStack B' B) {N N' : CComp B'} → N ↦red N' → plug S N ↦h plug S N'
  opS : (S : CStack B' B) → NonEmptyS S → (o : Op) (k : Fin (ar o) → CComp B')
      → plug S (op o k) ↦h op o (λ j → plug S (k j))

data _↦_ : CComp B → CComp B → Type where
  head  : {M M' : CComp B} → M ↦h M' → M ↦ M'
  opArg : (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) {N : CComp B}
        → k j ↦ N → op o k ↦ op o (k [ j ≔ N ])

data _↦*_ : CComp B → CComp B → Type where
  done : {M : CComp B} → M ↦* M
  _∷_  : {L M N : CComp B} → L ↦ M → M ↦* N → L ↦* N

infixr 5 _∷_ _++*_

_++*_ : {L M N : CComp B} → L ↦* M → M ↦* N → L ↦* N
done     ++* q = q
(s ∷ p)  ++* q = s ∷ (p ++* q)

------------------------------------------------------------------------
-- Head reduction as a function

-- The result of a head step: a plain contractum, or op(k) produced by
-- the operation rule.
data HStep (B : CTy) : Type where
  none : HStep B
  red  : CComp B → HStep B
  opc  : (o : Op) → (Fin (ar o) → CComp B) → HStep B

mapH : (CComp B → CComp B') → HStep B → HStep B'
mapH f none      = none
mapH f (red M)   = red (f M)
mapH f (opc o k) = opc o (f ∘ k)

toComp : HStep B → Maybe (CComp B)
toComp none      = nothing
toComp (red M)   = just M
toComp (opc o k) = just (op o k)

hs : CComp B → HStep B
hs (bind (ret V) N)         = red (N [ V ]₀)
hs (bind (op o k) N)        = opc o (λ j → bind (k j) N)
hs (bind M N)               = mapH (λ X → bind X N) (hs M)
hs (force (thunk M))        = red M
hs (case+ (inl V) M₁ M₂)    = red (M₁ [ V ]₀)
hs (case+ (inr V) M₁ M₂)    = red (M₂ [ V ]₀)
hs (case× (vpair V W) M)    = red (M [ V , W ]₁)
hs (app (lam M) V)          = red (M [ V ]₀)
hs (app (op o k) V)         = opc o (λ j → app (k j) V)
hs (app M V)                = mapH (λ X → app X V) (hs M)
hs (π₁ (cpair M N))         = red M
hs (π₁ (op o k))            = opc o (λ j → π₁ (k j))
hs (π₁ M)                   = mapH π₁ (hs M)
hs (π₂ (cpair M N))         = red N
hs (π₂ (op o k))            = opc o (λ j → π₂ (k j))
hs (π₂ M)                   = mapH π₂ (hs M)
hs _                        = none

-- Computations headed by an elimination frame.
Frame : CComp B → Type
Frame (bind _ _)   = Unit
Frame (app _ _)    = Unit
Frame (π₁ _)       = Unit
Frame (π₂ _)       = Unit
Frame (force _)    = Unit
Frame (case+ _ _ _) = Unit
Frame (case× _ _)  = Unit
Frame _            = ⊥

-- On frame-headed computations hs commutes with the next frame out.
hs-bind : (M : CComp (F A)) (N : Comp cbpv (ε ▸ A) B) → Frame M
  → hs (bind M N) ≡ mapH (λ X → bind X N) (hs M)
hs-bind (bind _ _)    N f = refl
hs-bind (app _ _)     N f = refl
hs-bind (π₁ _)        N f = refl
hs-bind (π₂ _)        N f = refl
hs-bind (force _)     N f = refl
hs-bind (case+ _ _ _) N f = refl
hs-bind (case× _ _)   N f = refl

hs-app : (M : CComp (A ⇒ B)) (V : CVal A) → Frame M
  → hs (app M V) ≡ mapH (λ X → app X V) (hs M)
hs-app (bind _ _)    V f = refl
hs-app (app _ _)     V f = refl
hs-app (π₁ _)        V f = refl
hs-app (π₂ _)        V f = refl
hs-app (force _)     V f = refl
hs-app (case+ _ _ _) V f = refl
hs-app (case× _ _)   V f = refl

hs-π₁ : (M : CComp (B₁ & B₂)) → Frame M → hs (π₁ M) ≡ mapH π₁ (hs M)
hs-π₁ (bind _ _)    f = refl
hs-π₁ (app _ _)     f = refl
hs-π₁ (π₁ _)        f = refl
hs-π₁ (π₂ _)        f = refl
hs-π₁ (force _)     f = refl
hs-π₁ (case+ _ _ _) f = refl
hs-π₁ (case× _ _)   f = refl

hs-π₂ : (M : CComp (B₁ & B₂)) → Frame M → hs (π₂ M) ≡ mapH π₂ (hs M)
hs-π₂ (bind _ _)    f = refl
hs-π₂ (app _ _)     f = refl
hs-π₂ (π₁ _)        f = refl
hs-π₂ (π₂ _)        f = refl
hs-π₂ (force _)     f = refl
hs-π₂ (case+ _ _ _) f = refl
hs-π₂ (case× _ _)   f = refl

redex-frame : {N N' : CComp B} → N ↦red N' → Frame N
redex-frame (bind-ret _ _)      = tt
redex-frame (force-thunk _)     = tt
redex-frame (case+-inl _ _ _)   = tt
redex-frame (case+-inr _ _ _)   = tt
redex-frame (case×-pair _ _ _)  = tt
redex-frame (app-lam _ _)       = tt
redex-frame (π₁-pair _ _)       = tt
redex-frame (π₂-pair _ _)       = tt

isNonEmpty-frame : (S : CStack B' B) (N : CComp B') → NonEmptyS S → Frame (plug S N)
isNonEmpty-frame (appˢ S V)  N _ = tt
isNonEmpty-frame (π₁ˢ S)     N _ = tt
isNonEmpty-frame (π₂ˢ S)     N _ = tt
isNonEmpty-frame (bindˢ S M) N _ = tt

-- Whether a stack is isEmpty.
data EmptyView : CStack B B' → Type where
  isEmpty    : EmptyView {B = B} •
  isNonEmpty : {S : CStack B B'} → NonEmptyS S → EmptyView S

emptyView : (S : CStack B B') → EmptyView S
emptyView •           = isEmpty
emptyView (appˢ S V)  = isNonEmpty tt
emptyView (π₁ˢ S)     = isNonEmpty tt
emptyView (π₂ˢ S)     = isNonEmpty tt
emptyView (bindˢ S N) = isNonEmpty tt

plug-frame : (S : CStack B' B) {N : CComp B'} → Frame N → Frame (plug S N)
plug-frame S {N} f with emptyView S
... | isEmpty      = f
... | isNonEmpty p = isNonEmpty-frame S N p

hs-red : {N N' : CComp B} → N ↦red N' → hs N ≡ red N'
hs-red (bind-ret V N)      = refl
hs-red (force-thunk M)     = refl
hs-red (case+-inl V M₁ M₂) = refl
hs-red (case+-inr V M₁ M₂) = refl
hs-red (case×-pair V W M)  = refl
hs-red (app-lam M V)       = refl
hs-red (π₁-pair M N)       = refl
hs-red (π₂-pair M N)       = refl

hs-stk : (S : CStack B' B) {N N' : CComp B'} → N ↦red N'
  → hs (plug S N) ≡ red (plug S N')
hs-stk • r = hs-red r
hs-stk (appˢ S V) {N} r =
  hs-app (plug S N) V (plug-frame S (redex-frame r)) ∙ cong (mapH (λ X → app X V)) (hs-stk S r)
hs-stk (π₁ˢ S) {N} r =
  hs-π₁ (plug S N) (plug-frame S (redex-frame r)) ∙ cong (mapH π₁) (hs-stk S r)
hs-stk (π₂ˢ S) {N} r =
  hs-π₂ (plug S N) (plug-frame S (redex-frame r)) ∙ cong (mapH π₂) (hs-stk S r)
hs-stk (bindˢ S M) {N} r =
  hs-bind (plug S N) M (plug-frame S (redex-frame r)) ∙ cong (mapH (λ X → bind X M)) (hs-stk S r)

hs-opS : (S : CStack B' B) → NonEmptyS S → (o : Op) (k : Fin (ar o) → CComp B')
  → hs (plug S (op o k)) ≡ opc o (λ j → plug S (k j))
hs-opS (appˢ S V) _ o k with emptyView S
... | isEmpty      = refl
... | isNonEmpty p = hs-app (plug S (op o k)) V (isNonEmpty-frame S _ p)
                 ∙ cong (mapH (λ X → app X V)) (hs-opS S p o k)
hs-opS (π₁ˢ S) _ o k with emptyView S
... | isEmpty      = refl
... | isNonEmpty p = hs-π₁ (plug S (op o k)) (isNonEmpty-frame S _ p)
                 ∙ cong (mapH π₁) (hs-opS S p o k)
hs-opS (π₂ˢ S) _ o k with emptyView S
... | isEmpty      = refl
... | isNonEmpty p = hs-π₂ (plug S (op o k)) (isNonEmpty-frame S _ p)
                 ∙ cong (mapH π₂) (hs-opS S p o k)
hs-opS (bindˢ S N) _ o k with emptyView S
... | isEmpty      = refl
... | isNonEmpty p = hs-bind (plug S (op o k)) N (isNonEmpty-frame S _ p)
                 ∙ cong (mapH (λ X → bind X N)) (hs-opS S p o k)

-- Every head step is the one computed by hs.
hs-sound : {M M' : CComp B} → M ↦h M' → toComp (hs M) ≡ just M'
hs-sound (stk S r)      = cong toComp (hs-stk S r)
hs-sound (opS S p o k)  = cong toComp (hs-opS S p o k)

just-inj : {X : Type} {x y : X} → just x ≡ just y → x ≡ y
just-inj {x = x} p = cong (Maybe.rec x (λ z → z)) p

-- Determinism of head reduction.
hdet : {M M₁ M₂ : CComp B} → M ↦h M₁ → M ↦h M₂ → M₁ ≡ M₂
hdet h₁ h₂ = just-inj (sym (hs-sound h₁) ∙ hs-sound h₂)

nothing≢just : {X : Type} {x : X} → ¬ (nothing ≡ just x)
nothing≢just p = subst (Maybe.rec Unit (λ _ → ⊥)) p tt

-- ret V and op(k) take no head step.
ret-no-hstep : {V : CVal A} {M' : CComp (F A)} → ¬ (ret V ↦h M')
ret-no-hstep h = nothing≢just (hs-sound h)

op-no-hstep : {o : Op} {k : Fin (ar o) → CComp B} {M' : CComp B} → ¬ (op o k ↦h M')
op-no-hstep h = nothing≢just (hs-sound h)

ret-no-step : {V : CVal A} {M' : CComp (F A)} → ¬ (ret V ↦ M')
ret-no-step (head h) = ret-no-hstep h

-- A head step's source is not an operation.
hstep-notOp : {M M' : CComp B} → M ↦h M' → NotOp M
hstep-notOp {M = op o k} h = op-no-hstep h
hstep-notOp {M = case+ _ _ _} h = tt
hstep-notOp {M = case× _ _}   h = tt
hstep-notOp {M = force _}     h = tt
hstep-notOp {M = triv}        h = tt
hstep-notOp {M = lam _}       h = tt
hstep-notOp {M = app _ _}     h = tt
hstep-notOp {M = cpair _ _}   h = tt
hstep-notOp {M = π₁ _}        h = tt
hstep-notOp {M = π₂ _}        h = tt
hstep-notOp {M = ret _}       h = tt
hstep-notOp {M = bind _ _}    h = tt

-- On non-operations every step is a head step.
notOp-head : {M M' : CComp B} → NotOp M → M ↦ M' → M ↦h M'
notOp-head n (head h) = h

-- Determinism of ↦tree on non-operations ("only one reduction applies").
det : {M M₁ M₂ : CComp B} → NotOp M → M ↦ M₁ → M ↦ M₂ → M₁ ≡ M₂
det n s₁ s₂ = hdet (notOp-head n s₁) (notOp-head n s₂)

------------------------------------------------------------------------
-- Head steps under a stack.
--
-- ↦tree is not a congruence for stacks: S[op(K⃗)] ↦ op(S[K⃗]) moves the
-- whole stack in one step, so S'[S[op K⃗]] and S'[op(S[K⃗])] are not one
-- step apart.  They have a common head reduct instead.  (This is the
-- gap in the paper's anti-reduction lemma; see CORRECTION C5 in
-- POPL.OpLogicalRelation.)

∘ˢ-nonempty : (S' : CStack B' B'') (S : CStack B B') → NonEmptyS S → NonEmptyS (S' ∘ˢ S)
∘ˢ-nonempty •           S p = p
∘ˢ-nonempty (appˢ S' V)  S p = tt
∘ˢ-nonempty (π₁ˢ S')     S p = tt
∘ˢ-nonempty (π₂ˢ S')     S p = tt
∘ˢ-nonempty (bindˢ S' N) S p = tt

Join : CComp B → CComp B → Type
Join M M' = (M ↦h M') ⊎ (Σ[ X ∈ CComp _ ] (M ↦h X) × (M' ↦h X))

hjoin : (S' : CStack B B') {M M' : CComp B} → M ↦h M' → Join (plug S' M) (plug S' M')
hjoin S' (stk S {N} {N'} r) =
  inl (subst2 _↦h_ (plug-∘ S' S N) (plug-∘ S' S N') (stk (S' ∘ˢ S) r))
hjoin S' (opS S p o k) with emptyView S'
... | isEmpty      = inl (opS S p o k)
... | isNonEmpty q =
  inr ( op o (λ j → plug S' (plug S (k j)))
      , subst2 _↦h_ (plug-∘ S' S (op o k)) (cong (Comp.op {Γ = ε} o) (funExt λ j → plug-∘ S' S (k j)))
          (opS (S' ∘ˢ S) (∘ˢ-nonempty S' S p) o k)
      , opS S' q o (λ j → plug S (k j)) )
