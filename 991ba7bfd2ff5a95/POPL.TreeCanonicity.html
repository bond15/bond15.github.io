{-
  Paper §4.3, last corollary (canonicity of tree reduction).

  CORRECTION C8: the paper states "for any type A such that ⟦·⟧_A is a
  bijection, and any M : Comp(F A), M ↦* reify(⟦M⟧)".  That does not
  type-check: ⟦M⟧ lives in the free model T⟦A⟧, while reify takes a term
  of Term_Σ(Val(A)).  When the theory has equations, T⟦A⟧ is not
  Term_Σ(⟦A⟧), so there is no reify(⟦M⟧).  What termination + soundness
  give, for every A (the bijection is not needed), is:

      ∀ M : Comp(F A). ∃ t. M ↦* reify(t)  and  ⌊t⌋ = ⟦M⟧ in T⟦A⟧,

  where ⌊t⌋ interprets t in T⟦A⟧ with leaves η(⟦V⟧): reduction reaches a
  syntactic tree whose denotation is ⟦M⟧.
-}
open import Cubical.Foundations.Prelude using (Type)
open import POPL.Theory
open import POPL.FreeModel

module POPL.TreeCanonicity (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Sigma

open Theory 𝒯
open FreeModel fm

open import POPL.Syntax sig
open import POPL.Denotation 𝒯 fm
open import POPL.TreeReduction sig
open import POPL.TreeSoundness 𝒯 fm
open import POPL.TreeNormalization sig

private variable
  A : VTy

-- ⌊t⌋ ∈ T⟦A⟧
⌊_⌋ : Term sig (CVal A) → T ⟦ A ⟧v
⌊ t ⌋ = eval sig ops (λ V → η (⟦ V ⟧V ρ∅)) t

den-reify : (t : Term sig (CVal A)) → ⟦ reify t ⟧C ρ∅ ≡ ⌊ t ⌋
den-reify (var V)   = refl
den-reify (op o ts) = cong (ops o) (funExt λ j → den-reify (ts j))

tree-canonicity : (M : CComp (F A))
  → Σ[ t ∈ Term sig (CVal A) ] (M ↦* reify t) × (⌊ t ⌋ ≡ ⟦ M ⟧C ρ∅)
tree-canonicity M =
  let (t , p) = termination M in
  t , p , sym (den-reify t) ∙ sym (sound↦* p)
