{-
  Paper §5.3, Example (errors): one nullary operation err and no
  equations.  Free model T(X) = X + 1, as a polynomial with two shapes:
  `run` with one position (the running computation) and `raised` with
  none (the error).

  Derived rule (§5.3):  S[err()] ↦T err.
-}
module POPL.Instances.Errors where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Bool using (Bool; true; false)
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin; zero; suc; ¬Fin0)

open import POPL.Theory
open import POPL.FreeModel
open import POPL.Polynomial renaming (plug to _⟪_⟫)
open import POPL.OperationalModel
open import POPL.Instances.Common
open import POPL.Instances.Writer using (isPropFin1; fin1-const)

sigE : Signature
sigE = record { Op = Unit ; ar = λ _ → 0 }

errTheory : Theory
errTheory = record { sig = sigE ; Eqn = ⊥ ; eqn = λ () }

open Model

-- run = true (one position), raised = false (no positions)
errPoly : Poly
errPoly = record { Sh = Bool ; P = λ { true → 1 ; false → 0 } }

private
  T : Type → Type
  T = ⟦ errPoly ⟧

  noArgs : {X : Type} → Fin 0 → X
  noArgs ()

  fin0-ext : {X : Type} (g : Fin 0 → X) → g ≡ noArgs
  fin0-ext g = funExt λ ()

  ηE : {X : Type} → X → T X
  ηE x = true , λ _ → x

  -- err
  opsE : {X : Type} → AlgStr sigE (T X)
  opsE _ _ = false , noArgs

  extE : {X : Type} (N : Model errTheory) → (X → N .Carrier) → T X → N .Carrier
  extE N f (true  , γ) = f (γ zero)
  extE N f (false , γ) = N .alg tt noArgs

  uniqE : {X : Type} (N : Model errTheory) (f : X → N .Carrier) (h : T X → N .Carrier)
    → IsHom sigE opsE (N .alg) h → (∀ x → h (ηE x) ≡ f x) → ∀ t → h t ≡ extE N f t
  uniqE N f h hh hη (true , γ) = cong h (ΣPathP (refl , fin1-const γ)) ∙ hη (γ zero)
  uniqE N f h hh hη (false , γ) =
    cong h (ΣPathP (refl , fin0-ext γ)) ∙ hh .pres tt noArgs ∙ cong (N .alg tt) (fin0-ext _)

errFree : FreeModel errTheory T
errFree = record
  { η        = ηE
  ; ops      = opsE
  ; ops-sat  = λ ()
  ; ext      = extE
  ; ext-η    = λ N f x → refl
  ; ext-hom  = λ N f → ishom λ _ args → cong (N .alg tt) (sym (fin0-ext _))
  ; ext-uniq = uniqE
  }

errOM : OperationalModel errTheory
errOM = mkOperationalModel errTheory errPoly errFree
  (λ f → ishom λ _ args → ΣPathP (refl , fin0-ext _)) (λ f x → refl)

open import POPL.ConfigReduction errTheory errOM using (Affine; ρ; _∘[_]_)

-- Errors are affine: every shape has at most one position.
err-affine : Affine
err-affine s i o = inj-from-prop (props (s ∘[ i ] o)) (ρ s i o)
  where
  props : (b : Bool) → isProp (Fin (errPoly .P b))
  props true  = isPropFin1
  props false = λ ()

-- The derived rule:  S[err()] ↦T err.  The Effect rule's right-hand side,
-- for C = ⟨run, •⟩, is the error configuration ⟨raised⟩.
err-effect : {X : Type} (γ : Fin∖ 1 zero → X) (k : Fin 0 → X)
  → FreeModel.μ errFree ((∂map ηE (true , zero , γ)) ⟪ opsE tt (λ j → ηE (k j)) ⟫)
    ≡ (false , noArgs)
err-effect γ k = refl
