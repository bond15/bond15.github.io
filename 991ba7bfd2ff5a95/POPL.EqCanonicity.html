{-
  Paper §3.4, second corollary: uniqueness of equational canonical forms.

  CORRECTION C3: the paper states, when ⟦-⟧ : Val(A) ≅ ⟦A⟧,
      ∀ M : Comp(F A). ∃! t : Term_Σ(Val(A)). M = reify(t).
  This is false for theories with equations: for writer,
  reify(tell_e(var V)) = tell_e(ret V) = ret V = reify(var V), and
  tell_e(var V) ≠ var V in Term_Σ.  The paper's proof identifies
  Term_Σ with the free model, which holds only when E is empty.  What
  holds (and what the proof sketch establishes) is uniqueness of the
  canonical form in the free model T(⟦A⟧):

      M ≈ reify t  and  M ≈ reify t'   ⇒   [t] = [t']  in T⟦A⟧,

  where [t] interprets t in T⟦A⟧ with leaves η(⟦V⟧).  Moreover [t] is
  the denotation ⟦M⟧, so the canonical form computes the denotation.
-}
open import Cubical.Foundations.Prelude using (Type)
open import POPL.Theory
open import POPL.FreeModel

module POPL.EqCanonicity (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma

open Theory 𝒯
open Model
open FreeModel fm

open import POPL.Syntax sig
open import POPL.Denotation 𝒯 fm
open import POPL.Equational 𝒯
open import POPL.EquationalSoundness 𝒯 fm
open import POPL.EqLogicalRelation 𝒯

private variable
  A : VTy

-- [t] : the image of a canonical form in the free model T⟦A⟧.
⌊_⌋ : Term sig (CVal A) → T ⟦ A ⟧v
⌊ t ⌋ = eval sig ops (λ V → η (⟦ V ⟧V ρ∅)) t

den-reify : (t : Term sig (CVal A)) → ⟦ reify t ⟧C ρ∅ ≡ ⌊ t ⌋
den-reify (var V)   = refl
den-reify (op o ts) = cong (ops o) (funExt λ j → den-reify (ts j))

-- The canonical form denotes ⟦M⟧.
canonical-form-denotes : (M : CComp (F A)) (t : Term sig (CVal A))
  → M ≈c reify t → ⟦ M ⟧C ρ∅ ≡ ⌊ t ⌋
canonical-form-denotes M t p = sound-c p ρ∅ ∙ den-reify t

-- Corrected uniqueness corollary (C3).
canonical-form-unique : (M : CComp (F A)) (t t' : Term sig (CVal A))
  → M ≈c reify t → M ≈c reify t' → ⌊ t ⌋ ≡ ⌊ t' ⌋
canonical-form-unique M t t' p p' =
  sym (canonical-form-denotes M t p) ∙ canonical-form-denotes M t' p'
