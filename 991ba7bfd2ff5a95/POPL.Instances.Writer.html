{-
  Paper §2.1 Example "Theory of Writer", §2.2 its free model, and the §5.3
  Writer example.

  Theory: one unary operation tell_m for each m in a monoid (M, ⊗, e), with
      tell_e(t) = t        tell_m(tell_n(t)) = tell_{m ⊗ n}(t)
  Free model: T(X) = M × X with tell_n(m, x) = (n ⊗ m, x), η(x) = (e, x).
  As a polynomial: shapes M, one position each.

  Derived rule (§5.3):  ⟨m, S[tell_n(M)]⟩ ↦T ⟨m ⊗ n, S[M]⟩.

  The monoid is a type with the unit and associativity laws; it need not
  be a set.
-}
module POPL.Instances.Writer where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin; zero; suc)

open import POPL.Theory
open import POPL.FreeModel
open import POPL.Polynomial renaming (plug to _⟪_⟫)
open import POPL.OperationalModel
open import POPL.Instances.Common

record MonoidOn (M : Type) : Type where
  field
    e     : M
    _⊗_   : M → M → M
    unitˡ : ∀ m → e ⊗ m ≡ m
    unitʳ : ∀ m → m ⊗ e ≡ m
    assoc : ∀ m n l → (m ⊗ n) ⊗ l ≡ m ⊗ (n ⊗ l)

isPropFin1 : isProp (Fin 1)
isPropFin1 zero zero = refl

fin1-const : {X : Type} (g : Fin 1 → X) → g ≡ (λ _ → g zero)
fin1-const g = funExt λ { zero → refl }

module _ (M : Type) (mon : MonoidOn M) where
  open MonoidOn mon

  sigW : Signature
  sigW = record { Op = M ; ar = λ _ → 1 }

  x₀ : Term sigW (Fin 1)
  x₀ = var zero

  tell : M → Term sigW (Fin 1) → Term sigW (Fin 1)
  tell m t = op m (λ _ → t)

  -- tell_e(t) = t ;  tell_m(tell_n(t)) = tell_{m⊗n}(t)
  writerTheory : Theory
  writerTheory = record
    { sig = sigW
    ; Eqn = Unit ⊎ (M × M)
    ; eqn = λ
      { (inl _)       → record { ctx = 1 ; lhs = tell e x₀ ; rhs = x₀ }
      ; (inr (m , n)) → record { ctx = 1 ; lhs = tell m (tell n x₀) ; rhs = tell (m ⊗ n) x₀ } }
    }

  open Theory writerTheory using () renaming (Eqn to EqnW)
  open Model

  writerPoly : Poly
  writerPoly = record { Sh = M ; P = λ _ → 1 }

  private
    T : Type → Type
    T = ⟦ writerPoly ⟧

    ηW : {X : Type} → X → T X
    ηW x = e , λ _ → x

    opsW : {X : Type} → AlgStr sigW (T X)
    opsW n args = n ⊗ args zero .fst , args zero .snd

    satW : {X : Type} → Sat writerTheory (opsW {X})
    satW (inl _)       γ = ΣPathP (unitˡ _ , refl)
    satW (inr (m , n)) γ = ΣPathP (sym (assoc m n _) , refl)

    extW : {X : Type} (N : Model writerTheory) → (X → N .Carrier) → T X → N .Carrier
    extW N f (m , γ) = N .alg m (λ _ → f (γ zero))

    -- In any model, tell_e(x) = x and tell_m(tell_n(x)) = tell_{m⊗n}(x).
    tell-e : (N : Model writerTheory) (x : N .Carrier) → N .alg e (λ _ → x) ≡ x
    tell-e N x = N .sat (inl tt) (λ _ → x)

    tell-tell : (N : Model writerTheory) (m n : M) (x : N .Carrier)
      → N .alg m (λ _ → N .alg n (λ _ → x)) ≡ N .alg (m ⊗ n) (λ _ → x)
    tell-tell N m n x = N .sat (inr (m , n)) (λ _ → x)

  writerFree : FreeModel writerTheory T
  writerFree = record
    { η        = ηW
    ; ops      = opsW
    ; ops-sat  = satW
    ; ext      = extW
    ; ext-η    = λ N f x → tell-e N (f x)
    ; ext-hom  = λ N f → ishom λ n args →
        sym (tell-tell N n (args zero .fst) _)
        ∙ cong (N .alg n) (sym (fin1-const (λ j → extW N f (args j))))
    ; ext-uniq = λ N f h hh hη → λ { (m , γ) →
        cong h (ΣPathP (sym (unitʳ m) , fin1-const γ))
        ∙ hh .pres m (λ _ → ηW (γ zero))
        ∙ cong (N .alg m) (funExt λ _ → hη (γ zero)) }
    }

  writerOM : OperationalModel writerTheory
  writerOM = mkOperationalModel writerTheory writerPoly writerFree
    (λ f → ishom λ _ _ → refl) (λ f x → refl)

  open import POPL.ConfigReduction writerTheory writerOM using (Affine; ρ)

  -- Writer is affine: every shape has one position (Example, §5.4).
  writer-affine : Affine
  writer-affine s i o = inj-from-prop isPropFin1 (ρ s i o)

  -- The derived rule:  ⟨m, S[tell_n(M)]⟩ ↦T ⟨m ⊗ n, S[M]⟩.
  -- The Effect rule's right-hand side μ((∂η C)⟪⟦tell_n⟧(η(S[M]))⟫), for
  -- C = ⟨m, •⟩, is ⟨m ⊗ n, S[M]⟩ (take k j = S[M]).
  writer-effect : {X : Type} (m n : M) (γ : Fin∖ 1 zero → X) (k : Fin 1 → X)
    → FreeModel.μ writerFree ((∂map ηW (m , zero , γ)) ⟪ opsW n (λ j → ηW (k j)) ⟫)
      ≡ (m ⊗ n , λ _ → k zero)
  writer-effect m n γ k = ΣPathP (cong (m ⊗_) (unitʳ n) , refl)
