{-
  Paper §2.3, Fig. 2: the equational theory of CBPV⁺(𝒯).

  DEVIATION D4 (agreed): the paper treats Comp as quotiented by the
  equations, so equations are equalities of terms.  We keep terms as a
  plain inductive type and define the equational theory as an inductive
  relation ≈ on CBPV⁺ terms: an equivalence relation, a congruence for
  every term former, and the axioms below.  Wherever the paper writes
  "M = N" for CBPV⁺ terms we write M ≈ N.  CBPV(𝒯) (mode cbpv) has no
  equations, as in §4.

  DEVIATION D5: Fig. 2 gives only a "fragment" of the β/η laws and says
  β and η laws hold "for all type connectives".  We take the standard
  full set (Levy):

    β:  case₊(σᵢ V){x.M₁ | x.M₂} = Mᵢ[V/x]       (values and computations)
        case×((V,W)){x y.M} = M[V/x, W/y]          (values and computations)
        force (thunk M) = M      (λx.M) V = M[V/x]      πᵢ⟨M₁,M₂⟩ = Mᵢ
        x ← ret V ; N = N[V/x]
    η:  M[V/z] = case₊(V){x. M[σ₁x/z] | y. M[σ₂y/z]}   (values and computations)
        M[V/z] = case×(V){x y. M[(x,y)/z]}               (values and computations)
        W[V/z] = absurd V   and   M[V/z] = force (absurd V)     for V : 0
        V = ⋆ (V : 1)    thunk (force V) = V    λx. M x = M
        ⟨π₁ M, π₂ M⟩ = M    M = triv (M : ⊤)
        S[M] = x ← M ; S[ret x]   (η for F, in the general stack form;
                                     S = • gives Fig. 2's x ← M ; ret x = M)

  The η laws for the sums and products are stated in the general
  (substitution) form, of which Fig. 2's case₊(V){x.σ₁x | y.σ₂y} = V is
  the instance M = z.

  Algebraic laws (Fig. 2): S[op(M⃗)] = op(S[M⃗]) for every stack S
  (including complex stacks), and every equation of 𝒯 holds in
  Γ | · ⊢ B.  op(M⃗)[γ] = op(M⃗[γ]) holds definitionally.
-}
open import POPL.Theory

module POPL.Equational (𝒯 : Theory) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.FinData using (Fin)

open Theory 𝒯

open import POPL.Syntax sig
open import POPL.Renaming sig
open import POPL.Substitution sig
open import POPL.Plug sig

private variable
  Γ : Ctx
  A A' A'' : VTy
  B B' B₁ B₂ : CTy

-- Closed and open CBPV⁺ terms.
Val⁺ : Ctx → VTy → Type
Val⁺ = Val cbpv⁺

Comp⁺ : Ctx → CTy → Type
Comp⁺ = Comp cbpv⁺

Stack⁺ : Ctx → CTy → CTy → Type
Stack⁺ = Stack cbpv⁺

-- (Γ | · ⊢ B) as a Σ-algebra, with the syntactic operations.
CompAlg : (m : Mode) (Γ : Ctx) (B : CTy) → AlgStr sig (Comp m Γ B)
CompAlg m Γ B o k = op o k

-- Substitutions used to state the η laws.
wk₁ : {Γ : Ctx} {A : VTy} → Subst cbpv⁺ (Γ ▸ A) Γ
wk₁ x = var (vs x)

wk₂ : {Γ : Ctx} {A A' : VTy} → Subst cbpv⁺ (Γ ▸ A ▸ A') Γ
wk₂ x = var (vs (vs x))

-- z ↦ σ₁ x, z ↦ σ₂ y, z ↦ (x , y)
inlS : {Γ : Ctx} {A A' : VTy} → Subst cbpv⁺ (Γ ▸ A) (Γ ▸ (A ⊕ A'))
inlS = wk₁ ,ₛ inl (var vz)

inrS : {Γ : Ctx} {A A' : VTy} → Subst cbpv⁺ (Γ ▸ A') (Γ ▸ (A ⊕ A'))
inrS = wk₁ ,ₛ inr (var vz)

pairS : {Γ : Ctx} {A A' : VTy} → Subst cbpv⁺ (Γ ▸ A ▸ A') (Γ ▸ (A ⊗ A'))
pairS = wk₂ ,ₛ vpair (var (vs vz)) (var vz)

infix 4 _≈v_ _≈c_

data _≈v_ : {Γ : Ctx} {A : VTy} → Val⁺ Γ A → Val⁺ Γ A → Type
data _≈c_ : {Γ : Ctx} {B : CTy} → Comp⁺ Γ B → Comp⁺ Γ B → Type

data _≈v_ where
  -- equivalence
  ≈refl  : {V : Val⁺ Γ A} → V ≈v V
  ≈sym   : {V W : Val⁺ Γ A} → V ≈v W → W ≈v V
  ≈trans : {U V W : Val⁺ Γ A} → U ≈v V → V ≈v W → U ≈v W
  -- congruence
  absurd≈ : {V V' : Val⁺ Γ 𝟘} → V ≈v V' → absurd {A = A} V ≈v absurd V'
  inl≈    : {V V' : Val⁺ Γ A} → V ≈v V' → inl {A' = A'} V ≈v inl V'
  inr≈    : {V V' : Val⁺ Γ A'} → V ≈v V' → inr {A = A} V ≈v inr V'
  vpair≈  : {V V' : Val⁺ Γ A} {W W' : Val⁺ Γ A'}
          → V ≈v V' → W ≈v W' → vpair V W ≈v vpair V' W'
  thunk≈  : {M M' : Comp⁺ Γ B} → M ≈c M' → thunk M ≈v thunk M'
  vcase+≈ : {V V' : Val⁺ Γ (A ⊕ A')} {W₁ W₁' : Val⁺ (Γ ▸ A) A''} {W₂ W₂' : Val⁺ (Γ ▸ A') A''}
          → V ≈v V' → W₁ ≈v W₁' → W₂ ≈v W₂' → vcase+ V W₁ W₂ ≈v vcase+ V' W₁' W₂'
  vcase×≈ : {V V' : Val⁺ Γ (A ⊗ A')} {W W' : Val⁺ (Γ ▸ A ▸ A') A''}
          → V ≈v V' → W ≈v W' → vcase× V W ≈v vcase× V' W'
  -- β
  vcase+-β₁ : (V : Val⁺ Γ A) (W₁ : Val⁺ (Γ ▸ A) A'') (W₂ : Val⁺ (Γ ▸ A') A'')
            → vcase+ (inl V) W₁ W₂ ≈v subV (idS ,ₛ V) W₁
  vcase+-β₂ : (V : Val⁺ Γ A') (W₁ : Val⁺ (Γ ▸ A) A'') (W₂ : Val⁺ (Γ ▸ A') A'')
            → vcase+ (inr V) W₁ W₂ ≈v subV (idS ,ₛ V) W₂
  vcase×-β  : (V : Val⁺ Γ A) (V' : Val⁺ Γ A') (W : Val⁺ (Γ ▸ A ▸ A') A'')
            → vcase× (vpair V V') W ≈v subV (idS ,ₛ V ,ₛ V') W
  -- η
  η𝟘v : (W : Val⁺ (Γ ▸ 𝟘) A) (V : Val⁺ Γ 𝟘) → subV (idS ,ₛ V) W ≈v absurd V
  η𝟙  : (V : Val⁺ Γ 𝟙) → V ≈v ⋆
  ηU  : (V : Val⁺ Γ (U B)) → thunk (force V) ≈v V
  η+v : (W : Val⁺ (Γ ▸ (A ⊕ A')) A'') (V : Val⁺ Γ (A ⊕ A'))
      → subV (idS ,ₛ V) W ≈v vcase+ V (subV inlS W) (subV inrS W)
  η×v : (W : Val⁺ (Γ ▸ (A ⊗ A')) A'') (V : Val⁺ Γ (A ⊗ A'))
      → subV (idS ,ₛ V) W ≈v vcase× V (subV pairS W)

data _≈c_ where
  -- equivalence
  ≈refl  : {M : Comp⁺ Γ B} → M ≈c M
  ≈sym   : {M N : Comp⁺ Γ B} → M ≈c N → N ≈c M
  ≈trans : {L M N : Comp⁺ Γ B} → L ≈c M → M ≈c N → L ≈c N
  -- congruence
  case+≈ : {V V' : Val⁺ Γ (A ⊕ A')} {M₁ M₁' : Comp⁺ (Γ ▸ A) B} {M₂ M₂' : Comp⁺ (Γ ▸ A') B}
         → V ≈v V' → M₁ ≈c M₁' → M₂ ≈c M₂' → case+ V M₁ M₂ ≈c case+ V' M₁' M₂'
  case×≈ : {V V' : Val⁺ Γ (A ⊗ A')} {M M' : Comp⁺ (Γ ▸ A ▸ A') B}
         → V ≈v V' → M ≈c M' → case× V M ≈c case× V' M'
  force≈ : {V V' : Val⁺ Γ (U B)} → V ≈v V' → force V ≈c force V'
  lam≈   : {M M' : Comp⁺ (Γ ▸ A) B} → M ≈c M' → lam M ≈c lam M'
  app≈   : {M M' : Comp⁺ Γ (A ⇒ B)} {V V' : Val⁺ Γ A}
         → M ≈c M' → V ≈v V' → app M V ≈c app M' V'
  cpair≈ : {M M' : Comp⁺ Γ B₁} {N N' : Comp⁺ Γ B₂}
         → M ≈c M' → N ≈c N' → cpair M N ≈c cpair M' N'
  π₁≈    : {M M' : Comp⁺ Γ (B₁ & B₂)} → M ≈c M' → π₁ M ≈c π₁ M'
  π₂≈    : {M M' : Comp⁺ Γ (B₁ & B₂)} → M ≈c M' → π₂ M ≈c π₂ M'
  ret≈   : {V V' : Val⁺ Γ A} → V ≈v V' → ret V ≈c ret V'
  bind≈  : {M M' : Comp⁺ Γ (F A)} {N N' : Comp⁺ (Γ ▸ A) B}
         → M ≈c M' → N ≈c N' → bind M N ≈c bind M' N'
  op≈    : (o : Op) {k k' : Fin (ar o) → Comp⁺ Γ B}
         → (∀ j → k j ≈c k' j) → op o k ≈c op o k'
  -- β
  case+-β₁ : (V : Val⁺ Γ A) (M₁ : Comp⁺ (Γ ▸ A) B) (M₂ : Comp⁺ (Γ ▸ A') B)
           → case+ (inl V) M₁ M₂ ≈c M₁ [ V ]₀
  case+-β₂ : (V : Val⁺ Γ A') (M₁ : Comp⁺ (Γ ▸ A) B) (M₂ : Comp⁺ (Γ ▸ A') B)
           → case+ (inr V) M₁ M₂ ≈c M₂ [ V ]₀
  case×-β  : (V : Val⁺ Γ A) (V' : Val⁺ Γ A') (M : Comp⁺ (Γ ▸ A ▸ A') B)
           → case× (vpair V V') M ≈c M [ V , V' ]₁
  U-β      : (M : Comp⁺ Γ B) → force (thunk M) ≈c M
  ⇒-β      : (M : Comp⁺ (Γ ▸ A) B) (V : Val⁺ Γ A) → app (lam M) V ≈c M [ V ]₀
  &-β₁     : (M : Comp⁺ Γ B₁) (N : Comp⁺ Γ B₂) → π₁ (cpair M N) ≈c M
  &-β₂     : (M : Comp⁺ Γ B₁) (N : Comp⁺ Γ B₂) → π₂ (cpair M N) ≈c N
  F-β      : (V : Val⁺ Γ A) (N : Comp⁺ (Γ ▸ A) B) → bind (ret V) N ≈c N [ V ]₀
  -- η
  η𝟘c : (M : Comp⁺ (Γ ▸ 𝟘) B) (V : Val⁺ Γ 𝟘) → M [ V ]₀ ≈c force (absurd V)
  η+c : (M : Comp⁺ (Γ ▸ (A ⊕ A')) B) (V : Val⁺ Γ (A ⊕ A'))
      → M [ V ]₀ ≈c case+ V (subC inlS M) (subC inrS M)
  η×c : (M : Comp⁺ (Γ ▸ (A ⊗ A')) B) (V : Val⁺ Γ (A ⊗ A'))
      → M [ V ]₀ ≈c case× V (subC pairS M)
  η⇒  : (M : Comp⁺ Γ (A ⇒ B)) → lam (app (wkC M) (var vz)) ≈c M
  η&  : (M : Comp⁺ Γ (B₁ & B₂)) → cpair (π₁ M) (π₂ M) ≈c M
  η⊤  : (M : Comp⁺ Γ ⊤) → M ≈c triv
  ηF  : (S : Stack⁺ Γ (F A) B) (M : Comp⁺ Γ (F A))
      → plug S M ≈c bind M (plug (wkS S) (ret (var vz)))
  -- algebraic laws
  op-stack : (S : Stack⁺ Γ B B') (o : Op) (k : Fin (ar o) → Comp⁺ Γ B)
           → plug S (op o k) ≈c op o (λ j → plug S (k j))
  theory   : (e : Eqn) (γ : Fin (ctx (eqn e)) → Comp⁺ Γ B)
           → eval sig (CompAlg cbpv⁺ Γ B) γ (lhs (eqn e))
             ≈c eval sig (CompAlg cbpv⁺ Γ B) γ (rhs (eqn e))

-- Reasoning helpers
≡→≈c : {M N : Comp⁺ Γ B} → M ≡ N → M ≈c N
≡→≈c {M = M} p = subst (M ≈c_) p ≈refl

≡→≈v : {V W : Val⁺ Γ A} → V ≡ W → V ≈v W
≡→≈v {V = V} p = subst (V ≈v_) p ≈refl
