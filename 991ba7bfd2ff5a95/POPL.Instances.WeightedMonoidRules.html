{-
  The derived weighted-monoid rules of §5.3, at the first position of a
  configuration ⟨(r, S[op(M⃗)]) | rest⟩, read as lists:

      ⟨(r, act_{r'}(M)) | ⋯⟩ ↦T ⟨(r·r', M) | ⋯⟩
      ⟨(r, e) | ⋯⟩           ↦T ⟨⋯⟩
      ⟨(r, M ⊗ M') | ⋯⟩      ↦T ⟨(r, M) | (r, M') | ⋯⟩

  Each lemma computes the Effect rule's right-hand side
  μ((∂η C)⟪⟦op⟧(η(S[M⃗]))⟫) for C = ⟨(r, •) | rest⟩.
-}
module POPL.Instances.WeightedMonoidRules where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Nat using (ℕ; zero; suc)
open import Cubical.Data.FinData using (Fin; zero; suc; snotz)
open import Cubical.Data.List using (List; []; _∷_; _++_; map)

open import POPL.Theory
open import POPL.FreeModel
open import POPL.Polynomial renaming (plug to _⟪_⟫)
open import POPL.Instances.Writer using (MonoidOn)
open import POPL.Instances.WeightedMonoid

module _ (R : Type) (mon : MonoidOn R) where
  open MonoidOn mon renaming (e to 1r; _⊗_ to _·_)
  open FreeModel (wmFree R mon) hiding (map)

  private
    T' = T R mon
    L' = L R mon
    toL'' = toL' R mon
    fromL' = fromL R mon

    -- the rest of the configuration, as a list
    restL : {X : Type} (m : ℕ) (w : Fin (suc m) → R) (γ : Fin∖ (suc m) zero → X) → L' X
    restL m w γ = toL'' m (w ∘ suc) (λ k → γ (suc k , snotz))

    -- μ of a configuration whose first entry is t and whose other
    -- entries are η(γ ·)
    μ-first : {X : Type} (m : ℕ) (w : Fin (suc m) → R) (γ : Fin∖ (suc m) zero → X) (t : T' X)
      → toL R mon (μ (((suc m , w) , zero , η ∘ γ) ⟪ t ⟫))
        ≡ map (scale R mon (w zero)) (toL R mon t) ++ restL m w γ
    μ-first {X} m w γ t =
      toL-fromL R mon _
      ∙ cong₂ _++_ (toL-fromL R mon _) (rest m (w ∘ suc) (λ k → γ (suc k , snotz)))
      where
      rest : (m : ℕ) (w' : Fin m → R) (h : Fin m → X)
        → toL R mon (extL R mon (FM _) (λ t → t) (toL'' m w' (η ∘ h))) ≡ toL'' m w' h
      rest zero    w' h = refl
      rest (suc m) w' h =
        toL-fromL R mon _
        ∙ cong₂ _++_ (toL-fromL R mon _ ∙ cong (_∷ []) (ΣPathP (unitʳ (w' zero) , refl)))
                     (rest m (w' ∘ suc) (h ∘ suc))

  -- ⟨(r, act_{r'}(M)) | ⋯⟩ ↦T ⟨(r·r', M) | ⋯⟩
  wm-act : {X : Type} (m : ℕ) (w : Fin (suc m) → R) (γ : Fin∖ (suc m) zero → X)
    (r' : R) (k : Fin 1 → X)
    → toL R mon (μ (((suc m , w) , zero , η ∘ γ) ⟪ ops (act r') (λ j → η (k j)) ⟫))
      ≡ (w zero · r' , k zero) ∷ restL m w γ
  wm-act m w γ r' k =
    μ-first m w γ (ops (act r') (λ j → η (k j)))
    ∙ cong (_++ restL m w γ) (cong (map (scale R mon (w zero)))
        (toL-fromL R mon (map (scale R mon r') ((1r , k zero) ∷ [])))
        ∙ cong (_∷ []) (ΣPathP (cong (w zero ·_) (unitʳ r') , refl)))

  -- ⟨(r, e) | ⋯⟩ ↦T ⟨⋯⟩
  wm-unit : {X : Type} (m : ℕ) (w : Fin (suc m) → R) (γ : Fin∖ (suc m) zero → X)
    (k : Fin 0 → X)
    → toL R mon (μ (((suc m , w) , zero , η ∘ γ) ⟪ ops unit (λ j → η (k j)) ⟫))
      ≡ restL m w γ
  wm-unit m w γ k = μ-first m w γ (ops unit (λ j → η (k j)))

  -- ⟨(r, M ⊗ M') | ⋯⟩ ↦T ⟨(r, M) | (r, M') | ⋯⟩
  wm-mul : {X : Type} (m : ℕ) (w : Fin (suc m) → R) (γ : Fin∖ (suc m) zero → X)
    (k : Fin 2 → X)
    → toL R mon (μ (((suc m , w) , zero , η ∘ γ) ⟪ ops mul (λ j → η (k j)) ⟫))
      ≡ (w zero , k zero) ∷ (w zero , k (suc zero)) ∷ restL m w γ
  wm-mul m w γ k =
    μ-first m w γ (ops mul (λ j → η (k j)))
    ∙ cong (_++ restL m w γ) (cong (map (scale R mon (w zero)))
        (toL-fromL R mon ((1r , k zero) ∷ (1r , k (suc zero)) ∷ []))
        ∙ cong₂ (λ a b → a ∷ b ∷ []) (ΣPathP (unitʳ (w zero) , refl)) (ΣPathP (unitʳ (w zero) , refl)))
