{-
  Paper §5.3 for operational models given as monad containers
  (DEVIATION D15).  This module repeats POPL.ConfigReduction, with the
  operational model of POPL.MonadContainer.OperationalModel in place of
  the paper's definition; the proofs are the same up to the lemma that
  the container's μ is the free model's μ (μₘ≡μ).

  The Effect rule keeps the derivative form,
      C⟪S[op(M⃗)]⟫ ↦T μ((∂η C)⟪⟦op⟧(η(S[M₁]), …)⟫),
  where η and μ are now the container's: η x = (e, λ_. x) and
  μ(s, v) = (s • v₀, λq. v₁ (↖ q) (↗ q)).  As a result the paper's §5
  definitions of s ∘ᵢ op and ρ_{s,i,op} hold by definition: the shape
  of the generic step is s • vᵢ, and its positions are given by ↖ and ↗
  (step-shape and step-pos below).
-}

private variable
  A : VTy
  B : CTy

-- Configurations over closed computations of type F A.
Config : VTy → Type
Config A = T (CComp (F A))

infix 4 _↦T_ _↦T*_

data _↦T_ : Config A → Config A → Type where
  pure   : (C : ∂ poly (CComp (F A))) {M M' : CComp (F A)}
         → M ↦ M' → C ⟪ M ⟫ ↦T C ⟪ M' ⟫
  effect : (C : ∂ poly (CComp (F A))) (S : CStack B (F A)) (o : Op) (k : Fin (ar o) → CComp B)
         → C ⟪ plug S (op o k) ⟫ ↦T μₘ ((∂map η C) ⟪ ops o (λ j → η (plug S (k j))) ⟫)

data _↦T*_ : Config A → Config A → Type where
  done : {t : Config A} → t ↦T* t
  _∷_  : {t t' t'' : Config A} → t ↦T t' → t' ↦T* t'' → t ↦T* t''

------------------------------------------------------------------------
-- Denotation of configurations

-- ⟦_⟧c on closed computations of type F A
den-c : CComp (F A) → T ⟦ A ⟧v
den-c M = ⟦ M ⟧C ρ∅

-- ⟦t̃⟧ = μ(T(⟦_⟧c)(t̃))
⟦_⟧T : Config A → T ⟦ A ⟧v
⟦ t ⟧T = μ (map den-c t)

-- Theorem (Soundness of initial configuration): ⟦η(M)⟧ = ⟦M⟧c
sound-init : (M : CComp (F A)) → ⟦ η M ⟧T ≡ den-c M
sound-init M = cong μ (map-η den-c M) ∙ μ-η (den-c M)

private
  -- The Kleisli extension of ⟦_⟧c, equal to ⟦_⟧T.
  E : Config A → T ⟦ A ⟧v
  E {A} = ext (FM ⟦ A ⟧v) den-c

  ⟦⟧T≡E : (t : Config A) → ⟦ t ⟧T ≡ E t
  ⟦⟧T≡E = ext-μmap den-c

  μmap-plug : {X Y : Type} (g : X → T Y) (C : ∂ poly X) (x : X)
    → μ (map g (C ⟪ x ⟫)) ≡ μ ((∂map g C) ⟪ g x ⟫)
  μmap-plug g C x = cong μ (map-plug g C x)

  ∂map-ext : (C : ∂ poly (CComp (F A)))
    → ∂map (E {A}) (∂map η C) ≡ ∂map den-c C
  ∂map-ext {A} (s , i , γ) =
    ΣPathP (refl , ΣPathP (refl , funExt λ x → ext-η (FM ⟦ A ⟧v) den-c (γ x)))

-- Theorem (Soundness of ↦T)
sound-T : {t t' : Config A} → t ↦T t' → ⟦ t ⟧T ≡ ⟦ t' ⟧T
sound-T (pure C {M} {M'} s) =
  μmap-plug den-c C M
  ∙ cong (λ x → μ ((∂map den-c C) ⟪ x ⟫)) (sound↦ s)
  ∙ sym (μmap-plug den-c C M')
sound-T {A = A} (effect C S o k) =
  μmap-plug den-c C (plug S (op o k))
  ∙ cong (λ x → μ ((∂map den-c C) ⟪ x ⟫)) (den-plug-op S o k ρ∅ ∙ sym opLemma)
  ∙ cong (λ C' → μ (C' ⟪ E X ⟫)) (sym (∂map-ext C))
  ∙ sym (μmap-plug E (∂map η C) X)
  ∙ ext-μmap E (∂map η C ⟪ X ⟫)
  ∙ sym (ext-μ (FM ⟦ A ⟧v) den-c (∂map η C ⟪ X ⟫))
  ∙ sym (⟦⟧T≡E (μ (∂map η C ⟪ X ⟫)))
  ∙ cong ⟦_⟧T (sym (μₘ≡μ (∂map η C ⟪ X ⟫)))
  where
  X : T (CComp (F A))
  X = ops o (λ j → η (plug S (k j)))
  opLemma : E X ≡ ops o (λ j → den-c (plug S (k j)))
  opLemma = ext-hom (FM ⟦ A ⟧v) den-c .pres o (λ j → η (plug S (k j)))
          ∙ cong (ops o) (funExt λ j → ext-η (FM ⟦ A ⟧v) den-c (plug S (k j)))

sound-T* : {t t' : Config A} → t ↦T* t' → ⟦ t ⟧T ≡ ⟦ t' ⟧T
sound-T* done    = refl
sound-T* (s ∷ p) = sound-T s ∙ sound-T* p

------------------------------------------------------------------------
-- The generic step and affinity

-- Cₛⁱ: the fresh one-hole context of shape s with hole at i.
C-fresh : (s : poly .Sh) (i : Fin (poly .P s)) → ∂ poly (Fin∖ (poly .P s) i)
C-fresh s i = s , i , λ x → x

-- ⌈op⌉ = ⟦op⟧(η(y₁), …, η(y_ar(op)))
⌈_⌉ : (o : Op) → T (Fin (ar o))
⌈ o ⌉ = ops o η

-- step(s, i, op) : T([P(s)] ∖ i ⊎ [ar(op)])
step : (s : poly .Sh) (i : Fin (poly .P s)) (o : Op) → T (Fin∖ (poly .P s) i ⊎ Fin (ar o))
step s i o = μₘ ((∂map (η ∘ inl) (C-fresh s i)) ⟪ map inr ⌈ o ⌉ ⟫)

-- s ∘ᵢ op and ρ_{s,i,op}
_∘[_]_ : (s : poly .Sh) (i : Fin (poly .P s)) (o : Op) → poly .Sh
s ∘[ i ] o = step s i o .fst

ρ : (s : poly .Sh) (i : Fin (poly .P s)) (o : Op)
  → Fin (poly .P (s ∘[ i ] o)) → Fin∖ (poly .P s) i ⊎ Fin (ar o)
ρ s i o = step s i o .snd

-- The paper's s ∘ᵢ op and ρ_{s,i,op}, in terms of the monad container:
-- vᵢ is the shape of ⌈op⌉ at position i and e elsewhere.
module _ (s : poly .Sh) (i : Fin (poly .P s)) (o : Op) where
  private
    g : Fin (poly .P s) → T (Fin∖ (poly .P s) i ⊎ Fin (ar o))
    g = fillAt i (η ∘ inl) (map inr ⌈ o ⌉)
    v : Fin (poly .P s) → poly .Sh
    v p = g p .fst

  step-shape : s ∘[ i ] o ≡ OperationalModel._•_ om s v
  step-shape = refl

  step-pos : (q : Fin (poly .P (s ∘[ i ] o)))
    → ρ s i o q ≡ g (OperationalModel.↖ om s v q) .snd (OperationalModel.↗ om s v q)
  step-pos q = refl

-- Definition (Affinity): every ρ_{s,i,op} is injective.
isInjective : {X Y : Type} → (X → Y) → Type
isInjective f = ∀ x y → f x ≡ f y → x ≡ y

Affine : Type
Affine = ∀ s i o → isInjective (ρ s i o)

-- results.tex, "Simplification of General Reduction Rule":
--   T(γ ⊎ δ)(step(s,i,op)) = μ((∂(η ∘ γ) Cₛⁱ)⟪⟦op⟧(η(δ y₁), …)⟫)
simplification : {X : Type} (s : poly .Sh) (i : Fin (poly .P s)) (o : Op)
  (γ : Fin∖ (poly .P s) i → X) (δ : Fin (ar o) → X)
  → map (⊎.rec γ δ) (step s i o) ≡ μₘ ((s , i , η ∘ γ) ⟪ ops o (η ∘ δ) ⟫)
simplification s i o γ δ =
  cong (map g) (μₘ≡μ _)
  ∙ μ-natural g _
  ∙ cong μ (map-plug (map g) (∂map (η ∘ inl) (C-fresh s i)) (map inr ⌈ o ⌉))
  ∙ cong₂ (λ C x → μ (C ⟪ x ⟫))
      (ΣPathP (refl , ΣPathP (refl , funExt λ x → map-η g (inl x))))
      (sym (map-∘ g inr ⌈ o ⌉) ∙ map-hom δ .pres o η ∙ ∘-η)
  ∙ sym (μₘ≡μ _)
  where
  g = ⊎.rec γ δ
  ∘-η : ops o (map δ ∘ η) ≡ ops o (η ∘ δ)
  ∘-η = cong (ops o) (funExt (map-η δ))

-- The Effect rule is the generic step with its variables instantiated.
effect-via-step : (s : poly .Sh) (i : Fin (poly .P s)) (γ : Fin∖ (poly .P s) i → CComp (F A))
  (S : CStack B (F A)) (o : Op) (k : Fin (ar o) → CComp B)
  → μₘ ((∂map η (s , i , γ)) ⟪ ops o (λ j → η (plug S (k j))) ⟫)
    ≡ map (⊎.rec γ (λ j → plug S (k j))) (step s i o)
effect-via-step s i γ S o k = sym (simplification s i o γ (λ j → plug S (k j)))
