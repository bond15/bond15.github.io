{-
  Paper §4.2–4.3, Fig. 7: the operational canonicity logical relation
  for CBPV(𝒯), its closure properties, and the Fundamental Lemma.

  The predicates are defined by hand, by recursion on types; 𝒞⟦F A⟧ is
  inductive with the three clauses of Fig. 7 (no gluing; DEVIATION D6).

  ======================================================================
  CORRECTION C5: the anti-reduction lemma (§4.2, Lemma "Computation
  Logical Predicates Anti-reduction").  This changes the statement and
  the proof of a lemma in the paper.
  ======================================================================

  What the paper says.
    Lemma.  For every B, if M ↦tree M' then 𝒞[B](M') ⇒ 𝒞[B](M).
    Proof.  "Immediate from inductive hypothesis for ⊤, →, &.  For F A we
    proceed by induction on M" (ret / op cases).

  Why the → and & cases are not immediate.
    𝒞[A → B](M) is ∀V. 𝒱[A](V) ⇒ 𝒞[B](M V).  Getting 𝒞[B](M V) from
    𝒞[B](M' V) by the inductive hypothesis at B needs M V ↦tree M' V,
    i.e. that ↦tree is a congruence for the frame (−) V.  Fig. 5 does not
    give that.  Its stack rule S[M] ↦ S[M'] only lifts primitive redexes
    M ↦red M', and the operation rule S[op(K⃗)] ↦ op(S[K⃗]) moves the
    whole stack S at once.  Counterexample, with S = (•) W:

        M    = (op(K₁,K₂)) W      ↦  op(K₁ W, K₂ W)       = M'
        M V  = (op(K₁,K₂)) W V    ↦  op(K₁ W V, K₂ W V)   (S = (•) W V)
        M' V = (op(K₁ W,K₂ W)) V  ↦  op(K₁ W V, K₂ W V)   (S = (•) V)

    M V does not step to M' V: it is not an operation, and its only step
    is the one shown.  The two terms only have a common one-step reduct.
    Steps under an operation, op(…Mᵢ…) ↦ op(…Mᵢ'…), break the same way:
    op(K⃗) V goes to op(K⃗ V), never to op(K⃗') V.  The paper's
    commented-out remark that "stacks are graph homomorphisms" is false
    for this relation.

  What we prove instead.
    (1) The lemma for head steps M ↦h M' only (POPL.TreeReduction): the
        redex rule and the operation rule, not steps under an operation.
        That is every use of the lemma in the paper:
          • the congruence lemma (closure under operations) uses
            op(M⃗) V ↦ op(M⃗ V), a head step;
          • the fundamental lemma uses β-steps and, in the bind case,
            x ← S[N]; N' ↦ x ← S[N''] and x ← S[op M⃗]; N' ↦ op(…), all
            head steps;
          • the back clause of 𝒞[F A] requires M ≠ op(N⃗), and a step from
            a non-operation is always a head step (notOp-head).
        The F A case under an operation, which the paper's proof treats
        by induction on M, is never needed.
    (2) The proof proves anti-reduction (AR) and forward preservation
        (FW: 𝒞[B](M) ⇒ 𝒞[B](M')) together, by induction on B.  It uses
        two facts the paper does not state:
          • hjoin (POPL.TreeReduction): for any stack S', if M ↦h M' then
            either S'[M] ↦h S'[M'], or S'[M] and S'[M'] both head-step to
            a common X (the counterexample above is the second case);
          • determinism of head steps (POPL.TreeReduction.det), proved by
            computing head reduction with a function and showing every
            head step agrees with it.  The paper assumes determinism
            ("only one reduction applies") in §4 and §5 but never
            proves it.
        Cases:
          • ⊤: trivial.
          • F A: AR is the back clause.  For FW, a proof of 𝒞[F A](M) with
            M head-stepping must be back with some step M ↦ M''; ret V
            and op(K⃗) take no head step.  Determinism gives M'' = M'.
          • A → B (and & with π₁, π₂): apply hjoin with S' = (•) V.  If
            M V ↦h M' V, use AR/FW at B.  Otherwise M V ↦h X and
            M' V ↦h X: for AR, FW at B takes 𝒞[B](M' V) to 𝒞[B](X), then
            AR at B takes it to 𝒞[B](M V); FW is symmetric.

  Effect on the rest of the paper: none.  The logical relation (Fig. 7)
  and the statements of the congruence lemma, the Fundamental Lemma and
  the corollaries are unchanged; only this lemma's statement is
  restricted to head steps and its proof replaced.

  CORRECTION C6: Fig. 7 defines 𝒱[1](⋆) := ⊤ and 𝒱[A+A'], 𝒱[A×A'] only
  on σᵢ W and pairs.  We read every undefined clause as ⊥.  (In CBPV(𝒯)
  the only other closed values of these types are absurd V, and there
  are no closed values of type 0.)
-}

-- 𝒞⟦F A⟧ (Fig. 7).
data FPred (P : CVal A → Type) : CComp (F A) → Type where
  ret  : {V : CVal A} → P V → FPred P (ret V)
  op   : (o : Op) {k : Fin (ar o) → CComp (F A)}
       → (∀ j → FPred P (k j)) → FPred P (op o k)
  back : {M M' : CComp (F A)} → NotOp M → M ↦ M' → FPred P M' → FPred P M

𝒱⟦_⟧ : (A : VTy) → CVal A → Type
𝒞⟦_⟧ : (B : CTy) → CComp B → Type

𝒱⟦ 𝟘 ⟧ V              = ⊥
𝒱⟦ 𝟙 ⟧ ⋆              = Unit
𝒱⟦ 𝟙 ⟧ _              = ⊥
𝒱⟦ A ⊕ A' ⟧ (inl W)   = 𝒱⟦ A ⟧ W
𝒱⟦ A ⊕ A' ⟧ (inr W)   = 𝒱⟦ A' ⟧ W
𝒱⟦ A ⊕ A' ⟧ _         = ⊥
𝒱⟦ A ⊗ A' ⟧ (vpair V W) = 𝒱⟦ A ⟧ V × 𝒱⟦ A' ⟧ W
𝒱⟦ A ⊗ A' ⟧ _         = ⊥
𝒱⟦ U B ⟧ V            = 𝒞⟦ B ⟧ (force V)

𝒞⟦ ⊤ ⟧ M      = Unit
𝒞⟦ A ⇒ B ⟧ M  = (V : CVal A) → 𝒱⟦ A ⟧ V → 𝒞⟦ B ⟧ (app M V)
𝒞⟦ B & B' ⟧ M = 𝒞⟦ B ⟧ (π₁ M) × 𝒞⟦ B' ⟧ (π₂ M)
𝒞⟦ F A ⟧ M    = FPred 𝒱⟦ A ⟧ M

-- Anti-reduction and forward preservation along head steps (C5)

AR : (B : CTy) {M M' : CComp B} → M ↦h M' → 𝒞⟦ B ⟧ M' → 𝒞⟦ B ⟧ M
FW : (B : CTy) {M M' : CComp B} → M ↦h M' → 𝒞⟦ B ⟧ M → 𝒞⟦ B ⟧ M'

FW-F : {P : CVal A → Type} {M M' : CComp (F A)} → M ↦h M' → FPred P M → FPred P M'
FW-F h (ret v)      = ⊥.rec (ret-no-hstep h)
FW-F h (op o cs)    = ⊥.rec (op-no-hstep h)
FW-F {P = P} h (back n s c) = subst (FPred P) (sym (det n (head h) s)) c

AR ⊤ h c = tt
AR (A ⇒ B) h c V v with hjoin (appˢ • V) h
... | inl h'             = AR B h' (c V v)
... | inr (X , h₁ , h₂)  = AR B h₁ (FW B h₂ (c V v))
AR (B₁ & B₂) h (c₁ , c₂) = ar₁ , ar₂
  where
  ar₁ : 𝒞⟦ B₁ ⟧ _
  ar₁ with hjoin (π₁ˢ •) h
  ... | inl h'            = AR B₁ h' c₁
  ... | inr (X , h₁ , h₂) = AR B₁ h₁ (FW B₁ h₂ c₁)
  ar₂ : 𝒞⟦ B₂ ⟧ _
  ar₂ with hjoin (π₂ˢ •) h
  ... | inl h'            = AR B₂ h' c₂
  ... | inr (X , h₁ , h₂) = AR B₂ h₁ (FW B₂ h₂ c₂)
AR (F A) h c = back (hstep-notOp h) (head h) c

FW ⊤ h c = tt
FW (A ⇒ B) h c V v with hjoin (appˢ • V) h
... | inl h'             = FW B h' (c V v)
... | inr (X , h₁ , h₂)  = AR B h₂ (FW B h₁ (c V v))
FW (B₁ & B₂) h (c₁ , c₂) = fw₁ , fw₂
  where
  fw₁ : 𝒞⟦ B₁ ⟧ _
  fw₁ with hjoin (π₁ˢ •) h
  ... | inl h'            = FW B₁ h' c₁
  ... | inr (X , h₁ , h₂) = AR B₁ h₂ (FW B₁ h₁ c₁)
  fw₂ : 𝒞⟦ B₂ ⟧ _
  fw₂ with hjoin (π₂ˢ •) h
  ... | inl h'            = FW B₂ h' c₂
  ... | inr (X , h₁ , h₂) = AR B₂ h₂ (FW B₂ h₁ c₂)
FW (F A) h c = FW-F h c

------------------------------------------------------------------------
-- Lemma "Computation Logical Predicates Congruence": 𝒞⟦B⟧ is closed
-- under the operations.

𝒞-op : (B : CTy) (o : Op) {k : Fin (ar o) → CComp B}
  → (∀ j → 𝒞⟦ B ⟧ (k j)) → 𝒞⟦ B ⟧ (op o k)
𝒞-op ⊤        o c = tt
𝒞-op (A ⇒ B)  o {k} c V v =
  AR B (opS (appˢ • V) tt o k) (𝒞-op B o (λ j → c j V v))
𝒞-op (B & B') o {k} c =
    AR B  (opS (π₁ˢ •) tt o k) (𝒞-op B  o (λ j → c j .fst))
  , AR B' (opS (π₂ˢ •) tt o k) (𝒞-op B' o (λ j → c j .snd))
𝒞-op (F A)    o c = op o c

-- Fundamental Lemma (Thm "Fundamental Lemma", §4.3)

𝒢⟦_⟧ : (Γ : Ctx) → Subst cbpv ε Γ → Type
𝒢⟦ Γ ⟧ γ = ∀ {A} (x : Var Γ A) → 𝒱⟦ A ⟧ (γ x)

𝒢-ext : {γ : Subst cbpv ε Γ} {V : CVal A}
  → 𝒢⟦ Γ ⟧ γ → 𝒱⟦ A ⟧ V → 𝒢⟦ Γ ▸ A ⟧ (γ ,ₛ V)
𝒢-ext g v vz     = v
𝒢-ext g v (vs x) = g x

-- Γ ⊨ V : A  and  Γ | · ⊨ M : B
_⊨v_ : (Γ : Ctx) → Val cbpv Γ A → Type
_⊨v_ {A = A} Γ V = (γ : Subst cbpv ε Γ) → 𝒢⟦ Γ ⟧ γ → 𝒱⟦ A ⟧ (subV γ V)

_⊨c_ : (Γ : Ctx) → Comp cbpv Γ B → Type
_⊨c_ {B = B} Γ M = (γ : Subst cbpv ε Γ) → 𝒢⟦ Γ ⟧ γ → 𝒞⟦ B ⟧ (subC γ M)

𝒞-≡ : (B : CTy) {M M' : CComp B} → M ≡ M' → 𝒞⟦ B ⟧ M' → 𝒞⟦ B ⟧ M
𝒞-≡ B p = subst 𝒞⟦ B ⟧ (sym p)

-- Eliminating related closed sums and pairs.
fund-case+ : (B : CTy) (W : CVal (A ⊕ A')) → 𝒱⟦ A ⊕ A' ⟧ W
  → (M₁ : Comp cbpv (ε ▸ A) B) (M₂ : Comp cbpv (ε ▸ A') B)
  → (∀ Y → 𝒱⟦ A ⟧ Y → 𝒞⟦ B ⟧ (M₁ [ Y ]₀))
  → (∀ Y → 𝒱⟦ A' ⟧ Y → 𝒞⟦ B ⟧ (M₂ [ Y ]₀))
  → 𝒞⟦ B ⟧ (case+ W M₁ M₂)
fund-case+ B (inl Y) w M₁ M₂ h₁ h₂ = AR B (stk • (case+-inl Y M₁ M₂)) (h₁ Y w)
fund-case+ B (inr Y) w M₁ M₂ h₁ h₂ = AR B (stk • (case+-inr Y M₁ M₂)) (h₂ Y w)
fund-case+ B (absurd _) () M₁ M₂ h₁ h₂

fund-case× : (B : CTy) (W : CVal (A ⊗ A')) → 𝒱⟦ A ⊗ A' ⟧ W
  → (M : Comp cbpv (ε ▸ A ▸ A') B)
  → (∀ Y Y' → 𝒱⟦ A ⟧ Y → 𝒱⟦ A' ⟧ Y' → 𝒞⟦ B ⟧ (M [ Y , Y' ]₁))
  → 𝒞⟦ B ⟧ (case× W M)
fund-case× B (vpair Y Y') (u , u') M h = AR B (stk • (case×-pair Y Y' M)) (h Y Y' u u')
fund-case× B (absurd _) () M h

-- The bind case: induction on the derivation of 𝒞⟦F A⟧(M[γ]).
fund-bind : (B : CTy) (N : Comp cbpv (ε ▸ A) B)
  → (∀ V → 𝒱⟦ A ⟧ V → 𝒞⟦ B ⟧ (N [ V ]₀))
  → {M : CComp (F A)} → 𝒞⟦ F A ⟧ M → 𝒞⟦ B ⟧ (bind M N)
fund-bind B N hN (ret {V = V} v) = AR B (stk • (bind-ret V N)) (hN V v)
fund-bind B N hN (op o {k} c) =
  AR B (opS (bindˢ • N) tt o k) (𝒞-op B o (λ j → fund-bind B N hN (c j)))
fund-bind B N hN (back n s c) with hjoin (bindˢ • N) (notOp-head n s)
... | inl h'            = AR B h' (fund-bind B N hN c)
... | inr (X , h₁ , h₂) = AR B h₁ (FW B h₂ (fund-bind B N hN c))

fundV : (V : Val cbpv Γ A) → Γ ⊨v V
fundC : (M : Comp cbpv Γ B) → Γ ⊨c M

fundV (var x)     γ g = g x
fundV (absurd V)  γ g = ⊥.rec (fundV V γ g)
fundV ⋆           γ g = tt
fundV (inl V)     γ g = fundV V γ g
fundV (inr V)     γ g = fundV V γ g
fundV (vpair V W) γ g = fundV V γ g , fundV W γ g
fundV (thunk {B = B} M) γ g = AR B (stk • (force-thunk _)) (fundC M γ g)

fundC (case+ {B = B} V M₁ M₂) γ g =
  fund-case+ B (subV γ V) (fundV V γ g) _ _
    (λ Y u → 𝒞-≡ B (sub₀-lift γ M₁ Y) (fundC M₁ (γ ,ₛ Y) (𝒢-ext g u)))
    (λ Y u → 𝒞-≡ B (sub₀-lift γ M₂ Y) (fundC M₂ (γ ,ₛ Y) (𝒢-ext g u)))
fundC (case× {B = B} V M) γ g =
  fund-case× B (subV γ V) (fundV V γ g) _
    (λ Y Y' u u' → 𝒞-≡ B (sub₁-lift γ M Y Y') (fundC M (γ ,ₛ Y ,ₛ Y') (𝒢-ext (𝒢-ext g u) u')))
fundC (force V)   γ g = fundV V γ g
fundC triv        γ g = tt
fundC (lam {B = B} M) γ g Y u =
  AR B (stk • (app-lam _ Y)) (𝒞-≡ B (sub₀-lift γ M Y) (fundC M (γ ,ₛ Y) (𝒢-ext g u)))
fundC (app M V)   γ g = fundC M γ g _ (fundV V γ g)
fundC (cpair {B₁ = B₁} {B₂ = B₂} M N) γ g =
  AR B₁ (stk • (π₁-pair _ _)) (fundC M γ g) , AR B₂ (stk • (π₂-pair _ _)) (fundC N γ g)
fundC (π₁ M)      γ g = fundC M γ g .fst
fundC (π₂ M)      γ g = fundC M γ g .snd
fundC (ret V)     γ g = ret (fundV V γ g)
fundC (bind {B = B} M N) γ g =
  fund-bind B _ (λ V v → 𝒞-≡ B (sub₀-lift γ N V) (fundC N (γ ,ₛ V) (𝒢-ext g v))) (fundC M γ g)
fundC (op {B = B} o k) γ g = 𝒞-op B o (λ j → fundC (k j) γ g)

-- Every closed computation is related.
related : (M : CComp B) → 𝒞⟦ B ⟧ M
related {B = B} M = 𝒞-≡ B (sym (sub-idC (λ _ → refl) M)) (fundC M idS (λ ()))
