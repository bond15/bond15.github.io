{-
  Paper §2.1 Example "Theory of Finite State" for S = 2, §2.2 Example
  (free model S → X × S), and the §5.3 state example.

  States are Fin 2 (the paper's {0, 1}).  Operations: get of arity 2,
  set_c of arity 1 for each state c.

  DEVIATION D13: the paper lists the lens laws only as "including"
  set₀(get(t,t')) = set₀(t) and set_s(set_s'(t)) = set_s'(t).  We take
  the standard complete set (Plotkin–Power):
      get(set₀(x), set₁(x)) = x                         (get-set)
      set_c(get(x₀, x₁))    = set_c(x_c)                (set-get)
      set_c(set_c'(x))      = set_c'(x)                 (set-set)
      get(get(x₀₀,x₀₁), get(x₁₀,x₁₁)) = get(x₀₀, x₁₁)    (get-get)

  Free model: (2 × X)² ≅ 2² × X², a polynomial with shapes 2 → 2 (the
  current state as a function of the initial state) and two positions
  (the term run from each initial state):
      η(x) = ⟨id, x, x⟩
      get(t₀, t₁) at initial state s runs t_s from s
      set_c(t)    at initial state s runs t from c

  Derived rules (§5.3), for the position of initial state 0:
      ⟨(b₀, S[get(M₀, M₁)]), (b₁, N)⟩ ↦T ⟨(b₀, S[M_{b₀}]), (b₁, N)⟩
      ⟨(b₀, S[set_c(M)]),    (b₁, N)⟩ ↦T ⟨(c,  S[M]),      (b₁, N)⟩
-}
module POPL.Instances.State where

open import Cubical.Foundations.Function

open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin; zero; suc; znots; snotz)

open import POPL.Theory
open import POPL.FreeModel

open import POPL.Instances.Common
open import POPL.Instances.Writer using (fin1-const)

St : Type
St = Fin 2

pattern s₀ = zero
pattern s₁ = suc zero

data StOp : Type where
  get : StOp
  set : St → StOp

sigS : Signature
sigS = record { Op = StOp ; ar = λ { get → 2 ; (set _) → 1 } }

data StEqn : Type where
  get-set : StEqn
  set-get : St → StEqn
  set-set : St → St → StEqn
  get-get : StEqn

-- x_{ij} for the get-get law
idx : St → St → Fin 4
idx s₀ s₀ = zero
idx s₀ s₁ = suc zero
idx s₁ s₀ = suc (suc zero)
idx s₁ s₁ = suc (suc (suc zero))

private
  getT : {n : ℕ} → (St → Term sigS (Fin n)) → Term sigS (Fin n)
  getT ts = op get ts

  setT : {n : ℕ} → St → Term sigS (Fin n) → Term sigS (Fin n)
  setT c t = op (set c) (λ _ → t)

stateTheory : Theory
stateTheory = record
  { sig = sigS
  ; Eqn = StEqn
  ; eqn = λ
    { get-set → record { ctx = 1 ; lhs = getT (λ s → setT s (var zero)) ; rhs = var zero }
    ; (set-get c) → record { ctx = 2 ; lhs = setT c (getT var) ; rhs = setT c (var c) }
    ; (set-set c c') → record { ctx = 1 ; lhs = setT c (setT c' (var zero)) ; rhs = setT c' (var zero) }
    ; get-get → record { ctx = 4 ; lhs = getT (λ i → getT (λ j → var (idx i j)))
                                 ; rhs = getT (λ i → var (idx i i)) } }
  }

open Model

statePoly : Poly
statePoly = record { Sh = St → St ; P = λ _ → 2 }

private
  T : Type → Type
  T = ⟦ statePoly ⟧

ηS : {X : Type} → X → T X
ηS x = (λ s → s) , (λ _ → x)

opsS : {X : Type} → AlgStr sigS (T X)
opsS get     args = (λ s → args s .fst s) , (λ s → args s .snd s)
opsS (set c) args = (λ _ → args zero .fst c) , (λ _ → args zero .snd c)

private
  satS : {X : Type} → Sat stateTheory (opsS {X})
  satS get-get       γ = ΣPathP (funExt (λ { s₀ → refl ; s₁ → refl }) , funExt (λ { s₀ → refl ; s₁ → refl }))
  satS get-set       γ = refl
  satS (set-get c)   γ = refl
  satS (set-set c c') γ = refl

  -- The laws in an arbitrary model N.
  module Laws (N : Model stateTheory) where
    Nget : (St → N .Carrier) → N .Carrier
    Nget = N .alg get
    Nset : St → N .Carrier → N .Carrier
    Nset c x = N .alg (set c) (λ _ → x)

    getset : ∀ x → Nget (λ s → Nset s x) ≡ x
    getset x = N .sat get-set (λ _ → x)

    setget : ∀ c (g : St → N .Carrier) → Nset c (Nget g) ≡ Nset c (g c)
    setget c g = N .sat (set-get c) g

    setset : ∀ c c' x → Nset c (Nset c' x) ≡ Nset c' x
    setset c c' x = N .sat (set-set c c') (λ _ → x)

    getget : (g : St → St → N .Carrier) → Nget (λ i → Nget (λ j → g i j)) ≡ Nget (λ i → g i i)
    getget g =
      cong Nget (funExt λ i → cong Nget (funExt λ j → sym (γidx i j)))
      ∙ N .sat get-get γ
      ∙ cong Nget (funExt λ i → γidx i i)
      where
      γ : Fin 4 → N .Carrier
      γ zero                   = g s₀ s₀
      γ (suc zero)             = g s₀ s₁
      γ (suc (suc zero))       = g s₁ s₀
      γ (suc (suc (suc zero))) = g s₁ s₁
      γidx : ∀ i j → γ (idx i j) ≡ g i j
      γidx s₀ s₀ = refl
      γidx s₀ s₁ = refl
      γidx s₁ s₀ = refl
      γidx s₁ s₁ = refl

    -- get(x, x) = x, derived.
    getconst : ∀ x → Nget (λ _ → x) ≡ x
    getconst x = cong Nget (funExt λ _ → sym (getset x))
               ∙ getget (λ i j → Nset j x)
               ∙ getset x

  extS : {X : Type} (N : Model stateTheory) → (X → N .Carrier) → T X → N .Carrier
  extS N f (σ , γ) = Nget (λ s → Nset (σ s) (f (γ s)))
    where open Laws N

  homS : {X : Type} (N : Model stateTheory) (f : X → N .Carrier)
    → IsHom sigS opsS (N .alg) (extS N f)
  homS N f .pres get args =
    sym (getget (λ i j → Nset (args i .fst j) (f (args i .snd j))))
    where open Laws N
  homS N f .pres (set c) args =
    getconst _
    ∙ sym (setset c (a .fst c) (f (a .snd c)))
    ∙ sym (setget c (λ s → Nset (a .fst s) (f (a .snd s))))
    ∙ cong (N .alg (set c)) (sym (fin1-const (λ j → extS N f (args j))))
    where
    open Laws N
    a = args zero

  uniqS : {X : Type} (N : Model stateTheory) (f : X → N .Carrier) (h : T X → N .Carrier)
    → IsHom sigS opsS (N .alg) h → (∀ x → h (ηS x) ≡ f x) → ∀ t → h t ≡ extS N f t
  uniqS N f h hh hη (σ , γ) =
    hh .pres get (λ s → opsS (set (σ s)) (λ _ → ηS (γ s)))
    ∙ cong (N .alg get) (funExt λ s →
        hh .pres (set (σ s)) (λ _ → ηS (γ s))
        ∙ cong (λ x → N .alg (set (σ s)) (λ _ → x)) (hη (γ s)))

stateFree : FreeModel stateTheory T
stateFree = record
  { η        = ηS
  ; ops      = opsS
  ; ops-sat  = satS
  ; ext      = extS
  ; ext-η    = λ N f x → Laws.getset N (f x)
  ; ext-hom  = homS
  ; ext-uniq = uniqS
  }

stateOM : OperationalModel stateTheory
stateOM = mkOperationalModel stateTheory statePoly stateFree
  (λ f → ishom λ { get args → refl ; (set c) args → refl }) (λ f x → refl)

open import POPL.ConfigReduction stateTheory stateOM using (Affine; ρ)

-- State is affine (Example, §5.4): get drops a subterm, set keeps them.

private
  isInl : {A B : Type} → A ⊎ B → Bool
  isInl (inl _) = true
  isInl (inr _) = false

  lr : {A B : Type} {a : A} {b : B} → inl a ≡ inr b → ⊥
  lr p = true≢false (cong isInl p)

state-affine : Affine
state-affine σ s₀ get     s₀ s₀ p = refl
state-affine σ s₀ get     s₁ s₁ p = refl
state-affine σ s₀ get     s₀ s₁ p = ⊥.rec (lr (sym p))
state-affine σ s₀ get     s₁ s₀ p = ⊥.rec (lr p)
state-affine σ s₁ get     s₀ s₀ p = refl
state-affine σ s₁ get     s₁ s₁ p = refl
state-affine σ s₁ get     s₀ s₁ p = ⊥.rec (lr p)
state-affine σ s₁ get     s₁ s₀ p = ⊥.rec (lr (sym p))
state-affine σ s₀ (set c) s₀ s₀ p = refl
state-affine σ s₀ (set c) s₁ s₁ p = refl
state-affine σ s₀ (set c) s₀ s₁ p = ⊥.rec (lr (sym p))
state-affine σ s₀ (set c) s₁ s₀ p = ⊥.rec (lr p)
state-affine σ s₁ (set c) s₀ s₀ p = refl
state-affine σ s₁ (set c) s₁ s₁ p = refl
state-affine σ s₁ (set c) s₀ s₁ p = ⊥.rec (lr p)
state-affine σ s₁ (set c) s₁ s₀ p = ⊥.rec (lr (sym p))

-- The derived rules (§5.3), at the position of initial state 0.  A
-- configuration is (σ, γ): from initial state s the current state is σ s
-- and the running term is γ s.

-- ⟨(b₀, S[get(M₀,M₁)]), (b₁, N)⟩ ↦T ⟨(b₀, S[M_{b₀}]), (b₁, N)⟩
state-get : {X : Type} (σ : St → St) (γ : Fin∖ 2 s₀ → X) (k : St → X)
  → FreeModel.μ stateFree ((∂map ηS (σ , s₀ , γ)) ⟪ opsS get (λ j → ηS (k j)) ⟫)
    ≡ (σ , λ { s₀ → k (σ s₀) ; s₁ → γ (s₁ , snotz) })
state-get σ γ k = ΣPathP (funExt (λ { s₀ → refl ; s₁ → refl }) , funExt (λ { s₀ → refl ; s₁ → refl }))

-- ⟨(b₀, S[set_c(M)]), (b₁, N)⟩ ↦T ⟨(c, S[M]), (b₁, N)⟩
state-set : {X : Type} (σ : St → St) (γ : Fin∖ 2 s₀ → X) (c : St) (k : Fin 1 → X)
  → FreeModel.μ stateFree ((∂map ηS (σ , s₀ , γ)) ⟪ opsS (set c) (λ j → ηS (k j)) ⟫)
    ≡ ((λ { s₀ → c ; s₁ → σ s₁ }) , λ { s₀ → k zero ; s₁ → γ (s₁ , snotz) })
state-set σ γ c k = ΣPathP (funExt (λ { s₀ → refl ; s₁ → refl }) , funExt (λ { s₀ → refl ; s₁ → refl }))
