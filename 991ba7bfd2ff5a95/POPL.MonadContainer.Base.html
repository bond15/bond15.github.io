{-
  Monad containers (Uustalu, "Container combinatorics: monads and lax
  monoidal functors", TTCS 2017, where they are called mnd-containers).

  A monad container is a container (polynomial) (S, P) with
      e : S
      s • v : S                         for v : P s → S
      ↖ s v : P (s • v) → P s           which outer position
      ↗ s v : (q : P (s • v)) → P (v (↖ s v q))   which inner position
  (Uustalu writes v ↖ₛ q and q ↗ᵥ s.)  Its monad on ⟦S, P⟧ is
      η x = (e, λ _. x)
      μ (s, v) = (s • v₀, λ q. v₁ (↖ q) (↗ q))     where v p = (v₀ p, v₁ p).

  Two ways of stating the laws:
    • MonadLaws (M): the three monad laws for η and μ, as paths in ⟦S,P⟧ X
      for every X.  This is the form used in the definition of an
      operational model (POPL.MonadContainer.OperationalModel).
    • UustaluLaws (U): Uustalu's eight laws, three on shapes and five on
      positions.  The position laws are dependent paths over the shape
      laws.
  M→U and U→M below show the two are equivalent (each implies the other,
  for the same data).  U→M computes both sides of each monad law; M→U
  instantiates each monad law at a generic element, whose leaves are its
  own positions, and reads the shape law off the first component and the
  position law off the second.
-}
module POPL.MonadContainer.Base where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin)

open import POPL.Polynomial

-- The operations of a monad container on a polynomial p.
record MCOps (p : Poly) : Type where
  Pos : p .Sh → Type
  Pos s = Fin (p .P s)
  field
    e   : p .Sh
    _•_ : (s : p .Sh) → (Pos s → p .Sh) → p .Sh
    ↖   : (s : p .Sh) (v : Pos s → p .Sh) → Pos (s • v) → Pos s
    ↗   : (s : p .Sh) (v : Pos s → p .Sh) (q : Pos (s • v)) → Pos (v (↖ s v q))

module _ {p : Poly} (c : MCOps p) where
  open MCOps c

  -- The induced η and μ on ⟦p⟧.
  ηᶜ : {X : Type} → X → ⟦ p ⟧ X
  ηᶜ x = e , λ _ → x

  μᶜ : {X : Type} → ⟦ p ⟧ (⟦ p ⟧ X) → ⟦ p ⟧ X
  μᶜ (s , γ) = s • (fst ∘ γ) , λ q → γ (↖ s (fst ∘ γ) q) .snd (↗ s (fst ∘ γ) q)

  -- (M) The monad laws.
  record MonadLaws : Type₁ where
    field
      μη    : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (ηᶜ t) ≡ t
      μmapη : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (pmap ηᶜ t) ≡ t
      μμ    : {X : Type} (t : ⟦ p ⟧ (⟦ p ⟧ (⟦ p ⟧ X))) → μᶜ (μᶜ t) ≡ μᶜ (pmap μᶜ t)

  -- (U) Uustalu's eight laws.
  record UustaluLaws : Type where
    field
      -- shapes
      unitʳ : (s : p .Sh) → s ≡ s • (λ _ → e)
      unitˡ : (s : p .Sh) → e • (λ _ → s) ≡ s
      assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → (s • v) • (λ q → w (↖ s v q) (↗ s v q)) ≡ s • (λ q → v q • w q)
      -- positions
      ↖-unitʳ : (s : p .Sh) → PathP (λ i → Pos (unitʳ s i) → Pos s) (λ q → q) (↖ s (λ _ → e))
      ↗-unitˡ : (s : p .Sh) → PathP (λ i → Pos (unitˡ s i) → Pos s) (↗ e (λ _ → s)) (λ q → q)
      ↖↖-assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → PathP (λ i → Pos (assoc s v w i) → Pos s)
                (λ q → ↖ s v (↖ (s • v) (λ q → w (↖ s v q) (↗ s v q)) q))
                (↖ s (λ q → v q • w q))
      ↗↖-assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → PathP (λ i → (q : Pos (assoc s v w i)) → Pos (v (↖↖-assoc s v w i q)))
                (λ q → ↗ s v (↖ (s • v) (λ q → w (↖ s v q) (↗ s v q)) q))
                (λ q → ↖ (v (↖ s (λ q → v q • w q) q)) (w (↖ s (λ q → v q • w q) q))
                         (↗ s (λ q → v q • w q) q))
      ↗↗-assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → PathP (λ i → (q : Pos (assoc s v w i))
                           → Pos (w (↖↖-assoc s v w i q) (↗↖-assoc s v w i q)))
                (λ q → ↗ (s • v) (λ q → w (↖ s v q) (↗ s v q)) q)
                (λ q → ↗ (v (↖ s (λ q → v q • w q) q)) (w (↖ s (λ q → v q • w q) q))
                         (↗ s (λ q → v q • w q) q))

  -- (U) ⇒ (M)
  U→M : UustaluLaws → MonadLaws
  U→M U = record { μη = μη' ; μmapη = μmapη' ; μμ = μμ' }
    where
    open UustaluLaws U
    μη' : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (ηᶜ t) ≡ t
    μη' (s , γ) i = unitˡ s i , λ q → γ (↗-unitˡ s i q)
    μmapη' : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (pmap ηᶜ t) ≡ t
    μmapη' (s , γ) i = unitʳ s (~ i) , λ q → γ (↖-unitʳ s (~ i) q)
    μμ' : {X : Type} (t : ⟦ p ⟧ (⟦ p ⟧ (⟦ p ⟧ X))) → μᶜ (μᶜ t) ≡ μᶜ (pmap μᶜ t)
    μμ' (s , γ) i =
      assoc s v w i ,
      λ q → γ (↖↖-assoc s v w i q) .snd (↗↖-assoc s v w i q) .snd (↗↗-assoc s v w i q)
      where
      v = λ q → γ q .fst
      w = λ q m → γ q .snd m .fst

  -- (M) ⇒ (U)
  M→U : MonadLaws → UustaluLaws
  M→U M = record
    { unitʳ    = λ s → sym (cong fst (μmapη (s , λ q → q)))
    ; unitˡ    = λ s → cong fst (μη (s , λ q → q))
    ; assoc    = λ s v w → cong fst (gen s v w)
    ; ↖-unitʳ  = λ s → symP (cong snd (μmapη (s , λ q → q)))
    ; ↗-unitˡ  = λ s → cong snd (μη (s , λ q → q))
    ; ↖↖-assoc = λ s v w i q → gen s v w i .snd q .fst
    ; ↗↖-assoc = λ s v w i q → gen s v w i .snd q .snd .fst
    ; ↗↗-assoc = λ s v w i q → gen s v w i .snd q .snd .snd
    }
    where
    open MonadLaws M
    -- the generic element: each leaf is its own position
    gen : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
      → _
    gen s v w = μμ {X = Σ[ q ∈ Pos s ] Σ[ m ∈ Pos (v q) ] Pos (w q m)}
                  (s , λ q → v q , λ m → w q m , λ r → q , m , r)

  -- U and M are logically equivalent.
  M↔U : (MonadLaws → UustaluLaws) × (UustaluLaws → MonadLaws)
  M↔U = M→U , U→M

-- A monad container.
record MonadContainer : Type₁ where
  field
    poly : Poly
    cops : MCOps poly
    laws : MonadLaws cops

  open MCOps cops public
  open MonadLaws laws public

  ηₘ : {X : Type} → X → ⟦ poly ⟧ X
  ηₘ = ηᶜ cops

  μₘ : {X : Type} → ⟦ poly ⟧ (⟦ poly ⟧ X) → ⟦ poly ⟧ X
  μₘ = μᶜ cops
