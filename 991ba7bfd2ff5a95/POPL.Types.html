{-
  Paper §2.3, Fig. 1: types and contexts of CBPV(𝒯) and CBPV⁺(𝒯).

    A ::= 0 | 1 | A + A' | A × A' | U B
    B ::= ⊤ | A → B | B & B' | F A

  Contexts are snoc lists of value types and variables are de Bruijn
  indices, so the syntax is non-unary (the paper's §3.1 doctrine is
  stated for unary syntax; its results are stated for full contexts).
-}
module POPL.Types where

open import Cubical.Foundations.Prelude

infixr 30 _⇒_
infixl 35 _⊕_
infixl 36 _⊗_
infixl 35 _&_
infixl 20 _▸_

data VTy : Type
data CTy : Type

data VTy where
  𝟘 𝟙       : VTy
  _⊕_ _⊗_   : VTy → VTy → VTy
  U         : CTy → VTy

data CTy where
  ⊤         : CTy
  _⇒_       : VTy → CTy → CTy
  _&_       : CTy → CTy → CTy
  F         : VTy → CTy

data Ctx : Type where
  ε   : Ctx
  _▸_ : Ctx → VTy → Ctx

data Var : Ctx → VTy → Type where
  vz : ∀ {Γ A}   → Var (Γ ▸ A) A
  vs : ∀ {Γ A B} → Var Γ A → Var (Γ ▸ B) A

{-
  CBPV(𝒯) and CBPV⁺(𝒯) share one syntax indexed by a mode.  The
  constructors highlighted grey in Fig. 1 (complex values and complex
  stacks) take an instance of `Plus m`, which exists only in mode cbpv⁺.
  So `Val cbpv Γ A` is exactly CBPV(𝒯), and the metatheory is proved
  once for both calculi.
-}
data Mode : Type where
  cbpv cbpv⁺ : Mode

data Plus : Mode → Type where
  instance plus : Plus cbpv⁺
