{-
  Paper §5.2, Definition "Operational Model", as a monad container.

  DEVIATION D15 (see the README): here an operational model is a monad
  container (POPL.MonadContainer.Base) whose monad is the free model
  monad of 𝒯.  The main development uses the paper's definition
  (POPL.OperationalModel); POPL.MonadContainer.ConfigReduction and
  POPL.MonadContainer.ConfigNormalization repeat §5 for this one.  Concretely:
    • a monad container (S, P, e, •, ↖, ↗) with the monad laws;
    • a generic effect ⌈op⌉ : ⟦S,P⟧(Fin ar(op)) for each operation, which
      gives the operations by  ⟦op⟧(args) = μ(T(args)(⌈op⌉))
      (algebraic operations from generic effects, Plotkin–Power);
    • the equations of 𝒯 hold in every ⟦S,P⟧ X;
    • ⟦S,P⟧ X is free on X, with unit the container's η.
  The paper's own definition (a free model monad whose functor is
  polynomial) is POPL.OperationalModel; POPL.MonadContainer.Equiv proves
  the two definitions equivalent.

  Derived here: the FreeModel record used by the denotational semantics,
  the fact that the container's μ is the free model's μ, and that the
  free model's functorial action is the polynomial one (the paper's
  defining condition, now a theorem).
-}
module POPL.MonadContainer.OperationalModel where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.FinData using (Fin)

open import POPL.Theory
open import POPL.FreeModel
open import POPL.Polynomial
open import POPL.MonadContainer.Base

module _ (𝒯 : Theory) where
  open Theory 𝒯
  open Model

  record OperationalModel : Type₁ where
    field
      mc : MonadContainer

    open MonadContainer mc public

    T : Type → Type
    T = ⟦ poly ⟧

    field
      -- ⌈op⌉ = ⟦op⟧(η y₁, …, η y_ar(op))
      gen : (o : Op) → T (Fin (ar o))

    -- ⟦op⟧(args) = μ(T(args)(⌈op⌉))
    opsᶜ : {X : Type} → AlgStr sig (T X)
    opsᶜ o args = μₘ (pmap args (gen o))

    field
      opsᶜ-sat  : {X : Type} → Sat 𝒯 (opsᶜ {X})
      extᶜ      : {X : Type} (M : Model 𝒯) → (X → M .Carrier) → T X → M .Carrier
      extᶜ-η    : {X : Type} (M : Model 𝒯) (f : X → M .Carrier) (x : X)
                → extᶜ M f (ηₘ x) ≡ f x
      extᶜ-hom  : {X : Type} (M : Model 𝒯) (f : X → M .Carrier)
                → IsHom sig opsᶜ (M .alg) (extᶜ M f)
      extᶜ-uniq : {X : Type} (M : Model 𝒯) (f : X → M .Carrier)
                  (h : T X → M .Carrier) → IsHom sig opsᶜ (M .alg) h
                → (∀ x → h (ηₘ x) ≡ f x)
                → ∀ t → h t ≡ extᶜ M f t

    -- The free model monad, for the denotational semantics.
    free : FreeModel 𝒯 T
    free = record
      { η        = ηₘ
      ; ops      = opsᶜ
      ; ops-sat  = opsᶜ-sat
      ; ext      = extᶜ
      ; ext-η    = extᶜ-η
      ; ext-hom  = extᶜ-hom
      ; ext-uniq = extᶜ-uniq
      }

    open FreeModel free public

    -- The container's μ is a homomorphism (by associativity) ...
    μₘ-hom : {X : Type} → IsHom sig ops ops (μₘ {X})
    μₘ-hom .pres o args = μμ (pmap args (gen o))

    -- ... so it is the free model's μ.
    μₘ≡μ : {X : Type} (t : T (T X)) → μₘ t ≡ μ t
    μₘ≡μ = ext-uniq (FM _) (λ t → t) μₘ μₘ-hom μη

    -- The paper's defining condition: the functorial action of the free
    -- model is the polynomial one.
    map-poly : {X Y : Type} (f : X → Y) (t : T X) → map f t ≡ pmap f t
    map-poly f t = sym (ext-uniq (FM _) (η ∘ f) (pmap f) (ishom λ _ _ → refl) (λ _ → refl) t)

    map-plug : {X Y : Type} (f : X → Y) (C : ∂ poly X) (x : X)
      → map f (plug C x) ≡ plug (∂map f C) (f x)
    map-plug f C x = map-poly f (plug C x) ∙ plug-map f C x
