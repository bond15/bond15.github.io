private variable
  m : Mode

  B B' : CTy

-- Δ ⇒ Γ in the paper: a value in Γ for each variable of Δ.
Subst : Mode → Ctx → Ctx → Type
Subst m Γ Δ = ∀ {A} → Var Δ A → Val m Γ A

liftS : Subst m Γ Δ → Subst m (Γ ▸ A) (Δ ▸ A)
liftS σ vz     = var vz
liftS σ (vs x) = wkV (σ x)

subV : Subst m Γ Δ → Val m Δ A → Val m Γ A
subC : Subst m Γ Δ → Comp m Δ B → Comp m Γ B
subS : Subst m Γ Δ → Stack m Δ B B' → Stack m Γ B B'

subV r (var x) = r x
subV r (absurd a0) = absurd (subV r a0)
subV r ⋆ = ⋆
subV r (inl a0) = inl (subV r a0)
subV r (inr a0) = inr (subV r a0)
subV r (vpair a0 a1) = vpair (subV r a0) (subV r a1)
subV r (thunk a0) = thunk (subC r a0)
subV r (vcase+ ⦃ p ⦄ a0 a1 a2) = vcase+ ⦃ p ⦄ (subV r a0) (subV (liftS r) a1) (subV (liftS r) a2)
subV r (vcase× ⦃ p ⦄ a0 a1) = vcase× ⦃ p ⦄ (subV r a0) (subV (liftS (liftS r)) a1)

subC r (case+ a0 a1 a2) = case+ (subV r a0) (subC (liftS r) a1) (subC (liftS r) a2)
subC r (case× a0 a1) = case× (subV r a0) (subC (liftS (liftS r)) a1)
subC r (force a0) = force (subV r a0)
subC r triv = triv
subC r (lam a0) = lam (subC (liftS r) a0)
subC r (app a0 a1) = app (subC r a0) (subV r a1)
subC r (cpair a0 a1) = cpair (subC r a0) (subC r a1)
subC r (π₁ a0) = π₁ (subC r a0)
subC r (π₂ a0) = π₂ (subC r a0)
subC r (ret a0) = ret (subV r a0)
subC r (bind a0 a1) = bind (subC r a0) (subC (liftS r) a1)
subC r (op o k) = op o (λ j → subC r (k j))

subS r • = •
subS r (appˢ a0 a1) = appˢ (subS r a0) (subV r a1)
subS r (π₁ˢ a0) = π₁ˢ (subS r a0)
subS r (π₂ˢ a0) = π₂ˢ (subS r a0)
subS r (bindˢ a0 a1) = bindˢ (subS r a0) (subC (liftS r) a1)
subS r (trivˢ ⦃ p ⦄) = trivˢ ⦃ p ⦄
subS r (lamˢ ⦃ p ⦄ a0) = lamˢ ⦃ p ⦄ (subS (liftS r) a0)
subS r (pairˢ ⦃ p ⦄ a0 a1) = pairˢ ⦃ p ⦄ (subS r a0) (subS r a1)

idS : Subst m Γ Γ
idS = var

-- γ , V/x
infixl 20 _,ₛ_
_,ₛ_ : Subst m Γ Δ → Val m Γ A → Subst m Γ (Δ ▸ A)
(σ ,ₛ V) vz     = V
(σ ,ₛ V) (vs x) = σ x

-- δ[γ]: composition of substitutions
_⊚_ : Subst m Γ Δ → Subst m Δ Θ → Subst m Γ Θ
(σ ⊚ σ') x = subV σ (σ' x)

-- M[V/x]
_[_]₀ : Comp m (Γ ▸ A) B → Val m Γ A → Comp m Γ B
M [ V ]₀ = subC (idS ,ₛ V) M

-- M[V/x, W/y]
_[_,_]₁ : Comp m (Γ ▸ A ▸ A') B → Val m Γ A → Val m Γ A' → Comp m Γ B
M [ V , W ]₁ = subC (idS ,ₛ V ,ₛ W) M

-- sub σ ∘ ren r = sub (σ ∘ r)

SubRen : Subst m Γ Δ → Ren Δ Θ → Subst m Γ Θ → Type
SubRen {Θ = Θ} σ r σ'' = ∀ {A} (x : Var Θ A) → σ (r x) ≡ σ'' x

liftS-ren : {σ : Subst m Γ Δ} {r : Ren Δ Θ} {σ'' : Subst m Γ Θ}
  → SubRen σ r σ'' → SubRen (liftS {A = A} σ) (liftR r) (liftS σ'')
liftS-ren h vz     = refl
liftS-ren h (vs x) = cong wkV (h x)

sub-renV : {σ : Subst m Γ Δ} {r : Ren Δ Θ} {σ'' : Subst m Γ Θ} → SubRen σ r σ''
  → (V : Val m Θ A) → subV σ (renV r V) ≡ subV σ'' V
sub-renC : {σ : Subst m Γ Δ} {r : Ren Δ Θ} {σ'' : Subst m Γ Θ} → SubRen σ r σ''
  → (M : Comp m Θ B) → subC σ (renC r M) ≡ subC σ'' M
sub-renS : {σ : Subst m Γ Δ} {r : Ren Δ Θ} {σ'' : Subst m Γ Θ} → SubRen σ r σ''
  → (S : Stack m Θ B B') → subS σ (renS r S) ≡ subS σ'' S

sub-renV {σ = σ} {r = r} {σ'' = σ''} h (var x) = h x
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (absurd a0) i = absurd (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h ⋆ = refl
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (inl a0) i = inl (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (inr a0) i = inr (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (vpair a0 a1) i = vpair (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (thunk a0) i = thunk (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (vcase+ ⦃ p ⦄ a0 a1 a2) i = vcase+ ⦃ p ⦄ (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a1 i) (sub-renV {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a2 i)
sub-renV {σ = σ} {r = r} {σ'' = σ''} h (vcase× ⦃ p ⦄ a0 a1) i = vcase× ⦃ p ⦄ (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = (liftS (liftS σ))} {r = (liftR (liftR r))} {σ'' = (liftS (liftS σ''))} (liftS-ren (liftS-ren h)) a1 i)

sub-renC {σ = σ} {r = r} {σ'' = σ''} h (case+ a0 a1 a2) i = case+ (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a1 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a2 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (case× a0 a1) i = case× (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS (liftS σ))} {r = (liftR (liftR r))} {σ'' = (liftS (liftS σ''))} (liftS-ren (liftS-ren h)) a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (force a0) i = force (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h triv = refl
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (lam a0) i = lam (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (app a0 a1) i = app (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (cpair a0 a1) i = cpair (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (π₁ a0) i = π₁ (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (π₂ a0) i = π₂ (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (ret a0) i = ret (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renC {σ = σ} {r = r} {σ'' = σ''} h (bind a0 a1) i = bind (sub-renC {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a1 i)

sub-renS {σ = σ} {r = r} {σ'' = σ''} h • = refl
sub-renS {σ = σ} {r = r} {σ'' = σ''} h (appˢ a0 a1) i = appˢ (sub-renS {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renV {σ = σ} {r = r} {σ'' = σ''} h a1 i)
sub-renS {σ = σ} {r = r} {σ'' = σ''} h (π₁ˢ a0) i = π₁ˢ (sub-renS {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renS {σ = σ} {r = r} {σ'' = σ''} h (π₂ˢ a0) i = π₂ˢ (sub-renS {σ = σ} {r = r} {σ'' = σ''} h a0 i)
sub-renS {σ = σ} {r = r} {σ'' = σ''} h (bindˢ a0 a1) i = bindˢ (sub-renS {σ = σ} {r = r} {σ'' = σ''} h a0 i) (sub-renC {σ = (liftS σ)} {r = (liftR r)} {σ'' = (liftS σ'')} (liftS-ren h) a1 i)
sub-renS {σ = σ} {r = r} {σ'' = σ''} h (trivˢ ⦃ p ⦄) = refl

-- ren r ∘ sub σ = sub (ren r ∘ σ)

RenSub : Ren Γ Δ → Subst m Δ Θ → Subst m Γ Θ → Type
RenSub {Θ = Θ} r σ σ'' = ∀ {A} (x : Var Θ A) → renV r (σ x) ≡ σ'' x

liftR-sub : {r : Ren Γ Δ} {σ : Subst m Δ Θ} {σ'' : Subst m Γ Θ}
  → RenSub r σ σ'' → RenSub (liftR {A = A} r) (liftS σ) (liftS σ'')
liftR-sub h vz     = refl
liftR-sub {r = r} {σ = σ} h (vs x) = wk-renV r (σ x) ∙ cong wkV (h x)

ren-subV : {r : Ren Γ Δ} {σ : Subst m Δ Θ} {σ'' : Subst m Γ Θ} → RenSub r σ σ''
  → (V : Val m Θ A) → renV r (subV σ V) ≡ subV σ'' V

  → (M : Comp m Θ B) → renC r (subC σ M) ≡ subC σ'' M

-- Weakening commutes with lifted substitutions.
wk-subV : (σ : Subst m Γ Δ) (V : Val m Δ A) → subV (liftS {A = A'} σ) (wkV V) ≡ wkV (subV σ V)
wk-subV σ V = sub-renV (λ _ → refl) V ∙ sym (ren-subV (λ _ → refl) V)

wk-subC : (σ : Subst m Γ Δ) (M : Comp m Δ B) → subC (liftS {A = A'} σ) (wkC M) ≡ wkC (subC σ M)
wk-subC σ M = sub-renC (λ _ → refl) M ∙ sym (ren-subC (λ _ → refl) M)

wk-subS : (σ : Subst m Γ Δ) (S : Stack m Δ B B') → subS (liftS {A = A'} σ) (wkS S) ≡ wkS (subS σ S)
wk-subS σ S = sub-renS (λ _ → refl) S ∙ sym (ren-subS (λ _ → refl) S)

-- σ = id  ⇒  sub σ = id

SubId : Subst m Γ Γ → Type
SubId {Γ = Γ} σ = ∀ {A} (x : Var Γ A) → σ x ≡ var x

liftS-id : {σ : Subst m Γ Γ} → SubId σ → SubId (liftS {A = A} σ)
liftS-id h vz     = refl
liftS-id h (vs x) = cong wkV (h x)

sub-idV : {σ : Subst m Γ Γ} → SubId σ → (V : Val m Γ A) → subV σ V ≡ V
sub-idC : {σ : Subst m Γ Γ} → SubId σ → (M : Comp m Γ B) → subC σ M ≡ M
sub-idS : {σ : Subst m Γ Γ} → SubId σ → (S : Stack m Γ B B') → subS σ S ≡ S

sub-idV {σ = σ} h (var x) = h x
sub-idV {σ = σ} h (absurd a0) i = absurd (sub-idV {σ = σ} h a0 i)
sub-idV {σ = σ} h ⋆ = refl
sub-idV {σ = σ} h (inl a0) i = inl (sub-idV {σ = σ} h a0 i)
sub-idV {σ = σ} h (inr a0) i = inr (sub-idV {σ = σ} h a0 i)
sub-idV {σ = σ} h (vpair a0 a1) i = vpair (sub-idV {σ = σ} h a0 i) (sub-idV {σ = σ} h a1 i)
sub-idV {σ = σ} h (thunk a0) i = thunk (sub-idC {σ = σ} h a0 i)
sub-idV {σ = σ} h (vcase+ ⦃ p ⦄ a0 a1 a2) i = vcase+ ⦃ p ⦄ (sub-idV {σ = σ} h a0 i) (sub-idV {σ = (liftS σ)} (liftS-id h) a1 i) (sub-idV {σ = (liftS σ)} (liftS-id h) a2 i)
sub-idV {σ = σ} h (vcase× ⦃ p ⦄ a0 a1) i = vcase× ⦃ p ⦄ (sub-idV {σ = σ} h a0 i) (sub-idV {σ = (liftS (liftS σ))} (liftS-id (liftS-id h)) a1 i)

sub-idC {σ = σ} h (case+ a0 a1 a2) i = case+ (sub-idV {σ = σ} h a0 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a1 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a2 i)
sub-idC {σ = σ} h (case× a0 a1) i = case× (sub-idV {σ = σ} h a0 i) (sub-idC {σ = (liftS (liftS σ))} (liftS-id (liftS-id h)) a1 i)
sub-idC {σ = σ} h (force a0) i = force (sub-idV {σ = σ} h a0 i)
sub-idC {σ = σ} h triv = refl
sub-idC {σ = σ} h (lam a0) i = lam (sub-idC {σ = (liftS σ)} (liftS-id h) a0 i)
sub-idC {σ = σ} h (app a0 a1) i = app (sub-idC {σ = σ} h a0 i) (sub-idV {σ = σ} h a1 i)
sub-idC {σ = σ} h (cpair a0 a1) i = cpair (sub-idC {σ = σ} h a0 i) (sub-idC {σ = σ} h a1 i)
sub-idC {σ = σ} h (π₁ a0) i = π₁ (sub-idC {σ = σ} h a0 i)
sub-idC {σ = σ} h (π₂ a0) i = π₂ (sub-idC {σ = σ} h a0 i)
sub-idC {σ = σ} h (ret a0) i = ret (sub-idV {σ = σ} h a0 i)
sub-idC {σ = σ} h (bind a0 a1) i = bind (sub-idC {σ = σ} h a0 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a1 i)
sub-idC {σ = σ} h (op o k) i = op o (λ j → sub-idC {σ = σ} h (k j) i)

sub-idS {σ = σ} h • = refl
sub-idS {σ = σ} h (appˢ a0 a1) i = appˢ (sub-idS {σ = σ} h a0 i) (sub-idV {σ = σ} h a1 i)
sub-idS {σ = σ} h (π₁ˢ a0) i = π₁ˢ (sub-idS {σ = σ} h a0 i)
sub-idS {σ = σ} h (π₂ˢ a0) i = π₂ˢ (sub-idS {σ = σ} h a0 i)
sub-idS {σ = σ} h (bindˢ a0 a1) i = bindˢ (sub-idS {σ = σ} h a0 i) (sub-idC {σ = (liftS σ)} (liftS-id h) a1 i)
sub-idS {σ = σ} h (trivˢ ⦃ p ⦄) = refl
sub-idS {σ = σ} h (lamˢ ⦃ p ⦄ a0) i = lamˢ ⦃ p ⦄ (sub-idS {σ = (liftS σ)} (liftS-id h) a0 i)
sub-idS {σ = σ} h (pairˢ ⦃ p ⦄ a0 a1) i = pairˢ ⦃ p ⦄ (sub-idS {σ = σ} h a0 i) (sub-idS {σ = σ} h a1 i)

-- Single substitution after a lifted substitution:
--   M[γ↑][V/x] = M[γ, V/x]

sub₀-lift : (σ : Subst m Γ Δ) (M : Comp m (Δ ▸ A) B) (V : Val m Γ A)
  → (subC (liftS σ) M) [ V ]₀ ≡ subC (σ ,ₛ V) M
sub₀-lift σ M V = sub-subC h M
  where
  h : SubSub (idS ,ₛ V) (liftS σ) (σ ,ₛ V)
  h vz     = refl
  h (vs x) = sub-renV (λ _ → refl) (σ x) ∙ sub-idV (λ _ → refl) (σ x)

sub₁-lift : (σ : Subst m Γ Δ) (M : Comp m (Δ ▸ A ▸ A') B) (V : Val m Γ A) (W : Val m Γ A')
  → (subC (liftS (liftS σ)) M) [ V , W ]₁ ≡ subC (σ ,ₛ V ,ₛ W) M
sub₁-lift σ M V W = sub-subC h M
  where
  h : SubSub (idS ,ₛ V ,ₛ W) (liftS (liftS σ)) (σ ,ₛ V ,ₛ W)
  h vz          = refl
  h (vs vz)     = refl
  h (vs (vs x)) = sub-renV (λ _ → refl) (wkV (σ x))
                ∙ sub-renV (λ _ → refl) (σ x) ∙ sub-idV (λ _ → refl) (σ x)

sub₀-liftV : (σ : Subst m Γ Δ) (W : Val m (Δ ▸ A) A') (V : Val m Γ A)
  → subV (idS ,ₛ V) (subV (liftS σ) W) ≡ subV (σ ,ₛ V) W
sub₀-liftV σ W V = sub-subV h W
  where
  h : SubSub (idS ,ₛ V) (liftS σ) (σ ,ₛ V)
  h vz     = refl
  h (vs x) = sub-renV (λ _ → refl) (σ x) ∙ sub-idV (λ _ → refl) (σ x)

sub₁-liftV : (σ : Subst m Γ Δ) (X : Val m (Δ ▸ A ▸ A') A'') (V : Val m Γ A) (W : Val m Γ A')
  → subV (idS ,ₛ V ,ₛ W) (subV (liftS (liftS σ)) X) ≡ subV (σ ,ₛ V ,ₛ W) X
sub₁-liftV σ X V W = sub-subV h X
  where
  h : SubSub (idS ,ₛ V ,ₛ W) (liftS (liftS σ)) (σ ,ₛ V ,ₛ W)
  h vz          = refl
  h (vs vz)     = refl
  h (vs (vs x)) = sub-renV (λ _ → refl) (wkV (σ x))
