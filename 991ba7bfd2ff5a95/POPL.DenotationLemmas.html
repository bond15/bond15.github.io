{-
  Properties of the denotation that the paper gets from initiality
  (§3.2: "it preserves substitution, stack plugging, the algebraic
  operations, and all type structure by construction"), proved here by
  induction (see DEVIATION D3 in POPL.Denotation):

    ⟦V[γ]⟧ = ⟦V⟧ ∘ ⟦γ⟧        ⟦S[M]⟧ = ⟦S⟧(⟦M⟧)
    ⟦S⟧ρ is a homomorphism of models (algebraicity)
-}

private variable
  m : Mode
  Γ Δ : Ctx
  A A' : VTy
  B B' : CTy

------------------------------------------------------------------------
-- Renaming

EnvRen : Ren Γ Δ → Env Γ → Env Δ → Type
EnvRen {Δ = Δ} r ρ ρ' = ∀ {A} (x : Var Δ A) → ρ (r x) ≡ ρ' x

liftER : {r : Ren Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} {a a' : ⟦ A ⟧v}
  → EnvRen r ρ ρ' → a ≡ a' → EnvRen (liftR {A = A} r) (ρ ▹ a) (ρ' ▹ a')
liftER h p vz     = p
liftER h p (vs x) = h x

den-renV : {r : Ren Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} → EnvRen r ρ ρ'
  → (V : Val m Δ A) → ⟦ renV r V ⟧V ρ ≡ ⟦ V ⟧V ρ'
den-renC : {r : Ren Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} → EnvRen r ρ ρ'
  → (M : Comp m Δ B) → ⟦ renC r M ⟧C ρ ≡ ⟦ M ⟧C ρ'
den-renS : {r : Ren Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} → EnvRen r ρ ρ'
  → (S : Stack m Δ B B') → ⟦ renS r S ⟧S ρ ≡ ⟦ S ⟧S ρ'

den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (var x) = h x
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (absurd V) i = ⊥.rec (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h ⋆ = refl
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (inl V) i = inl (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (inr V) i = inr (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (vpair V W) i =
  den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i , den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h W i
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (thunk M) = den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h M
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (vcase+ {A = A} {A' = A'} V W W') i =
  ⊎.rec (λ a → den-renV {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A} h refl) W i)
        (λ a → den-renV {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A'} h refl) W' i)
        (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h (vcase× {A = A} {A' = A'} V W) =
  den-renV {r = liftR (liftR f)}
    (liftER {A = A'} (liftER {A = A} h (cong fst (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V)))
                  (cong snd (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V))) W

den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (case+ {A = A} {A' = A'} V M N) i =
  ⊎.rec (λ a → den-renC {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A} h refl) M i)
        (λ a → den-renC {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A'} h refl) N i)
        (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (case× {A = A} {A' = A'} V M) =
  den-renC {r = liftR (liftR f)}
    (liftER {A = A'} (liftER {A = A} h (cong fst (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V)))
                  (cong snd (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V))) M
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (force V) = den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h triv = refl
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (lam {A = A} M) i a =
  den-renC {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A} h refl) M i
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (app M V) i =
  den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h M i (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (cpair M N) i =
  den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h M i , den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h N i
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (π₁ M) i = den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h M i .fst
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (π₂ M) i = den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h M i .snd
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (ret V) i = η (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (bind {A = A} {B = B} M N) i =
  ext ⟦ B ⟧c (λ a → den-renC {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A} h refl) N i)
    (den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h M i)
den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (op {B = B} o k) i =
  ⟦ B ⟧c .alg o (λ j → den-renC {r = f} {ρ = ρ} {ρ' = ρ'} h (k j) i)

den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h • = refl
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h (appˢ S V) i b =
  den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h S i b (den-renV {r = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h (π₁ˢ S) i b = den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h S i b .fst
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h (π₂ˢ S) i b = den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h S i b .snd
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h (bindˢ {A = A} {B' = B'} S N) i b =
  ext ⟦ B' ⟧c (λ a → den-renC {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A} h refl) N i)
    (den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h S i b)
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h trivˢ = refl
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h (lamˢ {A = A} S) i b a =
  den-renS {r = liftR f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftER {A = A} h refl) S i b
den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h (pairˢ S T) i b =
  den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h S i b , den-renS {r = f} {ρ = ρ} {ρ' = ρ'} h T i b

den-wkV : (ρ : Env Γ) (a : ⟦ A' ⟧v) (V : Val m Γ A)
  → ⟦ wkV {A' = A'} V ⟧V (_▹_ {A = A'} ρ a) ≡ ⟦ V ⟧V ρ
den-wkV {A' = A'} ρ a = den-renV {r = vs} {ρ = _▹_ {A = A'} ρ a} {ρ' = ρ} (λ _ → refl)

den-wkC : (ρ : Env Γ) (a : ⟦ A' ⟧v) (M : Comp m Γ B)
  → ⟦ wkC {A' = A'} M ⟧C (_▹_ {A = A'} ρ a) ≡ ⟦ M ⟧C ρ
den-wkC {A' = A'} ρ a = den-renC {r = vs} {ρ = _▹_ {A = A'} ρ a} {ρ' = ρ} (λ _ → refl)

------------------------------------------------------------------------
-- Substitution

EnvSub : Subst m Γ Δ → Env Γ → Env Δ → Type
EnvSub {Δ = Δ} σ ρ ρ' = ∀ {A} (x : Var Δ A) → ⟦ σ x ⟧V ρ ≡ ρ' x

liftES : {σ : Subst m Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} {a a' : ⟦ A ⟧v}
  → EnvSub σ ρ ρ' → a ≡ a' → EnvSub (liftS {A = A} σ) (ρ ▹ a) (ρ' ▹ a')
liftES h p vz                  = p
liftES {A = A} {σ = σ} {ρ} {a = a} h p (vs x) = den-wkV {A' = A} ρ a (σ x) ∙ h x

den-subV : {σ : Subst m Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} → EnvSub σ ρ ρ'
  → (V : Val m Δ A) → ⟦ subV σ V ⟧V ρ ≡ ⟦ V ⟧V ρ'
den-subC : {σ : Subst m Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} → EnvSub σ ρ ρ'
  → (M : Comp m Δ B) → ⟦ subC σ M ⟧C ρ ≡ ⟦ M ⟧C ρ'
den-subS : {σ : Subst m Γ Δ} {ρ : Env Γ} {ρ' : Env Δ} → EnvSub σ ρ ρ'
  → (S : Stack m Δ B B') → ⟦ subS σ S ⟧S ρ ≡ ⟦ S ⟧S ρ'

den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h (var x) = h x

den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h (inr V) i = inr (den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h V i)

den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h (thunk M) = den-subC {σ = f} {ρ = ρ} {ρ' = ρ'} h M
den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h (vcase+ {A = A} {A' = A'} V W W') i =
  ⊎.rec (λ a → den-subV {σ = liftS f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftES {A = A} h refl) W i)
        (λ a → den-subV {σ = liftS f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftES {A = A'} h refl) W' i)
        (den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h (vcase× {A = A} {A' = A'} V W) =
  den-subV {σ = liftS (liftS f)}
    (liftES {A = A'} (liftES {A = A} h (cong fst (den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h V)))
                  (cong snd (den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h V))) W

den-subC {σ = f} {ρ = ρ} {ρ' = ρ'} h (case+ {A = A} {A' = A'} V M N) i =
  ⊎.rec (λ a → den-subC {σ = liftS f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftES {A = A} h refl) M i)
        (λ a → den-subC {σ = liftS f} {ρ = ρ ▹ a} {ρ' = ρ' ▹ a} (liftES {A = A'} h refl) N i)
        (den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h V i)
den-subC {σ = f} {ρ = ρ} {ρ' = ρ'} h (case× {A = A} {A' = A'} V M) =
  den-subC {σ = liftS (liftS f)}
    (liftES {A = A'} (liftES {A = A} h (cong fst (den-subV {σ = f} {ρ = ρ} {ρ' = ρ'} h V)))

den-subS {σ = f} {ρ = ρ} {ρ' = ρ'} h • = refl
den-subS {σ = f} {ρ = ρ} {ρ' = ρ'} h (appˢ S V) i b =

den-subS {σ = f} {ρ = ρ} {ρ' = ρ'} h (lamˢ {A = A} S) i b a =

-- ⟦M[V/x]⟧ρ = ⟦M⟧(ρ, ⟦V⟧ρ)
den-sub₀ : (ρ : Env Γ) (M : Comp m (Γ ▸ A) B) (V : Val m Γ A)
  → ⟦ M [ V ]₀ ⟧C ρ ≡ ⟦ M ⟧C (ρ ▹ ⟦ V ⟧V ρ)
den-sub₀ ρ M V = den-subC h M
  where
  h : EnvSub (idS ,ₛ V) ρ (ρ ▹ ⟦ V ⟧V ρ)
  h vz     = refl
  h (vs x) = refl

den-sub₁ : (ρ : Env Γ) (M : Comp m (Γ ▸ A ▸ A') B) (V : Val m Γ A) (W : Val m Γ A')
  → ⟦ M [ V , W ]₁ ⟧C ρ ≡ ⟦ M ⟧C (ρ ▹ ⟦ V ⟧V ρ ▹ ⟦ W ⟧V ρ)
den-sub₁ ρ M V W = den-subC h M
  where
  h : EnvSub (idS ,ₛ V ,ₛ W) ρ (ρ ▹ ⟦ V ⟧V ρ ▹ ⟦ W ⟧V ρ)
  h vz          = refl
  h (vs vz)     = refl
  h (vs (vs x)) = refl

------------------------------------------------------------------------
-- Plugging: ⟦S[M]⟧ = ⟦S⟧(⟦M⟧)

den-plug : (S : Stack m Γ B B') (M : Comp m Γ B) (ρ : Env Γ)
  → ⟦ plug S M ⟧C ρ ≡ ⟦ S ⟧S ρ (⟦ M ⟧C ρ)
den-plug •           M ρ = refl
den-plug (appˢ S V)  M ρ i = den-plug S M ρ i (⟦ V ⟧V ρ)
den-plug (π₁ˢ S)     M ρ i = den-plug S M ρ i .fst
den-plug (π₂ˢ S)     M ρ i = den-plug S M ρ i .snd
den-plug (bindˢ {B' = B'} S N) M ρ i =
  ext ⟦ B' ⟧c (λ a → ⟦ N ⟧C (ρ ▹ a)) (den-plug S M ρ i)
den-plug trivˢ       M ρ = refl
den-plug (lamˢ S)    M ρ i a =
  (den-plug S (wkC M) (ρ ▹ a) ∙ cong (⟦ S ⟧S (ρ ▹ a)) (den-wkC ρ a M)) i
den-plug (pairˢ S T) M ρ i = den-plug S M ρ i , den-plug T M ρ i

------------------------------------------------------------------------
-- Algebraicity: ⟦S⟧ρ is a homomorphism ⟦B⟧ → ⟦B'⟧ (Fig. 2:
-- S[op(M⃗)] = op(S[M⃗]) holds semantically).

den-stack-hom : (S : Stack m Γ B B') (ρ : Env Γ)
  → IsModelHom 𝒯 ⟦ B ⟧c ⟦ B' ⟧c (⟦ S ⟧S ρ)
den-stack-hom •           ρ = ishom λ _ _ → refl
den-stack-hom (appˢ S V)  ρ = ishom λ o args →
  cong (λ f → f (⟦ V ⟧V ρ)) (den-stack-hom S ρ .pres o args)
den-stack-hom (π₁ˢ S)     ρ = ishom λ o args → cong fst (den-stack-hom S ρ .pres o args)
den-stack-hom (π₂ˢ S)     ρ = ishom λ o args → cong snd (den-stack-hom S ρ .pres o args)
den-stack-hom (bindˢ {B' = B'} S N) ρ = ishom λ o args →
  cong (ext ⟦ B' ⟧c (λ a → ⟦ N ⟧C (ρ ▹ a))) (den-stack-hom S ρ .pres o args)
  ∙ ext-hom ⟦ B' ⟧c (λ a → ⟦ N ⟧C (ρ ▹ a)) .pres o (λ j → ⟦ S ⟧S ρ (args j))
den-stack-hom trivˢ       ρ = ishom λ _ _ → refl
den-stack-hom (lamˢ S)    ρ = ishom λ o args i a → den-stack-hom S (ρ ▹ a) .pres o args i
den-stack-hom (pairˢ S T) ρ = ishom λ o args i →
  den-stack-hom S ρ .pres o args i , den-stack-hom T ρ .pres o args i

-- ⟦S[op(M⃗)]⟧ = ⟦op⟧(⟦S[M⃗]⟧)
den-plug-op : (S : Stack m Γ B B') (o : Op) (k : _ → Comp m Γ B) (ρ : Env Γ)
  → ⟦ plug S (op o k) ⟧C ρ ≡ ⟦ B' ⟧c .alg o (λ j → ⟦ plug S (k j) ⟧C ρ)
den-plug-op {B' = B'} S o k ρ =
  den-plug S (op o k) ρ
  ∙ den-stack-hom S ρ .pres o (λ j → ⟦ k j ⟧C ρ)
  ∙ cong (⟦ B' ⟧c .alg o) (funExt λ j → sym (den-plug S (k j) ρ))

-- Value versions of den-sub₀ / den-sub₁ (for the complex value eliminators).
den-sub₀V : (ρ : Env Γ) (W : Val m (Γ ▸ A) A') (V : Val m Γ A)
  → ⟦ subV (idS ,ₛ V) W ⟧V ρ ≡ ⟦ W ⟧V (ρ ▹ ⟦ V ⟧V ρ)
den-sub₀V ρ W V = den-subV h W
  where
  h : EnvSub (idS ,ₛ V) ρ (ρ ▹ ⟦ V ⟧V ρ)
  h vz     = refl
  h (vs x) = refl

den-sub₁V : {A'' : VTy} (ρ : Env Γ) (X : Val m (Γ ▸ A ▸ A') A'') (V : Val m Γ A) (W : Val m Γ A')
  → ⟦ subV (idS ,ₛ V ,ₛ W) X ⟧V ρ ≡ ⟦ X ⟧V (ρ ▹ ⟦ V ⟧V ρ ▹ ⟦ W ⟧V ρ)
den-sub₁V ρ X V W = den-subV h X
  where
  h : EnvSub (idS ,ₛ V ,ₛ W) ρ (ρ ▹ ⟦ V ⟧V ρ ▹ ⟦ W ⟧V ρ)
  h vz          = refl
  h (vs vz)     = refl
  h (vs (vs x)) = refl
