  Paper §3.2, Fig. 3: the denotational semantics of CBPV⁺(𝒯) (and so of
  CBPV(𝒯), its cbpv-mode fragment).

    ⟦A⟧ : Set        ⟦B⟧ : Alg 𝒯
    ⟦Γ ⊢ V : A⟧           : ⟦Γ⟧ → ⟦A⟧
    ⟦Γ | · ⊢ M : B⟧        : ⟦Γ⟧ → U⟦B⟧
    ⟦Γ | • : B ⊢ S : B'⟧   : ⟦Γ⟧ → ⟦B⟧ ⊸ ⟦B'⟧

  F A is interpreted by the given free model T⟦A⟧, and x ← M ; N by the
  unique homomorphism !_{curry ⟦N⟧ γ}.

  DEVIATION D3: the paper obtains ⟦-⟧ from initiality of the syntactic
  CBPV⁺ doctrine (§3.1), and only "gives the recursive equations".  We
  define ⟦-⟧ directly by those recursive equations and prove the
  properties initiality would give (substitution, plugging, algebraicity
  of stacks, soundness for ≈) by hand.  The doctrine is not formalized.

  CORRECTION C2: Fig. 3 writes ⟦tt⟧(γ) = {⋆}; the denotation of the unit
  value is the element ⋆, not the set {⋆}.
-}
open import Cubical.Foundations.Prelude using (Type)
open import POPL.Theory
open import POPL.FreeModel

module POPL.Denotation (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Unit
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.Sigma

open import POPL.Models
open Models-of 𝒯

open Theory 𝒯
open Model
open FreeModel fm

open import POPL.Syntax sig
open import POPL.Renaming sig
open import POPL.Substitution sig
open import POPL.Plug sig

private variable
  m : Mode
  Γ Δ : Ctx
  A A' : VTy
  B B' : CTy

------------------------------------------------------------------------
-- Types

⟦_⟧v : VTy → Type
⟦_⟧c : CTy → Model 𝒯

⟦ 𝟘 ⟧v      = ⊥
⟦ 𝟙 ⟧v      = Unit
⟦ A ⊕ A' ⟧v = ⟦ A ⟧v ⊎ ⟦ A' ⟧v
⟦ A ⊗ A' ⟧v = ⟦ A ⟧v × ⟦ A' ⟧v
⟦ U B ⟧v    = ⟦ B ⟧c .Carrier

⟦ ⊤ ⟧c      = unitModel
⟦ A ⇒ B ⟧c  = powModel ⟦ A ⟧v ⟦ B ⟧c
⟦ B & B' ⟧c = prodModel ⟦ B ⟧c ⟦ B' ⟧c
⟦ F A ⟧c    = FM ⟦ A ⟧v

-- U⟦B⟧
∣_∣ : CTy → Type
∣ B ∣ = ⟦ B ⟧c .Carrier

-- ⟦Γ⟧, as environments.
Env : Ctx → Type
Env Γ = ∀ {A} → Var Γ A → ⟦ A ⟧v

infixl 20 _▹_
_▹_ : Env Γ → ⟦ A ⟧v → Env (Γ ▸ A)
(ρ ▹ a) vz     = a
(ρ ▹ a) (vs x) = ρ x

------------------------------------------------------------------------
-- Terms

⟦_⟧V : Val m Γ A → Env Γ → ⟦ A ⟧v
⟦_⟧C : Comp m Γ B → Env Γ → ∣ B ∣
⟦_⟧S : Stack m Γ B B' → Env Γ → ∣ B ∣ → ∣ B' ∣

⟦ var x ⟧V ρ          = ρ x
⟦ absurd V ⟧V ρ       = ⊥.rec (⟦ V ⟧V ρ)
⟦ ⋆ ⟧V ρ              = tt
⟦ inl V ⟧V ρ          = inl (⟦ V ⟧V ρ)
⟦ inr V ⟧V ρ          = inr (⟦ V ⟧V ρ)
⟦ vpair V W ⟧V ρ      = ⟦ V ⟧V ρ , ⟦ W ⟧V ρ
⟦ thunk M ⟧V ρ        = ⟦ M ⟧C ρ
⟦ vcase+ V W W' ⟧V ρ  = ⊎.rec (λ a → ⟦ W ⟧V (ρ ▹ a)) (λ a → ⟦ W' ⟧V (ρ ▹ a)) (⟦ V ⟧V ρ)
⟦ vcase× V W ⟧V ρ     = ⟦ W ⟧V (ρ ▹ ⟦ V ⟧V ρ .fst ▹ ⟦ V ⟧V ρ .snd)

⟦ case+ V M N ⟧C ρ    = ⊎.rec (λ a → ⟦ M ⟧C (ρ ▹ a)) (λ a → ⟦ N ⟧C (ρ ▹ a)) (⟦ V ⟧V ρ)
⟦ case× V M ⟧C ρ      = ⟦ M ⟧C (ρ ▹ ⟦ V ⟧V ρ .fst ▹ ⟦ V ⟧V ρ .snd)
⟦ force V ⟧C ρ        = ⟦ V ⟧V ρ
⟦ triv ⟧C ρ           = tt
⟦ lam M ⟧C ρ          = λ a → ⟦ M ⟧C (ρ ▹ a)
⟦ app M V ⟧C ρ        = ⟦ M ⟧C ρ (⟦ V ⟧V ρ)
⟦ cpair M N ⟧C ρ      = ⟦ M ⟧C ρ , ⟦ N ⟧C ρ
⟦ π₁ M ⟧C ρ           = ⟦ M ⟧C ρ .fst
⟦ π₂ M ⟧C ρ           = ⟦ M ⟧C ρ .snd
⟦ ret V ⟧C ρ          = η (⟦ V ⟧V ρ)
⟦ bind {B = B} M N ⟧C ρ = ext ⟦ B ⟧c (λ a → ⟦ N ⟧C (ρ ▹ a)) (⟦ M ⟧C ρ)
⟦ op {B = B} o k ⟧C ρ = ⟦ B ⟧c .alg o (λ j → ⟦ k j ⟧C ρ)

⟦ • ⟧S ρ b                 = b
⟦ appˢ S V ⟧S ρ b          = ⟦ S ⟧S ρ b (⟦ V ⟧V ρ)
⟦ π₁ˢ S ⟧S ρ b             = ⟦ S ⟧S ρ b .fst
⟦ π₂ˢ S ⟧S ρ b             = ⟦ S ⟧S ρ b .snd
⟦ bindˢ {B' = B'} S N ⟧S ρ b = ext ⟦ B' ⟧c (λ a → ⟦ N ⟧C (ρ ▹ a)) (⟦ S ⟧S ρ b)
⟦ trivˢ ⟧S ρ b             = tt
⟦ lamˢ S ⟧S ρ b            = λ a → ⟦ S ⟧S (ρ ▹ a) b
⟦ pairˢ S T ⟧S ρ b         = ⟦ S ⟧S ρ b , ⟦ T ⟧S ρ b

-- The empty environment, for closed terms.
ρ∅ : Env ε
ρ∅ ()
