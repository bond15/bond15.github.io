{-
  Configuration strong normalization for every interleaving, by a
  multiset measure (Dershowitz–Manna), when no generic step copies a
  neighbor (NoCopyOps).  Copies of continuations are allowed.

    _≺_              M' ≺ M := M ↦ M', or M' is an argument of the op
                     at the head of M; well founded by induction on
                     𝒞⟦F A⟧(M) (wf≺)
    cont≺            S[Mₖ] ≺⁺ S[op(M⃗)]  (whole stack, then an argument)
    step⋖            every ↦T step is a multiset step on the subterms:
                     Pure replaces N by a reduct; Effect replaces S[op(M⃗)]
                     by continuations and keeps each neighbor at most once
    config-sn        every ↦T sequence is finite
    config-adequacy  every ↦T sequence from η(M) is finite; a configuration
                     reached from η(M) with no step is T(ret)(ṽ) with
                     T(⟦–⟧v)(ṽ) = ⟦M⟧; and one is reached
-}

------------------------------------------------------------------------
-- The order on terms

data _≺_ {A : VTy} : CComp (F A) → CComp (F A) → Type where
  red≺ : {M M' : CComp (F A)} → M ↦ M' → M' ≺ M
  arg≺ : (o : Op) (k : Fin (ar o) → CComp (F A)) (j : Fin (ar o)) → k j ≺ op o k

_≺⁺_ : CComp (F A) → CComp (F A) → Type
_≺⁺_ = TC _≺_

private
  ret≺ : {V : CVal A} {M' : CComp (F A)} → M' ≺ ret V → ⊥.⊥
  ret≺ (red≺ s) = ret-no-step s

  op≺ : {o : Op} {k : Fin (ar o) → CComp (F A)}
    → (∀ j → Acc _≺_ (k j)) → (∀ M' → op o k ↦ M' → Acc _≺_ M')
    → {M' : CComp (F A)} → M' ≺ op o k → Acc _≺_ M'
  op≺ c r (red≺ s)     = r _ s
  op≺ c r (arg≺ o k j) = c j

  neu≺ : {M M' : CComp (F A)} → Neutral M → (∀ M' → M ↦ M' → Acc _≺_ M') → M' ≺ M → Acc _≺_ M'
  neu≺ n r (red≺ s)     = r _ s
  neu≺ n r (arg≺ o k j) = ⊥.rec n

-- ≺ is well founded: the premises of 𝒞⟦F A⟧(M) are its reducts and,
-- for an operation, its arguments.
wf≺ : {M : CComp (F A)} → 𝒞⟦ F A ⟧ .S M → Acc _≺_ M
wf≺ (sret v)    = acc λ M' p → ⊥.rec (ret≺ p)
wf≺ (sop o c r) = acc λ M' p → op≺ (λ j → wf≺ (c j)) (λ M'' s → wf≺ (r M'' s)) p
wf≺ (sneu n r)  = acc λ M' p → neu≺ n (λ M'' s → wf≺ (r M'' s)) p

private
  steps≺ : {X Y Z : CComp (F A)} → X ↦* Y → Z ≺⁺ Y → Z ≺⁺ X
  steps≺ done    z = z
  steps≺ (s ∷ p) z = steps≺ p z ⊳ red≺ s

-- Effect continuations are smaller.
cont≺ : (S : CStack B (F A)) (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o))
  → plug S (k j) ≺⁺ plug S (op o k)
cont≺ S o k j = steps≺ (whole-stack S o k) [ arg≺ o (λ j → plug S (k j)) j ]

------------------------------------------------------------------------
-- Every ↦T step is a multiset step on the subterms

private
  SameL : {n : ℕ} {i : Fin n} {b : ℕ} → Fin∖ n i ⊎ Fin b → Fin∖ n i ⊎ Fin b → Type
  SameL (inl p) (inl p') = p .fst ≡ p' .fst
  SameL (inl _) (inr _)  = ⊥.⊥
  SameL (inr _) _        = ⊥.⊥

  same-pos : NoCopyOps → ∀ s i o (q q' : Fin (poly .P (s ∘[ i ] o)))
    → SameL (ρ s i o q) (ρ s i o q') → q ≡ q'
  same-pos nc s i o q q' = go (ρ s i o q) (ρ s i o q') refl refl
    where
    go : (v v' : Fin∖ (poly .P s) i ⊎ Fin (ar o)) → ρ s i o q ≡ v → ρ s i o q' ≡ v' → SameL v v' → q ≡ q'
    go (inl p) (inl p') x x' e = nc s i o q q' p x (x' ∙ cong inlP (Σ≡Prop (λ _ → isProp¬ _) (sym e)))
      where
      inlP : Fin∖ (poly .P s) i → Fin∖ (poly .P s) i ⊎ Fin (ar o)
      inlP = inl
    go (inl _) (inr _) _ _ ()
    go (inr _) _       _ _ ()

fam : Config A → Fam (_≺⁺_ {A})
fam (s , γ) = poly .P s , γ

step⋖ : NoCopyOps → {t t' : Config A} → t ↦T t' → _⋖_ _≺⁺_ (fam t') (fam t)
step⋖ nc (pure (s , i , γ) {M} {M'} st) = mk⋖ i srcP okP injP
  where
  srcP : Fin (poly .P s) → Fin∖ (poly .P s) i ⊎ Unit
  srcP j with discreteFin j i
  ... | yes _ = inr tt
  ... | no ne = inl (j , ne)
  okP : ∀ j → Ok _≺⁺_ (poly .P s , fillAt i γ M) i (fillAt i γ M' j) (srcP j)
  okP j with discreteFin j i
  ... | yes _  = subst (M' ≺⁺_) (sym (fill-hole i γ M)) [ red≺ st ]
  ... | no ne  = sym (fill-other i γ M j ne)
  injP : ∀ j j' → Same (srcP j) (srcP j') → j ≡ j'
  injP j j' e with discreteFin j i | discreteFin j' i
  ... | yes _ | _     = ⊥.rec e
  ... | no _  | yes _ = ⊥.rec e
  ... | no _  | no _  = e
step⋖ nc (effect (s , i , γ) S o k) =
  subst (λ z → _⋖_ _≺⁺_ (fam z) (fam ((s , i , γ) ⟪ x ⟫))) (sym path) explicit
  where
  x : CComp (F _)
  x = plug S (op o k)
  k' : Fin (ar o) → CComp (F _)
  k' j = plug S (k j)
  path : μ ((∂map η (s , i , γ)) ⟪ ops o (λ j → η (plug S (k j))) ⟫) ≡ pmap (⊎.rec γ k') (step s i o)
  path = effect-via-step s i γ S o k ∙ map-poly _ (step s i o)
  mapv : Fin∖ (poly .P s) i ⊎ Fin (ar o) → Fin∖ (poly .P s) i ⊎ Unit
  mapv = ⊎.map (λ p → p) (λ _ → tt)
  okE : (v : Fin∖ (poly .P s) i ⊎ Fin (ar o))
    → Ok _≺⁺_ (poly .P s , fillAt i γ x) i (⊎.rec γ k' v) (mapv v)
  okE (inl p) = sym (fill-other i γ x (p .fst) (p .snd))
  okE (inr j) = subst (k' j ≺⁺_) (sym (fill-hole i γ x)) (cont≺ S o k j)
  sameL : (v v' : Fin∖ (poly .P s) i ⊎ Fin (ar o)) → Same (mapv v) (mapv v') → SameL v v'
  sameL (inl _) (inl _) e = e
  sameL (inl _) (inr _) ()
  sameL (inr _) _       ()
  explicit : _⋖_ _≺⁺_ (fam (pmap (⊎.rec γ k') (step s i o))) (poly .P s , fillAt i γ x)
  explicit = mk⋖ i (λ q → mapv (ρ s i o q)) (λ q → okE (ρ s i o q))
    (λ q q' e → same-pos nc s i o q q' (sameL (ρ s i o q) (ρ s i o q') e))

------------------------------------------------------------------------
-- Configuration strong normalization

_↤T_ : Config A → Config A → Type
t' ↤T t = t ↦T t'

-- SN t := every ↦T sequence from t is finite.
SN-T : Config A → Type
SN-T = Acc _↤T_

config-sn : NoCopyOps → (t : Config A) → SN-T t
config-sn nc t = go t (dm _≺⁺_ _ (t .snd) (λ p → acc-plus (wf≺ (related (t .snd p)))))
  where
  go : (t : Config _) → Acc (_⋖_ _≺⁺_) (fam t) → SN-T t
  go t (acc r) = acc λ t' s → go t' (r (fam t') (step⋖ nc s))

------------------------------------------------------------------------
-- Configuration adequacy

reach-final : NoCopyOps → (M : CComp (F A))
  → Σ[ ṽ ∈ T (CVal A) ] (η M ↦T* pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M)
reach-final nc M = go (η M) done (config-sn nc (η M))
  where
  go : (t : Config _) → η M ↦T* t → SN-T t
    → Σ[ ṽ ∈ T (CVal _) ] (η M ↦T* pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M)
  go t p (acc r) with progress-T t
  ... | inl (t' , s) = go t' (p ++T* (s ∷ done)) (r t' s)
  ... | inr (ṽ , e)  =
    ṽ , subst (η M ↦T*_) e p
      , sym (den-final ṽ) ∙ cong ⟦_⟧T (sym e) ∙ sym (sound-T* p) ∙ sound-init M

config-adequacy : NoCopyOps → (M : CComp (F A))
  → SN-T (η M)
  × (∀ t → η M ↦T* t → (∀ t' → ¬ t ↦T t')
       → Σ[ ṽ ∈ T (CVal A) ] (t ≡ pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M))
  × (Σ[ ṽ ∈ T (CVal A) ] (η M ↦T* pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M))
config-adequacy nc M =
  config-sn nc (η M) , (λ t p stuck → stuck-final M p stuck) , reach-final nc M
