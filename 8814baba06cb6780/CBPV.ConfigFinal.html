{-
  Final configurations T(ret)(ṽ).

    den-final     ⟦T(ret)(ṽ)⟧ = T(⟦–⟧v)(ṽ)
    stuck-final   a configuration reached from η(M) with no step is
                  T(ret)(ṽ) with T(⟦–⟧v)(ṽ) = ⟦M⟧ (by configuration progress)
    uniqueness    for A built from 0, 1, +, ×, the final configuration
                  reached from η(M) is unique (by soundness: T(⟦–⟧v) is
                  injective there)
-}
open import CBPV.Theory
open import CBPV.OperationalModel

module CBPV.ConfigFinal (𝒯 : Theory) (om : OperationalModel 𝒯) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Relation.Nullary using (¬_)

open Theory 𝒯
open OperationalModel om

open import CBPV.Polynomial using (pmap)
open import CBPV.Syntax sig
open import CBPV.Reduction sig using (CVal; CComp)
open import CBPV.Denotation 𝒯 free
open import CBPV.ConfigReduction 𝒯 om
open import CBPV.ConfigProgress 𝒯 om using (progress-T)

private variable
  A : VTy

-- ⟦T(ret)(ṽ)⟧ = T(⟦–⟧v)(ṽ)
den-final : (ṽ : T (CVal A)) → ⟦ pmap ret ṽ ⟧T ≡ pmap (λ V → ⟦ V ⟧V ρ∅) ṽ
den-final ṽ =
  cong (μ ∘ map den-c) (sym (map-poly ret ṽ))
  ∙ cong μ (sym (map-∘ den-c ret ṽ))
  ∙ cong μ (map-∘ η (λ V → ⟦ V ⟧V ρ∅) ṽ)
  ∙ μ-mapη _
  ∙ map-poly _ ṽ

stuck-final : (M : CComp (F A)) {t : Config A} → η M ↦T* t → (∀ t' → ¬ t ↦T t')
  → Σ[ ṽ ∈ T (CVal A) ] (t ≡ pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M)
stuck-final M {t} p stuck with progress-T t
... | inl (t' , s) = ⊥.rec (stuck t' s)
... | inr (ṽ , e)  =
  ṽ , e , sym (den-final ṽ) ∙ cong ⟦_⟧T (sym e) ∙ sym (sound-T* p) ∙ sound-init M

-- T(⟦–⟧v) is injective when ⟦–⟧v has a left inverse r.
T⟦⟧v-inj : (r : ⟦ A ⟧v → CVal A) → (∀ V → r (⟦ V ⟧V ρ∅) ≡ V)
  → (ṽ₁ ṽ₂ : T (CVal A))
  → pmap (λ V → ⟦ V ⟧V ρ∅) ṽ₁ ≡ pmap (λ V → ⟦ V ⟧V ρ∅) ṽ₂ → ṽ₁ ≡ ṽ₂
T⟦⟧v-inj r retr ṽ₁ ṽ₂ same = sym (readback ṽ₁) ∙ cong (pmap r) same ∙ readback ṽ₂
  where
  readback : (ṽ : T (CVal _)) → pmap r (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ) ≡ ṽ
  readback (s , γ) = ΣPathP (refl , funExt λ p → retr (γ p))

final-unique : (r : ⟦ A ⟧v → CVal A) → (∀ V → r (⟦ V ⟧V ρ∅) ≡ V)
  → (M : CComp (F A)) (ṽ₁ ṽ₂ : T (CVal A))
  → η M ↦T* pmap ret ṽ₁ → η M ↦T* pmap ret ṽ₂ → ṽ₁ ≡ ṽ₂
final-unique r retr M ṽ₁ ṽ₂ p₁ p₂ =
  T⟦⟧v-inj r retr ṽ₁ ṽ₂ (sym (den-final ṽ₁) ∙ sym (sound-T* p₁) ∙ sound-T* p₂ ∙ den-final ṽ₂)

-- Ground types: built from 0, 1, +, ×.
data Ground : VTy → Type where
  g𝟘 : Ground 𝟘
  g𝟙 : Ground 𝟙
  g⊕ : {A A' : VTy} → Ground A → Ground A' → Ground (A ⊕ A')
  g⊗ : {A A' : VTy} → Ground A → Ground A' → Ground (A ⊗ A')

reflect : Ground A → ⟦ A ⟧v → CVal A
reflect g𝟘 ()
reflect g𝟙 _ = ⋆
reflect (g⊕ g g') (inl a) = inl (reflect g a)
reflect (g⊕ g g') (inr a) = inr (reflect g' a)
reflect (g⊗ g g') (a , a') = vpair (reflect g a) (reflect g' a')

reflect-den : (g : Ground A) (V : CVal A) → reflect g (⟦ V ⟧V ρ∅) ≡ V
reflect-den g (var ())
reflect-den g (absurd W) = ⊥.rec (⟦ W ⟧V ρ∅)
reflect-den g𝟙 ⋆ = refl
reflect-den (g⊕ g g') (inl V) = cong inl (reflect-den g V)
reflect-den (g⊕ g g') (inr V) = cong inr (reflect-den g' V)
reflect-den (g⊗ g g') (vpair V W) = cong₂ vpair (reflect-den g V) (reflect-den g' W)

-- Uniqueness at ground types.
uniqueness : Ground A → (M : CComp (F A)) {ṽ₁ ṽ₂ : T (CVal A)}
  → η M ↦T* pmap ret ṽ₁ → η M ↦T* pmap ret ṽ₂ → ṽ₁ ≡ ṽ₂
uniqueness g M {ṽ₁} {ṽ₂} = final-unique (reflect g) (reflect-den g) M ṽ₁ ṽ₂
