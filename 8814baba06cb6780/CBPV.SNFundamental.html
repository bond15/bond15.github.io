{-
  Strong normalization, via the reducibility candidates of
  CBPV.SNRelation.

  Fundamental Lemma for 𝒞: every well-typed term, under related closing
  substitutions, is in its candidate.  Each elimination of an
  introduction form is neutral and has exactly one reduct, its β-reduct,
  so it is in the candidate by CR3.  Bind is by induction on the
  derivation of 𝒞⟦F A⟧(M).

  Corollary: every closed computation is strongly normalizing (CR1).
-}
open import CBPV.Theory

module CBPV.SNFundamental (Sg : Signature) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Unit
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Sum using (_⊎_; inl; inr)
open import Cubical.Data.Sigma
open import Cubical.Data.FinData using (Fin)
open import Cubical.Induction.WellFounded

open Signature Sg
open import CBPV.Syntax Sg
open import CBPV.Substitution Sg
open import CBPV.Reduction Sg
open import CBPV.SNRelation Sg

private variable
  Γ : Ctx
  A A' : VTy
  B B₁ B₂ : CTy

------------------------------------------------------------------------
-- The only reduct of a β-redex is its β-reduct (introduction forms and
-- values do not step, and there is no congruence under force or case).

force-thunk-step : {M X : CComp B} → force (thunk M) ↦ X → X ≡ M
force-thunk-step (red (β (force-thunk _))) = refl

case+-inl-step : {V : CVal A} {M₁ : Comp (ε ▸ A) B} {M₂ : Comp (ε ▸ A') B} {X : CComp B}
  → case+ (inl V) M₁ M₂ ↦ X → X ≡ M₁ [ V ]₀
case+-inl-step (red (β (case+-inl _ _ _))) = refl

case+-inr-step : {V : CVal A'} {M₁ : Comp (ε ▸ A) B} {M₂ : Comp (ε ▸ A') B} {X : CComp B}
  → case+ (inr V) M₁ M₂ ↦ X → X ≡ M₂ [ V ]₀
case+-inr-step (red (β (case+-inr _ _ _))) = refl

case×-step : {V : CVal A} {W : CVal A'} {M : Comp (ε ▸ A ▸ A') B} {X : CComp B}
  → case× (vpair V W) M ↦ X → X ≡ M [ V , W ]₁
case×-step (red (β (case×-pair _ _ _))) = refl

app-lam-step : {M : Comp (ε ▸ A) B} {V : CVal A} {X : CComp B}
  → app (lam M) V ↦ X → X ≡ M [ V ]₀
app-lam-step (red (β (app-lam _ _))) = refl
app-lam-step (appF _ s)               = ⊥.rec (lam-no-step s)

π₁-pair-step : {M : CComp B₁} {N : CComp B₂} {X : CComp B₁} → π₁ (cpair M N) ↦ X → X ≡ M
π₁-pair-step (red (β (π₁-pair _ _))) = refl
π₁-pair-step (π₁F s)                 = ⊥.rec (cpair-no-step s)

π₂-pair-step : {M : CComp B₁} {N : CComp B₂} {X : CComp B₂} → π₂ (cpair M N) ↦ X → X ≡ N
π₂-pair-step (red (β (π₂-pair _ _))) = refl
π₂-pair-step (π₂F s)                 = ⊥.rec (cpair-no-step s)

bind-ret-step : {V : CVal A} {N : Comp (ε ▸ A) B} {X : CComp B}
  → bind (ret V) N ↦ X → X ≡ N [ V ]₀
bind-ret-step (red (β (bind-ret _ _))) = refl
bind-ret-step (bindF _ s)               = ⊥.rec (ret-no-step s)

-- A step of bind (op o k) N moves the operation out, or is a step of op o k.
bind-op-step : {o : Op} {k : Fin (ar o) → CComp (F A)} {N : Comp (ε ▸ A) B} {X : CComp B}
  → bind (op o k) N ↦ X
  → (X ≡ op o (λ j → bind (k j) N)) ⊎ (Σ[ M' ∈ CComp (F A) ] (op o k ↦ M') × (X ≡ bind M' N))
bind-op-step (red (alg-bind _ _ _)) = inl refl
bind-op-step (bindF _ s)            = inr (_ , s , refl)

-- Introduction forms are strongly normalizing (they are normal).
canonical-sn : {M : CComp B} → Canonical M → SN M
canonical-sn c = acc λ _ s → ⊥.rec (canonical-no-step c s)

------------------------------------------------------------------------
-- Semantic typing for 𝒞

𝒢⟦_⟧ : (Γ : Ctx) → Subst ε Γ → Type
𝒢⟦ Γ ⟧ γ = ∀ {A} (x : Var Γ A) → 𝒱⟦ A ⟧ (γ x)

𝒢-ext : {γ : Subst ε Γ} {V : CVal A}
  → 𝒢⟦ Γ ⟧ γ → 𝒱⟦ A ⟧ V → 𝒢⟦ Γ ▸ A ⟧ (γ ,ₛ V)
𝒢-ext g v vz     = v
𝒢-ext g v (vs x) = g x

_⊨v_ : (Γ : Ctx) → Val Γ A → Type
_⊨v_ {A = A} Γ V = (γ : Subst ε Γ) → 𝒢⟦ Γ ⟧ γ → 𝒱⟦ A ⟧ (subV γ V)

_⊨c_ : (Γ : Ctx) → Comp Γ B → Type
_⊨c_ {B = B} Γ M = (γ : Subst ε Γ) → 𝒢⟦ Γ ⟧ γ → 𝒞⟦ B ⟧ .S (subC γ M)

𝒞-≡ : (B : CTy) {M M' : CComp B} → M ≡ M' → 𝒞⟦ B ⟧ .S M' → 𝒞⟦ B ⟧ .S M
𝒞-≡ B p = subst (𝒞⟦ B ⟧ .S) (sym p)

-- A neutral term whose only reduct is in the candidate (CR3).
one-reduct : (B : CTy) {M R : CComp B} → Neutral M
  → (∀ {X} → M ↦ X → X ≡ R) → 𝒞⟦ B ⟧ .S R → 𝒞⟦ B ⟧ .S M
one-reduct B n only r = 𝒞⟦ B ⟧ .neutral-closed n λ X s → 𝒞-≡ B (only s) r

------------------------------------------------------------------------
-- Eliminating related closed sums and pairs

fund-case+ : (B : CTy) (W : CVal (A ⊕ A')) → 𝒱⟦ A ⊕ A' ⟧ W
  → (M₁ : Comp (ε ▸ A) B) (M₂ : Comp (ε ▸ A') B)
  → (∀ Y → 𝒱⟦ A ⟧ Y → 𝒞⟦ B ⟧ .S (M₁ [ Y ]₀))
  → (∀ Y → 𝒱⟦ A' ⟧ Y → 𝒞⟦ B ⟧ .S (M₂ [ Y ]₀))
  → 𝒞⟦ B ⟧ .S (case+ W M₁ M₂)
fund-case+ B (inl Y) w M₁ M₂ h₁ h₂ = one-reduct B tt case+-inl-step (h₁ Y w)
fund-case+ B (inr Y) w M₁ M₂ h₁ h₂ = one-reduct B tt case+-inr-step (h₂ Y w)
fund-case+ B (absurd _) () M₁ M₂ h₁ h₂

fund-case× : (B : CTy) (W : CVal (A ⊗ A')) → 𝒱⟦ A ⊗ A' ⟧ W
  → (M : Comp (ε ▸ A ▸ A') B)
  → (∀ Y Y' → 𝒱⟦ A ⟧ Y → 𝒱⟦ A' ⟧ Y' → 𝒞⟦ B ⟧ .S (M [ Y , Y' ]₁))
  → 𝒞⟦ B ⟧ .S (case× W M)
fund-case× B (vpair Y Y') (u , u') M h = one-reduct B tt case×-step (h Y Y' u u')
fund-case× B (absurd _) () M h

------------------------------------------------------------------------
-- The bind case, by induction on the derivation of 𝒞⟦F A⟧(M).
-- bind M N is neutral; its reducts are, for each shape of M:
--   ret V:    the β-reduct N[V/x];
--   op o k:   op o (λ j → bind (k j) N) by moving the operation out (CR4 at B and the
--             induction hypothesis on the arguments), or bind M' N for a
--             step of op o k (induction hypothesis on the reduct);
--   neutral:  bind M' N for a step of M (Lemma 0, induction hypothesis).

fund-bind : (B : CTy) (N : Comp (ε ▸ A) B)
  → (∀ V → 𝒱⟦ A ⟧ V → 𝒞⟦ B ⟧ .S (N [ V ]₀))
  → {M : CComp (F A)} → 𝒞⟦ F A ⟧ .S M → 𝒞⟦ B ⟧ .S (bind M N)
fund-bind B N hN (sret {V = V} v) = one-reduct B tt bind-ret-step (hN V v)
fund-bind B N hN (sop o {k} c r)  = 𝒞⟦ B ⟧ .neutral-closed tt λ X s → go (bind-op-step s)
  where
  go : {X : CComp _}
     → (X ≡ op o (λ j → bind (k j) N)) ⊎ (Σ[ M' ∈ CComp _ ] (op o k ↦ M') × (X ≡ bind M' N))
     → 𝒞⟦ B ⟧ .S X
  go (inl p)            = 𝒞-≡ B p (𝒞⟦ B ⟧ .op-closed o (λ j → fund-bind B N hN (c j)))
  go (inr (M' , s , p)) = 𝒞-≡ B p (fund-bind B N hN (r M' s))
fund-bind B N hN (sneu n r)       = 𝒞⟦ B ⟧ .neutral-closed tt λ X s →
  let (M' , s' , p) = bind-neutral-step n s in 𝒞-≡ B p (fund-bind B N hN (r M' s'))

------------------------------------------------------------------------
-- Fundamental Lemma for 𝒞

fundV : (V : Val Γ A) → Γ ⊨v V
fundC : (M : Comp Γ B) → Γ ⊨c M

fundV (var x)     γ g = g x
fundV (absurd V)  γ g = ⊥.rec (fundV V γ g)
fundV ⋆           γ g = tt
fundV (inl V)     γ g = fundV V γ g
fundV (inr V)     γ g = fundV V γ g
fundV (vpair V W) γ g = fundV V γ g , fundV W γ g
fundV (thunk {B = B} M) γ g = one-reduct B tt force-thunk-step (fundC M γ g)

fundC (case+ {B = B} V M₁ M₂) γ g =
  fund-case+ B (subV γ V) (fundV V γ g) _ _
    (λ Y u → 𝒞-≡ B (sub₀-lift γ M₁ Y) (fundC M₁ (γ ,ₛ Y) (𝒢-ext g u)))
    (λ Y u → 𝒞-≡ B (sub₀-lift γ M₂ Y) (fundC M₂ (γ ,ₛ Y) (𝒢-ext g u)))
fundC (case× {B = B} V M) γ g =
  fund-case× B (subV γ V) (fundV V γ g) _
    (λ Y Y' u u' → 𝒞-≡ B (sub₁-lift γ M Y Y') (fundC M (γ ,ₛ Y ,ₛ Y') (𝒢-ext (𝒢-ext g u) u')))
fundC (force V)   γ g = fundV V γ g
fundC triv        γ g = canonical-sn tt
fundC (lam {B = B} M) γ g =
  canonical-sn tt , λ Y u →
    one-reduct B tt app-lam-step (𝒞-≡ B (sub₀-lift γ M Y) (fundC M (γ ,ₛ Y) (𝒢-ext g u)))
fundC (app M V)   γ g = fundC M γ g .snd _ (fundV V γ g)
fundC (cpair {B₁ = B₁} {B₂ = B₂} M N) γ g =
  canonical-sn tt , one-reduct B₁ tt π₁-pair-step (fundC M γ g)
                  , one-reduct B₂ tt π₂-pair-step (fundC N γ g)
fundC (π₁ M)      γ g = fundC M γ g .snd .fst
fundC (π₂ M)      γ g = fundC M γ g .snd .snd
fundC (ret V)     γ g = sret (fundV V γ g)
fundC (bind {B = B} M N) γ g =
  fund-bind B _ (λ V v → 𝒞-≡ B (sub₀-lift γ N V) (fundC N (γ ,ₛ V) (𝒢-ext g v))) (fundC M γ g)
fundC (op {B = B} o k) γ g = 𝒞⟦ B ⟧ .op-closed o (λ j → fundC (k j) γ g)

-- Every closed computation is in its candidate.
related : (M : CComp B) → 𝒞⟦ B ⟧ .S M
related {B = B} M = 𝒞-≡ B (sym (sub-idC (λ _ → refl) M)) (fundC M idS (λ ()))

------------------------------------------------------------------------
-- Corollary (strong normalization): every closed computation, of any
-- type, is strongly normalizing for ↦tree.

strong-normalization : (M : CComp B) → SN M
strong-normalization {B = B} M = 𝒞⟦ B ⟧ .sn (related M)
