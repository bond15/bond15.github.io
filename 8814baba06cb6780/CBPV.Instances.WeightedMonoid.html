{-
  Weighted monoids with an operation twice.

  Theory, over a monoid of weights (R, ·, 1): nullary e, binary ⊗,
  unary act_r (r : R) and twice, with
      t ⊗ (t' ⊗ t'') = (t ⊗ t') ⊗ t''     e ⊗ t = t      t ⊗ e = t
      act₁(t) = t     act_r(act_r'(t)) = act_{r·r'}(t)
      act_r(t ⊗ t') = act_r(t) ⊗ act_r(t')     act_r(e) = e
      twice(t) = t ⊗ t
  Free model: List(R × X), with e = [], ⊗ = append, act_r scales the
  weights, twice(l) = l ++ l, η(x) = [(1, x)].  As a polynomial: shapes
  Σ n. (Fin n → R) (lists of weights), P := n.

  Monadic container: ι := [1], σ(s, v) := the concatenation of the v(p),
  each scaled by the weight of p, pr(q) := (p, m) for q the m-th position
  of the block of p.  ⌈act_r⌉ := ⟨[r], (0)⟩, ⌈⊗⌉ := ⟨[1,1], (0,1)⟩,
  ⌈twice⌉ := ⟨[1,1], (0,0)⟩: twice copies its continuation.  Affine:
  each neighbor of the hole is σ-multiplied by ι, which has one position.
-}

data WOp (R : Type) : Type where
  unit mul twice : WOp R
  act : R → WOp R

data WEqn (R : Type) : Type where
  e-assoc e-unitl e-unitr e-act1 e-twice : WEqn R
  e-actact  : R → R → WEqn R
  e-actdist : R → WEqn R
  e-acte    : R → WEqn R

module _ (R : Type) (mon : MonoidOn R) where
  open MonoidOn mon renaming (e to 1r; _⊗_ to _·_)

  sigWM : Signature
  sigWM = record { Op = WOp R ; ar = λ { unit → 0 ; mul → 2 ; twice → 1 ; (act _) → 1 } }

  private
    eT : {n : ℕ} → Term sigWM (Fin n)
    eT = op unit noArgs
    _⊗T_ : {n : ℕ} → Term sigWM (Fin n) → Term sigWM (Fin n) → Term sigWM (Fin n)
    a ⊗T b = op mul (pair2 a b)
    actT : {n : ℕ} → R → Term sigWM (Fin n) → Term sigWM (Fin n)
    actT r a = op (act r) (λ _ → a)
    twiceT : {n : ℕ} → Term sigWM (Fin n) → Term sigWM (Fin n)
    twiceT a = op twice (λ _ → a)
    x : {n : ℕ} → Fin n → Term sigWM (Fin n)
    x = var

    eqnWM : WEqn R → Equation sigWM
    eqnWM e-assoc   = record { ctx = 3 ; lhs = x zero ⊗T (x (suc zero) ⊗T x (suc (suc zero)))
                                       ; rhs = (x zero ⊗T x (suc zero)) ⊗T x (suc (suc zero)) }
    eqnWM e-unitl   = record { ctx = 1 ; lhs = eT ⊗T x zero ; rhs = x zero }
    eqnWM e-unitr   = record { ctx = 1 ; lhs = x zero ⊗T eT ; rhs = x zero }
    eqnWM e-act1    = record { ctx = 1 ; lhs = actT 1r (x zero) ; rhs = x zero }
    eqnWM (e-actact r r') = record { ctx = 1 ; lhs = actT r (actT r' (x zero)) ; rhs = actT (r · r') (x zero) }
    eqnWM (e-actdist r) = record { ctx = 2 ; lhs = actT r (x zero ⊗T x (suc zero))
                                           ; rhs = actT r (x zero) ⊗T actT r (x (suc zero)) }
    eqnWM (e-acte r)  = record { ctx = 0 ; lhs = actT r eT ; rhs = eT }
    eqnWM e-twice     = record { ctx = 1 ; lhs = twiceT (x zero) ; rhs = x zero ⊗T x zero }

  wmTheory : Theory
  wmTheory = record { sig = sigWM ; Eqn = WEqn R ; eqn = eqnWM }

  open Model
  open Equation

  ------------------------------------------------------------------------
  -- The list model

  L : Type → Type
  L X = List (R × X)

  scale : {X : Type} → R → R × X → R × X
  scale r (q , y) = r · q , y

  opsL : {X : Type} → AlgStr sigWM (L X)
  opsL unit    a = []
  opsL mul     a = a zero ++ a (suc zero)
  opsL twice   a = a zero ++ a zero
  opsL (act r) a = map (scale r) (a zero)

  private
    ++-assoc' : {A : Type} (a b c : List A) → (a ++ b) ++ c ≡ a ++ (b ++ c)
    ++-assoc' []      b c = refl
    ++-assoc' (y ∷ a) b c = cong (y ∷_) (++-assoc' a b c)

    ++-unit' : {A : Type} (a : List A) → a ++ [] ≡ a
    ++-unit' []      = refl
    ++-unit' (y ∷ a) = cong (y ∷_) (++-unit' a)

    map-++' : {A B : Type} (f : A → B) (a b : List A) → map f (a ++ b) ≡ map f a ++ map f b
    map-++' f []      b = refl
    map-++' f (y ∷ a) b = cong (f y ∷_) (map-++' f a b)

    map-scale1 : {X : Type} (a : L X) → map (scale 1r) a ≡ a
    map-scale1 []            = refl
    map-scale1 ((q , y) ∷ a) = cong₂ _∷_ (ΣPathP (unitˡ q , refl)) (map-scale1 a)

    map-scale² : {X : Type} (r r' : R) (a : L X) → map (scale r) (map (scale r') a) ≡ map (scale (r · r')) a
    map-scale² r r' []            = refl
    map-scale² r r' ((q , y) ∷ a) = cong₂ _∷_ (ΣPathP (sym (assoc r r' q) , refl)) (map-scale² r r' a)

    eval-pair2 : {A X : Type} (α : AlgStr sigWM A) (γ : X → A) (a b : Term sigWM X)
      → (λ j → eval sigWM α γ (pair2 a b j)) ≡ pair2 (eval sigWM α γ a) (eval sigWM α γ b)
    eval-pair2 α γ a b = funExt λ { zero → refl ; (suc zero) → refl }

  satL : {X : Type} → Sat wmTheory (opsL {X})
  satL e-assoc γ = sym (++-assoc' (γ zero) (γ (suc zero)) (γ (suc (suc zero))))
  satL e-unitl γ = refl
  satL e-unitr γ = ++-unit' (γ zero)
  satL e-act1  γ = map-scale1 (γ zero)
  satL (e-actact r r') γ = map-scale² r r' (γ zero)
  satL (e-actdist r) γ = map-++' (scale r) (γ zero) (γ (suc zero))
  satL (e-acte r) γ = refl
  satL e-twice γ = refl

  ------------------------------------------------------------------------
  -- The polynomial and the bijection with lists

  wmPoly : Poly
  wmPoly = record { Sh = Σ[ n ∈ ℕ ] (Fin n → R) ; P = fst }

  T : Type → Type
  T = ⟦ wmPoly ⟧

  fromL : {X : Type} → L X → T X
  fromL []            = (0 , noArgs) , noArgs
  fromL ((r , y) ∷ l) = (suc (fromL l .fst .fst) , cons r (fromL l .fst .snd)) , cons y (fromL l .snd)

  toL' : {X : Type} (n : ℕ) → (Fin n → R) → (Fin n → X) → L X
  toL' zero    w γ = []
  toL' (suc n) w γ = (w zero , γ zero) ∷ toL' n (w ∘ suc) (γ ∘ suc)

  toL : {X : Type} → T X → L X
  toL ((n , w) , γ) = toL' n w γ

  toL-fromL : {X : Type} (l : L X) → toL (fromL l) ≡ l
  toL-fromL []            = refl
  toL-fromL ((r , y) ∷ l) = cong ((r , y) ∷_) (toL-fromL l)

  fromL-toL' : {X : Type} (n : ℕ) (w : Fin n → R) (γ : Fin n → X) → fromL (toL' n w γ) ≡ ((n , w) , γ)
  fromL-toL' zero w γ = ΣPathP (ΣPathP (refl , funExt λ ()) , funExt λ ())
  fromL-toL' {X} (suc n) w γ =
    cong g (fromL-toL' n (w ∘ suc) (γ ∘ suc))
    ∙ ΣPathP (ΣPathP (refl , cons-η w) , cons-η γ)
    where
    g : T X → T X
    g u = (suc (u .fst .fst) , cons (w zero) (u .fst .snd)) , cons (γ zero) (u .snd)

  fromL-toL : {X : Type} (t : T X) → fromL (toL t) ≡ t
  fromL-toL ((n , w) , γ) = fromL-toL' n w γ

  ------------------------------------------------------------------------
  -- The free model on T

  ηW : {X : Type} → X → T X
  ηW y = fromL ((1r , y) ∷ [])

  opsW : {X : Type} → AlgStr sigWM (T X)
  opsW o args = fromL (opsL o (λ j → toL (args j)))

  private
    eval-T : {X Y : Type} (γ : Y → T X) (t : Term sigWM Y)
      → toL (eval sigWM opsW γ t) ≡ eval sigWM opsL (toL ∘ γ) t
    eval-T γ (var y)   = refl
    eval-T γ (op o ts) =
      toL-fromL _ ∙ cong (opsL o) (funExt λ j → eval-T γ (ts j))

    satW : {X : Type} → Sat wmTheory (opsW {X})
    satW e γ =
      sym (fromL-toL _)
      ∙ cong fromL (eval-T γ (lhs (eqnWM e)) ∙ satL e (toL ∘ γ) ∙ sym (eval-T γ (rhs (eqnWM e))))
      ∙ fromL-toL _

  module NLaws (N : Model wmTheory) where
    Nunit : N .Carrier
    Nunit = N .alg unit noArgs
    Nmul : N .Carrier → N .Carrier → N .Carrier
    Nmul a b = N .alg mul (pair2 a b)
    Nact : R → N .Carrier → N .Carrier
    Nact r a = N .alg (act r) (λ _ → a)

    private
      law : (e : WEqn R) (γ : Fin (ctx (eqnWM e)) → N .Carrier)
        → eval sigWM (N .alg) γ (lhs (eqnWM e)) ≡ eval sigWM (N .alg) γ (rhs (eqnWM e))
      law = N .sat

    N-assoc : ∀ a b c → Nmul a (Nmul b c) ≡ Nmul (Nmul a b) c
    N-assoc a b c =
      cong (N .alg mul) (fin2-ext _ ∙ cong (pair2 a) (cong (N .alg mul) (sym (fin2-ext _))))
      ∙ sym (cong (N .alg mul) (eval-pair2 _ γ _ _)) ∙ law e-assoc γ ∙ cong (N .alg mul) (eval-pair2 _ γ _ _)
      ∙ cong (N .alg mul) (cong (λ z → pair2 z c) (cong (N .alg mul) (eval-pair2 _ γ _ _)))
      where
      γ : Fin 3 → N .Carrier
      γ zero = a
      γ (suc zero) = b
      γ (suc (suc zero)) = c

    unit-eval : {n : ℕ} (γ : Fin n → N .Carrier) → Nunit ≡ eval sigWM (N .alg) γ eT
    unit-eval γ = cong (N .alg unit) (funExt λ ())

    N-unitl : ∀ a → Nmul Nunit a ≡ a
    N-unitl a = cong (λ z → Nmul z a) (unit-eval (λ _ → a))
              ∙ cong (N .alg mul) (sym (eval-pair2 _ (λ _ → a) _ _)) ∙ law e-unitl (λ _ → a)

    N-unitr : ∀ a → Nmul a Nunit ≡ a
    N-unitr a = cong (Nmul a) (unit-eval (λ _ → a))
              ∙ cong (N .alg mul) (sym (eval-pair2 _ (λ _ → a) _ _)) ∙ law e-unitr (λ _ → a)

    N-act1 : ∀ a → Nact 1r a ≡ a
    N-act1 a = law e-act1 (λ _ → a)

    N-actact : ∀ r r' a → Nact r (Nact r' a) ≡ Nact (r · r') a
    N-actact r r' a = law (e-actact r r') (λ _ → a)

    N-actdist : ∀ r a b → Nact r (Nmul a b) ≡ Nmul (Nact r a) (Nact r b)
    N-actdist r a b =
      cong (Nact r) (cong (N .alg mul) (sym (eval-pair2 _ γ _ _)))
      ∙ law (e-actdist r) γ
      ∙ cong (N .alg mul) (eval-pair2 _ γ _ _)
      where
      γ : Fin 2 → N .Carrier
      γ = pair2 a b

    N-acte : ∀ r → Nact r Nunit ≡ Nunit
    N-acte r = cong (Nact r) (unit-eval noArgs) ∙ law (e-acte r) noArgs ∙ sym (unit-eval noArgs)

    N-twice : ∀ a → N .alg twice (λ _ → a) ≡ Nmul a a
    N-twice a = law e-twice (λ _ → a) ∙ cong (N .alg mul) (eval-pair2 _ (λ _ → a) _ _)

  -- !_f on lists
  extL : {X : Type} (N : Model wmTheory) → (X → N .Carrier) → L X → N .Carrier
  extL N f []            = NLaws.Nunit N
  extL N f ((r , y) ∷ l) = NLaws.Nmul N (NLaws.Nact N r (f y)) (extL N f l)

  extW : {X : Type} (N : Model wmTheory) → (X → N .Carrier) → T X → N .Carrier
  extW N f t = extL N f (toL t)

  private
    module _ {X : Type} (N : Model wmTheory) (f : X → N .Carrier) where
      open NLaws N

      extL-++ : (a b : L X) → extL N f (a ++ b) ≡ Nmul (extL N f a) (extL N f b)
      extL-++ []            b = sym (N-unitl _)
      extL-++ ((r , y) ∷ a) b = cong (Nmul _) (extL-++ a b) ∙ N-assoc _ _ _

      extL-map : (r : R) (a : L X) → extL N f (map (scale r) a) ≡ Nact r (extL N f a)
      extL-map r []            = sym (N-acte r)
      extL-map r ((q , y) ∷ a) =
        cong₂ Nmul (sym (N-actact r q (f y))) (extL-map r a) ∙ sym (N-actdist r _ _)

      homW : IsHom sigWM opsW (N .alg) (extW N f)
      homW .pres unit    args = cong (N .alg unit) (funExt λ ())
      homW .pres mul     args =
        cong (extL N f) (toL-fromL (toL (args zero) ++ toL (args (suc zero))))
        ∙ extL-++ (toL (args zero)) (toL (args (suc zero)))
        ∙ cong (N .alg mul) (sym (fin2-ext (λ j → extW N f (args j))))
      homW .pres twice   args =
        cong (extL N f) (toL-fromL (toL (args zero) ++ toL (args zero)))
        ∙ extL-++ (toL (args zero)) (toL (args zero))
        ∙ sym (N-twice _)
        ∙ cong (N .alg twice) (sym (fin1-const (λ j → extW N f (args j))))
      homW .pres (act r) args =
        cong (extL N f) (toL-fromL (map (scale r) (toL (args zero))))
        ∙ extL-map r (toL (args zero))
        ∙ cong (N .alg (act r)) (sym (fin1-const (λ j → extW N f (args j))))

      η-law : (y : X) → extW N f (ηW y) ≡ f y
      η-law y = N-unitr _ ∙ N-act1 _

      uniq : (h : T X → N .Carrier) → IsHom sigWM opsW (N .alg) h
        → (∀ y → h (ηW y) ≡ f y) → ∀ t → h t ≡ extW N f t
      uniq h hh hη t = cong h (sym (fromL-toL t)) ∙ onL (toL t)
        where
        onL : (l : L X) → h (fromL l) ≡ extL N f l
        onL [] = hh .pres unit noArgs ∙ cong (N .alg unit) (funExt λ ())
        onL ((r , y) ∷ l) =
          cong h (cong fromL (cong₂ _∷_ (ΣPathP (sym (unitʳ r) , refl)) (sym (toL-fromL l))
                    ∙ cong (_++ toL (fromL l)) (sym (toL-fromL _))))
          ∙ hh .pres mul (pair2 (opsW (act r) (λ _ → ηW y)) (fromL l))
          ∙ cong (N .alg mul) (fin2-ext _)
          ∙ cong₂ Nmul
              (hh .pres (act r) (λ _ → ηW y) ∙ cong (Nact r) (hη y))
              (onL l)

  wmFree : FreeModel wmTheory T
  wmFree = record
    { η        = ηW
    ; ops      = opsW
    ; ops-sat  = satW
    ; ext      = extW
    ; ext-η    = λ N f → η-law N f
    ; ext-hom  = homW
    ; ext-uniq = λ N f → uniq N f
    }

  ------------------------------------------------------------------------
  -- The polynomial action is the list map

  private
    second : {X Y : Type} → (X → Y) → R × X → R × Y
    second f (q , y) = q , f y

    pmap-fromL : {X Y : Type} (f : X → Y) (l : L X) → pmap {p = wmPoly} f (fromL l) ≡ fromL (map (second f) l)
    pmap-fromL f [] = ΣPathP (refl , funExt λ ())
    pmap-fromL {Y = Y} f ((r , y) ∷ l) =
      ΣPathP (refl , funExt λ { zero → refl ; (suc j) → refl }) ∙ cong g (pmap-fromL f l)
      where
      g : T Y → T Y
      g u = (suc (u .fst .fst) , cons r (u .fst .snd)) , cons (f y) (u .snd)

    toL-pmap : {X Y : Type} (f : X → Y) (t : T X) → toL (pmap {p = wmPoly} f t) ≡ map (second f) (toL t)
    toL-pmap f ((n , w) , γ) = go n w γ
      where
      go : (n : ℕ) (w : Fin n → R) (γ : Fin n → _) → toL' n w (f ∘ γ) ≡ map (second f) (toL' n w γ)
      go zero    w γ = refl
      go (suc n) w γ = cong (_ ∷_) (go n (w ∘ suc) (γ ∘ suc))

    map-second-scale : {X Y : Type} (f : X → Y) (r : R) (a : L X)
      → map (second f) (map (scale r) a) ≡ map (scale r) (map (second f) a)
    map-second-scale f r []            = refl
    map-second-scale f r ((q , y) ∷ a) = cong (_ ∷_) (map-second-scale f r a)

    opsL-nat : {X Y : Type} (f : X → Y) (o : WOp R) (a : Fin (Signature.ar sigWM o) → L X)
      → map (second f) (opsL o a) ≡ opsL o (map (second f) ∘ a)
    opsL-nat f unit    a = refl
    opsL-nat f mul     a = map-++' (second f) (a zero) (a (suc zero))
    opsL-nat f twice   a = map-++' (second f) (a zero) (a zero)
    opsL-nat f (act r) a = map-second-scale f r (a zero)

  wmOM : OperationalModel wmTheory
  wmOM = mkOperationalModel wmTheory wmPoly wmFree
    (λ f → ishom λ o args →
      pmap-fromL f _
      ∙ cong fromL (opsL-nat f o _ ∙ cong (opsL o) (funExt λ j → sym (toL-pmap f (args j)))))
    (λ f y → ΣPathP (refl , funExt λ { zero → refl }))

  ------------------------------------------------------------------------
  -- The monadic container

  -- pr: the position q of a concatenation of blocks of sizes m p is the
  -- m-th position of block p.
  prW : (n : ℕ) (m : Fin n → ℕ) → Fin (sumFin n m) → Σ[ p ∈ Fin n ] Fin (m p)
  prW zero    m ()
  prW (suc n) m q =
    ⊎.rec (λ x → zero , x) (λ y → suc (prW n (m ∘ suc) y .fst) , prW n (m ∘ suc) y .snd)
      (splitFin {m = m zero} q)

  wmC : MCOps wmPoly
  wmC = record
    { ι  = 1 , λ _ → 1r
    ; σ  = λ { (n , w) v →
               sumFin n (λ p → v p .fst)
             , λ q → w (prW n (λ p → v p .fst) q .fst)
                     · v (prW n (λ p → v p .fst) q .fst) .snd (prW n (λ p → v p .fst) q .snd) }
    ; pr = λ { (n , w) v q → prW n (λ p → v p .fst) q } }

  -- ⌈e⌉ := ⟨[]⟩, ⌈⊗⌉ := ⟨[1,1], (0,1)⟩, ⌈twice⌉ := ⟨[1,1], (0,0)⟩, ⌈act_r⌉ := ⟨[r], (0)⟩
  wmGen : (o : WOp R) → T (Fin (Signature.ar sigWM o))
  wmGen unit    = (0 , noArgs) , noArgs
  wmGen mul     = (2 , cons 1r (cons 1r noArgs)) , cons zero (cons (suc zero) noArgs)
  wmGen twice   = (2 , cons 1r (cons 1r noArgs)) , cons zero (cons zero noArgs)
  wmGen (act r) = (1 , λ _ → r) , λ _ → zero

  private
    open FreeModel wmFree using (μ)

    toL'-+ : {X : Type} (a b : ℕ) (w : Fin (a + b) → R) (γ : Fin (a + b) → X)
      → toL' (a + b) w γ ≡ toL' a (w ∘ finL) (γ ∘ finL) ++ toL' b (w ∘ finR {m = a}) (γ ∘ finR {m = a})
    toL'-+ zero    b w γ = refl
    toL'-+ (suc a) b w γ = cong (_ ∷_) (toL'-+ a b (w ∘ suc) (γ ∘ suc))

    toL'-scale : {X : Type} (n : ℕ) (r : R) (u : Fin n → R) (δ : Fin n → X)
      → toL' n (λ x → r · u x) δ ≡ map (scale r) (toL' n u δ)
    toL'-scale zero    r u δ = refl
    toL'-scale (suc n) r u δ = cong (_ ∷_) (toL'-scale n r (u ∘ suc) (δ ∘ suc))

    μᶜ-step : {X : Type} (n : ℕ) (w : Fin (suc n) → R) (γ : Fin (suc n) → T X)
      → toL (μᶜ wmC ((suc n , w) , γ))
        ≡ map (scale (w zero)) (toL (γ zero)) ++ toL (μᶜ wmC ((n , w ∘ suc) , γ ∘ suc))
    μᶜ-step {X} n w γ =
      toL'-+ m₀ _ W Γ
      ∙ cong₂ _++_
          (cong₂ (toL' m₀) (funExt λ x → cong fW (pL x)) (funExt λ x → cong fΓ (pL x))
           ∙ toL'-scale m₀ (w zero) (γ zero .fst .snd) (γ zero .snd))
          (cong₂ (toL' _) (funExt λ y → cong fW (pR y)) (funExt λ y → cong fΓ (pR y)))
      where
      m : Fin (suc n) → ℕ
      m p = γ p .fst .fst
      m₀ = m zero
      Q = Σ[ p ∈ Fin (suc n) ] Fin (m p)
      fW : Q → R
      fW (p , r) = w p · γ p .fst .snd r
      fΓ : Q → X
      fΓ (p , r) = γ p .snd r
      W = λ q → fW (prW (suc n) m q)
      Γ = λ q → fΓ (prW (suc n) m q)
      pL : (x : Fin m₀) → prW (suc n) m (finL x) ≡ (zero , x)
      pL x = cong (⊎.rec (λ x → zero , x) (λ y → suc (prW n (m ∘ suc) y .fst) , prW n (m ∘ suc) y .snd))
               (splitFin-L x)
      pR : (y : Fin (sumFin n (m ∘ suc)))
        → prW (suc n) m (finR {m = m₀} y) ≡ (suc (prW n (m ∘ suc) y .fst) , prW n (m ∘ suc) y .snd)
      pR y = cong (⊎.rec (λ x → zero , x) (λ y → suc (prW n (m ∘ suc) y .fst) , prW n (m ∘ suc) y .snd))
               (splitFin-R {m = m₀} y)

    μ-step : {X : Type} (n : ℕ) (w : Fin (suc n) → R) (γ : Fin (suc n) → T X)
      → toL (μ ((suc n , w) , γ)) ≡ map (scale (w zero)) (toL (γ zero)) ++ toL (μ ((n , w ∘ suc) , γ ∘ suc))
    μ-step n w γ =
      toL-fromL _ ∙ cong (_++ toL (μ ((n , w ∘ suc) , γ ∘ suc))) (toL-fromL _)

    μ-agree-toL : {X : Type} (n : ℕ) (w : Fin n → R) (γ : Fin n → T X)
      → toL (μ ((n , w) , γ)) ≡ toL (μᶜ wmC ((n , w) , γ))
    μ-agree-toL zero    w γ = refl
    μ-agree-toL (suc n) w γ =
      μ-step n w γ ∙ cong (map (scale (w zero)) (toL (γ zero)) ++_) (μ-agree-toL n (w ∘ suc) (γ ∘ suc))
      ∙ sym (μᶜ-step n w γ)

    μ-agree : {X : Type} (t : T (T X)) → μ t ≡ μᶜ wmC t
    μ-agree ((n , w) , γ) = sym (fromL-toL _) ∙ cong fromL (μ-agree-toL n w γ) ∙ fromL-toL _

    η-agree : {X : Type} (x : X) → ηW x ≡ ηᶜ wmC x
    η-agree x = ΣPathP (ΣPathP (refl , funExt λ { zero → refl }) , funExt λ { zero → refl })

    gen-agree : (o : WOp R) → wmGen o ≡ opsW o ηW
    gen-agree unit    = refl
    gen-agree mul     = refl
    gen-agree twice   = refl
    gen-agree (act r) = ΣPathP (ΣPathP (refl , funExt λ { zero → sym (unitʳ r) }) , funExt λ { zero → refl })

  wmCM : ContainerModel wmTheory
  wmCM = FromPolyWith.With.cm wmTheory wmOM wmC η-agree μ-agree wmGen gen-agree

  -- Affinity

  private
    embW : (n : ℕ) (m : Fin n → ℕ) → Σ[ p ∈ Fin n ] Fin (m p) → Fin (sumFin n m)
    embW (suc n) m (zero  , x) = finL x
    embW (suc n) m (suc p , r) = finR {m = m zero} (embW n (m ∘ suc) (p , r))

    split-retract : (a b : ℕ) (q : Fin (a + b)) → ⊎.rec finL (finR {m = a}) (splitFin {m = a} {b = b} q) ≡ q
    split-retract zero    b q       = refl
    split-retract (suc a) b zero    = refl
    split-retract (suc a) b (suc k) = rm (splitFin {m = a} {b = b} k) ∙ cong suc (split-retract a b k)
      where
      rm : (x : Fin a ⊎ Fin b)
        → ⊎.rec finL (finR {m = suc a}) (⊎.map suc (λ y → y) x) ≡ suc (⊎.rec finL (finR {m = a}) x)
      rm (inl _) = refl
      rm (inr _) = refl

    emb-pr : (n : ℕ) (m : Fin n → ℕ) (q : Fin (sumFin n m)) → embW n m (prW n m q) ≡ q
    emb-pr zero    m ()
    emb-pr (suc n) m q = h (splitFin {m = m zero} q) ∙ split-retract (m zero) _ q
      where
      h : (x : Fin (m zero) ⊎ Fin (sumFin n (m ∘ suc)))
        → embW (suc n) m (⊎.rec (λ x → zero , x) (λ y → suc (prW n (m ∘ suc) y .fst) , prW n (m ∘ suc) y .snd) x)
          ≡ ⊎.rec finL (finR {m = m zero}) x
      h (inl x) = refl
      h (inr y) = cong (finR {m = m zero}) (emb-pr n (m ∘ suc) y)

    prW-inj : (n : ℕ) (m : Fin n → ℕ) (q q' : Fin (sumFin n m)) → prW n m q ≡ prW n m q' → q ≡ q'
    prW-inj n m q q' e = sym (emb-pr n m q) ∙ cong (embW n m) e ∙ emb-pr n m q'

  wm-affine : Affine wmC
  wm-affine = Inj₁→Affine wmC λ { (n , w) i t q q' e ne →
    let v  = fillAt i (λ _ → ι') t
        m  = λ p → v p .fst
        p' = prW n m q' .fst
        ne' : ¬ p' ≡ i
        ne' y = ne (e ∙ y)
        prop : isProp (Fin (m p'))
        prop = subst (λ z → isProp (Fin (z .fst))) (sym (fill-other i (λ _ → ι') t p' ne')) isPropFin1
    in prW-inj n m q q' (ΣPathP (e , toPathP (prop _ _))) }
    where
    ι' : Σ[ n ∈ ℕ ] (Fin n → R)
    ι' = 1 , λ _ → 1r
