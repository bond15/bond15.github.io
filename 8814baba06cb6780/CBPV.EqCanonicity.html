{-
  Equational canonicity: every closed M : F A is ≈ to reify(t) for a
  tree t : Term_Σ(Val A).  By a logical relation closed under ≈, with
  𝒞⟦F A⟧ inductively generated by ret, op and ≈.

  The tree is unique only up to the equations of 𝒯: for writer,
  tell_e(ret V) ≈ ret V.  What is unique is its image ⌊t⌋ in the free
  model T⟦A⟧, which is ⟦M⟧ (canonical-form-unique).
-}

private variable
  Γ : Ctx
  A A' : VTy
  B B' : CTy

-- 𝒞⟦F A⟧ := generated by ret, op and ≈.
data FPred (P : CVal A → Type) : CComp (F A) → Type where
  ret  : {V : CVal A} → P V → FPred P (ret V)
  op   : (o : Op) {k : Fin (ar o) → CComp (F A)} → (∀ j → FPred P (k j)) → FPred P (op o k)
  conv : {M M' : CComp (F A)} → M ≈c M' → FPred P M' → FPred P M

𝒱⟦_⟧ : (A : VTy) → CVal A → Type
𝒞⟦_⟧ : (B : CTy) → CComp B → Type

𝒱⟦ 𝟘 ⟧ V      = ⊥
𝒱⟦ 𝟙 ⟧ V      = Unit
𝒱⟦ A ⊕ A' ⟧ V = (Σ[ W ∈ CVal A ] (V ≈v inl W) × 𝒱⟦ A ⟧ W) ⊎ (Σ[ W ∈ CVal A' ] (V ≈v inr W) × 𝒱⟦ A' ⟧ W)
𝒱⟦ A ⊗ A' ⟧ V = Σ[ W ∈ CVal A ] Σ[ W' ∈ CVal A' ] (V ≈v vpair W W') × 𝒱⟦ A ⟧ W × 𝒱⟦ A' ⟧ W'
𝒱⟦ U B ⟧ V    = 𝒞⟦ B ⟧ (force V)

𝒞⟦ ⊤ ⟧ M      = Unit
𝒞⟦ A ⇒ B ⟧ M  = (V : CVal A) → 𝒱⟦ A ⟧ V → 𝒞⟦ B ⟧ (app M V)
𝒞⟦ B & B' ⟧ M = 𝒞⟦ B ⟧ (π₁ M) × 𝒞⟦ B' ⟧ (π₂ M)
𝒞⟦ F A ⟧ M    = FPred 𝒱⟦ A ⟧ M

𝒱-conv : (A : VTy) {V V' : CVal A} → V ≈v V' → 𝒱⟦ A ⟧ V' → 𝒱⟦ A ⟧ V
𝒞-conv : (B : CTy) {M M' : CComp B} → M ≈c M' → 𝒞⟦ B ⟧ M' → 𝒞⟦ B ⟧ M

𝒱-conv 𝟘        p ()
𝒱-conv 𝟙        p v = tt
𝒱-conv (A ⊕ A') p (inl (W , q , w)) = inl (W , ≈trans p q , w)
𝒱-conv (A ⊕ A') p (inr (W , q , w)) = inr (W , ≈trans p q , w)
𝒱-conv (A ⊗ A') p (W , W' , q , w , w') = W , W' , ≈trans p q , w , w'
𝒱-conv (U B)    p v = 𝒞-conv B (force≈ p) v

𝒞-conv ⊤        p c = tt
𝒞-conv (A ⇒ B)  p c V v = 𝒞-conv B (app≈ p ≈refl) (c V v)
𝒞-conv (B & B') p (c , c') = 𝒞-conv B (π₁≈ p) c , 𝒞-conv B' (π₂≈ p) c'
𝒞-conv (F A)    p c = conv p c

-- Closed under the operations, by algebraicity of stacks.
𝒞-op : (B : CTy) (o : Op) {k : Fin (ar o) → CComp B} → (∀ j → 𝒞⟦ B ⟧ (k j)) → 𝒞⟦ B ⟧ (op o k)
𝒞-op ⊤        o c = tt
𝒞-op (A ⇒ B)  o {k} c V v = 𝒞-conv B (op-stack (appˢ • V) o k) (𝒞-op B o (λ j → c j V v))
𝒞-op (B & B') o {k} c =
    𝒞-conv B  (op-stack (π₁ˢ •) o k) (𝒞-op B  o (λ j → c j .fst))
  , 𝒞-conv B' (op-stack (π₂ˢ •) o k) (𝒞-op B' o (λ j → c j .snd))
𝒞-op (F A)    o c = op o c

𝒢⟦_⟧ : (Γ : Ctx) → Subst ε Γ → Type
𝒢⟦ Γ ⟧ γ = ∀ {A} (x : Var Γ A) → 𝒱⟦ A ⟧ (γ x)

𝒢-ext : {γ : Subst ε Γ} {V : CVal A} → 𝒢⟦ Γ ⟧ γ → 𝒱⟦ A ⟧ V → 𝒢⟦ Γ ▸ A ⟧ (γ ,ₛ V)
𝒢-ext g v vz     = v
𝒢-ext g v (vs x) = g x

𝒞-≡ : (B : CTy) {M M' : CComp B} → M ≡ M' → 𝒞⟦ B ⟧ M' → 𝒞⟦ B ⟧ M
𝒞-≡ B p = subst 𝒞⟦ B ⟧ (sym p)

------------------------------------------------------------------------
-- Fundamental lemma

fund-bind : (B : CTy) (γ : Subst ε Γ) (N : Comp (Γ ▸ A) B)
  → (∀ V → 𝒱⟦ A ⟧ V → 𝒞⟦ B ⟧ (subC (γ ,ₛ V) N))
  → {M : CComp (F A)} → 𝒞⟦ F A ⟧ M → 𝒞⟦ B ⟧ (bind M (subC (liftS γ) N))
fund-bind B γ N hN (ret {V = V} v) = 𝒞-conv B (F-β V _) (𝒞-≡ B (sub₀-lift γ N V) (hN V v))
fund-bind B γ N hN (op o {k} c) =
  𝒞-conv B (op-stack (bindˢ • (subC (liftS γ) N)) o k) (𝒞-op B o λ j → fund-bind B γ N hN (c j))
fund-bind B γ N hN (conv p c) = 𝒞-conv B (bind≈ p ≈refl) (fund-bind B γ N hN c)

fundV : (V : Val Γ A) (γ : Subst ε Γ) → 𝒢⟦ Γ ⟧ γ → 𝒱⟦ A ⟧ (subV γ V)
fundC : (M : Comp Γ B) (γ : Subst ε Γ) → 𝒢⟦ Γ ⟧ γ → 𝒞⟦ B ⟧ (subC γ M)

fundV (var x)     γ g = g x
fundV (absurd V)  γ g = ⊥.rec (fundV V γ g)
fundV ⋆           γ g = tt
fundV (inl V)     γ g = inl (_ , ≈refl , fundV V γ g)
fundV (inr V)     γ g = inr (_ , ≈refl , fundV V γ g)
fundV (vpair V W) γ g = _ , _ , ≈refl , fundV V γ g , fundV W γ g
fundV (thunk {B = B} M) γ g = 𝒞-conv B (U-β _) (fundC M γ g)

fundC (case+ {B = B} V M₁ M₂) γ g with fundV V γ g
... | inl (X , p , u) =
  𝒞-conv B (≈trans (case+≈ p ≈refl ≈refl) (case+-β₁ X _ _))
    (𝒞-≡ B (sub₀-lift γ M₁ X) (fundC M₁ (γ ,ₛ X) (𝒢-ext g u)))
... | inr (X , p , u) =
  𝒞-conv B (≈trans (case+≈ p ≈refl ≈refl) (case+-β₂ X _ _))
    (𝒞-≡ B (sub₀-lift γ M₂ X) (fundC M₂ (γ ,ₛ X) (𝒢-ext g u)))
fundC (case× {B = B} V M) γ g with fundV V γ g
... | (X , U' , p , u , u') =
  𝒞-conv B (≈trans (case×≈ p ≈refl) (case×-β X U' _))
    (𝒞-≡ B (sub₁-lift γ M X U') (fundC M (γ ,ₛ X ,ₛ U') (𝒢-ext (𝒢-ext g u) u')))
fundC (force V)   γ g = fundV V γ g
fundC triv        γ g = tt
fundC (lam {B = B} M) γ g X u =
  𝒞-conv B (⇒-β _ X) (𝒞-≡ B (sub₀-lift γ M X) (fundC M (γ ,ₛ X) (𝒢-ext g u)))
fundC (app M V)   γ g = fundC M γ g _ (fundV V γ g)
fundC (cpair {B₁ = B₁} {B₂ = B₂} M N) γ g =
  𝒞-conv B₁ (&-β₁ _ _) (fundC M γ g) , 𝒞-conv B₂ (&-β₂ _ _) (fundC N γ g)
fundC (π₁ M)      γ g = fundC M γ g .fst
fundC (π₂ M)      γ g = fundC M γ g .snd
fundC (ret V)     γ g = ret (fundV V γ g)
fundC (bind {B = B} M N) γ g = fund-bind B γ N (λ V v → fundC N (γ ,ₛ V) (𝒢-ext g v)) (fundC M γ g)
fundC (op {B = B} o k) γ g = 𝒞-op B o (λ j → fundC (k j) γ g)

related : (M : CComp B) → 𝒞⟦ B ⟧ M
related {B = B} M = 𝒞-≡ B (sym (sub-idC (λ _ → refl) M)) (fundC M idS (λ ()))

------------------------------------------------------------------------
-- Equational canonicity

canonical-form : {M : CComp (F A)} → 𝒞⟦ F A ⟧ M → Σ[ t ∈ Term sig (CVal A) ] M ≈c reify t
canonical-form (ret {V = V} v) = var V , ≈refl
canonical-form (op o c) = op o (λ j → canonical-form (c j) .fst) , op≈ o (λ j → canonical-form (c j) .snd)
canonical-form (conv p c) = canonical-form c .fst , ≈trans p (canonical-form c .snd)

equational-canonicity : (M : CComp (F A)) → Σ[ t ∈ Term sig (CVal A) ] M ≈c reify t
equational-canonicity M = canonical-form (related M)

-- ⌊t⌋ := t interpreted in T⟦A⟧ with leaves η(⟦V⟧).
⌊_⌋ : Term sig (CVal A) → T ⟦ A ⟧v
⌊ t ⌋ = eval sig ops (λ V → η (⟦ V ⟧V ρ∅)) t

den-reify : (t : Term sig (CVal A)) → ⟦ reify t ⟧C ρ∅ ≡ ⌊ t ⌋
den-reify (var V)   = refl
den-reify (op o ts) = cong (ops o) (funExt λ j → den-reify (ts j))

-- The canonical form is unique in the free model, where it is ⟦M⟧.
canonical-form-unique : (M : CComp (F A)) (t t' : Term sig (CVal A))
  → M ≈c reify t → M ≈c reify t' → ⌊ t ⌋ ≡ ⌊ t' ⌋
canonical-form-unique M t t' p p' =
  sym (sound-c p ρ∅ ∙ den-reify t) ∙ sound-c p' ρ∅ ∙ den-reify t'

canonical-form-denotes : (M : CComp (F A)) (t : Term sig (CVal A)) → M ≈c reify t → ⟦ M ⟧C ρ∅ ≡ ⌊ t ⌋
canonical-form-denotes M t p = sound-c p ρ∅ ∙ den-reify t
