{-
  Errors: one nullary operation err, no equations.  Free model
  T X = X + 1.

  Monadic container: Sh := {run, raised} (Bool), P(run) := 1,
  P(raised) := 0, ι := run, σ(run, v) := v(0), σ(raised, v) := raised,
  pr(q) := (0, q).  ⌈err⌉ := ⟨raised⟩.  Affine: no shape has a neighbor
  of the hole.
-}
module CBPV.Instances.Errors where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Bool using (Bool; true; false)
open import Cubical.Data.FinData using (Fin; zero; suc)

open import CBPV.Theory
open import CBPV.FreeModel
open import CBPV.Polynomial
open import CBPV.OperationalModel
open import CBPV.MonadicContainer
open import CBPV.ContainerModel
open import CBPV.ContainerAffinity using (Affine; Inj₁→Affine)
open import CBPV.Instances.Common

sigE : Signature
sigE = record { Op = Unit ; ar = λ _ → 0 }

errTheory : Theory
errTheory = record { sig = sigE ; Eqn = ⊥ ; eqn = λ () }

open Model

-- run = true, raised = false
errPoly : Poly
errPoly = record { Sh = Bool ; P = λ { true → 1 ; false → 0 } }

private
  T : Type → Type
  T = ⟦ errPoly ⟧

  ηE : {X : Type} → X → T X
  ηE x = true , λ _ → x

  opsE : {X : Type} → AlgStr sigE (T X)
  opsE _ _ = false , noArgs

  extE : {X : Type} (N : Model errTheory) → (X → N .Carrier) → T X → N .Carrier
  extE N f (true  , γ) = f (γ zero)
  extE N f (false , γ) = N .alg tt noArgs

  uniqE : {X : Type} (N : Model errTheory) (f : X → N .Carrier) (h : T X → N .Carrier)
    → IsHom sigE opsE (N .alg) h → (∀ x → h (ηE x) ≡ f x) → ∀ t → h t ≡ extE N f t
  uniqE N f h hh hη (true , γ) = cong h (ΣPathP (refl , fin1-const γ)) ∙ hη (γ zero)
  uniqE N f h hh hη (false , γ) =
    cong h (ΣPathP (refl , fin0-ext γ)) ∙ hh .pres tt noArgs ∙ cong (N .alg tt) (fin0-ext _)

errFree : FreeModel errTheory T
errFree = record
  { η        = ηE
  ; ops      = opsE
  ; ops-sat  = λ ()
  ; ext      = extE
  ; ext-η    = λ N f x → refl
  ; ext-hom  = λ N f → ishom λ _ args → cong (N .alg tt) (sym (fin0-ext _))
  ; ext-uniq = uniqE }

errOM : OperationalModel errTheory
errOM = mkOperationalModel errTheory errPoly errFree (λ f → ishom λ _ args → ΣPathP (refl , fin0-ext _)) (λ f x → refl)

errC : MCOps errPoly
errC = record
  { ι  = true
  ; σ  = λ { true v → v zero ; false v → false }
  ; pr = λ { true v q → zero , q ; false v () } }

-- ⌈err⌉ := ⟨raised⟩
errGen : Unit → T (Fin 0)
errGen _ = false , noArgs

private
  μ-agree : {X : Type} (t : T (T X)) → FreeModel.μ errFree t ≡ μᶜ errC t
  μ-agree (true  , γ) = refl
  μ-agree (false , γ) = ΣPathP (refl , funExt λ ())

errCM : ContainerModel errTheory
errCM = FromPolyWith.With.cm errTheory errOM errC (λ _ → refl) μ-agree errGen (λ _ → refl)

err-affine : Affine errC
err-affine = Inj₁→Affine errC λ { true i t q q' e ne → ⊥.rec (ne (isPropFin1 _ _)) ; false () }
