-- A free model on a polynomial whose polynomial action is a homomorphism
-- commuting with η is an operational model.
mkOperationalModel : (𝒯 : Theory) (p : Poly) (fm : FreeModel 𝒯 ⟦ p ⟧)
  → (∀ {X Y} (f : X → Y) → IsHom (Theory.sig 𝒯) (FreeModel.ops fm) (FreeModel.ops fm) (pmap {p = p} f))
  → (∀ {X Y} (f : X → Y) (x : X) → pmap {p = p} f (FreeModel.η fm x) ≡ FreeModel.η fm (f x))
  → OperationalModel 𝒯
mkOperationalModel 𝒯 p fm hom nat = record
  { poly     = p
  ; free     = fm
  ; map-poly = λ f t → sym (ext-uniq (FM _) (η ∘ f) (pmap f) (hom f) (nat f) t) }
  where open FreeModel fm

record MonoidOn (M : Type) : Type where
  field
    e     : M
    _⊗_   : M → M → M
    unitˡ : ∀ m → e ⊗ m ≡ m
    unitʳ : ∀ m → m ⊗ e ≡ m
    assoc : ∀ m n l → (m ⊗ n) ⊗ l ≡ m ⊗ (n ⊗ l)

noArgs : {X : Type} → Fin 0 → X
noArgs ()

fin0-ext : {X : Type} (g : Fin 0 → X) → g ≡ noArgs
fin0-ext g = funExt λ ()

isPropFin1 : isProp (Fin 1)
isPropFin1 zero zero = refl

fin1-const : {X : Type} (g : Fin 1 → X) → g ≡ (λ _ → g zero)
fin1-const g = funExt λ { zero → refl }

pair2 : {X : Type} → X → X → Fin 2 → X
pair2 a b zero       = a
pair2 a b (suc zero) = b

fin2-ext : {X : Type} (g : Fin 2 → X) → g ≡ pair2 (g zero) (g (suc zero))
fin2-ext g = funExt λ { zero → refl ; (suc zero) → refl }
