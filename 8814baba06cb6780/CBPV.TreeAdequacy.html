{-
  Tree adequacy: from the Fundamental Lemma for 𝒞, by induction on the
  derivation of 𝒞⟦F A⟧(M), then soundness.  Strong canonicity adds
  uniqueness of the tree (confluence).
-}
open import Cubical.Foundations.Prelude using (Type)
open import CBPV.Theory
open import CBPV.FreeModel

module CBPV.TreeAdequacy (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Sigma

open Theory 𝒯
open FreeModel fm
open import CBPV.Syntax sig
open import CBPV.Denotation 𝒯 fm
open import CBPV.Reduction sig
open import CBPV.Soundness 𝒯 fm
open import CBPV.Trees sig using (reify; op-cong*)
open import CBPV.Confluence sig using (tree-unique)
open import CBPV.SNFundamental sig using (related)
open import CBPV.NormalForms sig using (progress; NF; neutral-not-NF)
open import CBPV.SNRelation sig using (𝒞⟦_⟧; S; SPred; sret; sop; sneu)
open import Cubical.Data.Empty as ⊥ using ()
open import Cubical.Data.Sum using (inl; inr)

private variable
  A : VTy

tree-adequacy : (M : CComp (F A))
  → Σ[ t ∈ Term sig (CVal A) ] (M ↦* reify t) × (⟦ reify t ⟧C ρ∅ ≡ ⟦ M ⟧C ρ∅)
tree-adequacy M =
  let (t , p) = tree (related M) in
  t , p , sym (sound↦* p)
  where
  tree : {M : CComp (F A)} → 𝒞⟦ F A ⟧ .S M → Σ[ t ∈ Term sig (CVal A) ] M ↦* reify t
  tree (sret {V = V} v) = var V , done
  tree (sop o {k} c r) =
    op o (λ j → tree (c j) .fst) , op-cong* o k _ (λ j → tree (c j) .snd)
  -- a neutral term is not a normal form, so it steps
  tree {M = M} (sneu n r) with progress M
  ... | inl (M' , s) = tree (r M' s) .fst , s ∷ tree (r M' s) .snd
  ... | inr h        = ⊥.rec (neutral-not-NF n h)

-- Strong canonicity: M ↦* reify t for a unique tree t, and ⟦reify t⟧ = ⟦M⟧.
strong-canonicity : (M : CComp (F A))
  → Σ[ t ∈ Term sig (CVal A) ] (M ↦* reify t) × (⟦ reify t ⟧C ρ∅ ≡ ⟦ M ⟧C ρ∅)
      × (∀ t' → M ↦* reify t' → t' ≡ t)
strong-canonicity M =
  let (t , p , d) = tree-adequacy M in
  t , p , d , λ t' p' → tree-unique t' t p' p
