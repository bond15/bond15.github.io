{-
  Soundness of ⟦–⟧ for the equational theory:  M ≈ N  ⇒  ⟦M⟧ = ⟦N⟧,
  by induction on ≈.
-}
open import Cubical.Foundations.Prelude using (Type)
open import CBPV.Theory
open import CBPV.FreeModel

module CBPV.EquationalSoundness (𝒯 : Theory) {T : Type → Type} (fm : FreeModel 𝒯 T) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.Sigma

open Theory 𝒯
open Model
open FreeModel fm

open import CBPV.Syntax sig
open import CBPV.Renaming sig
open import CBPV.Substitution sig
open import CBPV.Plug sig
open import CBPV.Denotation 𝒯 fm
open import CBPV.DenotationLemmas 𝒯 fm
open import CBPV.Equational 𝒯

private variable
  Γ : Ctx
  A A' : VTy
  B : CTy

-- ⟦–⟧ is a homomorphism out of the syntactic algebra Γ ⊢ B.
den-eval : {X : Type} (γ : X → Comp Γ B) (t : Term sig X) (ρ : Env Γ)
  → ⟦ eval sig (CompAlg Γ B) γ t ⟧C ρ ≡ eval sig (⟦ B ⟧c .alg) (λ x → ⟦ γ x ⟧C ρ) t
den-eval γ (var x)           ρ = refl
den-eval {B = B} γ (op o ts) ρ = cong (⟦ B ⟧c .alg o) (funExt λ j → den-eval γ (ts j) ρ)

private
  inlS-env : (ρ : Env Γ) (a : ⟦ A ⟧v)
    → EnvSub (inlS {A = A} {A' = A'}) (ρ ▹ a) (_▹_ {A = A ⊕ A'} ρ (inl a))
  inlS-env ρ a vz     = refl
  inlS-env ρ a (vs x) = refl

  inrS-env : (ρ : Env Γ) (a : ⟦ A' ⟧v)
    → EnvSub (inrS {A = A} {A' = A'}) (ρ ▹ a) (_▹_ {A = A ⊕ A'} ρ (inr a))
  inrS-env ρ a vz     = refl
  inrS-env ρ a (vs x) = refl

  pairS-env : (ρ : Env Γ) (s : ⟦ A ⟧v × ⟦ A' ⟧v)
    → EnvSub (pairS {A = A} {A' = A'}) (ρ ▹ s .fst ▹ s .snd) (_▹_ {A = A ⊗ A'} ρ s)
  pairS-env ρ s vz     = refl
  pairS-env ρ s (vs x) = refl

sound-v : {V V' : Val Γ A} → V ≈v V' → (ρ : Env Γ) → ⟦ V ⟧V ρ ≡ ⟦ V' ⟧V ρ
sound-c : {M M' : Comp Γ B} → M ≈c M' → (ρ : Env Γ) → ⟦ M ⟧C ρ ≡ ⟦ M' ⟧C ρ

sound-v ≈refl        ρ = refl
sound-v (≈sym p)     ρ = sym (sound-v p ρ)
sound-v (≈trans p q) ρ = sound-v p ρ ∙ sound-v q ρ
sound-v (absurd≈ p)  ρ = cong ⊥.rec (sound-v p ρ)
sound-v (inl≈ p)     ρ = cong inl (sound-v p ρ)
sound-v (inr≈ p)     ρ = cong inr (sound-v p ρ)
sound-v (vpair≈ p q) ρ = cong₂ _,_ (sound-v p ρ) (sound-v q ρ)
sound-v (thunk≈ p)   ρ = sound-c p ρ
sound-v (η𝟘v W V)    ρ = ⊥.rec (⟦ V ⟧V ρ)
sound-v (η𝟙 V)       ρ = refl
sound-v (ηU V)       ρ = refl

sound-c ≈refl          ρ = refl
sound-c (≈sym p)       ρ = sym (sound-c p ρ)
sound-c (≈trans p q)   ρ = sound-c p ρ ∙ sound-c q ρ
sound-c (case+≈ p q r) ρ i =
  ⊎.rec (λ a → sound-c q (ρ ▹ a) i) (λ a → sound-c r (ρ ▹ a) i) (sound-v p ρ i)
sound-c (case×≈ p q)   ρ i = sound-c q (ρ ▹ sound-v p ρ i .fst ▹ sound-v p ρ i .snd) i
sound-c (force≈ p)     ρ = sound-v p ρ
sound-c (lam≈ p)       ρ i a = sound-c p (ρ ▹ a) i
sound-c (app≈ p q)     ρ i = sound-c p ρ i (sound-v q ρ i)
sound-c (cpair≈ p q)   ρ = cong₂ _,_ (sound-c p ρ) (sound-c q ρ)
sound-c (π₁≈ p)        ρ = cong fst (sound-c p ρ)
sound-c (π₂≈ p)        ρ = cong snd (sound-c p ρ)
sound-c (ret≈ p)       ρ = cong η (sound-v p ρ)
sound-c (bind≈ {B = B} p q) ρ i = ext ⟦ B ⟧c (λ a → sound-c q (ρ ▹ a) i) (sound-c p ρ i)
sound-c (op≈ {B = B} o ps)  ρ i = ⟦ B ⟧c .alg o (λ j → sound-c (ps j) ρ i)
sound-c (case+-β₁ V M₁ M₂) ρ = sym (den-sub₀ ρ M₁ V)
sound-c (case+-β₂ V M₁ M₂) ρ = sym (den-sub₀ ρ M₂ V)
sound-c (case×-β V V' M)   ρ = sym (den-sub₁ ρ M V V')
sound-c (U-β M)            ρ = refl
sound-c (⇒-β M V)          ρ = sym (den-sub₀ ρ M V)
sound-c (&-β₁ M N)         ρ = refl
sound-c (&-β₂ M N)         ρ = refl
sound-c (F-β {B = B} V N)  ρ = ext-η ⟦ B ⟧c (λ a → ⟦ N ⟧C (ρ ▹ a)) (⟦ V ⟧V ρ) ∙ sym (den-sub₀ ρ N V)
sound-c (η𝟘c M V)          ρ = ⊥.rec (⟦ V ⟧V ρ)
sound-c (η+c M V)          ρ = den-sub₀ ρ M V ∙ cases (⟦ V ⟧V ρ)
  where
  cases : (s : _) → ⟦ M ⟧C (ρ ▹ s)
    ≡ ⊎.rec (λ a → ⟦ subC inlS M ⟧C (ρ ▹ a)) (λ a → ⟦ subC inrS M ⟧C (ρ ▹ a)) s
  cases (inl a) = sym (den-subC (inlS-env ρ a) M)
  cases (inr a) = sym (den-subC (inrS-env ρ a) M)
sound-c (η×c M V)          ρ = den-sub₀ ρ M V ∙ sym (den-subC (pairS-env ρ (⟦ V ⟧V ρ)) M)
sound-c (η⇒ {A = A} M)     ρ i a = den-wkC {A' = A} ρ a M i a
sound-c (η& M)             ρ = refl
sound-c (η⊤ M)             ρ = refl
sound-c (ηF {B = B} S M)   ρ =
  den-plug S M ρ
  ∙ ext-uniq ⟦ B ⟧c (λ a → ⟦ S ⟧S ρ (η a)) (⟦ S ⟧S ρ) (den-stack-hom S ρ) (λ _ → refl) (⟦ M ⟧C ρ)
  ∙ cong (λ f → ext ⟦ B ⟧c f (⟦ M ⟧C ρ)) (funExt λ a → sym (body a))
  where
  body : ∀ a → ⟦ plug (wkS S) (ret (var vz)) ⟧C (ρ ▹ a) ≡ ⟦ S ⟧S ρ (η a)
  body a = den-plug (wkS S) (ret (var vz)) (ρ ▹ a) ∙ cong (λ f → f (η a)) (den-renS (λ _ → refl) S)
sound-c (op-stack S o k)   ρ = den-plug-op S o k ρ
sound-c (theory {B = B} e γ) ρ =
  den-eval γ (lhs (eqn e)) ρ ∙ ⟦ B ⟧c .sat e (λ x → ⟦ γ x ⟧C ρ) ∙ sym (den-eval γ (rhs (eqn e)) ρ)
