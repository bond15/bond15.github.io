{-
  A free model monad for a theory 𝒯.

  T X is a model of 𝒯 for every X, with a unit η : X → T X, such that
  for every model M and f : X → |M| there is a unique homomorphism
  !_f : T X → M with !_f ∘ η = f.  Here !_f is called `ext M f`.

  The monad structure (functor action, μ, Kleisli extension) and its
  laws are derived from the universal property.
-}
module CBPV.FreeModel where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function

open import CBPV.Theory

module _ (𝒯 : Theory) where
  open Theory 𝒯
  open Model

  -- The carrier functor T is a parameter so that an operational model
  -- (CBPV.OperationalModel) can fix it to a polynomial definitionally.
  record FreeModel (T : Type → Type) : Type₁ where
    field
      η       : {X : Type} → X → T X
      ops     : {X : Type} → AlgStr sig (T X)
      ops-sat : {X : Type} → Sat 𝒯 (ops {X})
      ext     : {X : Type} (M : Model 𝒯) → (X → Carrier M) → T X → Carrier M
      ext-η   : {X : Type} (M : Model 𝒯) (f : X → Carrier M) (x : X)
              → ext M f (η x) ≡ f x
      ext-hom : {X : Type} (M : Model 𝒯) (f : X → Carrier M)
              → IsHom sig ops (M .alg) (ext M f)
      ext-uniq : {X : Type} (M : Model 𝒯) (f : X → Carrier M)
               (h : T X → Carrier M) → IsHom sig ops (M .alg) h
               → (∀ x → h (η x) ≡ f x)
               → ∀ t → h t ≡ ext M f t

    -- The free model on X.
    FM : Type → Model 𝒯
    FM X .Carrier = T X
    FM X .alg     = ops
    FM X .sat     = ops-sat

    map : {X Y : Type} → (X → Y) → T X → T Y
    map f = ext (FM _) (η ∘ f)

    μ : {X : Type} → T (T X) → T X
    μ = ext (FM _) (λ t → t)

    -- Two homomorphisms out of T X that agree on generators are equal.
    hom-ext : {X : Type} (M : Model 𝒯) (h k : T X → Carrier M)
      → IsHom sig ops (M .alg) h → IsHom sig ops (M .alg) k
      → (∀ x → h (η x) ≡ k (η x)) → ∀ t → h t ≡ k t
    hom-ext M h k hh hk p t =
      ext-uniq M (k ∘ η) h hh p t ∙ sym (ext-uniq M (k ∘ η) k hk (λ _ → refl) t)

    ∘-hom : {A B C : Type} {α : AlgStr sig A} {β : AlgStr sig B} {γ : AlgStr sig C}
      (g : B → C) (h : A → B)
      → IsHom sig β γ g → IsHom sig α β h → IsHom sig α γ (g ∘ h)
    ∘-hom g h hg hh = ishom λ o args → cong g (hh .pres o args) ∙ hg .pres o (h ∘ args)

    map-hom : {X Y : Type} (f : X → Y) → IsHom sig ops ops (map f)
    map-hom f = ext-hom (FM _) (η ∘ f)

    map-η : {X Y : Type} (f : X → Y) (x : X) → map f (η x) ≡ η (f x)
    map-η f = ext-η (FM _) (η ∘ f)

    μ-hom : {X : Type} → IsHom sig ops ops (μ {X})
    μ-hom = ext-hom (FM _) (λ t → t)

    μ-η : {X : Type} (t : T X) → μ (η t) ≡ t
    μ-η = ext-η (FM _) (λ t → t)

    μ-mapη : {X : Type} (t : T X) → μ (map η t) ≡ t
    μ-mapη t = hom-ext (FM _) (μ ∘ map η) (λ t → t)
      (∘-hom μ (map η) μ-hom (map-hom η)) (ishom λ _ _ → refl)
      (λ x → cong μ (map-η η x) ∙ μ-η (η x)) t

    -- ext M f = ext M id ∘ map f
    ext-map : {X Y : Type} (M : Model 𝒯) (f : Y → Carrier M) (g : X → Y)
      → ∀ t → ext M f (map g t) ≡ ext M (f ∘ g) t
    ext-map M f g = ext-uniq M (f ∘ g) (ext M f ∘ map g)
      (∘-hom (ext M f) (map g) (ext-hom M f) (map-hom g))
      (λ x → cong (ext M f) (map-η g x) ∙ ext-η M f (g x))

    -- ext M f ∘ μ = ext M (ext M f)
    ext-μ : {X : Type} (M : Model 𝒯) (f : X → Carrier M)
      → ∀ t → ext M f (μ t) ≡ ext M (ext M f) t
    ext-μ M f = ext-uniq M (ext M f) (ext M f ∘ μ)
      (∘-hom (ext M f) μ (ext-hom M f) μ-hom)
      (λ t → cong (ext M f) (μ-η t))

    -- ext into a free model is μ after map.
    ext-FM : {X Y : Type} (f : X → T Y) → ∀ t → ext (FM Y) f t ≡ μ (map f t)
    ext-FM f t = sym (ext-map (FM _) (λ t → t) f t)

    map-∘ : {X Y Z : Type} (g : Y → Z) (f : X → Y) (t : T X)
      → map (g ∘ f) t ≡ map g (map f t)
    map-∘ g f t = sym (ext-map (FM _) (η ∘ g) f t)

    -- μ is natural: T(f) ∘ μ = μ ∘ T(T(f))
    μ-natural : {X Y : Type} (f : X → Y) (t : T (T X)) → map f (μ t) ≡ μ (map (map f) t)
    μ-natural f = hom-ext (FM _) (map f ∘ μ) (μ ∘ map (map f))
      (∘-hom (map f) μ (map-hom f) μ-hom) (∘-hom μ (map (map f)) μ-hom (map-hom (map f)))
      (λ t → cong (map f) (μ-η t) ∙ sym (cong μ (map-η (map f) t) ∙ μ-η (map f t)))

    μ-assoc : {X : Type} (t : T (T (T X))) → μ (μ t) ≡ μ (map μ t)
    μ-assoc t = ext-μ (FM _) (λ t → t) t ∙ ext-FM μ t

    -- ext (FM Y) f = μ ∘ map f
    ext-μmap : {X Y : Type} (f : X → T Y) (t : T X) → μ (map f t) ≡ ext (FM Y) f t
    ext-μmap f t = sym (ext-FM f t)
