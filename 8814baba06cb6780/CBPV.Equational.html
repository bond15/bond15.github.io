{-
  The equational theory of CBPV(𝒯): an inductive relation ≈ on terms
  (terms are not quotiented).  It is an equivalence, a congruence, and
  contains

    β   case+(inl V){x.M₁ | y.M₂} = M₁[V/x]  (and inr)   case×((V,W)){x y.M} = M[V/x,W/y]
        force(thunk M) = M     (λx.M) V = M[V/x]     πᵢ⟨M₁,M₂⟩ = Mᵢ
        x ← ret V; N = N[V/x]
    η   M[V/z] = case+ V {x. M[inl x/z] | y. M[inr y/z]}
        M[V/z] = case× V {x y. M[(x,y)/z]}       M[V/z] = force(absurd V)  (V : 0)
        V = ⋆ (V : 1)    W[V/z] = absurd V (V : 0)    thunk(force V) = V
        λx. M x = M      ⟨π₁ M, π₂ M⟩ = M      M = triv (M : ⊤)
        S[M] = x ← M; S[ret x]      (S = • gives x ← M; ret x = M)
    algebraicity of stacks   S[op(M⃗)] = op(S[M⃗])
    the equations of 𝒯       in every Γ ⊢ B

  The η laws for 0, +, × are in substitution form, since plain CBPV has no
  value eliminators.
-}
open import CBPV.Theory

module CBPV.Equational (𝒯 : Theory) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.FinData using (Fin)

open Theory 𝒯

open import CBPV.Syntax sig
open import CBPV.Renaming sig
open import CBPV.Substitution sig
open import CBPV.Plug sig

private variable
  Γ : Ctx
  A A' A'' : VTy
  B B' B₁ B₂ : CTy

-- Γ ⊢ B as a Σ-algebra.
CompAlg : (Γ : Ctx) (B : CTy) → AlgStr sig (Comp Γ B)
CompAlg Γ B o k = op o k

wk₁ : {Γ : Ctx} {A : VTy} → Subst (Γ ▸ A) Γ
wk₁ x = var (vs x)

wk₂ : {Γ : Ctx} {A A' : VTy} → Subst (Γ ▸ A ▸ A') Γ
wk₂ x = var (vs (vs x))

-- z := inl x, z := inr y, z := (x , y)
inlS : {Γ : Ctx} {A A' : VTy} → Subst (Γ ▸ A) (Γ ▸ (A ⊕ A'))
inlS = wk₁ ,ₛ inl (var vz)

inrS : {Γ : Ctx} {A A' : VTy} → Subst (Γ ▸ A') (Γ ▸ (A ⊕ A'))
inrS = wk₁ ,ₛ inr (var vz)

pairS : {Γ : Ctx} {A A' : VTy} → Subst (Γ ▸ A ▸ A') (Γ ▸ (A ⊗ A'))
pairS = wk₂ ,ₛ vpair (var (vs vz)) (var vz)

infix 4 _≈v_ _≈c_

data _≈v_ : Val Γ A → Val Γ A → Type
data _≈c_ : Comp Γ B → Comp Γ B → Type

data _≈v_ where
  ≈refl  : {V : Val Γ A} → V ≈v V
  ≈sym   : {V W : Val Γ A} → V ≈v W → W ≈v V
  ≈trans : {U V W : Val Γ A} → U ≈v V → V ≈v W → U ≈v W
  absurd≈ : {V V' : Val Γ 𝟘} → V ≈v V' → absurd {A = A} V ≈v absurd V'
  inl≈    : {V V' : Val Γ A} → V ≈v V' → inl {A' = A'} V ≈v inl V'
  inr≈    : {V V' : Val Γ A'} → V ≈v V' → inr {A = A} V ≈v inr V'
  vpair≈  : {V V' : Val Γ A} {W W' : Val Γ A'} → V ≈v V' → W ≈v W' → vpair V W ≈v vpair V' W'
  thunk≈  : {M M' : Comp Γ B} → M ≈c M' → thunk M ≈v thunk M'
  η𝟘v : (W : Val (Γ ▸ 𝟘) A) (V : Val Γ 𝟘) → subV (idS ,ₛ V) W ≈v absurd V
  η𝟙  : (V : Val Γ 𝟙) → V ≈v ⋆
  ηU  : (V : Val Γ (U B)) → thunk (force V) ≈v V

data _≈c_ where
  ≈refl  : {M : Comp Γ B} → M ≈c M
  ≈sym   : {M N : Comp Γ B} → M ≈c N → N ≈c M
  ≈trans : {L M N : Comp Γ B} → L ≈c M → M ≈c N → L ≈c N
  case+≈ : {V V' : Val Γ (A ⊕ A')} {M₁ M₁' : Comp (Γ ▸ A) B} {M₂ M₂' : Comp (Γ ▸ A') B}
         → V ≈v V' → M₁ ≈c M₁' → M₂ ≈c M₂' → case+ V M₁ M₂ ≈c case+ V' M₁' M₂'
  case×≈ : {V V' : Val Γ (A ⊗ A')} {M M' : Comp (Γ ▸ A ▸ A') B}
         → V ≈v V' → M ≈c M' → case× V M ≈c case× V' M'
  force≈ : {V V' : Val Γ (U B)} → V ≈v V' → force V ≈c force V'
  lam≈   : {M M' : Comp (Γ ▸ A) B} → M ≈c M' → lam M ≈c lam M'
  app≈   : {M M' : Comp Γ (A ⇒ B)} {V V' : Val Γ A} → M ≈c M' → V ≈v V' → app M V ≈c app M' V'
  cpair≈ : {M M' : Comp Γ B₁} {N N' : Comp Γ B₂} → M ≈c M' → N ≈c N' → cpair M N ≈c cpair M' N'
  π₁≈    : {M M' : Comp Γ (B₁ & B₂)} → M ≈c M' → π₁ M ≈c π₁ M'
  π₂≈    : {M M' : Comp Γ (B₁ & B₂)} → M ≈c M' → π₂ M ≈c π₂ M'
  ret≈   : {V V' : Val Γ A} → V ≈v V' → ret V ≈c ret V'
  bind≈  : {M M' : Comp Γ (F A)} {N N' : Comp (Γ ▸ A) B} → M ≈c M' → N ≈c N' → bind M N ≈c bind M' N'
  op≈    : (o : Op) {k k' : Fin (ar o) → Comp Γ B} → (∀ j → k j ≈c k' j) → op o k ≈c op o k'
  -- β
  case+-β₁ : (V : Val Γ A) (M₁ : Comp (Γ ▸ A) B) (M₂ : Comp (Γ ▸ A') B) → case+ (inl V) M₁ M₂ ≈c M₁ [ V ]₀
  case+-β₂ : (V : Val Γ A') (M₁ : Comp (Γ ▸ A) B) (M₂ : Comp (Γ ▸ A') B) → case+ (inr V) M₁ M₂ ≈c M₂ [ V ]₀
  case×-β  : (V : Val Γ A) (V' : Val Γ A') (M : Comp (Γ ▸ A ▸ A') B) → case× (vpair V V') M ≈c M [ V , V' ]₁
  U-β      : (M : Comp Γ B) → force (thunk M) ≈c M
  ⇒-β      : (M : Comp (Γ ▸ A) B) (V : Val Γ A) → app (lam M) V ≈c M [ V ]₀
  &-β₁     : (M : Comp Γ B₁) (N : Comp Γ B₂) → π₁ (cpair M N) ≈c M
  &-β₂     : (M : Comp Γ B₁) (N : Comp Γ B₂) → π₂ (cpair M N) ≈c N
  F-β      : (V : Val Γ A) (N : Comp (Γ ▸ A) B) → bind (ret V) N ≈c N [ V ]₀
  -- η
  η𝟘c : (M : Comp (Γ ▸ 𝟘) B) (V : Val Γ 𝟘) → M [ V ]₀ ≈c force (absurd V)
  η+c : (M : Comp (Γ ▸ (A ⊕ A')) B) (V : Val Γ (A ⊕ A')) → M [ V ]₀ ≈c case+ V (subC inlS M) (subC inrS M)
  η×c : (M : Comp (Γ ▸ (A ⊗ A')) B) (V : Val Γ (A ⊗ A')) → M [ V ]₀ ≈c case× V (subC pairS M)
  η⇒  : (M : Comp Γ (A ⇒ B)) → lam (app (wkC M) (var vz)) ≈c M
  η&  : (M : Comp Γ (B₁ & B₂)) → cpair (π₁ M) (π₂ M) ≈c M
  η⊤  : (M : Comp Γ ⊤) → M ≈c triv
  ηF  : (S : Stack Γ (F A) B) (M : Comp Γ (F A)) → plug S M ≈c bind M (plug (wkS S) (ret (var vz)))
  -- algebraicity and the theory
  op-stack : (S : Stack Γ B B') (o : Op) (k : Fin (ar o) → Comp Γ B)
           → plug S (op o k) ≈c op o (λ j → plug S (k j))
  theory   : (e : Eqn) (γ : Fin (ctx (eqn e)) → Comp Γ B)
           → eval sig (CompAlg Γ B) γ (lhs (eqn e)) ≈c eval sig (CompAlg Γ B) γ (rhs (eqn e))
