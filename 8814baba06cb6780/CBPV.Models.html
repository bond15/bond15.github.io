{-
  The models of 𝒯 interpreting ⊤, A → B and B & B', with pointwise
  algebra structure.
-}
module CBPV.Models where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Unit
open import Cubical.Data.Sigma

open import CBPV.Theory

module Models-of (𝒯 : Theory) where
  open Theory 𝒯
  open Model

  unitModel : Model 𝒯
  unitModel .Carrier     = Unit
  unitModel .alg o args  = tt
  unitModel .sat e γ     = refl

  -- Pointwise algebra on A → |M|
  module _ (A : Type) (M : Model 𝒯) where
    powAlg : AlgStr sig (A → M .Carrier)
    powAlg o fs a = M .alg o (λ j → fs j a)

    eval-pow : {X : Type} (γ : X → A → M .Carrier) (t : Term sig X) (a : A)
      → eval sig powAlg γ t a ≡ eval sig (M .alg) (λ x → γ x a) t
    eval-pow γ (var x)   a = refl
    eval-pow γ (op o ts) a = cong (M .alg o) (funExt λ j → eval-pow γ (ts j) a)

    powModel : Model 𝒯
    powModel .Carrier = A → M .Carrier
    powModel .alg     = powAlg
    powModel .sat e γ = funExt λ a →
      eval-pow γ (lhs (eqn e)) a ∙ M .sat e (λ x → γ x a) ∙ sym (eval-pow γ (rhs (eqn e)) a)

  -- Pointwise algebra on |M| × |N|
  module _ (M N : Model 𝒯) where
    prodAlg : AlgStr sig (M .Carrier × N .Carrier)
    prodAlg o ps = M .alg o (λ j → ps j .fst) , N .alg o (λ j → ps j .snd)

    eval-prod₁ : {X : Type} (γ : X → M .Carrier × N .Carrier) (t : Term sig X)
      → eval sig prodAlg γ t .fst ≡ eval sig (M .alg) (fst ∘ γ) t
    eval-prod₁ γ (var x)   = refl
    eval-prod₁ γ (op o ts) = cong (M .alg o) (funExt λ j → eval-prod₁ γ (ts j))

    eval-prod₂ : {X : Type} (γ : X → M .Carrier × N .Carrier) (t : Term sig X)
      → eval sig prodAlg γ t .snd ≡ eval sig (N .alg) (snd ∘ γ) t
    eval-prod₂ γ (var x)   = refl
    eval-prod₂ γ (op o ts) = cong (N .alg o) (funExt λ j → eval-prod₂ γ (ts j))

    prodModel : Model 𝒯
    prodModel .Carrier = M .Carrier × N .Carrier
    prodModel .alg     = prodAlg
    prodModel .sat e γ = ΣPathP
      ( eval-prod₁ γ (lhs (eqn e)) ∙ M .sat e (fst ∘ γ) ∙ sym (eval-prod₁ γ (rhs (eqn e)))
      , eval-prod₂ γ (lhs (eqn e)) ∙ N .sat e (snd ∘ γ) ∙ sym (eval-prod₂ γ (rhs (eqn e))) )
