{-
  Types and contexts of CBPV(𝒯).

    A ::= 0 | 1 | A + A' | A × A' | U B
    B ::= ⊤ | A → B | B & B' | F A

  Contexts are snoc lists; variables are de Bruijn indices.
-}
module CBPV.Types where

open import Cubical.Foundations.Prelude

infixr 30 _⇒_
infixl 35 _⊕_
infixl 36 _⊗_
infixl 35 _&_
infixl 20 _▸_

data VTy : Type
data CTy : Type

data VTy where
  𝟘 𝟙       : VTy
  _⊕_ _⊗_   : VTy → VTy → VTy
  U         : CTy → VTy

data CTy where
  ⊤         : CTy
  _⇒_       : VTy → CTy → CTy
  _&_       : CTy → CTy → CTy
  F         : VTy → CTy

data Ctx : Type where
  ε   : Ctx
  _▸_ : Ctx → VTy → Ctx

data Var : Ctx → VTy → Type where
  vz : ∀ {Γ A}   → Var (Γ ▸ A) A
  vs : ∀ {Γ A B} → Var Γ A → Var (Γ ▸ B) A
