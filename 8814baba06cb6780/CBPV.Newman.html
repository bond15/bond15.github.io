{-
  Newman's lemma: a strongly normalizing, locally confluent relation is
  confluent.  By well-founded induction on the source.
-}
module CBPV.Newman where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Sigma
open import Cubical.Induction.WellFounded

module _ {X : Type} (_⟶_ : X → X → Type) where

  infixr 5 _∷_ _++_

  data Star : X → X → Type where
    done : {a : X} → Star a a
    _∷_  : {a b c : X} → a ⟶ b → Star b c → Star a c

  _++_ : {a b c : X} → Star a b → Star b c → Star a c
  done    ++ q = q
  (s ∷ p) ++ q = s ∷ (p ++ q)

  Joinable : X → X → Type
  Joinable b c = Σ[ d ∈ X ] Star b d × Star c d

  LocallyConfluent : Type
  LocallyConfluent = ∀ {a b c} → a ⟶ b → a ⟶ c → Joinable b c

  Confluent : Type
  Confluent = ∀ {a b c} → Star a b → Star a c → Joinable b c

  _⟵_ : X → X → Type
  b ⟵ a = a ⟶ b

  newman : (∀ a → Acc _⟵_ a) → LocallyConfluent → Confluent
  newman sn lc {a} = go a (sn a)
    where
    go : ∀ a → Acc _⟵_ a → ∀ {b c} → Star a b → Star a c → Joinable b c
    go a _ done q = _ , q , done
    go a _ p done = _ , done , p
    go a (acc rec) (s₁ ∷ p₁) (s₂ ∷ p₂) =
      let (d , q₁ , q₂) = lc s₁ s₂
          (e , r₁ , r₂) = go _ (rec _ s₁) p₁ q₁
          (f , t₁ , t₂) = go _ (rec _ s₂) (q₂ ++ r₂) p₂
      in f , r₁ ++ t₁ , t₂
