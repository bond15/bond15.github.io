{-
  Terms of CBPV(𝒯), intrinsically typed.

    Val   Γ A      Γ ⊢ V : A
    Comp  Γ B      Γ | · ⊢ M : B
    Stack Γ B B'   Γ | • : B ⊢ S : B'

  Names: σ₁/σ₂ are inl/inr, (V , V') is vpair, ⟨M , N⟩ is cpair,
  x ← M ; N is bind M N.
-}
open import CBPV.Theory

module CBPV.Syntax (Sg : Signature) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.FinData using (Fin)

open import CBPV.Types public

open Signature Sg

private variable
  Γ Δ : Ctx
  A A' A'' : VTy
  B B' B'' B₁ B₂ : CTy

data Val   (Γ : Ctx) : VTy → Type
data Comp  (Γ : Ctx) : CTy → Type
data Stack (Γ : Ctx) : CTy → CTy → Type

data Val Γ where
  var    : Var Γ A → Val Γ A
  absurd : Val Γ 𝟘 → Val Γ A
  ⋆      : Val Γ 𝟙
  inl    : Val Γ A → Val Γ (A ⊕ A')
  inr    : Val Γ A' → Val Γ (A ⊕ A')
  vpair  : Val Γ A → Val Γ A' → Val Γ (A ⊗ A')
  thunk  : Comp Γ B → Val Γ (U B)

data Comp Γ where
  case+ : Val Γ (A ⊕ A') → Comp (Γ ▸ A) B → Comp (Γ ▸ A') B → Comp Γ B
  case× : Val Γ (A ⊗ A') → Comp (Γ ▸ A ▸ A') B → Comp Γ B
  force : Val Γ (U B) → Comp Γ B
  triv  : Comp Γ ⊤
  lam   : Comp (Γ ▸ A) B → Comp Γ (A ⇒ B)
  app   : Comp Γ (A ⇒ B) → Val Γ A → Comp Γ B
  cpair : Comp Γ B₁ → Comp Γ B₂ → Comp Γ (B₁ & B₂)
  π₁    : Comp Γ (B₁ & B₂) → Comp Γ B₁
  π₂    : Comp Γ (B₁ & B₂) → Comp Γ B₂
  ret   : Val Γ A → Comp Γ (F A)
  bind  : Comp Γ (F A) → Comp (Γ ▸ A) B → Comp Γ B
  op    : (o : Op) → (Fin (ar o) → Comp Γ B) → Comp Γ B

-- Stack Γ B B' : a stack with hole of type B and result type B'.
data Stack Γ where
  •      : Stack Γ B B
  appˢ   : Stack Γ B (A ⇒ B') → Val Γ A → Stack Γ B B'
  π₁ˢ    : Stack Γ B (B₁ & B₂) → Stack Γ B B₁
  π₂ˢ    : Stack Γ B (B₁ & B₂) → Stack Γ B B₂
  bindˢ  : Stack Γ B (F A) → Comp (Γ ▸ A) B' → Stack Γ B B'
