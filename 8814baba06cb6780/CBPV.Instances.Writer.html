{-
  Writer, over a monoid (M, ⊗, e).

  Theory: tell_m (unary) for m : M, with tell_e(x) = x and
  tell_m(tell_n(x)) = tell_{m⊗n}(x).  Free model T X = M × X.

  Monadic container: Sh := M, P := 1, ι := e, σ(m, v) := m ⊗ v(0),
  pr(q) := (0, q).  ⌈tell_m⌉ := ⟨m, 0⟩.  Affine: no shape has a
  neighbor of the hole.
-}
module CBPV.Instances.Writer where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.FinData using (Fin; zero; suc)
import Cubical.Data.Empty as ⊥

open import CBPV.Theory
open import CBPV.FreeModel
open import CBPV.Polynomial
open import CBPV.OperationalModel
open import CBPV.MonadicContainer
open import CBPV.ContainerModel
open import CBPV.ContainerAffinity using (Affine; Inj₁→Affine)
open import CBPV.Instances.Common

module _ (M : Type) (mon : MonoidOn M) where
  open MonoidOn mon

  sigW : Signature
  sigW = record { Op = M ; ar = λ _ → 1 }

  private
    x₀ : Term sigW (Fin 1)
    x₀ = var zero

    tell : M → Term sigW (Fin 1) → Term sigW (Fin 1)
    tell m t = op m (λ _ → t)

  writerTheory : Theory
  writerTheory = record
    { sig = sigW
    ; Eqn = Unit ⊎ (M × M)
    ; eqn = λ
      { (inl _)       → record { ctx = 1 ; lhs = tell e x₀ ; rhs = x₀ }
      ; (inr (m , n)) → record { ctx = 1 ; lhs = tell m (tell n x₀) ; rhs = tell (m ⊗ n) x₀ } } }

  open Model

  writerPoly : Poly
  writerPoly = record { Sh = M ; P = λ _ → 1 }

  private
    T : Type → Type
    T = ⟦ writerPoly ⟧

    ηW : {X : Type} → X → T X
    ηW x = e , λ _ → x

    opsW : {X : Type} → AlgStr sigW (T X)
    opsW n args = n ⊗ args zero .fst , args zero .snd

    satW : {X : Type} → Sat writerTheory (opsW {X})
    satW (inl _)       γ = ΣPathP (unitˡ _ , refl)
    satW (inr (m , n)) γ = ΣPathP (sym (assoc m n _) , refl)

    extW : {X : Type} (N : Model writerTheory) → (X → N .Carrier) → T X → N .Carrier
    extW N f (m , γ) = N .alg m (λ _ → f (γ zero))

    tell-e : (N : Model writerTheory) (x : N .Carrier) → N .alg e (λ _ → x) ≡ x
    tell-e N x = N .sat (inl tt) (λ _ → x)

    tell-tell : (N : Model writerTheory) (m n : M) (x : N .Carrier)
      → N .alg m (λ _ → N .alg n (λ _ → x)) ≡ N .alg (m ⊗ n) (λ _ → x)
    tell-tell N m n x = N .sat (inr (m , n)) (λ _ → x)

  writerFree : FreeModel writerTheory T
  writerFree = record
    { η        = ηW
    ; ops      = opsW
    ; ops-sat  = satW
    ; ext      = extW
    ; ext-η    = λ N f x → tell-e N (f x)
    ; ext-hom  = λ N f → ishom λ n args →
        sym (tell-tell N n (args zero .fst) _)
        ∙ cong (N .alg n) (sym (fin1-const (λ j → extW N f (args j))))
    ; ext-uniq = λ N f h hh hη → λ { (m , γ) →
        cong h (ΣPathP (sym (unitʳ m) , fin1-const γ))
        ∙ hh .pres m (λ _ → ηW (γ zero))
        ∙ cong (N .alg m) (funExt λ _ → hη (γ zero)) } }

  writerOM : OperationalModel writerTheory
  writerOM = mkOperationalModel writerTheory writerPoly writerFree (λ f → ishom λ _ _ → refl) (λ f x → refl)

  writerC : MCOps writerPoly
  writerC = record { ι = e ; σ = λ m v → m ⊗ v zero ; pr = λ m v q → zero , q }

  -- ⌈tell_m⌉ := ⟨m, 0⟩
  writerGen : (m : M) → T (Fin 1)
  writerGen m = m , λ _ → zero

  writerCM : ContainerModel writerTheory
  writerCM = FromPolyWith.With.cm writerTheory writerOM writerC (λ _ → refl) (λ _ → refl) writerGen
    (λ m → ΣPathP (sym (unitʳ m) , refl))

  writer-affine : Affine writerC
  writer-affine = Inj₁→Affine writerC λ s i t q q' e ne → ⊥.rec (ne (isPropFin1 _ _))
