{-
  Affinity, on the monadic container alone, and its bridge to the
  termination proof.

  For a one-hole shape c = (s, i) : ∂Sh := Σ s. P s and a shape t:
      P∂(s, i)   := P s ∖ {i}
      v_{c,t}    := t at i, ι elsewhere
      σ∂(c, t)   := σ(s, v_{c,t})
      pr∂        := split ∘ pr : P(σ∂(c,t)) → (Σ p : P∂ c. P ι) ⊎ P t
  Affine := for all c, t:  π₁ ∘ pr∂ is injective on pr∂⁻¹(inl),
  i.e. plugging a shape t into a hole never copies a neighbor.

  Bridge (for a container model, through toPoly):
      Affine ⇔ NoCopyShapes ⇔ NoCopyOps          (affine⇔nocopy)
  using that stepG s i (t , _) is μ of the context with neighbors
  η(inl p) and t at the hole, computed by the container formula.
  Hence config-sn and config-adequacy for affine container models.

  Derivative rule in container form: the Effect rule's right-hand side
  is μ((s, i, η ∘ γ)⟪T(k)(⌈op⌉)⟫) with the container's μ (effect-container),
  of shape σ∂((s,i), s_op) (effect-shape); a position q holds γ(pr₁ q) off
  the hole (effect-off) and k(ℓ_op m) at it (effect-hole).
-}

private
  inl-inj : {X Y : Type} {a b : X} → Path (X ⊎ Y) (inl a) (inl b) → a ≡ b
  inl-inj {a = a} p = cong (⊎.rec (λ x → x) (λ _ → a)) p

  inr≢inl : {X Y : Type} {a : X} {b : Y} → ¬ Path (X ⊎ Y) (inr b) (inl a)
  inr≢inl p = subst (⊎.rec (λ _ → ⊥) (λ _ → Unit)) p tt

------------------------------------------------------------------------
-- Affinity of a monadic container

module _ {p : Poly} (c : MCOps p) where
  open MCOps c

  ∂Sh : Type
  ∂Sh = Σ[ s ∈ p .Sh ] Fin (p .P s)

  P∂ : ∂Sh → Type
  P∂ (s , i) = Fin∖ (p .P s) i

  -- v_{c,t} := t at the hole, ι elsewhere
  v∂ : (c : ∂Sh) → p .Sh → Fin (p .P (c .fst)) → p .Sh
  v∂ (s , i) t = fillAt i (λ _ → ι) t

  σ∂ : ∂Sh → p .Sh → p .Sh
  σ∂ (s , i) t = σ s (v∂ (s , i) t)

  split : (c : ∂Sh) (t : p .Sh) → Σ[ q ∈ Fin (p .P (c .fst)) ] Fin (p .P (v∂ c t q))
    → (Σ[ q ∈ P∂ c ] Fin (p .P ι)) ⊎ Fin (p .P t)
  split (s , i) t (q , m) with discreteFin q i
  ... | yes _ = inr m
  ... | no ne = inl ((q , ne) , m)

  pr∂ : (c : ∂Sh) (t : p .Sh) → Fin (p .P (σ∂ c t)) → (Σ[ q ∈ P∂ c ] Fin (p .P ι)) ⊎ Fin (p .P t)
  pr∂ c t q = split c t (pr (c .fst) (v∂ c t) q)

  -- Affine := π₁ ∘ pr∂ is injective on pr∂⁻¹(inl).
  Affine : Type
  Affine = ∀ (c : ∂Sh) (t : p .Sh) (q q' : Fin (p .P (σ∂ c t))) (a a' : Σ[ r ∈ P∂ c ] Fin (p .P ι))
    → pr∂ c t q ≡ inl a → pr∂ c t q' ≡ inl a' → a .fst ≡ a' .fst → q ≡ q'

  -- Equivalently: pr₁ is injective off the hole.
  Inj₁ : (s : p .Sh) (i : Fin (p .P s)) (v : Fin (p .P s) → p .Sh) → Type
  Inj₁ s i v = ∀ q q' → ↖ s v q ≡ ↖ s v q' → ¬ (↖ s v q ≡ i) → q ≡ q'

  private
    split-inl : (c : ∂Sh) (t : p .Sh) (x : Σ[ q ∈ Fin (p .P (c .fst)) ] Fin (p .P (v∂ c t q)))
      (a : Σ[ q ∈ P∂ c ] Fin (p .P ι)) → split c t x ≡ inl a → x .fst ≡ a .fst .fst
    split-inl (s , i) t (q , m) a e with discreteFin q i
    ... | yes _ = ⊥.rec (inr≢inl e)
    ... | no ne = cong (fst ∘ fst) (inl-inj e)

    split-off : (c : ∂Sh) (t : p .Sh) (x : Σ[ q ∈ Fin (p .P (c .fst)) ] Fin (p .P (v∂ c t q)))
      → ¬ (x .fst ≡ c .snd) → Σ[ a ∈ Σ[ q ∈ P∂ c ] Fin (p .P ι) ] (split c t x ≡ inl a) × (a .fst .fst ≡ x .fst)
    split-off (s , i) t (q , m) ne with discreteFin q i
    ... | yes e   = ⊥.rec (ne e)
    ... | no ne'  = ((q , ne') , m) , refl , refl

  Affine→Inj₁ : Affine → ∀ s i t → Inj₁ s i (v∂ (s , i) t)
  Affine→Inj₁ aff s i t q q' e ne =
    let (a  , x  , ea)  = split-off (s , i) t (pr s v q) ne
        (a' , x' , ea') = split-off (s , i) t (pr s v q') (λ y → ne (e ∙ y))
    in aff (s , i) t q q' a a' x x' (Σ≡Prop (λ _ → isProp¬ _) (ea ∙ e ∙ sym ea'))
    where v = v∂ (s , i) t

  Inj₁→Affine : (∀ s i t → Inj₁ s i (v∂ (s , i) t)) → Affine
  Inj₁→Affine inj (s , i) t q q' a a' x x' af =
    inj s i t q q' (e ∙ cong fst af ∙ sym e') (λ y → a .fst .snd (sym e ∙ y))
    where
    v = v∂ (s , i) t
    e  = split-inl (s , i) t (pr s v q) a x
    e' = split-inl (s , i) t (pr s v q') a' x'

-- Bridge to the polynomial presentation

module Bridge (𝒯 : Theory) (cm : ContainerModel 𝒯) where
  open Theory 𝒯
  open ContainerModel cm
  import CBPV.OperationalModel as Poly

  om : Poly.OperationalModel 𝒯
  om = toPoly 𝒯 cm

  open import CBPV.Syntax sig using (VTy; F; ret)
  open import CBPV.Reduction sig using (CVal; CComp)
  open import CBPV.Denotation 𝒯 free using (⟦_⟧V; ρ∅)
  open import CBPV.ConfigReduction 𝒯 om using (NoCopy; NoCopyOps; Config; _↦T_; _↦T*_; den-c)
  import CBPV.ConfigSN 𝒯 om as SN
  open import CBPV.AffinityShapes 𝒯 om using (stepG; NoCopyShapes; shapes→ops; ops→shapes)

  -- The context of stepG s i (t , _): neighbors η(inl p), t at the hole.
  private
    G : (s : poly .Sh) (i : Fin (poly .P s)) (t : poly .Sh) → Fin (poly .P s) → T (Fin∖ (poly .P s) i ⊎ Unit)
    G s i t = fillAt i (λ q → ηₘ (inl q)) (t , λ _ → inr tt)

    stepG≡ : ∀ s i t → stepG {Y = Unit} s i (t , λ _ → tt) ≡ μₘ (s , G s i t)
    stepG≡ s i t =
      cong (λ x → μ ((s , i , ηₘ ∘ inl) ⟪ x ⟫)) (map-poly inr (t , λ _ → tt)) ∙ sym (μₘ≡μ _)

    G-inl : ∀ s i t j m (q : Fin∖ (poly .P s) i) → G s i t j .snd m ≡ inl q → j ≡ q .fst
    G-inl s i t j m q e with discreteFin j i
    ... | yes _ = ⊥.rec (inr≢inl e)
    ... | no _  = cong fst (inl-inj e)

    G-off : ∀ s i t j m (ne : ¬ j ≡ i) → G s i t j .snd m ≡ inl (j , ne)
    G-off s i t j m ne with discreteFin j i
    ... | yes e   = ⊥.rec (ne e)
    ... | no ne'  = cong (λ z → inlU (j , z)) (isProp¬ _ ne' ne)
      where
      inlU : Fin∖ (poly .P s) i → Fin∖ (poly .P s) i ⊎ Unit
      inlU = inl

    v-path : ∀ s i t → (λ j → G s i t j .fst) ≡ v∂ cops (s , i) t
    v-path s i t = funExt (fill-map fst i (λ q → ηₘ (inl q)) (t , λ _ → inr tt))

    nocopy→inj : ∀ s i t → NoCopy (μₘ (s , G s i t)) → Inj₁ cops s i (λ j → G s i t j .fst)
    nocopy→inj s i t nc q q' e ne =
      nc q q' (↖ s v q , ne) (G-off s i t _ _ ne)
        (G-off s i t _ _ (λ y → ne (e ∙ y)) ∙ cong inlP (Σ≡Prop (λ _ → isProp¬ _) (sym e)))
      where
      v = λ j → G s i t j .fst
      inlP : Fin∖ (poly .P s) i → Fin∖ (poly .P s) i ⊎ Unit
      inlP = inl

    inj→nocopy : ∀ s i t → Inj₁ cops s i (λ j → G s i t j .fst) → NoCopy (μₘ (s , G s i t))
    inj→nocopy s i t inj q q' r x x' =
      inj q q' (G-inl s i t _ _ r x ∙ sym (G-inl s i t _ _ r x')) (λ y → r .snd (sym (G-inl s i t _ _ r x) ∙ y))

  affine→shapes : Affine cops → NoCopyShapes
  affine→shapes aff s i t =
    subst NoCopy (sym (stepG≡ s i t))
      (inj→nocopy s i t (subst (Inj₁ cops s i) (sym (v-path s i t)) (Affine→Inj₁ cops aff s i t)))

  shapes→affine : NoCopyShapes → Affine cops
  shapes→affine nc = Inj₁→Affine cops λ s i t →
    subst (Inj₁ cops s i) (v-path s i t) (nocopy→inj s i t (subst NoCopy (stepG≡ s i t) (nc s i t)))

  -- Container affinity is exactly what the termination proof uses.
  affine⇔nocopy : (Affine cops → NoCopyOps) × (NoCopyOps → Affine cops)
  affine⇔nocopy = shapes→ops ∘ affine→shapes , shapes→affine ∘ ops→shapes

  -- Configuration strong normalization and adequacy, for affine containers.
  config-sn : Affine cops → {A : VTy} (t : Config A) → SN.SN-T t
  config-sn aff = SN.config-sn (affine⇔nocopy .fst aff)

  config-adequacy : Affine cops → {A : VTy} (M : CComp (F A))
    → SN.SN-T (η M)
    × (∀ t → η M ↦T* t → (∀ t' → ¬ t ↦T t')
         → Σ[ ṽ ∈ T (CVal A) ] (t ≡ pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M))
    × (Σ[ ṽ ∈ T (CVal A) ] (η M ↦T* pmap ret ṽ) × (pmap (λ V → ⟦ V ⟧V ρ∅) ṽ ≡ den-c M))
  config-adequacy aff = SN.config-adequacy (affine⇔nocopy .fst aff)

  ----------------------------------------------------------------------
  -- The derivative Effect rule in container form

  module _ {X : Type} (s : poly .Sh) (i : Fin (poly .P s)) (γ : Fin∖ (poly .P s) i → X)
    (o : Op) (k : Fin (ar o) → X) where

    -- the context of the right-hand side
    Gₑ : Fin (poly .P s) → T X
    Gₑ = fillAt i (ηₘ ∘ γ) (pmap k (gen o))

    effect-container : μ ((∂map η (s , i , γ)) ⟪ ops o (η ∘ k) ⟫) ≡ μₘ (s , Gₑ)
    effect-container =
      sym (μₘ≡μ _) ∙ cong (λ x → μₘ ((s , i , ηₘ ∘ γ) ⟪ x ⟫)) (μmapη (pmap k (gen o)))

    -- its shape is σ∂((s, i), s_op)
    effect-shape : μₘ (s , Gₑ) .fst ≡ σ∂ cops (s , i) (gen o .fst)
    effect-shape = cong (σ s) (funExt (fill-map fst i (ηₘ ∘ γ) (pmap k (gen o))))

    -- off the hole, position q holds the neighbor γ(pr₁ q) ...
    effect-off : (q : Fin (poly .P (μₘ (s , Gₑ) .fst))) (ne : ¬ ↖ s (fst ∘ Gₑ) q ≡ i)
      → μₘ (s , Gₑ) .snd q ≡ γ (↖ s (fst ∘ Gₑ) q , ne)
    effect-off q ne = go (↖ s (fst ∘ Gₑ) q) (↗ s (fst ∘ Gₑ) q) ne
      where
      go : (j : Fin (poly .P s)) (m : Fin (poly .P (Gₑ j .fst))) (ne : ¬ j ≡ i) → Gₑ j .snd m ≡ γ (j , ne)
      go j m ne with discreteFin j i
      ... | yes e   = ⊥.rec (ne e)
      ... | no ne'  = cong (λ z → γ (j , z)) (isProp¬ _ ne' ne)

    -- ... and at the hole a continuation k(ℓ_op m).
    effect-hole : (q : Fin (poly .P (μₘ (s , Gₑ) .fst))) → ↖ s (fst ∘ Gₑ) q ≡ i
      → Σ[ m ∈ Fin (poly .P (gen o .fst)) ] μₘ (s , Gₑ) .snd q ≡ k (gen o .snd m)
    effect-hole q e = go (↖ s (fst ∘ Gₑ) q) (↗ s (fst ∘ Gₑ) q) e
      where
      go : (j : Fin (poly .P s)) (m : Fin (poly .P (Gₑ j .fst))) → j ≡ i
        → Σ[ m' ∈ Fin (poly .P (gen o .fst)) ] Gₑ j .snd m ≡ k (gen o .snd m')
      go j m e with discreteFin j i
      ... | yes _  = m , refl
      ... | no ne  = ⊥.rec (ne e)
