{-
  Finitely branching polynomials p = (Sh, P), ⟦p⟧(X) := Σ (s : Sh) (X^P(s)),
  and their one-hole contexts, the derivative ∂p: a shape s, a hole
  position i, and fillers for the other positions Fin∖ n i := P(s) ∖ {i}.
  plug decides j ≡ i with discreteFin, so it computes on closed positions.
-}
module CBPV.Polynomial where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin; discreteFin)
open import Cubical.Data.Sigma
open import Cubical.Relation.Nullary
import Cubical.Data.Empty as ⊥

record Poly : Type₁ where
  field
    Sh : Type
    P  : Sh → ℕ

open Poly public

⟦_⟧ : Poly → Type → Type
⟦ p ⟧ X = Σ[ s ∈ p .Sh ] (Fin (p .P s) → X)

pmap : {p : Poly} {X Y : Type} → (X → Y) → ⟦ p ⟧ X → ⟦ p ⟧ Y
pmap f (s , γ) = s , f ∘ γ

-- The positions of Fin n other than i.
Fin∖ : (n : ℕ) → Fin n → Type
Fin∖ n i = Σ[ j ∈ Fin n ] ¬ (j ≡ i)

-- ∂p(X): one-hole contexts.
∂ : Poly → Type → Type
∂ p X = Σ[ s ∈ p .Sh ] Σ[ i ∈ Fin (p .P s) ] (Fin∖ (p .P s) i → X)

∂map : {p : Poly} {X Y : Type} → (X → Y) → ∂ p X → ∂ p Y
∂map f (s , i , γ) = s , i , f ∘ γ

-- Fill position i with x and every other position j from γ.
fillAt : {n : ℕ} {X : Type} (i : Fin n) → (Fin∖ n i → X) → X → Fin n → X
fillAt i γ x j with discreteFin j i
... | yes _  = x
... | no j≢i = γ (j , j≢i)

-- C[x]
plug : {p : Poly} {X : Type} → ∂ p X → X → ⟦ p ⟧ X
plug (s , i , γ) x = s , fillAt i γ x

fill-hole : {n : ℕ} {X : Type} (i : Fin n) (γ : Fin∖ n i → X) (x : X)
  → fillAt i γ x i ≡ x
fill-hole i γ x with discreteFin i i
... | yes _ = refl
... | no ¬p = ⊥.rec (¬p refl)

fill-other : {n : ℕ} {X : Type} (i : Fin n) (γ : Fin∖ n i → X) (x : X) (j : Fin n) (ne : ¬ j ≡ i)
  → fillAt i γ x j ≡ γ (j , ne)
fill-other i γ x j ne with discreteFin j i
... | yes p   = ⊥.rec (ne p)
... | no ne'  = cong (λ q → γ (j , q)) (isProp¬ _ ne' ne)

fill-map : {n : ℕ} {X Y : Type} (f : X → Y) (i : Fin n) (γ : Fin∖ n i → X) (x : X)
  → (j : Fin n) → f (fillAt i γ x j) ≡ fillAt i (f ∘ γ) (f x) j
fill-map f i γ x j with discreteFin j i
... | yes _ = refl
... | no _  = refl

-- T(f)(C[x]) = (∂f C)[f x]
plug-map : {p : Poly} {X Y : Type} (f : X → Y) (C : ∂ p X) (x : X)
  → pmap f (plug C x) ≡ plug (∂map f C) (f x)
plug-map f (s , i , γ) x = ΣPathP (refl , funExt (fill-map f i γ x))

-- Every configuration is a one-hole context plugged with the subterm at
-- any chosen position.
unplug : {p : Poly} {X : Type} (t : ⟦ p ⟧ X) → Fin (p .P (t .fst)) → ∂ p X
unplug (s , γ) i = s , i , γ ∘ fst

fill-unplug : {n : ℕ} {X : Type} (γ : Fin n → X) (i j : Fin n)
  → fillAt i (γ ∘ fst) (γ i) j ≡ γ j
fill-unplug γ i j with discreteFin j i
... | yes j≡i = cong γ (sym j≡i)
... | no _    = refl

plug-unplug : {p : Poly} {X : Type} (t : ⟦ p ⟧ X) (i : Fin (p .P (t .fst)))
  → plug (unplug t i) (t .snd i) ≡ t
plug-unplug (s , γ) i = ΣPathP (refl , funExt (fill-unplug γ i))
