{-
  Dershowitz–Manna for finite families.

  A finite multiset over X is a family (n , f) with f : Fin n → X.
  Y ⋖ Z (Y is a multiset step below Z): one element Z(at) is replaced by
  finitely many elements below it, and every other element of Z is kept at
  most once or dropped.

    acc-plus   Acc R x → Acc R⁺ x
    reindex    an injective reindexing of a family has no more predecessors
    dm         if every element is accessible for <, the family is
               accessible for ⋖
-}
module CBPV.Multiset where

------------------------------------------------------------------------
-- Searching a finite family of alternatives.

findFin : {n : ℕ} {P Q : Fin n → Type} → (∀ j → P j ⊎ Q j) → (Σ[ j ∈ Fin n ] P j) ⊎ (∀ j → Q j)
findFin {zero}  d = inr λ ()
findFin {suc n} d with d zero | findFin (λ j → d (suc j))
... | inl p | _            = inl (zero , p)
... | inr q | inl (j , p)  = inl (suc j , p)
... | inr q | inr qs       = inr λ { zero → q ; (suc j) → qs j }

------------------------------------------------------------------------
-- Transitive closure

module _ {X : Type} where

  data TC (R : X → X → Type) : X → X → Type where
    [_] : {y x : X} → R y x → TC R y x
    _⊳_ : {y z x : X} → TC R y z → R z x → TC R y x

  acc-plus : {R : X → X → Type} {x : X} → Acc R x → Acc (TC R) x
  acc-down : {R : X → X → Type} {x y : X} → Acc R x → TC R y x → Acc (TC R) y
  acc-plus a = acc λ y p → acc-down a p
  acc-down (acc r) [ s ]   = acc-plus (r _ s)
  acc-down (acc r) (p ⊳ s) = acc-down (r _ s) p

------------------------------------------------------------------------
-- Positions: an old element (inl) or a new one (inr)

module _ {n : ℕ} {at : Fin n} where

  Same : Fin∖ n at ⊎ Unit → Fin∖ n at ⊎ Unit → Type
  Same (inl p) (inl p') = p .fst ≡ p' .fst
  Same (inl p) (inr _)  = ⊥.⊥
  Same (inr _) _        = ⊥.⊥

  IsNew IsOld : Fin∖ n at ⊎ Unit → Type
  IsNew (inl _) = ⊥.⊥
  IsNew (inr _) = Unit
  IsOld (inl _) = Unit
  IsOld (inr _) = ⊥.⊥

  newOrOld : (v : Fin∖ n at ⊎ Unit) → IsNew v ⊎ IsOld v
  newOrOld (inl _) = inr tt
  newOrOld (inr _) = inl tt

------------------------------------------------------------------------
-- The multiset step

module _ {X : Type} (_<_ : X → X → Type) where

  Fam : Type
  Fam = Σ[ n ∈ ℕ ] (Fin n → X)

  -- What may sit at a position of the smaller family: an old element, or
  -- a new element below the replaced one.
  Ok : (Z : Fam) (at : Fin (Z .fst)) → X → Fin∖ (Z .fst) at ⊎ Unit → Type
  Ok Z at y (inl p) = y ≡ Z .snd (p .fst)
  Ok Z at y (inr _) = y < Z .snd at

  record _⋖_ (Y Z : Fam) : Type where
    constructor mk⋖
    field
      at  : Fin (Z .fst)
      src : Fin (Y .fst) → Fin∖ (Z .fst) at ⊎ Unit
      ok  : ∀ j → Ok Z at (Y .snd j) (src j)
      inj : ∀ j j' → Same (src j) (src j') → j ≡ j'

  ----------------------------------------------------------------------
  -- Reindexing

  -- Every predecessor of an injectively reindexed subfamily is a
  -- predecessor of the whole family.
  reindex : (Y Z : Fam) (τ : Fin (Y .fst) → Fin (Z .fst))
    → (∀ j j' → τ j ≡ τ j' → j ≡ j') → (∀ j → Y .snd j ≡ Z .snd (τ j))
    → ∀ W → W ⋖ Y → W ⋖ Z
  reindex Y Z τ τinj e W (mk⋖ at src ok inj) = mk⋖ (τ at) src' ok' inj'
    where
    liftP : Fin∖ (Y .fst) at → Fin∖ (Z .fst) (τ at)
    liftP (p , ne) = τ p , λ q → ne (τinj p at q)
    mapv : Fin∖ (Y .fst) at ⊎ Unit → Fin∖ (Z .fst) (τ at) ⊎ Unit
    mapv = ⊎.map liftP (λ u → u)
    src' : Fin (W .fst) → Fin∖ (Z .fst) (τ at) ⊎ Unit
    src' j = mapv (src j)
    okL : (w : X) (v : Fin∖ (Y .fst) at ⊎ Unit) → Ok Y at w v → Ok Z (τ at) w (mapv v)
    okL w (inl (p , ne)) o = o ∙ e p
    okL w (inr u)        o = subst (w <_) (e at) o
    ok' : ∀ j → Ok Z (τ at) (W .snd j) (src' j)
    ok' j = okL (W .snd j) (src j) (ok j)
    sameL : (v v' : Fin∖ (Y .fst) at ⊎ Unit) → Same (mapv v) (mapv v') → Same v v'
    sameL (inl (p , _)) (inl (p' , _)) q = τinj p p' q
    sameL (inl _)       (inr _)        ()
    sameL (inr _)       _              ()
    inj' : ∀ j j' → Same (src' j) (src' j') → j ≡ j'
    inj' j j' s = inj j j' (sameL (src j) (src j') s)

  acc-reindex : (Y Z : Fam) (τ : Fin (Y .fst) → Fin (Z .fst))
    → (∀ j j' → τ j ≡ τ j' → j ≡ j') → (∀ j → Y .snd j ≡ Z .snd (τ j))
    → Acc _⋖_ Z → Acc _⋖_ Y
  acc-reindex Y Z τ τinj e (acc r) = acc λ W w → r W (reindex Y Z τ τinj e W w)

  -- Move position j₀ to the front.
  toFront : {k : ℕ} (j₀ : Fin (suc k)) → Fin (suc k) → Fin (suc k)
  toFront j₀ p with discreteFin p j₀
  ... | yes _ = zero
  ... | no ne = suc (punchOut j₀ p ne)

  toFront-inj : {k : ℕ} (j₀ p p' : Fin (suc k)) → toFront j₀ p ≡ toFront j₀ p' → p ≡ p'
  toFront-inj j₀ p p' q with discreteFin p j₀ | discreteFin p' j₀
  ... | yes e | yes e' = e ∙ sym e'
  ... | yes _ | no _   = ⊥.rec (znots q)
  ... | no _  | yes _  = ⊥.rec (snotz q)
  ... | no ne | no ne' =
    sym (punchIn-punchOut j₀ p ne) ∙ cong (punchIn j₀) (injSucFin q) ∙ punchIn-punchOut j₀ p' ne'

  toFront-eq : {k : ℕ} (y : Fin (suc k) → X) (j₀ : Fin (suc k)) (a : X) → y j₀ ≡ a
    → ∀ p → y p ≡ cons a (y ∘ punchIn j₀) (toFront j₀ p)
  toFront-eq y j₀ a e p with discreteFin p j₀
  ... | yes q = cong y q ∙ e
  ... | no ne = cong y (sym (punchIn-punchOut j₀ p ne))

  ----------------------------------------------------------------------
  -- Adding an accessible element to an accessible family

  -- One step of the induction: on x (IHa), and inside on the family (IHg).
  add-step : (x : X)
    → (∀ y → y < x → ∀ {m} {h : Fin m → X} → Acc _⋖_ (m , h) → Acc _⋖_ (suc m , cons y h))
    → {n : ℕ} {g : Fin n → X} → Acc _⋖_ (n , g)
    → (∀ (Y : Fam) → Y ⋖ (n , g) → Acc _⋖_ (suc (Y .fst) , cons x (Y .snd)))
    → Acc _⋖_ (suc n , cons x g)
  add-step x IHa {n} {g} ag@(acc rg) IHg = acc pred
    where
    Z : Fam
    Z = suc n , cons x g

    -- x is replaced: the other elements come from g (injectively) or are below x.
    old-index : (v : Fin∖ (suc n) zero ⊎ Unit) → IsOld v → Fin n
    old-index (inl (zero  , ne)) _ = ⊥.rec (ne refl)
    old-index (inl (suc r , _))  _ = r
    old-index (inr _)            ()

    old-eq : (w : X) (v : Fin∖ (suc n) zero ⊎ Unit) (o : IsOld v) → Ok Z zero w v → w ≡ g (old-index v o)
    old-eq w (inl (zero  , ne)) _ _ = ⊥.rec (ne refl)
    old-eq w (inl (suc r , _))  _ e = e
    old-eq w (inr _)            ()

    old-same : (v v' : Fin∖ (suc n) zero ⊎ Unit) (o : IsOld v) (o' : IsOld v')
      → old-index v o ≡ old-index v' o' → Same v v'
    old-same (inl (zero  , ne)) _                   _ _ _ = ⊥.rec (ne refl)
    old-same (inl (suc r , _))  (inl (zero  , ne')) _ _ _ = ⊥.rec (ne' refl)
    old-same (inl (suc r , _))  (inl (suc r' , _))  _ _ e = cong suc e
    old-same (inl (suc r , _))  (inr _)             _ ()
    old-same (inr _)            _                   ()

    new-lt : (w : X) (v : Fin∖ (suc n) zero ⊎ Unit) → IsNew v → Ok Z zero w v → w < x
    new-lt w (inl _) ()
    new-lt w (inr _) _ o = o

    mix : (k : ℕ) (y : Fin k → X) (src : Fin k → Fin∖ (suc n) zero ⊎ Unit)
      → (∀ j → Ok Z zero (y j) (src j)) → (∀ j j' → Same (src j) (src j') → j ≡ j')
      → Acc _⋖_ (k , y)
    mix-new : (k : ℕ) (j₀ : Fin k) (y : Fin k → X) (src : Fin k → Fin∖ (suc n) zero ⊎ Unit)
      → IsNew (src j₀)
      → (∀ j → Ok Z zero (y j) (src j)) → (∀ j j' → Same (src j) (src j') → j ≡ j')
      → Acc _⋖_ (k , y)

    mix k y src ok inj with findFin (λ j → newOrOld (src j))
    ... | inl (j₀ , nw) = mix-new k j₀ y src nw ok inj
    ... | inr olds =
      acc-reindex (k , y) (n , g) (λ j → old-index (src j) (olds j))
        (λ j j' e → inj j j' (old-same (src j) (src j') (olds j) (olds j') e))
        (λ j → old-eq (y j) (src j) (olds j) (ok j)) ag

    mix-new zero    ()
    mix-new (suc k') j₀ y src nw ok inj =
      acc-reindex (suc k' , y) (suc k' , cons (y j₀) (y ∘ punchIn j₀))
        (toFront j₀) (toFront-inj j₀) (toFront-eq y j₀ (y j₀) refl)
        (IHa (y j₀) (new-lt (y j₀) (src j₀) nw (ok j₀))
          (mix k' (y ∘ punchIn j₀) (src ∘ punchIn j₀) (ok ∘ punchIn j₀)
            (λ r r' s → punchIn-inj j₀ r r' (inj _ _ s))))

    -- an element of g is replaced: x is kept (at most once) or dropped.
    IsX NotX : {q : Fin n} → Fin∖ (suc n) (suc q) ⊎ Unit → Type
    IsX (inl (zero  , _)) = Unit
    IsX (inl (suc _ , _)) = ⊥.⊥
    IsX (inr _)           = ⊥.⊥
    NotX (inl (zero  , _)) = ⊥.⊥
    NotX (inl (suc _ , _)) = Unit
    NotX (inr _)           = Unit

    xOrNot : {q : Fin n} (v : Fin∖ (suc n) (suc q) ⊎ Unit) → IsX v ⊎ NotX v
    xOrNot (inl (zero  , _)) = inl tt
    xOrNot (inl (suc _ , _)) = inr tt
    xOrNot (inr _)           = inr tt

    both-x : {q : Fin n} (v v' : Fin∖ (suc n) (suc q) ⊎ Unit) → IsX v → IsX v' → Same v v'
    both-x (inl (zero  , _)) (inl (zero  , _)) _ _ = refl
    both-x (inl (zero  , _)) (inl (suc _ , _)) _ ()
    both-x (inl (zero  , _)) (inr _)           _ ()
    both-x (inl (suc _ , _)) _                 ()
    both-x (inr _)           _                 ()

    x-eq : {q : Fin n} (w : X) (v : Fin∖ (suc n) (suc q) ⊎ Unit) → IsX v → Ok Z (suc q) w v → w ≡ x
    x-eq w (inl (zero  , _)) _ e = e
    x-eq w (inl (suc _ , _)) ()
    x-eq w (inr _)           ()

    shift : {q : Fin n} (v : Fin∖ (suc n) (suc q) ⊎ Unit) → NotX v → Fin∖ n q ⊎ Unit
    shift (inl (zero  , _))  ()
    shift (inl (suc r , ne)) _ = inl (r , λ e → ne (cong suc e))
    shift (inr u)            _ = inr u

    ok-shift : {q : Fin n} (w : X) (v : Fin∖ (suc n) (suc q) ⊎ Unit) (nx : NotX v)
      → Ok Z (suc q) w v → Ok (n , g) q w (shift v nx)
    ok-shift w (inl (zero  , _)) ()
    ok-shift w (inl (suc r , _)) _ e = e
    ok-shift w (inr u)           _ o = o

    same-shift : {q : Fin n} (v v' : Fin∖ (suc n) (suc q) ⊎ Unit) (nx : NotX v) (nx' : NotX v')
      → Same (shift v nx) (shift v' nx') → Same v v'
    same-shift (inl (zero  , _)) _                  ()
    same-shift (inl (suc _ , _)) (inl (zero  , _))  _ ()
    same-shift (inl (suc _ , _)) (inl (suc _ , _))  _ _ e = cong suc e
    same-shift (inl (suc _ , _)) (inr _)            _ _ ()
    same-shift (inr _)           _                  _ _ ()

    kept : (k : ℕ) (q : Fin n) (j₀ : Fin k) (y : Fin k → X) (src : Fin k → Fin∖ (suc n) (suc q) ⊎ Unit)
      → IsX (src j₀)
      → (∀ j → Ok Z (suc q) (y j) (src j)) → (∀ j j' → Same (src j) (src j') → j ≡ j')
      → Acc _⋖_ (k , y)
    kept zero    q ()
    kept (suc k') q j₀ y src isx ok inj =
      acc-reindex (suc k' , y) (suc k' , cons x (y ∘ punchIn j₀))
        (toFront j₀) (toFront-inj j₀) (toFront-eq y j₀ x (x-eq (y j₀) (src j₀) isx (ok j₀)))
        (IHg (k' , y ∘ punchIn j₀) (mk⋖ q src' ok' inj'))
      where
      notx : ∀ r → NotX (src (punchIn j₀ r))
      notx r with xOrNot (src (punchIn j₀ r))
      ... | inl isx' =
        ⊥.rec (punchIn-avoid j₀ r (sym (inj j₀ (punchIn j₀ r) (both-x _ _ isx isx'))))
      ... | inr nx = nx
      src' : Fin k' → Fin∖ n q ⊎ Unit
      src' r = shift (src (punchIn j₀ r)) (notx r)
      ok' : ∀ r → Ok (n , g) q (y (punchIn j₀ r)) (src' r)
      ok' r = ok-shift _ (src (punchIn j₀ r)) (notx r) (ok (punchIn j₀ r))
      inj' : ∀ r r' → Same (src' r) (src' r') → r ≡ r'
      inj' r r' s = punchIn-inj j₀ r r' (inj _ _ (same-shift _ _ (notx r) (notx r') s))

    pred : (Y : Fam) → Y ⋖ Z → Acc _⋖_ Y
    pred (k , y) (mk⋖ zero src ok inj) = mix k y src ok inj
    pred (k , y) (mk⋖ (suc q) src ok inj) with findFin (λ j → xOrNot (src j))
    ... | inl (j₀ , isx) = kept k q j₀ y src isx ok inj
    ... | inr nxs =
      rg (k , y) (mk⋖ q (λ j → shift (src j) (nxs j))
        (λ j → ok-shift (y j) (src j) (nxs j) (ok j))
        (λ j j' s → inj j j' (same-shift (src j) (src j') (nxs j) (nxs j') s)))

  add : {x : X} → Acc _<_ x → {n : ℕ} {g : Fin n → X} → Acc _⋖_ (n , g) → Acc _⋖_ (suc n , cons x g)
  add {x} ax@(acc rx) ag@(acc rg) =
    add-step x (λ y lt → add (rx y lt)) ag (λ Y w → add ax (rg Y w))

  ----------------------------------------------------------------------
  -- Dershowitz–Manna

  dm : (n : ℕ) (f : Fin n → X) → (∀ j → Acc _<_ (f j)) → Acc _⋖_ (n , f)
  dm zero    f a = acc λ W w → ⊥.rec (¬Fin0 (_⋖_.at w))
  dm (suc n) f a =
    subst (λ h → Acc _⋖_ (suc n , h)) (cons-η f) (add (a zero) (dm n (f ∘ suc) (a ∘ suc)))
