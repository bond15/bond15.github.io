{-
  The trilemma: if some operation has two distinct argument positions,
  no relation R on closed computations that contains the β-steps is
  at the same time

    closed under stacks       M R M' ⇒ S[M] R S[M']
    closed under operations   Mⱼ R N ⇒ op(…, Mⱼ, …) R op(…, N, …)
    deterministic off ops     L not an operation, L R L₁, L R L₂ ⇒ L₁ = L₂

  Witness: M := force(thunk(ret ⋆)) : F 1 at both positions, and
  S := x ← •; ret x.  S[op(M, M)] steps to S[op(ret ⋆, M)] and to
  S[op(M, ret ⋆)].  Tree reduction ↦ has the first two properties, so
  it is not deterministic (tree-not-det).
-}
open import CBPV.Theory

module CBPV.Trilemma (Sg : Signature) where

open import Cubical.Foundations.Prelude
open import Cubical.Data.Unit
open import Cubical.Data.Bool using (Bool; true; false; true≢false)
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Sigma
open import Cubical.Data.Sum as ⊎ using (_⊎_; inl; inr)
open import Cubical.Data.Nat using (ℕ; isSetℕ)
open import Cubical.Data.FinData using (Fin)
open import Cubical.Relation.Nullary

open Signature Sg
open import CBPV.Fin using (_[_≔_]; upd-yes; upd-no)
open import CBPV.Syntax Sg
open import CBPV.Plug Sg
open import CBPV.Reduction Sg

Rel : Type₁
Rel = ∀ {B} → CComp B → CComp B → Type

NotOp : {B : CTy} → CComp B → Type
NotOp (op _ _) = ⊥
NotOp _        = Unit

module _ (R : Rel) where
  HasBeta StackClosed OpClosed DetOffOps : Type
  HasBeta     = ∀ {B} {M M' : CComp B} → M ↦red M' → R M M'
  StackClosed = ∀ {B B'} (S : CStack B B') {M M' : CComp B} → R M M' → R (plug S M) (plug S M')
  OpClosed    = ∀ {B} (o : Op) (k : Fin (ar o) → CComp B) (j : Fin (ar o)) {N : CComp B}
              → R (k j) N → R (op o k) (op o (k [ j ≔ N ]))
  DetOffOps   = ∀ {B} {L L₁ L₂ : CComp B} → NotOp L → R L L₁ → R L L₂ → L₁ ≡ L₂

private
  isRet : {B : CTy} → CComp B → Bool
  isRet (ret _) = true
  isRet _       = false

  probe : CComp (F 𝟙) → (Σ[ n ∈ ℕ ] (Fin n → Bool)) ⊎ Unit
  probe (bind (op o k) N) = inl (ar o , λ j → isRet (k j))
  probe _                 = inr tt

  inl-inj : {X Y : Type} {a b : X} → Path (X ⊎ Y) (inl a) (inl b) → a ≡ b
  inl-inj {a = a} p = cong (⊎.rec (λ x → x) (λ _ → a)) p

  snd-inj : {n : ℕ} {f g : Fin n → Bool} → Path (Σ[ n ∈ ℕ ] (Fin n → Bool)) (n , f) (n , g) → f ≡ g
  snd-inj {f = f} {g} p = subst (λ q → PathP (λ i → Fin (q i) → Bool) f g) (isSetℕ _ _ (cong fst p) refl) (cong snd p)

trilemma : (o : Op) (j₁ j₂ : Fin (ar o)) → ¬ j₁ ≡ j₂
  → (R : Rel) → HasBeta R → StackClosed R → OpClosed R → DetOffOps R → ⊥
trilemma o j₁ j₂ ne R beta stk opc det = true≢false (sym e₁ ∙ funExt⁻ same j₁ ∙ e₂)
  where
  M₀ : CComp (F 𝟙)
  M₀ = force (thunk (ret ⋆))
  k : Fin (ar o) → CComp (F 𝟙)
  k _ = M₀
  S : CStack (F 𝟙) (F 𝟙)
  S = bindˢ • (ret (var vz))
  step : (j : Fin (ar o)) → R (plug S (op o k)) (plug S (op o (k [ j ≔ ret ⋆ ])))
  step j = stk S (opc o k j (beta (force-thunk (ret ⋆))))
  L≡ : plug S (op o (k [ j₁ ≔ ret ⋆ ])) ≡ plug S (op o (k [ j₂ ≔ ret ⋆ ]))
  L≡ = det tt (step j₁) (step j₂)
  same : (λ j → isRet ((k [ j₁ ≔ ret ⋆ ]) j)) ≡ (λ j → isRet ((k [ j₂ ≔ ret ⋆ ]) j))
  same = snd-inj (inl-inj (cong probe L≡))
  e₁ : isRet ((k [ j₁ ≔ ret ⋆ ]) j₁) ≡ true
  e₁ = cong isRet (upd-yes k j₁ (ret ⋆) refl)
  e₂ : isRet ((k [ j₂ ≔ ret ⋆ ]) j₁) ≡ false
  e₂ = cong isRet (upd-no k j₂ (ret ⋆) ne)

-- Tree reduction is closed under stacks and operations, hence not
-- deterministic off operations.
tree-not-det : (o : Op) (j₁ j₂ : Fin (ar o)) → ¬ j₁ ≡ j₂ → ¬ DetOffOps _↦_
tree-not-det o j₁ j₂ ne = trilemma o j₁ j₂ ne _↦_ (λ r → red (β r)) step-plug opArg
