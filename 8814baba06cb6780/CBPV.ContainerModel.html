{-
  Operational models given by a monadic container.

  A container model of 𝒯 is
    • a monadic container (Sh, P, ι, σ, pr) with the monad laws;
    • a generic effect ⌈op⌉ = ⟨s_op, ℓ_op⟩ : T(Fin ar(op)) for each op,
      giving the operations  ⟦op⟧(a⃗) := μ(T(a⃗)(⌈op⌉));
    • the equations of 𝒯 hold in every T X, and T X is free on X with
      unit the container's η.

  The rest of the development uses the polynomial presentation
  (CBPV.OperationalModel: a free model monad whose functor is
  polynomial).  toPoly and fromPoly convert; both keep the polynomial
  and preserve η, μ and the operations, and fromPoly ∘ toPoly gives back
  ι, σ and the generic effects.

  fromPoly reads the container off the free model monad:
    ι := the shape of η(tt)       σ s v , pr := the shape and positions of μ(G),
  where G := (s, λp. (v p, λm. (p, m))); by naturality μ of any element is
  an image of μ(G), which is the container formula.
-}
open import CBPV.Theory

module CBPV.ContainerModel (𝒯 : Theory) where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Unit
open import Cubical.Data.FinData using (Fin)

open import CBPV.FreeModel
open import CBPV.Polynomial renaming (plug to _⟪_⟫)
open import CBPV.MonadicContainer
import CBPV.OperationalModel as Poly

open Theory 𝒯
open Model

record ContainerModel : Type₁ where
  field
    mc : MonadicContainer

  open MonadicContainer mc public

  T : Type → Type
  T = ⟦ poly ⟧

  field
    -- ⌈op⌉ = ⟨s_op , ℓ_op⟩
    gen : (o : Op) → T (Fin (ar o))

  -- ⟦op⟧(a⃗) := μ(T(a⃗)(⌈op⌉))
  opsᶜ : {X : Type} → AlgStr sig (T X)
  opsᶜ o args = μₘ (pmap args (gen o))

  field
    opsᶜ-sat  : {X : Type} → Sat 𝒯 (opsᶜ {X})
    extᶜ      : {X : Type} (M : Model 𝒯) → (X → M .Carrier) → T X → M .Carrier
    extᶜ-η    : {X : Type} (M : Model 𝒯) (f : X → M .Carrier) (x : X) → extᶜ M f (ηₘ x) ≡ f x
    extᶜ-hom  : {X : Type} (M : Model 𝒯) (f : X → M .Carrier) → IsHom sig opsᶜ (M .alg) (extᶜ M f)
    extᶜ-uniq : {X : Type} (M : Model 𝒯) (f : X → M .Carrier) (h : T X → M .Carrier)
              → IsHom sig opsᶜ (M .alg) h → (∀ x → h (ηₘ x) ≡ f x) → ∀ t → h t ≡ extᶜ M f t

  free : FreeModel 𝒯 T
  free = record
    { η = ηₘ ; ops = opsᶜ ; ops-sat = opsᶜ-sat
    ; ext = extᶜ ; ext-η = extᶜ-η ; ext-hom = extᶜ-hom ; ext-uniq = extᶜ-uniq }

  open FreeModel free public

  -- The container's μ is the free model's.
  μₘ-hom : {X : Type} → IsHom sig ops ops (μₘ {X})
  μₘ-hom .pres o args = μμ (pmap args (gen o))

  μₘ≡μ : {X : Type} (t : T (T X)) → μₘ t ≡ μ t
  μₘ≡μ = ext-uniq (FM _) (λ t → t) μₘ μₘ-hom μη

  -- The functorial action of the free model is the polynomial one.
  map-poly : {X Y : Type} (f : X → Y) (t : T X) → map f t ≡ pmap f t
  map-poly f t = sym (ext-uniq (FM _) (η ∘ f) (pmap f) (ishom λ _ _ → refl) (λ _ → refl) t)

  -- ⌈op⌉ = ⟦op⟧(η y₁, …)
  gen≡ : (o : Op) → ops o η ≡ gen o
  gen≡ o = μmapη (gen o)

------------------------------------------------------------------------
-- Container ⇒ polynomial

toPoly : ContainerModel → Poly.OperationalModel 𝒯
toPoly cm = record
  { poly     = ContainerModel.poly cm
  ; free     = ContainerModel.free cm
  ; map-poly = ContainerModel.map-poly cm }

------------------------------------------------------------------------
-- Polynomial ⇒ container

-- A polynomial model with a monadic container whose η and μ agree with
-- its own, and generic effects ⌈op⌉ = ⟦op⟧(η y₁, …), is a container model.
module FromPolyWith (PF : Poly.OperationalModel 𝒯) where
  open Poly.OperationalModel PF

  module With (c : MCOps poly)
    (η-agree : {X : Type} (x : X) → η x ≡ ηᶜ c x)
    (μ-agree : {X : Type} (t : ⟦ poly ⟧ (⟦ poly ⟧ X)) → μ t ≡ μᶜ c t)
    (genC : (o : Op) → ⟦ poly ⟧ (Fin (ar o)))
    (gen-agree : (o : Op) → genC o ≡ ops o η) where

    laws : MonadLaws c
    laws = record
      { μη    = λ t → sym (μ-agree _) ∙ cong μ (sym (η-agree t)) ∙ μ-η t
      ; μmapη = λ t →
          sym (μ-agree _)
          ∙ cong μ (cong (λ f → pmap f t) (funExt λ x → sym (η-agree x)) ∙ sym (map-poly η t))
          ∙ μ-mapη t
      ; μμ    = λ t →
          sym (μ-agree _) ∙ cong μ (sym (μ-agree t)) ∙ μ-assoc t
          ∙ cong μ (map-poly μ t ∙ cong (λ f → pmap f t) (funExt μ-agree))
          ∙ μ-agree _ }

    mc : MonadicContainer
    mc = record { poly = poly ; cops = c ; laws = laws }

    opsC : {X : Type} → AlgStr sig (⟦ poly ⟧ X)
    opsC o args = μᶜ c (pmap args (genC o))

    -- algebraic operations from generic effects
    ops-agree : {X : Type} (o : Op) (args : Fin (ar o) → ⟦ poly ⟧ X) → opsC o args ≡ ops o args
    ops-agree o args =
      sym (μ-agree _)
      ∙ cong μ (cong (pmap args) (gen-agree o) ∙ sym (map-poly args (ops o η)))
      ∙ cong μ (map-hom args .pres o η)
      ∙ cong (λ a → μ (ops o a)) (funExt λ j → map-η args j)
      ∙ μ-hom .pres o (η ∘ args)
      ∙ cong (ops o) (funExt λ j → μ-η (args j))

    cm : ContainerModel
    cm = record
      { mc        = mc
      ; gen       = genC
      ; opsᶜ-sat  = subst (Sat 𝒯) (sym (funExt λ o → funExt λ args → ops-agree o args)) ops-sat
      ; extᶜ      = ext
      ; extᶜ-η    = λ M f x → cong (ext M f) (sym (η-agree x)) ∙ ext-η M f x
      ; extᶜ-hom  = λ M f → ishom λ o args → cong (ext M f) (ops-agree o args) ∙ ext-hom M f .pres o args
      ; extᶜ-uniq = λ M f h hh hη → ext-uniq M f h
          (ishom λ o args → cong h (sym (ops-agree o args)) ∙ hh .pres o args)
          (λ x → cong h (η-agree x) ∙ hη x) }

-- Polynomial ⇒ container, reading the container off the free model.
module FromPoly (PF : Poly.OperationalModel 𝒯) where
  open Poly.OperationalModel PF

  private
    Pos : poly .Sh → Type
    Pos s = Fin (poly .P s)

    G : (s : poly .Sh) (v : Pos s → poly .Sh) → ⟦ poly ⟧ (⟦ poly ⟧ (Σ[ q ∈ Pos s ] Pos (v q)))
    G s v = s , λ q → v q , λ m → q , m

  cops : MCOps poly
  cops = record
    { ι  = η tt .fst
    ; σ  = λ s v → μ (G s v) .fst
    ; pr = λ s v q → μ (G s v) .snd q }

  η-agree : {X : Type} (x : X) → η x ≡ ηᶜ cops x
  η-agree x = sym (map-η (λ _ → x) tt) ∙ map-poly (λ _ → x) (η tt)

  μ-agree : {X : Type} (t : ⟦ poly ⟧ (⟦ poly ⟧ X)) → μ t ≡ μᶜ cops t
  μ-agree (s , γ) = cong μ (sym mm) ∙ sym (μ-natural h (G s v)) ∙ map-poly h (μ (G s v))
    where
    v = λ q → γ q .fst
    h : (Σ[ q ∈ Pos s ] Pos (v q)) → _
    h (q , m) = γ q .snd m
    mm : map (map h) (G s v) ≡ (s , γ)
    mm = map-poly (map h) (G s v) ∙ cong (λ f → pmap f (G s v)) (funExt (map-poly h))

  open FromPolyWith.With PF cops η-agree μ-agree (λ o → ops o η) (λ _ → refl) public

fromPoly : Poly.OperationalModel 𝒯 → ContainerModel
fromPoly = FromPoly.cm

------------------------------------------------------------------------
-- Agreement

module _ (PF : Poly.OperationalModel 𝒯) where
  private
    module P = Poly.OperationalModel PF
    module O = ContainerModel (fromPoly PF)

  fromPoly-η : {X : Type} (x : X) → O.η x ≡ P.η x
  fromPoly-η x = sym (FromPoly.η-agree PF x)

  fromPoly-μ : {X : Type} (t : _) → O.μₘ {X} t ≡ P.μ t
  fromPoly-μ t = sym (FromPoly.μ-agree PF t)

  fromPoly-ops : {X : Type} (o : Op) (args : _) → O.ops {X} o args ≡ P.ops o args
  fromPoly-ops = FromPoly.ops-agree PF

module _ (cm : ContainerModel) where
  private
    module O  = ContainerModel cm
    module O' = ContainerModel (fromPoly (toPoly cm))

  -- toPoly keeps η, μ and the operations; going around gives back ι, σ, ⌈op⌉.
  roundtrip-ι : O'.ι ≡ O.ι
  roundtrip-ι = refl

  roundtrip-σ : (s : O.poly .Sh) (v : _) → O'.σ s v ≡ O.σ s v
  roundtrip-σ s v = cong fst (sym (O.μₘ≡μ _))

  roundtrip-gen : (o : Op) → O'.gen o ≡ O.gen o
  roundtrip-gen o = O.gen≡ o
