{-
  The strong-normalization logical relation: reducibility
  candidates (Tait, Girard).  Only the definitions and the candidate
  properties CR1–CR4; no Fundamental Lemma here.

    SN M         every ↦ sequence from M is finite (accessibility of M
                 for the converse of ↦)
    Neutral M    M is an elimination form: app, π₁, π₂, bind, force,
                 case+, case×

  A candidate at B (record SCand) is a predicate S on closed
  computations of type B with

    CR1  S M ⇒ SN M
    CR2  S M and M ↦ M' ⇒ S M'
    CR3  Neutral M and S M' for every M ↦ M' ⇒ S M
    CR4  S (k j) for every j ⇒ S (op o k)

  𝒱⟦A⟧ is the value relation, with 𝒱⟦U B⟧ V = 𝒞⟦B⟧ (force V).  The candidates 𝒞⟦B⟧:

    𝒞⟦⊤⟧ M      = SN M
    𝒞⟦A ⇒ B⟧ M  = SN M × (∀ V ∈ 𝒱⟦A⟧. 𝒞⟦B⟧ (M V))
    𝒞⟦B₁ & B₂⟧ M = SN M × 𝒞⟦B₁⟧ (π₁ M) × 𝒞⟦B₂⟧ (π₂ M)
    𝒞⟦F A⟧      = SPred 𝒱⟦A⟧

  where SPred P is generated by ret (P V), op (all arguments related,
  all reducts related) and neutral terms (all reducts related).  Each
  clause of 𝒞⟦_⟧ proves its own CR1–CR4.

  Every proof of CR4 is a nested induction on the family of arguments
  (args-ind): a step of op o k changes exactly one argument (op-step),
  and a family of strongly normalizing terms is accessible for
  one-position updates (Upd-acc).
-}

private variable
  A : VTy
  B B₁ B₂ : CTy

------------------------------------------------------------------------
-- Strong normalization

-- The converse of reduction: M' ↤ M  iff  M ↦ M'.
_↤_ : CComp B → CComp B → Type
M' ↤ M = M ↦ M'

-- SN M: M is accessible for ↤, i.e. every ↦ sequence from M is finite.
SN : CComp B → Type
SN = Acc _↤_

-- SN is closed under reduction.
sn-step : {M M' : CComp B} → SN M → M ↦ M' → SN M'
sn-step (acc r) s = r _ s

------------------------------------------------------------------------
-- Neutral terms: the elimination forms

Neutral : CComp B → Type
Neutral (app _ _)     = Unit
Neutral (π₁ _)        = Unit
Neutral (π₂ _)        = Unit
Neutral (bind _ _)    = Unit
Neutral (force _)     = Unit
Neutral (case+ _ _ _) = Unit
Neutral (case× _ _)   = Unit
Neutral (ret _)       = ⊥
Neutral (lam _)       = ⊥
Neutral (cpair _ _)   = ⊥
Neutral triv          = ⊥
Neutral (op _ _)      = ⊥

------------------------------------------------------------------------
-- Lemma 0: a frame around a neutral term steps only inside the frame.
-- (The redexes at the frame need the term to be λ, ⟨_,_⟩, ret or op.)

app-neutral-step : {M : CComp (A ⇒ B)} {V : CVal A} {X : CComp B}
  → Neutral M → app M V ↦ X → Σ[ M' ∈ CComp (A ⇒ B) ] (M ↦ M') × (X ≡ app M' V)
app-neutral-step () (red (β (app-lam _ _)))
app-neutral-step () (red (alg-app _ _ _))
app-neutral-step n  (appF _ s) = _ , s , refl

π₁-neutral-step : {M : CComp (B₁ & B₂)} {X : CComp B₁}
  → Neutral M → π₁ M ↦ X → Σ[ M' ∈ CComp (B₁ & B₂) ] (M ↦ M') × (X ≡ π₁ M')
π₁-neutral-step () (red (β (π₁-pair _ _)))
π₁-neutral-step () (red (alg-π₁ _ _))
π₁-neutral-step n  (π₁F s) = _ , s , refl

π₂-neutral-step : {M : CComp (B₁ & B₂)} {X : CComp B₂}
  → Neutral M → π₂ M ↦ X → Σ[ M' ∈ CComp (B₁ & B₂) ] (M ↦ M') × (X ≡ π₂ M')
π₂-neutral-step () (red (β (π₂-pair _ _)))
π₂-neutral-step () (red (alg-π₂ _ _))
π₂-neutral-step n  (π₂F s) = _ , s , refl

bind-neutral-step : {M : CComp (F A)} {N : Comp (ε ▸ A) B} {X : CComp B}
  → Neutral M → bind M N ↦ X → Σ[ M' ∈ CComp (F A) ] (M ↦ M') × (X ≡ bind M' N)
bind-neutral-step () (red (β (bind-ret _ _)))
bind-neutral-step () (red (alg-bind _ _ _))
bind-neutral-step n  (bindF _ s) = _ , s , refl

------------------------------------------------------------------------
-- Case analysis of steps (for CR4)

-- A step of op o k changes one argument (op-step).
op-elim : (G : CComp B → Type) {o : Op} {k : Fin (ar o) → CComp B}
  → ((j : Fin (ar o)) {N : CComp B} → k j ↦ N → G (op o (k [ j ≔ N ])))
  → {M' : CComp B} → op o k ↦ M' → G M'
op-elim G g s with op-step s
... | j , N , s' , p = subst G (sym p) (g j s')

-- A step of (op o k) V moves the operation out, or is a step of op o k.
app-op-elim : (G : CComp B → Type) {o : Op} {k : Fin (ar o) → CComp (A ⇒ B)} {V : CVal A}
  → G (op o (λ j → app (k j) V))
  → ({M' : CComp (A ⇒ B)} → op o k ↦ M' → G (app M' V))
  → {X : CComp B} → app (op o k) V ↦ X → G X
app-op-elim G ga gf (red (β ()))
app-op-elim G ga gf (red (alg-app _ _ _)) = ga
app-op-elim G ga gf (appF _ s)            = gf s

-- A step of πᵢ (op o k) moves the operation out, or is a step of op o k.
π₁-op-elim : (G : CComp B₁ → Type) {o : Op} {k : Fin (ar o) → CComp (B₁ & B₂)}
  → G (op o (λ j → π₁ (k j)))
  → ({M' : CComp (B₁ & B₂)} → op o k ↦ M' → G (π₁ M'))
  → {X : CComp B₁} → π₁ (op o k) ↦ X → G X
π₁-op-elim G ga gf (red (β ()))
π₁-op-elim G ga gf (red (alg-π₁ _ _)) = ga
π₁-op-elim G ga gf (π₁F s)            = gf s

π₂-op-elim : (G : CComp B₂ → Type) {o : Op} {k : Fin (ar o) → CComp (B₁ & B₂)}
  → G (op o (λ j → π₂ (k j)))
  → ({M' : CComp (B₁ & B₂)} → op o k ↦ M' → G (π₂ M'))
  → {X : CComp B₂} → π₂ (op o k) ↦ X → G X
π₂-op-elim G ga gf (red (β ()))
π₂-op-elim G ga gf (red (alg-π₂ _ _)) = ga
π₂-op-elim G ga gf (π₂F s)            = gf s

------------------------------------------------------------------------
-- Induction on a family of arguments, one argument at a time

-- Updating one position of a finite family along R.
module _ {X : Type} (R : X → X → Type) where

  Upd : (n : ℕ) → (Fin n → X) → (Fin n → X) → Type
  Upd n g f = Σ[ j ∈ Fin n ] Σ[ x ∈ X ] R x (f j) × (g ≡ f [ j ≔ x ])

  private
    acc-cons : {n : ℕ} {x : X} {g : Fin n → X}
      → Acc R x → Acc (Upd n) g → Acc (Upd (suc n)) (cons x g)
    acc-cons {n} {x} {g} ax@(acc rx) ag@(acc rg) = acc λ
      { h (zero , y , r , p) →
          subst (Acc (Upd (suc n))) (sym (p ∙ upd-zero (cons x g) y))
            (acc-cons (rx y r) ag)
      ; h (suc j , y , r , p) →
          subst (Acc (Upd (suc n))) (sym (p ∙ upd-cons-suc x g j y))
            (acc-cons ax (rg (g [ j ≔ y ]) (j , y , r , refl))) }

  -- A family of accessible elements is accessible for one-position updates.
  Upd-acc : (n : ℕ) (f : Fin n → X) → (∀ j → Acc R (f j)) → Acc (Upd n) f
  Upd-acc zero    f _ = acc λ { _ (() , _) }
  Upd-acc (suc n) f a =
    subst (Acc (Upd (suc n))) (cons-η f)
      (acc-cons (a zero) (Upd-acc n (f ∘ suc) (a ∘ suc)))

-- If every entry of f and x satisfy P, so does every entry of f [ j ≔ x ].
upd-all : {X : Type} {n : ℕ} (P : X → Type) (f : Fin n → X) (j : Fin n) (x : X)
  → (∀ i → P (f i)) → P x → ∀ i → P ((f [ j ≔ x ]) i)
upd-all P f j x c p i with discreteFin i j
... | yes _ = p
... | no _  = c i

-- Induction on a family k of terms satisfying a predicate S that
-- implies SN and is closed under reduction: to prove G k, one may
-- assume G for every family obtained by one step of one argument.
module _ {B : CTy} (S : CComp B → Type)
  (S-sn : {M : CComp B} → S M → SN M)
  (S-step : {M M' : CComp B} → S M → M ↦ M' → S M')
  (n : ℕ) (G : (Fin n → CComp B) → Type)
  (h : (k : Fin n → CComp B) → (∀ j → S (k j))
     → ((j : Fin n) {N : CComp B} → k j ↦ N → G (k [ j ≔ N ]))
     → G k) where

  private
    go : (k : Fin n → CComp B) → (∀ j → S (k j)) → Acc (Upd _↤_ n) k → G k
    go k c (acc r) = h k c λ j {N} s →
      go (k [ j ≔ N ]) (upd-all S k j N c (S-step (c j) s)) (r _ (j , N , s , refl))

  args-ind : (k : Fin n → CComp B) → (∀ j → S (k j)) → G k
  args-ind k c = go k c (Upd-acc _↤_ n k (λ j → S-sn (c j)))

-- SN under operations.
sn-op : (o : Op) {k : Fin (ar o) → CComp B} → (∀ j → SN (k j)) → SN (op o k)
sn-op o {k} =
  args-ind SN (λ a → a) sn-step (ar o) (λ k → SN (op o k))
    (λ k _ ih → acc λ M' s → op-elim SN ih s) k

------------------------------------------------------------------------
-- Candidates

record SCand (B : CTy) : Type₁ where
  field
    S              : CComp B → Type
    sn             : {M : CComp B} → S M → SN M                                        -- CR1
    step-closed    : {M M' : CComp B} → S M → M ↦ M' → S M'                            -- CR2
    neutral-closed : {M : CComp B} → Neutral M → (∀ M' → M ↦ M' → S M') → S M          -- CR3
    op-closed      : (o : Op) {k : Fin (ar o) → CComp B} → (∀ j → S (k j)) → S (op o k) -- CR4
open SCand public

------------------------------------------------------------------------
-- The candidate at F A.  A closed computation of type F A is ret, op or
-- neutral; the reduct premise of sop makes every reduct's derivation a
-- sub-derivation (ret has no reducts).

data SPred (P : CVal A → Type) : CComp (F A) → Type where
  sret : {V : CVal A} → P V → SPred P (ret V)
  sop  : (o : Op) {k : Fin (ar o) → CComp (F A)}
       → (∀ j → SPred P (k j)) → (∀ M' → op o k ↦ M' → SPred P M') → SPred P (op o k)
  sneu : {M : CComp (F A)} → Neutral M → (∀ M' → M ↦ M' → SPred P M') → SPred P M

module _ {P : CVal A → Type} where

  -- CR1 at F A, by induction on the derivation.
  spred-sn : {M : CComp (F A)} → SPred P M → SN M
  spred-sn (sret v)    = acc λ _ s → ⊥.rec (ret-no-step s)
  spred-sn (sop o c r) = acc λ M' s → spred-sn (r M' s)
  spred-sn (sneu n r)  = acc λ M' s → spred-sn (r M' s)

  -- CR2 at F A, from the reduct premises.
  spred-step : {M M' : CComp (F A)} → SPred P M → M ↦ M' → SPred P M'
  spred-step (sret v)    s = ⊥.rec (ret-no-step s)
  spred-step (sop o c r) s = r _ s
  spred-step (sneu n r)  s = r _ s

  -- CR4 at F A: the reduct premise of sop by induction on the argument
  -- family (a reduct is op o (k [ j ≔ N ]), whose arguments are related
  -- by CR2).
  spred-op : (o : Op) {k : Fin (ar o) → CComp (F A)} → (∀ j → SPred P (k j)) → SPred P (op o k)
  spred-op o {k} =
    args-ind (SPred P) spred-sn spred-step (ar o) (λ k → SPred P (op o k))
      (λ k c ih → sop o c (λ M' s → op-elim (SPred P) ih s)) k

------------------------------------------------------------------------
-- The relation

𝒱⟦_⟧ : (A : VTy) → CVal A → Type
𝒞⟦_⟧ : (B : CTy) → SCand B

𝒱⟦ 𝟘 ⟧ V                = ⊥
𝒱⟦ 𝟙 ⟧ ⋆                = Unit
𝒱⟦ 𝟙 ⟧ _                = ⊥
𝒱⟦ A ⊕ A' ⟧ (inl W)     = 𝒱⟦ A ⟧ W
𝒱⟦ A ⊕ A' ⟧ (inr W)     = 𝒱⟦ A' ⟧ W
𝒱⟦ A ⊕ A' ⟧ _           = ⊥
𝒱⟦ A ⊗ A' ⟧ (vpair V W) = 𝒱⟦ A ⟧ V × 𝒱⟦ A' ⟧ W
𝒱⟦ A ⊗ A' ⟧ _           = ⊥
𝒱⟦ U B ⟧ V              = 𝒞⟦ B ⟧ .S (force V)

-- ⊤: the strongly normalizing terms.
𝒞⟦ ⊤ ⟧ .S M                  = SN M
𝒞⟦ ⊤ ⟧ .sn m                 = m
𝒞⟦ ⊤ ⟧ .step-closed          = sn-step
𝒞⟦ ⊤ ⟧ .neutral-closed _ r   = acc r
𝒞⟦ ⊤ ⟧ .op-closed            = sn-op

-- A ⇒ B: strongly normalizing, and related arguments give related results.
𝒞⟦ A ⇒ B ⟧ .S M = SN M × ((V : CVal A) → 𝒱⟦ A ⟧ V → 𝒞⟦ B ⟧ .S (app M V))
-- CR1
𝒞⟦ A ⇒ B ⟧ .sn = fst
-- CR2: SN is closed under reduction; appF and CR2 at B.
𝒞⟦ A ⇒ B ⟧ .step-closed (m , c) s =
  sn-step m s , λ V v → 𝒞⟦ B ⟧ .step-closed (c V v) (appF V s)
-- CR3: app M V is neutral, and its reducts are app M' V (Lemma 0).
𝒞⟦ A ⇒ B ⟧ .neutral-closed n r =
    acc (λ M' s → r M' s .fst)
  , λ V v → 𝒞⟦ B ⟧ .neutral-closed tt λ X s →
      let (M' , s' , p) = app-neutral-step n s in
      subst (𝒞⟦ B ⟧ .S) (sym p) (r M' s' .snd V v)
-- CR4: app (op o k) V is neutral; by induction on the argument family,
-- its reducts are op o (λ j → app (k j) V) (CR4 at B) and
-- app (op o (k [ j ≔ N ])) V (induction hypothesis).
𝒞⟦ A ⇒ B ⟧ .op-closed o {k} c =
    sn-op o (λ j → c j .fst)
  , λ V v →
      args-ind (𝒞⟦ A ⇒ B ⟧ .S) (𝒞⟦ A ⇒ B ⟧ .sn) (𝒞⟦ A ⇒ B ⟧ .step-closed) (ar o)
        (λ k → 𝒞⟦ B ⟧ .S (app (op o k) V))
        (λ k c ih → 𝒞⟦ B ⟧ .neutral-closed tt λ X →
           app-op-elim (𝒞⟦ B ⟧ .S)
             (𝒞⟦ B ⟧ .op-closed o (λ j → c j .snd V v))
             (op-elim (λ M' → 𝒞⟦ B ⟧ .S (app M' V)) ih))
        k c

-- B₁ & B₂: strongly normalizing, and both projections are related.
𝒞⟦ B₁ & B₂ ⟧ .S M = SN M × 𝒞⟦ B₁ ⟧ .S (π₁ M) × 𝒞⟦ B₂ ⟧ .S (π₂ M)
-- CR1
𝒞⟦ B₁ & B₂ ⟧ .sn = fst
-- CR2: π₁F, π₂F and CR2 at B₁, B₂.
𝒞⟦ B₁ & B₂ ⟧ .step-closed (m , c₁ , c₂) s =
  sn-step m s , 𝒞⟦ B₁ ⟧ .step-closed c₁ (π₁F s) , 𝒞⟦ B₂ ⟧ .step-closed c₂ (π₂F s)
-- CR3: πᵢ M is neutral, and its reducts are πᵢ M' (Lemma 0).
𝒞⟦ B₁ & B₂ ⟧ .neutral-closed n r =
    acc (λ M' s → r M' s .fst)
  , 𝒞⟦ B₁ ⟧ .neutral-closed tt (λ X s →
      let (M' , s' , p) = π₁-neutral-step n s in
      subst (𝒞⟦ B₁ ⟧ .S) (sym p) (r M' s' .snd .fst))
  , 𝒞⟦ B₂ ⟧ .neutral-closed tt (λ X s →
      let (M' , s' , p) = π₂-neutral-step n s in
      subst (𝒞⟦ B₂ ⟧ .S) (sym p) (r M' s' .snd .snd))
-- CR4: πᵢ (op o k) is neutral; by induction on the argument family, its
-- reducts are op o (λ j → πᵢ (k j)) (CR4 at Bᵢ) and πᵢ (op o (k [ j ≔ N ]))
-- (induction hypothesis).
𝒞⟦ B₁ & B₂ ⟧ .op-closed o {k} c =
    sn-op o (λ j → c j .fst)
  , args-ind (𝒞⟦ B₁ & B₂ ⟧ .S) (𝒞⟦ B₁ & B₂ ⟧ .sn) (𝒞⟦ B₁ & B₂ ⟧ .step-closed) (ar o)
      (λ k → 𝒞⟦ B₁ ⟧ .S (π₁ (op o k)))
      (λ k c ih → 𝒞⟦ B₁ ⟧ .neutral-closed tt λ X →
         π₁-op-elim (𝒞⟦ B₁ ⟧ .S)
           (𝒞⟦ B₁ ⟧ .op-closed o (λ j → c j .snd .fst))
           (op-elim (λ M' → 𝒞⟦ B₁ ⟧ .S (π₁ M')) ih))
      k c
  , args-ind (𝒞⟦ B₁ & B₂ ⟧ .S) (𝒞⟦ B₁ & B₂ ⟧ .sn) (𝒞⟦ B₁ & B₂ ⟧ .step-closed) (ar o)
      (λ k → 𝒞⟦ B₂ ⟧ .S (π₂ (op o k)))
      (λ k c ih → 𝒞⟦ B₂ ⟧ .neutral-closed tt λ X →
         π₂-op-elim (𝒞⟦ B₂ ⟧ .S)
           (𝒞⟦ B₂ ⟧ .op-closed o (λ j → c j .snd .snd))
           (op-elim (λ M' → 𝒞⟦ B₂ ⟧ .S (π₂ M')) ih))
      k c

-- F A: SPred 𝒱⟦A⟧.
𝒞⟦ F A ⟧ .S              = SPred 𝒱⟦ A ⟧
𝒞⟦ F A ⟧ .sn             = spred-sn
𝒞⟦ F A ⟧ .step-closed    = spred-step
𝒞⟦ F A ⟧ .neutral-closed = sneu
𝒞⟦ F A ⟧ .op-closed      = spred-op
