{-
  No neighbor is copied: from generic effects to shapes.

  NoCopy t (CBPV.ConfigReduction): no label inl p occurs at two
  positions of t.  stepG s i g := μ((∂(η ∘ inl) Cₛⁱ)⟪T(inr)(g)⟫)
  generalizes step s i o = stepG s i ⌈o⌉ to any filler g : T Y.

    NoCopyOps    := ∀ s i op. NoCopy (step s i op)
    NoCopyShapes := ∀ s i t.  NoCopy (stepG s i (t , λ _ → tt))
    NoCopyAll    := ∀ s i g.  NoCopy (stepG s i g)

  All three are equivalent (ops→all, ops→shapes, shapes→ops,
  shapes→all).  NoCopyShapes mentions no operation: it is a property of
  the monad on shapes (see CBPV.ContainerAffinity).

  ops→all is by induction on g along the free model (FreeInduction):
    η y        stepG s i (η y) is Cₛⁱ plugged with inr y;
    ops o g⃗   stepG s i (ops o g⃗) = μ(T(h)(step s i o)); the argument
               positions of step s i o are expanded one at a time
               (expand), each expansion a one-hole stepG by g k.
  Only the monad laws, map-poly and plugging are used.
-}

------------------------------------------------------------------------
-- Sums

inl-inj : {A B : Type} {a a' : A} → inl {B = B} a ≡ inl a' → a ≡ a'
inl-inj {a = a} p = cong (⊎.rec (λ x → x) (λ _ → a)) p

inr-inj : {A B : Type} {b b' : B} → inr {A = A} b ≡ inr b' → b ≡ b'
inr-inj {b = b} p = cong (⊎.rec (λ _ → b) (λ x → x)) p

inl≢inr : {A B : Type} {a : A} {b : B} → ¬ (inl a ≡ inr b)
inl≢inr p = subst (⊎.rec (λ _ → Unit) (λ _ → ⊥)) p tt

inr≢inl : {A B : Type} {a : A} {b : B} → ¬ (inr b ≡ inl a)
inr≢inl p = inl≢inr (sym p)

------------------------------------------------------------------------
-- Searching Fin n

findFin : {n : ℕ} (A : Fin n → Type) → (∀ q → Dec (A q))
  → (Σ[ q ∈ Fin n ] A q) ⊎ (∀ q → ¬ A q)
findFin {zero}  A d = inr λ ()
findFin {suc n} A d with d zero
... | yes a = inl (zero , a)
... | no ¬a with findFin (A ∘ suc) (d ∘ suc)
...   | inl (q , a) = inl (suc q , a)
...   | inr none    = inr λ { zero → ¬a ; (suc q) → none q }

------------------------------------------------------------------------
-- Labels occurring at most once

-- The label z occurs at most at one position of t.
Once : {Z : Type} → T Z → Z → Type
Once t z = ∀ q q' → t .snd q ≡ z → t .snd q' ≡ z → q ≡ q'

isPropNoCopy : {N Y : Type} (t : T (N ⊎ Y)) → isProp (NoCopy t)
isPropNoCopy t = isPropΠ3 λ q q' p → isPropΠ2 λ _ _ → isSetFin q q'

NoCopy→Once : {N Y : Type} (t : T (N ⊎ Y)) → NoCopy t → ∀ p → Once t (inl p)
NoCopy→Once t nc p q q' = nc q q' p

Once→NoCopy : {N Y : Type} (t : T (N ⊎ Y)) → (∀ p → Once t (inl p)) → NoCopy t
Once→NoCopy t o q q' p = o p q q'

Once-subst : {Z : Type} {t u : T Z} → t ≡ u → (z : Z) → Once t z → Once u z
Once-subst p z = subst (λ u → Once u z) p

-- Relabelling along f transfers Once when f x ≡ b forces x ≡ a.
Once-pmap : {Z W : Type} (f : Z → W) (t : T Z) (a : Z) (b : W)
  → (∀ x → f x ≡ b → x ≡ a) → Once t a → Once (pmap {p = poly} f t) b
Once-pmap f t a b r o q q' e e' = o q q' (r _ e) (r _ e')

Once-map : {Z W : Type} (f : Z → W) (t : T Z) (a : Z) (b : W)
  → (∀ x → f x ≡ b → x ≡ a) → Once t a → Once (map f t) b
Once-map f t a b r o = Once-subst (sym (map-poly f t)) b (Once-pmap f t a b r o)

------------------------------------------------------------------------
-- Monad laws not in CBPV.FreeModel

μ-map-η : {X Z : Type} (f : X → T Z) (x : X) → μ (map f (η x)) ≡ f x
μ-map-η f x = cong μ (map-η f x) ∙ μ-η (f x)

-- μ(T(g⃗)(⌈o⌉)) = ⟦o⟧(g⃗)
μ-map-op : {Y : Type} (o : Op) (gs : Fin (ar o) → T Y) → μ (map gs ⌈ o ⌉) ≡ ops o gs
μ-map-op o gs =
  cong μ (map-hom gs .pres o η)
  ∙ μ-hom .pres o (map gs ∘ η)
  ∙ cong (ops o) (funExt λ k → μ-map-η gs k)

plugμ : {Z : Type} → ∂ poly (T Z) → T Z → T Z
plugμ C x = μ (C ⟪ x ⟫)

------------------------------------------------------------------------
-- Induction along the free model

-- A proposition on T Y that holds on generators and is closed under the
-- operations holds everywhere: the elements satisfying it form a
-- submodel, and the unique homomorphism from T Y into it is a section.
module FreeInduction {Y : Type} (A : T Y → Type) (isPropA : ∀ t → isProp (A t))
  (Aη : ∀ y → A (η y)) (Aops : ∀ o gs → (∀ k → A (gs k)) → A (ops o gs)) where

  private
    SubC : Type
    SubC = Σ (T Y) A

    subAlg : AlgStr sig SubC
    subAlg o args = ops o (fst ∘ args) , Aops o (fst ∘ args) (snd ∘ args)

    eval-fst : {X : Type} (γ : X → SubC) (t : Term sig X)
      → eval sig subAlg γ t .fst ≡ eval sig ops (fst ∘ γ) t
    eval-fst γ (var x)   = refl
    eval-fst γ (op o ts) = cong (ops o) (funExt λ j → eval-fst γ (ts j))

    SubM : Model 𝒯
    SubM .Carrier = SubC
    SubM .alg     = subAlg
    SubM .sat e γ = Σ≡Prop isPropA
      (eval-fst γ (lhs (eqn e)) ∙ ops-sat e (fst ∘ γ) ∙ sym (eval-fst γ (rhs (eqn e))))

    f : T Y → SubC
    f = ext SubM (λ y → η y , Aη y)

    fst-f : ∀ t → f t .fst ≡ t
    fst-f = hom-ext (FM Y) (fst ∘ f) (λ t → t)
      (ishom λ o args → cong fst (ext-hom SubM _ .pres o args))
      (ishom λ _ _ → refl)
      (λ y → cong fst (ext-η SubM _ y))

  induction : ∀ t → A t
  induction t = subst A (fst-f t) (f t .snd)

------------------------------------------------------------------------
-- The generic step with an arbitrary filler

stepG : {Y : Type} (s : poly .Sh) (i : Fin (poly .P s)) → T Y → T (Fin∖ (poly .P s) i ⊎ Y)
stepG s i g = μ ((∂map (η ∘ inl) (C-fresh s i)) ⟪ map inr g ⟫)

step≡stepG : ∀ s i o → step s i o ≡ stepG s i ⌈ o ⌉
step≡stepG s i o = refl

NoCopyShapes : Type
NoCopyShapes = ∀ s i (t : poly .Sh) → NoCopy (stepG {Y = Unit} s i (t , λ _ → tt))

NoCopyAll : Type₁
NoCopyAll = ∀ s i {Y : Type} (g : T Y) → NoCopy (stepG s i g)

-- T(γ ⊎ δ)(stepG s i g) = μ((∂(η ∘ γ) Cₛⁱ)⟪T(δ)(g)⟫)   (cf. simplification)
stepG-simp : {Y Z : Type} (s : poly .Sh) (i : Fin (poly .P s))
  (γ : Fin∖ (poly .P s) i → Z) (δ : Y → Z) (g : T Y)
  → map (⊎.rec γ δ) (stepG s i g) ≡ μ ((s , i , η ∘ γ) ⟪ map δ g ⟫)
stepG-simp s i γ δ g =
  μ-natural G _
  ∙ cong μ (map-plug (map G) (s , i , η ∘ inl) (map inr g))
  ∙ cong₂ plugμ
      (ΣPathP (refl , ΣPathP (refl , funExt λ x → map-η G (inl x))))
      (sym (map-∘ G inr g))
  where
  G = ⊎.rec γ δ

-- Naturality in the filler.
stepG-map : {Y Y' : Type} (s : poly .Sh) (i : Fin (poly .P s)) (f : Y → Y') (g : T Y)
  → stepG s i (map f g) ≡ map (⊎.rec inl (inr ∘ f)) (stepG s i g)
stepG-map s i f g =
  cong (plugμ (s , i , η ∘ inl)) (sym (map-∘ inr f g))
  ∙ sym (stepG-simp s i inl (inr ∘ f) g)

------------------------------------------------------------------------
-- NoCopy is invariant under relabelling the filler

rec-inl-refl : {N Y Y' : Type} (f : Y → Y') (p : N) (x : N ⊎ Y)
  → ⊎.rec inl (inr ∘ f) x ≡ inl p → x ≡ inl p
rec-inl-refl f p (inl p') e = cong inl (inl-inj e)
rec-inl-refl f p (inr y)  e = ⊥.rec (inr≢inl e)

NoCopy-map→ : {N Y Y' : Type} (f : Y → Y') (t : T (N ⊎ Y))
  → NoCopy t → NoCopy (map (⊎.rec inl (inr ∘ f)) t)
NoCopy-map→ f t nc = Once→NoCopy _ λ p →
  Once-map _ t (inl p) (inl p) (rec-inl-refl f p) (NoCopy→Once t nc p)

NoCopy-map← : {N Y Y' : Type} (f : Y → Y') (t : T (N ⊎ Y))
  → NoCopy (map (⊎.rec inl (inr ∘ f)) t) → NoCopy t
NoCopy-map← f t nc q q' p e e' =
  subst NoCopy (map-poly G t) nc q q' p (cong G e) (cong G e')
  where
  G = ⊎.rec inl (inr ∘ f)

-- NoCopy (stepG s i g) depends only on the shape of g.
module _ {Y : Type} (s : poly .Sh) (i : Fin (poly .P s)) (g : T Y) where
  private
    gen : T (Fin (poly .P (g .fst)))
    gen = g .fst , λ q → q

    via : {Y' : Type} (f : Fin (poly .P (g .fst)) → Y')
      → NoCopy (stepG s i (map f gen)) → NoCopy (stepG s i gen)
    via f nc = NoCopy-map← f _ (subst NoCopy (stepG-map s i f gen) nc)

    back : {Y' : Type} (f : Fin (poly .P (g .fst)) → Y')
      → NoCopy (stepG s i gen) → NoCopy (stepG s i (map f gen))
    back f nc = subst NoCopy (sym (stepG-map s i f gen)) (NoCopy-map→ f _ nc)

  NoCopy-shape→ : NoCopy (stepG s i g) → NoCopy (stepG s i (g .fst , λ _ → tt))
  NoCopy-shape→ nc =
    subst (λ x → NoCopy (stepG s i x)) (map-poly (λ _ → tt) gen)
      (back (λ _ → tt) (via (g .snd) (subst (λ x → NoCopy (stepG s i x)) (sym (map-poly (g .snd) gen)) nc)))

  NoCopy-shape← : NoCopy (stepG s i (g .fst , λ _ → tt)) → NoCopy (stepG s i g)
  NoCopy-shape← nc =
    subst (λ x → NoCopy (stepG s i x)) (map-poly (g .snd) gen)
      (back (g .snd) (via (λ _ → tt) (subst (λ x → NoCopy (stepG s i x)) (sym (map-poly (λ _ → tt) gen)) nc)))

------------------------------------------------------------------------
-- Base case: filler η y

stepG-η : {Y : Type} (s : poly .Sh) (i : Fin (poly .P s)) (y : Y)
  → stepG s i (η y) ≡ (s , i , inl) ⟪ inr y ⟫
stepG-η s i y =
  cong (plugμ (s , i , η ∘ inl)) (map-η inr y)
  ∙ cong μ (sym (map-plug η (s , i , inl) (inr y)))
  ∙ μ-mapη _

fill-inl : {Y : Type} {n : ℕ} (i : Fin n) (y : Y) (q : Fin n) (p : Fin∖ n i)
  → fillAt i inl (inr y) q ≡ inl p → q ≡ p .fst
fill-inl i y q p e with discreteFin q i
... | yes _ = ⊥.rec (inr≢inl e)
... | no _  = cong fst (inl-inj e)

NoCopy-η : {Y : Type} (s : poly .Sh) (i : Fin (poly .P s)) (y : Y) → NoCopy (stepG s i (η y))
NoCopy-η s i y = subst NoCopy (sym (stepG-η s i y))
  λ q q' p e e' → fill-inl i y q p e ∙ sym (fill-inl i y q' p e')

------------------------------------------------------------------------
-- Operation case: stepG s i (ops o g⃗) = μ(T(h)(step s i o))

stepG-op : {Y : Type} (s : poly .Sh) (i : Fin (poly .P s)) (o : Op) (gs : Fin (ar o) → T Y)
  → stepG s i (ops o gs) ≡ μ (map (⊎.rec (η ∘ inl) (map inr ∘ gs)) (step s i o))
stepG-op {Y} s i o gs = sym
  ( cong μ (μ-natural h X)
  ∙ μ-assoc (map (map h) X)
  ∙ cong μ (sym (map-∘ μ (map h) X))
  ∙ cong μ (map-plug (μ ∘ map h) C (map inr ⌈ o ⌉))
  ∙ cong₂ plugμ
      (ΣPathP (refl , ΣPathP (refl , funExt λ x → μ-map-η h (inl x))))
      ( cong μ (sym (map-∘ h inr ⌈ o ⌉))
      ∙ cong μ (map-∘ (map inr) gs ⌈ o ⌉)
      ∙ sym (μ-natural inr (map gs ⌈ o ⌉))
      ∙ cong (map inr) (μ-map-op o gs)))
  where
  h : Fin∖ (poly .P s) i ⊎ Fin (ar o) → T (Fin∖ (poly .P s) i ⊎ Y)
  h = ⊎.rec (η ∘ inl) (map inr ∘ gs)
  C : ∂ poly (T (Fin∖ (poly .P s) i ⊎ Fin (ar o)))
  C = s , i , η ∘ inl
  X : T (T (Fin∖ (poly .P s) i ⊎ Fin (ar o)))
  X = C ⟪ map inr ⌈ o ⌉ ⟫

------------------------------------------------------------------------
-- One-hole expansion copies no label that its context has at most once

Once-onehole : {Nb Y W : Type} (u : T (Nb ⊎ Y)) → NoCopy u
  → (γ : Nb → W) (δ : Y → W) (L : W)
  → (∀ y → ¬ (δ y ≡ L))
  → (∀ j j' → γ j ≡ L → γ j' ≡ L → j ≡ j')
  → Once (map (⊎.rec γ δ) u) L
Once-onehole {Nb} {Y} {W} u nc γ δ L nδ γinj =
  Once-subst (sym (map-poly G u)) L λ r r' e e' →
    let (j  , x  , g ) = pre (u .snd r)  e
        (j' , x' , g') = pre (u .snd r') e'
    in nc r r' j x (x' ∙ cong inl (γinj j' j g' g))
  where
  G = ⊎.rec γ δ
  pre : (x : Nb ⊎ Y) → G x ≡ L → Σ[ j ∈ Nb ] (x ≡ inl j) × (γ j ≡ L)
  pre (inl j) e = j , refl , e
  pre (inr y) e = ⊥.rec (nδ y e)

------------------------------------------------------------------------
-- Expanding the pending labels inr q one at a time

module _ {Y : Type} where

  R : {N : Type} {n : ℕ} → (Fin n → T Y) → N ⊎ Fin n → T (N ⊎ Y)
  R F = ⊎.rec (η ∘ inl) (map inr ∘ F)

  -- Relabelling after expanding inr zero: old neighbors, and the
  -- remaining pending labels shifted down.
  shift : {N : Type} {n : ℕ} (x : N ⊎ Fin (suc n)) → ¬ (x ≡ inr zero) → (N ⊎ Y) ⊎ Fin n
  shift (inl p)       _  = inl (inl p)
  shift (inr zero)    ne = ⊥.rec (ne refl)
  shift (inr (suc q)) _  = inr q

  -- Expand inr zero with F zero.
  e : {N : Type} {n : ℕ} → (Fin (suc n) → T Y) → N ⊎ Fin (suc n) → T ((N ⊎ Y) ⊎ Fin n)
  e F (inl p)       = η (inl (inl p))
  e F (inr zero)    = map (inl ∘ inr) (F zero)
  e F (inr (suc q)) = η (inr q)

  e-shift : {N : Type} {n : ℕ} (F : Fin (suc n) → T Y) (x : N ⊎ Fin (suc n))
    (ne : ¬ (x ≡ inr zero)) → e F x ≡ η (shift x ne)
  e-shift F (inl p)       ne = refl
  e-shift F (inr zero)    ne = ⊥.rec (ne refl)
  e-shift F (inr (suc q)) ne = refl

  shift-inl : {N : Type} {n : ℕ} (p : N) (x : N ⊎ Fin (suc n)) (ne : ¬ (x ≡ inr zero))
    → shift x ne ≡ inl (inl p) → x ≡ inl p
  shift-inl p (inl p')      ne s = cong inl (inl-inj (inl-inj s))
  shift-inl p (inr zero)    ne s = ⊥.rec (ne refl)
  shift-inl p (inr (suc q)) ne s = ⊥.rec (inr≢inl s)

  shift-inr : {N : Type} {n : ℕ} (q : Fin n) (x : N ⊎ Fin (suc n)) (ne : ¬ (x ≡ inr zero))
    → shift x ne ≡ inr q → x ≡ inr (suc q)
  shift-inr q (inl p)        ne s = ⊥.rec (inl≢inr s)
  shift-inr q (inr zero)     ne s = ⊥.rec (ne refl)
  shift-inr q (inr (suc q')) ne s = cong (inr ∘ suc) (inr-inj s)

  dec-inr0 : {N : Type} {n : ℕ} (x : N ⊎ Fin (suc n)) → Dec (x ≡ inr zero)
  dec-inr0 (inl p) = no inl≢inr
  dec-inr0 (inr q) with discreteFin q zero
  ... | yes e  = yes (cong inr e)
  ... | no ne  = no λ e → ne (inr-inj e)

  -- A care label L of μ(T(e F)(w)) occurs at most once, provided its
  -- preimage L₀ occurs at most once in w and L is not a filler label.
  expand-Once : {N : Type} {n : ℕ} (F : Fin (suc n) → T Y)
    → (∀ s i → NoCopy (stepG s i (F zero)))
    → (w : T (N ⊎ Fin (suc n))) → Once w (inr zero)
    → (L : (N ⊎ Y) ⊎ Fin n) (L₀ : N ⊎ Fin (suc n))
    → (∀ x ne → shift x ne ≡ L → x ≡ L₀)
    → (∀ y → ¬ (inl (inr y) ≡ L))
    → Once w L₀
    → Once (μ (map (e F) w)) L
  expand-Once {N} {n} F ncF w o0 L L₀ sh nf oL =
    go (findFin (λ r → w .snd r ≡ inr zero) (λ r → dec-inr0 (w .snd r)))
    where
    s = w .fst

    go : (Σ[ r ∈ Fin (poly .P s) ] w .snd r ≡ inr zero)
         ⊎ (∀ r → ¬ (w .snd r ≡ inr zero))
       → Once (μ (map (e F) w)) L
    -- inr zero occurs (exactly once, at r): a one-hole step.
    go (inl (r , wr)) = Once-subst (sym path) L
      (Once-onehole (stepG s r (F zero)) (ncF s r) γ' (inl ∘ inr) L nf γ'inj)
      where
      γ : Fin∖ (poly .P s) r → N ⊎ Fin (suc n)
      γ j = w .snd (j .fst)
      ne : (j : Fin∖ (poly .P s) r) → ¬ (γ j ≡ inr zero)
      ne j x = j .snd (o0 (j .fst) r x wr)
      γ' : Fin∖ (poly .P s) r → (N ⊎ Y) ⊎ Fin n
      γ' j = shift (γ j) (ne j)
      γ'inj : ∀ j j' → γ' j ≡ L → γ' j' ≡ L → j ≡ j'
      γ'inj j j' x x' = Σ≡Prop (λ _ → isProp¬ _)
        (oL (j .fst) (j' .fst) (sh (γ j) (ne j) x) (sh (γ j') (ne j') x'))
      path : μ (map (e F) w) ≡ map (⊎.rec γ' (inl ∘ inr)) (stepG s r (F zero))
      path =
        cong (μ ∘ map (e F)) (sym (plug-unplug w r) ∙ cong ((s , r , γ) ⟪_⟫) wr)
        ∙ cong μ (map-plug (e F) (s , r , γ) (inr zero))
        ∙ cong₂ plugμ
            (ΣPathP (refl , ΣPathP (refl , funExt λ j → e-shift F (γ j) (ne j))))
            refl
        ∙ sym (stepG-simp s r γ' (inl ∘ inr) (F zero))
    -- inr zero does not occur: a relabelling.
    go (inr none) = Once-subst (sym path) L λ r r' x x' →
      oL r r' (sh (w .snd r) (none r) x) (sh (w .snd r') (none r') x')
      where
      σ : Fin (poly .P s) → (N ⊎ Y) ⊎ Fin n
      σ r = shift (w .snd r) (none r)
      path : μ (map (e F) w) ≡ (s , σ)
      path =
        cong μ (map-poly (e F) w
               ∙ ΣPathP (refl , funExt λ r → e-shift F (w .snd r) (none r))
               ∙ sym (map-poly η (s , σ)))
        ∙ μ-mapη (s , σ)

  -- Expanding all pending labels: if every care neighbor and every
  -- pending label occurs at most once in w, and no one-hole expansion
  -- by a filler copies a neighbor, then care neighbors occur at most
  -- once in μ(T(R F)(w)).
  expand : (n : ℕ) {N : Type} (D : N → Type) (F : Fin n → T Y)
    → (∀ q s i → NoCopy (stepG s i (F q)))
    → (w : T (N ⊎ Fin n))
    → (∀ p → D p → Once w (inl p))
    → (∀ q → Once w (inr q))
    → ∀ p → D p → Once (μ (map (R F) w)) (inl p)
  expand zero {N} D F ncF w hD hQ p d =
    Once-subst (sym path) (inl p) (Once-map E0 w (inl p) (inl p) refl0 (hD p d))
    where
    E0 : N ⊎ Fin zero → N ⊎ Y
    E0 = ⊎.rec inl λ ()
    refl0 : ∀ x → E0 x ≡ inl p → x ≡ inl p
    refl0 (inl p') x = cong inl (inl-inj x)
    path : μ (map (R F) w) ≡ map E0 w
    path =
      cong μ (cong (λ f → map f w) (funExt λ { (inl p') → refl ; (inr ()) })
             ∙ map-∘ η E0 w)
      ∙ μ-mapη (map E0 w)
  expand (suc n) {N} D F ncF w hD hQ p d =
    Once-subst path (inl p)
      (Once-map j (μ (map (R (F ∘ suc)) w')) (inl (inl p)) (inl p) j-refl IH)
    where
    w' : T ((N ⊎ Y) ⊎ Fin n)
    w' = μ (map (e F) w)
    D' : N ⊎ Y → Type
    D' = ⊎.rec D (λ _ → ⊥)
    hD' : ∀ p' → D' p' → Once w' (inl p')
    hD' (inl p') d' = expand-Once F (ncF zero) w (hQ zero) (inl (inl p')) (inl p')
      (shift-inl p') (λ y x → inr≢inl (inl-inj x)) (hD p' d')
    hD' (inr y) ()
    hQ' : ∀ q → Once w' (inr q)
    hQ' q = expand-Once F (ncF zero) w (hQ zero) (inr q) (inr (suc q))
      (shift-inr q) (λ y → inl≢inr) (hQ (suc q))
    IH : Once (μ (map (R (F ∘ suc)) w')) (inl (inl p))
    IH = expand n D' (F ∘ suc) (λ q → ncF (suc q)) w' hD' hQ' (inl p) d
    j : (N ⊎ Y) ⊎ Y → N ⊎ Y
    j = ⊎.rec (⊎.rec inl inr) inr
    j-refl : ∀ x → j x ≡ inl p → x ≡ inl (inl p)
    j-refl (inl (inl p')) x = cong (inl ∘ inl) (inl-inj x)
    j-refl (inl (inr y))  x = ⊥.rec (inr≢inl x)
    j-refl (inr y)        x = ⊥.rec (inr≢inl x)
    K : (N ⊎ Y) ⊎ Fin n → T (N ⊎ Y)
    K = map j ∘ R (F ∘ suc)
    pw : ∀ x → μ (map K (e F x)) ≡ R F x
    pw (inl p')      = μ-map-η K (inl (inl p')) ∙ map-η j (inl (inl p'))
    pw (inr zero)    =
      cong μ (sym (map-∘ K (inl ∘ inr) (F zero))
             ∙ cong (λ f → map f (F zero)) (funExt λ y → map-η j (inl (inr y)))
             ∙ map-∘ η inr (F zero))
      ∙ μ-mapη (map inr (F zero))
    pw (inr (suc q)) = μ-map-η K (inr q) ∙ sym (map-∘ j inr (F (suc q)))
    path : map j (μ (map (R (F ∘ suc)) w')) ≡ μ (map (R F) w)
    path =
      μ-natural j _
      ∙ cong μ (sym (map-∘ (map j) (R (F ∘ suc)) w'))
      ∙ cong μ (μ-natural K _)
      ∙ μ-assoc (map (map K) (map (e F) w))
      ∙ cong μ (sym (map-∘ μ (map K) (map (e F) w)))
      ∙ cong μ (sym (map-∘ (μ ∘ map K) (e F) w))
      ∙ cong (λ f → μ (map f w)) (funExt pw)

-- The operation case of the induction

AStep : {Y : Type} → T Y → Type
AStep g = ∀ s i → NoCopy (stepG s i g)

isPropAStep : {Y : Type} (g : T Y) → isProp (AStep g)
isPropAStep g = isPropΠ2 λ s i → isPropNoCopy _

module _ (aff : NoCopyOps) {Y : Type} (o : Op) (gs : Fin (ar o) → T Y)
  (ih : ∀ k → AStep (gs k)) (s : poly .Sh) (i : Fin (poly .P s)) where

  private
    Nb : Type
    Nb = Fin∖ (poly .P s) i

    u : T (Nb ⊎ Fin (ar o))
    u = step s i o

    n : ℕ
    n = poly .P (u .fst)

    h : Nb ⊎ Fin (ar o) → T (Nb ⊎ Y)
    h = ⊎.rec (η ∘ inl) (map inr ∘ gs)

    -- Label each argument position of u by itself.
    tag : Fin n → Nb ⊎ Fin (ar o) → Nb ⊎ Fin n
    tag q (inl p) = inl p
    tag q (inr k) = inr q

    w : T (Nb ⊎ Fin n)
    w = u .fst , λ q → tag q (u .snd q)

    -- Fillers; neighbor positions of u are never expanded, so they get
    -- the dummy filler η (inr tt).
    Fsel : Nb ⊎ Fin (ar o) → T (Y ⊎ Unit)
    Fsel (inl p) = η (inr tt)
    Fsel (inr k) = map inl (gs k)

    F : Fin n → T (Y ⊎ Unit)
    F q = Fsel (u .snd q)

    ncSel : (x : Nb ⊎ Fin (ar o)) → AStep (Fsel x)
    ncSel (inl p) s' i' = NoCopy-η s' i' (inr tt)
    ncSel (inr k) s' i' =
      subst NoCopy (sym (stepG-map s' i' inl (gs k))) (NoCopy-map→ inl _ (ih k s' i'))

    tag-inl : (p : Nb) (q : Fin n) (x : Nb ⊎ Fin (ar o)) → tag q x ≡ inl p → x ≡ inl p
    tag-inl p q (inl p') x = cong inl (inl-inj x)
    tag-inl p q (inr k)  x = ⊥.rec (inr≢inl x)

    tag-inr : (q q' : Fin n) (x : Nb ⊎ Fin (ar o)) → tag q' x ≡ inr q → q' ≡ q
    tag-inr q q' (inl p) x = ⊥.rec (inl≢inr x)
    tag-inr q q' (inr k) x = inr-inj x

    hD : ∀ p → Unit → Once w (inl p)
    hD p _ q q' x x' =
      aff s i o q q' p (tag-inl p q (u .snd q) x) (tag-inl p q' (u .snd q') x')

    hQ : ∀ q → Once w (inr q)
    hQ q q₁ q₂ x x' = tag-inr q q₁ (u .snd q₁) x ∙ sym (tag-inr q q₂ (u .snd q₂) x')

    G : Nb ⊎ Y → Nb ⊎ (Y ⊎ Unit)
    G = ⊎.rec inl (inr ∘ inl)

    pwg : (q : Fin n) (x : Nb ⊎ Fin (ar o)) → u .snd q ≡ x → map G (h x) ≡ R F (tag q x)
    pwg q (inl p) _ = map-η G (inl p)
    pwg q (inr k) x =
      sym (map-∘ G inr (gs k))
      ∙ map-∘ inr inl (gs k)
      ∙ cong (map inr ∘ Fsel) (sym x)

    path : map G (μ (map h u)) ≡ μ (map (R F) w)
    path =
      μ-natural G _
      ∙ cong μ ( sym (map-∘ (map G) h u)
               ∙ map-poly (map G ∘ h) u
               ∙ ΣPathP (refl , funExt λ q → pwg q (u .snd q) refl)
               ∙ sym (map-poly (R F) w))

    nc : NoCopy (μ (map h u))
    nc = NoCopy-map← inl (μ (map h u))
      (subst NoCopy (sym path)
        (Once→NoCopy _ λ p → expand n (λ _ → Unit) F (λ q → ncSel (u .snd q)) w hD hQ p tt))

  op-case : NoCopy (stepG s i (ops o gs))
  op-case = subst NoCopy (sym (stepG-op s i o gs)) nc

-- Main results

ops→all : NoCopyOps → NoCopyAll
ops→all aff s i {Y} g =
  FreeInduction.induction AStep isPropAStep
    (λ y s' i' → NoCopy-η s' i' y)
    (λ o gs ih s' i' → op-case aff o gs ih s' i')
    g s i

ops→shapes : NoCopyOps → NoCopyShapes
ops→shapes aff s i t = ops→all aff s i (t , λ _ → tt)

all→shapes : NoCopyAll → NoCopyShapes
all→shapes aff s i t = aff s i (t , λ _ → tt)

shapes→all : NoCopyShapes → NoCopyAll
shapes→all aff s i g = NoCopy-shape← s i g (aff s i (g .fst))

shapes→ops : NoCopyShapes → NoCopyOps
shapes→ops aff s i o = shapes→all aff s i ⌈ o ⌉
