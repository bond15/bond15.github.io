{-
  Monadic containers (Uustalu's mnd-containers).

  A monadic container on a container (Sh, P) is
      ι  : Sh                                         unit shape
      σ  : (s : Sh) → (P s → Sh) → Sh                 multiplication
      pr : P (σ s v) → Σ (p : P s). P (v p)           where a position comes from
  Its container monad on ⟦Sh, P⟧ is
      η x      := (ι , λ _ → x)
      μ (s, γ) := (σ s (fst ∘ γ) , λ q → γ (pr₁ q) .snd (pr₂ q)).

  The laws can be stated syntactically, on (ι, σ, pr) (UustaluLaws: three
  shape laws, five position laws over them), or semantically, as the
  monad laws of (η, μ) (MonadLaws).  M→U and U→M: they are equivalent.
-}
module CBPV.MonadicContainer where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin)

open import CBPV.Polynomial

record MCOps (p : Poly) : Type where
  Pos : p .Sh → Type
  Pos s = Fin (p .P s)
  field
    ι  : p .Sh
    σ  : (s : p .Sh) → (Pos s → p .Sh) → p .Sh
    pr : (s : p .Sh) (v : Pos s → p .Sh) → Pos (σ s v) → Σ[ q ∈ Pos s ] Pos (v q)

  -- pr₁ and pr₂
  ↖ : (s : p .Sh) (v : Pos s → p .Sh) → Pos (σ s v) → Pos s
  ↖ s v q = pr s v q .fst

  ↗ : (s : p .Sh) (v : Pos s → p .Sh) (q : Pos (σ s v)) → Pos (v (↖ s v q))
  ↗ s v q = pr s v q .snd

module _ {p : Poly} (c : MCOps p) where
  open MCOps c

  ηᶜ : {X : Type} → X → ⟦ p ⟧ X
  ηᶜ x = ι , λ _ → x

  μᶜ : {X : Type} → ⟦ p ⟧ (⟦ p ⟧ X) → ⟦ p ⟧ X
  μᶜ (s , γ) = σ s (fst ∘ γ) , λ q → γ (↖ s (fst ∘ γ) q) .snd (↗ s (fst ∘ γ) q)

  -- The container monad laws.
  record MonadLaws : Type₁ where
    field
      μη    : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (ηᶜ t) ≡ t
      μmapη : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (pmap ηᶜ t) ≡ t
      μμ    : {X : Type} (t : ⟦ p ⟧ (⟦ p ⟧ (⟦ p ⟧ X))) → μᶜ (μᶜ t) ≡ μᶜ (pmap μᶜ t)

  -- The monadic container laws.
  record UustaluLaws : Type where
    field
      unitʳ : (s : p .Sh) → s ≡ σ s (λ _ → ι)
      unitˡ : (s : p .Sh) → σ ι (λ _ → s) ≡ s
      assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → σ (σ s v) (λ q → w (↖ s v q) (↗ s v q)) ≡ σ s (λ q → σ (v q) (w q))
      ↖-unitʳ : (s : p .Sh) → PathP (λ i → Pos (unitʳ s i) → Pos s) (λ q → q) (↖ s (λ _ → ι))
      ↗-unitˡ : (s : p .Sh) → PathP (λ i → Pos (unitˡ s i) → Pos s) (↗ ι (λ _ → s)) (λ q → q)
      ↖↖-assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → PathP (λ i → Pos (assoc s v w i) → Pos s)
                (λ q → ↖ s v (↖ (σ s v) (λ q → w (↖ s v q) (↗ s v q)) q))
                (↖ s (λ q → σ (v q) (w q)))
      ↗↖-assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → PathP (λ i → (q : Pos (assoc s v w i)) → Pos (v (↖↖-assoc s v w i q)))
                (λ q → ↗ s v (↖ (σ s v) (λ q → w (↖ s v q) (↗ s v q)) q))
                (λ q → ↖ (v (↖ s (λ q → σ (v q) (w q)) q)) (w (↖ s (λ q → σ (v q) (w q)) q))
                         (↗ s (λ q → σ (v q) (w q)) q))
      ↗↗-assoc : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh)
            → PathP (λ i → (q : Pos (assoc s v w i))
                           → Pos (w (↖↖-assoc s v w i q) (↗↖-assoc s v w i q)))
                (λ q → ↗ (σ s v) (λ q → w (↖ s v q) (↗ s v q)) q)
                (λ q → ↗ (v (↖ s (λ q → σ (v q) (w q)) q)) (w (↖ s (λ q → σ (v q) (w q)) q))
                         (↗ s (λ q → σ (v q) (w q)) q))

  U→M : UustaluLaws → MonadLaws
  U→M U = record { μη = μη' ; μmapη = μmapη' ; μμ = μμ' }
    where
    open UustaluLaws U
    μη' : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (ηᶜ t) ≡ t
    μη' (s , γ) i = unitˡ s i , λ q → γ (↗-unitˡ s i q)
    μmapη' : {X : Type} (t : ⟦ p ⟧ X) → μᶜ (pmap ηᶜ t) ≡ t
    μmapη' (s , γ) i = unitʳ s (~ i) , λ q → γ (↖-unitʳ s (~ i) q)
    μμ' : {X : Type} (t : ⟦ p ⟧ (⟦ p ⟧ (⟦ p ⟧ X))) → μᶜ (μᶜ t) ≡ μᶜ (pmap μᶜ t)
    μμ' (s , γ) i =
      assoc s v w i ,
      λ q → γ (↖↖-assoc s v w i q) .snd (↗↖-assoc s v w i q) .snd (↗↗-assoc s v w i q)
      where
      v = λ q → γ q .fst
      w = λ q m → γ q .snd m .fst

  -- Each monad law at a generic element (leaves are their own positions)
  -- gives a shape law (first component) and a position law (second).
  M→U : MonadLaws → UustaluLaws
  M→U M = record
    { unitʳ    = λ s → sym (cong fst (μmapη (s , λ q → q)))
    ; unitˡ    = λ s → cong fst (μη (s , λ q → q))
    ; assoc    = λ s v w → cong fst (gen s v w)
    ; ↖-unitʳ  = λ s → symP (cong snd (μmapη (s , λ q → q)))
    ; ↗-unitˡ  = λ s → cong snd (μη (s , λ q → q))
    ; ↖↖-assoc = λ s v w i q → gen s v w i .snd q .fst
    ; ↗↖-assoc = λ s v w i q → gen s v w i .snd q .snd .fst
    ; ↗↗-assoc = λ s v w i q → gen s v w i .snd q .snd .snd
    }
    where
    open MonadLaws M
    gen : (s : p .Sh) (v : Pos s → p .Sh) (w : (q : Pos s) → Pos (v q) → p .Sh) → _
    gen s v w = μμ {X = Σ[ q ∈ Pos s ] Σ[ m ∈ Pos (v q) ] Pos (w q m)}
                  (s , λ q → v q , λ m → w q m , λ r → q , m , r)

record MonadicContainer : Type₁ where
  field
    poly : Poly
    cops : MCOps poly
    laws : MonadLaws cops

  open MCOps cops public
  open MonadLaws laws public

  ηₘ : {X : Type} → X → ⟦ poly ⟧ X
  ηₘ = ηᶜ cops

  μₘ : {X : Type} → ⟦ poly ⟧ (⟦ poly ⟧ X) → ⟦ poly ⟧ X
  μₘ = μᶜ cops
