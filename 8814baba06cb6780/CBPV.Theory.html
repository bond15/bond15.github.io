{-
  Finitary algebraic theories, algebras, models and
  homomorphisms.

  A signature has a type of operations and an arity in ℕ for each.
  Terms over a signature with variables in X are trees of operations
  whose leaves are variables.  An equation has a context size n and two
  terms over Fin n.  A model is an algebra validating every equation in
  every environment.

  Carriers of algebras are arbitrary types: no result below needs them to
  be sets.
-}
module CBPV.Theory where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin)

record Signature : Type₁ where
  field
    Op : Type
    ar : Op → ℕ

module _ (Sg : Signature) where
  open Signature Sg

  -- Term Σ X
  data Term (X : Type) : Type where
    var : X → Term X
    op  : (o : Op) → (Fin (ar o) → Term X) → Term X

  record Equation : Type where
    field
      ctx : ℕ
      lhs rhs : Term (Fin ctx)

  -- An algebra structure on A: an interpretation ⟦op⟧ : A^ar(op) → A of
  -- every operation.
  AlgStr : Type → Type
  AlgStr A = (o : Op) → (Fin (ar o) → A) → A

  -- ⟦t⟧(γ): interpretation of a term in an algebra and environment.
  eval : {A X : Type} → AlgStr A → (X → A) → Term X → A
  eval α γ (var x)   = γ x
  eval α γ (op o ts) = α o (λ j → eval α γ (ts j))

  -- A record rather than a Π-type, so that α and β can be inferred.
  record IsHom {A B : Type} (α : AlgStr A) (β : AlgStr B) (h : A → B) : Type where
    constructor ishom
    field pres : ∀ o args → h (α o args) ≡ β o (h ∘ args)
  open IsHom public

record Theory : Type₁ where
  field
    sig : Signature
    Eqn : Type
    eqn : Eqn → Equation sig

  open Signature sig public
  open Equation public

module _ (𝒯 : Theory) where
  open Theory 𝒯

  -- A ⊨ 𝒯
  Sat : {A : Type} → AlgStr sig A → Type
  Sat {A} α = ∀ e (γ : Fin (ctx (eqn e)) → A)
    → eval sig α γ (lhs (eqn e)) ≡ eval sig α γ (rhs (eqn e))

  record Model : Type₁ where
    field
      Carrier : Type
      alg     : AlgStr sig Carrier
      sat     : Sat alg

  open Model

  IsModelHom : (M N : Model) → (Carrier M → Carrier N) → Type
  IsModelHom M N = IsHom sig (alg M) (alg N)
