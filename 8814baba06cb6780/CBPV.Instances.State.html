{-
  Global state over two states St = Fin 2.

  Theory: get (binary) and set_c (unary) for c : St, with the complete
  lens laws (get-set, set-get, set-set, get-get).  Free model
  T X = St → St × X, as the polynomial Σ (St → St) (St → X).

  Monadic container: Sh := St → St, P := St (Fin 2), ι := id,
  σ(f, v) := λ s. v(s)(f s), pr(q) := (q, f q).  ⌈get⌉ := ⟨id, id⟩,
  ⌈set_c⌉ := ⟨const c, const 0⟩.  Affine: pr₁ is the identity.
-}
module CBPV.Instances.State where

open import Cubical.Foundations.Prelude
open import Cubical.Foundations.Function
open import Cubical.Data.Sigma
open import Cubical.Data.Empty as ⊥ using (⊥)
open import Cubical.Data.Nat using (ℕ)
open import Cubical.Data.FinData using (Fin; zero; suc)

open import CBPV.Theory
open import CBPV.FreeModel
open import CBPV.Polynomial
open import CBPV.OperationalModel
open import CBPV.MonadicContainer
open import CBPV.ContainerModel
open import CBPV.ContainerAffinity using (Affine; Inj₁→Affine)
open import CBPV.Instances.Common

St : Type
St = Fin 2

pattern s₀ = zero
pattern s₁ = suc zero

data StOp : Type where
  get : StOp
  set : St → StOp

sigS : Signature
sigS = record { Op = StOp ; ar = λ { get → 2 ; (set _) → 1 } }

data StEqn : Type where
  get-set : StEqn
  set-get : St → StEqn
  set-set : St → St → StEqn
  get-get : StEqn

-- x_{ij} for the get-get law
idx : St → St → Fin 4
idx s₀ s₀ = zero
idx s₀ s₁ = suc zero
idx s₁ s₀ = suc (suc zero)
idx s₁ s₁ = suc (suc (suc zero))

private
  getT : {n : ℕ} → (St → Term sigS (Fin n)) → Term sigS (Fin n)
  getT ts = op get ts

  setT : {n : ℕ} → St → Term sigS (Fin n) → Term sigS (Fin n)
  setT c t = op (set c) (λ _ → t)

stateTheory : Theory
stateTheory = record
  { sig = sigS
  ; Eqn = StEqn
  ; eqn = λ
    { get-set → record { ctx = 1 ; lhs = getT (λ s → setT s (var zero)) ; rhs = var zero }
    ; (set-get c) → record { ctx = 2 ; lhs = setT c (getT var) ; rhs = setT c (var c) }
    ; (set-set c c') → record { ctx = 1 ; lhs = setT c (setT c' (var zero)) ; rhs = setT c' (var zero) }
    ; get-get → record { ctx = 4 ; lhs = getT (λ i → getT (λ j → var (idx i j)))
                                 ; rhs = getT (λ i → var (idx i i)) } }
  }

open Model

statePoly : Poly
statePoly = record { Sh = St → St ; P = λ _ → 2 }

private
  T : Type → Type
  T = ⟦ statePoly ⟧

ηS : {X : Type} → X → T X
ηS x = (λ s → s) , (λ _ → x)

opsS : {X : Type} → AlgStr sigS (T X)
opsS get     args = (λ s → args s .fst s) , (λ s → args s .snd s)
opsS (set c) args = (λ _ → args zero .fst c) , (λ _ → args zero .snd c)

private
  satS : {X : Type} → Sat stateTheory (opsS {X})
  satS get-get       γ = ΣPathP (funExt (λ { s₀ → refl ; s₁ → refl }) , funExt (λ { s₀ → refl ; s₁ → refl }))
  satS get-set       γ = refl
  satS (set-get c)   γ = refl
  satS (set-set c c') γ = refl

  -- The laws in an arbitrary model N.
  module Laws (N : Model stateTheory) where
    Nget : (St → N .Carrier) → N .Carrier
    Nget = N .alg get
    Nset : St → N .Carrier → N .Carrier
    Nset c x = N .alg (set c) (λ _ → x)

    getset : ∀ x → Nget (λ s → Nset s x) ≡ x
    getset x = N .sat get-set (λ _ → x)

    setget : ∀ c (g : St → N .Carrier) → Nset c (Nget g) ≡ Nset c (g c)
    setget c g = N .sat (set-get c) g

    setset : ∀ c c' x → Nset c (Nset c' x) ≡ Nset c' x
    setset c c' x = N .sat (set-set c c') (λ _ → x)

    getget : (g : St → St → N .Carrier) → Nget (λ i → Nget (λ j → g i j)) ≡ Nget (λ i → g i i)
    getget g =
      cong Nget (funExt λ i → cong Nget (funExt λ j → sym (γidx i j)))
      ∙ N .sat get-get γ
      ∙ cong Nget (funExt λ i → γidx i i)
      where
      γ : Fin 4 → N .Carrier
      γ zero                   = g s₀ s₀
      γ (suc zero)             = g s₀ s₁
      γ (suc (suc zero))       = g s₁ s₀
      γ (suc (suc (suc zero))) = g s₁ s₁
      γidx : ∀ i j → γ (idx i j) ≡ g i j
      γidx s₀ s₀ = refl
      γidx s₀ s₁ = refl
      γidx s₁ s₀ = refl
      γidx s₁ s₁ = refl

    -- get(x, x) = x, derived.
    getconst : ∀ x → Nget (λ _ → x) ≡ x
    getconst x = cong Nget (funExt λ _ → sym (getset x))
               ∙ getget (λ i j → Nset j x)
               ∙ getset x

  extS : {X : Type} (N : Model stateTheory) → (X → N .Carrier) → T X → N .Carrier
  extS N f (σ , γ) = Nget (λ s → Nset (σ s) (f (γ s)))
    where open Laws N

  homS : {X : Type} (N : Model stateTheory) (f : X → N .Carrier)
    → IsHom sigS opsS (N .alg) (extS N f)
  homS N f .pres get args =
    sym (getget (λ i j → Nset (args i .fst j) (f (args i .snd j))))
    where open Laws N
  homS N f .pres (set c) args =
    getconst _
    ∙ sym (setset c (a .fst c) (f (a .snd c)))
    ∙ sym (setget c (λ s → Nset (a .fst s) (f (a .snd s))))
    ∙ cong (N .alg (set c)) (sym (fin1-const (λ j → extS N f (args j))))
    where
    open Laws N
    a = args zero

  uniqS : {X : Type} (N : Model stateTheory) (f : X → N .Carrier) (h : T X → N .Carrier)
    → IsHom sigS opsS (N .alg) h → (∀ x → h (ηS x) ≡ f x) → ∀ t → h t ≡ extS N f t
  uniqS N f h hh hη (σ , γ) =
    hh .pres get (λ s → opsS (set (σ s)) (λ _ → ηS (γ s)))
    ∙ cong (N .alg get) (funExt λ s →
        hh .pres (set (σ s)) (λ _ → ηS (γ s))
        ∙ cong (λ x → N .alg (set (σ s)) (λ _ → x)) (hη (γ s)))

stateFree : FreeModel stateTheory T
stateFree = record
  { η        = ηS
  ; ops      = opsS
  ; ops-sat  = satS
  ; ext      = extS
  ; ext-η    = λ N f x → Laws.getset N (f x)
  ; ext-hom  = homS
  ; ext-uniq = uniqS
  }

stateOM : OperationalModel stateTheory
stateOM = mkOperationalModel stateTheory statePoly stateFree
  (λ f → ishom λ { get args → refl ; (set c) args → refl }) (λ f x → refl)


stateC : MCOps statePoly
stateC = record { ι = λ s → s ; σ = λ f v s → v s (f s) ; pr = λ f v q → q , f q }

-- ⌈get⌉ := ⟨id, id⟩,  ⌈set_c⌉ := ⟨const c, const 0⟩
stateGen : (o : StOp) → T (Fin (Signature.ar sigS o))
stateGen get     = (λ s → s) , (λ s → s)
stateGen (set c) = (λ _ → c) , (λ _ → zero)

stateCM : ContainerModel stateTheory
stateCM = FromPolyWith.With.cm stateTheory stateOM stateC (λ _ → refl) (λ _ → refl) stateGen
  (λ { get → refl ; (set c) → refl })

state-affine : Affine stateC
state-affine = Inj₁→Affine stateC λ f i t q q' e ne → e
